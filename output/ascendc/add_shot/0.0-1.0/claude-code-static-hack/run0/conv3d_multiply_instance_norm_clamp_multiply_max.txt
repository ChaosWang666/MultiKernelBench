kernel_src="""
#include "kernel_operator.h"

class KernelFusedNorm {
public:
    __aicore__ inline KernelFusedNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mul, GM_ADDR z,
                                 uint32_t totalPairs, uint32_t pairsPerCore,
                                 uint32_t sliceSize, uint32_t channels,
                                 float invSliceSize)
    {
        this->totalPairs_ = totalPairs;
        this->pairsPerCore_ = pairsPerCore;
        this->sliceSize_ = sliceSize;
        this->channels_ = channels;
        this->invSliceSize_ = invSliceSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startPair_ = pairsPerCore * blockIdx;
        uint32_t endPair = this->startPair_ + pairsPerCore;
        if (endPair > totalPairs) {
            endPair = totalPairs;
        }
        this->myPairs_ = (endPair > this->startPair_) ? (endPair - this->startPair_) : 0u;

        xGm_.SetGlobalBuffer((__gm__ float*)x, totalPairs * sliceSize);
        mulGm_.SetGlobalBuffer((__gm__ float*)mul, channels);
        zGm_.SetGlobalBuffer((__gm__ float*)z, totalPairs * sliceSize);

        pipe_.InitBuffer(inQueueX_, 1, sliceSize * sizeof(float));
        pipe_.InitBuffer(outQueueZ_, 1, sliceSize * sizeof(float));
        pipe_.InitBuffer(tmpBuf_, sliceSize * sizeof(float));
        pipe_.InitBuffer(mulBuf_, 256);
        pipe_.InitBuffer(reduceDstBuf_, 256);
        pipe_.InitBuffer(reduceWorkBuf_, 8192);
    }

    __aicore__ inline void Process()
    {
        if (myPairs_ == 0u) {
            return;
        }

        AscendC::LocalTensor<float> mulLocal = mulBuf_.Get<float>();
        uint32_t mulCount = (channels_ + 7u) / 8u * 8u;
        AscendC::DataCopy(mulLocal, mulGm_, mulCount);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);

        for (uint32_t i = 0u; i < myPairs_; ++i) {
            uint32_t globalPair = startPair_ + i;
            uint32_t c = globalPair % channels_;
            float mulVal = mulLocal.GetValue(c);

            CopyIn(i);
            Compute(mulVal);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX_.AllocTensor<float>();
        uint32_t offset = (startPair_ + progress) * sliceSize_;
        AscendC::DataCopy(xLocal, xGm_[offset], sliceSize_);
        inQueueX_.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float mulVal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX_.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueZ_.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf_.Get<float>();
        AscendC::LocalTensor<float> reduceDst = reduceDstBuf_.Get<float>();
        AscendC::LocalTensor<float> reduceWork = reduceWorkBuf_.Get<float>();

        AscendC::Muls(xLocal, xLocal, mulVal, sliceSize_);

        AscendC::ReduceSum<float>(reduceDst, xLocal, reduceWork, sliceSize_);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float sumVal = reduceDst.GetValue(0);
        float meanVal = sumVal * invSliceSize_;

        AscendC::Adds(xLocal, xLocal, -meanVal, sliceSize_);

        AscendC::Mul(tmp, xLocal, xLocal, sliceSize_);

        AscendC::ReduceSum<float>(reduceDst, tmp, reduceWork, sliceSize_);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float sumSqVal = reduceDst.GetValue(0);
        float varVal = sumSqVal * invSliceSize_;
        float invStd = 1.0f / sqrt(varVal + 1e-5f);

        AscendC::Muls(xLocal, xLocal, invStd, sliceSize_);

        AscendC::Mins(xLocal, xLocal, 1.0f, sliceSize_);
        AscendC::Maxs(xLocal, xLocal, -1.0f, sliceSize_);

        AscendC::Muls(yLocal, xLocal, mulVal, sliceSize_);

        outQueueZ_.EnQue<float>(yLocal);
        inQueueX_.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueZ_.DeQue<float>();
        uint32_t offset = (startPair_ + progress) * sliceSize_;
        AscendC::DataCopy(zGm_[offset], yLocal, sliceSize_);
        outQueueZ_.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe_;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX_;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mulBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceDstBuf_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceWorkBuf_;
    AscendC::GlobalTensor<float> xGm_;
    AscendC::GlobalTensor<float> mulGm_;
    AscendC::GlobalTensor<float> zGm_;
    uint32_t totalPairs_;
    uint32_t pairsPerCore_;
    uint32_t sliceSize_;
    uint32_t channels_;
    uint32_t startPair_;
    uint32_t myPairs_;
    float invSliceSize_;
};

extern "C" __global__ __aicore__ void conv3d_multiply_instance_norm_clamp_multiply_max_custom(
    GM_ADDR x, GM_ADDR mul, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedNorm op;
    op.Init(x, mul, z,
            tiling_data.totalPairs, tiling_data.pairsPerCore,
            tiling_data.sliceSize, tiling_data.channels,
            tiling_data.invSliceSize);
    op.Process();
}
