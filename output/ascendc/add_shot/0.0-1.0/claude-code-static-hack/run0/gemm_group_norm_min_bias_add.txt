kernel_src="""
#include "kernel_operator.h"

constexpr int32_t M_TOTAL = 1024;
constexpr int32_t N_TOTAL = 8192;
constexpr int32_t BLOCK_DIM = 32;
constexpr int32_t ROWS_PER_CORE = M_TOTAL / BLOCK_DIM;

class KernelGNMinBias {
public:
    __aicore__ inline KernelGNMinBias() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z)
    {
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, M_TOTAL * N_TOTAL);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, N_TOTAL);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z, M_TOTAL * N_TOTAL);

        pipe.InitBuffer(inQueueX, 2, N_TOTAL * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueBias, 1, N_TOTAL * sizeof(DTYPE_BIAS));
        pipe.InitBuffer(outQueue, 2, N_TOTAL * sizeof(DTYPE_Z));
        pipe.InitBuffer(reduceBuf, 32 * sizeof(DTYPE_X));
        pipe.InitBuffer(workBuf, 2048 * sizeof(DTYPE_X));
    }
    __aicore__ inline void Process()
    {
        int32_t blockIdx = AscendC::GetBlockIdx();
        int32_t rowStart = blockIdx * ROWS_PER_CORE;

        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = inQueueBias.AllocTensor<DTYPE_BIAS>();
        AscendC::DataCopy(biasLocal, biasGm, N_TOTAL);
        inQueueBias.EnQue(biasLocal);
        biasLocal = inQueueBias.DeQue<DTYPE_BIAS>();

        for (int32_t i = 0; i < ROWS_PER_CORE; i++) {
            int32_t n = rowStart + i;

            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xLocal, xGm[n * N_TOTAL], N_TOTAL);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<DTYPE_X>();

            AscendC::LocalTensor<DTYPE_X> reduceLocal = reduceBuf.Get<DTYPE_X>();
            AscendC::LocalTensor<DTYPE_X> workLocal = workBuf.Get<DTYPE_X>();
            AscendC::ReduceMin<DTYPE_X>(reduceLocal, xLocal, workLocal, N_TOTAL, false);

            AscendC::PipeBarrier<PIPE_ALL>();
            float rowMin = reduceLocal.GetValue(0);

            inQueueX.FreeTensor(xLocal);

            AscendC::LocalTensor<DTYPE_Z> outLocal = outQueue.AllocTensor<DTYPE_Z>();
            AscendC::Adds(outLocal, biasLocal, rowMin, N_TOTAL);
            outQueue.EnQue<DTYPE_Z>(outLocal);
            outLocal = outQueue.DeQue<DTYPE_Z>();
            AscendC::DataCopy(zGm[n * N_TOTAL], outLocal, N_TOTAL);
            outQueue.FreeTensor(outLocal);
        }

        inQueueBias.FreeTensor(biasLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, 2> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
};

extern "C" __global__ __aicore__ void gemm_group_norm_min_bias_add_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGNMinBias op;
    op.Init(x, bias, z);
    op.Process();
}
"""

import torch

        super(ModelNew, self).__init__()

        out = custom_ops_lib.gemm_group_norm_min_bias_add_custom(x.contiguous(), self.bias.contiguous())
