kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t batchSize, uint32_t hiddenSize,
                                 float scale, float clampMin, float clampMax)
    {
        this->batchSize = batchSize;
        this->hiddenSize = hiddenSize;
        this->scale = scale;
        this->clampMin = clampMin;
        this->clampMax = clampMax;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->rowsPerCore = batchSize / blockNum;
        uint32_t startRow = blockIdx * this->rowsPerCore;
        this->actualRows = this->rowsPerCore;

        if (this->actualRows == 0) return;

        xGm.SetGlobalBuffer((__gm__ float *)x + startRow * hiddenSize,
                            this->actualRows * hiddenSize);
        yGm.SetGlobalBuffer((__gm__ float *)y + startRow,
                            this->actualRows);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, hiddenSize * sizeof(float));
        pipe.InitBuffer(workBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(reduceBuf, 256 * sizeof(float));

        uint32_t outBytes = ((this->actualRows * sizeof(float) + 31) / 32) * 32;
        pipe.InitBuffer(outBuf, outBytes);
    }

    __aicore__ inline void Process()
    {
        if (this->actualRows == 0) return;
        AscendC::LocalTensor<float> resultLocal = outBuf.Get<float>();

        for (uint32_t i = 0; i < this->actualRows; i++) {
            float r = ProcessRow(i);
            resultLocal.SetValue(i, r);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        uint32_t alignedCount = ((this->actualRows + 7) / 8) * 8;
        AscendC::DataCopy(yGm, resultLocal, alignedCount);
    }

private:
    __aicore__ inline float ProcessRow(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * this->hiddenSize], this->hiddenSize);
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduce = reduceBuf.Get<float>();

        // scale + clamp
        AscendC::Muls(work, xLocal, this->scale, this->hiddenSize);
        AscendC::Mins(work, work, this->clampMax, this->hiddenSize);
        AscendC::Maxs(work, work, this->clampMin, this->hiddenSize);

        // ReduceMax for numerical stability
        AscendC::ReduceMax<float>(reduce, work, reduce[64], this->hiddenSize, false);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float maxVal = reduce.GetValue(0);
        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        // exp(work - max)
        AscendC::Adds(work, work, -maxVal, this->hiddenSize);
        AscendC::Exp(work, work, this->hiddenSize);

        // ReduceSum
        AscendC::ReduceSum<float>(reduce, work, reduce[64], this->hiddenSize);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float sumVal = reduce.GetValue(0);

        // ln(sumVal) via vector op (process 8 elements, only [0] meaningful)
        reduce.SetValue(0, sumVal);
        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::Ln(reduce, reduce, 8);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float lnSum = reduce.GetValue(0);
        float lse = lnSum + maxVal;

        // softplus(lse) = ln(1 + exp(lse))
        reduce.SetValue(0, lse);
        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::Exp(reduce, reduce, 8);
        AscendC::Adds(reduce, reduce, 1.0f, 8);
        AscendC::Ln(reduce, reduce, 8);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float softplus = reduce.GetValue(0);

        // tanh(softplus) via (exp(2sp) - 1) / (exp(2sp) + 1)
        reduce.SetValue(0, 2.0f * softplus);
        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::Exp(reduce, reduce, 8);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float exp2sp = reduce.GetValue(0);
        float tanhVal = (exp2sp - 1.0f) / (exp2sp + 1.0f);

        float mishLse = lse * tanhVal;
        float result = lse * mishLse;

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        inQueueX.FreeTensor(xLocal);
        return result;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t hiddenSize;
    float scale;
    float clampMin;
    float clampMax;
    uint32_t rowsPerCore;
    uint32_t actualRows;
};

extern "C" __global__ __aicore__ void matmul_scale_residual_add_clamp_log_sum_exp_mish_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.hiddenSize,
            tiling_data.scale, tiling_data.clampMin, tiling_data.clampMax);
    op.Process();
}
