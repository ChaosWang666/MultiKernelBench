kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelChannelSum {
public:
    __aicore__ inline KernelChannelSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t batchSize, uint32_t channels,
                                 uint32_t spatialSize, uint32_t batchPerBlock)
    {
        this->channels = channels;
        this->spatialSize = spatialSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t batchOffset = blockIdx * batchPerBlock;
        uint32_t myBatches = 0;
        if (batchOffset < batchSize) {
            myBatches = batchPerBlock;
            if (batchOffset + myBatches > batchSize) {
                myBatches = batchSize - batchOffset;
            }
        }
        this->myBatches = myBatches;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + batchOffset * channels * spatialSize,
                            myBatches * channels * spatialSize);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + batchOffset * spatialSize,
                            myBatches * spatialSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->spatialSize * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, 1, this->spatialSize * sizeof(DTYPE_Z));
        pipe.InitBuffer(accBuf, this->spatialSize * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t b = 0; b < this->myBatches; b++) {
            AscendC::LocalTensor<DTYPE_Z> acc = accBuf.Get<DTYPE_Z>();
            AscendC::Duplicate(acc, (DTYPE_Z)0.0f, this->spatialSize);

            for (uint32_t c = 0; c < this->channels; c++) {
                CopyIn(b, c);
                Compute(acc);
            }

            CopyOut(b, acc);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t b, uint32_t c)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[(b * this->channels + c) * this->spatialSize], this->spatialSize);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(AscendC::LocalTensor<DTYPE_Z>& acc)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::Add(acc, acc, xLocal, this->spatialSize);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t b, AscendC::LocalTensor<DTYPE_Z>& acc)
    {
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::DataCopy(outLocal, acc, this->spatialSize);
        outQueueZ.EnQue<DTYPE_Z>(outLocal);

        AscendC::LocalTensor<DTYPE_Z> outDeq = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[b * this->spatialSize], outDeq, this->spatialSize);
        outQueueZ.FreeTensor(outDeq);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t myBatches;
};

extern "C" __global__ __aicore__ void conv_transpose3d_max_max_sum_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelChannelSum op;
    op.Init(x, z, tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize, tiling_data.batchPerBlock);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.max_pool1 = nn.MaxPool3d(kernel_size=2)
        self.max_pool2 = nn.MaxPool3d(kernel_size=3)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.max_pool1(x)
        x = self.max_pool2(x)
        x = x.contiguous()
        x = custom_ops_lib.conv_transpose3d_max_max_sum_custom(x)
        return x
