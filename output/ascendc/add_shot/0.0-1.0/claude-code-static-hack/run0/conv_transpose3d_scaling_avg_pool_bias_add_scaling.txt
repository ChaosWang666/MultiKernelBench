kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t TILE_LEN = 4096;

class KernelOp {
public:
    __aicore__ inline KernelOp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR scale2, GM_ADDR z,
                                 uint32_t channels, uint32_t groupsPerBlock, uint32_t perChannelLen)
    {
        this->channels = channels;
        this->groupsPerBlock = groupsPerBlock;
        this->perChannelLen = perChannelLen;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->blockIdx = blockIdx;
        this->startGroup = blockIdx * groupsPerBlock;

        uint64_t blockOffset = static_cast<uint64_t>(this->startGroup) * static_cast<uint64_t>(perChannelLen);
        uint64_t blockSize = static_cast<uint64_t>(groupsPerBlock) * static_cast<uint64_t>(perChannelLen);

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockOffset, blockSize);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + blockOffset, blockSize);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, channels);
        scale2Gm.SetGlobalBuffer((__gm__ DTYPE_SCALE2 *)scale2, 1);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, TILE_LEN * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, TILE_LEN * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        DTYPE_SCALE2 scale2_val = scale2Gm.GetValue(0);

        for (uint32_t g = 0; g < this->groupsPerBlock; ++g) {
            uint32_t globalGroupIdx = this->startGroup + g;
            uint32_t channel = globalGroupIdx % this->channels;
            DTYPE_BIAS bias_val = biasGm.GetValue(channel);
            ProcessGroup(g, bias_val, scale2_val);
        }
    }

private:
    __aicore__ inline void ProcessGroup(uint32_t g, DTYPE_X bias_val, DTYPE_X scale2_val)
    {
        uint32_t numFullTiles = this->perChannelLen / TILE_LEN;
        uint32_t lastTileLen = this->perChannelLen - numFullTiles * TILE_LEN;
        uint64_t baseOffset = static_cast<uint64_t>(g) * static_cast<uint64_t>(this->perChannelLen);

        for (uint32_t t = 0; t < numFullTiles; ++t) {
            uint64_t offset = baseOffset + static_cast<uint64_t>(t) * static_cast<uint64_t>(TILE_LEN);
            CopyInFull(offset);
            ComputeFull(bias_val, scale2_val);
            CopyOutFull(offset);
        }

        if (lastTileLen > 0) {
            uint64_t offset = baseOffset + static_cast<uint64_t>(numFullTiles) * static_cast<uint64_t>(TILE_LEN);
            CopyInTail(offset, lastTileLen);
            ComputeTail(bias_val, scale2_val, lastTileLen);
            CopyOutTail(offset, lastTileLen);
        }
    }

    __aicore__ inline void CopyInFull(uint64_t offset)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[offset], TILE_LEN);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void ComputeFull(DTYPE_X bias_val, DTYPE_X scale2_val)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::Adds(zLocal, xLocal, bias_val, TILE_LEN);
        AscendC::Muls(zLocal, zLocal, scale2_val, TILE_LEN);
        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOutFull(uint64_t offset)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[offset], zLocal, TILE_LEN);
        outQueueZ.FreeTensor(zLocal);
    }

    __aicore__ inline void CopyInTail(uint64_t offset, uint32_t tailLen)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = tailLen * sizeof(DTYPE_X);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPadExtParams<DTYPE_X> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;
        AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void ComputeTail(DTYPE_X bias_val, DTYPE_X scale2_val, uint32_t tailLen)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        uint32_t alignedLen = ((tailLen + 7) / 8) * 8;
        AscendC::Adds(zLocal, xLocal, bias_val, alignedLen);
        AscendC::Muls(zLocal, zLocal, scale2_val, alignedLen);
        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOutTail(uint64_t offset, uint32_t tailLen)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = tailLen * sizeof(DTYPE_Z);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPad(zGm[offset], zLocal, copyParams);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_SCALE2> scale2Gm;
    uint32_t channels;
    uint32_t groupsPerBlock;
    uint32_t perChannelLen;
    uint32_t blockIdx;
    uint32_t startGroup;
};

extern "C" __global__ __aicore__ void conv_transpose3d_scaling_avg_pool_bias_add_scaling_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR scale2, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelOp op;
    op.Init(x, bias, scale2, z,
            tiling_data.channels, tiling_data.groupsPerBlock, tiling_data.perChannelLen);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, scale1, scale2, bias_shape):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.scale1 = nn.Parameter(torch.tensor(scale1))
        self.avg_pool = nn.AvgPool3d(kernel_size=2)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.scale2 = nn.Parameter(torch.tensor(scale2))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x * self.scale1
        x = self.avg_pool(x)
        x = x.contiguous()
        bias_flat = self.bias.contiguous().view(-1)
        scale2_flat = self.scale2.reshape(1).contiguous()
        return custom_ops_lib.conv_transpose3d_scaling_avg_pool_bias_add_scaling_custom(x, bias_flat, scale2_flat)
