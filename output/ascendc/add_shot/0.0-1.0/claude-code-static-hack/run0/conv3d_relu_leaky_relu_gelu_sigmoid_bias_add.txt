kernel_src="""
#include "kernel_operator.h"

class KernelFusedActBias {
public:
    __aicore__ inline KernelFusedActBias() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                uint32_t channelLength,
                                uint32_t channelsPerCore,
                                uint32_t outChannels,
                                uint32_t tileLength,
                                uint32_t tileNum)
    {
        this->channelLength = channelLength;
        this->channelsPerCore = channelsPerCore;
        this->outChannels = outChannels;
        this->tileLength = tileLength;
        this->tileNum = tileNum;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockOffset = blockIdx * channelsPerCore * channelLength;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockOffset, channelsPerCore * channelLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + blockOffset, channelsPerCore * channelLength);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, outChannels);

        pipe.InitBuffer(inQueueX, 1, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, 1, this->tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpBuf, this->tileLength * sizeof(DTYPE_X));
        uint32_t biasBytes = ((outChannels * sizeof(DTYPE_BIAS) + 31) / 32) * 32;
        pipe.InitBuffer(biasBuf, biasBytes);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();
        AscendC::DataCopy(biasLocal, biasGm, this->outChannels);

        event_t eventIDMte2ToS = static_cast<event_t>(GetTPipePtr()->FetchEventID(HardEvent::MTE2_S));
        SetFlag<HardEvent::MTE2_S>(eventIDMte2ToS);
        WaitFlag<HardEvent::MTE2_S>(eventIDMte2ToS);

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startChannel = blockIdx * this->channelsPerCore;

        for (uint32_t ch = 0; ch < this->channelsPerCore; ch++) {
            uint32_t globalChannelIdx = startChannel + ch;
            uint32_t cIdx = globalChannelIdx % this->outChannels;
            float biasScalar = biasLocal.GetValue(cIdx);

            for (uint32_t t = 0; t < this->tileNum; t++) {
                uint32_t offset = ch * this->channelLength + t * this->tileLength;
                CopyIn(offset);
                Compute(biasScalar);
                CopyOut(offset);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[offset], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float biasScalar)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_X> tmpLocal = tmpBuf.Get<DTYPE_X>();

        AscendC::Maxs(zLocal, xLocal, (DTYPE_X)0.0f, this->tileLength);

        AscendC::Mul(tmpLocal, zLocal, zLocal, this->tileLength);
        AscendC::Mul(tmpLocal, tmpLocal, zLocal, this->tileLength);
        AscendC::Muls(tmpLocal, tmpLocal, (DTYPE_X)0.044715f, this->tileLength);
        AscendC::Add(tmpLocal, tmpLocal, zLocal, this->tileLength);
        AscendC::Muls(tmpLocal, tmpLocal, (DTYPE_X)0.7978845608028654f, this->tileLength);
        AscendC::Tanh(tmpLocal, tmpLocal, this->tileLength);
        AscendC::Adds(tmpLocal, tmpLocal, (DTYPE_X)1.0f, this->tileLength);
        AscendC::Muls(tmpLocal, tmpLocal, (DTYPE_X)0.5f, this->tileLength);
        AscendC::Mul(zLocal, zLocal, tmpLocal, this->tileLength);

        AscendC::Muls(tmpLocal, zLocal, (DTYPE_X)-1.0f, this->tileLength);
        AscendC::Exp(tmpLocal, tmpLocal, this->tileLength);
        AscendC::Adds(tmpLocal, tmpLocal, (DTYPE_X)1.0f, this->tileLength);
        AscendC::Duplicate(zLocal, (DTYPE_X)1.0f, this->tileLength);
        AscendC::Div(zLocal, zLocal, tmpLocal, this->tileLength);

        AscendC::Adds(zLocal, zLocal, biasScalar, this->tileLength);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[offset], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t channelLength;
    uint32_t channelsPerCore;
    uint32_t outChannels;
    uint32_t tileLength;
    uint32_t tileNum;
};

extern "C" __global__ __aicore__ void conv3d_relu_leaky_relu_gelu_sigmoid_bias_add_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedActBias op;
    op.Init(x, bias, z,
            tiling_data.channelLength,
            tiling_data.channelsPerCore,
            tiling_data.outChannels,
            tiling_data.tileLength,
            tiling_data.tileNum);
    op.Process();
}
