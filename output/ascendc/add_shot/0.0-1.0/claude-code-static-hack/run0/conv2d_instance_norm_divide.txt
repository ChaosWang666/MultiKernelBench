kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelInstanceNormDivide {
public:
    __aicore__ inline KernelInstanceNormDivide() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                  uint32_t channelsPerBlock,
                                  uint32_t spatialSize,
                                  uint32_t alignedSpatialSize,
                                  float invSpatialSize,
                                  float divideBy,
                                  float epsilon)
    {
        this->channelsPerBlock = channelsPerBlock;
        this->spatialSize = spatialSize;
        this->alignedSpatialSize = alignedSpatialSize;
        this->invSpatialSize = invSpatialSize;
        this->divideBy = divideBy;
        this->epsilon = epsilon;

        uint64_t blockIdx64 = static_cast<uint64_t>(AscendC::GetBlockIdx());
        uint64_t channelsPerBlock64 = static_cast<uint64_t>(channelsPerBlock);
        uint64_t spatialSize64 = static_cast<uint64_t>(spatialSize);
        uint64_t blockOffset = channelsPerBlock64 * spatialSize64 * blockIdx64;
        uint64_t blockTotalElements = channelsPerBlock64 * spatialSize64;

        xGm.SetGlobalBuffer((__gm__ float*)x + blockOffset, blockTotalElements);
        zGm.SetGlobalBuffer((__gm__ float*)z + blockOffset, blockTotalElements);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedSpatialSize * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, alignedSpatialSize * sizeof(float));
        pipe.InitBuffer(workTBuf, 4096);
        pipe.InitBuffer(sumTBuf, 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->channelsPerBlock; i++) {
            CopyIn(i);
            Compute();
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = this->spatialSize * sizeof(float);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        AscendC::DataCopyPadExtParams<float> padParams;
        padParams.isPad = true;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;
        uint64_t offset = static_cast<uint64_t>(progress) * static_cast<uint64_t>(this->spatialSize);
        AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> workLocal = workTBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumTBuf.Get<float>();

        // sum(x)
        AscendC::ReduceSum<float>(sumLocal, xLocal, workLocal, this->spatialSize);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float mean = sumLocal.GetValue(0) * this->invSpatialSize;
        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        // z = x - mean
        AscendC::Adds(zLocal, xLocal, -mean, this->spatialSize);
        AscendC::PipeBarrier<AscendC::PIPE_V>();

        // x = z * z
        AscendC::Mul(xLocal, zLocal, zLocal, this->spatialSize);
        AscendC::PipeBarrier<AscendC::PIPE_V>();

        // sum((x-mean)^2)
        AscendC::ReduceSum<float>(sumLocal, xLocal, workLocal, this->spatialSize);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
        float variance = sumLocal.GetValue(0) * this->invSpatialSize;

        float invStd = 1.0f / sqrtf(variance + this->epsilon);
        float scale = invStd / this->divideBy;

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID1);
        AscendC::Muls(zLocal, zLocal, scale, this->spatialSize);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = this->spatialSize * sizeof(float);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        uint64_t offset = static_cast<uint64_t>(progress) * static_cast<uint64_t>(this->spatialSize);
        AscendC::DataCopyPad(zGm[offset], zLocal, copyParams);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workTBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumTBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t channelsPerBlock;
    uint32_t spatialSize;
    uint32_t alignedSpatialSize;
    float invSpatialSize;
    float divideBy;
    float epsilon;
};

extern "C" __global__ __aicore__ void conv2d_instance_norm_divide_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelInstanceNormDivide op;
    op.Init(x, z,
            tiling_data.channelsPerBlock,
            tiling_data.spatialSize,
            tiling_data.alignedSpatialSize,
            tiling_data.invSpatialSize,
            tiling_data.divideBy,
            tiling_data.epsilon);
    op.Process();
}
