kernel_src="""
#include "kernel_operator.h"

class KernelLseAct {
public:
    __aicore__ inline KernelLseAct() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t cols, uint32_t rowsPerBlock)
    {
        this->cols = cols;
        this->rowsPerBlock = rowsPerBlock;
        uint32_t startRow = AscendC::GetBlockIdx() * rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ float *)x + startRow * cols, rowsPerBlock * cols);
        zGm.SetGlobalBuffer((__gm__ float *)z + startRow, rowsPerBlock);

        pipe.InitBuffer(xBuf, cols * sizeof(float));
        pipe.InitBuffer(workBuf, cols * sizeof(float));
        pipe.InitBuffer(redBuf, 64);
        pipe.InitBuffer(resultBuf, 512);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> redLocal = redBuf.Get<float>();
        AscendC::LocalTensor<float> resultLocal = resultBuf.Get<float>();

        for (uint32_t r = 0; r < this->rowsPerBlock; r++) {
            AscendC::DataCopy(xLocal, xGm[r * this->cols], this->cols);
            AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(0);

            AscendC::ReduceMax<float>(redLocal, xLocal, workLocal, this->cols, false);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::SetFlag<AscendC::HardEvent::V_S>(0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(0);
            float maxVal = redLocal.GetValue(0);
            AscendC::SetFlag<AscendC::HardEvent::S_V>(0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(0);

            AscendC::Adds(xLocal, xLocal, -maxVal, this->cols);
            AscendC::Exp(xLocal, xLocal, this->cols);
            AscendC::ReduceSum<float>(redLocal, xLocal, workLocal, this->cols);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Ln(redLocal, redLocal, 8);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::SetFlag<AscendC::HardEvent::V_S>(0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(0);
            float lnSum = redLocal.GetValue(0);
            float lse = lnSum + maxVal;
            resultLocal.SetValue(r, lse);

            AscendC::SetFlag<AscendC::HardEvent::V_MTE2>(0);
            AscendC::WaitFlag<AscendC::HardEvent::V_MTE2>(0);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_V>(0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(0);

        uint32_t alignedN = ((this->rowsPerBlock + 7) / 8) * 8;

        AscendC::Muls(workLocal, resultLocal, (float)0.01f, alignedN);
        AscendC::Max(resultLocal, resultLocal, workLocal, alignedN);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Muls(workLocal, resultLocal, (float)0.01f, alignedN);
        AscendC::Max(resultLocal, resultLocal, workLocal, alignedN);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Muls(workLocal, resultLocal, (float)0.7071067811865475f, alignedN);
        AscendC::Erf(workLocal, workLocal, alignedN);
        AscendC::Adds(workLocal, workLocal, (float)1.0f, alignedN);
        AscendC::Muls(workLocal, workLocal, (float)0.5f, alignedN);
        AscendC::Mul(resultLocal, resultLocal, workLocal, alignedN);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Muls(workLocal, resultLocal, (float)0.7071067811865475f, alignedN);
        AscendC::Erf(workLocal, workLocal, alignedN);
        AscendC::Adds(workLocal, workLocal, (float)1.0f, alignedN);
        AscendC::Muls(workLocal, workLocal, (float)0.5f, alignedN);
        AscendC::Mul(resultLocal, resultLocal, workLocal, alignedN);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::SetFlag<AscendC::HardEvent::V_MTE3>(0);
        AscendC::WaitFlag<AscendC::HardEvent::V_MTE3>(0);

        AscendC::DataCopy(zGm, resultLocal, this->rowsPerBlock);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> redBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> resultBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t cols;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void gemm_log_sum_exp_leaky_relu_leaky_relu_gelu_gelu_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelLseAct op;
    op.Init(x, z, tiling_data.cols, tiling_data.rowsPerBlock);
    op.Process();
}
