kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulSwishSumGroupNorm {
public:
    __aicore__ inline KernelMatmulSwishSumGroupNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                 uint32_t batchSize, uint32_t outFeatures, uint32_t numGroups,
                                 uint32_t channelsPerGroup, uint32_t rowsPerCore,
                                 float epsilon, float invChannels)
    {
        this->batchSize = batchSize;
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->channelsPerGroup = channelsPerGroup;
        this->rowsPerCore = rowsPerCore;
        this->epsilon = epsilon;
        this->invChannels = invChannels;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockOffset = blockIdx * rowsPerCore * outFeatures;

        xGm.SetGlobalBuffer((__gm__ float *)x + blockOffset, rowsPerCore * outFeatures);
        yGm.SetGlobalBuffer((__gm__ float *)y + blockOffset, rowsPerCore * outFeatures);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);
        gammaGm.SetGlobalBuffer((__gm__ float *)gamma, outFeatures);
        betaGm.SetGlobalBuffer((__gm__ float *)beta, outFeatures);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(biasQueue, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(gammaQueue, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(betaQueue, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf1, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf2, outFeatures * sizeof(float));
        pipe.InitBuffer(reduceWorkBuf, 1024);
        pipe.InitBuffer(scalarBuf, 64);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasAlloc = biasQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> gammaAlloc = gammaQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> betaAlloc = betaQueue.AllocTensor<float>();
        AscendC::DataCopy(biasAlloc, biasGm, outFeatures);
        AscendC::DataCopy(gammaAlloc, gammaGm, outFeatures);
        AscendC::DataCopy(betaAlloc, betaGm, outFeatures);
        biasQueue.EnQue(biasAlloc);
        gammaQueue.EnQue(gammaAlloc);
        betaQueue.EnQue(betaAlloc);

        AscendC::LocalTensor<float> biasL = biasQueue.DeQue<float>();
        AscendC::LocalTensor<float> gammaL = gammaQueue.DeQue<float>();
        AscendC::LocalTensor<float> betaL = betaQueue.DeQue<float>();

        for (uint32_t row = 0; row < rowsPerCore; row++) {
            CopyIn(row);
            Compute(row, biasL, gammaL, betaL);
            CopyOut(row);
        }

        biasQueue.FreeTensor(biasL);
        gammaQueue.FreeTensor(gammaL);
        betaQueue.FreeTensor(betaL);
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * outFeatures], outFeatures);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t progress,
                                    AscendC::LocalTensor<float> &biasL,
                                    AscendC::LocalTensor<float> &gammaL,
                                    AscendC::LocalTensor<float> &betaL)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();
        AscendC::LocalTensor<float> workLocal = reduceWorkBuf.Get<float>();
        AscendC::LocalTensor<float> scalarL = scalarBuf.Get<float>();

        // Swish: tmp1 = x * sigmoid(x) computed via 1 / (1 + exp(-x))
        AscendC::Muls(tmp1, xLocal, -1.0f, outFeatures);
        AscendC::Exp(tmp1, tmp1, outFeatures);
        AscendC::Adds(tmp1, tmp1, 1.0f, outFeatures);
        AscendC::Div(tmp1, xLocal, tmp1, outFeatures);

        // Add bias
        AscendC::Add(tmp1, tmp1, biasL, outFeatures);

        // GroupNorm per group
        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t offset = g * channelsPerGroup;

            AscendC::ReduceSum(scalarL, tmp1[offset], workLocal, channelsPerGroup);
            AscendC::SetFlag<AscendC::HardEvent::V_S>((event_t)0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>((event_t)0);
            float sumVal = scalarL.GetValue(0);
            float meanVal = sumVal * invChannels;

            AscendC::Adds(tmp2[offset], tmp1[offset], -meanVal, channelsPerGroup);
            AscendC::Mul(yLocal[offset], tmp2[offset], tmp2[offset], channelsPerGroup);
            AscendC::ReduceSum(scalarL, yLocal[offset], workLocal, channelsPerGroup);
            AscendC::SetFlag<AscendC::HardEvent::V_S>((event_t)0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>((event_t)0);
            float sumSqVal = scalarL.GetValue(0);
            float varVal = sumSqVal * invChannels;

            AscendC::Duplicate(scalarL, varVal + epsilon, 8);
            AscendC::Sqrt(scalarL, scalarL, 8);
            AscendC::SetFlag<AscendC::HardEvent::V_S>((event_t)0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>((event_t)0);
            float stdVal = scalarL.GetValue(0);
            float invStd = 1.0f / stdVal;

            AscendC::Muls(yLocal[offset], tmp2[offset], invStd, channelsPerGroup);
            AscendC::Mul(yLocal[offset], yLocal[offset], gammaL[offset], channelsPerGroup);
            AscendC::Add(yLocal[offset], yLocal[offset], betaL[offset], channelsPerGroup);
        }

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * outFeatures], yLocal, outFeatures);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> biasQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> gammaQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> betaQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf2;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceWorkBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    uint32_t batchSize;
    uint32_t outFeatures;
    uint32_t numGroups;
    uint32_t channelsPerGroup;
    uint32_t rowsPerCore;
    float epsilon;
    float invChannels;
};

extern "C" __global__ __aicore__ void matmul_swish_sum_group_norm_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSwishSumGroupNorm op;
    op.Init(x, bias, gamma, beta, y,
            tiling_data.batchSize, tiling_data.outFeatures, tiling_data.numGroups,
            tiling_data.channelsPerGroup, tiling_data.rowsPerCore,
            tiling_data.epsilon, tiling_data.invChannels);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, bias_shape):
        super().__init__()
        self.matmul = nn.Linear(in_features, out_features)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.group_norm = nn.GroupNorm(num_groups, out_features)

    def forward(self, x):
        x = self.matmul(x)
        x = custom_ops_lib.matmul_swish_sum_group_norm_custom(
            x.contiguous(), self.bias, self.group_norm.weight, self.group_norm.bias
        )
        return x
