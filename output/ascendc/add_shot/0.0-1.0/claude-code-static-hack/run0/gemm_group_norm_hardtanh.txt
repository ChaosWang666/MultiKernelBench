kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelGemmGroupNormHardtanh {
public:
    __aicore__ inline KernelGemmGroupNormHardtanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                uint32_t batchSize, uint32_t channels,
                                uint32_t numGroups, uint32_t groupSize,
                                uint32_t blockBatches,
                                float invGroupSize, float epsilon,
                                float hardtanhMin, float hardtanhMax)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->numGroups = numGroups;
        this->groupSize = groupSize;
        this->invGroupSize = invGroupSize;
        this->epsilon = epsilon;
        this->hardtanhMin = hardtanhMin;
        this->hardtanhMax = hardtanhMax;

        uint32_t blockId = AscendC::GetBlockIdx();
        uint32_t startBatch = blockId * blockBatches;
        uint32_t endBatch = startBatch + blockBatches;
        if (endBatch > batchSize) {
            endBatch = batchSize;
        }
        if (startBatch >= batchSize) {
            this->localBatches = 0;
        } else {
            this->localBatches = endBatch - startBatch;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x + startBatch * channels, this->localBatches * channels);
        yGm.SetGlobalBuffer((__gm__ float *)y + startBatch * channels, this->localBatches * channels);
        gammaGm.SetGlobalBuffer((__gm__ float *)gamma, channels);
        betaGm.SetGlobalBuffer((__gm__ float *)beta, channels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, channels * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, channels * sizeof(float));
        pipe.InitBuffer(gammaQueue, 1, channels * sizeof(float));
        pipe.InitBuffer(betaQueue, 1, channels * sizeof(float));
        pipe.InitBuffer(workBuf, 1024);
    }

    __aicore__ inline void Process()
    {
        if (this->localBatches == 0) {
            return;
        }

        AscendC::LocalTensor<float> gammaTmp = gammaQueue.AllocTensor<float>();
        AscendC::DataCopy(gammaTmp, gammaGm, this->channels);
        gammaQueue.EnQue(gammaTmp);
        AscendC::LocalTensor<float> gammaLocal = gammaQueue.DeQue<float>();

        AscendC::LocalTensor<float> betaTmp = betaQueue.AllocTensor<float>();
        AscendC::DataCopy(betaTmp, betaGm, this->channels);
        betaQueue.EnQue(betaTmp);
        AscendC::LocalTensor<float> betaLocal = betaQueue.DeQue<float>();

        for (uint32_t b = 0; b < this->localBatches; b++) {
            CopyIn(b);
            Compute(gammaLocal, betaLocal);
            CopyOut(b);
        }

        gammaQueue.FreeTensor(gammaLocal);
        betaQueue.FreeTensor(betaLocal);
    }

private:
    __aicore__ inline void CopyIn(uint32_t b)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[b * this->channels], this->channels);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(AscendC::LocalTensor<float> &gammaLocal,
                                   AscendC::LocalTensor<float> &betaLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();

        for (uint32_t g = 0; g < this->numGroups; g++) {
            uint32_t offset = g * this->groupSize;

            AscendC::ReduceSum(workLocal, xLocal[offset], workLocal[64], this->groupSize);
            float sum = workLocal.GetValue(0);
            float mean = sum * this->invGroupSize;

            AscendC::Adds(xLocal[offset], xLocal[offset], -mean, this->groupSize);

            AscendC::Mul(yLocal[offset], xLocal[offset], xLocal[offset], this->groupSize);

            AscendC::ReduceSum(workLocal, yLocal[offset], workLocal[64], this->groupSize);
            float sumSq = workLocal.GetValue(0);
            float var = sumSq * this->invGroupSize;

            float invStd = 1.0f / sqrt(var + this->epsilon);

            AscendC::Muls(yLocal[offset], xLocal[offset], invStd, this->groupSize);

            AscendC::Mul(yLocal[offset], yLocal[offset], gammaLocal[offset], this->groupSize);
            AscendC::Add(yLocal[offset], yLocal[offset], betaLocal[offset], this->groupSize);

            AscendC::Maxs(yLocal[offset], yLocal[offset], this->hardtanhMin, this->groupSize);
            AscendC::Mins(yLocal[offset], yLocal[offset], this->hardtanhMax, this->groupSize);
        }

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t b)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[b * this->channels], yLocal, this->channels);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> gammaQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> betaQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;

    uint32_t batchSize;
    uint32_t channels;
    uint32_t numGroups;
    uint32_t groupSize;
    uint32_t localBatches;
    float invGroupSize;
    float epsilon;
    float hardtanhMin;
    float hardtanhMax;
};

extern "C" __global__ __aicore__ void gemm_group_norm_hardtanh_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormHardtanh op;
    op.Init(x, gamma, beta, y,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.numGroups, tiling_data.groupSize,
            tiling_data.blockBatches,
            tiling_data.invGroupSize, tiling_data.epsilon,
            tiling_data.hardtanhMin, tiling_data.hardtanhMax);
    op.Process();
}
"""

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, hardtanh_min, hardtanh_max):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.group_norm = nn.GroupNorm(num_groups, out_features)
        self.hardtanh_min = hardtanh_min
        self.hardtanh_max = hardtanh_max
        self.num_groups = num_groups

    def forward(self, x):
        x = self.gemm(x)
        x = custom_ops_lib.gemm_group_norm_hardtanh_custom(
            x.contiguous(),
            self.group_norm.weight.contiguous(),
            self.group_norm.bias.contiguous(),
        )
        return x
'''
