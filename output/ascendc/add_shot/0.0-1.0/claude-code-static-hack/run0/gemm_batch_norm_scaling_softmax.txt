kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelGemmBnScaleSoftmax {
public:
    __aicore__ inline KernelGemmBnScaleSoftmax() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                uint32_t batchSize, uint32_t featureSize)
    {
        uint32_t blockIdx = (uint32_t)AscendC::GetBlockIdx();
        uint32_t blockNum = (uint32_t)AscendC::GetBlockNum();

        this->featureSize = featureSize;
        this->rowsPerCore = batchSize / blockNum;

        uint32_t coreOffset = blockIdx * this->rowsPerCore * featureSize;
        uint32_t coreSize = this->rowsPerCore * featureSize;

        xGm.SetGlobalBuffer((__gm__ float *)x + coreOffset, coreSize);
        zGm.SetGlobalBuffer((__gm__ float *)z + coreOffset, coreSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, featureSize * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, featureSize * sizeof(float));
        pipe.InitBuffer(workBuf, 8192);
        pipe.InitBuffer(resultBuf, 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsPerCore; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * this->featureSize], this->featureSize);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> resultLocal = resultBuf.Get<float>();

        int32_t count = (int32_t)this->featureSize;

        AscendC::ReduceMax<float>(resultLocal, xLocal, workLocal, count, false);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float maxVal = resultLocal.GetValue(0);

        AscendC::Adds<float>(xLocal, xLocal, -maxVal, count);
        AscendC::Exp<float>(xLocal, xLocal, count);

        AscendC::ReduceSum<float>(resultLocal, xLocal, workLocal, count);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float sumVal = resultLocal.GetValue(0);

        float invSum = 1.0f / sumVal;

        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::Muls<float>(zLocal, xLocal, invSum, count);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[row * this->featureSize], zLocal, this->featureSize);
        outQueueZ.FreeTensor(zLocal);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> resultBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t featureSize;
    uint32_t rowsPerCore;
};

extern "C" __global__ __aicore__ void gemm_batch_norm_scaling_softmax_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmBnScaleSoftmax op;
    op.Init(x, z, tiling_data.batchSize, tiling_data.featureSize);
    op.Process();
}
"""

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, bn_eps=1e-5, bn_momentum=0.1, scale_shape=(1,)):
        super().__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.bn = nn.BatchNorm1d(out_features, eps=bn_eps, momentum=bn_momentum)
        self.scale = nn.Parameter(torch.ones(scale_shape))

    def forward(self, x):
        x = self.gemm(x)
        x = self.bn(x)
        x = self.scale * x
        x = x.contiguous()
        return custom_ops_lib.gemm_batch_norm_scaling_softmax_custom(x)
'''
