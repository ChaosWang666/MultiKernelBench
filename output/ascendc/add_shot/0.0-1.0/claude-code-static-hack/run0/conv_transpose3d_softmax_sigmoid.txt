kernel_src="""
#include "kernel_operator.h"

constexpr uint32_t BUFFER_NUM = 2;

class KernelConvSoftSig {
public:
    __aicore__ inline KernelConvSoftSig() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batchPerBlock, uint32_t channels,
                                uint32_t spatial, uint32_t tileSize)
    {
        this->batchPerBlock = batchPerBlock;
        this->channels = channels;
        this->spatial = spatial;
        this->tileSize = tileSize;
        this->numTiles = spatial / tileSize;

        uint32_t blockBaseOffset = AscendC::GetBlockIdx() * batchPerBlock * channels * spatial;
        uint32_t blockSize = batchPerBlock * channels * spatial;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockBaseOffset, blockSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + blockBaseOffset, blockSize);

        pipe.InitBuffer(inQueue, BUFFER_NUM, channels * tileSize * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueue, BUFFER_NUM, channels * tileSize * sizeof(DTYPE_Y));
        pipe.InitBuffer(maxBuf, tileSize * sizeof(float));
        pipe.InitBuffer(sumBuf, tileSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, tileSize * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t b = 0; b < batchPerBlock; b++) {
            for (uint32_t t = 0; t < numTiles; t++) {
                CopyIn(b, t);
                Compute();
                CopyOut(b, t);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t b, uint32_t t)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueue.AllocTensor<DTYPE_X>();
        AscendC::DataCopyParams copyParams;
        copyParams.blockCount = static_cast<uint16_t>(channels);
        copyParams.blockLen = static_cast<uint16_t>(tileSize / 8);
        copyParams.srcStride = static_cast<uint16_t>((spatial - tileSize) / 8);
        copyParams.dstStride = 0;
        AscendC::DataCopy(xLocal, xGm[b * channels * spatial + t * tileSize], copyParams);
        inQueue.EnQue(xLocal);
    }
    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueue.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueue.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        AscendC::Duplicate(maxLocal, -1e30f, tileSize);
        for (uint32_t c = 0; c < channels; c++) {
            AscendC::Max(maxLocal, maxLocal, xLocal[c * tileSize], tileSize);
        }

        AscendC::Duplicate(sumLocal, 0.0f, tileSize);
        for (uint32_t c = 0; c < channels; c++) {
            AscendC::Sub(yLocal[c * tileSize], xLocal[c * tileSize], maxLocal, tileSize);
            AscendC::Exp(yLocal[c * tileSize], yLocal[c * tileSize], tileSize);
            AscendC::Add(sumLocal, sumLocal, yLocal[c * tileSize], tileSize);
        }

        for (uint32_t c = 0; c < channels; c++) {
            AscendC::Div(yLocal[c * tileSize], yLocal[c * tileSize], sumLocal, tileSize);
            AscendC::Muls(tmpLocal, yLocal[c * tileSize], -1.0f, tileSize);
            AscendC::Exp(tmpLocal, tmpLocal, tileSize);
            AscendC::Adds(tmpLocal, tmpLocal, 1.0f, tileSize);
            AscendC::Reciprocal(yLocal[c * tileSize], tmpLocal, tileSize);
        }

        inQueue.FreeTensor(xLocal);
        outQueue.EnQue<DTYPE_Y>(yLocal);
    }
    __aicore__ inline void CopyOut(uint32_t b, uint32_t t)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueue.DeQue<DTYPE_Y>();
        AscendC::DataCopyParams copyParams;
        copyParams.blockCount = static_cast<uint16_t>(channels);
        copyParams.blockLen = static_cast<uint16_t>(tileSize / 8);
        copyParams.srcStride = 0;
        copyParams.dstStride = static_cast<uint16_t>((spatial - tileSize) / 8);
        AscendC::DataCopy(yGm[b * channels * spatial + t * tileSize], yLocal, copyParams);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t batchPerBlock;
    uint32_t channels;
    uint32_t spatial;
    uint32_t tileSize;
    uint32_t numTiles;
};

extern "C" __global__ __aicore__ void conv_transpose3d_softmax_sigmoid_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvSoftSig op;
    op.Init(x, y, tiling_data.batchPerBlock, tiling_data.channels,
            tiling_data.spatial, tiling_data.tileSize);
    op.Process();
}
