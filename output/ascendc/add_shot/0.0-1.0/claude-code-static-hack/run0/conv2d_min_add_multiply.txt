kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv2dMinAddMultiplyCustom {
public:
    __aicore__ inline KernelConv2dMinAddMultiplyCustom() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                uint32_t batchSize, uint32_t batchesPerCore,
                                uint32_t tileLength, uint32_t tileNum,
                                float constantValue, float scalingFactor)
    {
        this->batchSize = batchSize;
        this->batchesPerCore = batchesPerCore;
        this->tileLength = tileLength;
        this->tileNum = tileNum;
        this->constantValue = constantValue;
        this->scalingFactor = scalingFactor;

        uint32_t coreIdx = AscendC::GetBlockIdx();
        uint32_t startBatch = coreIdx * batchesPerCore;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startBatch * batchSize, batchesPerCore * batchSize);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, batchSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + startBatch * batchSize, batchesPerCore * batchSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, this->tileLength * sizeof(DTYPE_BIAS));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t b = 0; b < this->batchesPerCore; b++) {
            for (uint32_t t = 0; t < this->tileNum; t++) {
                CopyIn(b, t);
                Compute();
                CopyOut(b, t);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t b, uint32_t t)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = inQueueBias.AllocTensor<DTYPE_BIAS>();
        AscendC::DataCopy(xLocal, xGm[b * this->batchSize + t * this->tileLength], this->tileLength);
        AscendC::DataCopy(biasLocal, biasGm[t * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueBias.EnQue(biasLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = inQueueBias.DeQue<DTYPE_BIAS>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::Mins(yLocal, xLocal, (DTYPE_X)this->constantValue, this->tileLength);
        AscendC::Add(yLocal, yLocal, biasLocal, this->tileLength);
        AscendC::Muls(yLocal, yLocal, (DTYPE_Y)this->scalingFactor, this->tileLength);
        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueBias.FreeTensor(biasLocal);
    }

    __aicore__ inline void CopyOut(uint32_t b, uint32_t t)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[b * this->batchSize + t * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batchSize;
    uint32_t batchesPerCore;
    uint32_t tileLength;
    uint32_t tileNum;
    float constantValue;
    float scalingFactor;
};

extern "C" __global__ __aicore__ void conv2d_min_add_multiply_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dMinAddMultiplyCustom op;
    op.Init(x, bias, y,
            tiling_data.batchSize, tiling_data.batchesPerCore,
            tiling_data.tileLength, tiling_data.tileNum,
            tiling_data.constantValue, tiling_data.scalingFactor);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, constant_value, bias_shape, scaling_factor):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.constant_value = float(constant_value)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.scaling_factor = float(scaling_factor)

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
        bias_expanded = self.bias.expand(self.bias.shape[0], x.shape[2], x.shape[3]).contiguous()
        return custom_ops_lib.conv2d_min_add_multiply_custom(x, bias_expanded, self.constant_value, self.scaling_factor)
