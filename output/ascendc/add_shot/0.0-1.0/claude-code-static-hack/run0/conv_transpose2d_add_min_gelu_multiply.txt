kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedPostConv {
public:
    __aicore__ inline KernelFusedPostConv() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t tileLength, float addValue, float mulValue)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileLength = tileLength;
        this->addValue = addValue;
        this->mulValue = mulValue;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(tmpBuf, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t numFullTiles = this->blockLength / this->tileLength;
        uint32_t tail = this->blockLength - numFullTiles * this->tileLength;

        for (uint32_t i = 0; i < numFullTiles; i++) {
            CopyIn(i, this->tileLength);
            Compute(this->tileLength);
            CopyOut(i, this->tileLength);
        }
        if (tail > 0) {
            CopyIn(numFullTiles, tail);
            Compute(tail);
            CopyOut(numFullTiles, tail);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t i, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[i * this->tileLength], len);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        // y = x + addValue
        AscendC::Adds(yLocal, xLocal, this->addValue, len);
        // y = min(y, 0.0f)
        AscendC::Mins(yLocal, yLocal, (float)0.0, len);

        // GELU(y) using tanh-form expressed via sigmoid:
        // GELU(y) = y / (1 + exp(-1.5957691 * (y + 0.044715 * y^3)))
        // tmp = y * y
        AscendC::Mul(tmp, yLocal, yLocal, len);
        // tmp = tmp * y = y^3
        AscendC::Mul(tmp, tmp, yLocal, len);
        // tmp = 0.044715 * y^3
        AscendC::Muls(tmp, tmp, (float)0.044715, len);
        // tmp = y + 0.044715 * y^3
        AscendC::Add(tmp, tmp, yLocal, len);
        // tmp = -1.5957691 * (y + 0.044715 * y^3)
        AscendC::Muls(tmp, tmp, (float)-1.5957691, len);
        // tmp = exp(tmp)
        AscendC::Exp(tmp, tmp, len);
        // tmp = 1 + exp(...)
        AscendC::Adds(tmp, tmp, (float)1.0, len);
        // y = y / (1 + exp(...)) = GELU(y)
        AscendC::Div(yLocal, yLocal, tmp, len);

        // y = y * mulValue
        AscendC::Muls(yLocal, yLocal, this->mulValue, len);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t i, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[i * this->tileLength], yLocal, len);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t blockLength;
    uint32_t tileLength;
    float addValue;
    float mulValue;
};

extern "C" __global__ __aicore__ void conv_transpose2d_add_min_gelu_multiply_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedPostConv op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.tileLength, tiling_data.addValue, tiling_data.mulValue);
    op.Process();
}
