kernel_src="""
#include "kernel_operator.h"

class KernelBmmInstanceNormFused {
public:
    __aicore__ inline KernelBmmInstanceNormFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                 uint32_t batchSize, uint32_t outFeatures,
                                 float invN, float eps)
    {
        this->outFeatures = outFeatures;
        this->invN = invN;
        this->eps = eps;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->rowsPerBlock = batchSize / blockNum;
        uint32_t rowOffset = blockIdx * this->rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + rowOffset * outFeatures,
                             this->rowsPerBlock * outFeatures);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + rowOffset * outFeatures,
                             this->rowsPerBlock * outFeatures);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + rowOffset * outFeatures,
                             this->rowsPerBlock * outFeatures);

        pipe.InitBuffer(inQueueX, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(inQueueY, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(reduceWorkBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(scalarBuf, 32 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsPerBlock; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> yLocal = inQueueY.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * this->outFeatures], this->outFeatures);
        AscendC::DataCopy(yLocal, yGm[rowIdx * this->outFeatures], this->outFeatures);
        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void Compute(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = inQueueY.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceWork = reduceWorkBuf.Get<float>();
        AscendC::LocalTensor<float> scalar = scalarBuf.Get<float>();

        AscendC::ReduceSum<float>(scalar, xLocal, reduceWork, this->outFeatures);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float sumX = scalar.GetValue(0);
        float mean = sumX * this->invN;

        AscendC::Adds(tmp, xLocal, -mean, this->outFeatures);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(zLocal, tmp, tmp, this->outFeatures);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::ReduceSum<float>(scalar, zLocal, reduceWork, this->outFeatures);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
        float varSum = scalar.GetValue(0);
        float var = varSum * this->invN;
        float invStd = 1.0f / sqrtf(var + this->eps);

        AscendC::Muls(tmp, tmp, invStd, this->outFeatures);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(tmp, tmp, yLocal, this->outFeatures);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(zLocal, tmp, yLocal, this->outFeatures);
        AscendC::PipeBarrier<PIPE_V>();

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[rowIdx * this->outFeatures], zLocal, this->outFeatures);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, reduceWorkBuf, scalarBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t outFeatures;
    uint32_t rowsPerBlock;
    float invN;
    float eps;
};

extern "C" __global__ __aicore__ void bmm_instance_norm_sum_residual_add_multiply_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelBmmInstanceNormFused op;
    op.Init(x, y, z, tiling_data.batchSize, tiling_data.outFeatures,
            tiling_data.invN, tiling_data.eps);
    op.Process();
}
