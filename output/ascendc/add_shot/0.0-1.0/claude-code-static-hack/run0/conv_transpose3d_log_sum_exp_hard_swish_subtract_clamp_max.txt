kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t C_NUM = 16;

class KernelLseHsSubClamp {
public:
    __aicore__ inline KernelLseHsSubClamp() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias_in, GM_ADDR z,
                                 uint32_t totalLen, uint32_t paddedSpatial,
                                 uint32_t tileLength, uint32_t batchPerCore,
                                 uint32_t totalBatches)
    {
        this->paddedSpatial = paddedSpatial;
        this->tileLength = tileLength;
        this->totalBatches = totalBatches;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->batchStart = blockIdx * batchPerCore;
        uint32_t bEnd = this->batchStart + batchPerCore;
        if (bEnd > totalBatches) {
            bEnd = totalBatches;
        }
        this->batchEnd = bEnd;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, totalLen);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias_in, 8);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z, totalBatches * paddedSpatial);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(mBuf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(sBuf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(newMBuf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(t1Buf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(t2Buf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(constBuf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(biasBuf, 32);

        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();
        AscendC::DataCopy(biasLocal, biasGm, 8);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        biasScalar = biasLocal.GetValue(0);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t b = this->batchStart; b < this->batchEnd; b++) {
            ProcessBatch(b);
        }
    }

private:
    __aicore__ inline void ProcessBatch(uint32_t batchIdx)
    {
        uint32_t numFullTiles = this->paddedSpatial / this->tileLength;
        uint32_t remainder = this->paddedSpatial - numFullTiles * this->tileLength;

        for (uint32_t t = 0; t < numFullTiles; t++) {
            uint32_t posOffset = t * this->tileLength;
            ProcessTile(batchIdx, posOffset, this->tileLength);
        }
        if (remainder > 0) {
            uint32_t posOffset = numFullTiles * this->tileLength;
            ProcessTile(batchIdx, posOffset, remainder);
        }
    }

    __aicore__ inline void ProcessTile(uint32_t batchIdx, uint32_t posOffset, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_X> mLocal = mBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> sLocal = sBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> newMLocal = newMBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> t1Local = t1Buf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> t2Local = t2Buf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> constLocal = constBuf.Get<DTYPE_X>();

        uint32_t channelStride = this->paddedSpatial;
        uint32_t batchInputBase = batchIdx * (uint32_t)C_NUM * this->paddedSpatial + posOffset;
        uint32_t batchOutputBase = batchIdx * this->paddedSpatial + posOffset;

        {
            AscendC::LocalTensor<DTYPE_X> xc = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xc, xGm[batchInputBase], curLen);
            inQueueX.EnQue(xc);
            AscendC::LocalTensor<DTYPE_X> xR = inQueueX.DeQue<DTYPE_X>();
            AscendC::Adds(mLocal, xR, (DTYPE_X)0.0f, curLen);
            AscendC::Duplicate(sLocal, (DTYPE_X)1.0f, curLen);
            inQueueX.FreeTensor(xR);
        }

        for (int32_t c = 1; c < C_NUM; c++) {
            AscendC::LocalTensor<DTYPE_X> xc = inQueueX.AllocTensor<DTYPE_X>();
            uint32_t cOffset = (uint32_t)c * channelStride;
            AscendC::DataCopy(xc, xGm[batchInputBase + cOffset], curLen);
            inQueueX.EnQue(xc);
            AscendC::LocalTensor<DTYPE_X> xR = inQueueX.DeQue<DTYPE_X>();

            AscendC::Max(newMLocal, mLocal, xR, curLen);
            AscendC::Sub(t1Local, mLocal, newMLocal, curLen);
            AscendC::Exp(t1Local, t1Local, curLen);
            AscendC::Mul(sLocal, sLocal, t1Local, curLen);
            AscendC::Sub(t2Local, xR, newMLocal, curLen);
            AscendC::Exp(t2Local, t2Local, curLen);
            AscendC::Add(sLocal, sLocal, t2Local, curLen);
            AscendC::Adds(mLocal, newMLocal, (DTYPE_X)0.0f, curLen);

            inQueueX.FreeTensor(xR);
        }

        AscendC::Ln(sLocal, sLocal, curLen);
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::Add(outLocal, sLocal, mLocal, curLen);

        AscendC::Adds(t1Local, outLocal, (DTYPE_X)3.0f, curLen);
        AscendC::Muls(t1Local, t1Local, (DTYPE_X)(-1.0f), curLen);
        AscendC::Exp(t1Local, t1Local, curLen);
        AscendC::Adds(t1Local, t1Local, (DTYPE_X)1.0f, curLen);
        AscendC::Reciprocal(t1Local, t1Local, curLen);
        AscendC::Mul(outLocal, outLocal, t1Local, curLen);
        AscendC::Muls(outLocal, outLocal, (DTYPE_X)(1.0f / 6.0f), curLen);

        AscendC::Adds(outLocal, outLocal, (DTYPE_X)(-biasScalar), curLen);

        AscendC::Duplicate(constLocal, (DTYPE_X)1.0f, curLen);
        AscendC::Min(outLocal, outLocal, constLocal, curLen);
        AscendC::Duplicate(constLocal, (DTYPE_X)(-1.0f), curLen);
        AscendC::Max(outLocal, outLocal, constLocal, curLen);

        outQueueZ.EnQue<DTYPE_Z>(outLocal);
        AscendC::LocalTensor<DTYPE_Z> zOut = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[batchOutputBase], zOut, curLen);
        outQueueZ.FreeTensor(zOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> newMBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> t1Buf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> t2Buf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> constBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t paddedSpatial;
    uint32_t tileLength;
    uint32_t totalBatches;
    uint32_t batchStart;
    uint32_t batchEnd;
    DTYPE_BIAS biasScalar;
};

extern "C" __global__ __aicore__ void conv_transpose3d_log_sum_exp_hard_swish_subtract_clamp_max_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelLseHsSubClamp op;
    op.Init(x, bias, z,
            tiling_data.totalLen,
            tiling_data.paddedSpatial,
            tiling_data.tileLength,
            tiling_data.batchPerCore,
            tiling_data.totalBatches);
    op.Process();
}
