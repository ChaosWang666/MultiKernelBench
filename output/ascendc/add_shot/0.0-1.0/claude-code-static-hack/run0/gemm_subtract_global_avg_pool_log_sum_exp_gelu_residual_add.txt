kernel_src="""
#include "kernel_operator.h"

constexpr int32_t SCALAR_BUF_LEN = 8;
constexpr int32_t WORK_BUF_LEN = 1024;

class KernelFusedOp {
public:
    __aicore__ inline KernelFusedOp() {}

    __aicore__ inline void Init(GM_ADDR gemm_out, GM_ADDR subtract, GM_ADDR original, GM_ADDR result,
                                uint32_t batchSize, uint32_t outFeatures, uint32_t rowsPerCore, float invOutFeatures)
    {
        this->batchSize = batchSize;
        this->outFeatures = outFeatures;
        this->invOutFeatures = invOutFeatures;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerCore;
        uint32_t rows;
        if (startRow >= batchSize) {
            rows = 0;
            startRow = 0;
        } else if (startRow + rowsPerCore > batchSize) {
            rows = batchSize - startRow;
        } else {
            rows = rowsPerCore;
        }
        this->actualRows = rows;
        this->startRow = startRow;

        gemmGm.SetGlobalBuffer((__gm__ float *)gemm_out + startRow * outFeatures, rows * outFeatures);
        subtractGm.SetGlobalBuffer((__gm__ float *)subtract, outFeatures);
        originalGm.SetGlobalBuffer((__gm__ float *)original + startRow * outFeatures, rows * outFeatures);
        resultGm.SetGlobalBuffer((__gm__ float *)result + startRow * outFeatures, rows * outFeatures);

        pipe.InitBuffer(inQueueGemm, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(inQueueOriginal, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(outQueueResult, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(subtractBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(workBuf, WORK_BUF_LEN * sizeof(float));
        pipe.InitBuffer(scalarBuf, SCALAR_BUF_LEN * sizeof(float));
        pipe.InitBuffer(tmpBuf, SCALAR_BUF_LEN * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->actualRows == 0) return;

        AscendC::LocalTensor<float> subLocal = subtractBuf.Get<float>();
        AscendC::DataCopy(subLocal, subtractGm, this->outFeatures);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

        for (uint32_t row = 0; row < this->actualRows; row++) {
            ProcessRow(row);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t row)
    {
        AscendC::LocalTensor<float> gemmLocal = inQueueGemm.AllocTensor<float>();
        AscendC::DataCopy(gemmLocal, gemmGm[row * this->outFeatures], this->outFeatures);
        inQueueGemm.EnQue(gemmLocal);

        gemmLocal = inQueueGemm.DeQue<float>();
        AscendC::LocalTensor<float> subLocal = subtractBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        AscendC::Sub(gemmLocal, gemmLocal, subLocal, this->outFeatures);
        AscendC::ReduceSum<float>(scalarLocal, gemmLocal, workLocal, this->outFeatures);
        inQueueGemm.FreeTensor(gemmLocal);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

        float sumVal = scalarLocal.GetValue(0);
        float meanVal = sumVal * this->invOutFeatures;
        float meanCube = meanVal * meanVal * meanVal;
        float negZ = -1.5957691216057308f * (meanVal + 0.044715f * meanCube);

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        AscendC::Duplicate(tmpLocal, negZ, SCALAR_BUF_LEN);
        AscendC::Exp(tmpLocal, tmpLocal, SCALAR_BUF_LEN);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID1);

        float expNegZ = tmpLocal.GetValue(0);
        float geluVal = meanVal / (1.0f + expNegZ);

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID1);

        AscendC::LocalTensor<float> origLocal = inQueueOriginal.AllocTensor<float>();
        AscendC::DataCopy(origLocal, originalGm[row * this->outFeatures], this->outFeatures);
        inQueueOriginal.EnQue(origLocal);

        origLocal = inQueueOriginal.DeQue<float>();
        AscendC::LocalTensor<float> resLocal = outQueueResult.AllocTensor<float>();
        AscendC::Adds(resLocal, origLocal, geluVal, this->outFeatures);
        outQueueResult.EnQue(resLocal);
        inQueueOriginal.FreeTensor(origLocal);

        resLocal = outQueueResult.DeQue<float>();
        AscendC::DataCopy(resultGm[row * this->outFeatures], resLocal, this->outFeatures);
        outQueueResult.FreeTensor(resLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueGemm;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueOriginal;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueResult;
    AscendC::TBuf<AscendC::TPosition::VECCALC> subtractBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;

    AscendC::GlobalTensor<float> gemmGm;
    AscendC::GlobalTensor<float> subtractGm;
    AscendC::GlobalTensor<float> originalGm;
    AscendC::GlobalTensor<float> resultGm;

    uint32_t batchSize;
    uint32_t outFeatures;
    uint32_t actualRows;
    uint32_t startRow;
    float invOutFeatures;
};

extern "C" __global__ __aicore__ void gemm_subtract_global_avg_pool_log_sum_exp_gelu_residual_add_custom(
    GM_ADDR gemm_out, GM_ADDR subtract, GM_ADDR original, GM_ADDR result, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedOp op;
    op.Init(gemm_out, subtract, original, result,
            tiling_data.batchSize, tiling_data.outFeatures, tiling_data.rowsPerCore, tiling_data.invOutFeatures);
    op.Process();
}
