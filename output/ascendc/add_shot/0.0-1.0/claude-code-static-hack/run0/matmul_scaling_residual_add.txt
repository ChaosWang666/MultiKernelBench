    uint32_t singleCoreM = (M + coreNum - 1) / coreNum;
    singleCoreM = (singleCoreM + alignM - 1) / alignM * alignM;
    if (singleCoreM == 0) {
        singleCoreM = alignM;
    }
    uint32_t usedCoreNum = (M + singleCoreM - 1) / singleCoreM;
    if (usedCoreNum == 0) {
        usedCoreNum = 1;
    }

    tiling.set_M(M);
    tiling.set_N(N);
    tiling.set_K(K);
    tiling.set_singleCoreM(singleCoreM);

    matmul_tiling::MatmulApiTiling cubeTiling(ascendcPlatform);
    cubeTiling.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                        matmul_tiling::DataType::DT_FLOAT, false);

    cubeTiling.SetBias(true);

kernel_src="""
#define ASCENDC_CUBE_ONLY
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

extern "C" __global__ __aicore__ void matmul_scaling_residual_add_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);

    using A_TYPE = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, DTYPE_X, false>;
    using B_TYPE = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, DTYPE_WEIGHT, true>;
    using C_TYPE = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, DTYPE_Y>;
    using BIAS_TYPE = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, DTYPE_BIAS>;

    AscendC::TPipe pipe;
    AscendC::Matmul<A_TYPE, B_TYPE, C_TYPE, BIAS_TYPE> mm;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm, &tilingData.cubeTiling);

    uint32_t blockIdx = static_cast<uint32_t>(AscendC::GetBlockIdx());
    uint32_t M = tilingData.M;
    uint32_t N = tilingData.N;
    uint32_t K = tilingData.K;
    uint32_t singleCoreM = tilingData.singleCoreM;

    uint32_t mStart = blockIdx * singleCoreM;
    if (mStart >= M) {
        return;
    }
    uint32_t actualM = singleCoreM;
    if (mStart + singleCoreM > M) {
        actualM = M - mStart;
    }

    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_WEIGHT> weightGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x);
    weightGm.SetGlobalBuffer((__gm__ DTYPE_WEIGHT*)weight);
    biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS*)bias);
    yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y);

    mm.SetTensorA(xGm[mStart * K]);
    mm.SetTensorB(weightGm);
    mm.SetBias(biasGm);
    if (actualM != singleCoreM) {
        mm.SetTail(static_cast<int>(actualM), static_cast<int>(N), static_cast<int>(K));
    }
    mm.IterateAll(yGm[mStart * N]);
    mm.End();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, scaling_factor):
        super().__init__()
        self.matmul = nn.Linear(in_features, out_features)
        self.scaling_factor = scaling_factor

    def forward(self, x):
        y = custom_ops_lib.matmul_scaling_residual_add_custom(
            x.contiguous(),
            self.matmul.weight.contiguous(),
            self.matmul.bias.contiguous(),
        )
        return y * (1.0 + self.scaling_factor)
