kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelMatmulGeluSoftmax {
public:
    __aicore__ inline KernelMatmulGeluSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t totalRows,
                                 uint32_t rowLength,
                                 uint32_t rowsPerBlock)
    {
        this->rowLength = rowLength;
        this->rowsPerBlock = rowsPerBlock;

        uint32_t blockId = AscendC::GetBlockIdx();
        uint32_t startRow = blockId * rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ float *)x + startRow * rowLength, rowsPerBlock * rowLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + startRow * rowLength, rowsPerBlock * rowLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, rowLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, rowLength * sizeof(float));
        pipe.InitBuffer(workBuf, rowLength * sizeof(float));
        pipe.InitBuffer(reduceBuf, 64 * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsPerBlock; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->rowLength], this->rowLength);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceBuf.Get<float>();

        int32_t cnt = static_cast<int32_t>(this->rowLength);

        AscendC::ReduceMax<float>(reduceTmp, xLocal, yLocal, cnt, false);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float maxVal = reduceTmp.GetValue(0);
        float negMax = -maxVal;
        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        AscendC::Adds(work, xLocal, negMax, cnt);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Exp(work, work, cnt);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::ReduceSum<float>(reduceTmp, work, yLocal, cnt);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float sumVal = reduceTmp.GetValue(0);
        float invSum = 1.0f / sumVal;
        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        AscendC::Muls(yLocal, work, invSum, cnt);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->rowLength], yLocal, this->rowLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t rowLength;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void matmul_gelu_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulGeluSoftmax op;
    op.Init(x, y, tiling_data.totalRows, tiling_data.rowLength, tiling_data.rowsPerBlock);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features):
        super().__init__()
        self.linear = torch.nn.Linear(in_features, out_features)

    def forward(self, x):
        x = self.linear(x)
        x = torch.nn.functional.gelu(x)
        x = custom_ops_lib.matmul_gelu_softmax_custom(x.contiguous())
