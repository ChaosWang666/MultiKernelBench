kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulAvgPoolGeluScaleMax {
public:
    __aicore__ inline KernelMatmulAvgPoolGeluScaleMax() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                 uint32_t batchSize,
                                 uint32_t inFeatures,
                                 uint32_t poolLen,
                                 float scaleFactor,
                                 float invPoolK)
    {
        this->inFeatures = inFeatures;
        this->poolLen = poolLen;
        this->scaleFactor = scaleFactor;
        this->invPoolK = invPoolK;

        uint32_t coreNum = AscendC::GetBlockNum();
        uint32_t coreIdx = AscendC::GetBlockIdx();
        this->rowsPerCore = batchSize / coreNum;
        uint32_t startRow = coreIdx * this->rowsPerCore;

        xGm.SetGlobalBuffer((__gm__ float*)x + startRow * inFeatures,
                            this->rowsPerCore * inFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z + startRow, this->rowsPerCore);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->inFeatures * sizeof(float));
        pipe.InitBuffer(midBuf, this->inFeatures * sizeof(float) / 2);
        pipe.InitBuffer(workBuf, 2048);
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsPerCore; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * this->inFeatures], this->inFeatures);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> midLocal = midBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outQueueZ.AllocTensor<float>();

        AscendC::PairReduceSum<float>(midLocal, xLocal, 128, 64, 4, 1, 8);
        AscendC::PairReduceSum<float>(xLocal, midLocal, 64, 64, 4, 1, 8);
        AscendC::PairReduceSum<float>(midLocal, xLocal, 32, 64, 4, 1, 8);
        AscendC::PairReduceSum<float>(xLocal, midLocal, 16, 64, 4, 1, 8);

        AscendC::Muls(xLocal, xLocal, this->invPoolK, this->poolLen);
        AscendC::Gelu(midLocal, xLocal, this->poolLen);
        AscendC::Muls(midLocal, midLocal, this->scaleFactor, this->poolLen);
        AscendC::ReduceMax<float>(outLocal, midLocal, workLocal, this->poolLen, false);

        outQueueZ.EnQue<float>(outLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> outLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopyExtParams params;
        params.blockCount = 1;
        params.blockLen = sizeof(float);
        params.srcStride = 0;
        params.dstStride = 0;
        AscendC::DataCopyPad(zGm[rowIdx], outLocal, params);
        outQueueZ.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> midBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t rowsPerCore;
    uint32_t inFeatures;
    uint32_t poolLen;
    float scaleFactor;
    float invPoolK;
};

extern "C" __global__ __aicore__ void matmul_avg_pool_gelu_scale_max_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulAvgPoolGeluScaleMax op;
    op.Init(x, z, tiling_data.batchSize, tiling_data.inFeatures,
            tiling_data.poolLen, tiling_data.scaleFactor, tiling_data.invPoolK);
    op.Process();
}
