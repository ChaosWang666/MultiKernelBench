kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;
constexpr int32_t OUT_BATCH = 8;

class KernelFusion {
public:
    __aicore__ inline KernelFusion() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t spatialSize, uint32_t pairsPerCore, float invSize)
    {
        this->spatialSize = spatialSize;
        this->pairsPerCore = pairsPerCore;
        this->invSize = invSize;

        uint32_t coreOffset = AscendC::GetBlockIdx() * pairsPerCore;
        xGm.SetGlobalBuffer((__gm__ float *)x + coreOffset * spatialSize, pairsPerCore * spatialSize);
        yGm.SetGlobalBuffer((__gm__ float *)y + coreOffset, pairsPerCore);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, spatialSize * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, OUT_BATCH * sizeof(float));
        pipe.InitBuffer(tmpBuf, OUT_BATCH * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t numBatches = pairsPerCore / OUT_BATCH;
        for (uint32_t b = 0; b < numBatches; b++) {
            ComputeBatch(b);
        }
    }

private:
    __aicore__ inline void ComputeBatch(uint32_t b)
    {
        AscendC::LocalTensor<float> outLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        for (uint32_t k = 0; k < OUT_BATCH; k++) {
            uint32_t pairIdx = b * OUT_BATCH + k;
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[pairIdx * spatialSize], spatialSize);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::Mins(xLocal, xLocal, 1.0f, spatialSize);
            AscendC::Maxs(xLocal, xLocal, -1.0f, spatialSize);

            AscendC::ReduceSum<float>(outLocal[k], xLocal, xLocal, spatialSize);

            inQueueX.FreeTensor(xLocal);
        }

        AscendC::Muls(outLocal, outLocal, invSize, OUT_BATCH);

        AscendC::Muls(outLocal, outLocal, 2.0f, OUT_BATCH);
        AscendC::Exp(outLocal, outLocal, OUT_BATCH);
        AscendC::Adds(tmp, outLocal, 1.0f, OUT_BATCH);
        AscendC::Adds(outLocal, outLocal, -1.0f, OUT_BATCH);
        AscendC::Reciprocal(tmp, tmp, OUT_BATCH);
        AscendC::Mul(outLocal, outLocal, tmp, OUT_BATCH);

        outQueueY.EnQue(outLocal);
        AscendC::LocalTensor<float> outDeq = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[b * OUT_BATCH], outDeq, OUT_BATCH);
        outQueueY.FreeTensor(outDeq);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t spatialSize;
    uint32_t pairsPerCore;
    float invSize;
};

extern "C" __global__ __aicore__ void conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusion op;
    op.Init(x, y, tiling_data.spatialSize, tiling_data.pairsPerCore, tiling_data.invSize);
    op.Process();
}
