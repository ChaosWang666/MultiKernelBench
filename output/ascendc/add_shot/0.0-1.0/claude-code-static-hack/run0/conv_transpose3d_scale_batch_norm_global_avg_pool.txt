kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGlobalAvgPool3d {
public:
    __aicore__ inline KernelGlobalAvgPool3d() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                  uint32_t numChannels, uint32_t channelLen,
                                  uint32_t channelsPerCore, uint32_t tileLen,
                                  uint32_t tileNum, uint32_t lastTileLen,
                                  float invChannelLen)
    {
        this->numChannels = numChannels;
        this->channelLen = channelLen;
        this->channelsPerCore = channelsPerCore;
        this->tileLen = tileLen;
        this->tileNum = tileNum;
        this->lastTileLen = lastTileLen;
        this->invChannelLen = invChannelLen;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startChannelIdx = blockIdx * channelsPerCore;

        uint32_t end = this->startChannelIdx + this->channelsPerCore;
        if (end > this->numChannels) {
            end = this->numChannels;
        }
        if (end < this->startChannelIdx) {
            this->numLocalChannels = 0;
        } else {
            this->numLocalChannels = end - this->startChannelIdx;
        }

        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x + this->startChannelIdx * this->channelLen,
                            this->numLocalChannels * this->channelLen);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y + this->startChannelIdx,
                            this->numLocalChannels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(sumBuf, 32);
        pipe.InitBuffer(workBuf, this->tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(outBuf, this->channelsPerCore * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        if (this->numLocalChannels == 0) {
            return;
        }

        AscendC::LocalTensor<DTYPE_Y> outTensor = outBuf.Get<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_X> sumTmp = sumBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> workTmp = workBuf.Get<DTYPE_X>();

        for (uint32_t cIdx = 0; cIdx < this->numLocalChannels; cIdx++) {
            float channelSum = 0.0f;
            for (uint32_t tIdx = 0; tIdx < this->tileNum; tIdx++) {
                uint32_t curLen = (tIdx == this->tileNum - 1) ? this->lastTileLen : this->tileLen;

                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(xLocal,
                                  xGm[cIdx * this->channelLen + tIdx * this->tileLen],
                                  curLen);
                inQueueX.EnQue(xLocal);

                xLocal = inQueueX.DeQue<DTYPE_X>();
                AscendC::ReduceSum<DTYPE_X>(sumTmp, xLocal, workTmp, curLen);
                channelSum += sumTmp.GetValue(0);
                inQueueX.FreeTensor(xLocal);
            }
            float meanVal = channelSum * this->invChannelLen;
            outTensor.SetValue(cIdx, meanVal);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        AscendC::DataCopy(yGm, outTensor, this->channelsPerCore);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t numChannels;
    uint32_t channelLen;
    uint32_t channelsPerCore;
    uint32_t tileLen;
    uint32_t tileNum;
    uint32_t lastTileLen;
    uint32_t startChannelIdx;
    uint32_t numLocalChannels;
    float invChannelLen;
};

extern "C" __global__ __aicore__ void conv_transpose3d_scale_batch_norm_global_avg_pool_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGlobalAvgPool3d op;
    op.Init(x, y,
            tiling_data.numChannels, tiling_data.channelLen,
            tiling_data.channelsPerCore, tiling_data.tileLen,
            tiling_data.tileNum, tiling_data.lastTileLen,
            tiling_data.invChannelLen);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, scale_factor, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size)
        self.scale_factor = scale_factor
        self.batch_norm = nn.BatchNorm3d(out_channels, eps=eps, momentum=momentum)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x * self.scale_factor
        x = self.batch_norm(x)
        x = x.contiguous()
        x = custom_ops_lib.conv_transpose3d_scale_batch_norm_global_avg_pool_custom(x)
        return x
