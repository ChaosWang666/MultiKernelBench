kernel_src="""
#include "kernel_operator.h"
#include <math.h>

constexpr int32_t BUFFER_NUM = 2;

class KernelLogSumExp {
public:
    __aicore__ inline KernelLogSumExp() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t rowSize, uint32_t rowsPerCore)
    {
        this->rowSize = rowSize;
        this->rowsPerCore = rowsPerCore;

        uint32_t coreOffset = AscendC::GetBlockIdx() * rowsPerCore;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + coreOffset * rowSize, rowsPerCore * rowSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + coreOffset, rowsPerCore);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, rowSize * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, 1, rowsPerCore * sizeof(DTYPE_Y));
        pipe.InitBuffer(workBuf, 1024);
        pipe.InitBuffer(scalarBuf, 64);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        for (uint32_t i = 0; i < rowsPerCore; i++) {
            CopyIn(i);
            ComputeRow(i, yLocal);
        }
        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(AscendC::EVENT_ID3);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(AscendC::EVENT_ID3);
        outQueueY.EnQue(yLocal);
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn(uint32_t row)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[row * rowSize], rowSize);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void ComputeRow(uint32_t row, AscendC::LocalTensor<DTYPE_Y>& yLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> scalar = scalarBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();

        AscendC::ReduceMax<float>(scalar, xLocal, work, rowSize, false);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(AscendC::EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(AscendC::EVENT_ID0);
        float maxVal = scalar.GetValue(0);
        AscendC::SetFlag<AscendC::HardEvent::S_V>(AscendC::EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(AscendC::EVENT_ID0);

        AscendC::Adds(xLocal, xLocal, -maxVal, rowSize);
        AscendC::Exp(xLocal, xLocal, rowSize);
        AscendC::ReduceSum<float>(scalar, xLocal, work, rowSize);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(AscendC::EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(AscendC::EVENT_ID1);
        float sumVal = scalar.GetValue(0);

        float result = log(sumVal) + maxVal;
        yLocal.SetValue(row, result);

        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm, yLocal, rowsPerCore);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf, scalarBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t rowSize;
    uint32_t rowsPerCore;
};

extern "C" __global__ __aicore__ void gemm_sigmoid_sum_log_sum_exp_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelLogSumExp op;
    op.Init(x, y, tiling_data.rowSize, tiling_data.rowsPerCore);
    op.Process();
}
