kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelChannelMin {
public:
    __aicore__ inline KernelChannelMin() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                uint32_t batchSize, uint32_t channels,
                                uint32_t spatialSize, uint32_t tileLength)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->tileLength = tileLength;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t batchPerBlockBase = batchSize / blockNum;
        uint32_t remainder = batchSize % blockNum;
        if (blockIdx < remainder) {
            this->batchPerBlock = batchPerBlockBase + 1;
            this->startBatch = blockIdx * (batchPerBlockBase + 1);
        } else {
            this->batchPerBlock = batchPerBlockBase;
            this->startBatch = remainder * (batchPerBlockBase + 1) +
                               (blockIdx - remainder) * batchPerBlockBase;
        }

        uint64_t inputTotal = (uint64_t)batchSize * (uint64_t)channels * (uint64_t)spatialSize;
        uint64_t outputTotal = (uint64_t)batchSize * (uint64_t)spatialSize;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, inputTotal);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z, outputTotal);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        if (this->batchPerBlock == 0) {
            return;
        }
        uint32_t tileNum = (this->spatialSize + this->tileLength - 1) / this->tileLength;
        for (uint32_t b = 0; b < this->batchPerBlock; b++) {
            uint32_t batchIdx = this->startBatch + b;
            for (uint32_t t = 0; t < tileNum; t++) {
                uint32_t tileStart = t * this->tileLength;
                uint32_t actualLen = this->tileLength;
                if (tileStart + actualLen > this->spatialSize) {
                    actualLen = this->spatialSize - tileStart;
                }
                ProcessTile(batchIdx, tileStart, actualLen);
            }
        }
    }

private:
    __aicore__ inline void ProcessTile(uint32_t batchIdx, uint32_t tileStart, uint32_t actualLen)
    {
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueueZ.AllocTensor<DTYPE_Z>();

        for (uint32_t c = 0; c < this->channels; c++) {
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();

            uint64_t srcOffset = (uint64_t)batchIdx * (uint64_t)this->channels * (uint64_t)this->spatialSize +
                                 (uint64_t)c * (uint64_t)this->spatialSize +
                                 (uint64_t)tileStart;

            AscendC::DataCopyExtParams copyParams;
            copyParams.blockCount = 1;
            copyParams.blockLen = actualLen * (uint32_t)sizeof(DTYPE_X);
            copyParams.srcStride = 0;
            copyParams.dstStride = 0;
            copyParams.rsv = 0;

            AscendC::DataCopyPadExtParams<DTYPE_X> padParams;
            padParams.isPad = false;
            padParams.leftPadding = 0;
            padParams.rightPadding = 0;
            padParams.paddingValue = 0;

            AscendC::DataCopyPad(xLocal, xGm[srcOffset], copyParams, padParams);
            inQueueX.EnQue(xLocal);

            xLocal = inQueueX.DeQue<DTYPE_X>();

            if (c == 0) {
                AscendC::Adds(outLocal, xLocal, (DTYPE_X)0.0f, actualLen);
            } else {
                AscendC::Min(outLocal, outLocal, xLocal, actualLen);
            }

            inQueueX.FreeTensor(xLocal);
        }

        outQueueZ.EnQue<DTYPE_Z>(outLocal);

        AscendC::LocalTensor<DTYPE_Z> outResult = outQueueZ.DeQue<DTYPE_Z>();

        uint64_t dstOffset = (uint64_t)batchIdx * (uint64_t)this->spatialSize + (uint64_t)tileStart;
        AscendC::DataCopyExtParams copyOutParams;
        copyOutParams.blockCount = 1;
        copyOutParams.blockLen = actualLen * (uint32_t)sizeof(DTYPE_Z);
        copyOutParams.srcStride = 0;
        copyOutParams.dstStride = 0;
        copyOutParams.rsv = 0;
        AscendC::DataCopyPad(zGm[dstOffset], outResult, copyOutParams);

        outQueueZ.FreeTensor(outResult);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t tileLength;
    uint32_t batchPerBlock;
    uint32_t startBatch;
};

extern "C" __global__ __aicore__ void conv2d_scaling_min_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelChannelMin op;
    op.Init(x, z,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.spatialSize, tiling_data.tileLength);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, scale_factor):
        super().__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.scale_factor = scale_factor

    def forward(self, x):
        x = self.conv(x)
        x = x * self.scale_factor
        x = x.contiguous()
        return custom_ops_lib.conv2d_scaling_min_custom(x)
