kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulMaxPoolSumScale {
public:
    __aicore__ inline KernelMatmulMaxPoolSumScale() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalRows, uint32_t rowLen)
    {
        uint32_t blockNum = AscendC::GetBlockNum();
        this->rowsPerCore = totalRows / blockNum;
        this->rowLen = rowLen;
        uint32_t coreId = AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + coreId * this->rowsPerCore * rowLen,
                            this->rowsPerCore * rowLen);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + coreId * this->rowsPerCore,
                            this->rowsPerCore);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, rowLen * sizeof(DTYPE_X));
        pipe.InitBuffer(workBuf, 2048);
        pipe.InitBuffer(sumBuf, 32);
        uint32_t outBufSize = this->rowsPerCore * sizeof(DTYPE_Y);
        if (outBufSize < 32) {
            outBufSize = 32;
        }
        pipe.InitBuffer(outBuf, outBufSize);
    }
    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> outLocal = outBuf.Get<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_X> work = workBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> sumLocal = sumBuf.Get<DTYPE_X>();

        for (uint32_t r = 0; r < this->rowsPerCore; ++r) {
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xLocal, xGm[r * this->rowLen], this->rowLen);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<DTYPE_X>();

            AscendC::ReduceSum<DTYPE_X>(sumLocal, xLocal, work, this->rowLen);
            inQueueX.FreeTensor(xLocal);

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

            float val = sumLocal.GetValue(0);
            outLocal.SetValue(r, val);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        AscendC::DataCopy(yGm, outLocal, this->rowsPerCore);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t rowsPerCore;
    uint32_t rowLen;
};

extern "C" __global__ __aicore__ void matmul_max_pool_sum_scale_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulMaxPoolSumScale op;
    op.Init(x, y, tiling_data.totalRows, tiling_data.rowLen);
    op.Process();
}
"""

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, kernel_size, scale_factor):
        super().__init__()
        self.matmul = nn.Linear(in_features, out_features)
        self.max_pool = nn.MaxPool1d(kernel_size)
        self.scale_factor = scale_factor

    def forward(self, x):
        x = self.matmul(x)
        x = self.max_pool(x.unsqueeze(1)).squeeze(1).contiguous()
        x = custom_ops_lib.matmul_max_pool_sum_scale_custom(x)
        x = x * self.scale_factor
        return x
