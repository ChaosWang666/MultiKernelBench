kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv3dGroupNormMean {
public:
    __aicore__ inline KernelConv3dGroupNormMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t batchLength, uint32_t tileLength, uint32_t tileNum, float invCount)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->batchesPerBlock = batchSize / blockNum;
        this->batchLength = batchLength;
        this->tileLength = tileLength;
        this->tileNum = tileNum;
        this->invCount = invCount;

        uint32_t batchOffset = blockIdx * this->batchesPerBlock;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + batchOffset * this->batchLength, this->batchesPerBlock * this->batchLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + batchOffset, this->batchesPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(workBuf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(sumQueue, 1, 32);
        pipe.InitBuffer(outBuf, this->batchesPerBlock * sizeof(DTYPE_Y));
    }
    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> outLocal = outBuf.Get<DTYPE_Y>();
        for (uint32_t b = 0; b < this->batchesPerBlock; b++) {
            float batchSum = 0.0f;
            for (uint32_t t = 0; t < this->tileNum; t++) {
                CopyIn(b, t);
                Compute();
                batchSum += DequeueAndRead();
            }
            outLocal.SetValue(b, batchSum * this->invCount);
        }
        AscendC::DataCopy(yGm, outLocal, this->batchesPerBlock);
    }

private:
    __aicore__ inline void CopyIn(uint32_t b, uint32_t t)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[b * this->batchLength + t * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> workLocal = workBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> sumLocal = sumQueue.AllocTensor<DTYPE_X>();

        AscendC::ReduceSum(sumLocal, xLocal, workLocal, this->tileLength);
        sumQueue.EnQue(sumLocal);

        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline float DequeueAndRead()
    {
        AscendC::LocalTensor<DTYPE_X> sumLocal = sumQueue.DeQue<DTYPE_X>();
        float val = sumLocal.GetValue(0);
        sumQueue.FreeTensor(sumLocal);
        return val;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> sumQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batchesPerBlock;
    uint32_t batchLength;
    uint32_t tileLength;
    uint32_t tileNum;
    float invCount;
};

extern "C" __global__ __aicore__ void conv3d_group_norm_mean_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dGroupNormMean op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.batchLength, tiling_data.tileLength, tiling_data.tileNum, tiling_data.invCount);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, num_groups):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.group_norm = nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        x = self.conv(x)
        x = self.group_norm(x)
        x = x.contiguous()
        x = custom_ops_lib.conv3d_group_norm_mean_custom(x)
        return x
