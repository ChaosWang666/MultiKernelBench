kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t OUT_BLOCK_FLOATS = 8;

class KernelFusedOp {
public:
    __aicore__ inline KernelFusedOp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t channelsArg, uint32_t spatialSizeArg,
                                 uint32_t tileLengthArg, uint32_t fullTilesArg, uint32_t tailLenArg,
                                 float hwInvArg)
    {
        this->channels = channelsArg;
        this->spatialSize = spatialSizeArg;
        this->tileLength = tileLengthArg;
        this->fullTiles = fullTilesArg;
        this->tailLen = tailLenArg;
        this->hwInv = hwInvArg;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t batchOffset = static_cast<uint64_t>(blockIdx)
                             * static_cast<uint64_t>(this->channels)
                             * static_cast<uint64_t>(this->spatialSize);

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + batchOffset,
                            static_cast<uint64_t>(this->channels) * this->spatialSize);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, this->channels);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + static_cast<uint64_t>(blockIdx) * OUT_BLOCK_FLOATS,
                            OUT_BLOCK_FLOATS);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(meansBuf, this->channels * sizeof(float));
        pipe.InitBuffer(biasBuf, this->channels * sizeof(float));
        pipe.InitBuffer(workBuf, 4096);
        pipe.InitBuffer(scalarBuf, 32);
        pipe.InitBuffer(outBuf, 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> means = meansBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> tmpScalar = scalarBuf.Get<float>();
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();

        for (uint32_t c = 0; c < this->channels; c++) {
            float channelSum = 0.0f;
            uint64_t channelBase = static_cast<uint64_t>(c) * static_cast<uint64_t>(this->spatialSize);

            for (uint32_t t = 0; t < this->fullTiles; t++) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopyExtParams cpyP{1, this->tileLength * (uint32_t)sizeof(float), 0, 0, 0};
                AscendC::DataCopyPadExtParams<float> padP{true, 0, 0, 0.0f};
                AscendC::DataCopyPad(xLocal,
                                     xGm[channelBase + static_cast<uint64_t>(t) * this->tileLength],
                                     cpyP, padP);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<float> xL = inQueueX.DeQue<float>();
                AscendC::ReduceSum<float>(tmpScalar, xL, work, this->tileLength);
                inQueueX.FreeTensor(xL);

                event_t evt = static_cast<event_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
                AscendC::SetFlag<AscendC::HardEvent::V_S>(evt);
                AscendC::WaitFlag<AscendC::HardEvent::V_S>(evt);
                channelSum += tmpScalar.GetValue(0);
            }

            if (this->tailLen > 0) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopyExtParams cpyP{1, this->tailLen * (uint32_t)sizeof(float), 0, 0, 0};
                AscendC::DataCopyPadExtParams<float> padP{true, 0, 0, 0.0f};
                AscendC::DataCopyPad(xLocal,
                                     xGm[channelBase + static_cast<uint64_t>(this->fullTiles) * this->tileLength],
                                     cpyP, padP);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<float> xL = inQueueX.DeQue<float>();
                AscendC::ReduceSum<float>(tmpScalar, xL, work, this->tailLen);
                inQueueX.FreeTensor(xL);

                event_t evt = static_cast<event_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
                AscendC::SetFlag<AscendC::HardEvent::V_S>(evt);
                AscendC::WaitFlag<AscendC::HardEvent::V_S>(evt);
                channelSum += tmpScalar.GetValue(0);
            }

            means.SetValue(c, channelSum * this->hwInv);
        }

        event_t evtSV0 = static_cast<event_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_V));
        AscendC::SetFlag<AscendC::HardEvent::S_V>(evtSV0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(evtSV0);

        AscendC::DataCopy(biasLocal, biasGm, this->channels);
        event_t evtMV = static_cast<event_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::MTE2_V));
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(evtMV);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(evtMV);

        AscendC::Add(means, means, biasLocal, this->channels);

        AscendC::ReduceMax<float>(tmpScalar, means, work, this->channels);

        event_t evtVS1 = static_cast<event_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(evtVS1);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(evtVS1);
        float maxVal = tmpScalar.GetValue(0);

        AscendC::Adds(means, means, -maxVal, this->channels);
        AscendC::Exp(means, means, this->channels);
        AscendC::ReduceSum<float>(tmpScalar, means, work, this->channels);

        event_t evtVS2 = static_cast<event_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(evtVS2);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(evtVS2);
        float sumExp = tmpScalar.GetValue(0);

        AscendC::Duplicate<float>(tmpScalar, sumExp, 8);
        AscendC::Ln<float>(tmpScalar, tmpScalar, 8);

        event_t evtVS3 = static_cast<event_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(evtVS3);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(evtVS3);
        float logVal = tmpScalar.GetValue(0);

        float result = (logVal + maxVal) * 10.0f;

        AscendC::LocalTensor<float> yLocal = outBuf.Get<float>();
        AscendC::Duplicate<float>(yLocal, 0.0f, OUT_BLOCK_FLOATS);

        event_t evtVS4 = static_cast<event_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(evtVS4);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(evtVS4);

        yLocal.SetValue(0, result);

        event_t evtSM = static_cast<event_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_MTE3));
        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(evtSM);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(evtSM);

        AscendC::DataCopy(yGm, yLocal, OUT_BLOCK_FLOATS);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> meansBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t channels;
    uint32_t spatialSize;
    uint32_t tileLength;
    uint32_t fullTiles;
    uint32_t tailLen;
    float hwInv;
};

extern "C" __global__ __aicore__ void convtranspose2d_globalavgpool_biasadd_logsumexp_sum_multiply_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedOp op;
    op.Init(x, bias, y,
            tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize,
            tiling_data.tileLength, tiling_data.fullTiles, tiling_data.tailLen,
            tiling_data.hwInv);
    op.Process();
}
