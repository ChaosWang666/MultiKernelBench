namespace optiling {
const uint32_t BLOCK_DIM = 32;
const uint32_t DEFAULT_TILE_LENGTH = 4096;
static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    Conv2dSubtractSubtractMishCustomTilingData tiling;
    uint32_t totalLength = context->GetInputShape(0)->GetOriginShape().GetShapeSize();

    uint32_t blockNum = BLOCK_DIM;
    if (totalLength % blockNum != 0) {
        blockNum = 1;
    }
    uint32_t blockLength = totalLength / blockNum;

    uint32_t tileLength = DEFAULT_TILE_LENGTH;
    if (tileLength > blockLength) {
        tileLength = blockLength;
    }
    while (tileLength > 0 && (tileLength % 8 != 0)) {
        tileLength -= 1;
    }
    if (tileLength == 0) {
        tileLength = blockLength;
    }
    uint32_t tileNum = blockLength / tileLength;
    uint32_t tailLength = blockLength - tileNum * tileLength;

    context->SetBlockDim(blockNum);
    tiling.set_totalLength(totalLength);
    tiling.set_blockLength(blockLength);
    tiling.set_tileLength(tileLength);
    tiling.set_tileNum(tileNum);
    tiling.set_tailLength(tailLength);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);

}

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMish {
public:
    __aicore__ inline KernelMish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                uint32_t blockLength, uint32_t tileLength,
                                uint32_t tileNum, uint32_t tailLength)
    {
        this->blockLength = blockLength;
        this->tileLength = tileLength;
        this->tileNum = tileNum;
        this->tailLength = tailLength;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpBufA, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(tmpBufB, this->tileLength * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->tileNum; i++) {
            CopyIn(i, this->tileLength);
            Compute(this->tileLength);
            CopyOut(i, this->tileLength);
        }
        if (this->tailLength > 0) {
            CopyIn(this->tileNum, this->tailLength);
            Compute(this->tailLength);
            CopyOut(this->tileNum, this->tailLength);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress, uint32_t length)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], length);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t length)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_X> tmpA = tmpBufA.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> tmpB = tmpBufB.Get<DTYPE_X>();

        // Mish(x) = x * tanh(softplus(x))
        // Let w = 1 + exp(x); then tanh(softplus(x)) = (w*w - 1) / (w*w + 1)
        AscendC::Exp(tmpA, xLocal, length);
        AscendC::Adds(tmpA, tmpA, (DTYPE_X)1.0f, length);
        AscendC::Mul(tmpA, tmpA, tmpA, length);
        AscendC::Adds(tmpB, tmpA, (DTYPE_X)(-1.0f), length);
        AscendC::Adds(tmpA, tmpA, (DTYPE_X)1.0f, length);
        AscendC::Div(tmpB, tmpB, tmpA, length);
        AscendC::Mul(zLocal, xLocal, tmpB, length);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress, uint32_t length)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, length);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBufA;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBufB;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t blockLength;
    uint32_t tileLength;
    uint32_t tileNum;
    uint32_t tailLength;
};

extern "C" __global__ __aicore__ void conv2d_subtract_subtract_mish_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMish op;
    op.Init(x, z, tiling_data.blockLength, tiling_data.tileLength, tiling_data.tileNum, tiling_data.tailLength);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, subtract_value_1, subtract_value_2):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.subtract_total = float(subtract_value_1) + float(subtract_value_2)

    def forward(self, x):
        x = self.conv(x)
        x = x - self.subtract_total
        x = custom_ops_lib.conv2d_subtract_subtract_mish_custom(x.contiguous())
        return x
