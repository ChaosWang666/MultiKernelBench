kernel_src="""
#include "kernel_operator.h"

class KernelMatmulSumChain {
public:
    __aicore__ inline KernelMatmulSumChain() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w_col_sum, GM_ADDR b_sum, GM_ADDR y,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t batchPerCore)
    {
        this->inFeatures = inFeatures;
        this->batchPerCore = batchPerCore;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startBatch = blockIdx * batchPerCore;

        xGm.SetGlobalBuffer((__gm__ float *)x + startBatch * inFeatures, batchPerCore * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float *)w_col_sum, inFeatures);
        bGm.SetGlobalBuffer((__gm__ float *)b_sum, 8);
        yGm.SetGlobalBuffer((__gm__ float *)y + startBatch, batchPerCore);

        pipe.InitBuffer(inQueueX, 2, inFeatures * sizeof(float));
        pipe.InitBuffer(wBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(workBuf, 2048);
        pipe.InitBuffer(sumBuf, 32);
        uint32_t yBytes = ((batchPerCore * sizeof(float) + 31) / 32) * 32;
        pipe.InitBuffer(yBuf, yBytes);
        pipe.InitBuffer(biasBuf, 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> bLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> yLocal = yBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();

        AscendC::DataCopy(wLocal, wGm, this->inFeatures);
        AscendC::DataCopy(bLocal, bGm, 8);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        float biasVal = bLocal.GetValue(0);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID1);

        for (uint32_t i = 0; i < this->batchPerCore; i++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[i * this->inFeatures], this->inFeatures);
            inQueueX.EnQue(xLocal);

            AscendC::LocalTensor<float> xDeq = inQueueX.DeQue<float>();
            AscendC::Mul(tmpLocal, xDeq, wLocal, this->inFeatures);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum<float>(sumLocal, tmpLocal, workLocal, this->inFeatures);
            inQueueX.FreeTensor(xDeq);

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID2);

            float val = sumLocal.GetValue(0) + biasVal;
            yLocal.SetValue(i, val);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID3);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID3);

        AscendC::DataCopy(yGm, yLocal, this->batchPerCore);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> yBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t inFeatures;
    uint32_t batchPerCore;
};

extern "C" __global__ __aicore__ void matmul_sum_max_avg_pool_log_sum_exp_log_sum_exp_custom(
    GM_ADDR x, GM_ADDR w_col_sum, GM_ADDR b_sum, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSumChain op;
    op.Init(x, w_col_sum, b_sum, y, tiling_data.batchSize, tiling_data.inFeatures, tiling_data.batchPerCore);
    op.Process();
}
