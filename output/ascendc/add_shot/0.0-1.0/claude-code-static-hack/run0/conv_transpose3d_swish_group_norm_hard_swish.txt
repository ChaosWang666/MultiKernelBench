kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t TILE_LENGTH = 8192;

class KernelHardSwish {
public:
    __aicore__ inline KernelHardSwish() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalLength)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t totalTiles = (totalLength + TILE_LENGTH - 1) / TILE_LENGTH;
        uint32_t tilesPerBlock = (totalTiles + blockNum - 1) / blockNum;

        this->myStartTile = blockIdx * tilesPerBlock;
        this->myEndTile = this->myStartTile + tilesPerBlock;
        if (this->myEndTile > totalTiles) {
            this->myEndTile = totalTiles;
        }
        if (this->myStartTile > totalTiles) {
            this->myStartTile = totalTiles;
        }
        this->totalLength = totalLength;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, totalLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z, totalLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, TILE_LENGTH * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, TILE_LENGTH * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpBuf, TILE_LENGTH * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = this->myStartTile; i < this->myEndTile; i++) {
            uint32_t offset = i * TILE_LENGTH;
            uint32_t curLen;
            if (offset + TILE_LENGTH <= this->totalLength) {
                curLen = TILE_LENGTH;
            } else {
                curLen = this->totalLength - offset;
            }
            CopyIn(offset, curLen);
            Compute();
            CopyOut(offset, curLen);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        if (curLen == TILE_LENGTH) {
            AscendC::DataCopy(xLocal, xGm[offset], TILE_LENGTH);
        } else {
            AscendC::DataCopyExtParams copyParams;
            copyParams.blockCount = 1;
            copyParams.blockLen = curLen * sizeof(DTYPE_X);
            copyParams.srcStride = 0;
            copyParams.dstStride = 0;
            AscendC::DataCopyPadExtParams<DTYPE_X> padParams;
            padParams.isPad = true;
            padParams.leftPadding = 0;
            padParams.rightPadding = TILE_LENGTH - curLen;
            padParams.paddingValue = 0;
            AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        }
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        AscendC::Adds(tmp, xLocal, (DTYPE_X)3.0f, TILE_LENGTH);
        AscendC::Maxs(tmp, tmp, (DTYPE_X)0.0f, TILE_LENGTH);
        AscendC::Mins(tmp, tmp, (DTYPE_X)6.0f, TILE_LENGTH);
        AscendC::Mul(tmp, tmp, xLocal, TILE_LENGTH);
        AscendC::Muls(zLocal, tmp, (DTYPE_X)(1.0f / 6.0f), TILE_LENGTH);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        if (curLen == TILE_LENGTH) {
            AscendC::DataCopy(zGm[offset], zLocal, TILE_LENGTH);
        } else {
            AscendC::DataCopyExtParams copyParams;
            copyParams.blockCount = 1;
            copyParams.blockLen = curLen * sizeof(DTYPE_Z);
            copyParams.srcStride = 0;
            copyParams.dstStride = 0;
            AscendC::DataCopyPad(zGm[offset], zLocal, copyParams);
        }
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t myStartTile;
    uint32_t myEndTile;
    uint32_t totalLength;
};

extern "C" __global__ __aicore__ void conv_transpose3d_swish_group_norm_hard_swish_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelHardSwish op;
    op.Init(x, z, tiling_data.totalLength);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, groups, eps, bias=True):
        super().__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, bias=bias)
        self.group_norm = nn.GroupNorm(num_groups=groups, num_channels=out_channels, eps=eps)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = torch.sigmoid(x) * x
        x = self.group_norm(x)
        x = x.contiguous()
        x = custom_ops_lib.conv_transpose3d_swish_group_norm_hard_swish_custom(x)
        return x
