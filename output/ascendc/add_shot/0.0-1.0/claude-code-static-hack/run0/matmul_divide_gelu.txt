kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

constexpr int32_t SYNC_AIC_AIV_EVENT = 4;
constexpr uint32_t TILE_SIZE = 4096;

class GeluPostProc {
public:
    __aicore__ inline GeluPostProc() {}

    __aicore__ inline void Init(GM_ADDR z, AscendC::TPipe* pipe,
                                 uint32_t startOffset, uint32_t totalLen)
    {
        this->myLen = totalLen;
        zGm.SetGlobalBuffer((__gm__ float*)z + startOffset, totalLen);

        pipe->InitBuffer(inQueue, 2, TILE_SIZE * sizeof(float));
        pipe->InitBuffer(outQueue, 2, TILE_SIZE * sizeof(float));
        pipe->InitBuffer(tmpBuf, TILE_SIZE * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t processed = 0;
        while (processed < myLen) {
            uint32_t curLen = (myLen - processed > TILE_SIZE) ? TILE_SIZE : (myLen - processed);
            CopyIn(processed, curLen);
            Compute(curLen);
            CopyOut(processed, curLen);
            processed += curLen;
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset, uint32_t curLen)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        AscendC::DataCopy(xLocal, zGm[offset], curLen);
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t curLen)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.DeQue<float>();
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        AscendC::Muls(tmp, xLocal, 0.70710678118654752f, curLen);
        AscendC::Erf(tmp, tmp, curLen);
        AscendC::Adds(tmp, tmp, 1.0f, curLen);
        AscendC::Mul(outLocal, xLocal, tmp, curLen);
        AscendC::Muls(outLocal, outLocal, 0.5f, curLen);

        outQueue.EnQue<float>(outLocal);
        inQueue.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset, uint32_t curLen)
    {
        AscendC::LocalTensor<float> outLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(zGm[offset], outLocal, curLen);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::GlobalTensor<float> zGm;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 2> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    uint32_t myLen;
};

extern "C" __global__ __aicore__ void matmul_divide_gelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);

    if (ASCEND_IS_AIC) {
        AscendC::TPipe pipe;

        using AType = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float>;
        using BType = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float>;
        using CType = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float>;
        using BiasType = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float>;

        AscendC::Matmul<AType, BType, CType, BiasType> mm;
        REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm, &tiling_data.cubeTilingData);

        AscendC::GlobalTensor<float> gmA, gmB, gmBias, gmZ;
        gmA.SetGlobalBuffer((__gm__ float*)x);
        gmB.SetGlobalBuffer((__gm__ float*)weight);
        gmBias.SetGlobalBuffer((__gm__ float*)bias);
        gmZ.SetGlobalBuffer((__gm__ float*)z);

        mm.SetTensorA(gmA);
        mm.SetTensorB(gmB);
        mm.SetBias(gmBias);
        mm.IterateAll(gmZ);
        mm.End();

        AscendC::CrossCoreSetFlag<0x2, PIPE_FIX>(SYNC_AIC_AIV_EVENT);
    }

    if (ASCEND_IS_AIV) {
        AscendC::CrossCoreWaitFlag(SYNC_AIC_AIV_EVENT);

        AscendC::TPipe pipe;

        uint32_t aivIdx = static_cast<uint32_t>(AscendC::GetBlockIdx());
        uint32_t aivNum = static_cast<uint32_t>(AscendC::GetBlockNum());
        uint32_t totalLength = tiling_data.totalLength;

        const uint32_t ALIGN = 8;
        uint32_t alignedTotal = (totalLength / ALIGN) * ALIGN;
        uint32_t basePerAiv = (alignedTotal / aivNum / ALIGN) * ALIGN;
        uint32_t remAfterBase = alignedTotal - basePerAiv * aivNum;
        uint32_t extraChunks = remAfterBase / ALIGN;

        uint32_t myStart;
        uint32_t myLen;
        if (aivIdx < extraChunks) {
            myLen = basePerAiv + ALIGN;
            myStart = aivIdx * myLen;
        } else {
            myLen = basePerAiv;
            myStart = extraChunks * (basePerAiv + ALIGN) + (aivIdx - extraChunks) * basePerAiv;
        }

        if (myLen > 0 && myStart < totalLength) {
            GeluPostProc op;
            op.Init(z, &pipe, myStart, myLen);
            op.Process();
        }
    }
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, input_size, output_size, divisor):
        super().__init__()
        self.linear = nn.Linear(input_size, output_size)
        self.divisor = divisor
        self._weight_t = None
        self._bias_scaled = None

    def forward(self, x):
        if self._weight_t is None:
            with torch.no_grad():
                self._weight_t = (self.linear.weight / self.divisor).t().contiguous()
                self._bias_scaled = (self.linear.bias / self.divisor).contiguous()
        return custom_ops_lib.matmul_divide_gelu_custom(x, self._weight_t, self._bias_scaled)
