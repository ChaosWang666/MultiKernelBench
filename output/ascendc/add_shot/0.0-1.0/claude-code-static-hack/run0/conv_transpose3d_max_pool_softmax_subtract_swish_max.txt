kernel_src="""
#include "kernel_operator.h"

class KernelFusedTail {
public:
    __aicore__ inline KernelFusedTail() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR sub, GM_ADDR y,
                                uint32_t totalN, uint32_t channels,
                                uint32_t spatialSize, uint32_t batchPerBlock,
                                uint32_t tileSize, uint32_t tileNum)
    {
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->batchPerBlock = batchPerBlock;
        this->tileSize = tileSize;
        this->tileNum = tileNum;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t batchStart = blockIdx * batchPerBlock;

        xGm.SetGlobalBuffer((__gm__ float*)x + (uint64_t)batchStart * channels * spatialSize,
                            (uint64_t)batchPerBlock * channels * spatialSize);
        subGm.SetGlobalBuffer((__gm__ float*)sub, channels);
        yGm.SetGlobalBuffer((__gm__ float*)y + (uint64_t)batchStart * spatialSize,
                            (uint64_t)batchPerBlock * spatialSize);

        pipe.InitBuffer(inQueueX, 1, channels * tileSize * sizeof(float));
        pipe.InitBuffer(inQueueSub, 1, channels * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, tileSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, 3 * tileSize * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> subLocal = inQueueSub.AllocTensor<float>();
        AscendC::DataCopy(subLocal, subGm, this->channels);
        inQueueSub.EnQue(subLocal);
        subLocal = inQueueSub.DeQue<float>();

        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> maxPerPos = tmp;
        AscendC::LocalTensor<float> sumPerPos = tmp[this->tileSize];
        AscendC::LocalTensor<float> sigTmp = tmp[2 * this->tileSize];

        for (uint32_t b = 0; b < this->batchPerBlock; b++) {
            for (uint32_t t = 0; t < this->tileNum; t++) {
                ProcessTile(subLocal, maxPerPos, sumPerPos, sigTmp, b, t);
            }
        }

        inQueueSub.FreeTensor(subLocal);
    }

private:
    __aicore__ inline void ProcessTile(
        AscendC::LocalTensor<float>& subLocal,
        AscendC::LocalTensor<float>& maxPerPos,
        AscendC::LocalTensor<float>& sumPerPos,
        AscendC::LocalTensor<float>& sigTmp,
        uint32_t b, uint32_t t)
    {
        uint32_t cs = this->channels;
        uint32_t ts = this->tileSize;
        uint32_t ss = this->spatialSize;

        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        uint64_t xBase = (uint64_t)b * cs * ss + (uint64_t)t * ts;
        for (uint32_t c = 0; c < cs; c++) {
            AscendC::DataCopy(xLocal[c * ts],
                              xGm[xBase + (uint64_t)c * ss],
                              ts);
        }
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        AscendC::Duplicate(maxPerPos, -1.0e30f, ts);
        for (uint32_t c = 0; c < cs; c++) {
            AscendC::Max(maxPerPos, maxPerPos, xLocal[c * ts], ts);
        }

        for (uint32_t c = 0; c < cs; c++) {
            AscendC::Sub(xLocal[c * ts], xLocal[c * ts], maxPerPos, ts);
        }

        AscendC::Exp(xLocal, xLocal, cs * ts);

        AscendC::Duplicate(sumPerPos, 0.0f, ts);
        for (uint32_t c = 0; c < cs; c++) {
            AscendC::Add(sumPerPos, sumPerPos, xLocal[c * ts], ts);
        }

        for (uint32_t c = 0; c < cs; c++) {
            AscendC::Div(xLocal[c * ts], xLocal[c * ts], sumPerPos, ts);
        }

        for (uint32_t c = 0; c < cs; c++) {
            float subVal = subLocal.GetValue(c);
            AscendC::Adds(xLocal[c * ts], xLocal[c * ts], -subVal, ts);
        }

        for (uint32_t c = 0; c < cs; c++) {
            AscendC::Muls(sigTmp, xLocal[c * ts], -1.0f, ts);
            AscendC::Exp(sigTmp, sigTmp, ts);
            AscendC::Adds(sigTmp, sigTmp, 1.0f, ts);
            AscendC::Div(xLocal[c * ts], xLocal[c * ts], sigTmp, ts);
        }

        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::Duplicate(yLocal, -1.0e30f, ts);
        for (uint32_t c = 0; c < cs; c++) {
            AscendC::Max(yLocal, yLocal, xLocal[c * ts], ts);
        }
        outQueueY.EnQue(yLocal);
        yLocal = outQueueY.DeQue<float>();

        uint64_t yOffset = (uint64_t)b * ss + (uint64_t)t * ts;
        AscendC::DataCopy(yGm[yOffset], yLocal, ts);
        outQueueY.FreeTensor(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueSub;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> subGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t batchPerBlock;
    uint32_t tileSize;
    uint32_t tileNum;
};

extern "C" __global__ __aicore__ void conv_transpose3d_max_pool_softmax_subtract_swish_max_custom(
    GM_ADDR x, GM_ADDR sub, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedTail op;
    op.Init(x, sub, y,
            tiling_data.totalN,
            tiling_data.channels,
            tiling_data.spatialSize,
            tiling_data.batchPerBlock,
            tiling_data.tileSize,
            tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, pool_kernel_size, pool_stride, pool_padding):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.max_pool = nn.MaxPool3d(kernel_size=pool_kernel_size, stride=pool_stride, padding=pool_padding)
        self.subtract = nn.Parameter(torch.randn(out_channels))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.max_pool(x)
        x = x.contiguous()
        sub = self.subtract.contiguous()
        return custom_ops_lib.conv_transpose3d_max_pool_softmax_subtract_swish_max_custom(x, sub)
