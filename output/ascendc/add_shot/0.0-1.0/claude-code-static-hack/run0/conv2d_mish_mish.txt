#include "conv2d_mish_mish_custom_tiling.h"

namespace optiling {
const uint32_t MAX_BLOCK_DIM = 32;
const uint32_t TILE_LENGTH = 4096;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    Conv2dMishMishCustomTilingData tiling;
    uint32_t totalLength = context->GetInputShape(0)->GetOriginShape().GetShapeSize();

    uint32_t blockDim = MAX_BLOCK_DIM;
    while (blockDim > 1 && (totalLength % (blockDim * 8) != 0)) {
        blockDim /= 2;
    }
    if (blockDim == 0) {
        blockDim = 1;
    }

    uint32_t blockLength = totalLength / blockDim;

    context->SetBlockDim(blockDim);
    tiling.set_totalLength(totalLength);
    tiling.set_blockLength(blockLength);
    tiling.set_tileLength(TILE_LENGTH);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 0;
    return ge::GRAPH_SUCCESS;
}
}


namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* x_shape = context->GetInputShape(0);
    gert::Shape* y_shape = context->GetOutputShape(0);
    *y_shape = *x_shape;
    return GRAPH_SUCCESS;
}
static ge::graphStatus InferDataType(gert::InferDataTypeContext *context)
{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}
}

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv2dMishMish {
public:
    __aicore__ inline KernelConv2dMishMish() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                uint32_t blockLength,
                                uint32_t tileLength)
    {
        this->blockLength = blockLength;
        this->tileLength = tileLength;

        uint32_t coreIdx = AscendC::GetBlockIdx();
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockLength * coreIdx, blockLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + blockLength * coreIdx, blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(tBuf1, tileLength * sizeof(float));
        pipe.InitBuffer(tBuf2, tileLength * sizeof(float));
        pipe.InitBuffer(tBuf3, tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t base = 0;
        while (base < this->blockLength) {
            uint32_t curLen = this->tileLength;
            if (base + curLen > this->blockLength) {
                curLen = this->blockLength - base;
            }
            CopyIn(base, curLen);
            Compute(curLen);
            CopyOut(base, curLen);
            base += curLen;
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t base, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[base], len);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> t1 = tBuf1.Get<float>();
        AscendC::LocalTensor<float> t2 = tBuf2.Get<float>();
        AscendC::LocalTensor<float> t3 = tBuf3.Get<float>();

        // First Mish: y = x * (u^2 + 2u) / (u^2 + 2u + 2), u = exp(x)
        AscendC::Exp(t1, xLocal, len);
        AscendC::Adds(t2, t1, (float)2.0f, len);
        AscendC::Mul(t1, t1, t2, len);
        AscendC::Adds(t2, t1, (float)2.0f, len);
        AscendC::Mul(t1, xLocal, t1, len);
        AscendC::Div(t3, t1, t2, len);

        // Second Mish: z = t3 * (u'^2 + 2u') / (u'^2 + 2u' + 2), u' = exp(t3)
        AscendC::Exp(t1, t3, len);
        AscendC::Adds(t2, t1, (float)2.0f, len);
        AscendC::Mul(t1, t1, t2, len);
        AscendC::Adds(t2, t1, (float)2.0f, len);
        AscendC::Mul(t1, t3, t1, len);
        AscendC::Div(zLocal, t1, t2, len);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t base, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[base], zLocal, len);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tBuf1;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tBuf2;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tBuf3;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t blockLength;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv2d_mish_mish_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dMishMish op;
    op.Init(x, z, tiling_data.blockLength, tiling_data.tileLength);
    op.Process();
}
