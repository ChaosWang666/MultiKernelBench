kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

constexpr int32_t BUFFER_NUM = 2;

using A_T = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float>;
using B_T = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float>;
using C_T = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float>;
using BIAS_T = AscendC::MatmulType<AscendC::TPosition::GM, CubeFormat::ND, float>;

extern "C" __global__ __aicore__ void gemm_multiply_leakyrelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);

    if (ASCEND_IS_AIC) {
        AscendC::TPipe pipe;
        AscendC::Matmul<A_T, B_T, C_T, BIAS_T> mm;
        REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm, &tilingData.cubeTiling);

        AscendC::GlobalTensor<float> aGm;
        AscendC::GlobalTensor<float> bGm;
        AscendC::GlobalTensor<float> biasGm;
        AscendC::GlobalTensor<float> cGm;
        aGm.SetGlobalBuffer((__gm__ float*)x);
        bGm.SetGlobalBuffer((__gm__ float*)weight);
        biasGm.SetGlobalBuffer((__gm__ float*)bias);
        cGm.SetGlobalBuffer((__gm__ float*)y);

        mm.SetTensorA(aGm);
        mm.SetTensorB(bGm);
        mm.SetBias(biasGm);
        mm.IterateAll(cGm);
        mm.End();

        AscendC::CrossCoreSetFlag<0x2, PIPE_FIX>(0x8);
    }

    if (ASCEND_IS_AIV) {
        AscendC::CrossCoreWaitFlag(0x8);

        const uint32_t totalLen = tilingData.totalLength;
        const uint32_t tileLen = tilingData.tileLength;
        const float multiplier = tilingData.multiplier;
        const float negSlope = tilingData.negSlope;

        const uint32_t blockNum = AscendC::GetBlockNum();
        const uint32_t blockIdx = AscendC::GetBlockIdx();
        const uint32_t lenPerCore = totalLen / blockNum;
        const uint32_t coreOffset = blockIdx * lenPerCore;
        const uint32_t numTiles = lenPerCore / tileLen;

        AscendC::GlobalTensor<float> yGm;
        yGm.SetGlobalBuffer((__gm__ float*)y);

        AscendC::TPipe vPipe;
        AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
        AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
        vPipe.InitBuffer(inQueue, BUFFER_NUM, tileLen * sizeof(float));
        vPipe.InitBuffer(outQueue, BUFFER_NUM, tileLen * sizeof(float));

        for (uint32_t i = 0; i < numTiles; i++) {
            uint32_t offset = coreOffset + i * tileLen;

            AscendC::LocalTensor<float> inLocal = inQueue.AllocTensor<float>();
            AscendC::DataCopy(inLocal, yGm[offset], tileLen);
            inQueue.EnQue(inLocal);

            AscendC::LocalTensor<float> inLocal2 = inQueue.DeQue<float>();
            AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();

            AscendC::Muls(outLocal, inLocal2, multiplier, tileLen);
            AscendC::LeakyRelu(outLocal, outLocal, negSlope, tileLen);

            outQueue.EnQue<float>(outLocal);
            inQueue.FreeTensor(inLocal2);

            AscendC::LocalTensor<float> outLocal2 = outQueue.DeQue<float>();
            AscendC::DataCopy(yGm[offset], outLocal2, tileLen);
            outQueue.FreeTensor(outLocal2);
        }
    }
}
