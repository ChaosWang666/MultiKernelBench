kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose3dBatchNormSubtract {
public:
    __aicore__ inline KernelConvTranspose3dBatchNormSubtract() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalRows, uint32_t spatialSize,
                                uint32_t rowsPerCore, uint32_t tileLen,
                                uint32_t tilesPerRow, uint32_t tailLen,
                                float invSpatialSize)
    {
        this->totalRows = totalRows;
        this->spatialSize = spatialSize;
        this->tileLen = tileLen;
        this->tilesPerRow = tilesPerRow;
        this->tailLen = tailLen;
        this->invSpatialSize = invSpatialSize;

        uint32_t coreIdx = static_cast<uint32_t>(AscendC::GetBlockIdx());
        uint32_t startRow = coreIdx * rowsPerCore;
        uint32_t endRow = startRow + rowsPerCore;
        if (startRow > totalRows) {
            startRow = totalRows;
        }
        if (endRow > totalRows) {
            endRow = totalRows;
        }
        this->rowStart = startRow;
        this->rowEnd = endRow;

        uint64_t totalElements = static_cast<uint64_t>(totalRows) * static_cast<uint64_t>(spatialSize);
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, totalElements);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, totalElements);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLen * sizeof(DTYPE_Y));
        pipe.InitBuffer(workBuf, this->tileLen * sizeof(float));
        pipe.InitBuffer(reduceBuf, 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t row = this->rowStart; row < this->rowEnd; ++row) {
            float mean = ComputeMean(row);
            SubtractMean(row, mean);
        }
    }

private:
    __aicore__ inline float ComputeMean(uint32_t row)
    {
        float sum = 0.0f;
        uint64_t base = static_cast<uint64_t>(row) * static_cast<uint64_t>(this->spatialSize);
        for (uint32_t t = 0; t < this->tilesPerRow; ++t) {
            uint32_t curLen = (t == this->tilesPerRow - 1) ? this->tailLen : this->tileLen;
            uint64_t offset = base + static_cast<uint64_t>(t) * static_cast<uint64_t>(this->tileLen);
            CopyInPass(offset, curLen);
            sum += ComputePartialSum(curLen);
        }
        return sum * this->invSpatialSize;
    }

    __aicore__ inline void CopyInPass(uint64_t offset, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = static_cast<uint32_t>(curLen * sizeof(DTYPE_X));
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPadExtParams<DTYPE_X> padParams;
        padParams.isPad = true;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;
        AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline float ComputePartialSum(uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduce = reduceBuf.Get<float>();
        AscendC::ReduceSum<float>(reduce, xLocal, work, static_cast<int32_t>(curLen));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float partial = reduce.GetValue(0);
        inQueueX.FreeTensor(xLocal);
        return partial;
    }

    __aicore__ inline void SubtractMean(uint32_t row, float mean)
    {
        uint64_t base = static_cast<uint64_t>(row) * static_cast<uint64_t>(this->spatialSize);
        for (uint32_t t = 0; t < this->tilesPerRow; ++t) {
            uint32_t curLen = (t == this->tilesPerRow - 1) ? this->tailLen : this->tileLen;
            uint64_t offset = base + static_cast<uint64_t>(t) * static_cast<uint64_t>(this->tileLen);
            CopyInPass(offset, curLen);
            ComputeSubtract(curLen, mean);
            CopyOutPass(offset, curLen);
        }
    }

    __aicore__ inline void ComputeSubtract(uint32_t curLen, float mean)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        DTYPE_X negMean = static_cast<DTYPE_X>(-mean);
        AscendC::Adds<DTYPE_Y>(yLocal, xLocal, negMean, static_cast<int32_t>(curLen));
        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOutPass(uint64_t offset, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = static_cast<uint32_t>(curLen * sizeof(DTYPE_Y));
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPad(yGm[offset], yLocal, copyParams);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t totalRows;
    uint32_t spatialSize;
    uint32_t tileLen;
    uint32_t tilesPerRow;
    uint32_t tailLen;
    float invSpatialSize;
    uint32_t rowStart;
    uint32_t rowEnd;
};

extern "C" __global__ __aicore__ void conv_transpose3d_batch_norm_subtract_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dBatchNormSubtract op;
    op.Init(x, y,
            tiling_data.totalRows,
            tiling_data.spatialSize,
            tiling_data.rowsPerCore,
            tiling_data.tileLen,
            tiling_data.tilesPerRow,
            tiling_data.tailLen,
            tiling_data.invSpatialSize);
    op.Process();
}
