kernel_src="""
#include "kernel_operator.h"

constexpr int32_t COLS_PER_GROUP = 8;
constexpr int32_t WORK_BUF_SIZE = 256;

class KernelGemmScaleBatchnorm {
public:
    __aicore__ inline KernelGemmScaleBatchnorm() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR scale,
                                 GM_ADDR bn_weight, GM_ADDR bn_bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures,
                                 uint32_t numSuperCols, float invBatchSize, float eps)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->numSuperCols = numSuperCols;
        this->invBatchSize = invBatchSize;
        this->eps = eps;

        uint32_t coreId = AscendC::GetBlockIdx();
        uint32_t numCores = AscendC::GetBlockNum();

        uint32_t superColsPerCore = (numSuperCols + numCores - 1) / numCores;
        this->startSuperCol = coreId * superColsPerCore;
        uint32_t endTmp = this->startSuperCol + superColsPerCore;
        this->endSuperCol = (endTmp < numSuperCols) ? endTmp : numSuperCols;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float*)weight, outFeatures * inFeatures);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, outFeatures);
        scaleGm.SetGlobalBuffer((__gm__ float*)scale, outFeatures);
        bnWeightGm.SetGlobalBuffer((__gm__ float*)bn_weight, outFeatures);
        bnBiasGm.SetGlobalBuffer((__gm__ float*)bn_bias, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z, batchSize * outFeatures);

        pipe.InitBuffer(wBuf, COLS_PER_GROUP * inFeatures * sizeof(float));
        pipe.InitBuffer(xBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(mulBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(yBuf, COLS_PER_GROUP * batchSize * sizeof(float));
        pipe.InitBuffer(workBuf, WORK_BUF_SIZE * sizeof(float));
        pipe.InitBuffer(transBuf, COLS_PER_GROUP * batchSize * sizeof(float));
        pipe.InitBuffer(reduceBuf, COLS_PER_GROUP * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (startSuperCol >= endSuperCol) {
            return;
        }

        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> mulLocal = mulBuf.Get<float>();
        AscendC::LocalTensor<float> yLocal = yBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> transLocal = transBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        for (uint32_t superJ = startSuperCol; superJ < endSuperCol; superJ++) {
            uint32_t baseCol = superJ * COLS_PER_GROUP;

            AscendC::DataCopy(wLocal, wGm[baseCol * inFeatures], COLS_PER_GROUP * inFeatures);
            pipe_barrier(PIPE_ALL);

            float biasVals[COLS_PER_GROUP];
            float scaleVals[COLS_PER_GROUP];
            float gammaVals[COLS_PER_GROUP];
            float betaVals[COLS_PER_GROUP];
            for (int32_t k = 0; k < COLS_PER_GROUP; k++) {
                biasVals[k] = biasGm.GetValue(baseCol + k);
                scaleVals[k] = scaleGm.GetValue(baseCol + k);
                gammaVals[k] = bnWeightGm.GetValue(baseCol + k);
                betaVals[k] = bnBiasGm.GetValue(baseCol + k);
            }

            for (uint32_t i = 0; i < batchSize; i++) {
                AscendC::DataCopy(xLocal, xGm[i * inFeatures], inFeatures);
                pipe_barrier(PIPE_ALL);

                for (int32_t k = 0; k < COLS_PER_GROUP; k++) {
                    AscendC::Mul(mulLocal, xLocal, wLocal[k * inFeatures], inFeatures);
                    pipe_barrier(PIPE_ALL);
                    AscendC::ReduceSum<float>(reduceLocal, mulLocal, workLocal, inFeatures);
                    pipe_barrier(PIPE_ALL);
                    float dot = reduceLocal.GetValue(0);
                    float val = (dot + biasVals[k]) * scaleVals[k];
                    yLocal.SetValue(k * batchSize + i, val);
                }
            }
            pipe_barrier(PIPE_ALL);

            for (int32_t k = 0; k < COLS_PER_GROUP; k++) {
                AscendC::LocalTensor<float> yCol = yLocal[k * batchSize];

                AscendC::ReduceSum<float>(reduceLocal, yCol, workLocal, batchSize);
                pipe_barrier(PIPE_ALL);
                float meanSum = reduceLocal.GetValue(0);
                float mean = meanSum * invBatchSize;

                AscendC::Adds(mulLocal, yCol, -mean, batchSize);
                pipe_barrier(PIPE_ALL);
                AscendC::Mul(mulLocal, mulLocal, mulLocal, batchSize);
                pipe_barrier(PIPE_ALL);
                AscendC::ReduceSum<float>(reduceLocal, mulLocal, workLocal, batchSize);
                pipe_barrier(PIPE_ALL);
                float varSum = reduceLocal.GetValue(0);
                float var = varSum * invBatchSize;
                float invStd = 1.0f / sqrt(var + eps);
                float gv = gammaVals[k];
                float bv = betaVals[k];

                AscendC::Adds(yCol, yCol, -mean, batchSize);
                pipe_barrier(PIPE_ALL);
                AscendC::Muls(yCol, yCol, invStd * gv, batchSize);
                pipe_barrier(PIPE_ALL);
                AscendC::Adds(yCol, yCol, bv, batchSize);
                pipe_barrier(PIPE_ALL);
            }

            for (uint32_t i = 0; i < batchSize; i++) {
                for (int32_t k = 0; k < COLS_PER_GROUP; k++) {
                    float val = yLocal.GetValue(k * batchSize + i);
                    transLocal.SetValue(i * COLS_PER_GROUP + k, val);
                }
            }
            pipe_barrier(PIPE_ALL);

            AscendC::DataCopyParams dcParams;
            dcParams.blockCount = static_cast<uint16_t>(batchSize);
            dcParams.blockLen = 1;
            dcParams.srcStride = 0;
            dcParams.dstStride = static_cast<uint16_t>((outFeatures - COLS_PER_GROUP) / COLS_PER_GROUP);

            AscendC::DataCopy(zGm[baseCol], transLocal, dcParams);
            pipe_barrier(PIPE_ALL);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mulBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> yBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> transBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> scaleGm;
    AscendC::GlobalTensor<float> bnWeightGm;
    AscendC::GlobalTensor<float> bnBiasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t inFeatures;
    uint32_t outFeatures;
    uint32_t numSuperCols;
    uint32_t startSuperCol;
    uint32_t endSuperCol;
    float invBatchSize;
    float eps;
};

extern "C" __global__ __aicore__ void gemm_scale_batchnorm_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR scale,
    GM_ADDR bn_weight, GM_ADDR bn_bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmScaleBatchnorm op;
    op.Init(x, weight, bias, scale, bn_weight, bn_bias, z,
            tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures,
            tiling_data.numSuperCols, tiling_data.invBatchSize, tiling_data.eps);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, scale_shape, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.scale = nn.Parameter(torch.rand(scale_shape))
        self.bn = nn.BatchNorm1d(out_features, eps=eps, momentum=momentum)

    def forward(self, x):
        return custom_ops_lib.gemm_scale_batchnorm_custom(
            x, self.gemm.weight, self.gemm.bias, self.scale,
            self.bn.weight, self.bn.bias
        )
