kernel_src="""
#include "kernel_operator.h"

class KernelGNLeakyRelu {
public:
    __aicore__ inline KernelGNLeakyRelu() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                 uint32_t batchSize, uint32_t hiddenSize, uint32_t numGroups,
                                 uint32_t groupSize, uint32_t rowsPerBlock,
                                 float invGroupSize, float epsVal, float negSlopeVal)
    {
        this->batchSize = batchSize;
        this->hiddenSize = hiddenSize;
        this->numGroups = numGroups;
        this->groupSize = groupSize;
        this->rowsPerBlock = rowsPerBlock;
        this->invGroupSize = invGroupSize;
        this->epsVal = epsVal;
        this->negSlopeVal = negSlopeVal;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowOffset = blockIdx * this->rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ float*)x + rowOffset * hiddenSize, this->rowsPerBlock * hiddenSize);
        yGm.SetGlobalBuffer((__gm__ float*)y + rowOffset * hiddenSize, this->rowsPerBlock * hiddenSize);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, hiddenSize);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, hiddenSize);

        pipe.InitBuffer(inQueueX, 1, hiddenSize * sizeof(float));
        pipe.InitBuffer(inQueueGamma, 1, hiddenSize * sizeof(float));
        pipe.InitBuffer(inQueueBeta, 1, hiddenSize * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, hiddenSize * sizeof(float));
        pipe.InitBuffer(diffBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(workBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.AllocTensor<float>();
        AscendC::DataCopy(gammaLocal, gammaGm, hiddenSize);
        inQueueGamma.EnQue(gammaLocal);
        gammaLocal = inQueueGamma.DeQue<float>();

        AscendC::LocalTensor<float> betaLocal = inQueueBeta.AllocTensor<float>();
        AscendC::DataCopy(betaLocal, betaGm, hiddenSize);
        inQueueBeta.EnQue(betaLocal);
        betaLocal = inQueueBeta.DeQue<float>();

        for (uint32_t row = 0; row < rowsPerBlock; row++) {
            ProcessRow(row, gammaLocal, betaLocal);
        }

        inQueueGamma.FreeTensor(gammaLocal);
        inQueueBeta.FreeTensor(betaLocal);
    }

private:
    __aicore__ inline void ProcessRow(uint32_t row,
                                       AscendC::LocalTensor<float>& gammaLocal,
                                       AscendC::LocalTensor<float>& betaLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * hiddenSize], hiddenSize);
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> diffLocal = diffBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();

        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t off = g * groupSize;

            AscendC::ReduceSum<float>(workLocal, xLocal[off], workLocal[32], groupSize);
            AscendC::PipeBarrier<PIPE_V>();
            float meanVal = workLocal.GetValue(0) * invGroupSize;

            AscendC::Adds(diffLocal[off], xLocal[off], -meanVal, groupSize);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(yLocal[off], diffLocal[off], diffLocal[off], groupSize);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum<float>(workLocal, yLocal[off], workLocal[32], groupSize);
            AscendC::PipeBarrier<PIPE_V>();
            float varVal = workLocal.GetValue(0) * invGroupSize;

            float invStd = 1.0f / sqrt(varVal + epsVal);

            AscendC::Muls(diffLocal[off], diffLocal[off], invStd, groupSize);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(diffLocal[off], diffLocal[off], gammaLocal[off], groupSize);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Add(diffLocal[off], diffLocal[off], betaLocal[off], groupSize);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::LeakyRelu(yLocal[off], diffLocal[off], negSlopeVal, groupSize);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Muls(yLocal[off], yLocal[off], 2.0f, groupSize);
            AscendC::PipeBarrier<PIPE_V>();
        }

        outQueueY.EnQue(yLocal);
        inQueueX.FreeTensor(xLocal);

        yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[row * hiddenSize], yLocal, hiddenSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueGamma;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueBeta;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> diffBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;

    uint32_t batchSize;
    uint32_t hiddenSize;
    uint32_t numGroups;
    uint32_t groupSize;
    uint32_t rowsPerBlock;
    float invGroupSize;
    float epsVal;
    float negSlopeVal;
};

extern "C" __global__ __aicore__ void matmul_group_norm_leaky_relu_sum_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGNLeakyRelu op;
    op.Init(x, gamma, beta, y,
            tiling_data.batchSize, tiling_data.hiddenSize, tiling_data.numGroups,
            tiling_data.groupSize, tiling_data.rowsPerBlock,
            tiling_data.invGroupSize, tiling_data.epsVal, tiling_data.negSlopeVal);
    op.Process();
}

import torch
import torch.nn as nn

import custom_ops_lib


class ModelNew(torch.nn.Module):
    def __init__(self, input_size, hidden_size, num_groups, eps=1e-5, negative_slope=0.01):

        self.gn = nn.GroupNorm(num_groups=num_groups, num_channels=hidden_size, eps=eps)
