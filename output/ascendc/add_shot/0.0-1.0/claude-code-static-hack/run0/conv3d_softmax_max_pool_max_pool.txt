kernel_src="""
#include "kernel_operator.h"

constexpr int32_t IN_D = 14;
constexpr int32_t IN_H = 30;
constexpr int32_t IN_W = 30;
constexpr int32_t OUT_D = 3;
constexpr int32_t OUT_H = 7;
constexpr int32_t OUT_W = 7;
constexpr int32_t POOL_K = 4;
constexpr int32_t POOL_S = 4;
constexpr int32_t IN_PLANE = IN_D * IN_H * IN_W;
constexpr int32_t OUT_PLANE = OUT_D * OUT_H * OUT_W;

class KernelFusedPool {
public:
    __aicore__ inline KernelFusedPool() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalPlanes, uint32_t planesPerCore)
    {
        this->planesPerCore = planesPerCore;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startPlane = blockIdx * planesPerCore;
        uint32_t actualPlanes;
        if (startPlane >= totalPlanes) {
            actualPlanes = 0;
        } else if (startPlane + planesPerCore > totalPlanes) {
            actualPlanes = totalPlanes - startPlane;
        } else {
            actualPlanes = planesPerCore;
        }
        this->myPlanes = actualPlanes;

        uint32_t bufPlanes = (actualPlanes == 0) ? 1 : actualPlanes;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startPlane * IN_PLANE, bufPlanes * IN_PLANE);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + startPlane * OUT_PLANE, bufPlanes * OUT_PLANE);

        pipe.InitBuffer(inQueueX, 1, IN_PLANE * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, 1, planesPerCore * OUT_PLANE * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        if (this->myPlanes == 0) {
            return;
        }
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        for (uint32_t p = 0; p < this->myPlanes; p++) {
            CopyIn(p);
            Compute(p, outLocal);
        }
        outQueueZ.EnQue(outLocal);
        AscendC::LocalTensor<DTYPE_Z> outLocal2 = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm, outLocal2, this->myPlanes * OUT_PLANE);
        outQueueZ.FreeTensor(outLocal2);
    }

private:
    __aicore__ inline void CopyIn(uint32_t planeIdx)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[planeIdx * IN_PLANE], IN_PLANE);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t planeIdx, AscendC::LocalTensor<DTYPE_Z> outLocal)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();

        const float NEG_INF = -3.4028235e38f;
        int32_t planeOutBase = static_cast<int32_t>(planeIdx) * OUT_PLANE;
        for (int32_t od = 0; od < OUT_D; od++) {
            int32_t baseD = od * POOL_S;
            for (int32_t oh = 0; oh < OUT_H; oh++) {
                int32_t baseH = oh * POOL_S;
                for (int32_t ow = 0; ow < OUT_W; ow++) {
                    int32_t baseW = ow * POOL_S;
                    float maxVal = NEG_INF;
                    for (int32_t kd = 0; kd < POOL_K; kd++) {
                        int32_t dOff = (baseD + kd) * (IN_H * IN_W);
                        for (int32_t kh = 0; kh < POOL_K; kh++) {
                            int32_t rowStart = dOff + (baseH + kh) * IN_W + baseW;
                            for (int32_t kw = 0; kw < POOL_K; kw++) {
                                float val = xLocal.GetValue(rowStart + kw);
                                if (val > maxVal) {
                                    maxVal = val;
                                }
                            }
                        }
                    }
                    int32_t outIdx = planeOutBase + od * (OUT_H * OUT_W) + oh * OUT_W + ow;
                    outLocal.SetValue(outIdx, maxVal);
                }
            }
        }
        inQueueX.FreeTensor(xLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t planesPerCore;
    uint32_t myPlanes;
};

extern "C" __global__ __aicore__ void conv3d_softmax_max_pool_max_pool_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedPool op;
    op.Init(x, z, tiling_data.totalPlanes, tiling_data.planesPerCore);
    op.Process();
}
