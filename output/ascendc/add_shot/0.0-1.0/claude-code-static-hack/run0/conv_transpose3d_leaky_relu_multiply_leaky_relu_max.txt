kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t MAX_CHANNELS = 256;

class KernelFusedActMul {
public:
    __aicore__ inline KernelFusedActMul() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mul, GM_ADDR y,
                                 uint32_t batchSize, uint32_t channelSize,
                                 uint32_t spatialSize, uint32_t tileLength,
                                 uint32_t alignedChannelSize)
    {
        this->batchSize = batchSize;
        this->channelSize = channelSize;
        this->spatialSize = spatialSize;
        this->tileLength = tileLength;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t totalSlices = batchSize * channelSize;
        this->slicesPerBlock = totalSlices / blockNum;
        this->sliceStart = this->slicesPerBlock * blockIdx;
        this->tilesPerSlice = spatialSize / tileLength;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        yGm.SetGlobalBuffer((__gm__ float*)y);
        mulGm.SetGlobalBuffer((__gm__ float*)mul, alignedChannelSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(mulBuf, MAX_CHANNELS * sizeof(float));

        AscendC::LocalTensor<float> mulLocal = mulBuf.Get<float>();
        AscendC::DataCopy(mulLocal, mulGm, alignedChannelSize);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> mulLocal = mulBuf.Get<float>();
        for (uint32_t s = 0; s < this->slicesPerBlock; ++s) {
            uint32_t globalSliceIdx = this->sliceStart + s;
            uint32_t channelIdx = globalSliceIdx % this->channelSize;
            float mulVal = mulLocal.GetValue(channelIdx);
            uint64_t baseOffset = (uint64_t)globalSliceIdx * (uint64_t)this->spatialSize;
            for (uint32_t t = 0; t < this->tilesPerSlice; ++t) {
                uint64_t tileOffset = baseOffset + (uint64_t)t * (uint64_t)this->tileLength;
                CopyIn(tileOffset);
                Compute(mulVal);
                CopyOut(tileOffset);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint64_t offset)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float mulVal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        const float negSlope = 0.2f;
        AscendC::LeakyRelu(yLocal, xLocal, negSlope, this->tileLength);
        AscendC::Muls(yLocal, yLocal, mulVal, this->tileLength);
        AscendC::LeakyRelu(yLocal, yLocal, negSlope, this->tileLength);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint64_t offset)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[offset], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mulBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> mulGm;
    uint32_t batchSize;
    uint32_t channelSize;
    uint32_t spatialSize;
    uint32_t tileLength;
    uint32_t slicesPerBlock;
    uint32_t sliceStart;
    uint32_t tilesPerSlice;
};

extern "C" __global__ __aicore__ void conv_transpose3d_leaky_relu_multiply_leaky_relu_max_custom(
    GM_ADDR x, GM_ADDR multiplier, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedActMul op;
    op.Init(x, multiplier, y,
            tiling_data.batchSize, tiling_data.channelSize,
            tiling_data.spatialSize, tiling_data.tileLength,
            tiling_data.alignedChannelSize);
    op.Process();
}
