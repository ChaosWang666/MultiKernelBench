kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMeanReduce {
public:
    __aicore__ inline KernelMeanReduce() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t bcPerBlock, uint32_t spatialSize,
                                 uint32_t tileLength, uint32_t numTiles,
                                 float scale)
    {
        this->bcPerBlock = bcPerBlock;
        this->spatialSize = spatialSize;
        this->tileLength = tileLength;
        this->numTiles = numTiles;
        this->scale = scale;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t startBC = static_cast<uint64_t>(blockIdx) * static_cast<uint64_t>(bcPerBlock);

        xGm.SetGlobalBuffer((__gm__ float*)x + startBC * static_cast<uint64_t>(spatialSize),
                            static_cast<uint64_t>(bcPerBlock) * static_cast<uint64_t>(spatialSize));
        yGm.SetGlobalBuffer((__gm__ float*)y + startBC, bcPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, bcPerBlock * sizeof(float));
        pipe.InitBuffer(workBuf, tileLength * sizeof(float));
        pipe.InitBuffer(reduceTmpBuf, 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> outLocal = outQueueY.AllocTensor<float>();

        for (uint32_t bc = 0; bc < bcPerBlock; bc++) {
            float partialSum = 0.0f;
            for (uint32_t t = 0; t < numTiles; t++) {
                CopyIn(bc, t);
                partialSum += ComputeTileSum();
            }
            outLocal.SetValue(bc, partialSum * scale);
        }

        outQueueY.EnQue(outLocal);

        AscendC::LocalTensor<float> outLocal2 = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm, outLocal2, bcPerBlock);
        outQueueY.FreeTensor(outLocal2);
    }

private:
    __aicore__ inline void CopyIn(uint32_t bc, uint32_t t)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        uint64_t offset = static_cast<uint64_t>(bc) * static_cast<uint64_t>(spatialSize)
                        + static_cast<uint64_t>(t) * static_cast<uint64_t>(tileLength);
        AscendC::DataCopy(xLocal, xGm[offset], tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline float ComputeTileSum()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceTmpBuf.Get<float>();

        AscendC::ReduceSum<float>(reduceTmp, xLocal, workLocal, static_cast<int32_t>(tileLength));
        float val = reduceTmp.GetValue(0);

        inQueueX.FreeTensor(xLocal);
        return val;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceTmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t bcPerBlock;
    uint32_t spatialSize;
    uint32_t tileLength;
    uint32_t numTiles;
    float scale;
};

extern "C" __global__ __aicore__ void conv_transpose2d_multiply_global_avg_pool_global_avg_pool_mean_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{

    KernelMeanReduce op;
    op.Init(x, y, tiling_data.bcPerBlock, tiling_data.spatialSize,
            tiling_data.tileLength, tiling_data.numTiles, tiling_data.scale);
    op.Process();
}
