kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedAct {
public:
    __aicore__ inline KernelFusedAct() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR add_value, GM_ADDR y,
                                 uint32_t M, uint32_t N, uint32_t rowsPerCore,
                                 uint32_t tileLen, uint32_t tilesPerRow)
    {
        this->N = N;
        this->tileLen = tileLen;
        this->tilesPerRow = tilesPerRow;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerCore;
        uint32_t endRow = startRow + rowsPerCore;
        if (endRow > M) {
            endRow = M;
        }
        this->myRows = (endRow > startRow) ? (endRow - startRow) : 0;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startRow * N, this->myRows * N);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + startRow * N, this->myRows * N);
        addValueGm.SetGlobalBuffer((__gm__ DTYPE_ADD_VALUE *)add_value, N);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLen * sizeof(DTYPE_Y));
        pipe.InitBuffer(addBuf, N * sizeof(DTYPE_ADD_VALUE));
    }

    __aicore__ inline void Process()
    {
        if (myRows == 0) {
            return;
        }

        AscendC::LocalTensor<DTYPE_ADD_VALUE> addLocal = addBuf.Get<DTYPE_ADD_VALUE>();
        AscendC::DataCopy(addLocal, addValueGm, N);

        for (uint32_t r = 0; r < myRows; r++) {
            for (uint32_t t = 0; t < tilesPerRow; t++) {
                uint32_t curOffset = r * N + t * tileLen;
                uint32_t addOffset = t * tileLen;
                CopyIn(curOffset);
                Compute(addLocal, addOffset);
                CopyOut(curOffset);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[offset], tileLen);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(AscendC::LocalTensor<DTYPE_ADD_VALUE>& addLocal, uint32_t addOffset)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        AscendC::Add(yLocal, xLocal, addLocal[addOffset], tileLen);

        AscendC::Sigmoid(xLocal, yLocal, tileLen);
        AscendC::Mul(yLocal, xLocal, yLocal, tileLen);

        AscendC::Tanh(xLocal, yLocal, tileLen);

        AscendC::Gelu(yLocal, xLocal, tileLen);

        AscendC::Mins(xLocal, yLocal, (float)1.0f, tileLen);
        AscendC::Maxs(yLocal, xLocal, (float)-1.0f, tileLen);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[offset], yLocal, tileLen);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> addBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_ADD_VALUE> addValueGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t N;
    uint32_t myRows;
    uint32_t tileLen;
    uint32_t tilesPerRow;
};

extern "C" __global__ __aicore__ void matmul_add_swish_tanh_gelu_hardtanh_custom(
    GM_ADDR x, GM_ADDR add_value, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedAct op;
    op.Init(x, add_value, y, tiling_data.M, tiling_data.N, tiling_data.rowsPerCore,
            tiling_data.tileLen, tiling_data.tilesPerRow);
    op.Process();
}
