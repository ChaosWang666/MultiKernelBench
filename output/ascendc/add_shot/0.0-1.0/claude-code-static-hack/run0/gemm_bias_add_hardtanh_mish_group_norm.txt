kernel_src="""
#include "kernel_operator.h"

constexpr int32_t MAX_FEATURES = 8192;
constexpr int32_t WORK_SIZE = 1024;
constexpr int32_t REDUCE_SIZE = 32;

class KernelGemmFused {
public:
    __aicore__ inline KernelGemmFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR linear_bias,
                                GM_ADDR custom_bias, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z,
                                uint32_t batchSizeArg, uint32_t inFeaturesArg, uint32_t outFeaturesArg,
                                uint32_t numGroupsArg, uint32_t groupSizeArg, uint32_t rowsPerBlockArg,
                                float invGroupSizeArg, float epsilonArg)
    {
        this->batchSize = batchSizeArg;
        this->inFeatures = inFeaturesArg;
        this->outFeatures = outFeaturesArg;
        this->numGroups = numGroupsArg;
        this->groupSize = groupSizeArg;
        this->rowsPerBlock = rowsPerBlockArg;
        this->invGroupSize = invGroupSizeArg;
        this->epsilon = epsilonArg;

        this->startRow = AscendC::GetBlockIdx() * this->rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        wGm.SetGlobalBuffer((__gm__ float*)weight);
        linBiasGm.SetGlobalBuffer((__gm__ float*)linear_bias);
        cusBiasGm.SetGlobalBuffer((__gm__ float*)custom_bias);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma);
        betaGm.SetGlobalBuffer((__gm__ float*)beta);
        outGm.SetGlobalBuffer((__gm__ float*)z);

        pipe.InitBuffer(xBuf, MAX_FEATURES * sizeof(float));
        pipe.InitBuffer(wBuf, MAX_FEATURES * sizeof(float));
        pipe.InitBuffer(preActBuf, MAX_FEATURES * sizeof(float));
        pipe.InitBuffer(mulBuf, MAX_FEATURES * sizeof(float));
        pipe.InitBuffer(workBuf, WORK_SIZE * sizeof(float));
        pipe.InitBuffer(reduceBuf, REDUCE_SIZE * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> preAct = preActBuf.Get<float>();
        AscendC::LocalTensor<float> mulLocal = mulBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        for (uint32_t r = 0; r < this->rowsPerBlock; r++) {
            uint32_t globalRow = this->startRow + r;
            ProcessRow(globalRow, xLocal, wLocal, preAct, mulLocal, workLocal, reduceLocal);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t globalRow,
                                       AscendC::LocalTensor<float>& xLocal,
                                       AscendC::LocalTensor<float>& wLocal,
                                       AscendC::LocalTensor<float>& preAct,
                                       AscendC::LocalTensor<float>& mulLocal,
                                       AscendC::LocalTensor<float>& workLocal,
                                       AscendC::LocalTensor<float>& reduceLocal)
    {
        uint32_t inF = this->inFeatures;
        uint32_t outF = this->outFeatures;
        uint32_t gSize = this->groupSize;
        uint32_t nGroups = this->numGroups;
        float invGS = this->invGroupSize;
        float eps = this->epsilon;

        AscendC::DataCopy(xLocal, xGm[globalRow * inF], inF);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

        for (uint32_t j = 0; j < outF; j++) {
            AscendC::DataCopy(wLocal, wGm[j * inF], inF);
            AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID1);
            AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID1);

            AscendC::Mul(mulLocal, xLocal, wLocal, inF);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum<float>(reduceLocal, mulLocal, workLocal, inF);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
            float val = reduceLocal.GetValue(0);
            preAct.SetValue(j, val);
            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID3);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID3);
        }

        AscendC::DataCopy(wLocal, linBiasGm, outF);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::Add(preAct, preAct, wLocal, outF);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::DataCopy(wLocal, cusBiasGm, outF);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::Add(preAct, preAct, wLocal, outF);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Duplicate(wLocal, 1.0f, outF);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Min(preAct, preAct, wLocal, outF);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Duplicate(wLocal, -1.0f, outF);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Max(preAct, preAct, wLocal, outF);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Exp(mulLocal, preAct, outF);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Adds(mulLocal, mulLocal, 1.0f, outF);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Ln(mulLocal, mulLocal, outF);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Muls(wLocal, mulLocal, 2.0f, outF);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Exp(wLocal, wLocal, outF);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Adds(wLocal, wLocal, 1.0f, outF);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Reciprocal(wLocal, wLocal, outF);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Muls(wLocal, wLocal, -2.0f, outF);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Adds(wLocal, wLocal, 1.0f, outF);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(preAct, preAct, wLocal, outF);
        AscendC::PipeBarrier<PIPE_V>();

        for (uint32_t g = 0; g < nGroups; g++) {
            uint32_t offset = g * gSize;

            AscendC::DataCopy(wLocal, gammaGm[offset], gSize);
            AscendC::DataCopy(mulLocal, betaGm[offset], gSize);
            AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

            AscendC::ReduceSum<float>(reduceLocal, preAct[offset], workLocal, gSize);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Muls(reduceLocal, reduceLocal, invGS, 1);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
            float mean = reduceLocal.GetValue(0);
            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID3);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID3);

            AscendC::Adds(xLocal, preAct[offset], -mean, gSize);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Mul(workLocal[64], xLocal, xLocal, gSize);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum<float>(reduceLocal, workLocal[64], workLocal[256], gSize);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Muls(reduceLocal, reduceLocal, invGS, 1);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Adds(reduceLocal, reduceLocal, eps, 1);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Rsqrt(reduceLocal, reduceLocal, 1);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
            float invStd = reduceLocal.GetValue(0);
            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID3);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID3);

            AscendC::Muls(xLocal, xLocal, invStd, gSize);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Mul(xLocal, xLocal, wLocal, gSize);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Add(preAct[offset], xLocal, mulLocal, gSize);
            AscendC::PipeBarrier<PIPE_V>();
        }

        AscendC::SetFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);
        AscendC::DataCopy(outGm[globalRow * outF], preAct, outF);
        AscendC::SetFlag<AscendC::HardEvent::MTE3_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE3_V>(EVENT_ID0);

        AscendC::DataCopy(xLocal, xGm[globalRow * inF], inF);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> preActBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mulBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> linBiasGm;
    AscendC::GlobalTensor<float> cusBiasGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    AscendC::GlobalTensor<float> outGm;
    uint32_t batchSize;
    uint32_t inFeatures;
    uint32_t outFeatures;
    uint32_t numGroups;
    uint32_t groupSize;
    uint32_t rowsPerBlock;
    uint32_t startRow;
    float invGroupSize;
    float epsilon;
};

extern "C" __global__ __aicore__ void gemm_bias_add_hardtanh_mish_group_norm_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR linear_bias, GM_ADDR custom_bias,
    GM_ADDR gamma, GM_ADDR beta, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)

            tiling_data.numGroups, tiling_data.groupSize, tiling_data.rowsPerBlock,
            tiling_data.invGroupSize, tiling_data.epsilon);
    op.Process();
}
