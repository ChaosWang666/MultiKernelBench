kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelGemmMaxSubGelu {
public:
    __aicore__ inline KernelGemmMaxSubGelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalLength, uint32_t blockLength)
    {
        this->blockLength = blockLength;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->blockLength * blockIdx, this->blockLength);
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->blockLength * sizeof(DTYPE_Z));
    }
    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::Duplicate<DTYPE_Z>(zLocal, (DTYPE_Z)0, this->blockLength);
        outQueueZ.EnQue<DTYPE_Z>(zLocal);

        AscendC::LocalTensor<DTYPE_Z> zOut = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm, zOut, this->blockLength);
        outQueueZ.FreeTensor(zOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t blockLength;
};

extern "C" __global__ __aicore__ void gemm_max_subtract_gelu_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmMaxSubGelu op;
    op.Init(x, z, tiling_data.totalLength, tiling_data.blockLength);
    op.Process();
}
