kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelSoftmaxTanhFused {
public:
    __aicore__ inline KernelSoftmaxTanhFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                uint32_t batchSize, uint32_t channels,
                                uint32_t spatialSize, uint32_t tilesPerBlock,
                                uint32_t tileSpatial)
    {
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->tilesPerBlock = tilesPerBlock;
        this->tileSpatial = tileSpatial;
        this->tilesPerBatch = spatialSize / tileSpatial;
        this->workUnitOffset = AscendC::GetBlockIdx() * tilesPerBlock;

        uint64_t totalSize = static_cast<uint64_t>(batchSize) *
                             static_cast<uint64_t>(channels) *
                             static_cast<uint64_t>(spatialSize);
        xGm.SetGlobalBuffer((__gm__ float*)x, totalSize);
        zGm.SetGlobalBuffer((__gm__ float*)z, totalSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, channels * tileSpatial * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, channels * tileSpatial * sizeof(float));
        pipe.InitBuffer(maxBuf, tileSpatial * sizeof(float));
        pipe.InitBuffer(sumBuf, tileSpatial * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();

        for (uint32_t i = 0; i < tilesPerBlock; i++) {
            uint32_t workUnit = workUnitOffset + i;
            uint32_t b = workUnit / tilesPerBatch;
            uint32_t hwTileIdx = workUnit % tilesPerBatch;
            uint32_t hwStart = hwTileIdx * tileSpatial;
            uint64_t baseOffset = static_cast<uint64_t>(b) *
                                  static_cast<uint64_t>(channels) *
                                  static_cast<uint64_t>(spatialSize) +
                                  static_cast<uint64_t>(hwStart);

            CopyIn(baseOffset);
            Compute(maxLocal, sumLocal);
            CopyOut(baseOffset);
        }
    }

private:
    __aicore__ inline void CopyIn(uint64_t baseOffset)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        for (uint32_t c = 0; c < channels; c++) {
            AscendC::DataCopy(xLocal[c * tileSpatial],
                              xGm[baseOffset + static_cast<uint64_t>(c) *
                                  static_cast<uint64_t>(spatialSize)],
                              tileSpatial);
        }
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(AscendC::LocalTensor<float>& maxLocal,
                                   AscendC::LocalTensor<float>& sumLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        AscendC::Duplicate(maxLocal, -1.0e30f, tileSpatial);
        for (uint32_t c = 0; c < channels; c++) {
            AscendC::Max(maxLocal, maxLocal, xLocal[c * tileSpatial], tileSpatial);
        }

        for (uint32_t c = 0; c < channels; c++) {
            AscendC::Sub(zLocal[c * tileSpatial], xLocal[c * tileSpatial],
                         maxLocal, tileSpatial);
        }
        inQueueX.FreeTensor(xLocal);

        for (uint32_t c = 0; c < channels; c++) {
            AscendC::Exp(zLocal[c * tileSpatial], zLocal[c * tileSpatial], tileSpatial);
        }

        AscendC::Duplicate(sumLocal, 0.0f, tileSpatial);
        for (uint32_t c = 0; c < channels; c++) {
            AscendC::Add(sumLocal, sumLocal, zLocal[c * tileSpatial], tileSpatial);
        }

        AscendC::Reciprocal(sumLocal, sumLocal, tileSpatial);

        for (uint32_t c = 0; c < channels; c++) {
            AscendC::Mul(zLocal[c * tileSpatial], zLocal[c * tileSpatial],
                         sumLocal, tileSpatial);
        }

        for (uint32_t c = 0; c < channels; c++) {
            AscendC::Tanh(zLocal[c * tileSpatial], zLocal[c * tileSpatial], tileSpatial);
        }

        outQueueZ.EnQue<float>(zLocal);
    }

    __aicore__ inline void CopyOut(uint64_t baseOffset)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        for (uint32_t c = 0; c < channels; c++) {
            AscendC::DataCopy(zGm[baseOffset + static_cast<uint64_t>(c) *
                                  static_cast<uint64_t>(spatialSize)],
                              zLocal[c * tileSpatial], tileSpatial);
        }
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t channels;
    uint32_t spatialSize;
    uint32_t tilesPerBlock;
    uint32_t tileSpatial;
    uint32_t tilesPerBatch;
    uint32_t workUnitOffset;
};

extern "C" __global__ __aicore__ void convtranspose3d_mean_add_softmax_tanh_scaling_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelSoftmaxTanhFused op;
    op.Init(x, z,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.spatialSize, tiling_data.tilesPerBlock,
            tiling_data.tileSpatial);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib


class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, scaling_factor):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose3d(
            in_channels, out_channels, kernel_size, stride=stride, padding=padding
        )
        self.bias = torch.nn.Parameter(torch.randn(1, out_channels, 1, 1, 1))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x.mean(dim=2, keepdim=True)
        x = x + self.bias
        x = x.contiguous()
        x = custom_ops_lib.convtranspose3d_mean_add_softmax_tanh_scaling_custom(x)
        x = x * self.scaling_factor
        return x
