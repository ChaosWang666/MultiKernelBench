kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;
using namespace matmul;

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t TILE_LEN = 1024;

class KernelGemmScalingHardTanhGelu {
public:
    __aicore__ inline KernelGemmScalingHardTanhGelu() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                uint32_t M, uint32_t N, uint32_t K,
                                float scalingFactor, float htMin, float htMax)
    {
        this->M_ = M;
        this->N_ = N;
        this->K_ = K;
        this->scalingFactor_ = scalingFactor;
        this->htMin_ = htMin;
        this->htMax_ = htMax;

        xGm_.SetGlobalBuffer((__gm__ float *)x, M * K);
        wGm_.SetGlobalBuffer((__gm__ float *)weight, N * K);
        biasGm_.SetGlobalBuffer((__gm__ float *)bias, N);
        yGm_.SetGlobalBuffer((__gm__ float *)y, M * N);
    }

    __aicore__ inline void ProcessCube(const TCubeTiling &cubeTiling)
    {
        if ASCEND_IS_AIC {
            REGIST_MATMUL_OBJ(&pipe_, GetSysWorkSpacePtr(), mm_, &cubeTiling);
            mm_.SetTensorA(xGm_);
            mm_.SetTensorB(wGm_, true);
            mm_.SetBias(biasGm_);
            while (mm_.Iterate()) {
                mm_.GetTensorC(yGm_);
            }
            mm_.End();
        }
    }

    __aicore__ inline void ProcessVector()
    {
        if ASCEND_IS_AIV {
            uint32_t blockIdx = GetBlockIdx();
            uint32_t blockNum = GetBlockNum();
            uint32_t totalLen = M_ * N_;
            uint32_t totalTiles = totalLen / TILE_LEN;
            uint32_t tilesPerBlock = (totalTiles + blockNum - 1) / blockNum;
            uint32_t startTile = blockIdx * tilesPerBlock;
            uint32_t endTile = startTile + tilesPerBlock;
            if (endTile > totalTiles) {
                endTile = totalTiles;
            }
            if (startTile >= totalTiles) {
                return;
            }

            pipe_.InitBuffer(inQueue_, BUFFER_NUM, TILE_LEN * sizeof(float));
            pipe_.InitBuffer(outQueue_, BUFFER_NUM, TILE_LEN * sizeof(float));

            for (uint32_t t = startTile; t < endTile; t++) {
                uint32_t offset = t * TILE_LEN;
                CopyIn(offset);
                Compute();
                CopyOut(offset);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset)
    {
        LocalTensor<float> inLocal = inQueue_.AllocTensor<float>();
        DataCopy(inLocal, yGm_[offset], TILE_LEN);
        inQueue_.EnQue(inLocal);
    }

    __aicore__ inline void Compute()
    {
        LocalTensor<float> inLocal = inQueue_.DeQue<float>();
        LocalTensor<float> outLocal = outQueue_.AllocTensor<float>();
        Muls(outLocal, inLocal, scalingFactor_, TILE_LEN);
        Maxs(outLocal, outLocal, htMin_, TILE_LEN);
        Mins(outLocal, outLocal, htMax_, TILE_LEN);
        Gelu(outLocal, outLocal, TILE_LEN);
        outQueue_.EnQue<float>(outLocal);
        inQueue_.FreeTensor(inLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset)
    {
        LocalTensor<float> outLocal = outQueue_.DeQue<float>();
        DataCopy(yGm_[offset], outLocal, TILE_LEN);
        outQueue_.FreeTensor(outLocal);
    }

private:
    TPipe pipe_;
    TQue<TPosition::VECIN, BUFFER_NUM> inQueue_;
    TQue<TPosition::VECOUT, BUFFER_NUM> outQueue_;
    Matmul<MatmulType<TPosition::GM, CubeFormat::ND, float>,
           MatmulType<TPosition::GM, CubeFormat::ND, float>,
           MatmulType<TPosition::GM, CubeFormat::ND, float>,
           MatmulType<TPosition::GM, CubeFormat::ND, float>> mm_;

    GlobalTensor<float> xGm_;
    GlobalTensor<float> wGm_;
    GlobalTensor<float> biasGm_;
    GlobalTensor<float> yGm_;
    uint32_t M_;
    uint32_t N_;
    uint32_t K_;
    float scalingFactor_;
    float htMin_;
    float htMax_;
};

extern "C" __global__ __aicore__ void gemm_scaling_hard_tanh_gelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmScalingHardTanhGelu op;
    op.Init(x, weight, bias, y,
            tiling_data.M, tiling_data.N, tiling_data.K,
            tiling_data.scalingFactor, tiling_data.hardtanhMin, tiling_data.hardtanhMax);
    op.ProcessCube(tiling_data.cubeTiling);
    AscendC::SyncAll();
    op.ProcessVector();
}
