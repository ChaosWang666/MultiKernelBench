kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using AscendC::CubeFormat;
using AscendC::MatmulType;
using AscendC::TPosition;

typedef MatmulType<TPosition::GM, CubeFormat::ND, float, false> aType;
typedef MatmulType<TPosition::GM, CubeFormat::ND, float, true>  bType;
typedef MatmulType<TPosition::GM, CubeFormat::ND, float>        cType;
typedef MatmulType<TPosition::GM, CubeFormat::ND, float>        biasType;

constexpr int32_t SYNC_FLAG_AIC_TO_AIV = 0;

extern "C" __global__ __aicore__ void gemm_relu_divide_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);

    uint32_t M = tiling_data.M;
    uint32_t N = tiling_data.N;
    uint32_t K = tiling_data.K;
    uint32_t totalElems = tiling_data.totalElems;
    uint32_t tileLen = tiling_data.tileLen;
    float divisorRecip = tiling_data.divisorRecip;

    if ASCEND_IS_AIC {
        AscendC::TPipe pipe;
        AscendC::Matmul<aType, bType, cType, biasType> mm;
        REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm, &tiling_data.cubeTilingData);

        AscendC::GlobalTensor<float> xGm;
        AscendC::GlobalTensor<float> wGm;
        AscendC::GlobalTensor<float> biasGm;
        AscendC::GlobalTensor<float> zGm;
        xGm.SetGlobalBuffer((__gm__ float*)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float*)weight, N * K);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, N);
        zGm.SetGlobalBuffer((__gm__ float*)z, M * N);

        mm.SetTensorA(xGm);
        mm.SetTensorB(wGm);
        mm.SetBias(biasGm);
        mm.IterateAll(zGm);
        mm.End();

        AscendC::CrossCoreSetFlag<0x2, PIPE_FIX>(SYNC_FLAG_AIC_TO_AIV);
    }

    if ASCEND_IS_AIV {
        AscendC::CrossCoreWaitFlag(SYNC_FLAG_AIC_TO_AIV);

        AscendC::TPipe pipe;
        AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueue;
        AscendC::TQue<AscendC::TPosition::VECOUT, 2> outQueue;
        pipe.InitBuffer(inQueue, 2, tileLen * sizeof(float));
        pipe.InitBuffer(outQueue, 2, tileLen * sizeof(float));

        uint32_t blockNumU = AscendC::GetBlockNum();
        uint32_t blockIdxU = AscendC::GetBlockIdx();
        int32_t  blockNum  = static_cast<int32_t>(blockNumU);
        int32_t  blockIdx  = static_cast<int32_t>(blockIdxU);
        int32_t  total     = static_cast<int32_t>(totalElems);
        int32_t  tile      = static_cast<int32_t>(tileLen);

        int32_t elemsPerCore = (total + blockNum - 1) / blockNum;
        elemsPerCore = ((elemsPerCore + tile - 1) / tile) * tile;

        int32_t startElem = blockIdx * elemsPerCore;
        if (startElem >= total) {
            return;
        }
        int32_t endElem = startElem + elemsPerCore;
        if (endElem > total) {
            endElem = total;
        }
        int32_t myElems = endElem - startElem;
        int32_t numTiles = (myElems + tile - 1) / tile;

        AscendC::GlobalTensor<float> zGm;
        zGm.SetGlobalBuffer((__gm__ float*)z, M * N);

        for (int32_t i = 0; i < numTiles; i++) {
            int32_t offset = startElem + i * tile;
            int32_t curLen = (i == numTiles - 1) ? (myElems - i * tile) : tile;

            AscendC::LocalTensor<float> inLocal = inQueue.AllocTensor<float>();
            AscendC::DataCopy(inLocal, zGm[offset], curLen);
            inQueue.EnQue(inLocal);

            AscendC::LocalTensor<float> inT = inQueue.DeQue<float>();
            AscendC::LocalTensor<float> outT = outQueue.AllocTensor<float>();
            AscendC::Relu(outT, inT, curLen);
            AscendC::Muls(outT, outT, divisorRecip, curLen);
            inQueue.FreeTensor(inT);
            outQueue.EnQue(outT);

            AscendC::LocalTensor<float> outResult = outQueue.DeQue<float>();
            AscendC::DataCopy(zGm[offset], outResult, curLen);
            outQueue.FreeTensor(outResult);
        }
    }
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor gemm_relu_divide_custom_impl_npu(const at::Tensor& x,
                                             const at::Tensor& weight,
                                             const at::Tensor& bias,
                                             double divisor) {
    int64_t M = x.size(0);
    int64_t N = weight.size(0);
    at::Tensor result = at::empty({M, N}, x.options());
    float divisor_f = static_cast<float>(divisor);
    EXEC_NPU_CMD(aclnnGemmReluDivideCustom, x, weight, bias, divisor_f, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("gemm_relu_divide_custom", &gemm_relu_divide_custom_impl_npu);

    m.def("gemm_relu_divide_custom", &gemm_relu_divide_custom_impl_npu,
