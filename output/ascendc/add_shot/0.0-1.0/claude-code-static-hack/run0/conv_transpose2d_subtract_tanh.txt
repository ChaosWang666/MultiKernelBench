kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTransposeSubTanh {
public:
    __aicore__ inline KernelConvTransposeSubTanh() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                 uint32_t totalLength, uint32_t channelSize,
                                 uint32_t numChannels, uint32_t coreChannels,
                                 uint32_t tileLen, uint32_t paddedNumChannels)
    {
        this->channelSize = channelSize;
        this->numChannels = numChannels;
        this->coreChannels = coreChannels;
        this->tileLen = tileLen;
        this->paddedNumChannels = paddedNumChannels;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->coreElemOffset = blockIdx * coreChannels * channelSize;
        this->channelStart = blockIdx * coreChannels;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, totalLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z, totalLength);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, numChannels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, tileLen * sizeof(DTYPE_Z));
        pipe.InitBuffer(biasBuf, paddedNumChannels * sizeof(DTYPE_BIAS));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();
        AscendC::DataCopyExtParams biasParams{1, (uint32_t)(numChannels * sizeof(DTYPE_BIAS)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<DTYPE_BIAS> biasPadParams{false, 0, 0, 0};
        AscendC::DataCopyPad(biasLocal, biasGm, biasParams, biasPadParams);

        event_t eventIdMte2ToS = static_cast<event_t>(AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::MTE2_S));
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(eventIdMte2ToS);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(eventIdMte2ToS);

        for (uint32_t k = 0; k < coreChannels; ++k) {
            uint32_t globalChannelIdx = channelStart + k;
            uint32_t c = globalChannelIdx % numChannels;
            DTYPE_BIAS biasVal = biasLocal.GetValue(c);
            DTYPE_X negBias = -biasVal;

            uint32_t channelGlobalOffset = coreElemOffset + k * channelSize;
            uint32_t fullTiles = channelSize / tileLen;
            uint32_t tail = channelSize - fullTiles * tileLen;

            for (uint32_t i = 0; i < fullTiles; ++i) {
                uint32_t off = channelGlobalOffset + i * tileLen;
                CopyInPad(off, tileLen);
                Compute(tileLen, negBias);
                CopyOutPad(off, tileLen);
            }
            if (tail > 0) {
                uint32_t off = channelGlobalOffset + fullTiles * tileLen;
                uint32_t alignedTail = ((tail + 7) / 8) * 8;
                CopyInPad(off, tail);
                Compute(alignedTail, negBias);
                CopyOutPad(off, tail);
            }
        }
    }

private:
    __aicore__ inline void CopyInPad(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopyExtParams copyParams{1, (uint32_t)(len * sizeof(DTYPE_X)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<DTYPE_X> padParams{false, 0, 0, 0};
        AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(uint32_t len, DTYPE_X negBias)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::Adds(zLocal, xLocal, negBias, len);
        AscendC::Tanh(zLocal, zLocal, len);
        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOutPad(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopyExtParams copyParams{1, (uint32_t)(len * sizeof(DTYPE_Z)), 0, 0, 0};
        AscendC::DataCopyPad(zGm[offset], zLocal, copyParams);
        outQueueZ.FreeTensor(zLocal);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    uint32_t channelSize;
    uint32_t numChannels;
    uint32_t coreChannels;
    uint32_t tileLen;
    uint32_t paddedNumChannels;
    uint32_t coreElemOffset;
    uint32_t channelStart;
};

extern "C" __global__ __aicore__ void conv_transpose2d_subtract_tanh_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTransposeSubTanh op;
    op.Init(x, bias, z,
            tiling_data.totalLength, tiling_data.channelSize,
            tiling_data.numChannels, tiling_data.coreChannels,
            tiling_data.tileLen, tiling_data.paddedNumChannels);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, bias_shape, stride=2, padding=1, output_padding=1):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.bias = nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x.contiguous()
        bias_flat = self.bias.reshape(-1).contiguous()
        return custom_ops_lib.conv_transpose2d_subtract_tanh_custom(x, bias_flat)
