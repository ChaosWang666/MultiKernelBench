kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv3dMinSoftmax {
public:
    __aicore__ inline KernelConv3dMinSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t depth, uint32_t hw, uint32_t hwPadded,
                                 uint32_t batchesPerCore, uint32_t coreNum)
    {
        this->B = batchSize;
        this->C = channels;
        this->D = depth;
        this->HW = hw;
        this->HWP = hwPadded;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->batchStart = blockIdx * batchesPerCore;
        uint32_t end = this->batchStart + batchesPerCore;
        this->batchEnd = (end > batchSize) ? batchSize : end;

        this->CDHW = this->C * this->D * this->HW;
        this->CHW = this->C * this->HW;
        this->DHW = this->D * this->HW;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, this->B * this->CDHW);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, this->B * this->CHW);

        pipe.InitBuffer(inQueue, BUFFER_NUM, this->HWP * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->HWP * sizeof(DTYPE_Y));
        pipe.InitBuffer(minsBuf, this->C * this->HWP * sizeof(DTYPE_X));
        pipe.InitBuffer(maxBuf, this->HWP * sizeof(DTYPE_X));
        pipe.InitBuffer(sumBuf, this->HWP * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        if (this->batchStart >= this->batchEnd) {
            return;
        }
        for (uint32_t b = this->batchStart; b < this->batchEnd; b++) {
            ProcessBatch(b);
        }
    }

private:
    __aicore__ inline void CopyInDSlice(uint32_t baseOffset)
    {
        AscendC::LocalTensor<DTYPE_X> in = inQueue.AllocTensor<DTYPE_X>();
        AscendC::DataCopyExtParams ext;
        ext.blockCount = 1;
        ext.blockLen = this->HW * sizeof(DTYPE_X);
        ext.srcStride = 0;
        ext.dstStride = 0;
        AscendC::DataCopyPadExtParams<DTYPE_X> pad;
        pad.isPad = false;
        pad.leftPadding = 0;
        pad.rightPadding = 0;
        pad.paddingValue = 0;
        AscendC::DataCopyPad(in, xGm[baseOffset], ext, pad);
        inQueue.EnQue(in);
    }

    __aicore__ inline void ProcessBatch(uint32_t b)
    {
        AscendC::LocalTensor<DTYPE_X> mins = minsBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> maxL = maxBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> sumL = sumBuf.Get<DTYPE_X>();

        for (uint32_t c = 0; c < this->C; c++) {
            AscendC::LocalTensor<DTYPE_X> minsC = mins[c * this->HWP];
            uint32_t baseOffset = b * this->CDHW + c * this->DHW;

            CopyInDSlice(baseOffset);
            {
                AscendC::LocalTensor<DTYPE_X> in = inQueue.DeQue<DTYPE_X>();
                AscendC::Adds(minsC, in, (DTYPE_X)0, this->HW);
                inQueue.FreeTensor(in);
            }

            for (uint32_t d = 1; d < this->D; d++) {
                CopyInDSlice(baseOffset + d * this->HW);
                AscendC::LocalTensor<DTYPE_X> in = inQueue.DeQue<DTYPE_X>();
                AscendC::Min(minsC, minsC, in, this->HW);
                inQueue.FreeTensor(in);
            }
        }

        AscendC::Adds(maxL, mins, (DTYPE_X)0, this->HW);
        for (uint32_t c = 1; c < this->C; c++) {
            AscendC::Max(maxL, maxL, mins[c * this->HWP], this->HW);
        }

        for (uint32_t c = 0; c < this->C; c++) {
            AscendC::LocalTensor<DTYPE_X> minsC = mins[c * this->HWP];
            AscendC::Sub(minsC, minsC, maxL, this->HW);
            AscendC::Exp(minsC, minsC, this->HW);
        }

        AscendC::Adds(sumL, mins, (DTYPE_X)0, this->HW);
        for (uint32_t c = 1; c < this->C; c++) {
            AscendC::Add(sumL, sumL, mins[c * this->HWP], this->HW);
        }

        for (uint32_t c = 0; c < this->C; c++) {
            AscendC::LocalTensor<DTYPE_X> minsC = mins[c * this->HWP];
            AscendC::Div(minsC, minsC, sumL, this->HW);
        }

        for (uint32_t c = 0; c < this->C; c++) {
            AscendC::LocalTensor<DTYPE_Y> out = outQueue.AllocTensor<DTYPE_Y>();
            AscendC::Adds(out, mins[c * this->HWP], (DTYPE_Y)0, this->HW);
            outQueue.EnQue(out);

            AscendC::LocalTensor<DTYPE_Y> outDeq = outQueue.DeQue<DTYPE_Y>();
            uint32_t outOffset = b * this->CHW + c * this->HW;
            AscendC::DataCopyExtParams extOut;
            extOut.blockCount = 1;
            extOut.blockLen = this->HW * sizeof(DTYPE_Y);
            extOut.srcStride = 0;
            extOut.dstStride = 0;
            AscendC::DataCopyPad(yGm[outOffset], outDeq, extOut);
            outQueue.FreeTensor(outDeq);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> minsBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t B;
    uint32_t C;
    uint32_t D;
    uint32_t HW;
    uint32_t HWP;
    uint32_t batchStart;
    uint32_t batchEnd;
    uint32_t CDHW;
    uint32_t CHW;
    uint32_t DHW;
};

extern "C" __global__ __aicore__ void conv3d_min_softmax_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dMinSoftmax op;
    op.Init(x, y,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.depth, tiling_data.hw, tiling_data.hwPadded,
            tiling_data.batchesPerCore, tiling_data.coreNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, dim):
        super().__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.dim = dim

    def forward(self, x):
        x = self.conv(x)
        if self.dim == 2:
            x = x.contiguous()
            return custom_ops_lib.conv3d_min_softmax_custom(x)
        x = torch.min(x, dim=self.dim)[0]
        return torch.softmax(x, dim=1)
'''
