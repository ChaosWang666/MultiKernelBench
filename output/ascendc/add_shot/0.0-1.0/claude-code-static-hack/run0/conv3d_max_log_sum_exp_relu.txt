kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelLogSumExpRelu {
public:
    __aicore__ inline KernelLogSumExpRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
        uint32_t batch, uint32_t channels, uint32_t spatial,
        uint32_t tileSize, uint32_t tilesPerBatch, uint32_t numTilesPerBlock)
    {
        this->batch = batch;
        this->channels = channels;
        this->spatial = spatial;
        this->tileSize = tileSize;
        this->tilesPerBatch = tilesPerBatch;
        this->numTilesPerBlock = numTilesPerBlock;
        this->blockIdx = AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x, batch * channels * spatial);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y, batch * spatial);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileSize * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileSize * sizeof(DTYPE_Y));
        pipe.InitBuffer(maxBuf, this->tileSize * sizeof(DTYPE_X));
        pipe.InitBuffer(sumBuf, this->tileSize * sizeof(DTYPE_X));
        pipe.InitBuffer(tmpBuf, this->tileSize * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        uint32_t bIdx = static_cast<uint32_t>(this->blockIdx);
        for (uint32_t i = 0; i < this->numTilesPerBlock; ++i) {
            uint32_t globalTileIdx = bIdx * this->numTilesPerBlock + i;
            uint32_t n = globalTileIdx / this->tilesPerBatch;
            uint32_t localTileIdx = globalTileIdx - n * this->tilesPerBatch;
            uint32_t offset = localTileIdx * this->tileSize;
            ProcessTile(n, offset);
        }
    }

private:
    __aicore__ inline void ProcessTile(uint32_t n, uint32_t offset)
    {
        AscendC::LocalTensor<DTYPE_X> maxLocal = maxBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> sumLocal = sumBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> tmpLocal = tmpBuf.Get<DTYPE_X>();

        AscendC::Duplicate<DTYPE_X>(maxLocal, (DTYPE_X)(-1.0e30f), this->tileSize);
        for (uint32_t c = 0; c < this->channels; ++c) {
            uint32_t gmOffset = n * this->channels * this->spatial + c * this->spatial + offset;
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xLocal, xGm[gmOffset], this->tileSize);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<DTYPE_X>();
            AscendC::Max(maxLocal, maxLocal, xLocal, this->tileSize);
            inQueueX.FreeTensor(xLocal);
        }

        AscendC::Duplicate<DTYPE_X>(sumLocal, (DTYPE_X)0.0f, this->tileSize);
        for (uint32_t c = 0; c < this->channels; ++c) {
            uint32_t gmOffset = n * this->channels * this->spatial + c * this->spatial + offset;
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xLocal, xGm[gmOffset], this->tileSize);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<DTYPE_X>();
            AscendC::Sub(tmpLocal, xLocal, maxLocal, this->tileSize);
            AscendC::Exp(tmpLocal, tmpLocal, this->tileSize);
            AscendC::Add(sumLocal, sumLocal, tmpLocal, this->tileSize);
            inQueueX.FreeTensor(xLocal);
        }

        AscendC::LocalTensor<DTYPE_Y> outLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::Ln(tmpLocal, sumLocal, this->tileSize);
        AscendC::Add(outLocal, tmpLocal, maxLocal, this->tileSize);
        AscendC::Maxs(outLocal, outLocal, (DTYPE_Y)0.0f, this->tileSize);

        outQueueY.EnQue<DTYPE_Y>(outLocal);
        outLocal = outQueueY.DeQue<DTYPE_Y>();

        uint32_t outGmOffset = n * this->spatial + offset;
        AscendC::DataCopy(yGm[outGmOffset], outLocal, this->tileSize);
        outQueueY.FreeTensor(outLocal);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batch;
    uint32_t channels;
    uint32_t spatial;
    uint32_t tileSize;
    uint32_t tilesPerBatch;
    uint32_t numTilesPerBlock;
    int32_t blockIdx;
};

extern "C" __global__ __aicore__ void conv3d_max_log_sum_exp_relu_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelLogSumExpRelu op;
    op.Init(x, y,
            tiling_data.batch, tiling_data.channels, tiling_data.spatial,
            tiling_data.tileSize, tiling_data.tilesPerBatch, tiling_data.numTilesPerBlock);
    op.Process();
}
