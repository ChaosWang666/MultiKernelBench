kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;
using namespace matmul;

constexpr int32_t POST_TILE = 4096;

extern "C" __global__ __aicore__ void gemm_sigmoid_scaling_residual_add_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling) {

    GET_TILING_DATA(tiling_data, tiling);

    if (ASCEND_IS_AIC) {
        TPipe pipeC;
        GlobalTensor<float> xGm, wGm, bGm, zGm;
        xGm.SetGlobalBuffer((__gm__ float*)x);
        wGm.SetGlobalBuffer((__gm__ float*)weight);
        bGm.SetGlobalBuffer((__gm__ float*)bias);
        zGm.SetGlobalBuffer((__gm__ float*)z);

        Matmul<MatmulType<TPosition::GM, CubeFormat::ND, float>,
               MatmulType<TPosition::GM, CubeFormat::ND, float, true>,
               MatmulType<TPosition::GM, CubeFormat::ND, float>,
               MatmulType<TPosition::GM, CubeFormat::ND, float>> mm;
        REGIST_MATMUL_OBJ(&pipeC, GetSysWorkSpacePtr(), mm, &tiling_data.cubeTiling);

        mm.SetTensorA(xGm);
        mm.SetTensorB(wGm);
        mm.SetBias(bGm);
        mm.IterateAll(zGm);
        mm.End();

        SyncAll();
        CrossCoreSetFlag<0x2, PIPE_FIX>(0);
    }

    if (ASCEND_IS_AIV) {
        CrossCoreWaitFlag(0);

        TPipe pipeV;
        GlobalTensor<float> zGm;
        zGm.SetGlobalBuffer((__gm__ float*)z);

        uint32_t totalLen = tiling_data.totalLen;
        float scaling = tiling_data.scalingFactor;

        TQue<QuePosition::VECIN, 2> inQueue;
        TQue<QuePosition::VECOUT, 2> outQueue;
        TBuf<QuePosition::VECCALC> tmpBuf;
        pipeV.InitBuffer(inQueue, 2, POST_TILE * sizeof(float));
        pipeV.InitBuffer(outQueue, 2, POST_TILE * sizeof(float));
        pipeV.InitBuffer(tmpBuf, POST_TILE * sizeof(float));

        int32_t totalAiv = static_cast<int32_t>(GetBlockNum());
        int32_t aivIdx = static_cast<int32_t>(GetBlockIdx());
        int32_t totalLenSigned = static_cast<int32_t>(totalLen);

        int32_t perCore = (totalLenSigned + totalAiv - 1) / totalAiv;
        perCore = ((perCore + POST_TILE - 1) / POST_TILE) * POST_TILE;
        int32_t startOff = aivIdx * perCore;
        if (startOff >= totalLenSigned) {
            return;
        }
        int32_t endOff = startOff + perCore;
        if (endOff > totalLenSigned) {
            endOff = totalLenSigned;
        }

        for (int32_t off = startOff; off < endOff; off += POST_TILE) {
            int32_t curLen = (off + POST_TILE <= endOff) ? POST_TILE : (endOff - off);

            LocalTensor<float> inLocal = inQueue.AllocTensor<float>();
            DataCopy(inLocal, zGm[off], curLen);
            inQueue.EnQue(inLocal);

            LocalTensor<float> in2 = inQueue.DeQue<float>();
            LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
            LocalTensor<float> tmp = tmpBuf.Get<float>();

            Sigmoid(tmp, in2, curLen);
            Muls(tmp, tmp, scaling, curLen);
            Add(outLocal, tmp, in2, curLen);

            outQueue.EnQue<float>(outLocal);
            inQueue.FreeTensor(in2);

            LocalTensor<float> out2 = outQueue.DeQue<float>();
            DataCopy(zGm[off], out2, curLen);
            outQueue.FreeTensor(out2);
        }
    }
}
