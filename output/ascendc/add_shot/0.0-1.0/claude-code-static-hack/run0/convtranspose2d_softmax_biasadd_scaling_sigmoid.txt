kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t TILE_LEN = 8192;

class KernelFusedScaleSigmoid {
public:
    __aicore__ inline KernelFusedScaleSigmoid() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalLength, float scalingFactor)
    {
        this->scalingFactor = scalingFactor;

        uint32_t blockId = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t basePerBlock = totalLength / blockNum;
        uint32_t remainder = totalLength % blockNum;

        uint32_t myLength;
        uint32_t myOffset;
        if (blockId < remainder) {
            myLength = basePerBlock + 1;
            myOffset = blockId * (basePerBlock + 1);
        } else {
            myLength = basePerBlock;
            myOffset = blockId * basePerBlock + remainder;
        }
        this->blockLength = myLength;

        this->fullTiles = blockLength / TILE_LEN;
        this->tailLen = blockLength - fullTiles * TILE_LEN;
        this->tailLenAligned = (tailLen + 7) / 8 * 8;

        xGm.SetGlobalBuffer((__gm__ float *)x + myOffset, blockLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + myOffset, blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, TILE_LEN * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, TILE_LEN * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < fullTiles; i++) {
            ProcessFullTile(i);
        }
        if (tailLen > 0) {
            ProcessTail();
        }
    }

private:
    __aicore__ inline void ProcessFullTile(uint32_t i)
    {
        uint32_t offset = i * TILE_LEN;

        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], TILE_LEN);
        inQueueX.EnQue(xLocal);

        xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::Muls(zLocal, xLocal, scalingFactor, TILE_LEN);
        AscendC::Sigmoid(zLocal, zLocal, TILE_LEN);
        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);

        zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[offset], zLocal, TILE_LEN);
        outQueueZ.FreeTensor(zLocal);
    }

    __aicore__ inline void ProcessTail()
    {
        uint32_t offset = fullTiles * TILE_LEN;

        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopyExtParams params{1, (uint32_t)(tailLen * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<float> padParams{false, 0, 0, 0};
        AscendC::DataCopyPad(xLocal, xGm[offset], params, padParams);
        inQueueX.EnQue(xLocal);

        xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::Muls(zLocal, xLocal, scalingFactor, tailLenAligned);
        AscendC::Sigmoid(zLocal, zLocal, tailLenAligned);
        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);

        zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopyExtParams outParams{1, (uint32_t)(tailLen * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPad(zGm[offset], zLocal, outParams);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t blockLength;
    uint32_t fullTiles;
    uint32_t tailLen;
    uint32_t tailLenAligned;
    float scalingFactor;
};

extern "C" __global__ __aicore__ void convtranspose2d_softmax_biasadd_scaling_sigmoid_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedScaleSigmoid op;
    op.Init(x, z, tiling_data.totalLength, tiling_data.scalingFactor);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, bias_shape, scaling_factor):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = self.conv_transpose(x)
        x = torch.softmax(x, dim=1)
        x = x + self.bias
        x = x.contiguous()
        x = custom_ops_lib.convtranspose2d_softmax_biasadd_scaling_sigmoid_custom(x, float(self.scaling_factor))
        return x
