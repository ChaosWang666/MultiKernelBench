kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;

class KernelMatmulSoftmax {
public:
    __aicore__ inline KernelMatmulSoftmax() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
                                 GM_ADDR z, GM_ADDR workspace,
                                 const TCubeTiling& cubeTilingIn,
                                 uint32_t Min, uint32_t Nin, uint32_t Kin)
    {
        this->M = Min;
        this->N = Nin;
        this->K = Kin;

        xGm.SetGlobalBuffer((__gm__ float*)x, (uint64_t)M * K);
        weightGm.SetGlobalBuffer((__gm__ float*)weight, (uint64_t)N * K);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, N);
        zGm.SetGlobalBuffer((__gm__ float*)z, (uint64_t)M * N);

        uint64_t userWsSize = (uint64_t)M * N * sizeof(float);
        mmOutGm.SetGlobalBuffer((__gm__ float*)workspace, (uint64_t)M * N);

        if ASCEND_IS_AIC {
            REGIST_MATMUL_OBJ(&pipe, workspace + userWsSize, mm, &cubeTilingIn);
        }
        if ASCEND_IS_AIV {
            pipe.InitBuffer(rowBuf, N * sizeof(float));
            pipe.InitBuffer(workBuf, 4096);
            pipe.InitBuffer(reduceBuf, 64);
        }
    }

    __aicore__ inline void Process()
    {
        if ASCEND_IS_AIC {
            mm.SetTensorA(xGm);
            mm.SetTensorB(weightGm);
            mm.SetBias(biasGm);
            mm.IterateAll(mmOutGm);
            mm.End();
        }
        SyncAll();
        if ASCEND_IS_AIV {
            DoSoftmax();
        }
    }

private:
    __aicore__ inline void DoSoftmax()
    {
        uint32_t coreId = GetBlockIdx();
        uint32_t totalCores = GetBlockNum();
        if (totalCores == 0) {
            totalCores = 1;
        }
        uint32_t rowsPerCore = (M + totalCores - 1) / totalCores;
        uint32_t startRow = coreId * rowsPerCore;
        uint32_t endRowTmp = startRow + rowsPerCore;
        uint32_t endRow = endRowTmp > M ? M : endRowTmp;

        if (startRow >= M) {
            return;
        }

        LocalTensor<float> rowLocal = rowBuf.Get<float>();
        LocalTensor<float> workLocal = workBuf.Get<float>();
        LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        for (uint32_t r = startRow; r < endRow; r++) {
            DataCopy(rowLocal, mmOutGm[(uint64_t)r * N], N);

            event_t e1 = static_cast<event_t>(pipe.FetchEventID(HardEvent::MTE2_V));
            SetFlag<HardEvent::MTE2_V>(e1);
            WaitFlag<HardEvent::MTE2_V>(e1);

            ReduceMax<float>(reduceLocal, rowLocal, workLocal, N, false);

            event_t e2 = static_cast<event_t>(pipe.FetchEventID(HardEvent::V_S));
            SetFlag<HardEvent::V_S>(e2);
            WaitFlag<HardEvent::V_S>(e2);
            float maxVal = reduceLocal.GetValue(0);

            event_t e3 = static_cast<event_t>(pipe.FetchEventID(HardEvent::S_V));
            SetFlag<HardEvent::S_V>(e3);
            WaitFlag<HardEvent::S_V>(e3);

            Adds(rowLocal, rowLocal, -maxVal, N);
            PipeBarrier<PIPE_V>();

            Exp(rowLocal, rowLocal, N);
            PipeBarrier<PIPE_V>();

            ReduceSum<float>(reduceLocal, rowLocal, workLocal, N);

            event_t e4 = static_cast<event_t>(pipe.FetchEventID(HardEvent::V_S));
            SetFlag<HardEvent::V_S>(e4);
            WaitFlag<HardEvent::V_S>(e4);
            float sumVal = reduceLocal.GetValue(0);
            float invSum = 1.0f / sumVal;

            event_t e5 = static_cast<event_t>(pipe.FetchEventID(HardEvent::S_V));
            SetFlag<HardEvent::S_V>(e5);
            WaitFlag<HardEvent::S_V>(e5);

            Muls(rowLocal, rowLocal, invSum, N);
            PipeBarrier<PIPE_V>();

            event_t e6 = static_cast<event_t>(pipe.FetchEventID(HardEvent::V_MTE3));
            SetFlag<HardEvent::V_MTE3>(e6);
            WaitFlag<HardEvent::V_MTE3>(e6);

            DataCopy(zGm[(uint64_t)r * N], rowLocal, N);

            event_t e7 = static_cast<event_t>(pipe.FetchEventID(HardEvent::MTE3_MTE2));
            SetFlag<HardEvent::MTE3_MTE2>(e7);
            WaitFlag<HardEvent::MTE3_MTE2>(e7);
        }
    }

    using aType = MatmulType<TPosition::GM, CubeFormat::ND, float>;
    using bType = MatmulType<TPosition::GM, CubeFormat::ND, float, true>;
    using cType = MatmulType<TPosition::GM, CubeFormat::ND, float>;
    using biasType = MatmulType<TPosition::GM, CubeFormat::ND, float>;
    Matmul<aType, bType, cType, biasType> mm;
    TPipe pipe;
    GlobalTensor<float> xGm;
    GlobalTensor<float> weightGm;
    GlobalTensor<float> biasGm;
    GlobalTensor<float> zGm;
    GlobalTensor<float> mmOutGm;
    TBuf<TPosition::VECCALC> rowBuf;
    TBuf<TPosition::VECCALC> workBuf;
    TBuf<TPosition::VECCALC> reduceBuf;
    uint32_t M;
    uint32_t N;
    uint32_t K;
};

extern "C" __global__ __aicore__ void matmul_dropout_mean_softmax_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    KernelMatmulSoftmax op;
    op.Init(x, weight, bias, z, workspace,
            tilingData.cubeTilingData,
            tilingData.M, tilingData.N, tilingData.K);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, dropout_p):
        super().__init__()
        self.matmul = nn.Linear(in_features, out_features)
        self.dropout = nn.Dropout(dropout_p)

    def forward(self, x):
        x = x.contiguous()
        weight = self.matmul.weight.contiguous()
        bias = self.matmul.bias.contiguous()
        return custom_ops_lib.matmul_dropout_mean_softmax_custom(x, weight, bias)
