kernel_src="""
#include "kernel_operator.h"

class KernelBiasAddSum {
public:
    __aicore__ inline KernelBiasAddSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t channels, uint32_t batchPerCore)
    {
        this->channels = channels;
        this->batchPerCore = batchPerCore;
        int32_t blockIdx = AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ float *)x + blockIdx * batchPerCore * channels,
                            batchPerCore * channels);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, channels);
        zGm.SetGlobalBuffer((__gm__ float *)z + blockIdx * batchPerCore, batchPerCore);

        pipe.InitBuffer(inQueueX, 1, batchPerCore * channels * sizeof(float));
        pipe.InitBuffer(inQueueB, 1, channels * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, batchPerCore * sizeof(float));
        pipe.InitBuffer(tmpBuf, channels * sizeof(float));
        pipe.InitBuffer(workBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm, batchPerCore * channels);
        AscendC::DataCopy(bLocal, biasGm, channels);
        inQueueX.EnQue(xLocal);
        inQueueB.EnQue(bLocal);

        AscendC::LocalTensor<float> x2 = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> b2 = inQueueB.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();

        for (uint32_t i = 0; i < batchPerCore; i++) {
            AscendC::Add(tmp, x2[i * channels], b2, channels);
            AscendC::ReduceSum<float>(zLocal[i], tmp, work, channels);
        }

        outQueueZ.EnQue(zLocal);
        inQueueX.FreeTensor(x2);
        inQueueB.FreeTensor(b2);

        AscendC::LocalTensor<float> z2 = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm, z2, batchPerCore);
        outQueueZ.FreeTensor(z2);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, workBuf;
    AscendC::GlobalTensor<float> xGm, biasGm, zGm;
    uint32_t channels;
    uint32_t batchPerCore;
};

extern "C" __global__ __aicore__ void conv3d_divide_max_global_avg_pool_bias_add_sum_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{

    KernelBiasAddSum op;
    op.Init(x, bias, z, tiling_data.batchSize, tiling_data.channels, tiling_data.batchPerCore);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, divisor, pool_size, bias_shape, sum_dim):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        self.divisor = divisor
        self.max_pool = torch.nn.MaxPool3d(pool_size)
        self.global_avg_pool = torch.nn.AdaptiveAvgPool3d((1, 1, 1))
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.sum_dim = sum_dim

    def forward(self, x):
        x = self.conv(x)
        x = x / self.divisor
        x = self.max_pool(x)
        x = self.global_avg_pool(x)
        b = x.shape[0]
        c = x.shape[1]
        x_flat = x.reshape(b, c).contiguous()
        bias_flat = self.bias.reshape(c).contiguous()
        out = custom_ops_lib.conv3d_divide_max_global_avg_pool_bias_add_sum_custom(x_flat, bias_flat)
        return out.reshape(b, 1, 1, 1)
