kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSigmoidSum {
public:
    __aicore__ inline KernelSigmoidSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t batchSize, uint32_t perBatchLen, uint32_t tileLength)
    {
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->batchPerCore = batchSize / blockNum;
        this->perBatchLen = perBatchLen;
        this->tileLength = tileLength;
        this->numTiles = perBatchLen / tileLength;

        uint32_t startBatch = blockIdx * this->batchPerCore;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startBatch * perBatchLen, this->batchPerCore * perBatchLen);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + startBatch, this->batchPerCore);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(sigmoidBuf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(expBuf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(partialAccumBuf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(reduceWorkBuf, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, 1, 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_X> partialAccum = partialAccumBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> sigmoidLocal = sigmoidBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> expLocal = expBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> reduceWorkLocal = reduceWorkBuf.Get<DTYPE_X>();

        AscendC::Duplicate(outLocal, (DTYPE_Z)0.0f, 8);

        for (uint32_t b = 0; b < this->batchPerCore; b++) {
            AscendC::Duplicate(partialAccum, (DTYPE_X)0.0f, this->tileLength);

            for (uint32_t t = 0; t < this->numTiles; t++) {
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(xLocal, xGm[b * this->perBatchLen + t * this->tileLength], this->tileLength);
                inQueueX.EnQue(xLocal);

                xLocal = inQueueX.DeQue<DTYPE_X>();
                AscendC::Muls(expLocal, xLocal, (DTYPE_X)(-1.0f), this->tileLength);
                AscendC::Exp(expLocal, expLocal, this->tileLength);
                AscendC::Adds(expLocal, expLocal, (DTYPE_X)(1.0f), this->tileLength);
                AscendC::Duplicate(sigmoidLocal, (DTYPE_X)(1.0f), this->tileLength);
                AscendC::Div(sigmoidLocal, sigmoidLocal, expLocal, this->tileLength);
                AscendC::Add(partialAccum, partialAccum, sigmoidLocal, this->tileLength);
                inQueueX.FreeTensor(xLocal);
            }

            AscendC::ReduceSum(sigmoidLocal, partialAccum, reduceWorkLocal, this->tileLength);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

            DTYPE_Z val = sigmoidLocal.GetValue(0);
            outLocal.SetValue(b, val);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        outQueueZ.EnQue(outLocal);
        AscendC::LocalTensor<DTYPE_Z> outLocal2 = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm, outLocal2, this->batchPerCore);
        outQueueZ.FreeTensor(outLocal2);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sigmoidBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> expBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> partialAccumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceWorkBuf;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t batchPerCore;
    uint32_t perBatchLen;
    uint32_t tileLength;
    uint32_t numTiles;
};

extern "C" __global__ __aicore__ void conv2d_avg_pool_sigmoid_sum_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSigmoidSum op;
    op.Init(x, z, tiling_data.batchSize, tiling_data.perBatchLen, tiling_data.tileLength);
    op.Process();
}
