kernel_src="""
#include "kernel_operator.h"

class KernelConv3dFused {
public:
    __aicore__ inline KernelConv3dFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR sum_t, GM_ADDR y,
                                 uint32_t N_, uint32_t Cin_, uint32_t Cout_,
                                 uint32_t D_, uint32_t H_, uint32_t W_,
                                 uint32_t kD_, uint32_t kH_, uint32_t kW_,
                                 uint32_t Dout_, uint32_t Hout_, uint32_t Wout_)
    {
        N = N_; Cin = Cin_; Cout = Cout_;
        D = D_; H = H_; W = W_;
        kD = kD_; kH = kH_; kW = kW_;
        Dout = Dout_; Hout = Hout_; Wout = Wout_;
        WoutPadded = ((Wout + 7) / 8) * 8;
        CoutAligned = ((Cout + 7) / 8) * 8;
        weightTotal = Cout * Cin * kD * kH * kW;
        weightAligned = ((weightTotal + 7) / 8) * 8;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t totalNCo = N * Cout;
        ncoStart = (totalNCo * blockIdx) / blockNum;
        ncoEnd = (totalNCo * (blockIdx + 1)) / blockNum;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        wGm.SetGlobalBuffer((__gm__ float*)weight);
        bGm.SetGlobalBuffer((__gm__ float*)bias);
        sGm.SetGlobalBuffer((__gm__ float*)sum_t);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        pipe.InitBuffer(inQueueX, 1, H * W * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, Hout * WoutPadded * sizeof(float));
        pipe.InitBuffer(wBuf, weightAligned * sizeof(float));
        pipe.InitBuffer(bBuf, CoutAligned * sizeof(float));
        pipe.InitBuffer(sBuf, CoutAligned * sizeof(float));
        pipe.InitBuffer(tmpBuf, Hout * WoutPadded * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> bLocal = bBuf.Get<float>();
        AscendC::LocalTensor<float> sLocal = sBuf.Get<float>();

        AscendC::DataCopy(wLocal, wGm, weightAligned);
        AscendC::DataCopy(bLocal, bGm, CoutAligned);
        AscendC::DataCopy(sLocal, sGm, CoutAligned);

        AscendC::TEventID evtMte2S = pipe.FetchEventID(AscendC::HardEvent::MTE2_S);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(evtMte2S);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(evtMte2S);

        for (uint32_t nco = ncoStart; nco < ncoEnd; nco++) {
            uint32_t n = nco / Cout;
            uint32_t co = nco % Cout;
            float biasVal = bLocal.GetValue(co);
            float sumVal = sLocal.GetValue(co);
            ProcessOneNCo(n, co, biasVal, sumVal, wLocal);
        }
    }

private:
    __aicore__ inline void ProcessOneNCo(uint32_t n, uint32_t co, float biasVal, float sumVal,
                                          AscendC::LocalTensor<float>& wLocal)
    {
        for (uint32_t d_out = 0; d_out < Dout; d_out++) {
            ProcessOneDout(n, co, d_out, biasVal, sumVal, wLocal);
        }
    }

    __aicore__ inline void ProcessOneDout(uint32_t n, uint32_t co, uint32_t d_out,
                                           float biasVal, float sumVal,
                                           AscendC::LocalTensor<float>& wLocal)
    {
        AscendC::LocalTensor<float> outLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        uint32_t totalElems = Hout * WoutPadded;
        AscendC::Duplicate(outLocal, biasVal, totalElems);

        for (uint32_t ci = 0; ci < Cin; ci++) {
            for (uint32_t kd = 0; kd < kD; kd++) {
                AscendC::LocalTensor<float> xPlane = inQueueX.AllocTensor<float>();
                uint64_t xOffset = ((uint64_t)n * Cin + ci) * D * H * W +
                                   (uint64_t)(d_out + kd) * H * W;
                AscendC::DataCopy(xPlane, xGm[xOffset], H * W);
                inQueueX.EnQue(xPlane);
                xPlane = inQueueX.DeQue<float>();

                for (uint32_t kh = 0; kh < kH; kh++) {
                    uint32_t wBase = ((co * Cin + ci) * kD + kd) * kH * kW + kh * kW;
                    for (uint32_t kw = 0; kw < kW; kw++) {
                        float wScalar = wLocal.GetValue(wBase + kw);
                        for (uint32_t h_out = 0; h_out < Hout; h_out++) {
                            AscendC::Axpy(outLocal[h_out * WoutPadded],
                                           xPlane[(h_out + kh) * W + kw],
                                           wScalar, Wout);
                        }
                    }
                }

                inQueueX.FreeTensor(xPlane);
            }
        }

        AscendC::Muls(tmpLocal, outLocal, (float)0.2f, totalElems);
        AscendC::Max(outLocal, outLocal, tmpLocal, totalElems);

        AscendC::Adds(outLocal, outLocal, sumVal, totalElems);

        AscendC::Maxs(outLocal, outLocal, (float)-1.0f, totalElems);
        AscendC::Mins(outLocal, outLocal, (float)1.0f, totalElems);

        AscendC::Muls(tmpLocal, outLocal, (float)0.7071067811865475f, totalElems);
        AscendC::Erf(tmpLocal, tmpLocal, totalElems);
        AscendC::Adds(tmpLocal, tmpLocal, (float)1.0f, totalElems);
        AscendC::Mul(outLocal, outLocal, tmpLocal, totalElems);
        AscendC::Muls(outLocal, outLocal, (float)0.5f, totalElems);

        outQueueZ.EnQue(outLocal);
        outLocal = outQueueZ.DeQue<float>();

        for (uint32_t h_out = 0; h_out < Hout; h_out++) {
            uint64_t outOffset = ((uint64_t)n * Cout + co) * Dout * Hout * Wout +
                                  (uint64_t)d_out * Hout * Wout +
                                  (uint64_t)h_out * Wout;
            AscendC::DataCopyExtParams copyParams;
            copyParams.blockCount = 1;
            copyParams.blockLen = Wout * sizeof(float);
            copyParams.srcStride = 0;
            copyParams.dstStride = 0;
            AscendC::DataCopyPad(yGm[outOffset], outLocal[h_out * WoutPadded], copyParams);
        }

        outQueueZ.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;

    AscendC::GlobalTensor<float> xGm, wGm, bGm, sGm, yGm;

    uint32_t N, Cin, Cout, D, H, W, kD, kH, kW, Dout, Hout, Wout;
    uint32_t WoutPadded, CoutAligned, weightTotal, weightAligned;
    uint32_t ncoStart, ncoEnd;
};

extern "C" __global__ __aicore__ void conv3d_leaky_relu_sum_clamp_gelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR sum_tensor, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dFused op;
    op.Init(x, weight, bias, sum_tensor, y,
            tiling_data.N, tiling_data.Cin, tiling_data.Cout,
            tiling_data.D, tiling_data.H, tiling_data.W,
            tiling_data.kD, tiling_data.kH, tiling_data.kW,
            tiling_data.Dout, tiling_data.Hout, tiling_data.Wout);
    op.Process();
}

import torch_npu
