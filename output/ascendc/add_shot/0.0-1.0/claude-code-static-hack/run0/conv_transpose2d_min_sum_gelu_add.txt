kernel_src="""
#include "kernel_operator.h"

constexpr int32_t IN_BUFFER_NUM = 2;

class KernelMinSumGeluAdd {
public:
    __aicore__ inline KernelMinSumGeluAdd() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                  uint32_t batch, uint32_t channel, uint32_t height, uint32_t width,
                                  uint32_t tileW, uint32_t numWChunks)
    {
        this->batch = batch;
        this->channel = channel;
        this->height = height;
        this->width = width;
        this->tileW = tileW;
        this->numWChunks = numWChunks;

        uint32_t blockId = AscendC::GetBlockIdx();
        this->bIdx = blockId / numWChunks;
        this->wChunkIdx = blockId % numWChunks;
        this->wStart = this->wChunkIdx * tileW;

        xGm.SetGlobalBuffer((__gm__ float*)x, batch * channel * height * width);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, 1);
        yGm.SetGlobalBuffer((__gm__ float*)y, batch * width);

        pipe.InitBuffer(inQueueX, IN_BUFFER_NUM, this->channel * this->tileW * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, this->tileW * sizeof(float));
        pipe.InitBuffer(sumBuf, this->tileW * sizeof(float));
        pipe.InitBuffer(minBuf, this->tileW * sizeof(float));
        pipe.InitBuffer(tmp1Buf, this->tileW * sizeof(float));
        pipe.InitBuffer(tmp2Buf, this->tileW * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::Duplicate(sumLocal, 0.0f, this->tileW);

        for (uint32_t h = 0; h < this->height; h++) {
            CopyIn(h);
            ComputeMinAndAdd();
        }

        ComputeGeluAndBias();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn(uint32_t h)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopyParams params;
        params.blockCount = (uint16_t)this->channel;
        params.blockLen = (uint16_t)(this->tileW / 8u);
        params.srcStride = (uint16_t)((this->height * this->width - this->tileW) / 8u);
        params.dstStride = 0;

        uint32_t offset = this->bIdx * this->channel * this->height * this->width
                          + h * this->width + this->wStart;
        AscendC::DataCopy(xLocal, xGm[offset], params);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void ComputeMinAndAdd()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> minLocal = minBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();

        AscendC::DataCopy(minLocal, xLocal, this->tileW);
        for (uint32_t c = 1; c < this->channel; c++) {
            AscendC::Min(minLocal, minLocal, xLocal[c * this->tileW], this->tileW);
        }

        AscendC::Add(sumLocal, sumLocal, minLocal, this->tileW);

        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void ComputeGeluAndBias()
    {
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> tmp1 = tmp1Buf.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmp2Buf.Get<float>();

        const float kSqrt2OverPi = 0.7978845608028654f;
        const float kCube = 0.044715f;

        AscendC::Mul(tmp1, sumLocal, sumLocal, this->tileW);
        AscendC::Mul(tmp1, tmp1, sumLocal, this->tileW);
        AscendC::Muls(tmp1, tmp1, kCube, this->tileW);
        AscendC::Add(tmp1, sumLocal, tmp1, this->tileW);
        AscendC::Muls(tmp1, tmp1, kSqrt2OverPi, this->tileW);

        AscendC::Muls(tmp2, tmp1, 2.0f, this->tileW);
        AscendC::Exp(tmp2, tmp2, this->tileW);
        AscendC::Adds(tmp2, tmp2, 1.0f, this->tileW);
        AscendC::Reciprocal(tmp2, tmp2, this->tileW);
        AscendC::Muls(tmp2, tmp2, -2.0f, this->tileW);
        AscendC::Adds(tmp2, tmp2, 1.0f, this->tileW);
        AscendC::Adds(tmp2, tmp2, 1.0f, this->tileW);

        AscendC::Muls(tmp1, sumLocal, 0.5f, this->tileW);
        AscendC::Mul(tmp1, tmp1, tmp2, this->tileW);

        float biasVal = biasGm.GetValue(0);
        AscendC::Adds(tmp1, tmp1, biasVal, this->tileW);

        AscendC::LocalTensor<float> outLocal = outQueueY.AllocTensor<float>();
        AscendC::DataCopy(outLocal, tmp1, this->tileW);
        outQueueY.EnQue(outLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<float> outLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[this->bIdx * this->width + this->wStart], outLocal, this->tileW);
        outQueueY.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, IN_BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> minBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmp1Buf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmp2Buf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batch;
    uint32_t channel;
    uint32_t height;
    uint32_t width;
    uint32_t tileW;
    uint32_t numWChunks;
    uint32_t bIdx;
    uint32_t wChunkIdx;
    uint32_t wStart;
};

extern "C" __global__ __aicore__ void conv_transpose2d_min_sum_gelu_add_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMinSumGeluAdd op;
    op.Init(x, bias, y,
            tiling_data.batch, tiling_data.channel, tiling_data.height, tiling_data.width,
            tiling_data.tileW, tiling_data.numWChunks);
    op.Process();
}
