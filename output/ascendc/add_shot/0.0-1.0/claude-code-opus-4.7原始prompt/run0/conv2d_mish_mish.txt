kernel_src="""
#include "kernel_operator.h"

class KernelConv2dMishMish {
public:
    __aicore__ inline KernelConv2dMishMish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inHeight, uint32_t inWidth,
                                 uint32_t outHeight, uint32_t outWidth,
                                 uint32_t kernelSize, uint32_t totalOutElements)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->kernelSize = kernelSize;
        this->totalOutElements = totalOutElements;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t perBlock = (totalOutElements + blockNum - 1) / blockNum;
        this->startIdx = blockIdx * perBlock;
        this->endIdx = (this->startIdx + perBlock > totalOutElements) ? totalOutElements : (this->startIdx + perBlock);

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * inHeight * inWidth);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outChannels * inChannels * kernelSize * kernelSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalOutElements);
    }

    __aicore__ inline float mish_scalar(float v)
    {
        float sp;
        if (v > 20.0f) {
            sp = v;
        } else if (v < -20.0f) {
            sp = expf(v);
        } else {
            sp = logf(1.0f + expf(v));
        }
        float th = tanhf(sp);
        return v * th;
    }

    __aicore__ inline void Process()
    {
        uint32_t outHW = outHeight * outWidth;
        uint32_t outCHW = outChannels * outHW;

        for (uint32_t idx = startIdx; idx < endIdx; idx++) {
            uint32_t ow = idx % outWidth;
            uint32_t oh = (idx / outWidth) % outHeight;
            uint32_t oc = (idx / outHW) % outChannels;
            uint32_t b  = idx / outCHW;

            float sum = bGm.GetValue(oc);

            for (uint32_t ic = 0; ic < inChannels; ic++) {
                for (uint32_t kh = 0; kh < kernelSize; kh++) {
                    for (uint32_t kw = 0; kw < kernelSize; kw++) {
                        uint32_t ih = oh + kh;
                        uint32_t iw = ow + kw;
                        uint32_t xOff = ((b * inChannels + ic) * inHeight + ih) * inWidth + iw;
                        uint32_t wOff = ((oc * inChannels + ic) * kernelSize + kh) * kernelSize + kw;
                        sum += xGm.GetValue(xOff) * wGm.GetValue(wOff);
                    }
                }
            }

            float v = mish_scalar(sum);
            v = mish_scalar(v);
            yGm.SetValue(idx, v);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inHeight;
    uint32_t inWidth;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t kernelSize;
    uint32_t totalOutElements;
    uint32_t startIdx;
    uint32_t endIdx;
};

extern "C" __global__ __aicore__ void conv2d_mish_mish_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                                                GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dMishMish op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inHeight, tiling_data.inWidth,
            tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.kernelSize, tiling_data.totalOutElements);
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor conv2d_mish_mish_custom_impl_npu(const at::Tensor& x, const at::Tensor& weight, const at::Tensor& bias) {
    auto xSizes = x.sizes();
    auto wSizes = weight.sizes();
    int64_t batchSize = xSizes[0];
    int64_t outChannels = wSizes[0];
    int64_t kernelSize = wSizes[2];
    int64_t outHeight = xSizes[2] - kernelSize + 1;
    int64_t outWidth = xSizes[3] - kernelSize + 1;
    at::Tensor result = at::empty({batchSize, outChannels, outHeight, outWidth}, x.options());
    EXEC_NPU_CMD(aclnnConv2dMishMishCustom, x, weight, bias, result);
    return result;
}

    m.impl("conv2d_mish_mish_custom", &conv2d_mish_mish_custom_impl_npu);

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)

    def forward(self, x):
        weight = self.conv.weight.contiguous()
        bias = self.conv.bias.contiguous()
        return custom_ops_lib.conv2d_mish_mish_custom(x, weight, bias)
