kernel_src="""
#include "kernel_operator.h"

class KernelConv3dHardswishGnMean {
public:
    __aicore__ inline KernelConv3dHardswishGnMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
                                GM_ADDR gn_weight, GM_ADDR gn_bias, GM_ADDR y,
                                uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                uint32_t inDepth, uint32_t inHeight, uint32_t inWidth,
                                uint32_t kernelSize, uint32_t outDepth, uint32_t outHeight,
                                uint32_t outWidth, uint32_t numGroups, uint32_t channelsPerGroup,
                                uint32_t spatialSize)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inDepth = inDepth;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->kernelSize = kernelSize;
        this->outDepth = outDepth;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->numGroups = numGroups;
        this->channelsPerGroup = channelsPerGroup;
        this->spatialSize = spatialSize;
        this->inputVol = inChannels * inDepth * inHeight * inWidth;
        this->outputVol = outChannels * spatialSize;
        this->weightVol = inChannels * kernelSize * kernelSize * kernelSize;

        xGm.SetGlobalBuffer((__gm__ float *)x);
        wGm.SetGlobalBuffer((__gm__ float *)weight);
        bGm.SetGlobalBuffer((__gm__ float *)bias);
        gnWGm.SetGlobalBuffer((__gm__ float *)gn_weight);
        gnBGm.SetGlobalBuffer((__gm__ float *)gn_bias);
        yGm.SetGlobalBuffer((__gm__ float *)y);
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        for (uint32_t b = blockIdx; b < batchSize; b += blockNum) {
            ProcessBatch(b);
        }
    }

private:
    __aicore__ inline float HardSwish(float v)
    {
        float t = v + 3.0f;
        if (t < 0.0f) t = 0.0f;
        if (t > 6.0f) t = 6.0f;
        return v * t / 6.0f;
    }

    __aicore__ inline void ProcessBatch(uint32_t b)
    {
        const uint32_t kVol = kernelSize * kernelSize * kernelSize;
        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t cStart = g * channelsPerGroup;
            uint32_t cEnd = cStart + channelsPerGroup;
            uint32_t groupElems = channelsPerGroup * spatialSize;

            float mean = 0.0f;
            float m2 = 0.0f;

            for (uint32_t c = cStart; c < cEnd; c++) {
                float bv = bGm.GetValue(c);
                for (uint32_t od = 0; od < outDepth; od++) {
                    for (uint32_t oh = 0; oh < outHeight; oh++) {
                        for (uint32_t ow = 0; ow < outWidth; ow++) {
                            float s = bv;
                            for (uint32_t ic = 0; ic < inChannels; ic++) {
                                for (uint32_t kd = 0; kd < kernelSize; kd++) {
                                    for (uint32_t kh = 0; kh < kernelSize; kh++) {
                                        for (uint32_t kw = 0; kw < kernelSize; kw++) {
                                            uint32_t xIdx = ((b * inChannels + ic) * inDepth + (od + kd)) * inHeight * inWidth
                                                          + (oh + kh) * inWidth + (ow + kw);
                                            uint32_t wIdx = ((c * inChannels + ic) * kernelSize + kd) * kernelSize * kernelSize
                                                          + kh * kernelSize + kw;
                                            s += xGm.GetValue(xIdx) * wGm.GetValue(wIdx);
                                        }
                                    }
                                }
                            }
                            float hs = HardSwish(s);
                            mean += hs;
                            m2 += hs * hs;
                        }
                    }
                }
            }

            mean = mean / (float)groupElems;
            float var = m2 / (float)groupElems - mean * mean;
            if (var < 0.0f) var = 0.0f;
            float invStd = 1.0f / AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                                      AscendC::ScalarGetSqrtVal<float>(var + 1e-5f));

            for (uint32_t c = cStart; c < cEnd; c++) {
                float gw = gnWGm.GetValue(c);
                float gb = gnBGm.GetValue(c);
                float bv = bGm.GetValue(c);
                float acc = 0.0f;
                for (uint32_t od = 0; od < outDepth; od++) {
                    for (uint32_t oh = 0; oh < outHeight; oh++) {
                        for (uint32_t ow = 0; ow < outWidth; ow++) {
                            float s = bv;
                            for (uint32_t ic = 0; ic < inChannels; ic++) {
                                for (uint32_t kd = 0; kd < kernelSize; kd++) {
                                    for (uint32_t kh = 0; kh < kernelSize; kh++) {
                                        for (uint32_t kw = 0; kw < kernelSize; kw++) {
                                            uint32_t xIdx = ((b * inChannels + ic) * inDepth + (od + kd)) * inHeight * inWidth
                                                          + (oh + kh) * inWidth + (ow + kw);
                                            uint32_t wIdx = ((c * inChannels + ic) * kernelSize + kd) * kernelSize * kernelSize
                                                          + kh * kernelSize + kw;
                                            s += xGm.GetValue(xIdx) * wGm.GetValue(wIdx);
                                        }
                                    }
                                }
                            }
                            float hs = HardSwish(s);
                            float norm = (hs - mean) * invStd * gw + gb;
                            acc += norm;
                        }
                    }
                }
                float meanOut = acc / (float)spatialSize;
                yGm.SetValue(b * outChannels + c, meanOut);
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> gnWGm;
    AscendC::GlobalTensor<float> gnBGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inDepth;
    uint32_t inHeight;
    uint32_t inWidth;
    uint32_t kernelSize;
    uint32_t outDepth;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t numGroups;
    uint32_t channelsPerGroup;
    uint32_t spatialSize;
    uint32_t inputVol;
    uint32_t outputVol;
    uint32_t weightVol;
};

extern "C" __global__ __aicore__ void conv3d_hardswish_relu_softmax_mean_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR gn_weight, GM_ADDR gn_bias,
    GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dHardswishGnMean op;
    op.Init(x, weight, bias, gn_weight, gn_bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inDepth, tiling_data.inHeight, tiling_data.inWidth,
            tiling_data.kernelSize, tiling_data.outDepth, tiling_data.outHeight,
            tiling_data.outWidth, tiling_data.numGroups, tiling_data.channelsPerGroup,
            tiling_data.spatialSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, num_groups=4, bias=True):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size, bias=bias)
        self.group_norm = nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        weight = self.conv.weight.contiguous()
        bias = self.conv.bias.contiguous() if self.conv.bias is not None else torch.zeros(
            self.conv.out_channels, device=x.device, dtype=x.dtype)
        gn_weight = self.group_norm.weight.contiguous()
        gn_bias = self.group_norm.bias.contiguous()
        return custom_ops_lib.conv3d_hardswish_relu_softmax_mean_custom(
            x.contiguous(), weight, bias, gn_weight, gn_bias)
'''
