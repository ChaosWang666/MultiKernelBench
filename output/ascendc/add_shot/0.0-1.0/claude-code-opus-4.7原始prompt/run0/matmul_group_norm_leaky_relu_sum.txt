kernel_src="""
#include "kernel_operator.h"

using namespace AscendC;

class KernelMatmulGNLeakyReluSum {
public:
    __aicore__ inline KernelMatmulGNLeakyReluSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
                                GM_ADDR gn_weight, GM_ADDR gn_bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize,
                                uint32_t numGroups, uint32_t groupSize,
                                float eps, float negativeSlope)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->numGroups = numGroups;
        this->groupSize = groupSize;
        this->eps = eps;
        this->negativeSlope = negativeSlope;

        uint32_t blockIdx = GetBlockIdx();
        uint32_t blockNum = GetBlockNum();

        this->rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->rowStart = blockIdx * this->rowsPerBlock;
        this->rowEnd = this->rowStart + this->rowsPerBlock;
        if (this->rowEnd > batchSize) this->rowEnd = batchSize;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inputSize);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, hiddenSize * inputSize);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, hiddenSize);
        gnWeightGm.SetGlobalBuffer((__gm__ float *)gn_weight, hiddenSize);
        gnBiasGm.SetGlobalBuffer((__gm__ float *)gn_bias, hiddenSize);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * hiddenSize);

        pipe.InitBuffer(xBuf, inputSize * sizeof(float));
        pipe.InitBuffer(wBuf, inputSize * sizeof(float));
        pipe.InitBuffer(outBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(biasBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(gnWBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(gnBBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, groupSize * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->rowStart >= this->rowEnd) return;

        LocalTensor<float> biasLocal = biasBuf.Get<float>();
        LocalTensor<float> gnWLocal = gnWBuf.Get<float>();
        LocalTensor<float> gnBLocal = gnBBuf.Get<float>();
        DataCopy(biasLocal, biasGm, hiddenSize);
        DataCopy(gnWLocal, gnWeightGm, hiddenSize);
        DataCopy(gnBLocal, gnBiasGm, hiddenSize);

        for (uint32_t row = this->rowStart; row < this->rowEnd; row++) {
            ProcessRow(row, biasLocal, gnWLocal, gnBLocal);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t row, LocalTensor<float>& biasLocal,
                                      LocalTensor<float>& gnWLocal, LocalTensor<float>& gnBLocal)
    {
        LocalTensor<float> xLocal = xBuf.Get<float>();
        LocalTensor<float> outLocal = outBuf.Get<float>();
        LocalTensor<float> wLocal = wBuf.Get<float>();
        LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        DataCopy(xLocal, xGm[row * inputSize], inputSize);

        for (uint32_t h = 0; h < hiddenSize; h++) {
            DataCopy(wLocal, weightGm[h * inputSize], inputSize);
            Mul(tmpLocal, xLocal, wLocal, groupSize);
            float sum = 0.0f;
            for (uint32_t i = 0; i < inputSize; i++) {
                sum += xLocal.GetValue(i) * wLocal.GetValue(i);
            }
            outLocal.SetValue(h, sum + biasLocal.GetValue(h));
        }

        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t gStart = g * groupSize;
            float mean = 0.0f;
            for (uint32_t i = 0; i < groupSize; i++) {
                mean += outLocal.GetValue(gStart + i);
            }
            mean /= (float)groupSize;

            float var = 0.0f;
            for (uint32_t i = 0; i < groupSize; i++) {
                float d = outLocal.GetValue(gStart + i) - mean;
                var += d * d;
            }
            var /= (float)groupSize;
            float invStd = 1.0f / sqrtf(var + eps);

            for (uint32_t i = 0; i < groupSize; i++) {
                uint32_t idx = gStart + i;
                float v = (outLocal.GetValue(idx) - mean) * invStd;
                v = v * gnWLocal.GetValue(idx) + gnBLocal.GetValue(idx);
                if (v < 0.0f) v = v * negativeSlope;
                v = v + v;
                outLocal.SetValue(idx, v);
            }
        }

        DataCopy(zGm[row * hiddenSize], outLocal, hiddenSize);
    }

private:
    TPipe pipe;
    TBuf<TPosition::VECCALC> xBuf, wBuf, outBuf, biasBuf, gnWBuf, gnBBuf, tmpBuf;
    GlobalTensor<float> xGm, weightGm, biasGm, gnWeightGm, gnBiasGm, zGm;
    uint32_t batchSize, inputSize, hiddenSize, numGroups, groupSize;
    uint32_t rowsPerBlock, rowStart, rowEnd;
    float eps, negativeSlope;
};

extern "C" __global__ __aicore__ void matmul_group_norm_leaky_relu_sum_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR gn_weight, GM_ADDR gn_bias,
    GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulGNLeakyReluSum op;
    op.Init(x, weight, bias, gn_weight, gn_bias, z,
            tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize,
            tiling_data.numGroups, tiling_data.groupSize,
            tiling_data.eps, tiling_data.negativeSlope);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, input_size, hidden_size, num_groups, eps=1e-5, negative_slope=0.01):
        super(ModelNew, self).__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.num_groups = num_groups
        self.eps = eps
        self.negative_slope = negative_slope
        self.fc = nn.Linear(input_size, hidden_size)
        self.gn = nn.GroupNorm(num_groups=num_groups, num_channels=hidden_size, eps=eps)

    def forward(self, x):
        return custom_ops_lib.matmul_group_norm_leaky_relu_sum_custom(
            x, self.fc.weight, self.fc.bias,
            self.gn.weight, self.gn.bias,
            self.input_size, self.hidden_size, self.num_groups,
