kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelGbnsSoftmax {
public:
    __aicore__ inline KernelGbnsSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t featSize)
    {
        this->featSize = featSize;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->rowsPerBlock = batchSize / blockNum;
        uint32_t remainder = batchSize % blockNum;
        uint32_t startRow;
        if (blockIdx < remainder) {
            this->rowsPerBlock += 1;
            startRow = blockIdx * this->rowsPerBlock;
        } else {
            startRow = blockIdx * this->rowsPerBlock + remainder;
        }

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startRow * featSize, this->rowsPerBlock * featSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + startRow * featSize, this->rowsPerBlock * featSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, featSize * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, featSize * sizeof(DTYPE_Y));
        pipe.InitBuffer(tmpBuf, 256);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsPerBlock; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->featSize], this->featSize);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        AscendC::ReduceMax<float>(tmp, xLocal, tmp, this->featSize, false);
        float maxVal = tmp.GetValue(0);

        AscendC::Adds(yLocal, xLocal, -maxVal, this->featSize);
        AscendC::Exp(yLocal, yLocal, this->featSize);

        AscendC::ReduceSum<float>(tmp, yLocal, tmp, this->featSize);
        float sumVal = tmp.GetValue(0);

        float invSum = 1.0f / sumVal;
        AscendC::Muls(yLocal, yLocal, invSum, this->featSize);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[progress * this->featSize], yLocal, this->featSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t featSize;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void gemm_batch_norm_scaling_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGbnsSoftmax op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.featSize);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    """
    Optimized model using a custom AscendC kernel for the final scale+softmax stage.
    The preceding Gemm and BatchNorm1d are executed with PyTorch operators; the
    custom kernel consumes the scaled activations and produces the softmax output
    in one fused pass over each row.
    """
    def __init__(self, in_features, out_features, bn_eps=1e-5, bn_momentum=0.1, scale_shape=(1,)):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.bn = nn.BatchNorm1d(out_features, eps=bn_eps, momentum=bn_momentum)
        self.scale = nn.Parameter(torch.ones(scale_shape))

    def forward(self, x):
        x = self.gemm(x)
        x = self.bn(x)
        x = self.scale * x
        x = x.contiguous()
        x = custom_ops_lib.gemm_batch_norm_scaling_softmax_custom(x)
        return x
'''
