kernel_src="""
#include "kernel_operator.h"

class KernelConv2dMinTanhTanh {
public:
    __aicore__ inline KernelConv2dMinTanhTanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inHeight, uint32_t inWidth,
                                 uint32_t outHeight, uint32_t outWidth,
                                 uint32_t kernelSize, uint32_t totalOutPixels)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->kernelSize = kernelSize;
        this->totalOutPixels = totalOutPixels;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * inHeight * inWidth);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outChannels * inChannels * kernelSize * kernelSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize * outHeight * outWidth);

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->pixelsPerBlock = (totalOutPixels + blockNum - 1) / blockNum;
        this->startPixel = blockIdx * this->pixelsPerBlock;
        this->endPixel = this->startPixel + this->pixelsPerBlock;
        if (this->endPixel > totalOutPixels) {
            this->endPixel = totalOutPixels;
        }
    }

    __aicore__ inline void Process()
    {
        for (uint32_t p = this->startPixel; p < this->endPixel; p++) {
            uint32_t n = p / (this->outHeight * this->outWidth);
            uint32_t hw = p % (this->outHeight * this->outWidth);
            uint32_t oh = hw / this->outWidth;
            uint32_t ow = hw % this->outWidth;

            float minVal = 3.402823466e+38f;

            for (uint32_t oc = 0; oc < this->outChannels; oc++) {
                float sum = bGm.GetValue(oc);
                for (uint32_t ic = 0; ic < this->inChannels; ic++) {
                    for (uint32_t kh = 0; kh < this->kernelSize; kh++) {
                        for (uint32_t kw = 0; kw < this->kernelSize; kw++) {
                            uint32_t ih = oh + kh;
                            uint32_t iw = ow + kw;
                            uint32_t xIdx = ((n * this->inChannels + ic) * this->inHeight + ih) * this->inWidth + iw;
                            uint32_t wIdx = ((oc * this->inChannels + ic) * this->kernelSize + kh) * this->kernelSize + kw;
                            float xVal = xGm.GetValue(xIdx);
                            float wVal = wGm.GetValue(wIdx);
                            sum += xVal * wVal;
                        }
                    }
                }
                if (sum < minVal) {
                    minVal = sum;
                }
            }

            float t1 = tanhf_approx(minVal);
            float t2 = tanhf_approx(t1);
            yGm.SetValue(p, t2);
        }
    }

private:
    __aicore__ inline float tanhf_approx(float x) {
        float e2x;
        float absx = x < 0 ? -x : x;
        if (absx > 10.0f) {
            return x < 0 ? -1.0f : 1.0f;
        }
        float ex = expf_approx(x);
        float enx = expf_approx(-x);
        return (ex - enx) / (ex + enx);
    }

    __aicore__ inline float expf_approx(float x) {
        if (x > 88.0f) return 1e38f;
        if (x < -88.0f) return 0.0f;
        float result = 1.0f;
        float term = 1.0f;
        for (int i = 1; i < 20; i++) {
            term *= x / (float)i;
            result += term;
        }
        return result;
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inHeight;
    uint32_t inWidth;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t kernelSize;
    uint32_t totalOutPixels;
    uint32_t pixelsPerBlock;
    uint32_t startPixel;
    uint32_t endPixel;
};

extern "C" __global__ __aicore__ void conv2d_min_tanh_tanh_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dMinTanhTanh op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inHeight, tiling_data.inWidth,
            tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.kernelSize, tiling_data.totalOutPixels);
    op.Process();
}
