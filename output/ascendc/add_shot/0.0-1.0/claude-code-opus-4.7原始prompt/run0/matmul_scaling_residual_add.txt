kernel_src = """
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;

template <typename aType, typename bType, typename cType, typename biasType>
class KernelMatmulScalingResidualAdd {
public:
    __aicore__ inline KernelMatmulScalingResidualAdd() {}

    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR bias, GM_ADDR c,
                                const TCubeTiling &tiling, float scalingFactor, TPipe *pipe)
    {
        this->tiling = tiling;
        this->multiplier = 1.0f + scalingFactor;

        aGlobal.SetGlobalBuffer((__gm__ aType *)a);
        bGlobal.SetGlobalBuffer((__gm__ bType *)b);
        cGlobal.SetGlobalBuffer((__gm__ cType *)c);
        biasGlobal.SetGlobalBuffer((__gm__ biasType *)bias);

        int offsetA = 0, offsetB = 0, offsetC = 0, offsetBias = 0;
        CalcOffset(GetBlockIdx(), tiling, offsetA, offsetB, offsetC, offsetBias);
        aGlobal = aGlobal[offsetA];
        bGlobal = bGlobal[offsetB];
        cGlobal = cGlobal[offsetC];
        biasGlobal = biasGlobal[offsetBias];

        this->singleM = tiling.singleCoreM;
        this->singleN = tiling.singleCoreN;
        if (tiling.M - GetMCoreIndex(GetBlockIdx(), tiling) * tiling.singleCoreM < tiling.singleCoreM) {
            this->singleM = tiling.M - GetMCoreIndex(GetBlockIdx(), tiling) * tiling.singleCoreM;
        }
        if (tiling.N - GetNCoreIndex(GetBlockIdx(), tiling) * tiling.singleCoreN < tiling.singleCoreN) {
            this->singleN = tiling.N - GetNCoreIndex(GetBlockIdx(), tiling) * tiling.singleCoreN;
        }

        pipe->InitBuffer(outQueue, 1, tiling.baseM * tiling.baseN * sizeof(cType));
    }

    __aicore__ inline void Process(TPipe *pipe)
    {
        REGIST_MATMUL_OBJ(pipe, GetSysWorkSpacePtr(), matmulObj, &tiling);
        matmulObj.SetTensorA(aGlobal);
        matmulObj.SetTensorB(bGlobal, true);
        matmulObj.SetBias(biasGlobal);
        matmulObj.SetTail(this->singleM, this->singleN);

        uint32_t baseM = tiling.baseM;
        uint32_t baseN = tiling.baseN;
        uint32_t mIter = (this->singleM + baseM - 1) / baseM;
        uint32_t nIter = (this->singleN + baseN - 1) / baseN;

        for (uint32_t i = 0; i < mIter; ++i) {
            for (uint32_t j = 0; j < nIter; ++j) {
                if (!matmulObj.Iterate()) {
                    break;
                }
                LocalTensor<cType> outLocal = outQueue.AllocTensor<cType>();
                matmulObj.GetTensorC(outLocal, false, true);

                uint32_t curM = (i == mIter - 1) ? (this->singleM - i * baseM) : baseM;
                uint32_t curN = (j == nIter - 1) ? (this->singleN - j * baseN) : baseN;
                uint32_t count = curM * curN;

                Muls(outLocal, outLocal, (cType)this->multiplier, count);
                outQueue.EnQue(outLocal);

                LocalTensor<cType> toCopy = outQueue.DeQue<cType>();
                uint64_t cOffset = i * baseM * tiling.N + j * baseN;
                DataCopyExtParams copyParams;
                copyParams.blockCount = curM;
                copyParams.blockLen = curN * sizeof(cType);
                copyParams.srcStride = 0;
                copyParams.dstStride = (tiling.N - curN) * sizeof(cType);
                DataCopyPad(cGlobal[cOffset], toCopy, copyParams);
                outQueue.FreeTensor(toCopy);
            }
        }
        matmulObj.End();
    }

private:
    __aicore__ inline uint32_t GetMCoreIndex(int32_t blockIdx, const TCubeTiling &t)
    {
        uint32_t mSingleBlocks = (t.M + t.singleCoreM - 1) / t.singleCoreM;
        return blockIdx % mSingleBlocks;
    }
    __aicore__ inline uint32_t GetNCoreIndex(int32_t blockIdx, const TCubeTiling &t)
    {
        uint32_t mSingleBlocks = (t.M + t.singleCoreM - 1) / t.singleCoreM;
        return blockIdx / mSingleBlocks;
    }

    __aicore__ inline void CalcOffset(int32_t blockIdx, const TCubeTiling &t,
                                      int &offsetA, int &offsetB, int &offsetC, int &offsetBias)
    {
        uint32_t mCoreIndex = GetMCoreIndex(blockIdx, t);
        uint32_t nCoreIndex = GetNCoreIndex(blockIdx, t);
        offsetA = mCoreIndex * t.singleCoreM * t.Ka;
        offsetB = nCoreIndex * t.singleCoreN * t.Kb;
        offsetC = mCoreIndex * t.singleCoreM * t.N + nCoreIndex * t.singleCoreN;
        offsetBias = nCoreIndex * t.singleCoreN;
    }

    matmul::Matmul<matmul::MatmulType<TPosition::GM, CubeFormat::ND, aType>,
                   matmul::MatmulType<TPosition::GM, CubeFormat::ND, bType>,
                   matmul::MatmulType<TPosition::GM, CubeFormat::ND, cType>,
                   matmul::MatmulType<TPosition::GM, CubeFormat::ND, biasType>> matmulObj;

    GlobalTensor<aType> aGlobal;
    GlobalTensor<bType> bGlobal;
    GlobalTensor<cType> cGlobal;
    GlobalTensor<biasType> biasGlobal;
    TQue<TPosition::VECOUT, 1> outQueue;
    TCubeTiling tiling;
    float multiplier;
    uint32_t singleM;
    uint32_t singleN;
};

extern "C" __global__ __aicore__ void matmul_scaling_residual_add_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    TPipe pipe;
    KernelMatmulScalingResidualAdd<float, float, float, float> op;
    op.Init(x, weight, bias, y, tilingData.cubeTilingData, tilingData.scalingFactor, &pipe);
    op.Process(&pipe);
}
"""

at::Tensor matmul_scaling_residual_add_custom_impl_npu(
    const at::Tensor& x, const at::Tensor& weight, const at::Tensor& bias, double scaling_factor) {
    auto M = x.size(0);
    auto N = weight.size(0);
    at::Tensor result = at::empty({M, N}, x.options());
    float sf = static_cast<float>(scaling_factor);
    EXEC_NPU_CMD(aclnnMatmulScalingResidualAddCustom, x, weight, bias, sf, result);
    return result;
}

    m.impl("matmul_scaling_residual_add_custom", &matmul_scaling_residual_add_custom_impl_npu);

model_src = '''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, scaling_factor):
        super(ModelNew, self).__init__()
        self.matmul = nn.Linear(in_features, out_features)
        self.scaling_factor = float(scaling_factor)

    def forward(self, x):
        weight = self.matmul.weight.contiguous()
        bias = self.matmul.bias.contiguous()
        return custom_ops_lib.matmul_scaling_residual_add_custom(
            x.contiguous(), weight, bias, self.scaling_factor
        )
