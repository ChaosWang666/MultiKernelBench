kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmDivideSumScaling {
public:
    __aicore__ inline KernelGemmDivideSumScaling() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR v, GM_ADDR y,
                                 uint32_t rowsPerBlock, uint32_t innerSize, uint32_t tileNum)
    {
        this->rowsPerBlock = rowsPerBlock;
        this->innerSize = innerSize;
        this->tileNum = tileNum;
        this->tileLength = innerSize / tileNum;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x + AscendC::GetBlockIdx() * rowsPerBlock * innerSize,
                            rowsPerBlock * innerSize);
        vGm.SetGlobalBuffer((__gm__ DTYPE_V*)v, innerSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y + AscendC::GetBlockIdx() * rowsPerBlock,
                            rowsPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueV, BUFFER_NUM, this->tileLength * sizeof(DTYPE_V));
        pipe.InitBuffer(accumBuf, this->tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(tmpBuf, this->tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(reduceBuf, 32);
        uint32_t yBufSize = ((this->rowsPerBlock * sizeof(DTYPE_Y) + 31) / 32) * 32;
        pipe.InitBuffer(outQueueY, 1, yBufSize);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        for (uint32_t row = 0; row < rowsPerBlock; row++) {
            AscendC::LocalTensor<DTYPE_Y> accum = accumBuf.Get<DTYPE_Y>();
            AscendC::Duplicate<DTYPE_Y>(accum, (DTYPE_Y)0, this->tileLength);

            for (uint32_t tile = 0; tile < this->tileNum; tile++) {
                CopyIn(row, tile);
                Compute(accum);
            }

            AscendC::LocalTensor<DTYPE_Y> reduceDst = reduceBuf.Get<DTYPE_Y>();
            AscendC::LocalTensor<DTYPE_Y> reduceWork = tmpBuf.Get<DTYPE_Y>();
            AscendC::ReduceSum<DTYPE_Y>(reduceDst, accum, reduceWork, this->tileLength);
            DTYPE_Y result = reduceDst.GetValue(0);
            yLocal.SetValue(row, result);
        }

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        AscendC::LocalTensor<DTYPE_Y> yOut = outQueueY.DeQue<DTYPE_Y>();
        uint32_t outCount = ((this->rowsPerBlock + 7) / 8) * 8;
        AscendC::DataCopy(yGm, yOut, outCount);
        outQueueY.FreeTensor(yOut);
    }

private:
    __aicore__ inline void CopyIn(uint32_t row, uint32_t tile)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_V> vLocal = inQueueV.AllocTensor<DTYPE_V>();
        AscendC::DataCopy(xLocal, xGm[row * this->innerSize + tile * this->tileLength], this->tileLength);
        AscendC::DataCopy(vLocal, vGm[tile * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueV.EnQue(vLocal);
    }

    __aicore__ inline void Compute(AscendC::LocalTensor<DTYPE_Y> accum)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_V> vLocal = inQueueV.DeQue<DTYPE_V>();
        AscendC::LocalTensor<DTYPE_Y> tmp = tmpBuf.Get<DTYPE_Y>();
        AscendC::Mul(tmp, xLocal, vLocal, this->tileLength);
        AscendC::Add(accum, accum, tmp, this->tileLength);
        inQueueX.FreeTensor(xLocal);
        inQueueV.FreeTensor(vLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueV;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accumBuf, tmpBuf, reduceBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_V> vGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t rowsPerBlock;
    uint32_t innerSize;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void gemm_divide_sum_scaling_custom(GM_ADDR x, GM_ADDR v, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmDivideSumScaling op;
    op.Init(x, v, y, tiling_data.rowsPerBlock, tiling_data.innerSize, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, input_size, hidden_size, scaling_factor):
        super().__init__()
        self.weight = nn.Parameter(torch.randn(hidden_size, input_size))
        self.scaling_factor = scaling_factor
        self._v_cache = None

    def forward(self, x):
        if self._v_cache is None or self._v_cache.device != x.device:
            with torch.no_grad():
                self._v_cache = (self.weight.sum(dim=0) * (self.scaling_factor / 2.0)).contiguous()
        y = custom_ops_lib.gemm_divide_sum_scaling_custom(x, self._v_cache)
        return y.view(-1, 1)
