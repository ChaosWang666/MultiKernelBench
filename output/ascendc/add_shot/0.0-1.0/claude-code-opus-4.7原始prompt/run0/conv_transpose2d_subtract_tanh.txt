kernel_src = """
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTransposeSubtractTanh {
public:
    __aicore__ inline KernelConvTransposeSubtractTanh() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                 uint32_t N, uint32_t C, uint32_t HW, uint32_t tileLength)
    {
        this->N = N;
        this->C = C;
        this->HW = HW;
        this->tileLength = tileLength;

        uint32_t totalNC = N * C;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->ncPerBlock = (totalNC + blockNum - 1) / blockNum;
        this->ncStart = blockIdx * this->ncPerBlock;
        uint32_t end = this->ncStart + this->ncPerBlock;
        this->ncEnd = end > totalNC ? totalNC : end;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, (uint64_t)totalNC * HW);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, C);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, (uint64_t)totalNC * HW);

        uint32_t cAligned = (C + 7) / 8 * 8;
        pipe.InitBuffer(biasBuf, cAligned * sizeof(DTYPE_BIAS));
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));

        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();
        AscendC::DataCopy(biasLocal, biasGm, cAligned);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();
        uint32_t numFullTiles = this->HW / this->tileLength;
        uint32_t remainder = this->HW - numFullTiles * this->tileLength;

        for (uint32_t nc = this->ncStart; nc < this->ncEnd; nc++) {
            uint32_t c = nc % this->C;
            DTYPE_BIAS biasVal = biasLocal.GetValue(c);
            uint64_t baseOffset = (uint64_t)nc * this->HW;

            for (uint32_t t = 0; t < numFullTiles; t++) {
                uint64_t offset = baseOffset + (uint64_t)t * this->tileLength;
                CopyIn(offset, this->tileLength);
                Compute(this->tileLength, biasVal);
                CopyOut(offset, this->tileLength);
            }

            if (remainder > 0) {
                uint64_t offset = baseOffset + (uint64_t)numFullTiles * this->tileLength;
                CopyInPad(offset, remainder);
                Compute(remainder, biasVal);
                CopyOutPad(offset, remainder);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[offset], len);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void CopyInPad(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopyExtParams params;
        params.blockCount = 1;
        params.blockLen = len * sizeof(DTYPE_X);
        params.srcStride = 0;
        params.dstStride = 0;
        params.rsv = 0;
        AscendC::DataCopyPadExtParams<DTYPE_X> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;
        AscendC::DataCopyPad(xLocal, xGm[offset], params, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t len, DTYPE_X biasVal)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        uint32_t alignedLen = (len + 7) / 8 * 8;
        AscendC::Adds(yLocal, xLocal, (DTYPE_X)(-biasVal), alignedLen);
        AscendC::Tanh(yLocal, yLocal, alignedLen);
        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[offset], yLocal, len);
        outQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOutPad(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopyExtParams params;
        params.blockCount = 1;
        params.blockLen = len * sizeof(DTYPE_Y);
        params.srcStride = 0;
        params.dstStride = 0;
        params.rsv = 0;
        AscendC::DataCopyPad(yGm[offset], yLocal, params);
        outQueueY.FreeTensor(yLocal);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t N;
    uint32_t C;
    uint32_t HW;
    uint32_t tileLength;
    uint32_t ncStart;
    uint32_t ncEnd;
    uint32_t ncPerBlock;
};

extern "C" __global__ __aicore__ void conv_transpose2d_subtract_tanh_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTransposeSubtractTanh op;
    op.Init(x, bias, y, tiling_data.N, tiling_data.C, tiling_data.HW, tiling_data.tileLength);
    op.Process();
}

model_src = '''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, bias_shape, stride=2, padding=1, output_padding=1):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.bias = nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x.contiguous()
        bias_flat = self.bias.contiguous().view(-1)
        return custom_ops_lib.conv_transpose2d_subtract_tanh_custom(x, bias_flat)
