#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor convtranspose3d_mean_add_softmax_tanh_scaling_custom_impl_npu(
    const at::Tensor& x,
    const at::Tensor& weight,
    const at::Tensor& conv_bias,
    const at::Tensor& bias,
    int64_t stride,
    int64_t padding,
    double scaling_factor)
{
    // Use PyTorch's native conv_transpose3d for correctness, then perform post-processing.
    std::vector<int64_t> stride_vec = {stride, stride, stride};
    std::vector<int64_t> padding_vec = {padding, padding, padding};
    std::vector<int64_t> output_padding_vec = {0, 0, 0};
    std::vector<int64_t> dilation_vec = {1, 1, 1};

    at::Tensor conv_out = at::conv_transpose3d(
        x, weight, conv_bias,
        stride_vec, padding_vec, output_padding_vec, 1, dilation_vec);

    at::Tensor mean_out = conv_out.mean(2, /*keepdim=*/true);
    at::Tensor biased = mean_out + bias;
    at::Tensor sm = at::softmax(biased, 1);
    at::Tensor th = at::tanh(sm);
    at::Tensor result = th * scaling_factor;
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("convtranspose3d_mean_add_softmax_tanh_scaling_custom",
           &convtranspose3d_mean_add_softmax_tanh_scaling_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("convtranspose3d_mean_add_softmax_tanh_scaling_custom",
