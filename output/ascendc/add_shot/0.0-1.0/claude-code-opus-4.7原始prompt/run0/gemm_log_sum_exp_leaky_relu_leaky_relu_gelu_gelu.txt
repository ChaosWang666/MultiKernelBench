kernel_src="""
#include "kernel_operator.h"

class KernelGemmLSE {
public:
    __aicore__ inline KernelGemmLSE() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR z,
                                  uint32_t M_, uint32_t N_, uint32_t K_, uint32_t rpb)
    {
        this->M = M_;
        this->N = N_;
        this->K = K_;
        this->rowsPerBlock = rpb;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowStart = blockIdx * rpb;
        uint32_t rowsThisBlock = rpb;
        if (rowStart + rowsThisBlock > M) {
            rowsThisBlock = (rowStart < M) ? (M - rowStart) : 0;
        }
        this->rowsThisBlock = rowsThisBlock;

        xGm.SetGlobalBuffer((__gm__ float *)x + rowStart * K, rowsThisBlock * K);
        wGm.SetGlobalBuffer((__gm__ float *)w, N * K);
        bGm.SetGlobalBuffer((__gm__ float *)b, N);
        zGm.SetGlobalBuffer((__gm__ float *)z + rowStart, rowsThisBlock);

        pipe.InitBuffer(xBuf, K * sizeof(float));
        pipe.InitBuffer(wBuf, K * sizeof(float));
        pipe.InitBuffer(accBuf, N * sizeof(float));
        pipe.InitBuffer(bBuf, N * sizeof(float));
        pipe.InitBuffer(tmpBuf, K * sizeof(float));
        pipe.InitBuffer(scalarBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (rowsThisBlock == 0) {
            return;
        }

        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> accLocal = accBuf.Get<float>();
        AscendC::LocalTensor<float> bLocal = bBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();

        AscendC::DataCopy(bLocal, bGm, N);

        for (uint32_t i = 0; i < rowsThisBlock; i++) {
            AscendC::DataCopy(xLocal, xGm[i * K], K);

            for (uint32_t n = 0; n < N; n++) {
                AscendC::DataCopy(wLocal, wGm[n * K], K);
                AscendC::Mul(tmpLocal, xLocal, wLocal, K);
                AscendC::ReduceSum<float>(scalarLocal, tmpLocal, tmpLocal, K);
                AscendC::PipeBarrier<PIPE_V>();
                float dot = scalarLocal.GetValue(0);
                accLocal.SetValue(n, dot);
            }

            AscendC::PipeBarrier<PIPE_ALL>();

            AscendC::Add(accLocal, accLocal, bLocal, N);

            AscendC::ReduceMax<float>(scalarLocal, accLocal, tmpLocal, N);
            AscendC::PipeBarrier<PIPE_V>();
            float max_val = scalarLocal.GetValue(0);

            AscendC::Adds(tmpLocal, accLocal, -max_val, N);
            AscendC::Exp(tmpLocal, tmpLocal, N);
            AscendC::ReduceSum<float>(scalarLocal, tmpLocal, tmpLocal, N);
            AscendC::PipeBarrier<PIPE_V>();
            float sum_exp = scalarLocal.GetValue(0);

            for (int p = 0; p < 8; p++) {
                scalarLocal.SetValue(p, sum_exp);
            }
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Ln(scalarLocal, scalarLocal, 8);
            AscendC::PipeBarrier<PIPE_V>();
            float log_sum = scalarLocal.GetValue(0);
            float lse = max_val + log_sum;

            for (int p = 0; p < 8; p++) {
                scalarLocal.SetValue(p, lse);
            }
            AscendC::PipeBarrier<PIPE_ALL>();

            AscendC::LeakyRelu(scalarLocal, scalarLocal, (float)0.01f, 8);
            AscendC::LeakyRelu(scalarLocal, scalarLocal, (float)0.01f, 8);

            AscendC::Muls(tmpLocal, scalarLocal, (float)0.7071067811865475f, 8);
            AscendC::Erf(tmpLocal, tmpLocal, 8);
            AscendC::Adds(tmpLocal, tmpLocal, (float)1.0f, 8);
            AscendC::Mul(scalarLocal, scalarLocal, tmpLocal, 8);
            AscendC::Muls(scalarLocal, scalarLocal, (float)0.5f, 8);

            AscendC::Muls(tmpLocal, scalarLocal, (float)0.7071067811865475f, 8);
            AscendC::Erf(tmpLocal, tmpLocal, 8);
            AscendC::Adds(tmpLocal, tmpLocal, (float)1.0f, 8);
            AscendC::Mul(scalarLocal, scalarLocal, tmpLocal, 8);
            AscendC::Muls(scalarLocal, scalarLocal, (float)0.5f, 8);

            AscendC::PipeBarrier<PIPE_V>();
            float result = scalarLocal.GetValue(0);
            zGm.SetValue(i, result);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t M;
    uint32_t N;
    uint32_t K;
    uint32_t rowsPerBlock;
    uint32_t rowsThisBlock;
};

extern "C" __global__ __aicore__ void gemm_log_sum_exp_leaky_relu_leaky_relu_gelu_gelu_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmLSE op;
    op.Init(x, w, b, z, tiling_data.M, tiling_data.N, tiling_data.K, tiling_data.rowsPerBlock);
    op.Process();
}
