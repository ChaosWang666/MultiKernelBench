#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedConvTransposeOp {
public:
    __aicore__ inline KernelFusedConvTransposeOp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                uint32_t totalLength, uint32_t spatialLength,
                                uint32_t channels, uint32_t tileLength,
                                uint32_t chunksPerBlock)
    {
        this->totalLength = totalLength;
        this->spatialLength = spatialLength;
        this->channels = channels;
        this->tileLength = tileLength;
        this->chunksPerBlock = chunksPerBlock;
        this->numTilesPerChunk = spatialLength / tileLength;
        this->tailLength = spatialLength - this->numTilesPerChunk * tileLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startChunk = blockIdx * chunksPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, totalLength);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, channels);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, totalLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(tmpBuf, this->tileLength * sizeof(DTYPE_X));

        // Pad channels up to 32-byte alignment for safe DataCopy (8 floats = 32 bytes)
        uint32_t alignedChannels = (channels + 7) / 8 * 8;
        pipe.InitBuffer(biasBuf, alignedChannels * sizeof(DTYPE_BIAS));
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();
        AscendC::DataCopy(biasLocal, biasGm, alignedChannels);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();
        for (uint32_t c = 0; c < this->chunksPerBlock; c++) {
            uint32_t chunkIdx = this->startChunk + c;
            uint32_t channelIdx = chunkIdx % this->channels;
            DTYPE_X biasVal = biasLocal.GetValue(channelIdx);
            DTYPE_X scalarConst = biasVal + (DTYPE_X)1.0f;
            uint64_t baseOffset = (uint64_t)chunkIdx * this->spatialLength;
            for (uint32_t t = 0; t < this->numTilesPerChunk; t++) {
                uint64_t offset = baseOffset + (uint64_t)t * this->tileLength;
                CopyIn(offset, this->tileLength);
                Compute(scalarConst, this->tileLength);
                CopyOut(offset, this->tileLength);
            }
            if (this->tailLength > 0) {
                uint64_t offset = baseOffset + (uint64_t)this->numTilesPerChunk * this->tileLength;
                CopyIn(offset, this->tailLength);
                Compute(scalarConst, this->tailLength);
                CopyOut(offset, this->tailLength);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint64_t offset, uint32_t length)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[offset], length);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(DTYPE_X scalarConst, uint32_t length)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_X> tmp = tmpBuf.Get<DTYPE_X>();
        // out = x * (2*x + bias + 1)
        AscendC::Muls(tmp, xLocal, (DTYPE_X)2.0f, length);
        AscendC::Adds(tmp, tmp, scalarConst, length);
        AscendC::Mul(yLocal, xLocal, tmp, length);
        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint64_t offset, uint32_t length)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[offset], yLocal, length);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t totalLength;
    uint32_t spatialLength;
    uint32_t channels;
    uint32_t tileLength;
    uint32_t chunksPerBlock;
    uint32_t startChunk;
    uint32_t numTilesPerChunk;
    uint32_t tailLength;
};

extern "C" __global__ __aicore__ void conv_transpose3d_sum_residual_add_multiply_residual_add_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedConvTransposeOp op;
    op.Init(x, bias, y, tiling_data.totalLength, tiling_data.spatialLength,
            tiling_data.channels, tiling_data.tileLength, tiling_data.chunksPerBlock);
    op.Process();
}

import torch
import torch.nn as nn

import custom_ops_lib

class ModelNew(nn.Module):

        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)

        x = self.conv_transpose(x)
        x = x.contiguous()
        bias_flat = self.bias.view(-1).contiguous()
