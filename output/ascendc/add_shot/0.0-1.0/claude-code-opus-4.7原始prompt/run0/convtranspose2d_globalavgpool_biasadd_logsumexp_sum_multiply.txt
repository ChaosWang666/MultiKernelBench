kernel_src="""
#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                uint32_t batchSize, uint32_t channels,
                                uint32_t spatialSize, uint32_t tileSize)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->tileSize = tileSize;
        this->invSpatial = 1.0f / (float)spatialSize;
        this->blockIdx = AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ float *)x);
        biasGm.SetGlobalBuffer((__gm__ float *)bias);
        yGm.SetGlobalBuffer((__gm__ float *)y);

        pipe.InitBuffer(inQueueX, 2, tileSize * sizeof(float));
        pipe.InitBuffer(biasBuf, 256 * sizeof(float));
        pipe.InitBuffer(valBuf, 256 * sizeof(float));
        pipe.InitBuffer(tmpBuf, 256 * sizeof(float));
        pipe.InitBuffer(redBuf, 64 * sizeof(float));
        pipe.InitBuffer(outBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (blockIdx >= batchSize) return;

        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::DataCopy(biasLocal, biasGm, channels);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);

        AscendC::LocalTensor<float> valLocal = valBuf.Get<float>();
        AscendC::LocalTensor<float> red = redBuf.Get<float>();

        for (uint32_t c = 0; c < channels; c++) {
            uint64_t channelBase = ((uint64_t)blockIdx * channels + c) * spatialSize;
            float channelSum = 0.0f;
            uint32_t pos = 0;

            while (pos < spatialSize) {
                uint32_t currentTile = (spatialSize - pos) < tileSize ? (spatialSize - pos) : tileSize;
                uint32_t alignedTile = (currentTile + 63) / 64 * 64;

                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopyExtParams copyParams{1, (uint32_t)(currentTile * sizeof(float)), 0, 0, 0};
                AscendC::DataCopyPadExtParams<float> padParams{true, 0, (uint8_t)(alignedTile - currentTile), 0.0f};
                AscendC::DataCopyPad(xLocal, xGm[channelBase + pos], copyParams, padParams);
                inQueueX.EnQue(xLocal);

                xLocal = inQueueX.DeQue<float>();
                AscendC::ReduceSum(red, xLocal, red, alignedTile);
                AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                channelSum += red.GetValue(0);
                inQueueX.FreeTensor(xLocal);

                pos += currentTile;
            }

            float biasVal = biasLocal.GetValue(c);
            valLocal.SetValue(c, channelSum * invSpatial + biasVal);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        AscendC::ReduceMax(red, valLocal, red, channels, false);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
        float maxVal = red.GetValue(0);

        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::Adds(tmp, valLocal, -maxVal, channels);
        AscendC::Exp(tmp, tmp, channels);
        AscendC::ReduceSum(red, tmp, red, channels);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
        float sumExp = red.GetValue(0);

        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        AscendC::Duplicate(outLocal, 1.0f, 64);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID3);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID3);
        outLocal.SetValue(0, sumExp);
        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID1);
        AscendC::Ln(outLocal, outLocal, 64);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float logSumExp = outLocal.GetValue(0);

        float result = (maxVal + logSumExp) * 10.0f;
        outLocal.SetValue(0, result);
        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        AscendC::DataCopyExtParams outParams{1, (uint32_t)sizeof(float), 0, 0, 0};
        AscendC::DataCopyPad(yGm[blockIdx], outLocal, outParams);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf, valBuf, tmpBuf, redBuf, outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t tileSize;
    uint32_t blockIdx;
    float invSpatial;
};

extern "C" __global__ __aicore__ void convtranspose2d_globalavgpool_biasadd_logsumexp_sum_multiply_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, y, tiling_data.batchSize, tiling_data.channels,
            tiling_data.spatialSize, tiling_data.tileSize);
    op.Process();
}
