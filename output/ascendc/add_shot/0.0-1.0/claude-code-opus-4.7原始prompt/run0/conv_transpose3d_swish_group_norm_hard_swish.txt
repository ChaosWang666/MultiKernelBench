kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t TILE_SIZE = 2048;

class KernelHardSwish {
public:
    __aicore__ inline KernelHardSwish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalLength)
    {
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t perBlock = totalLength / blockNum;
        uint32_t remainder = totalLength % blockNum;

        uint32_t startOffset;
        if (blockIdx < remainder) {
            this->myLength = perBlock + 1;
            startOffset = blockIdx * this->myLength;
        } else {
            this->myLength = perBlock;
            startOffset = remainder * (perBlock + 1) + (blockIdx - remainder) * perBlock;
        }

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startOffset, this->myLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + startOffset, this->myLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, TILE_SIZE * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, TILE_SIZE * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpBuf, TILE_SIZE * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        uint32_t offset = 0;
        while (offset < this->myLength) {
            uint32_t remaining = this->myLength - offset;
            uint32_t curLen = (TILE_SIZE < remaining) ? TILE_SIZE : remaining;
            CopyIn(offset, curLen);
            Compute(curLen);
            CopyOut(offset, curLen);
            offset += curLen;
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[offset], len);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_X> tmp = tmpBuf.Get<DTYPE_X>();

        // HardSwish: z = x * clamp(x + 3, 0, 6) / 6
        AscendC::Adds(tmp, xLocal, (DTYPE_X)3.0f, len);
        AscendC::Maxs(tmp, tmp, (DTYPE_X)0.0f, len);
        AscendC::Mins(tmp, tmp, (DTYPE_X)6.0f, len);
        AscendC::Muls(tmp, tmp, (DTYPE_X)(1.0f / 6.0f), len);
        AscendC::Mul(zLocal, xLocal, tmp, len);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[offset], zLocal, len);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t myLength;
};

extern "C" __global__ __aicore__ void conv_transpose3d_swish_group_norm_hard_swish_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelHardSwish op;
    op.Init(x, z, tiling_data.totalLength);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, groups, eps, bias=True):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, bias=bias)
        self.group_norm = nn.GroupNorm(num_groups=groups, num_channels=out_channels, eps=eps)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = torch.sigmoid(x) * x
        x = self.group_norm(x)
        x = x.contiguous()
        x = custom_ops_lib.conv_transpose3d_swish_group_norm_hard_swish_custom(x)
        return x
