kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;
constexpr float EPS = 1e-5f;

class KernelBmmInstanceNormSumResidualAddMultiply {
public:
    __aicore__ inline KernelBmmInstanceNormSumResidualAddMultiply() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z, uint32_t batchSize, uint32_t featureSize)
    {
        this->featureSize = featureSize;
        uint32_t blockNum = AscendC::GetBlockNum();
        this->batchPerBlock = batchSize / blockNum;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t offset = blockIdx * this->batchPerBlock * featureSize;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + offset, this->batchPerBlock * featureSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + offset, this->batchPerBlock * featureSize);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + offset, this->batchPerBlock * featureSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, featureSize * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueY, BUFFER_NUM, featureSize * sizeof(DTYPE_Y));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, featureSize * sizeof(DTYPE_Z));
        pipe.InitBuffer(normBuf, featureSize * sizeof(float));
        pipe.InitBuffer(workBuf, featureSize * sizeof(float));
        pipe.InitBuffer(reduceBuf, 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->batchPerBlock; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.AllocTensor<DTYPE_Y>();
        AscendC::DataCopy(xLocal, xGm[progress * this->featureSize], this->featureSize);
        AscendC::DataCopy(yLocal, yGm[progress * this->featureSize], this->featureSize);
        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> normed = normBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduceOut = reduceBuf.Get<float>();

        float invN = 1.0f / (float)this->featureSize;

        AscendC::ReduceSum<float>(reduceOut, xLocal, work, this->featureSize);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float mean = reduceOut.GetValue(0) * invN;
        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        AscendC::Adds(normed, xLocal, -mean, this->featureSize);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(work, normed, normed, this->featureSize);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::ReduceSum<float>(reduceOut, work, work, this->featureSize);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float var = reduceOut.GetValue(0) * invN;
        float invStd = 1.0f / sqrt(var + EPS);
        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        AscendC::Muls(normed, normed, invStd, this->featureSize);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(normed, normed, yLocal, this->featureSize);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(zLocal, normed, yLocal, this->featureSize);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * this->featureSize], zLocal, this->featureSize);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> normBuf, workBuf, reduceBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t featureSize;
    uint32_t batchPerBlock;
};

extern "C" __global__ __aicore__ void bmm_instance_norm_sum_residual_add_multiply_custom(GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelBmmInstanceNormSumResidualAddMultiply op;
    op.Init(x, y, z, tiling_data.batchSize, tiling_data.featureSize);
    op.Process();
}
