import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    def __init__(self, in_features, out_features, scale_shape, eps=1e-5, momentum=0.1):
        super(Model, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.scale = nn.Parameter(torch.rand(scale_shape))
        self.bn = nn.BatchNorm1d(out_features, eps=eps, momentum=momentum)

    def forward(self, x):
        x = self.gemm(x)
        x = x * self.scale
        x = self.bn(x)
        return x

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmScaleBN {
public:
    __aicore__ inline KernelGemmScaleBN() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR y,
                                uint32_t totalLength, uint32_t featureSize,
                                uint32_t rowsPerBlock)
    {
        this->featureSize = featureSize;
        this->rowsPerBlock = rowsPerBlock;
        this->tileLength = featureSize;

        uint32_t blockOffset = rowsPerBlock * featureSize * AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockOffset, rowsPerBlock * featureSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + blockOffset, rowsPerBlock * featureSize);
        scaleGm.SetGlobalBuffer((__gm__ DTYPE_SCALE *)scale, featureSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(scaleBuf, this->featureSize * sizeof(DTYPE_SCALE));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_SCALE> scaleLocal = scaleBuf.Get<DTYPE_SCALE>();
        AscendC::DataCopy(scaleLocal, scaleGm, this->featureSize);

        for (uint32_t i = 0; i < this->rowsPerBlock; i++) {
            CopyIn(i);
            Compute(i, scaleLocal);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(int32_t progress, AscendC::LocalTensor<DTYPE_SCALE>& scaleLocal)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::Mul(yLocal, xLocal, scaleLocal, this->tileLength);
        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scaleBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_SCALE> scaleGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t featureSize;
    uint32_t rowsPerBlock;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void gemm_scale_batchnorm_custom(GM_ADDR x, GM_ADDR scale, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmScaleBN op;
    op.Init(x, scale, y, tiling_data.totalLength, tiling_data.featureSize, tiling_data.rowsPerBlock);
    op.Process();
}
