#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose2dMaxPoolHardtanhMeanTanh {
public:
    __aicore__ inline KernelConvTranspose2dMaxPoolHardtanhMeanTanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalPairs, uint32_t H, uint32_t W)
    {
        this->H = H;
        this->W = W;
        this->pooledH = H / 2;
        this->pooledW = W / 2;
        this->twoRows = 2 * W;
        this->poolCount = this->pooledH * this->pooledW;
        this->invPoolCount = 1.0f / (float)this->poolCount;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t pairsPerBlock = (totalPairs + blockNum - 1) / blockNum;
        this->startPair = blockIdx * pairsPerBlock;
        this->endPair = this->startPair + pairsPerBlock;
        if (this->endPair > totalPairs) this->endPair = totalPairs;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, (uint64_t)totalPairs * H * W);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, totalPairs);

        pipe.InitBuffer(inQueue, BUFFER_NUM, this->twoRows * sizeof(DTYPE_X));
        pipe.InitBuffer(scalarBuf, 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t p = this->startPair; p < this->endPair; p++) {
            ProcessPair(p);
        }
    }

private:
    __aicore__ inline void ProcessPair(uint32_t pairIdx) {
        uint64_t baseOffset = (uint64_t)pairIdx * this->H * this->W;
        float totalSum = 0.0f;

        for (uint32_t i = 0; i < this->pooledH; i++) {
            AscendC::LocalTensor<DTYPE_X> rows = inQueue.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(rows, xGm[baseOffset + (uint64_t)i * 2 * this->W], this->twoRows);
            inQueue.EnQue(rows);
            AscendC::LocalTensor<DTYPE_X> r = inQueue.DeQue<DTYPE_X>();

            for (uint32_t j = 0; j < this->pooledW; j++) {
                float v0 = (float)r.GetValue(2 * j);
                float v1 = (float)r.GetValue(2 * j + 1);
                float v2 = (float)r.GetValue(this->W + 2 * j);
                float v3 = (float)r.GetValue(this->W + 2 * j + 1);
                float m = v0;
                if (v1 > m) m = v1;
                if (v2 > m) m = v2;
                if (v3 > m) m = v3;
                if (m > 1.0f) m = 1.0f;
                if (m < -1.0f) m = -1.0f;
                totalSum += m;
            }

            inQueue.FreeTensor(r);
        }

        float mean = totalSum * this->invPoolCount;

        AscendC::LocalTensor<float> sb = scalarBuf.Get<float>();
        for (int k = 0; k < 8; k++) {
            sb.SetValue(k, mean);
        }
        AscendC::Tanh(sb, sb, 8);
        float tanhVal = sb.GetValue(0);

        yGm.SetValue(pairIdx, tanhVal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t H;
    uint32_t W;
    uint32_t pooledH;
    uint32_t pooledW;
    uint32_t twoRows;
    uint32_t poolCount;
    float invPoolCount;
    uint32_t startPair;
    uint32_t endPair;
};

extern "C" __global__ __aicore__ void conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);

}
