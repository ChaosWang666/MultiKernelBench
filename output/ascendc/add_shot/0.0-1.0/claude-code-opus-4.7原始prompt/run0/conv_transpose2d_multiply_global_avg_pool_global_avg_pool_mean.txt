kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMulMean {
public:
    __aicore__ inline KernelMulMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                 uint32_t totalChannels,
                                 uint32_t spatialSize,
                                 uint32_t channelsPerBlock,
                                 uint32_t tileLength,
                                 float scale)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startCh = blockIdx * channelsPerBlock;
        uint32_t actualChannels = 0;
        if (startCh < totalChannels) {
            actualChannels = (startCh + channelsPerBlock > totalChannels)
                             ? (totalChannels - startCh)
                             : channelsPerBlock;
        }
        this->startCh = startCh;
        this->actualChannels = actualChannels;
        this->channelsPerBlock = channelsPerBlock;
        this->spatialSize = spatialSize;
        this->tileLength = tileLength;
        this->scale = scale;

        xGm.SetGlobalBuffer((__gm__ float*)x + (uint64_t)startCh * spatialSize,
                            (uint64_t)channelsPerBlock * spatialSize);
        zGm.SetGlobalBuffer((__gm__ float*)z + startCh, channelsPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(workBuf, tileLength * sizeof(float));
        pipe.InitBuffer(outQueue, 1, channelsPerBlock * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::Duplicate(outLocal, (float)0.0f, this->channelsPerBlock);

        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();

        for (uint32_t ch = 0; ch < this->actualChannels; ch++) {
            uint64_t baseOffset = (uint64_t)ch * this->spatialSize;
            uint32_t numTiles = this->spatialSize / this->tileLength;
            uint32_t remainder = this->spatialSize - numTiles * this->tileLength;
            float acc = 0.0f;

            for (uint32_t t = 0; t < numTiles; t++) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[baseOffset + t * this->tileLength], this->tileLength);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();
                AscendC::ReduceSum<float>(workLocal, xLocal, workLocal, this->tileLength);
                AscendC::PipeBarrier<PIPE_V>();
                acc += workLocal.GetValue(0);
                inQueueX.FreeTensor(xLocal);
            }

            if (remainder > 0) {
                uint32_t alignedLen = ((remainder + 7) / 8) * 8;
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[baseOffset + numTiles * this->tileLength], alignedLen);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();
                for (uint32_t i = remainder; i < alignedLen; i++) {
                    xLocal.SetValue(i, (float)0.0f);
                }
                AscendC::ReduceSum<float>(workLocal, xLocal, workLocal, alignedLen);
                AscendC::PipeBarrier<PIPE_V>();
                acc += workLocal.GetValue(0);
                inQueueX.FreeTensor(xLocal);
            }

            outLocal.SetValue(ch, acc * this->scale);
        }

        outQueue.EnQue(outLocal);
        outLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(zGm, outLocal, this->channelsPerBlock);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t startCh;
    uint32_t actualChannels;
    uint32_t channelsPerBlock;
    uint32_t spatialSize;
    uint32_t tileLength;
    float scale;
};

extern "C" __global__ __aicore__ void conv_transpose2d_multiply_global_avg_pool_global_avg_pool_mean_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMulMean op;
    op.Init(x, z,
            tiling_data.totalChannels,
            tiling_data.spatialSize,
            tiling_data.channelsPerBlock,
            tiling_data.tileLength,
            tiling_data.scale);
    op.Process();
}
