kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvOp {
public:
    __aicore__ inline KernelConvOp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                  uint32_t N_, uint32_t C_, uint32_t S_, uint32_t tileSize_)
    {
        this->N = N_;
        this->C = C_;
        this->S = S_;
        this->tileSize = tileSize_;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t batchesPerBlock = (this->N + blockNum - 1) / blockNum;
        this->n_start = blockIdx * batchesPerBlock;
        uint32_t n_end_tmp = this->n_start + batchesPerBlock;
        this->n_end = (n_end_tmp < this->N) ? n_end_tmp : this->N;

        xGm.SetGlobalBuffer((__gm__ float *)x, (uint64_t)this->N * this->C * this->S);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, 1);
        yGm.SetGlobalBuffer((__gm__ float *)y, (uint64_t)this->N * this->S);

        pipe.InitBuffer(inQue, BUFFER_NUM, this->C * this->tileSize * sizeof(float));
        pipe.InitBuffer(outQue, BUFFER_NUM, this->tileSize * sizeof(float));
        pipe.InitBuffer(maxBuf, this->tileSize * sizeof(float));
        pipe.InitBuffer(sumBuf, this->tileSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->tileSize * sizeof(float));
        pipe.InitBuffer(biasBuf, 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::DataCopyExtParams biasParams;
        biasParams.blockCount = 1;
        biasParams.blockLen = sizeof(float);
        biasParams.srcStride = 0;
        biasParams.dstStride = 0;
        AscendC::DataCopyPadExtParams<float> biasPadParams;
        biasPadParams.isPad = true;
        biasPadParams.leftPadding = 0;
        biasPadParams.rightPadding = 7;
        biasPadParams.paddingValue = 0;
        AscendC::DataCopyPad(biasLocal, biasGm, biasParams, biasPadParams);
        AscendC::PipeBarrier<PIPE_ALL>();
        float biasVal = biasLocal.GetValue(0);

        for (uint32_t n = this->n_start; n < this->n_end; n++) {
            uint32_t numFullTiles = this->S / this->tileSize;
            for (uint32_t t = 0; t < numFullTiles; t++) {
                ProcessTile(n, t * this->tileSize, this->tileSize, biasVal);
            }
            uint32_t tailStart = numFullTiles * this->tileSize;
            uint32_t tailSize = this->S - tailStart;
            if (tailSize > 0) {
                ProcessTile(n, tailStart, tailSize, biasVal);
            }
        }
    }

private:
    __aicore__ inline void ProcessTile(uint32_t n, uint32_t s_start, uint32_t curTileSize, float biasVal)
    {
        AscendC::LocalTensor<float> dataLocal = inQue.AllocTensor<float>();

        for (uint32_t c = 0; c < this->C; c++) {
            uint64_t srcOffset = (uint64_t)n * this->C * this->S + (uint64_t)c * this->S + s_start;
            AscendC::DataCopyExtParams params;
            params.blockCount = 1;
            params.blockLen = curTileSize * sizeof(float);
            params.srcStride = 0;
            params.dstStride = 0;
            AscendC::DataCopyPadExtParams<float> padParams;
            padParams.isPad = true;
            padParams.leftPadding = 0;
            padParams.rightPadding = 0;
            padParams.paddingValue = 0;
            AscendC::DataCopyPad(dataLocal[c * this->tileSize], xGm[srcOffset], params, padParams);
        }
        inQue.EnQue(dataLocal);

        AscendC::LocalTensor<float> data = inQue.DeQue<float>();
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        AscendC::DataCopy(maxLocal, data, this->tileSize);
        for (uint32_t c = 1; c < this->C; c++) {
            AscendC::Max(maxLocal, maxLocal, data[c * this->tileSize], curTileSize);
        }

        AscendC::Duplicate(sumLocal, 0.0f, this->tileSize);
        for (uint32_t c = 0; c < this->C; c++) {
            AscendC::Sub(tmpLocal, data[c * this->tileSize], maxLocal, curTileSize);
            AscendC::Exp(tmpLocal, tmpLocal, curTileSize);
            AscendC::Add(sumLocal, sumLocal, tmpLocal, curTileSize);
        }

        inQue.FreeTensor(data);

        AscendC::Ln(sumLocal, sumLocal, curTileSize);
        AscendC::LocalTensor<float> outLocal = outQue.AllocTensor<float>();
        AscendC::Add(outLocal, sumLocal, maxLocal, curTileSize);

        AscendC::Adds(tmpLocal, outLocal, 3.0f, curTileSize);
        AscendC::Muls(tmpLocal, tmpLocal, -1.0f, curTileSize);
        AscendC::Exp(tmpLocal, tmpLocal, curTileSize);
        AscendC::Adds(tmpLocal, tmpLocal, 1.0f, curTileSize);
        AscendC::Muls(tmpLocal, tmpLocal, 6.0f, curTileSize);
        AscendC::Div(outLocal, outLocal, tmpLocal, curTileSize);

        AscendC::Adds(outLocal, outLocal, -biasVal, curTileSize);

        AscendC::Mins(outLocal, outLocal, 1.0f, curTileSize);
        AscendC::Maxs(outLocal, outLocal, -1.0f, curTileSize);

        outQue.EnQue(outLocal);

        AscendC::LocalTensor<float> out = outQue.DeQue<float>();
        AscendC::DataCopyExtParams outParams;
        outParams.blockCount = 1;
        outParams.blockLen = curTileSize * sizeof(float);
        outParams.srcStride = 0;
        outParams.dstStride = 0;
        AscendC::DataCopyPad(yGm[(uint64_t)n * this->S + s_start], out, outParams);
        outQue.FreeTensor(out);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t N;
    uint32_t C;
    uint32_t S;
    uint32_t tileSize;
    uint32_t n_start;
    uint32_t n_end;
};

extern "C" __global__ __aicore__ void conv_transpose3d_log_sum_exp_hard_swish_subtract_clamp_max_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvOp op;
    op.Init(x, bias, y, tiling_data.N, tiling_data.C, tiling_data.S, tiling_data.tileSize);
    op.Process();
}
