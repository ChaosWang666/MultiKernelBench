kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose3dClampMinDivide {
public:
    __aicore__ inline KernelConvTranspose3dClampMinDivide() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalLength, uint32_t tileNum,
                                float minValue, float divisor)
    {
        this->totalLength = totalLength;
        this->blockNum = AscendC::GetBlockNum();
        this->blockIdx = AscendC::GetBlockIdx();
        this->minValue = minValue;
        this->scale = 1.0f / divisor;

        uint32_t baseLen = totalLength / this->blockNum;
        uint32_t remainder = totalLength % this->blockNum;
        uint32_t offset;
        if (this->blockIdx < remainder) {
            this->blockLength = baseLen + 1;
            offset = this->blockIdx * this->blockLength;
        } else {
            this->blockLength = baseLen;
            offset = remainder * (baseLen + 1) + (this->blockIdx - remainder) * baseLen;
        }

        this->tileNum = tileNum;
        if (this->tileNum == 0) {
            this->tileNum = 1;
        }
        if (this->blockLength == 0) {
            this->tileLength = 0;
            this->tailLength = 0;
            this->loopCount = 0;
            return;
        }
        uint32_t rawTile = this->blockLength / (this->tileNum * BUFFER_NUM);
        if (rawTile == 0) {
            rawTile = this->blockLength;
            this->tileNum = 1;
            this->loopCount = 1;
        } else {
            this->loopCount = this->tileNum * BUFFER_NUM;
        }
        this->tileLength = rawTile;
        uint32_t consumed = this->tileLength * this->loopCount;
        this->tailLength = this->blockLength - consumed;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + offset, this->blockLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + offset, this->blockLength);

        uint32_t bufLen = this->tileLength + this->tailLength;
        pipe.InitBuffer(inQueueX, BUFFER_NUM, bufLen * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, bufLen * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        if (this->blockLength == 0) {
            return;
        }
        for (int32_t i = 0; i < this->loopCount; i++) {
            uint32_t curLen = this->tileLength;
            if (i == this->loopCount - 1) {
                curLen += this->tailLength;
            }
            CopyIn(i, curLen);
            Compute(i, curLen);
            CopyOut(i, curLen);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], curLen);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t progress, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::Maxs(zLocal, xLocal, (DTYPE_X)this->minValue, curLen);
        AscendC::Muls(zLocal, zLocal, (DTYPE_Z)this->scale, curLen);
        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, curLen);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t totalLength;
    uint32_t blockNum;
    uint32_t blockIdx;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t tailLength;
    int32_t loopCount;
    float minValue;
    float scale;
};

extern "C" __global__ __aicore__ void conv_transpose3d_clamp_min_divide_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dClampMinDivide op;
    op.Init(x, z, tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.minValue, tiling_data.divisor);
    op.Process();
}

class ModelNew(nn.Module):

        self.divisor = float(divisor)

        x = self.conv_transpose(x)
        x = x.contiguous()
