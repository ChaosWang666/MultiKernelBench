kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedActBias {
public:
    __aicore__ inline KernelFusedActBias() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                 uint32_t slicesPerBlock, uint32_t sliceSize,
                                 uint32_t tileLength, uint32_t tilesPerSlice,
                                 uint32_t numChannels)
    {
        this->slicesPerBlock = slicesPerBlock;
        this->sliceSize = sliceSize;
        this->tileLength = tileLength;
        this->tilesPerSlice = tilesPerSlice;
        this->numChannels = numChannels;

        uint32_t blockOffset = slicesPerBlock * sliceSize * AscendC::GetBlockIdx();
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockOffset, slicesPerBlock * sliceSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + blockOffset, slicesPerBlock * sliceSize);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, numChannels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(inQueueBias, 1, numChannels * sizeof(DTYPE_BIAS));
        pipe.InitBuffer(tmpBuf, tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_BIAS> biasAlloc = inQueueBias.AllocTensor<DTYPE_BIAS>();
        AscendC::DataCopy(biasAlloc, biasGm, this->numChannels);
        inQueueBias.EnQue(biasAlloc);
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = inQueueBias.DeQue<DTYPE_BIAS>();

        for (uint32_t s = 0; s < this->slicesPerBlock; s++) {
            uint32_t channel = s % this->numChannels;
            float biasVal = (float)biasLocal.GetValue(channel);
            for (uint32_t t = 0; t < this->tilesPerSlice; t++) {
                uint32_t offset = s * this->sliceSize + t * this->tileLength;
                CopyIn(offset);
                Compute(biasVal);
                CopyOut(offset);
            }
        }
        inQueueBias.FreeTensor(biasLocal);
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[offset], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float biasVal)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        // ReLU (LeakyReLU with slope 0.01 is identity after ReLU since values are >= 0)
        AscendC::Maxs(yLocal, xLocal, (DTYPE_X)0.0f, this->tileLength);
        inQueueX.FreeTensor(xLocal);

        // GELU via tanh approximation: 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
        AscendC::Mul(tmp, yLocal, yLocal, this->tileLength);
        AscendC::Mul(tmp, tmp, yLocal, this->tileLength);
        AscendC::Muls(tmp, tmp, 0.044715f, this->tileLength);
        AscendC::Add(tmp, tmp, yLocal, this->tileLength);
        AscendC::Muls(tmp, tmp, 0.7978845608f, this->tileLength);
        AscendC::Tanh(tmp, tmp, this->tileLength);
        AscendC::Adds(tmp, tmp, 1.0f, this->tileLength);
        AscendC::Mul(yLocal, yLocal, tmp, this->tileLength);
        AscendC::Muls(yLocal, yLocal, 0.5f, this->tileLength);

        // Sigmoid: 1 / (1 + exp(-x))
        AscendC::Muls(tmp, yLocal, -1.0f, this->tileLength);
        AscendC::Exp(tmp, tmp, this->tileLength);
        AscendC::Adds(tmp, tmp, 1.0f, this->tileLength);
        AscendC::Duplicate(yLocal, 1.0f, this->tileLength);
        AscendC::Div(yLocal, yLocal, tmp, this->tileLength);

        // Add bias (scalar per channel)
        AscendC::Adds(yLocal, yLocal, (DTYPE_Y)biasVal, this->tileLength);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[offset], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueBias;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    uint32_t slicesPerBlock;
    uint32_t sliceSize;
    uint32_t tileLength;
    uint32_t tilesPerSlice;
    uint32_t numChannels;
};

extern "C" __global__ __aicore__ void conv3d_relu_leaky_relu_gelu_sigmoid_bias_add_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedActBias op;
    op.Init(x, bias, y,
            tiling_data.slicesPerBlock, tiling_data.sliceSize,
            tiling_data.tileLength, tiling_data.tilesPerSlice,
            tiling_data.numChannels);
    op.Process();
}
