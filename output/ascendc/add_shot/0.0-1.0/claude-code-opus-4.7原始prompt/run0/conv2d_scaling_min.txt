kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv2dScalingMin {
public:
    __aicore__ inline KernelConv2dScalingMin() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batchSize, uint32_t channels,
                                uint32_t height, uint32_t width, float scaleFactor)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->height = height;
        this->width = width;
        this->scaleFactor = scaleFactor;
        this->spatialSize = height * width;

        // Pad width to a multiple of 8 floats (32 bytes) for UB allocation.
        this->alignedWidth = ((width + 7) / 8) * 8;

        uint32_t totalRows = batchSize * height;
        this->totalRows = totalRows;
        this->rowsPerBlock = (totalRows + AscendC::GetBlockNum() - 1) / AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, batchSize * channels * height * width);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, batchSize * height * width);

        pipe.InitBuffer(inQueue, BUFFER_NUM, this->alignedWidth * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->alignedWidth * sizeof(float));
        pipe.InitBuffer(accBuf, this->alignedWidth * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t startRow = AscendC::GetBlockIdx() * this->rowsPerBlock;
        uint32_t endRow = startRow + this->rowsPerBlock;
        if (endRow > this->totalRows) {
            endRow = this->totalRows;
        }
        for (uint32_t r = startRow; r < endRow; r++) {
            ProcessRow(r);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t rowIdx)
    {
        uint32_t b = rowIdx / this->height;
        uint32_t h = rowIdx % this->height;
        uint32_t outOffset = b * this->spatialSize + h * this->width;
        uint32_t inRowBase = b * this->channels * this->spatialSize + h * this->width;

        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = static_cast<uint32_t>(this->width * sizeof(float));
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;

        AscendC::DataCopyPadExtParams<float> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;

        AscendC::LocalTensor<float> accLocal = accBuf.Get<float>();

        // Initialize accumulator with channel 0 scaled.
        {
            AscendC::LocalTensor<float> chanLocal = inQueue.AllocTensor<float>();
            AscendC::DataCopyPad(chanLocal, xGm[inRowBase + 0 * this->spatialSize], copyParams, padParams);
            inQueue.EnQue(chanLocal);
            chanLocal = inQueue.DeQue<float>();
            AscendC::Muls(accLocal, chanLocal, this->scaleFactor, this->width);
            inQueue.FreeTensor(chanLocal);
        }

        for (uint32_t c = 1; c < this->channels; c++) {
            AscendC::LocalTensor<float> chanLocal = inQueue.AllocTensor<float>();
            AscendC::DataCopyPad(chanLocal, xGm[inRowBase + c * this->spatialSize], copyParams, padParams);
            inQueue.EnQue(chanLocal);
            chanLocal = inQueue.DeQue<float>();
            AscendC::Muls(chanLocal, chanLocal, this->scaleFactor, this->width);
            AscendC::Min(accLocal, accLocal, chanLocal, this->width);
            inQueue.FreeTensor(chanLocal);
        }

        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::Adds(outLocal, accLocal, (float)0.0f, this->width);
        outQueue.EnQue<float>(outLocal);
        outLocal = outQueue.DeQue<float>();
        AscendC::DataCopyPad(yGm[outOffset], outLocal, copyParams);
        outQueue.FreeTensor(outLocal);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t height;
    uint32_t width;
    uint32_t spatialSize;
    uint32_t totalRows;
    uint32_t rowsPerBlock;
    uint32_t alignedWidth;
    float scaleFactor;
};

extern "C" __global__ __aicore__ void conv2d_scaling_min_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dScalingMin op;
    op.Init(x, y,
            tiling_data.batchSize,
            tiling_data.channels,
            tiling_data.height,
            tiling_data.width,
            tiling_data.scaleFactor);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, scale_factor):
        super().__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.scale_factor = float(scale_factor)

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
        return custom_ops_lib.conv2d_scaling_min_custom(x, self.scale_factor)
