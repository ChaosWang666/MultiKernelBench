

Looking at this architecture, it performs a transposed 2D convolution. Implementing a full conv_transpose2d kernel from scratch in AscendC is extremely complex. The practical approach is to leverage the existing ACLNN operators for convolution transpose while wrapping them in our custom operator framework.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConvTranspose2d {
public:
    __aicore__ inline KernelConvTranspose2d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inputH, uint32_t inputW,
                                 uint32_t kernelH, uint32_t kernelW,
                                 uint32_t strideH, uint32_t strideW,
                                 uint32_t padH, uint32_t padW,
                                 uint32_t outputPadH, uint32_t outputPadW,
                                 uint32_t groups,
                                 uint32_t outputH, uint32_t outputW,
                                 uint32_t totalLength)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inputH = inputH;
        this->inputW = inputW;
        this->kernelH = kernelH;
        this->kernelW = kernelW;
        this->strideH = strideH;
        this->strideW = strideW;
        this->padH = padH;
        this->padW = padW;
        this->groups = groups;
        this->outputH = outputH;
        this->outputW = outputW;
        this->totalLength = totalLength;

        this->inChannelsPerGroup = inChannels / groups;
        this->outChannelsPerGroup = outChannels / groups;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inChannels * inputH * inputW);
        wGm.SetGlobalBuffer((__gm__ float*)weight, inChannels * outChannelsPerGroup * kernelH * kernelW);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalLength);
    }

    __aicore__ inline void Process()
    {
        // Simple direct implementation: iterate over output elements
        for (uint32_t n = 0; n < batchSize; n++) {
            for (uint32_t oc = 0; oc < outChannels; oc++) {
                uint32_t g = oc / outChannelsPerGroup;
                uint32_t oc_in_group = oc % outChannelsPerGroup;
                for (uint32_t oh = 0; oh < outputH; oh++) {
                    for (uint32_t ow = 0; ow < outputW; ow++) {
                        float val = 0.0f;
                        for (uint32_t ic_local = 0; ic_local < inChannelsPerGroup; ic_local++) {
                            uint32_t ic = g * inChannelsPerGroup + ic_local;
                            for (uint32_t kh = 0; kh < kernelH; kh++) {
                                for (uint32_t kw = 0; kw < kernelW; kw++) {
                                    int32_t ih_s = (int32_t)oh + (int32_t)padH - (int32_t)kh;
                                    int32_t iw_s = (int32_t)ow + (int32_t)padW - (int32_t)kw;
                                    if (ih_s % (int32_t)strideH != 0) continue;
                                    if (iw_s % (int32_t)strideW != 0) continue;
                                    int32_t ih = ih_s / (int32_t)strideH;
                                    int32_t iw = iw_s / (int32_t)strideW;
                                    if (ih < 0 || ih >= (int32_t)inputH) continue;
                                    if (iw < 0 || iw >= (int32_t)inputW) continue;

                                    uint32_t x_idx = ((n * inChannels + ic) * inputH + (uint32_t)ih) * inputW + (uint32_t)iw;
                                    // weight layout: [inChannels, outChannelsPerGroup, kH, kW]
                                    uint32_t w_idx = ((ic * outChannelsPerGroup + oc_in_group) * kernelH + kh) * kernelW + kw;
                                    val += (float)xGm.GetValue(x_idx) * (float)wGm.GetValue(w_idx);
                                }
                            }
                        }
                        uint32_t y_idx = ((n * outChannels + oc) * outputH + oh) * outputW + ow;
                        yGm.SetValue(y_idx, (float)val);
                    }
                }
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize, inChannels, outChannels;
    uint32_t inputH, inputW;
    uint32_t kernelH, kernelW;
    uint32_t strideH, strideW;
    uint32_t padH, padW;
    uint32_t groups;
    uint32_t outputH, outputW;
    uint32_t totalLength;
    uint32_t inChannelsPerGroup, outChannelsPerGroup;
};

extern "C" __global__ __aicore__ void conv_transposed_2d_square_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose2d op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inputH, tiling_data.inputW,
            tiling_data.kernelH, tiling_data.kernelW,
            tiling_data.strideH, tiling_data.strideW,
            tiling_data.padH, tiling_data.padW,
            tiling_data.outputPadH, tiling_data.outputPadW,
            tiling_data.groups,
            tiling_data.outputH, tiling_data.outputW,
            tiling_data.totalLength);
    op.Process();
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor conv_transposed_2d_custom_impl_npu(const at::Tensor& input, const at::Tensor& weight,
    const c10::optional<at::Tensor>& bias_opt,
    int64_t stride_h, int64_t stride_w,
    int64_t pad_h, int64_t pad_w,
    int64_t output_pad_h, int64_t output_pad_w,
    int64_t groups) {

    auto input_contig = input.contiguous();
    auto weight_contig = weight.contiguous();

    int64_t batch = input_contig.size(0);
    int64_t in_channels = input_contig.size(1);
    int64_t inputH = input_contig.size(2);
    int64_t inputW = input_contig.size(3);

    int64_t out_channels_per_group = weight_contig.size(1);
    int64_t kernelH = weight_contig.size(2);
    int64_t kernelW = weight_contig.size(3);
    int64_t out_channels = out_channels_per_group * groups;

    int64_t outputH = (inputH - 1) * stride_h - 2 * pad_h + kernelH + output_pad_h;
    int64_t outputW = (inputW - 1) * stride_w - 2 * pad_w + kernelW + output_pad_w;

    at::Tensor result = at::zeros({batch, out_channels, outputH, outputW}, input_contig.options());

    if (bias_opt.has_value() && bias_opt.value().defined()) {
        auto bias_val = bias_opt.value().contiguous();
        EXEC_NPU_CMD(aclnnConvTransposed2dSquareInputAsymmetricKernelCustom,
            input_contig, weight_contig, bias_val, result,
            stride_h, stride_w, pad_h, pad_w, output_pad_h, output_pad_w, groups);
    } else {
        // Create a dummy empty bias tensor
        at::Tensor bias_empty = at::Tensor();
        EXEC_NPU_CMD(aclnnConvTransposed2dSquareInputAsymmetricKernelCustom,
            input_contig, weight_contig, bias_empty, result,
            stride_h, stride_w, pad_h, pad_w, output_pad_h, output_pad_w, groups);
    }

    if (bias_opt.has_value() && bias_opt.value().defined()) {
        result = result + bias_opt.value().reshape({1, -1, 1, 1});
    }

    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("conv_transposed_2d_square_input_asymmetric_kernel_custom", &conv_transposed_2d_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("conv_transposed_2d_square_input_asymmetric_kernel_custom", &conv_transposed_2d_custom_impl_npu,
          "ConvTranspose2d custom op");
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels: int, out_channels: int, kernel_size: tuple, stride: int = 1, padding: int = 0, output_padding: int = 0, groups: int = 1, bias: bool = False):
        super().__init__()
        self.conv_transpose2d = torch.nn.ConvTranspose2d(
            in_channels, out_channels, kernel_size,
            stride=stride, padding=padding, output_padding=output_padding,
            groups=groups, bias=bias
        )
        self.stride = stride if isinstance(stride, tuple) else (stride, stride)
        self.padding = padding if isinstance(padding, tuple) else (padding, padding)
        self.output_padding = output_padding if isinstance(output_padding, tuple) else (output_padding, output_padding)
        self.groups = groups
        self.has_bias = bias

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        weight = self.conv_transpose2d.weight
        bias_tensor = self.conv_transpose2d.bias if self.has_bias else None
        return custom_ops_lib.conv_transposed_2d_square_input_asymmetric_kernel_custom(
            x, weight, bias_tensor,
            self.stride[0], self.stride[1],
            self.padding[0], self.padding[1],
            self.output_padding[0], self.output_padding[1],
            self.groups
        )
'''
