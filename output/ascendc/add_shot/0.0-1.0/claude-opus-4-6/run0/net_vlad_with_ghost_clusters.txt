

Looking at the architecture, I'll identify the key computational bottleneck that can benefit from a custom AscendC kernel. The main expensive operations are:

1. Matrix multiplication + batch norm + softmax + slicing + aggregation
2. The VLAD computation involves softmax, assignment slicing, sum, matmul, subtraction, and normalization

I'll create a fused kernel that handles the softmax followed by ghost cluster removal (slicing), which is a common pattern that can be optimized. The softmax is computed over `K+G` clusters but only the first `K` are kept, so we can fuse the softmax computation with the slicing to avoid writing unused ghost cluster softmax values.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSoftmaxSlice {
public:
    __aicore__ inline KernelSoftmaxSlice() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalRows, uint32_t fullCols, uint32_t keepCols, uint32_t tileNum)
    {
        this->totalRows = totalRows;
        this->fullCols = fullCols;
        this->keepCols = keepCols;

        // Distribute rows across blocks
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->rowsPerBlock = totalRows / blockNum;
        uint32_t remainder = totalRows % blockNum;
        if (blockIdx < remainder) {
            this->rowsPerBlock += 1;
            this->startRow = blockIdx * this->rowsPerBlock;
        } else {
            this->startRow = remainder * (this->rowsPerBlock + 1) + (blockIdx - remainder) * this->rowsPerBlock;
        }

        if (this->rowsPerBlock == 0) {
            return;
        }

        // Align fullCols to 32 bytes (8 floats)
        this->fullColsAligned = (fullCols + 7) / 8 * 8;
        this->keepColsAligned = (keepCols + 7) / 8 * 8;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->startRow * fullCols, this->rowsPerBlock * fullCols);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->startRow * keepCols, this->rowsPerBlock * keepCols);

        // Process one row at a time, allocate buffer for one row
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->fullColsAligned * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->keepColsAligned * sizeof(float));
        pipe.InitBuffer(tmpBuf1, this->fullColsAligned * sizeof(float));
        pipe.InitBuffer(tmpBuf2, this->fullColsAligned * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->rowsPerBlock == 0) return;
        for (uint32_t i = 0; i < this->rowsPerBlock; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        // Copy one row of fullCols elements
        AscendC::DataCopy(xLocal, xGm[row * fullCols], this->fullColsAligned);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();

        // Step 1: Find max for numerical stability
        AscendC::ReduceMax(tmp1, xLocal, tmp2, this->fullColsAligned);
        float maxVal = tmp1.GetValue(0);

        // Step 2: Subtract max -> exp
        AscendC::Adds(xLocal, xLocal, -maxVal, this->fullColsAligned);
        AscendC::Exp(xLocal, xLocal, this->fullColsAligned);

        // Zero out elements beyond fullCols (if fullColsAligned > fullCols)
        for (uint32_t j = fullCols; j < fullColsAligned; j++) {
            xLocal.SetValue(j, 0.0f);
        }

        // Step 3: Sum of exp
        AscendC::ReduceSum(tmp1, xLocal, tmp2, this->fullColsAligned);
        float sumVal = tmp1.GetValue(0);

        // Step 4: Divide by sum
        float invSum = 1.0f / sumVal;
        AscendC::Muls(xLocal, xLocal, invSum, this->fullColsAligned);

        // Step 5: Copy only keepCols elements to output
        AscendC::DataCopy(yLocal, xLocal, this->keepColsAligned);

        outQueueY.EnQue(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[row * keepCols], yLocal, this->keepColsAligned);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t totalRows;
    uint32_t fullCols;
    uint32_t fullColsAligned;
    uint32_t keepCols;
    uint32_t keepColsAligned;
    uint32_t rowsPerBlock;
    uint32_t startRow;
};

extern "C" __global__ __aicore__ void net_vlad_with_ghost_clusters_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSoftmaxSlice op;
    op.Init(x, y, tiling_data.totalRows, tiling_data.fullCols, tiling_data.keepCols, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import torch.nn.functional as F
import torch as th
import math
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, cluster_size, feature_size, ghost_clusters):
        super(ModelNew, self).__init__()

        self.feature_size = feature_size
        self.cluster_size = cluster_size
        self.ghost_clusters = ghost_clusters

        init_sc = (1 / math.sqrt(feature_size))
        clusters = cluster_size + ghost_clusters

        self.clusters = nn.Parameter(init_sc * th.randn(feature_size, clusters))
        self.batch_norm = nn.BatchNorm1d(clusters)
        self.clusters2 = nn.Parameter(init_sc * th.randn(1, feature_size, cluster_size))
        self.out_dim = self.cluster_size * feature_size

    def forward(self, x, mask=None):
        max_sample = x.size()[1]
        x = x.view(-1, self.feature_size)

        assignment = th.matmul(x, self.clusters)
        assignment = self.batch_norm(assignment)

        # Fused softmax + ghost cluster slicing using custom kernel
        assignment = custom_ops_lib.net_vlad_with_ghost_clusters_custom(assignment, self.cluster_size)

        assignment = assignment.view(-1, max_sample, self.cluster_size)
        a_sum = th.sum(assignment, dim=1, keepdim=True)
        a = a_sum * self.clusters2

        assignment = assignment.transpose(1, 2)

        x = x.view(-1, max_sample, self.feature_size)
        vlad = th.matmul(assignment, x)
        vlad = vlad.transpose(1, 2)
        vlad = vlad - a

        vlad = F.normalize(vlad)

        vlad = vlad.reshape(-1, self.cluster_size * self.feature_size)
        vlad = F.normalize(vlad)
        return vlad
'''
