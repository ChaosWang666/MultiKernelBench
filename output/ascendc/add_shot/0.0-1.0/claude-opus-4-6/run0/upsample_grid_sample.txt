

Looking at this architecture, I need to implement a fused upsample + grid_sample kernel. The operation involves:
1. Bilinear interpolation (upsample by 2x)
2. Affine grid generation from theta
3. Grid sampling with bilinear interpolation

This is a complex operation. Let me implement it as a custom AscendC kernel.

kernel_src="""
#include "kernel_operator.h"

// Fused upsample(2x bilinear) + affine_grid + grid_sample(bilinear) kernel
// Input x: [N, C, H, W], theta: [N, 2, 3]
// Output y: [N, C, 2H, 2W]
// 
// With align_corners=False, scale_factor=2:
// For upsample bilinear 2x with align_corners=False:
//   src_y = (dst_y + 0.5) / 2.0 - 0.5
//   src_x = (dst_x + 0.5) / 2.0 - 0.5
//
// For affine_grid with align_corners=False on output size [N,C,oH,oW]:
//   normalized coords: nx = (2*ox + 1)/oW - 1, ny = (2*oy + 1)/oH - 1
//   grid_x = theta[0]*nx + theta[1]*ny + theta[2]
//   grid_y = theta[3]*nx + theta[4]*ny + theta[5]
//
// For grid_sample bilinear with align_corners=False on upsampled [N,C,uH,uW]:
//   ix = ((grid_x + 1) * uW - 1) / 2
//   iy = ((grid_y + 1) * uH - 1) / 2
//
// Since we fuse upsample into grid_sample, we compute the final source coords
// in the original input space [N,C,H,W].

extern "C" __global__ __aicore__ void upsample_grid_sample_custom(
    GM_ADDR x, GM_ADDR theta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    
    uint32_t batchSize = tiling_data.batchSize;
    uint32_t channels = tiling_data.channels;
    uint32_t inH = tiling_data.inputH;
    uint32_t inW = tiling_data.inputW;
    uint32_t outH = tiling_data.outputH;
    uint32_t outW = tiling_data.outputW;
    
    uint32_t blockIdx = AscendC::GetBlockIdx();
    uint32_t blockNum = AscendC::GetBlockNum();
    
    // Total output pixels across all batches and channels
    uint32_t totalOutPixels = batchSize * channels * outH * outW;
    
    // Distribute pixels across blocks
    uint32_t pixelsPerBlock = (totalOutPixels + blockNum - 1) / blockNum;
    uint32_t startPixel = blockIdx * pixelsPerBlock;
    uint32_t endPixel = startPixel + pixelsPerBlock;
    if (endPixel > totalOutPixels) endPixel = totalOutPixels;
    
    __gm__ float* xGm = (__gm__ float*)x;
    __gm__ float* thetaGm = (__gm__ float*)theta;
    __gm__ float* yGm = (__gm__ float*)y;
    
    // Process each output pixel assigned to this block
    for (uint32_t pix = startPixel; pix < endPixel; pix++) {
        // Decode pixel index: pix -> (n, c, oy, ox)
        uint32_t rem = pix;
        uint32_t ox = rem % outW;
        rem = rem / outW;
        uint32_t oy = rem % outH;
        rem = rem / outH;
        uint32_t c = rem % channels;
        uint32_t n = rem / channels;
        
        // Step 1: Compute normalized coordinates for affine_grid (align_corners=False)
        // Output size is [N,C,outH,outW]
        float nx = (2.0f * (float)ox + 1.0f) / (float)outW - 1.0f;
        float ny = (2.0f * (float)oy + 1.0f) / (float)outH - 1.0f;
        
        // Step 2: Apply affine transform: theta is [N, 2, 3]
        // grid_x = theta[n][0][0]*nx + theta[n][0][1]*ny + theta[n][0][2]
        // grid_y = theta[n][1][0]*nx + theta[n][1][1]*ny + theta[n][1][2]
        uint32_t thetaBase = n * 6;
        float t00 = *((__gm__ float*)(thetaGm + thetaBase + 0));
        float t01 = *((__gm__ float*)(thetaGm + thetaBase + 1));
        float t02 = *((__gm__ float*)(thetaGm + thetaBase + 2));
        float t10 = *((__gm__ float*)(thetaGm + thetaBase + 3));
        float t11 = *((__gm__ float*)(thetaGm + thetaBase + 4));
        float t12 = *((__gm__ float*)(thetaGm + thetaBase + 5));
        
        float grid_x = t00 * nx + t01 * ny + t02;
        float grid_y = t10 * nx + t11 * ny + t12;
        
        // Step 3: Convert grid coords to pixel coords in upsampled space (align_corners=False)
        // ix = ((grid_x + 1) * outW - 1) / 2
        // iy = ((grid_y + 1) * outH - 1) / 2
        float uix = ((grid_x + 1.0f) * (float)outW - 1.0f) / 2.0f;
        float uiy = ((grid_y + 1.0f) * (float)outH - 1.0f) / 2.0f;
        
        // Step 4: Map from upsampled space to original input space
        // For bilinear upsample 2x with align_corners=False:
        //   src = (dst + 0.5) / scale - 0.5
        float src_x = (uix + 0.5f) / 2.0f - 0.5f;
        float src_y = (uiy + 0.5f) / 2.0f - 0.5f;
        
        // Step 5: Bilinear interpolation in input space
        int32_t x0 = (int32_t)floorf(src_x);
        int32_t y0 = (int32_t)floorf(src_y);
        int32_t x1 = x0 + 1;
        int32_t y1 = y0 + 1;
        
        float xw = src_x - (float)x0;
        float yw = src_y - (float)y0;
        
        // Clamp to input bounds
        auto clampInt = [](int32_t v, int32_t lo, int32_t hi) -> int32_t {
            if (v < lo) return lo;
            if (v > hi) return hi;
            return v;
        };
        
        int32_t cx0 = clampInt(x0, 0, (int32_t)inW - 1);
        int32_t cx1 = clampInt(x1, 0, (int32_t)inW - 1);
        int32_t cy0 = clampInt(y0, 0, (int32_t)inH - 1);
        int32_t cy1 = clampInt(y1, 0, (int32_t)inH - 1);
        
        // Check bounds for zero-padding (grid_sample default padding_mode='zeros')
        float v00 = 0.0f, v01 = 0.0f, v10 = 0.0f, v11 = 0.0f;
        
        uint32_t inputBase = n * channels * inH * inW + c * inH * inW;
        
        // For grid_sample with padding_mode='zeros': out of bounds -> 0
        bool x0_in = (x0 >= 0 && x0 < (int32_t)inW);
        bool x1_in = (x1 >= 0 && x1 < (int32_t)inW);
        bool y0_in = (y0 >= 0 && y0 < (int32_t)inH);
        bool y1_in = (y1 >= 0 && y1 < (int32_t)inH);
        
        if (y0_in && x0_in) {
            v00 = *((__gm__ float*)(xGm + inputBase + (uint32_t)y0 * inW + (uint32_t)x0));
        }
        if (y0_in && x1_in) {
            v01 = *((__gm__ float*)(xGm + inputBase + (uint32_t)y0 * inW + (uint32_t)x1));
        }
        if (y1_in && x0_in) {
            v10 = *((__gm__ float*)(xGm + inputBase + (uint32_t)y1 * inW + (uint32_t)x0));
        }
        if (y1_in && x1_in) {
            v11 = *((__gm__ float*)(xGm + inputBase + (uint32_t)y1 * inW + (uint32_t)x1));
        }
        
        float result = v00 * (1.0f - xw) * (1.0f - yw) +
                        v01 * xw * (1.0f - yw) +
                        v10 * (1.0f - xw) * yw +
                        v11 * xw * yw;
        
        // Write output
        uint32_t outIdx = n * channels * outH * outW + c * outH * outW + oy * outW + ox;
        *((__gm__ float*)(yGm + outIdx)) = result;
    }
}

at::Tensor upsample_grid_sample_custom_impl_npu(const at::Tensor& x, const at::Tensor& theta) {
    auto N = x.size(0);
    auto C = x.size(1);
    auto H = x.size(2);

    auto outH = H * 2;
    auto outW = W * 2;

    return result;
}
