kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelHuberLoss {
public:
    __aicore__ inline KernelHuberLoss() {}
    __aicore__ inline void Init(GM_ADDR predictions, GM_ADDR targets, GM_ADDR loss,
                                 uint32_t totalLength, uint32_t tileNum)
    {
        this->totalLength = totalLength;
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        predGm.SetGlobalBuffer((__gm__ float *)predictions + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        targGm.SetGlobalBuffer((__gm__ float *)targets + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        lossGm.SetGlobalBuffer((__gm__ float *)loss + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueuePred, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueTarg, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueLoss, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf1, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> predLocal = inQueuePred.AllocTensor<float>();
        AscendC::LocalTensor<float> targLocal = inQueueTarg.AllocTensor<float>();
        AscendC::DataCopy(predLocal, predGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(targLocal, targGm[progress * this->tileLength], this->tileLength);
        inQueuePred.EnQue(predLocal);
        inQueueTarg.EnQue(targLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> predLocal = inQueuePred.DeQue<float>();
        AscendC::LocalTensor<float> targLocal = inQueueTarg.DeQue<float>();
        AscendC::LocalTensor<float> lossLocal = outQueueLoss.AllocTensor<float>();
        AscendC::LocalTensor<float> diffLocal = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> absDiffLocal = tmpBuf2.Get<float>();

        // diff = predictions - targets
        AscendC::Sub(diffLocal, predLocal, targLocal, this->tileLength);
        // absDiff = |diff|
        AscendC::Abs(absDiffLocal, diffLocal, this->tileLength);

        // Smooth L1 / Huber loss with beta=1.0:
        // if |diff| < 1.0: 0.5 * diff^2
        // else: |diff| - 0.5

        // Compute quadratic part: 0.5 * diff^2
        AscendC::Mul(lossLocal, diffLocal, diffLocal, this->tileLength);
        AscendC::Muls(lossLocal, lossLocal, 0.5f, this->tileLength);

        // Compute linear part: |diff| - 0.5
        AscendC::Adds(diffLocal, absDiffLocal, -0.5f, this->tileLength);

        // Select: where absDiff < 1.0 use quadratic, else use linear
        // We use Compare + Select
        // Create mask: absDiff < 1.0
        AscendC::LocalTensor<float> onesLocal = tmpBuf2.Get<float>();
        // We already have absDiffLocal in tmpBuf2, need to be careful
        // Actually let's recompute: absDiffLocal is in tmpBuf2, diffLocal (linear part) is in tmpBuf1
        // lossLocal has quadratic part
        // We need the comparison based on original absDiff which is in tmpBuf2
        // But we overwrote diffLocal (tmpBuf1) with linear part, and absDiffLocal (tmpBuf2) still has |diff|
        // Wait - we did Adds(diffLocal, absDiffLocal, -0.5f) which writes to tmpBuf1 = linear part
        // absDiffLocal (tmpBuf2) still has |diff|
        
        // Actually let me re-check the buffer aliasing:
        // diffLocal = tmpBuf1, absDiffLocal = tmpBuf2
        // After Adds: diffLocal (tmpBuf1) = |diff| - 0.5 (linear part)
        // absDiffLocal (tmpBuf2) still = |diff|
        // lossLocal = quadratic part

        // Compare: need to check absDiffLocal < 1.0
        // Use: if absDiff < 1, result = quadratic, else result = linear
        // Approach: compute both, then blend
        // result = quadratic * (absDiff < 1) + linear * (absDiff >= 1)
        // Using Sel with Compare

        // We'll use a simpler approach with Min/Max style:
        // smooth_l1 = where(|d|<1, 0.5*d^2, |d|-0.5)
        // Note: at |d|=1, both give 0.5, so it's continuous

        // Use the fact that for |d| < 1: 0.5*d^2 < 0.5 and |d|-0.5 < 0.5
        // For |d| >= 1: |d|-0.5 >= 0.5 and 0.5*d^2 >= 0.5
        // At boundary they're equal
        // So smooth_l1 = min(0.5*d^2, |d|-0.5) only works if... no that's wrong for large |d|

        // Let's just use Min: smooth_l1 = min(0.5*d^2, |d| - 0.5) ? 
        // For |d|=2: 0.5*4=2, |d|-0.5=1.5 -> min=1.5 correct
        // For |d|=0.5: 0.5*0.25=0.125, 0.5-0.5=0 -> min=0, but correct is 0.125. WRONG.

        // Need to handle negative linear part. Let's clamp linear to 0:
        // max(|d|-0.5, 0) won't help either.

        // Better approach: smooth_l1 = where(|d|<1, 0.5*d^2, |d|-0.5)
        // Since both are non-negative for the respective domains, we can use:
        // result = min(0.5*d^2, max(|d|-0.5, 0.5*d^2)) -- no, too convoluted

        // Just compute it as: huber = 0.5*d^2/max(|d|,1)  ... hmm not standard

        // Let me use a cleaner formula that avoids branching:
        // smooth_l1(d) = where(|d|<1, 0.5*d^2, |d|-0.5)
        // Equivalent: Let c = clamp(|d|, 0, 1), then smooth_l1 = |d| - 0.5*c
        // Check: |d|<1: c=|d|, result=|d|-0.5*|d|=0.5*|d|... no that's 0.5|d| not 0.5d^2. Wrong.

        // Actually the correct identity is:
        // smooth_l1(d) = 0.5 * (|d| - clamp(|d|-1, min=0) )^2 ... no

        // Let me try yet another approach. We know:
        // if |d| < 1: loss = 0.5*d^2
        // if |d| >= 1: loss = |d| - 0.5
        // We can write: loss = |d| - 0.5 + 0.5*(1-|d|)^2  when |d|<1
        //   = |d| - 0.5 + 0.5 - |d| + 0.5*d^2 = 0.5*d^2 ✓
        // When |d|>=1: loss should be |d|-0.5, so we need the correction term to be 0.
        // correction = 0.5*(max(1-|d|,0))^2
        // loss = |d| - 0.5 + 0.5*(max(1-|d|,0))^2
        // Check |d|>=1: max(1-|d|,0)=0, loss=|d|-0.5 ✓
        // Check |d|<1: max(1-|d|,0)=1-|d|, loss=|d|-0.5+0.5*(1-|d|)^2=|d|-0.5+0.5-|d|+0.5d^2=0.5d^2 ✓
        // Great!

        // Recompute using this formula:
        // absDiffLocal (tmpBuf2) = |diff|
        // Step 1: t = 1 - |diff|, clamped to >= 0
        // Step 2: loss = |diff| - 0.5 + 0.5 * t^2

        // We have: lossLocal = 0.5*d^2 (quadratic, but we'll overwrite)
        // diffLocal (tmpBuf1) = |diff| - 0.5 (linear part)

        // Let's redo from absDiffLocal:
        // tmpBuf1 (diffLocal) can be reused
        // Compute t = 1 - |diff|
        AscendC::Muls(lossLocal, absDiffLocal, -1.0f, this->tileLength);
        AscendC::Adds(lossLocal, lossLocal, 1.0f, this->tileLength);
        // lossLocal = 1 - |diff|
        // Clamp to >= 0: use Maxs if available, or Relu
        // Actually we need max(lossLocal, 0). We can use Relu-like or Maxs.
        // AscendC doesn't have Maxs with scalar easily. Let's use:
        // We'll use Abs and Add trick: max(x,0) = (x + |x|) / 2
        AscendC::Abs(diffLocal, lossLocal, this->tileLength); // diffLocal = |1-|diff||
        AscendC::Add(lossLocal, lossLocal, diffLocal, this->tileLength); // lossLocal = (1-|diff|) + |1-|diff||
        AscendC::Muls(lossLocal, lossLocal, 0.5f, this->tileLength); // lossLocal = max(1-|diff|, 0) = t

        // Now compute t^2
        AscendC::Mul(lossLocal, lossLocal, lossLocal, this->tileLength); // t^2
        AscendC::Muls(lossLocal, lossLocal, 0.5f, this->tileLength); // 0.5*t^2

        // Add |diff| - 0.5
        AscendC::Adds(diffLocal, absDiffLocal, -0.5f, this->tileLength); // |diff| - 0.5
        AscendC::Add(lossLocal, lossLocal, diffLocal, this->tileLength); // loss per element

        // Divide by totalLength for mean reduction
        float invN = 1.0f / (float)this->totalLength;
        AscendC::Muls(lossLocal, lossLocal, invN, this->tileLength);

        outQueueLoss.EnQue<float>(lossLocal);
        inQueuePred.FreeTensor(predLocal);
        inQueueTarg.FreeTensor(targLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> lossLocal = outQueueLoss.DeQue<float>();
        AscendC::DataCopy(lossGm[progress * this->tileLength], lossLocal, this->tileLength);
        outQueueLoss.FreeTensor(lossLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueuePred, inQueueTarg;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueLoss;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> predGm;
    AscendC::GlobalTensor<float> targGm;
    AscendC::GlobalTensor<float> lossGm;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void huber_loss_custom(GM_ADDR predictions, GM_ADDR targets, GM_ADDR loss, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelHuberLoss op;
    op.Init(predictions, targets, loss, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self) -> None:
        super().__init__()

    def forward(self, predictions, targets):
        # The kernel computes per-element loss / N and stores it in an output tensor of same shape.
        # We need to sum to get the scalar mean loss.
        per_element_loss = custom_ops_lib.huber_loss_custom(predictions, targets)
        return per_element_loss.sum()
