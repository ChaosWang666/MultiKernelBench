

Looking at the ShuffleNet architecture, the key custom operation I can optimize is the **Channel Shuffle** operation, which involves reshape, transpose, and reshape operations. This is a memory-bound operation that can benefit from a fused custom kernel on Ascend hardware.

The channel shuffle takes a tensor of shape (batch, channels, H, W), reshapes to (batch, groups, channels_per_group, H, W), transposes dimensions 1 and 2, and reshapes back. This can be implemented as a single data movement kernel.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelChannelShuffle {
public:
    __aicore__ inline KernelChannelShuffle() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t totalLength, uint32_t tileNum,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t height, uint32_t width,
                                 uint32_t groups)
    {
        this->totalLength = totalLength;
        this->batchSize = batchSize;
        this->channels = channels;
        this->height = height;
        this->width = width;
        this->groups = groups;
        this->channelsPerGroup = channels / groups;
        this->spatialSize = height * width;

        // Each block processes a portion of batch*channels
        uint32_t totalSlices = batchSize * channels;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        // Distribute slices (each slice = one channel plane = H*W elements)
        this->sliceStart = (totalSlices / blockNum) * blockIdx;
        uint32_t sliceEnd = (blockIdx == blockNum - 1) ? totalSlices : (totalSlices / blockNum) * (blockIdx + 1);
        this->sliceCount = sliceEnd - this->sliceStart;

        xGm.SetGlobalBuffer((__gm__ float*)x, totalLength);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalLength);

        // Compute tile length: each tile processes some number of spatial elements
        this->tileNum = tileNum;
        if (this->sliceCount == 0) {
            this->tileLength = 0;
        } else {
            // We'll process one slice at a time, tiling the spatial dimension
            uint32_t alignedSpatial = (this->spatialSize + 31) / 32 * 32;
            this->tileLength = alignedSpatial;
        }

        if (this->tileLength > 0) {
            pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
            pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        }
    }

    __aicore__ inline void Process()
    {
        if (this->sliceCount == 0 || this->tileLength == 0) return;

        for (uint32_t i = 0; i < this->sliceCount; i++) {
            uint32_t globalSlice = this->sliceStart + i;
            // globalSlice = batch_idx * channels + old_channel
            uint32_t batchIdx = globalSlice / this->channels;
            uint32_t oldChannel = globalSlice % this->channels;

            // Channel shuffle mapping:
            // old layout: (batch, groups, channels_per_group, H, W)
            // old_channel = g * channels_per_group + c
            // new_channel = c * groups + g
            uint32_t g = oldChannel / this->channelsPerGroup;
            uint32_t c = oldChannel % this->channelsPerGroup;
            uint32_t newChannel = c * this->groups + g;

            uint32_t srcOffset = (batchIdx * this->channels + oldChannel) * this->spatialSize;
            uint32_t dstOffset = (batchIdx * this->channels + newChannel) * this->spatialSize;

            CopyIn(srcOffset);
            CopyOut(dstOffset);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t srcOffset)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[srcOffset], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t dstOffset)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        AscendC::DataCopy(yLocal, xLocal, this->tileLength);
        outQueueY.EnQue(yLocal);
        inQueueX.FreeTensor(xLocal);

        AscendC::LocalTensor<float> yOut = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[dstOffset], yOut, this->tileLength);
        outQueueY.FreeTensor(yOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t totalLength;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t height;
    uint32_t width;
    uint32_t groups;
    uint32_t channelsPerGroup;
    uint32_t spatialSize;
    uint32_t sliceStart;
    uint32_t sliceCount;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void shufflenet_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelChannelShuffle op;
    op.Init(x, y,
            tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.height, tiling_data.width,
            tiling_data.groups);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

class ChannelShuffleCustom(nn.Module):
    def __init__(self, groups):
        super(ChannelShuffleCustom, self).__init__()
        self.groups = groups

    def forward(self, x):
        return custom_ops_lib.shufflenet_custom(x, self.groups)

class ShuffleNetUnit(nn.Module):
    def __init__(self, in_channels, out_channels, groups=3):
        super(ShuffleNetUnit, self).__init__()
        assert out_channels % 4 == 0
        mid_channels = out_channels // 4

        self.conv1 = nn.Conv2d(in_channels, mid_channels, kernel_size=1, stride=1, padding=0, groups=groups, bias=False)
        self.bn1 = nn.BatchNorm2d(mid_channels)

        self.conv2 = nn.Conv2d(mid_channels, mid_channels, kernel_size=3, stride=1, padding=1, groups=mid_channels, bias=False)
        self.bn2 = nn.BatchNorm2d(mid_channels)

        self.conv3 = nn.Conv2d(mid_channels, out_channels, kernel_size=1, stride=1, padding=0, groups=groups, bias=False)
        self.bn3 = nn.BatchNorm2d(out_channels)

        self.shuffle = ChannelShuffleCustom(groups)

        if in_channels == out_channels:
            self.shortcut = nn.Sequential()
        else:
            self.shortcut = nn.Sequential(
                nn.Conv2d(in_channels, out_channels, kernel_size=1, stride=1, padding=0, bias=False),
                nn.BatchNorm2d(out_channels)
            )

    def forward(self, x):
        out = F.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out = self.shuffle(out)
        out = F.relu(self.bn3(self.conv3(out)))
        out += self.shortcut(x)
        return out

class ModelNew(nn.Module):
    def __init__(self, num_classes=1000, groups=3, stages_repeats=[3, 7, 3], stages_out_channels=[24, 240, 480, 960]):
        super(ModelNew, self).__init__()

        self.conv1 = nn.Conv2d(3, stages_out_channels[0], kernel_size=3, stride=2, padding=1, bias=False)
        self.bn1 = nn.BatchNorm2d(stages_out_channels[0])
        self.maxpool = nn.MaxPool2d(kernel_size=3, stride=2, padding=1)

        self.stage2 = self._make_stage(stages_out_channels[0], stages_out_channels[1], stages_repeats[0], groups)
        self.stage3 = self._make_stage(stages_out_channels[1], stages_out_channels[2], stages_repeats[1], groups)
        self.stage4 = self._make_stage(stages_out_channels[2], stages_out_channels[3], stages_repeats[2], groups)

        self.conv5 = nn.Conv2d(stages_out_channels[3], 1024, kernel_size=1, stride=1, padding=0, bias=False)
        self.bn5 = nn.BatchNorm2d(1024)

        self.fc = nn.Linear(1024, num_classes)

    def _make_stage(self, in_channels, out_channels, repeats, groups):
        layers = []
        layers.append(ShuffleNetUnit(in_channels, out_channels, groups))
        for _ in range(1, repeats):
            layers.append(ShuffleNetUnit(out_channels, out_channels, groups))
        return nn.Sequential(*layers)

    def forward(self, x):
        x = F.relu(self.bn1(self.conv1(x)))
        x = self.maxpool(x)

        x = self.stage2(x)
        x = self.stage3(x)
        x = self.stage4(x)

        x = F.relu(self.bn5(self.conv5(x)))
        x = F.adaptive_avg_pool2d(x, (1, 1))
        x = x.view(x.size(0), -1)
        x = self.fc(x)

        return x
'''
