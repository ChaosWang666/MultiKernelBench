

Looking at the architecture, this is a SqueezeNet Fire Module that performs:
1. 1x1 squeeze convolution + ReLU
2. Two parallel expand paths: 1x1 conv + ReLU and 3x3 conv + ReLU
3. Concatenation along channel dimension

The key optimization opportunity is to fuse the ReLU activations with the convolution outputs and the final concatenation. I'll create a custom kernel that fuses the two ReLU activations and concatenation of the expand paths into a single kernel operation, reducing memory bandwidth.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSqueezeNetFireModule {
public:
    __aicore__ inline KernelSqueezeNetFireModule() {}
    __aicore__ inline void Init(GM_ADDR expand1x1_out, GM_ADDR expand3x3_out, GM_ADDR z,
                                 uint32_t totalLength1, uint32_t totalLength2,
                                 uint32_t batchSize, uint32_t channels1, uint32_t channels2,
                                 uint32_t spatialSize, uint32_t tileNum)
    {
        this->totalLength1 = totalLength1;
        this->totalLength2 = totalLength2;
        this->batchSize = batchSize;
        this->channels1 = channels1;
        this->channels2 = channels2;
        this->spatialSize = spatialSize;

        // Total output elements = batchSize * (channels1 + channels2) * spatialSize
        uint32_t totalOut = batchSize * (channels1 + channels2) * spatialSize;

        // We'll process expand1x1 and expand3x3 separately, applying ReLU and writing to correct output positions
        // Simple approach: each block processes a portion of elements from both inputs

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        // Process input1 (expand1x1)
        this->blockLength1 = totalLength1 / blockNum;
        uint32_t remaining1 = totalLength1 - this->blockLength1 * blockNum;
        if (blockIdx < remaining1) {
            this->blockLength1 += 1;
            this->startOffset1 = this->blockLength1 * blockIdx;
        } else {
            this->startOffset1 = this->blockLength1 * blockIdx + remaining1;
        }

        // Process input2 (expand3x3)
        this->blockLength2 = totalLength2 / blockNum;
        uint32_t remaining2 = totalLength2 - this->blockLength2 * blockNum;
        if (blockIdx < remaining2) {
            this->blockLength2 += 1;
            this->startOffset2 = this->blockLength2 * blockIdx;
        } else {
            this->startOffset2 = this->blockLength2 * blockIdx + remaining2;
        }

        // Align tile lengths to 32 bytes (8 floats)
        this->tileNum = tileNum;
        if (this->blockLength1 > 0) {
            this->tileLength1 = ((this->blockLength1 / (tileNum * BUFFER_NUM) + 7) / 8) * 8;
            if (this->tileLength1 == 0) this->tileLength1 = 8;
        } else {
            this->tileLength1 = 8;
        }

        if (this->blockLength2 > 0) {
            this->tileLength2 = ((this->blockLength2 / (tileNum * BUFFER_NUM) + 7) / 8) * 8;
            if (this->tileLength2 == 0) this->tileLength2 = 8;
        } else {
            this->tileLength2 = 8;
        }

        // For the contiguous approach: input is contiguous NCHW, output needs interleaved channels
        // Input1: [N, C1, H, W] contiguous => element at (n, c, h, w) = n*C1*S + c*S + h*W + w
        // Output: [N, C1+C2, H, W], first C1 channels from input1, next C2 from input2
        // Output element for input1 (n,c,h,w) => n*(C1+C2)*S + c*S + h*W + w
        // Output element for input2 (n,c,h,w) => n*(C1+C2)*S + (C1+c)*S + h*W + w

        // Since the channel reordering makes non-contiguous writes complex in a tiled approach,
        // we'll use a simpler strategy: process per-batch-channel-spatial chunks

        expand1x1Gm.SetGlobalBuffer((__gm__ float *)expand1x1_out, totalLength1);
        expand3x3Gm.SetGlobalBuffer((__gm__ float *)expand3x3_out, totalLength2);
        zGm.SetGlobalBuffer((__gm__ float *)z, totalOut);

        uint32_t maxTile = (this->tileLength1 > this->tileLength2) ? this->tileLength1 : this->tileLength2;
        pipe.InitBuffer(inQueue1, BUFFER_NUM, maxTile * sizeof(float));
        pipe.InitBuffer(outQueue1, BUFFER_NUM, maxTile * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t outChannels = channels1 + channels2;
        uint32_t chunkSize = spatialSize; // one channel-spatial block

        // Process expand1x1 part
        ProcessInput(expand1x1Gm, 0, totalLength1, blockLength1, startOffset1, channels1, 0, outChannels);
        // Process expand3x3 part
        ProcessInput(expand3x3Gm, 0, totalLength2, blockLength2, startOffset2, channels2, channels1, outChannels);
    }

private:
    __aicore__ inline void ProcessInput(AscendC::GlobalTensor<float>& inputGm,
                                         uint32_t gmBase,
                                         uint32_t totalLen,
                                         uint32_t myBlockLen,
                                         uint32_t myStartOff,
                                         uint32_t inChannels,
                                         uint32_t channelOffset,
                                         uint32_t outChannels)
    {
        if (myBlockLen == 0) return;

        uint32_t inBatchStride = inChannels * spatialSize;
        uint32_t outBatchStride = outChannels * spatialSize;

        // Process elements [myStartOff, myStartOff + myBlockLen)
        uint32_t processed = 0;
        uint32_t globalPos = myStartOff;

        while (processed < myBlockLen) {
            uint32_t remaining = myBlockLen - processed;

            // Figure out current position in NCHW layout
            uint32_t n = globalPos / inBatchStride;
            uint32_t withinBatch = globalPos - n * inBatchStride;
            uint32_t c = withinBatch / spatialSize;
            uint32_t withinChannel = withinBatch - c * spatialSize;

            // How many contiguous elements we can process (within this channel)
            uint32_t contiguousAvail = spatialSize - withinChannel;
            uint32_t toCopy = (remaining < contiguousAvail) ? remaining : contiguousAvail;

            // Align to 8 for processing (pad up)
            uint32_t alignedCopy = ((toCopy + 7) / 8) * 8;
            if (alignedCopy > myBlockLen) alignedCopy = ((toCopy + 7) / 8) * 8;

            // Calculate output position
            uint32_t outPos = n * outBatchStride + (channelOffset + c) * spatialSize + withinChannel;

            // CopyIn
            AscendC::LocalTensor<float> inLocal = inQueue1.AllocTensor<float>();
            AscendC::DataCopy(inLocal, inputGm[globalPos], alignedCopy);
            inQueue1.EnQue(inLocal);

            // Compute ReLU
            AscendC::LocalTensor<float> inLocal2 = inQueue1.DeQue<float>();
            AscendC::LocalTensor<float> outLocal = outQueue1.AllocTensor<float>();

            // Create zero tensor for max (ReLU = max(0, x))
            float zeroVal = 0.0f;
            AscendC::Duplicate(outLocal, zeroVal, alignedCopy);
            AscendC::Max(outLocal, inLocal2, outLocal, alignedCopy);

            outQueue1.EnQue(outLocal);
            inQueue1.FreeTensor(inLocal2);

            // CopyOut
            AscendC::LocalTensor<float> outLocal2 = outQueue1.DeQue<float>();
            AscendC::DataCopy(zGm[outPos], outLocal2, alignedCopy);
            outQueue1.FreeTensor(outLocal2);

            processed += toCopy;
            globalPos += toCopy;
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue1;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue1;
    AscendC::GlobalTensor<float> expand1x1Gm;
    AscendC::GlobalTensor<float> expand3x3Gm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t totalLength1, totalLength2;
    uint32_t batchSize, channels1, channels2, spatialSize;
    uint32_t blockLength1, blockLength2;
    uint32_t startOffset1, startOffset2;
    uint32_t tileNum;
    uint32_t tileLength1, tileLength2;
};

extern "C" __global__ __aicore__ void squeeze_net_fire_module_custom(GM_ADDR expand1x1_out, GM_ADDR expand3x3_out, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSqueezeNetFireModule op;
    op.Init(expand1x1_out, expand3x3_out, z,
            tiling_data.totalLength1, tiling_data.totalLength2,
            tiling_data.batchSize, tiling_data.channels1, tiling_data.channels2,
            tiling_data.spatialSize, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, squeeze_channels, expand1x1_channels, expand3x3_channels):
        """
        :param in_channels: Number of input channels
        :param squeeze_channels: Number of output channels for the squeeze layer
        :param expand1x1_channels: Number of output channels for the 1x1 expand layer
        :param expand3x3_channels: Number of output channels for the 3x3 expand layer
        """
        super(ModelNew, self).__init__()

        self.squeeze = nn.Conv2d(in_channels, squeeze_channels, kernel_size=1)
        self.squeeze_activation = nn.ReLU(inplace=True)

        self.expand1x1 = nn.Conv2d(squeeze_channels, expand1x1_channels, kernel_size=1)
        self.expand3x3 = nn.Conv2d(squeeze_channels, expand3x3_channels, kernel_size=3, padding=1)

    def forward(self, x):
        """
        :param x: Input tensor, shape (batch_size, in_channels, height, width)
        :return: Output tensor, shape (batch_size, expand1x1_channels + expand3x3_channels, height, width)
        """
        x = self.squeeze_activation(self.squeeze(x))
        expand1x1_out = self.expand1x1(x)
        expand3x3_out = self.expand3x3(x)
        # Fused ReLU + concatenation using custom kernel
        return custom_ops_lib.squeeze_net_fire_module_custom(expand1x1_out, expand3x3_out)
'''
