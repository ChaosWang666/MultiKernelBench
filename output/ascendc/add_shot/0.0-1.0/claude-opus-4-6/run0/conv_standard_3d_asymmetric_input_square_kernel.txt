

Looking at this architecture, it performs a 3D convolution with an asymmetric kernel (kernel_size x kernel_size x 1). This is essentially a 2D convolution applied across the depth dimension. Implementing a full 3D convolution in AscendC from scratch is complex, so I'll create a custom operator that wraps the convolution operation efficiently on the Ascend NPU.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv3d {
public:
    __aicore__ inline KernelConv3d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t height, uint32_t width, uint32_t depth,
                                 uint32_t kernelSize, uint32_t stride, uint32_t padding,
                                 uint32_t dilation, uint32_t groups,
                                 uint32_t outHeight, uint32_t outWidth,
                                 uint32_t totalOutput, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->height = height;
        this->width = width;
        this->depth = depth;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->dilation = dilation;
        this->groups = groups;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->totalOutput = totalOutput;

        this->inChannelsPerGroup = inChannels / groups;
        this->outChannelsPerGroup = outChannels / groups;

        // Each block handles a portion of the total output elements
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->blockLength = totalOutput / blockNum;
        this->blockStart = blockIdx * this->blockLength;
        if (blockIdx == blockNum - 1) {
            this->blockLength = totalOutput - this->blockStart;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * height * width * depth);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outChannels * inChannelsPerGroup * kernelSize * kernelSize * 1);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalOutput);
    }

    __aicore__ inline void Process()
    {
        // Process each output element assigned to this block
        for (uint32_t i = 0; i < this->blockLength; i++) {
            uint32_t outIdx = this->blockStart + i;
            
            // Decompose outIdx into (n, oc, oh, ow, od)
            uint32_t rem = outIdx;
            uint32_t od = rem % depth;
            rem = rem / depth;
            uint32_t ow = rem % outWidth;
            rem = rem / outWidth;
            uint32_t oh = rem % outHeight;
            rem = rem / outHeight;
            uint32_t oc = rem % outChannels;
            uint32_t n = rem / outChannels;

            // Determine group
            uint32_t g = oc / outChannelsPerGroup;
            uint32_t ocInGroup = oc % outChannelsPerGroup;

            float sum = 0.0f;

            // Convolve over inChannelsPerGroup and kernel spatial dims (kh, kw)
            // kernel is (kernelSize, kernelSize, 1), so kd is always 0
            for (uint32_t ic = 0; ic < inChannelsPerGroup; ic++) {
                uint32_t icGlobal = g * inChannelsPerGroup + ic;
                for (uint32_t kh = 0; kh < kernelSize; kh++) {
                    for (uint32_t kw = 0; kw < kernelSize; kw++) {
                        int32_t ih = (int32_t)(oh * stride + kh * dilation) - (int32_t)padding;
                        int32_t iw = (int32_t)(ow * stride + kw * dilation) - (int32_t)padding;
                        int32_t id = (int32_t)od; // kd=0, no padding on depth

                        if (ih >= 0 && ih < (int32_t)height && iw >= 0 && iw < (int32_t)width && id >= 0 && id < (int32_t)depth) {
                            // x: (N, C, H, W, D) in row-major
                            uint32_t xOffset = ((((uint32_t)n * inChannels + icGlobal) * height + (uint32_t)ih) * width + (uint32_t)iw) * depth + (uint32_t)id;
                            // weight: (outChannels, inChannelsPerGroup, kH, kW, 1)
                            uint32_t wOffset = (((oc * inChannelsPerGroup + ic) * kernelSize + kh) * kernelSize + kw);
                            float xVal = xGm.GetValue(xOffset);
                            float wVal = wGm.GetValue(wOffset);
                            sum += xVal * wVal;
                        }
                    }
                }
            }
            yGm.SetValue(outIdx, sum);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize, inChannels, outChannels;
    uint32_t height, width, depth;
    uint32_t kernelSize, stride, padding, dilation, groups;
    uint32_t outHeight, outWidth;
    uint32_t totalOutput;
    uint32_t inChannelsPerGroup, outChannelsPerGroup;
    uint32_t blockLength, blockStart;
};

extern "C" __global__ __aicore__ void conv_standard_3d_asymmetric_input_square_kernel_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3d op;
    op.Init(x, weight, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.height, tiling_data.width, tiling_data.depth,
            tiling_data.kernelSize, tiling_data.stride, tiling_data.padding,
            tiling_data.dilation, tiling_data.groups,
            tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.totalOutput, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import math
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized 3D convolution using custom AscendC kernel.
    Performs convolution with kernel (kernel_size, kernel_size, 1).
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int,
                 stride: int = 1, padding: int = 0, dilation: int = 1,
                 groups: int = 1, bias: bool = False):
        super(ModelNew, self).__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        self.dilation = dilation
        self.groups = groups
        self.has_bias = bias

        # Create weight parameter matching nn.Conv3d layout: (out_channels, in_channels/groups, kH, kW, kD)
        self.weight = torch.nn.Parameter(
            torch.empty(out_channels, in_channels // groups, kernel_size, kernel_size, 1)
        )
        if bias:
            self.bias = torch.nn.Parameter(torch.empty(out_channels))
        else:
            self.register_parameter('bias', None)

        # Initialize weights like nn.Conv3d
        torch.nn.init.kaiming_uniform_(self.weight, a=math.sqrt(5))
        if self.bias is not None:
            fan_in, _ = torch.nn.init._calculate_fan_in_and_fan_out(self.weight)
            bound = 1 / math.sqrt(fan_in)
            torch.nn.init.uniform_(self.bias, -bound, bound)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        batch_size = x.shape[0]
        height = x.shape[2]
        width = x.shape[3]
        depth = x.shape[4]

        out_height = (height + 2 * self.padding - self.dilation * (self.kernel_size - 1) - 1) // self.stride + 1
        out_width = (width + 2 * self.padding - self.dilation * (self.kernel_size - 1) - 1) // self.stride + 1

        # Reshape weight to 4D for the kernel: (out_channels, in_channels/groups, kH, kW)
        weight_4d = self.weight.squeeze(-1).contiguous()

        result = custom_ops_lib.conv_standard_3d_asymmetric_input_square_kernel_custom(
            x.contiguous(), weight_4d,
            batch_size, self.in_channels, self.out_channels,
            height, width, depth,
            self.kernel_size, self.stride, self.padding,
            self.dilation, self.groups,
            out_height, out_width
        )

        if self.bias is not None:
            result = result + self.bias.view(1, -1, 1, 1, 1)

        return result
'''
