

Looking at this architecture, it performs: ConvTranspose3d → BatchNorm3d → AvgPool3d(2) → AvgPool3d(2). The two consecutive AvgPool3d(kernel_size=2) operations can be fused into a single custom kernel that performs average pooling with an effective kernel size of 4 (or two sequential pool operations in one pass). However, given the complexity of ConvTranspose3d and BatchNorm3d, those are best left to the existing optimized implementations. I'll fuse the two average pooling layers into a single custom AscendC kernel.

Actually, let me reconsider. Two consecutive AvgPool3d(2) with stride 2 each reduce dimensions by 4x total. Each pools over 2x2x2=8 elements. Fusing them means pooling over 4x4x4=64 elements with stride 4. This is a good fusion candidate.

However, implementing a full 3D average pooling in AscendC with proper tiling is very complex. A more practical approach is to fuse just the two avg_pool operations as a single custom operator that internally does the computation.

Let me create a custom kernel that performs two sequential 2x2x2 average pooling operations (equivalent to one 4x4x4 average pooling with stride 4) on 5D tensors.

Given the complexity, I'll implement a kernel that treats the data as flat and performs the fused pooling operation.

kernel_src="""
#include "kernel_operator.h"

// Fused double AvgPool3d(kernel_size=2): equivalent to AvgPool3d(kernel_size=4, stride=4)
// Input: [N, C, D, H, W], Output: [N, C, D/4, H/4, W/4]
// Each output element = average of 4x4x4 = 64 input elements

extern "C" __global__ __aicore__ void conv_transpose3d_batch_norm_avg_pool_avg_pool_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(t, tiling);

    uint32_t batchSize = t.batchSize;
    uint32_t channels = t.channels;
    uint32_t depth = t.depth;
    uint32_t height = t.height;
    uint32_t width = t.width;
    uint32_t outDepth = t.outDepth;
    uint32_t outHeight = t.outHeight;
    uint32_t outWidth = t.outWidth;
    uint32_t totalOutput = t.totalOutput;

    uint32_t blockIdx = AscendC::GetBlockIdx();
    uint32_t blockNum = AscendC::GetBlockNum();

    // Each block processes a portion of total output elements
    uint32_t outputPerBlock = (totalOutput + blockNum - 1) / blockNum;
    uint32_t startOut = blockIdx * outputPerBlock;
    uint32_t endOut = startOut + outputPerBlock;
    if (endOut > totalOutput) endOut = totalOutput;
    if (startOut >= totalOutput) return;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * channels * depth * height * width);
    yGm.SetGlobalBuffer((__gm__ float*)y, totalOutput);

    uint32_t outSpatial = outDepth * outHeight * outWidth;
    uint32_t inSpatial = depth * height * width;

    // Process each output element
    for (uint32_t outIdx = startOut; outIdx < endOut; outIdx++) {
        // Decode output index: [n, c, od, oh, ow]
        uint32_t remaining = outIdx;
        uint32_t ow = remaining % outWidth;
        remaining /= outWidth;
        uint32_t oh = remaining % outHeight;
        remaining /= outHeight;
        uint32_t od = remaining % outDepth;
        remaining /= outDepth;
        uint32_t c = remaining % channels;
        uint32_t n = remaining / channels;

        // Compute sum over 4x4x4 block
        float sum = 0.0f;
        uint32_t baseIdx = n * channels * inSpatial + c * inSpatial;
        uint32_t dStart = od * 4;
        uint32_t hStart = oh * 4;
        uint32_t wStart = ow * 4;

        for (uint32_t dd = 0; dd < 4; dd++) {
            for (uint32_t hh = 0; hh < 4; hh++) {
                for (uint32_t ww = 0; ww < 4; ww++) {
                    uint32_t inIdx = baseIdx + (dStart + dd) * height * width + (hStart + hh) * width + (wStart + ww);
                    sum += xGm.GetValue(inIdx);
                }
            }
        }
        // Average: divide by 64 (4*4*4)
        yGm.SetValue(outIdx, sum / 64.0f);
    }
}

import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    A model that performs a 3D transposed convolution, followed by batch normalization,
    then uses a custom fused kernel for the two average pooling layers.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, bias_shape):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.batch_norm = torch.nn.BatchNorm3d(out_channels)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.batch_norm(x)
        # Fused double AvgPool3d(kernel_size=2) using custom AscendC kernel
        x = x.contiguous()
        x = custom_ops_lib.conv_transpose3d_batch_norm_avg_pool_avg_pool_custom(x)
