kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv2dSubTanhSubAvgPool {
public:
    __aicore__ inline KernelConv2dSubTanhSubAvgPool() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalLength, uint32_t tileNum, float sub1, float sub2)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->sub1 = sub1;
        this->sub2 = sub2;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        // subtract1
        AscendC::Adds(zLocal, xLocal, -this->sub1, this->tileLength);
        // tanh
        AscendC::Tanh(zLocal, zLocal, this->tileLength);
        // subtract2
        AscendC::Adds(zLocal, zLocal, -this->sub2, this->tileLength);
        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    float sub1;
    float sub2;
};

extern "C" __global__ __aicore__ void conv2d_subtract_tanh_subtract_avg_pool_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dSubTanhSubAvgPool op;
    op.Init(x, z, tiling_data.totalLength, tiling_data.tileNum, tiling_data.subtract1_value, tiling_data.subtract2_value);
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor conv2d_subtract_tanh_subtract_avg_pool_custom_impl_npu(
    const at::Tensor& x,
    const at::Tensor& weight,
    const at::Tensor& bias,
    float subtract1_value,
    float subtract2_value,
    int64_t in_channels,
    int64_t out_channels,
    int64_t kernel_size,
    int64_t kernel_size_pool,
    int64_t batch_size,
    int64_t height,
    int64_t width) {

    // Perform conv2d using PyTorch
    auto conv_out = at::conv2d(x, weight, bias);

    // Output tensor for fused subtract-tanh-subtract
    at::Tensor fused_result = at::empty_like(conv_out);

    EXEC_NPU_CMD(aclnnConv2dSubtractTanhSubtractAvgPoolCustom, conv_out, weight, bias, fused_result,
                  subtract1_value, subtract2_value, in_channels, out_channels, kernel_size, kernel_size_pool,
                  batch_size, height, width);

    // Perform avg_pool2d using PyTorch
    auto result = at::avg_pool2d(fused_result, {kernel_size_pool, kernel_size_pool});

    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("conv2d_subtract_tanh_subtract_avg_pool_custom", &conv2d_subtract_tanh_subtract_avg_pool_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("conv2d_subtract_tanh_subtract_avg_pool_custom",
          &conv2d_subtract_tanh_subtract_avg_pool_custom_impl_npu,

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model that uses a custom AscendC kernel for fused subtract-tanh-subtract
    and uses standard PyTorch for conv2d and avg_pool2d.
    """
    def __init__(self, in_channels, out_channels, kernel_size, subtract1_value, subtract2_value, kernel_size_pool):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.subtract1_value = subtract1_value
        self.subtract2_value = subtract2_value
        self.kernel_size_pool = kernel_size_pool
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size

    def forward(self, x):
        batch_size = x.shape[0]
        height = x.shape[2]
        width = x.shape[3]
        return custom_ops_lib.conv2d_subtract_tanh_subtract_avg_pool_custom(
            x,
            self.conv.weight,
            self.conv.bias,
            self.subtract1_value,
            self.subtract2_value,
            self.in_channels,
            self.out_channels,
            self.kernel_size,
            self.kernel_size_pool,
            batch_size,
            height,
            width
