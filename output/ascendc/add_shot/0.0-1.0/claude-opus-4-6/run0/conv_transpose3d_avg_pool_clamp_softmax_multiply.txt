

Looking at the architecture, I need to analyze which operations can be fused into a custom kernel. The model does:
1. AvgPool3d
2. ConvTranspose3d
3. Clamp
4. Softmax over spatial dims
5. Multiply by scale

The AvgPool3d and ConvTranspose3d are complex operations best left to PyTorch/NPU built-in ops. I'll fuse the **clamp + softmax + multiply** into a custom AscendC kernel, as these are element-wise/reduction operations that benefit from fusion.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelClampSoftmaxMul {
public:
    __aicore__ inline KernelClampSoftmaxMul() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR z, 
                                 uint32_t batchSize, uint32_t channels, uint32_t spatialSize,
                                 float clampMin, float clampMax)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->clampMin = clampMin;
        this->clampMax = clampMax;
        
        uint32_t totalBC = batchSize * channels;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        uint32_t bcPerBlock = (totalBC + blockNum - 1) / blockNum;
        this->bcStart = blockIdx * bcPerBlock;
        this->bcEnd = (this->bcStart + bcPerBlock > totalBC) ? totalBC : (this->bcStart + bcPerBlock);
        
        // Align spatialSize to 32 bytes (8 floats)
        this->alignedSpatial = (spatialSize + 7) / 8 * 8;
        
        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * channels * spatialSize);
        scaleGm.SetGlobalBuffer((__gm__ float *)scale, channels);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * channels * spatialSize);
        
        pipe.InitBuffer(inQueue, BUFFER_NUM, this->alignedSpatial * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->alignedSpatial * sizeof(float));
        pipe.InitBuffer(tmpBuf1, this->alignedSpatial * sizeof(float));
        pipe.InitBuffer(tmpBuf2, this->alignedSpatial * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        for (uint32_t bc = this->bcStart; bc < this->bcEnd; bc++) {
            CopyIn(bc);
            Compute(bc);
            CopyOut(bc);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t bc)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[bc * spatialSize], this->alignedSpatial);
        inQueue.EnQue(xLocal);
    }
    
    __aicore__ inline void Compute(uint32_t bc)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();
        
        uint32_t len = this->alignedSpatial;
        
        // Step 1: Clamp - Mins(x, clampMax) then Maxs(x, clampMin)
        AscendC::Mins(tmp1, xLocal, this->clampMax, len);
        AscendC::Maxs(zLocal, tmp1, this->clampMin, len);
        
        // Step 2: Softmax over spatial dimension
        // Find max for numerical stability
        AscendC::ReduceMax(tmp1, zLocal, tmp2, len);
        float maxVal = tmp1.GetValue(0);
        
        // Subtract max
        AscendC::Adds(tmp1, zLocal, -maxVal, len);
        
        // Exp
        AscendC::Exp(zLocal, tmp1, len);
        
        // Sum of exp
        AscendC::ReduceSum(tmp1, zLocal, tmp2, len);
        float sumVal = tmp1.GetValue(0);
        
        // Divide by sum
        float invSum = 1.0f / sumVal;
        AscendC::Muls(tmp1, zLocal, invSum, len);
        
        // Step 3: Multiply by scale
        uint32_t c = bc % channels;
        float scaleVal = scaleGm.GetValue(c);
        AscendC::Muls(zLocal, tmp1, scaleVal, len);
        
        outQueue.EnQue(zLocal);
        inQueue.FreeTensor(xLocal);
    }
    
    __aicore__ inline void CopyOut(uint32_t bc)
    {
        AscendC::LocalTensor<float> zLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(zGm[bc * spatialSize], zLocal, this->alignedSpatial);
        outQueue.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> scaleGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t alignedSpatial;
    uint32_t bcStart;
    uint32_t bcEnd;
    float clampMin;
    float clampMax;
};

extern "C" __global__ __aicore__ void conv_transpose3d_avg_pool_clamp_softmax_multiply_custom(GM_ADDR x, GM_ADDR scale, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelClampSoftmaxMul op;
    op.Init(x, scale, z, tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize,
            tiling_data.clampMin, tiling_data.clampMax);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    """
    Optimized model that uses custom AscendC kernel for fused clamp+softmax+multiply.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, pool_kernel_size, clamp_min, clamp_max):
        super(ModelNew, self).__init__()
        self.avg_pool = nn.AvgPool3d(pool_kernel_size)
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.clamp_min = clamp_min
        self.clamp_max = clamp_max
        self.scale = nn.Parameter(torch.ones(1, out_channels, 1, 1, 1))

    def forward(self, x):
        x = self.avg_pool(x)
        x = self.conv_transpose(x)
        # Flatten scale to 1D for the kernel (channels,)
        scale_flat = self.scale.view(-1)
        # Use custom fused kernel for clamp + softmax + scale multiply
        x = custom_ops_lib.conv_transpose3d_avg_pool_clamp_softmax_multiply_custom(
            x.contiguous(), scale_flat.contiguous(), self.clamp_min, self.clamp_max)
        return x
'''
