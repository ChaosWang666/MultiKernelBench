kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelHardswishGroupNormMean {
public:
    __aicore__ inline KernelHardswishGroupNormMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                 uint32_t batchSize, uint32_t channels, uint32_t spatialSize, uint32_t numGroups)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->numGroups = numGroups;
        this->channelsPerGroup = channels / numGroups;
        
        uint32_t totalBatches = batchSize;
        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        this->batchStart = blockIdx * ((totalBatches + numBlocks - 1) / numBlocks);
        this->batchEnd = (blockIdx + 1) * ((totalBatches + numBlocks - 1) / numBlocks);
        if (this->batchEnd > totalBatches) this->batchEnd = totalBatches;
        
        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * channels * spatialSize);
        gammaGm.SetGlobalBuffer((__gm__ float *)gamma, channels);
        betaGm.SetGlobalBuffer((__gm__ float *)beta, channels);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize * channels);
        
        // Align spatialSize to 8 for float (32 bytes = 8 floats)
        uint32_t alignedSpatial = ((spatialSize + 7) / 8) * 8;
        
        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedSpatial * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, alignedSpatial * sizeof(float));
        pipe.InitBuffer(tmpBuf1, BUFFER_NUM, alignedSpatial * sizeof(float));
        pipe.InitBuffer(tmpBuf2, BUFFER_NUM, alignedSpatial * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        uint32_t alignedSpatial = ((spatialSize + 7) / 8) * 8;
        
        for (uint32_t b = batchStart; b < batchEnd; b++) {
            // Step 1: Apply hardswish to all channels for this batch
            // hardswish(x) = x * relu6(x+3) / 6
            // Step 2: GroupNorm per group
            // Step 3: Mean over spatial dims
            
            for (uint32_t g = 0; g < numGroups; g++) {
                // First pass: compute group mean and variance for hardswish output
                float groupSum = 0.0f;
                float groupSqSum = 0.0f;
                uint32_t groupElements = channelsPerGroup * spatialSize;
                
                // Compute mean of hardswish outputs for this group
                for (uint32_t c = g * channelsPerGroup; c < (g + 1) * channelsPerGroup; c++) {
                    uint32_t offset = b * channels * spatialSize + c * spatialSize;
                    
                    AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                    AscendC::DataCopy(xLocal, xGm[offset], alignedSpatial);
                    inQueueX.EnQue(xLocal);
                    xLocal = inQueueX.DeQue<float>();
                    
                    // Compute hardswish: x * clamp(x+3, 0, 6) / 6
                    AscendC::LocalTensor<float> tmpLocal = tmpBuf1.AllocTensor<float>();
                    AscendC::LocalTensor<float> tmpLocal2 = tmpBuf2.AllocTensor<float>();
                    
                    // tmp = x + 3
                    AscendC::Adds(tmpLocal, xLocal, 3.0f, alignedSpatial);
                    // clamp to [0, 6]: first max with 0
                    AscendC::Maxs(tmpLocal, tmpLocal, 0.0f, alignedSpatial);
                    // then min with 6
                    AscendC::Mins(tmpLocal, tmpLocal, 6.0f, alignedSpatial);
                    // tmp = tmp / 6
                    AscendC::Muls(tmpLocal, tmpLocal, 1.0f/6.0f, alignedSpatial);
                    // result = x * tmp
                    AscendC::Mul(tmpLocal, xLocal, tmpLocal, alignedSpatial);
                    
                    // Sum for group stats
                    float channelSum = 0.0f;
                    float channelSqSum = 0.0f;
                    // Use ReduceSum
                    AscendC::LocalTensor<float> workLocal = outQueueY.AllocTensor<float>();
                    AscendC::ReduceSum(workLocal, tmpLocal, tmpLocal2, spatialSize);
                    channelSum = workLocal.GetValue(0);
                    
                    // sq sum
                    AscendC::Mul(tmpLocal2, tmpLocal, tmpLocal, alignedSpatial);
                    AscendC::LocalTensor<float> workLocal2 = tmpBuf2.AllocTensor<float>();
                    // reuse
                    tmpBuf2.FreeTensor(workLocal2);
                    AscendC::LocalTensor<float> tmpLocal3 = tmpBuf2.AllocTensor<float>();
                    AscendC::ReduceSum(workLocal, tmpLocal2, tmpLocal3, spatialSize);
                    channelSqSum = workLocal.GetValue(0);
                    
                    groupSum += channelSum;
                    groupSqSum += channelSqSum;
                    
                    // Store hardswish result back temporarily (reuse xGm as we read it already)
                    AscendC::DataCopy(xGm[offset], tmpLocal, alignedSpatial);
                    
                    outQueueY.FreeTensor(workLocal);
                    tmpBuf1.FreeTensor(tmpLocal);
                    tmpBuf2.FreeTensor(tmpLocal3);
                    inQueueX.FreeTensor(xLocal);
                }
                
                float groupMean = groupSum / (float)groupElements;
                float groupVar = groupSqSum / (float)groupElements - groupMean * groupMean;
                float invStd = 1.0f / (float)sqrt((double)(groupVar + 1e-5f));
                
                // Second pass: normalize and compute mean over spatial
                for (uint32_t c = g * channelsPerGroup; c < (g + 1) * channelsPerGroup; c++) {
                    uint32_t offset = b * channels * spatialSize + c * spatialSize;
                    
                    AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                    AscendC::DataCopy(xLocal, xGm[offset], alignedSpatial);
                    inQueueX.EnQue(xLocal);
                    xLocal = inQueueX.DeQue<float>();
                    
                    AscendC::LocalTensor<float> tmpLocal = tmpBuf1.AllocTensor<float>();
                    AscendC::LocalTensor<float> tmpLocal2 = tmpBuf2.AllocTensor<float>();
                    
                    // normalize: (x - mean) * invStd
                    AscendC::Adds(tmpLocal, xLocal, -groupMean, alignedSpatial);
                    AscendC::Muls(tmpLocal, tmpLocal, invStd, alignedSpatial);
                    
                    // Apply gamma and beta: gamma[c] * normalized + beta[c]
                    // Read gamma and beta scalar values
                    // We need to copy gamma/beta for channel c
                    float gammaVal = 0.0f;
                    float betaVal = 0.0f;
                    
                    // Copy single values - use aligned copy
                    AscendC::LocalTensor<float> paramLocal = outQueueY.AllocTensor<float>();
                    AscendC::DataCopy(paramLocal, gammaGm[c / 8 * 8], 8);
                    gammaVal = paramLocal.GetValue(c % 8);
                    AscendC::DataCopy(paramLocal, betaGm[c / 8 * 8], 8);
                    betaVal = paramLocal.GetValue(c % 8);
                    
                    AscendC::Muls(tmpLocal, tmpLocal, gammaVal, alignedSpatial);
                    AscendC::Adds(tmpLocal, tmpLocal, betaVal, alignedSpatial);
                    
                    // Mean over spatial dimensions
                    AscendC::ReduceSum(paramLocal, tmpLocal, tmpLocal2, spatialSize);
                    float spatialMean = paramLocal.GetValue(0) / (float)spatialSize;
                    
                    // Write output
                    uint32_t outOffset = b * channels + c;
                    uint32_t alignedOutBase = (outOffset / 8) * 8;
                    AscendC::DataCopy(paramLocal, yGm[alignedOutBase], 8);
                    paramLocal.SetValue(outOffset % 8, spatialMean);
                    AscendC::DataCopy(yGm[alignedOutBase], paramLocal, 8);
                    
                    outQueueY.FreeTensor(paramLocal);
                    tmpBuf1.FreeTensor(tmpLocal);
                    tmpBuf2.FreeTensor(tmpLocal2);
                    inQueueX.FreeTensor(xLocal);
                }
            }
        }
    }
    
private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm, gammaGm, betaGm, yGm;
    uint32_t batchSize, channels, spatialSize, numGroups, channelsPerGroup;
    uint32_t batchStart, batchEnd;
};

extern "C" __global__ __aicore__ void conv3d_hardswish_relu_softmax_mean_custom(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelHardswishGroupNormMean op;
    op.Init(x, gamma, beta, y, tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize, tiling_data.numGroups);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    """
    Optimized model that performs:
    1. Conv3D (standard PyTorch)
    2. HardSwish + GroupNorm + Mean fused in custom AscendC kernel
    """
    def __init__(self, in_channels, out_channels, kernel_size, num_groups=4, bias=True):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size, bias=bias)
        self.group_norm = nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        x = self.conv(x)
        # Fused hardswish + groupnorm + spatial mean
        gamma = self.group_norm.weight
        beta = self.group_norm.bias
        x = custom_ops_lib.conv3d_hardswish_relu_softmax_mean_custom(x, gamma, beta)
        return x
