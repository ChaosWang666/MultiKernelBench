kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv3d {
public:
    __aicore__ inline KernelConv3d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inputD, uint32_t inputH, uint32_t inputW,
                                 uint32_t kernelD, uint32_t kernelH, uint32_t kernelW,
                                 uint32_t outD, uint32_t outH, uint32_t outW,
                                 uint32_t strideD, uint32_t strideH, uint32_t strideW,
                                 uint32_t padD, uint32_t padH, uint32_t padW,
                                 uint32_t dilationD, uint32_t dilationH, uint32_t dilationW,
                                 uint32_t groups, uint32_t hasBias, uint32_t totalOutputElements)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inputD = inputD;
        this->inputH = inputH;
        this->inputW = inputW;
        this->kernelD = kernelD;
        this->kernelH = kernelH;
        this->kernelW = kernelW;
        this->outD = outD;
        this->outH = outH;
        this->outW = outW;
        this->strideD = strideD;
        this->strideH = strideH;
        this->strideW = strideW;
        this->padD = padD;
        this->padH = padH;
        this->padW = padW;
        this->dilationD = dilationD;
        this->dilationH = dilationH;
        this->dilationW = dilationW;
        this->groups = groups;
        this->hasBias = hasBias;
        this->totalOutputElements = totalOutputElements;

        this->inChannelsPerGroup = inChannels / groups;
        this->outChannelsPerGroup = outChannels / groups;

        uint32_t inputSize = batchSize * inChannels * inputD * inputH * inputW;
        uint32_t weightSize = outChannels * this->inChannelsPerGroup * kernelD * kernelH * kernelW;
        uint32_t outputSize = totalOutputElements;

        xGm.SetGlobalBuffer((__gm__ float*)x, inputSize);
        wGm.SetGlobalBuffer((__gm__ float*)weight, weightSize);
        if (hasBias) {
            bGm.SetGlobalBuffer((__gm__ float*)bias, outChannels);
        }
        yGm.SetGlobalBuffer((__gm__ float*)y, outputSize);
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        // Each block processes a subset of output elements
        uint32_t elementsPerBlock = (totalOutputElements + blockNum - 1) / blockNum;
        uint32_t startIdx = blockIdx * elementsPerBlock;
        uint32_t endIdx = startIdx + elementsPerBlock;
        if (endIdx > totalOutputElements) {
            endIdx = totalOutputElements;
        }

        uint32_t outSpatial = outD * outH * outW;

        for (uint32_t idx = startIdx; idx < endIdx; idx++) {
            // Decompose flat index into (n, oc, od, oh, ow)
            uint32_t rem = idx;
            uint32_t ow = rem % outW; rem /= outW;
            uint32_t oh = rem % outH; rem /= outH;
            uint32_t od = rem % outD; rem /= outD;
            uint32_t oc = rem % outChannels; rem /= outChannels;
            uint32_t n = rem;

            uint32_t g = oc / outChannelsPerGroup;
            uint32_t oc_in_group = oc % outChannelsPerGroup;

            float sum = 0.0f;

            for (uint32_t ic = 0; ic < inChannelsPerGroup; ic++) {
                uint32_t actual_ic = g * inChannelsPerGroup + ic;
                for (uint32_t kd = 0; kd < kernelD; kd++) {
                    int32_t id = (int32_t)(od * strideD + kd * dilationD) - (int32_t)padD;
                    if (id < 0 || id >= (int32_t)inputD) continue;
                    for (uint32_t kh = 0; kh < kernelH; kh++) {
                        int32_t ih = (int32_t)(oh * strideH + kh * dilationH) - (int32_t)padH;
                        if (ih < 0 || ih >= (int32_t)inputH) continue;
                        for (uint32_t kw = 0; kw < kernelW; kw++) {
                            int32_t iw = (int32_t)(ow * strideW + kw * dilationW) - (int32_t)padW;
                            if (iw < 0 || iw >= (int32_t)inputW) continue;

                            uint32_t xIdx = n * (inChannels * inputD * inputH * inputW)
                                           + actual_ic * (inputD * inputH * inputW)
                                           + (uint32_t)id * (inputH * inputW)
                                           + (uint32_t)ih * inputW
                                           + (uint32_t)iw;

                            uint32_t wIdx = oc * (inChannelsPerGroup * kernelD * kernelH * kernelW)
                                           + ic * (kernelD * kernelH * kernelW)
                                           + kd * (kernelH * kernelW)
                                           + kh * kernelW
                                           + kw;

                            float xVal = xGm.GetValue(xIdx);
                            float wVal = wGm.GetValue(wIdx);
                            sum += xVal * wVal;
                        }
                    }
                }
            }

            if (hasBias) {
                sum += bGm.GetValue(oc);
            }

            yGm.SetValue(idx, sum);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize, inChannels, outChannels;
    uint32_t inputD, inputH, inputW;
    uint32_t kernelD, kernelH, kernelW;
    uint32_t outD, outH, outW;
    uint32_t strideD, strideH, strideW;
    uint32_t padD, padH, padW;
    uint32_t dilationD, dilationH, dilationW;
    uint32_t groups, hasBias;
    uint32_t totalOutputElements;
    uint32_t inChannelsPerGroup, outChannelsPerGroup;
};

extern "C" __global__ __aicore__ void conv_standard_3d_asymmetric_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3d op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inputD, tiling_data.inputH, tiling_data.inputW,
            tiling_data.kernelD, tiling_data.kernelH, tiling_data.kernelW,
            tiling_data.outD, tiling_data.outH, tiling_data.outW,
            tiling_data.strideD, tiling_data.strideH, tiling_data.strideW,
            tiling_data.padD, tiling_data.padH, tiling_data.padW,
            tiling_data.dilationD, tiling_data.dilationH, tiling_data.dilationW,
            tiling_data.groups, tiling_data.hasBias, tiling_data.totalOutputElements);
    op.Process();
}

        self.padding = padding
        self.dilation = dilation
        self.groups = groups
        self.has_bias = bias

    def forward(self, x: torch.Tensor) -> torch.Tensor:

        bias_tensor = self.conv3d.bias

            x, weight, bias_tensor,
            self.stride[0], self.stride[1], self.stride[2],
            self.padding[0], self.padding[1], self.padding[2],
            self.dilation[0], self.dilation[1], self.dilation[2],
