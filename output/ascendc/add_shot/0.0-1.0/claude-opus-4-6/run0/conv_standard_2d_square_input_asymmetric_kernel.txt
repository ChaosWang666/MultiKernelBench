

Looking at this architecture, I need to implement a 2D convolution with square input and asymmetric kernel using custom AscendC kernels. The convolution is a standard operation that takes input (batch_size, in_channels, H, W) and produces output (batch_size, out_channels, H_out, W_out).

Given the complexity of implementing a full 2D convolution in AscendC (which involves nested loops, weight management, and careful tiling), I'll implement this using the im2col + matrix multiplication approach, which is more amenable to hardware acceleration on Ascend AI cores.

However, given the practical constraints, I'll implement a direct convolution kernel that processes the data in tiles.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv2d {
public:
    __aicore__ inline KernelConv2d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR y,
                                uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                uint32_t inputH, uint32_t inputW,
                                uint32_t kernelH, uint32_t kernelW,
                                uint32_t outputH, uint32_t outputW,
                                uint32_t stride, uint32_t padding,
                                uint32_t dilation, uint32_t groups,
                                uint32_t totalOutput, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inputH = inputH;
        this->inputW = inputW;
        this->kernelH = kernelH;
        this->kernelW = kernelW;
        this->outputH = outputH;
        this->outputW = outputW;
        this->strideVal = stride;
        this->paddingVal = padding;
        this->dilationVal = dilation;
        this->groups = groups;
        this->totalOutput = totalOutput;

        // Each block processes a portion of output elements
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        // Distribute output channels across blocks
        this->outCPerBlock = (outChannels + blockNum - 1) / blockNum;
        this->startOutC = blockIdx * this->outCPerBlock;
        if (this->startOutC >= outChannels) {
            this->outCPerBlock = 0;
        } else if (this->startOutC + this->outCPerBlock > outChannels) {
            this->outCPerBlock = outChannels - this->startOutC;
        }

        uint32_t inputSize = batchSize * inChannels * inputH * inputW;
        uint32_t weightSize = outChannels * inChannels * kernelH * kernelW;
        uint32_t outputSize = batchSize * outChannels * outputH * outputW;

        xGm.SetGlobalBuffer((__gm__ float *)x, inputSize);
        wGm.SetGlobalBuffer((__gm__ float *)weight, weightSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, outputSize);

        // Allocate local buffers for a row of output and kernel weights
        uint32_t alignedW = ((outputW + 7) / 8) * 8;
        pipe.InitBuffer(outQueue, BUFFER_NUM, alignedW * sizeof(float));
        pipe.InitBuffer(inQueue, BUFFER_NUM, alignedW * sizeof(float));
        pipe.InitBuffer(tmpBuf, BUFFER_NUM, alignedW * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->outCPerBlock == 0) return;

        for (uint32_t b = 0; b < batchSize; b++) {
            for (uint32_t oc = this->startOutC; oc < this->startOutC + this->outCPerBlock; oc++) {
                for (uint32_t oh = 0; oh < outputH; oh++) {
                    ComputeRow(b, oc, oh);
                }
            }
        }
    }

private:
    __aicore__ inline void ComputeRow(uint32_t b, uint32_t oc, uint32_t oh)
    {
        uint32_t alignedW = ((outputW + 7) / 8) * 8;

        // Allocate output accumulator and zero it
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::Duplicate(outLocal, (float)0, alignedW);

        for (uint32_t ic = 0; ic < inChannels; ic++) {
            for (uint32_t kh = 0; kh < kernelH; kh++) {
                for (uint32_t kw = 0; kw < kernelW; kw++) {
                    uint32_t ih = oh * strideVal + kh * dilationVal - paddingVal;
                    if (ih >= inputH) continue;

                    float wVal = wGm[oc * inChannels * kernelH * kernelW +
                                     ic * kernelH * kernelW +
                                     kh * kernelW + kw];

                    // Load input row and multiply
                    AscendC::LocalTensor<float> inLocal = inQueue.AllocTensor<float>();
                    AscendC::LocalTensor<float> tmpLocal = tmpBuf.AllocTensor<float>();

                    // We need to gather input values for each output column
                    // For ow in [0, outputW): input_col = ow * stride + kw * dilation - padding
                    // Copy contiguous section from input
                    uint32_t baseIW = kw * dilationVal - paddingVal;
                    uint32_t inputRowBase = b * inChannels * inputH * inputW +
                                           ic * inputH * inputW +
                                           ih * inputW;

                    // When stride==1 and padding==0 and dilation==1, input cols are contiguous starting at baseIW
                    if (strideVal == 1 && dilationVal == 1) {
                        int32_t srcStart = (int32_t)baseIW;
                        // Clamp and copy
                        if (srcStart >= 0 && srcStart + outputW <= inputW) {
                            AscendC::DataCopy(inLocal, xGm[inputRowBase + srcStart], alignedW);
                            AscendC::SetMaskCount();
                            AscendC::SetVectorMask(alignedW);
                            AscendC::Muls(tmpLocal, inLocal, wVal, alignedW);
                            AscendC::Add(outLocal, outLocal, tmpLocal, alignedW);
                            AscendC::ResetMask();
                        } else {
                            // Handle boundary: set to 0 then copy valid parts
                            AscendC::Duplicate(inLocal, (float)0, alignedW);
                            if (srcStart >= 0 && (uint32_t)srcStart < inputW) {
                                uint32_t validLen = inputW - srcStart;
                                if (validLen > outputW) validLen = outputW;
                                uint32_t alignedValid = ((validLen + 7) / 8) * 8;
                                AscendC::DataCopy(inLocal, xGm[inputRowBase + srcStart], alignedValid);
                            } else if (srcStart < 0 && (int32_t)(srcStart + outputW) > 0) {
                                uint32_t offset = (uint32_t)(-srcStart);
                                uint32_t validLen = outputW - offset;
                                if (validLen > inputW) validLen = inputW;
                                uint32_t alignedValid = ((validLen + 7) / 8) * 8;
                                // Cannot easily offset local tensor for DataCopy, handle scalar
                                for (uint32_t ow = 0; ow < outputW; ow++) {
                                    int32_t iw = (int32_t)ow + srcStart;
                                    if (iw >= 0 && (uint32_t)iw < inputW) {
                                        inLocal.SetValue(ow, xGm[inputRowBase + iw]);
                                    }
                                }
                            }
                            AscendC::SetMaskCount();
                            AscendC::SetVectorMask(alignedW);
                            AscendC::Muls(tmpLocal, inLocal, wVal, alignedW);
                            AscendC::Add(outLocal, outLocal, tmpLocal, alignedW);
                            AscendC::ResetMask();
                        }
                    } else {
                        // General case: scalar gather
                        AscendC::Duplicate(inLocal, (float)0, alignedW);
                        for (uint32_t ow = 0; ow < outputW; ow++) {
                            int32_t iw = (int32_t)(ow * strideVal) + (int32_t)(kw * dilationVal) - (int32_t)paddingVal;
                            if (iw >= 0 && (uint32_t)iw < inputW) {
                                inLocal.SetValue(ow, xGm[inputRowBase + iw]);
                            }
                        }
                        AscendC::SetMaskCount();
                        AscendC::SetVectorMask(alignedW);
                        AscendC::Muls(tmpLocal, inLocal, wVal, alignedW);
                        AscendC::Add(outLocal, outLocal, tmpLocal, alignedW);
                        AscendC::ResetMask();
                    }

                    tmpBuf.FreeTensor(tmpLocal);
                    inQueue.FreeTensor(inLocal);
                }
            }
        }

        // Write output row
        uint32_t outBase = b * outChannels * outputH * outputW +
                          oc * outputH * outputW +
                          oh * outputW;
        AscendC::DataCopy(yGm[outBase], outLocal, alignedW);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpBuf;
    AscendC::GlobalTensor<float> xGm, wGm, yGm;

    uint32_t batchSize, inChannels, outChannels;
    uint32_t inputH, inputW, kernelH, kernelW;
    uint32_t outputH, outputW;
    uint32_t strideVal, paddingVal, dilationVal, groups;
    uint32_t totalOutput;
    uint32_t outCPerBlock, startOutC;
};

extern "C" __global__ __aicore__ void conv_standard_2d_square_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2d op;
    op.Init(x, weight, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inputH, tiling_data.inputW,
            tiling_data.kernelH, tiling_data.kernelW,
            tiling_data.outputH, tiling_data.outputW,
            tiling_data.stride, tiling_data.padding,
            tiling_data.dilation, tiling_data.groups,
            tiling_data.totalOutput, tiling_data.tileNum);
    op.Process();
}
