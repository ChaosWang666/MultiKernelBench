class KernelDropoutMeanSoftmax {
public:
    __aicore__ inline KernelDropoutMeanSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mask, GM_ADDR z,
                                 uint32_t batchSize, uint32_t outFeatures, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->outFeatures = outFeatures;

        uint32_t totalRows = batchSize;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->rowsPerBlock = (totalRows + blockNum - 1) / blockNum;
        this->rowStart = blockIdx * this->rowsPerBlock;
        if (this->rowStart + this->rowsPerBlock > totalRows) {
            this->rowsPerBlock = (totalRows > this->rowStart) ? (totalRows - this->rowStart) : 0;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * outFeatures);
        maskGm.SetGlobalBuffer((__gm__ float*)mask, batchSize * outFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z, batchSize * outFeatures);

        uint32_t alignedFeatures = ((outFeatures + 7) / 8) * 8;
        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedFeatures * sizeof(float));
        pipe.InitBuffer(inQueueM, BUFFER_NUM, alignedFeatures * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, alignedFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf1, 1, alignedFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf2, 1, alignedFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsPerBlock; i++) {
            uint32_t globalRow = this->rowStart + i;
            if (globalRow >= this->batchSize) break;
            CopyIn(globalRow);
            Compute(globalRow);
            CopyOut(globalRow);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row)
    {
        uint32_t alignedFeatures = ((outFeatures + 7) / 8) * 8;
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> mLocal = inQueueM.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * outFeatures], alignedFeatures);
        AscendC::DataCopy(mLocal, maskGm[row * outFeatures], alignedFeatures);
        inQueueX.EnQue(xLocal);
        inQueueM.EnQue(mLocal);
    }

    __aicore__ inline void Compute(uint32_t row)
    {
        uint32_t alignedFeatures = ((outFeatures + 7) / 8) * 8;
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> mLocal = inQueueM.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();

        // Step 1: Apply dropout mask: x = x * mask * scale(1.25 for p=0.2)
        AscendC::Mul(zLocal, xLocal, mLocal, alignedFeatures);

        AscendC::Muls(zLocal, zLocal, scale, alignedFeatures);

        // Step 2: Compute mean for numerical stability and subtract
        // Find max for numerical stability of softmax
        AscendC::ReduceMax(tmp1, zLocal, tmp2, alignedFeatures);
        float maxVal = tmp1.GetValue(0);
        AscendC::Adds(zLocal, zLocal, -maxVal, alignedFeatures);

        // Step 3: exp
        AscendC::Exp(zLocal, zLocal, alignedFeatures);

        // Step 4: sum of exp
        AscendC::ReduceSum(tmp1, zLocal, tmp2, alignedFeatures);
        float sumVal = tmp1.GetValue(0);

        // Step 5: divide by sum
        if (sumVal > 0.0f) {
            float invSum = 1.0f / sumVal;
            AscendC::Muls(zLocal, zLocal, invSum, alignedFeatures);
        }

        outQueueZ.EnQue(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueM.FreeTensor(mLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row)
    {
        uint32_t alignedFeatures = ((outFeatures + 7) / 8) * 8;
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[row * outFeatures], zLocal, alignedFeatures);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueM;
