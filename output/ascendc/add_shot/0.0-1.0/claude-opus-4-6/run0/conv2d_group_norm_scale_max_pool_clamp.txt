

Looking at this architecture, I need to analyze the operations: Conv2d → GroupNorm → Scale → MaxPool2d → Clamp. 

Given the complexity of Conv2d and GroupNorm (which involve weight matrices, running stats, etc.), the most practical optimization is to fuse the post-convolution operations (GroupNorm + Scale + MaxPool + Clamp) or at least fuse the simpler element-wise operations. However, implementing full Conv2d or GroupNorm in AscendC from scratch would be extremely complex and error-prone.

A practical approach is to fuse the Scale + MaxPool + Clamp operations into a custom kernel, while keeping Conv2d and GroupNorm as standard PyTorch operations. But for a more meaningful fusion, I'll fuse Scale, MaxPool2d, and Clamp into one kernel.

Let me implement a fused kernel that takes the group-normed tensor, applies scaling, max pooling, and clamping in one pass.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelScaleMaxPoolClamp {
public:
    __aicore__ inline KernelScaleMaxPoolClamp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR z,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t height, uint32_t width,
                                 uint32_t poolKernel,
                                 uint32_t outHeight, uint32_t outWidth,
                                 float clampMin, float clampMax,
                                 uint32_t totalOutputLength, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->height = height;
        this->width = width;
        this->poolKernel = poolKernel;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->clampMin = clampMin;
        this->clampMax = clampMax;

        // Each block processes a subset of (batch, channel) pairs
        uint32_t totalBC = batchSize * channels;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->bcStart = (totalBC * blockIdx) / blockNum;
        this->bcEnd = (totalBC * (blockIdx + 1)) / blockNum;
        this->bcCount = this->bcEnd - this->bcStart;

        uint32_t inSpatial = height * width;
        uint32_t outSpatial = outHeight * outWidth;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * channels * inSpatial);
        scaleGm.SetGlobalBuffer((__gm__ float *)scale, channels);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * channels * outSpatial);

        // Allocate buffers for one row of pooling windows in width direction
        // We process one output row at a time: outWidth elements
        // For input we need poolKernel rows of width elements
        // We'll process per output pixel in a simple loop
        // Use buffer for one row of output: outWidth elements, aligned to 32 bytes
        uint32_t alignedOutWidth = ((outWidth + 7) / 8) * 8;
        this->alignedOutWidth = alignedOutWidth;

        pipe.InitBuffer(inQueueX, 1, alignedOutWidth * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, alignedOutWidth * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t bc = this->bcStart; bc < this->bcEnd; bc++) {
            uint32_t b = bc / this->channels;
            uint32_t c = bc % this->channels;

            uint32_t inOffset = (b * this->channels + c) * this->height * this->width;
            uint32_t outOffset = (b * this->channels + c) * this->outHeight * this->outWidth;

            // Read scale for this channel
            // Since scale shape is (channels, 1, 1), just index by c
            // We can't easily do single-element GM read with AscendC DataCopy
            // So we'll handle scale in compute

            for (uint32_t oh = 0; oh < this->outHeight; oh++) {
                ProcessRow(inOffset, outOffset, oh, c);
            }
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t inOffset, uint32_t outOffset, uint32_t oh, uint32_t c)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        // For each output position in this row, compute max over poolKernel x poolKernel window
        for (uint32_t ow = 0; ow < this->outWidth; ow++) {
            float maxVal = -3.402823e+38f; // -FLT_MAX
            for (uint32_t ph = 0; ph < this->poolKernel; ph++) {
                for (uint32_t pw = 0; pw < this->poolKernel; pw++) {
                    uint32_t ih = oh * this->poolKernel + ph;
                    uint32_t iw = ow * this->poolKernel + pw;
                    uint32_t idx = inOffset + ih * this->width + iw;
                    // Scalar read from GM
                    float val = *(((__gm__ float*)xGm.GetPhyAddr()) + idx);
                    if (val > maxVal) {
                        maxVal = val;
                    }
                }
            }
            // Apply scale
            float scaleVal = *(((__gm__ float*)scaleGm.GetPhyAddr()) + c);
            maxVal = maxVal * scaleVal;

            // Apply clamp
            if (maxVal < this->clampMin) maxVal = this->clampMin;
            if (maxVal > this->clampMax) maxVal = this->clampMax;

            zLocal.SetValue(ow, maxVal);
        }
        // Pad remaining
        for (uint32_t ow = this->outWidth; ow < this->alignedOutWidth; ow++) {
            zLocal.SetValue(ow, 0.0f);
        }

        outQueueZ.EnQue(zLocal);
        AscendC::LocalTensor<float> zOut = outQueueZ.DeQue<float>();

        // Copy out - need aligned size, but only outWidth meaningful
        // DataCopy requires 32-byte aligned count (8 floats)
        AscendC::DataCopy(zGm[outOffset + oh * this->outWidth], zOut, this->alignedOutWidth);
        outQueueZ.FreeTensor(zOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> scaleGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t batchSize, channels, height, width;
    uint32_t poolKernel, outHeight, outWidth;
    float clampMin, clampMax;
    uint32_t bcStart, bcEnd, bcCount;
    uint32_t alignedOutWidth;
};

extern "C" __global__ __aicore__ void conv2d_group_norm_scale_max_pool_clamp_custom(GM_ADDR x, GM_ADDR scale, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelScaleMaxPoolClamp op;
    op.Init(x, scale, z,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.height, tiling_data.width,
            tiling_data.poolKernel,
            tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.clampMin, tiling_data.clampMax,
            tiling_data.totalOutputLength, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    """
    Optimized model that performs convolution, group normalization, then uses a custom
    fused kernel for scaling, max pooling, and clamping.
    """
    def __init__(self, in_channels, out_channels, kernel_size, num_groups, scale_shape, maxpool_kernel_size, clamp_min, clamp_max):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.group_norm = nn.GroupNorm(num_groups, out_channels)
        self.scale = nn.Parameter(torch.ones(scale_shape))
        self.maxpool_kernel_size = maxpool_kernel_size
        self.clamp_min = clamp_min
        self.clamp_max = clamp_max

    def forward(self, x):
        x = self.conv(x)
        x = self.group_norm(x)
        # Fused: scale * x -> maxpool -> clamp
        x = custom_ops_lib.conv2d_group_norm_scale_max_pool_clamp_custom(x, self.scale)
        return x
'''
