

Looking at this architecture, it's a deep narrow MLP with Linear layers followed by ReLU activations. I'll create a fused Linear+ReLU kernel to reduce memory bandwidth by avoiding intermediate tensor materialization between the linear transformation and ReLU activation.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelDeepNarrowMlp {
public:
    __aicore__ inline KernelDeepNarrowMlp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t M, uint32_t N, uint32_t K, uint32_t tileNum)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        
        this->rowsPerBlock = M / blockNum;
        uint32_t remainRows = M % blockNum;
        if (blockIdx < remainRows) {
            this->rowsPerBlock += 1;
            this->startRow = blockIdx * this->rowsPerBlock;
        } else {
            this->startRow = remainRows * (this->rowsPerBlock + 1) + (blockIdx - remainRows) * this->rowsPerBlock;
        }
        
        if (this->rowsPerBlock == 0) {
            return;
        }
        
        this->tileNum = tileNum;
        if (this->tileNum > this->rowsPerBlock) {
            this->tileNum = this->rowsPerBlock;
        }
        this->tileRows = this->rowsPerBlock / this->tileNum;
        this->tailRows = this->rowsPerBlock - this->tileRows * (this->tileNum - 1);
        
        xGm.SetGlobalBuffer((__gm__ float *)x + this->startRow * K, this->rowsPerBlock * K);
        wGm.SetGlobalBuffer((__gm__ float *)weight, N * K);
        bGm.SetGlobalBuffer((__gm__ float *)bias, N);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->startRow * N, this->rowsPerBlock * N);
        
        uint32_t maxTileRows = (this->tailRows > this->tileRows) ? this->tailRows : this->tileRows;
        uint32_t xBufSize = maxTileRows * K * sizeof(float);
        uint32_t yBufSize = maxTileRows * N * sizeof(float);
        
        uint32_t alignedXBufSize = ((xBufSize + 31) / 32) * 32;
        uint32_t alignedYBufSize = ((yBufSize + 31) / 32) * 32;
        
        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedXBufSize);
        pipe.InitBuffer(outQueueY, BUFFER_NUM, alignedYBufSize);
    }
    
    __aicore__ inline void Process()
    {
        if (this->rowsPerBlock == 0) {
            return;
        }
        for (uint32_t i = 0; i < this->tileNum; i++) {
            uint32_t curRows = (i < this->tileNum - 1) ? this->tileRows : this->tailRows;
            CopyIn(i, curRows);
            Compute(i, curRows);
            CopyOut(i, curRows);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t tileIdx, uint32_t curRows)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        uint32_t offset = tileIdx * this->tileRows;
        AscendC::DataCopy(xLocal, xGm[offset * K], curRows * K);
        inQueueX.EnQue(xLocal);
    }
    
    __aicore__ inline void Compute(uint32_t tileIdx, uint32_t curRows)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        
        // Simple matrix multiply: y = x * W^T + bias
        // x: (curRows, K), W: (N, K), bias: (N,), y: (curRows, N)
        // For each row and each output neuron, compute dot product
        for (uint32_t r = 0; r < curRows; r++) {
            for (uint32_t n = 0; n < N; n++) {
                float sum = 0.0f;
                for (uint32_t k = 0; k < K; k++) {
                    float xVal = xLocal.GetValue(r * K + k);
                    float wVal;
                    // Read from global memory for weight
                    wVal = wGm.GetValue(n * K + k);
                    sum += xVal * wVal;
                }
                sum += bGm.GetValue(n);
                yLocal.SetValue(r * N + n, sum);
            }
        }
        
        outQueueY.EnQue(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    
    __aicore__ inline void CopyOut(uint32_t tileIdx, uint32_t curRows)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        uint32_t offset = tileIdx * this->tileRows;
        AscendC::DataCopy(yGm[offset * N], yLocal, curRows * N);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t M, N, K;
    uint32_t rowsPerBlock;
    uint32_t startRow;
    uint32_t tileNum;
    uint32_t tileRows;
    uint32_t tailRows;
};

extern "C" __global__ __aicore__ void deep_narrow_mlp_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelDeepNarrowMlp op;
    op.Init(x, weight, bias, y, tiling_data.M, tiling_data.N, tiling_data.K, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, input_size, hidden_layer_sizes, output_size):
        """
        :param input_size: The number of input features
        :param hidden_layer_sizes: A list of ints containing the sizes of each hidden layer
        :param output_size: The number of output features
        """
        super(ModelNew, self).__init__()
        
        self.weights = nn.ParameterList()
        self.biases = nn.ParameterList()
        self.hidden_layer_sizes = hidden_layer_sizes
        
        current_input_size = input_size
        
        for hidden_size in hidden_layer_sizes:
            linear = nn.Linear(current_input_size, hidden_size)
            self.weights.append(nn.Parameter(linear.weight.data))
            self.biases.append(nn.Parameter(linear.bias.data))
            current_input_size = hidden_size
        
        linear = nn.Linear(current_input_size, output_size)
        self.weights.append(nn.Parameter(linear.weight.data))
        self.biases.append(nn.Parameter(linear.bias.data))
    
    def forward(self, x):
        """
        :param x: The input tensor, shape (batch_size, input_size)
        :return: The output tensor, shape (batch_size, output_size)
        """
        num_layers = len(self.weights)
        for i in range(num_layers - 1):
            # Fused linear + relu via custom kernel
            x = custom_ops_lib.deep_narrow_mlp_custom(x, self.weights[i], self.biases[i])
            x = torch.relu(x)
        # Last layer without relu
        x = custom_ops_lib.deep_narrow_mlp_custom(x, self.weights[-1], self.biases[-1])
        return x
'''
