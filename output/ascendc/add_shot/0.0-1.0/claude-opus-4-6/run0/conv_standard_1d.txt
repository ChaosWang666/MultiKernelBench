kernel_src="""
#include "kernel_operator.h"

extern "C" __global__ __aicore__ void conv_standard_1d_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);

    uint32_t batchSize = tiling_data.batchSize;
    uint32_t inChannels = tiling_data.inChannels;
    uint32_t outChannels = tiling_data.outChannels;
    uint32_t inLength = tiling_data.inLength;
    uint32_t outLength = tiling_data.outLength;
    uint32_t kernelSize = tiling_data.kernelSize;
    uint32_t stride = tiling_data.stride;
    uint32_t padding = tiling_data.padding;
    uint32_t dilation = tiling_data.dilation;
    uint32_t groups = tiling_data.groups;

    uint32_t totalOutElements = batchSize * outChannels * outLength;
    uint32_t blockNum = AscendC::GetBlockNum();
    uint32_t blockIdx = AscendC::GetBlockIdx();

    uint32_t elementsPerBlock = (totalOutElements + blockNum - 1) / blockNum;
    uint32_t startIdx = blockIdx * elementsPerBlock;
    uint32_t endIdx = startIdx + elementsPerBlock;
    if (endIdx > totalOutElements) endIdx = totalOutElements;

    uint32_t inChannelsPerGroup = inChannels / groups;
    uint32_t outChannelsPerGroup = outChannels / groups;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> yGm;
    xGm.SetGlobalBuffer((__gm__ float*)x);
    wGm.SetGlobalBuffer((__gm__ float*)weight);
    yGm.SetGlobalBuffer((__gm__ float*)y);

    // Process each output element assigned to this block
    for (uint32_t idx = startIdx; idx < endIdx; idx++) {
        uint32_t oPos = idx % outLength;
        uint32_t oc = (idx / outLength) % outChannels;
        uint32_t b = idx / (outChannels * outLength);

        uint32_t g = oc / outChannelsPerGroup;
        uint32_t ocInGroup = oc % outChannelsPerGroup;

        float sum = 0.0f;
        for (uint32_t ic = 0; ic < inChannelsPerGroup; ic++) {
            uint32_t globalIc = g * inChannelsPerGroup + ic;
            for (uint32_t k = 0; k < kernelSize; k++) {
                int32_t inPos = (int32_t)(oPos * stride) - (int32_t)padding + (int32_t)(k * dilation);
                if (inPos >= 0 && inPos < (int32_t)inLength) {
                    uint32_t xIdx = b * inChannels * inLength + globalIc * inLength + (uint32_t)inPos;
                    uint32_t wIdx = oc * inChannelsPerGroup * kernelSize + ic * kernelSize + k;
                    float xVal = xGm.GetValue(xIdx);
                    float wVal = wGm.GetValue(wIdx);
                    sum += xVal * wVal;
                }
            }
        }
        yGm.SetValue(idx, sum);
    }
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, stride: int = 1, padding: int = 0, dilation: int = 1, groups: int = 1, bias: bool = False):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        self.dilation = dilation
        self.groups = groups
        # Use nn.Conv1d to initialize weights properly, then extract them
        conv_tmp = torch.nn.Conv1d(in_channels, out_channels, kernel_size,
                                    stride=stride, padding=padding, dilation=dilation,
                                    groups=groups, bias=bias)
        self.weight = torch.nn.Parameter(conv_tmp.weight.clone())
        if bias:
            self.bias = torch.nn.Parameter(conv_tmp.bias.clone())
        else:
            self.bias = None

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        batch_size = x.shape[0]
        in_length = x.shape[2]
        result = custom_ops_lib.conv_standard_1d_custom(
            x, self.weight,
            batch_size, self.in_channels, self.out_channels,
            in_length, self.kernel_size, self.stride,
            self.padding, self.dilation, self.groups
        )
        if self.bias is not None:
            result = result + self.bias.unsqueeze(0).unsqueeze(2)
        return result
'''
