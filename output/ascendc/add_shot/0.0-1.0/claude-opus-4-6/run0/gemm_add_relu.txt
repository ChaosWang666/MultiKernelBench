kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmAddRelu {
public:
    __aicore__ inline KernelGemmAddRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                 uint32_t M, uint32_t N, uint32_t K, uint32_t tileNum)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        this->tileNum = tileNum;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        // Each block processes a subset of rows
        this->rowsPerBlock = M / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        if (blockIdx == blockNum - 1) {
            this->rowsPerBlock = M - this->startRow;
        }

        // tile rows
        this->rowsPerTile = this->rowsPerBlock / tileNum;
        if (this->rowsPerTile < 1) this->rowsPerTile = 1;

        xGm.SetGlobalBuffer((__gm__ float *)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float *)weight, N * K);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, N);
        zGm.SetGlobalBuffer((__gm__ float *)z, M * N);

        // Allocate buffers for one row of x (K floats), one column tile of results, and bias
        uint32_t xBufSize = K * sizeof(float);
        uint32_t biasBufSize = N * sizeof(float);
        // We'll process one row at a time, computing dot products with all N weight rows
        // For each row of x, we need K floats; for each weight row, K floats; output N floats
        // We'll tile over N as well to fit in local memory

        // Determine tile size for N dimension
        // Local memory is limited, so we tile over N
        // Each weight tile: tileN * K floats
        // We want tileN * K * 4 bytes to fit comfortably
        this->tileN = 256; // process 256 output features at a time
        if (this->tileN > N) this->tileN = N;

        // Align tileN to 8 for vectorized operations
        this->tileN = (this->tileN / 8) * 8;
        if (this->tileN == 0) this->tileN = 8;

        // tile K for dot product accumulation
        this->tileK = 256;
        if (this->tileK > K) this->tileK = K;
        this->tileK = (this->tileK / 8) * 8;
        if (this->tileK == 0) this->tileK = 8;

        // Allocate pipe buffers
        // xBuf: tileK floats for a chunk of one row of x
        // wBuf: tileN * tileK floats for a chunk of weights
        // accBuf: tileN floats for accumulated results
        // biasBuf: tileN floats for bias chunk
        // outBuf: tileN floats for output after add+relu
        pipe.InitBuffer(inQueueX, 1, this->tileK * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, this->tileN * this->tileK * sizeof(float));
        pipe.InitBuffer(inQueueBias, 1, this->tileN * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, this->tileN * sizeof(float));
        pipe.InitBuffer(accBuf, 1, this->tileN * sizeof(float));
        pipe.InitBuffer(tmpBuf, 1, this->tileN * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t row = 0; row < this->rowsPerBlock; row++) {
            uint32_t globalRow = this->startRow + row;
            // Process all output features in tiles of tileN
            for (uint32_t nOff = 0; nOff < N; nOff += this->tileN) {
                uint32_t curTileN = this->tileN;
                if (nOff + curTileN > N) curTileN = N - nOff;

                // Initialize accumulator to zero
                AscendC::LocalTensor<float> accLocal = accBuf.Get<float>();
                AscendC::Duplicate(accLocal, (float)0.0f, curTileN);

                // Accumulate dot product over K dimension
                for (uint32_t kOff = 0; kOff < K; kOff += this->tileK) {
                    uint32_t curTileK = this->tileK;
                    if (kOff + curTileK > K) curTileK = K - kOff;

                    // Load x[globalRow, kOff:kOff+curTileK]
                    AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                    AscendC::DataCopy(xLocal, xGm[globalRow * K + kOff], curTileK);
                    inQueueX.EnQue(xLocal);
                    xLocal = inQueueX.DeQue<float>();

                    // For each weight row in [nOff, nOff+curTileN), compute partial dot product
                    // weight is [N, K], so weight[n, k] = wGm[n * K + k]
                    // Load weight block: curTileN rows, each curTileK elements
                    AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                    // Copy curTileN rows of curTileK elements each
                    for (uint32_t ni = 0; ni < curTileN; ni++) {
                        // Copy one row segment of weight
                        AscendC::DataCopy(wLocal[ni * curTileK], wGm[(nOff + ni) * K + kOff], curTileK);
                    }
                    inQueueW.EnQue(wLocal);
                    wLocal = inQueueW.DeQue<float>();

                    // Compute partial dot products
                    AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
                    for (uint32_t ni = 0; ni < curTileN; ni++) {
                        // Multiply element-wise x[kOff:kOff+curTileK] * w[ni, kOff:kOff+curTileK]
                        AscendC::Mul(tmpLocal, xLocal, wLocal[ni * curTileK], curTileK);
                        // Sum the products - use ReduceSum if available, otherwise accumulate
                        float sum = 0.0f;
                        // We'll use vector reduce
                        AscendC::LocalTensor<float> sumTensor = tmpLocal;
                        // Manual reduction: not ideal but functional
                        // Use WholeReduceSum
                        float partialSum = 0.0f;
                        for (uint32_t ki = 0; ki < curTileK; ki++) {
                            partialSum += tmpLocal.GetValue(ki);
                        }
                        accLocal.SetValue(ni, accLocal.GetValue(ni) + partialSum);
                    }

                    inQueueX.FreeTensor(xLocal);
                    inQueueW.FreeTensor(wLocal);
                }

                // Add bias
                AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
                AscendC::DataCopy(biasLocal, biasGm[nOff], curTileN);
                inQueueBias.EnQue(biasLocal);
                biasLocal = inQueueBias.DeQue<float>();

                AscendC::LocalTensor<float> outLocal = outQueueZ.AllocTensor<float>();
                AscendC::Add(outLocal, accLocal, biasLocal, curTileN);

                // ReLU: max(0, x)
                AscendC::LocalTensor<float> zeroLocal = tmpBuf.Get<float>();
                AscendC::Duplicate(zeroLocal, (float)0.0f, curTileN);
                AscendC::Max(outLocal, outLocal, zeroLocal, curTileN);

                outQueueZ.EnQue(outLocal);
                outLocal = outQueueZ.DeQue<float>();

                // Copy out
                AscendC::DataCopy(zGm[globalRow * N + nOff], outLocal, curTileN);

                outQueueZ.FreeTensor(outLocal);
                inQueueBias.FreeTensor(biasLocal);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueW, inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accBuf, tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t M, N, K;
    uint32_t tileNum;
    uint32_t rowsPerBlock;
    uint32_t startRow;
    uint32_t tileN;
    uint32_t tileK;
};

extern "C" __global__ __aicore__ void gemm_add_relu_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmAddRelu op;
    op.Init(x, weight, bias, z, tiling_data.M, tiling_data.N, tiling_data.K, tiling_data.tileNum);
    op.Process();
}
