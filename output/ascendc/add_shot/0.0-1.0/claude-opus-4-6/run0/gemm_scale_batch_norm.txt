

Looking at the architecture, I need to fuse the GEMM (linear), scale multiplication, and batch normalization operations into a custom AscendC kernel. Let me analyze the operations:

1. `x = gemm(x)` → Linear: `x = x @ weight.T + bias`
2. `x = x * scale` → Element-wise multiplication with scale parameter
3. `x = bn(x)` → Batch normalization: `(x - mean) / sqrt(var + eps) * gamma + beta`

Given the complexity, I'll create a fused kernel for the scale and batch norm parts (post-GEMM), since the GEMM itself is best left to optimized BLAS libraries. The custom kernel will fuse: element-wise scale multiplication followed by batch normalization.

Actually, let me reconsider - fusing all three into one kernel on AscendC for large matrix multiply is complex. A practical approach is to fuse scale + batchnorm into one custom kernel, keeping GEMM as the standard operation.

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmScaleBatchNorm {
public:
    __aicore__ inline KernelGemmScaleBatchNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR scale,
                                 GM_ADDR bn_weight, GM_ADDR bn_bias,
                                 GM_ADDR running_mean, GM_ADDR running_var,
                                 GM_ADDR z, GM_ADDR mean_out, GM_ADDR var_out,
                                 GM_ADDR running_mean_out, GM_ADDR running_var_out,
                                 GM_ADDR workspace,
                                 uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures,
                                 float eps, float momentum, uint32_t isTraining, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->eps = eps;
        this->momentum = momentum;
        this->isTraining = isTraining;

        uint32_t totalOut = batchSize * outFeatures;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        // Each block processes a portion of the outFeatures dimension
        this->featPerBlock = outFeatures / blockNum;
        this->featStart = blockIdx * this->featPerBlock;
        if (blockIdx == blockNum - 1) {
            this->featPerBlock = outFeatures - this->featStart;
        }

        this->tileNum = tileNum;
        // tile along batch dimension
        this->tileBatch = batchSize / tileNum;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, outFeatures * inFeatures);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);
        scaleGm.SetGlobalBuffer((__gm__ float *)scale, outFeatures);
        bnWeightGm.SetGlobalBuffer((__gm__ float *)bn_weight, outFeatures);
        bnBiasGm.SetGlobalBuffer((__gm__ float *)bn_bias, outFeatures);
        runMeanGm.SetGlobalBuffer((__gm__ float *)running_mean, outFeatures);
        runVarGm.SetGlobalBuffer((__gm__ float *)running_var, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * outFeatures);
        meanOutGm.SetGlobalBuffer((__gm__ float *)mean_out, outFeatures);
        varOutGm.SetGlobalBuffer((__gm__ float *)var_out, outFeatures);
        runMeanOutGm.SetGlobalBuffer((__gm__ float *)running_mean_out, outFeatures);
        runVarOutGm.SetGlobalBuffer((__gm__ float *)running_var_out, outFeatures);

        // workspace: use for intermediate gemm result storage
        workGm.SetGlobalBuffer((__gm__ float *)workspace, batchSize * outFeatures + outFeatures * 2);

        uint32_t bufSize = this->featPerBlock * sizeof(float);
        if (bufSize < 32) bufSize = 32;
        pipe.InitBuffer(inQueueX, BUFFER_NUM, bufSize);
        pipe.InitBuffer(inQueueY, BUFFER_NUM, bufSize);
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, bufSize);
    }

    __aicore__ inline void Process()
    {
        // Phase 1: Compute GEMM + scale for our assigned features, store in zGm
        // For each batch row and our assigned features, compute dot product with weight row
        // This is simplified - we process feature by feature for each batch element
        
        // Phase 1: For each batch element, compute linear output * scale for assigned features
        // Then Phase 2: compute batch norm statistics and normalize
        
        // Due to memory constraints, we do this in tiles along feature dimension
        // For simplicity with AscendC constraints, process one feature at a time
        
        // Phase 1: GEMM + Scale
        // For each feature f in [featStart, featStart + featPerBlock):
        //   For each batch b: z[b, f] = (sum_k x[b,k] * weight[f,k] + bias[f]) * scale[f]
        
        // Phase 2: BatchNorm
        // For each feature f:
        //   mean_f = mean over b of z[b, f]  
        //   var_f = var over b of z[b, f]
        //   z[b, f] = bn_weight[f] * (z[b, f] - mean_f) / sqrt(var_f + eps) + bn_bias[f]

        // We process per-feature to keep things manageable
        for (uint32_t fi = 0; fi < this->featPerBlock; fi++) {
            uint32_t f = this->featStart + fi;
            
            // Load bias and scale for this feature
            float biasVal = 0.0f;
            float scaleVal = 1.0f;
            float bnW = 1.0f;
            float bnB = 0.0f;
            
            // Use vector copy to load single values
            {
                AscendC::LocalTensor<float> tmpLocal = inQueueX.AllocTensor<float>();
                // Copy bias
                uint32_t copyLen = 8; // minimum alignment
                AscendC::DataCopy(tmpLocal, biasGm[f & 0xFFFFFFF8u], copyLen);
                pipe.InitBuffer(inQueueX, 0, 0); // barrier
                // We'll use a simpler approach
                inQueueX.FreeTensor(tmpLocal);
            }
            
            // Simplified: compute gemm output for all batch rows for feature f
            // Store intermediate in zGm[b * outFeatures + f]
            
            float sumVal = 0.0f;
            float sumSqVal = 0.0f;
            
            for (uint32_t b = 0; b < batchSize; b++) {
                // Compute dot product of x[b, :] and weight[f, :]
                float dot = 0.0f;
                // This would need tiled processing for large inFeatures
                // For now, process in tiles
                for (uint32_t k = 0; k < inFeatures; k += 8) {
                    uint32_t len = 8;
                    if (k + len > inFeatures) len = inFeatures - k;
                    // simplified scalar computation
                }
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm, weightGm, biasGm, scaleGm;
    AscendC::GlobalTensor<float> bnWeightGm, bnBiasGm, runMeanGm, runVarGm;
    AscendC::GlobalTensor<float> zGm, meanOutGm, varOutGm, runMeanOutGm, runVarOutGm;
    AscendC::GlobalTensor<float> workGm;
    uint32_t batchSize, inFeatures, outFeatures;
    float eps, momentum;
    uint32_t isTraining;
    uint32_t featPerBlock, featStart;
    uint32_t tileNum, tileBatch;
};

extern "C" __global__ __aicore__ void gemm_scale_batch_norm_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
    GM_ADDR scale, GM_ADDR bn_weight, GM_ADDR bn_bias,
    GM_ADDR running_mean, GM_ADDR running_var,
    GM_ADDR z, GM_ADDR mean_out, GM_ADDR var_out,
    GM_ADDR running_mean_out, GM_ADDR running_var_out,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);

    op.Init(x, weight, bias, scale, bn_weight, bn_bias, running_mean, running_var,
