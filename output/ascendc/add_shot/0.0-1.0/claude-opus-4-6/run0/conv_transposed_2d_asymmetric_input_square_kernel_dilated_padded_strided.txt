

Looking at this architecture, it performs a 2D transposed convolution. Implementing a full transposed convolution from scratch in AscendC is extremely complex. The practical approach is to leverage the existing NPU convolution transpose capabilities through the ACL (Ascend Computing Language) operator interface while wrapping it as a custom operator.

Let me analyze the parameters:
- Input: (batch_size=16, in_channels=32, height_in=64, width_in=128)
- Weight: (in_channels=32, out_channels=64, kernel_size=3, kernel_size=3)
- stride=5, padding=1, dilation=2, bias=False

I'll create a custom operator that performs the transposed convolution using the NPU's built-in conv2d_transpose capabilities but wrapped as a custom AscendC operator.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose2d {
public:
    __aicore__ inline KernelConvTranspose2d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t heightIn, uint32_t widthIn,
                                 uint32_t kernelSize, uint32_t stride,
                                 uint32_t padding, uint32_t dilation,
                                 uint32_t heightOut, uint32_t widthOut,
                                 uint32_t totalOutput, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->heightIn = heightIn;
        this->widthIn = widthIn;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->dilation = dilation;
        this->heightOut = heightOut;
        this->widthOut = widthOut;
        this->totalOutput = totalOutput;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->outputPerBlock = totalOutput / blockNum;
        uint32_t remainder = totalOutput % blockNum;
        if (blockIdx < remainder) {
            this->outputPerBlock += 1;
            this->startIdx = blockIdx * this->outputPerBlock;
        } else {
            this->startIdx = blockIdx * this->outputPerBlock + remainder;
        }

        this->tileNum = tileNum;
        if (this->outputPerBlock < tileNum) {
            this->tileNum = this->outputPerBlock > 0 ? this->outputPerBlock : 1;
        }
        this->tileLength = this->outputPerBlock / this->tileNum;
        this->tailLength = this->outputPerBlock - this->tileLength * this->tileNum;

        uint32_t xSize = batchSize * inChannels * heightIn * widthIn;
        uint32_t wSize = inChannels * outChannels * kernelSize * kernelSize;
        
        xGm.SetGlobalBuffer((__gm__ float*)x, xSize);
        wGm.SetGlobalBuffer((__gm__ float*)weight, wSize);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalOutput);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t tile = 0; tile < this->tileNum; tile++) {
            uint32_t curLen = this->tileLength;
            if (tile == this->tileNum - 1) {
                curLen += this->tailLength;
            }
            uint32_t curStart = this->startIdx + tile * this->tileLength;
            ComputeTile(curStart, curLen);
        }
    }

private:
    __aicore__ inline void ComputeTile(uint32_t outputStart, uint32_t length)
    {
        for (uint32_t i = 0; i < length; i++) {
            uint32_t outIdx = outputStart + i;
            
            uint32_t ow = outIdx % widthOut;
            uint32_t tmp = outIdx / widthOut;
            uint32_t oh = tmp % heightOut;
            tmp = tmp / heightOut;
            uint32_t oc = tmp % outChannels;
            uint32_t n = tmp / outChannels;
            
            float sum = 0.0f;
            
            for (uint32_t ic = 0; ic < inChannels; ic++) {
                for (uint32_t kh = 0; kh < kernelSize; kh++) {
                    for (uint32_t kw = 0; kw < kernelSize; kw++) {
                        int32_t h_offset = (int32_t)oh + (int32_t)padding - (int32_t)(kh * dilation);
                        int32_t w_offset = (int32_t)ow + (int32_t)padding - (int32_t)(kw * dilation);
                        
                        if (h_offset % (int32_t)stride != 0) continue;
                        if (w_offset % (int32_t)stride != 0) continue;
                        
                        int32_t ih = h_offset / (int32_t)stride;
                        int32_t iw = w_offset / (int32_t)stride;
                        
                        if (ih < 0 || ih >= (int32_t)heightIn) continue;
                        if (iw < 0 || iw >= (int32_t)widthIn) continue;
                        
                        uint32_t xIdx = n * inChannels * heightIn * widthIn + ic * heightIn * widthIn + (uint32_t)ih * widthIn + (uint32_t)iw;
                        uint32_t wIdx = ic * outChannels * kernelSize * kernelSize + oc * kernelSize * kernelSize + kh * kernelSize + kw;
                        
                        float xVal = xGm.GetValue(xIdx);
                        float wVal = wGm.GetValue(wIdx);
                        sum += xVal * wVal;
                    }
                }
            }
            
            yGm.SetValue(outIdx, sum);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize, inChannels, outChannels;
    uint32_t heightIn, widthIn;
    uint32_t kernelSize, stride, padding, dilation;
    uint32_t heightOut, widthOut;
    uint32_t totalOutput;
    uint32_t outputPerBlock;
    uint32_t startIdx;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t tailLength;
};

extern "C" __global__ __aicore__ void conv_transposed_2d_asymmetric_input_square_kernel_dilated_padded_strided_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose2d op;
    op.Init(x, weight, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.heightIn, tiling_data.widthIn,
            tiling_data.kernelSize, tiling_data.stride,
            tiling_data.padding, tiling_data.dilation,
            tiling_data.heightOut, tiling_data.widthOut,
            tiling_data.totalOutput, tiling_data.tileNum);
    op.Process();
}
