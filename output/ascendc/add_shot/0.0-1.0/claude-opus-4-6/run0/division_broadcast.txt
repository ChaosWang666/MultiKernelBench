kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelDivisionBroadcast {
public:
    __aicore__ inline KernelDivisionBroadcast() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR z,
                                 uint32_t totalLength, uint32_t tileNum,
                                 uint32_t N, uint32_t C, uint32_t H, uint32_t W)
    {
        this->N = N;
        this->C = C;
        this->H = H;
        this->W = W;
        this->HW = H * W;
        this->CHW = C * H * W;

        // Each block processes a subset of the N dimension
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        // Distribute N across blocks
        this->nPerBlock = N / blockNum;
        uint32_t nRemainder = N % blockNum;
        if (blockIdx < nRemainder) {
            this->nPerBlock += 1;
            this->nStart = blockIdx * this->nPerBlock;
        } else {
            this->nStart = nRemainder * (this->nPerBlock + 1) + (blockIdx - nRemainder) * this->nPerBlock;
        }

        this->blockLength = this->nPerBlock * this->CHW;
        this->tileNum = tileNum;

        // Each tile processes one channel slice (H*W elements) at a time
        // We'll iterate over nPerBlock * C channel slices
        this->totalSlices = this->nPerBlock * C;

        xGm.SetGlobalBuffer((__gm__ float *)x, totalLength);
        scaleGm.SetGlobalBuffer((__gm__ float *)scale, C);
        zGm.SetGlobalBuffer((__gm__ float *)z, totalLength);

        // Align HW to 8 for float (32 bytes)
        uint32_t alignedHW = ((this->HW + 7) / 8) * 8;

        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedHW * sizeof(float));
        pipe.InitBuffer(inQueueScale, BUFFER_NUM, alignedHW * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, alignedHW * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t slice = 0; slice < this->totalSlices; slice++) {
            CopyIn(slice);
            Compute(slice);
            CopyOut(slice);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t sliceIdx)
    {
        uint32_t n = sliceIdx / this->C;
        uint32_t c = sliceIdx % this->C;
        uint32_t globalN = this->nStart + n;
        uint32_t offset = globalN * this->CHW + c * this->HW;

        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], this->HW);
        inQueueX.EnQue(xLocal);

        // Broadcast scale value across HW
        AscendC::LocalTensor<float> scaleLocal = inQueueScale.AllocTensor<float>();
        // We need to fill scaleLocal with scale[c] repeated HW times
        // Use Duplicate to broadcast
        AscendC::Duplicate(scaleLocal, scaleGm.GetValue(c), this->HW);
        inQueueScale.EnQue(scaleLocal);
    }

    __aicore__ inline void Compute(uint32_t sliceIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> scaleLocal = inQueueScale.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        AscendC::Div(zLocal, xLocal, scaleLocal, this->HW);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueScale.FreeTensor(scaleLocal);
    }

    __aicore__ inline void CopyOut(uint32_t sliceIdx)
    {
        uint32_t n = sliceIdx / this->C;
        uint32_t c = sliceIdx % this->C;
        uint32_t globalN = this->nStart + n;
        uint32_t offset = globalN * this->CHW + c * this->HW;

        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[offset], zLocal, this->HW);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueScale;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> scaleGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t N, C, H, W, HW, CHW;
    uint32_t nPerBlock, nStart;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t totalSlices;
};

extern "C" __global__ __aicore__ void division_broadcast_custom(GM_ADDR x, GM_ADDR scale, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelDivisionBroadcast op;
    op.Init(x, scale, z, tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.N, tiling_data.C, tiling_data.H, tiling_data.W);
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor division_broadcast_custom_impl_npu(const at::Tensor& x, const at::Tensor& scale) {
    // Ensure scale is contiguous and flattened to C elements
    auto scale_contig = scale.contiguous().reshape({-1});
    at::Tensor result = at::empty_like(x);
    EXEC_NPU_CMD(aclnnDivisionBroadcastCustom, x, scale_contig, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("division_broadcast_custom", &division_broadcast_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("division_broadcast_custom", &division_broadcast_custom_impl_npu, "x / scale with broadcast");
}
