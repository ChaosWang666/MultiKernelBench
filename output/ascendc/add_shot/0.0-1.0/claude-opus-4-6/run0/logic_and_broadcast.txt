kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelLogicAndBroadcast {
public:
    __aicore__ inline KernelLogicAndBroadcast() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                 uint32_t totalLength, uint32_t tileNum,
                                 uint32_t dim0, uint32_t dim1, uint32_t dim2,
                                 uint32_t bDim0, uint32_t bDim1, uint32_t bDim2)
    {
        this->totalLength = totalLength;
        this->dim0 = dim0;
        this->dim1 = dim1;
        this->dim2 = dim2;
        this->bDim0 = bDim0;
        this->bDim1 = bDim1;
        this->bDim2 = bDim2;
        this->bTotalLength = bDim0 * bDim1 * bDim2;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->blockLength = totalLength / blockNum;
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        // Align tileLength to 32 bytes for int8/bool (32 elements)
        uint32_t align = 32;
        this->tileLength = (this->tileLength / align) * align;
        if (this->tileLength == 0) {
            this->tileLength = align;
        }

        this->startOffset = blockIdx * this->blockLength;

        xGm.SetGlobalBuffer((__gm__ int8_t *)x, totalLength);
        yGm.SetGlobalBuffer((__gm__ int8_t *)y, this->bTotalLength);
        zGm.SetGlobalBuffer((__gm__ int8_t *)z, totalLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(int8_t));
        pipe.InitBuffer(inQueueY, BUFFER_NUM, this->tileLength * sizeof(int8_t));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(int8_t));
    }

    __aicore__ inline void Process()
    {
        uint32_t loopCount = this->blockLength / this->tileLength;
        uint32_t remaining = this->blockLength % this->tileLength;
        for (uint32_t i = 0; i < loopCount; i++) {
            CopyIn(i, this->tileLength);
            Compute(i, this->tileLength);
            CopyOut(i, this->tileLength);
        }
        if (remaining > 0) {
            // Handle remaining with alignment padding
            uint32_t align = 32;
            uint32_t paddedRemaining = ((remaining + align - 1) / align) * align;
            CopyInRemaining(loopCount, remaining, paddedRemaining);
            ComputeRemaining(loopCount, remaining, paddedRemaining);
            CopyOutRemaining(loopCount, remaining, paddedRemaining);
        }
    }

private:
    __aicore__ inline void GetBroadcastIndex(uint32_t flatIdx, uint32_t &bFlatIdx)
    {
        uint32_t i = flatIdx / (dim1 * dim2);
        uint32_t rem = flatIdx % (dim1 * dim2);
        uint32_t j = rem / dim2;
        uint32_t k = rem % dim2;

        uint32_t bi = (bDim0 == 1) ? 0 : i;
        uint32_t bj = (bDim1 == 1) ? 0 : j;
        uint32_t bk = (bDim2 == 1) ? 0 : k;

        bFlatIdx = bi * (bDim1 * bDim2) + bj * bDim2 + bk;
    }

    __aicore__ inline void CopyIn(uint32_t progress, uint32_t len)
    {
        AscendC::LocalTensor<int8_t> xLocal = inQueueX.AllocTensor<int8_t>();
        AscendC::LocalTensor<int8_t> yLocal = inQueueY.AllocTensor<int8_t>();

        uint32_t globalOffset = this->startOffset + progress * this->tileLength;
        AscendC::DataCopy(xLocal, xGm[globalOffset], len);

        // For y, we need to handle broadcasting by copying element by element
        // But for efficiency, if b shape is (1, dim1, 1), we can compute mapped indices
        for (uint32_t t = 0; t < len; t++) {
            uint32_t flatIdx = globalOffset + t;
            uint32_t bIdx;
            GetBroadcastIndex(flatIdx, bIdx);
            yLocal.SetValue(t, yGm.GetValue(bIdx));
        }

        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void Compute(uint32_t progress, uint32_t len)
    {
        AscendC::LocalTensor<int8_t> xLocal = inQueueX.DeQue<int8_t>();
        AscendC::LocalTensor<int8_t> yLocal = inQueueY.DeQue<int8_t>();
        AscendC::LocalTensor<int8_t> zLocal = outQueueZ.AllocTensor<int8_t>();

        AscendC::And(zLocal, xLocal, yLocal, len);

        outQueueZ.EnQue<int8_t>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress, uint32_t len)
    {
        AscendC::LocalTensor<int8_t> zLocal = outQueueZ.DeQue<int8_t>();
        uint32_t globalOffset = this->startOffset + progress * this->tileLength;
        AscendC::DataCopy(zGm[globalOffset], zLocal, len);
        outQueueZ.FreeTensor(zLocal);
    }

    __aicore__ inline void CopyInRemaining(uint32_t progress, uint32_t remaining, uint32_t padded)
    {
        AscendC::LocalTensor<int8_t> xLocal = inQueueX.AllocTensor<int8_t>();
        AscendC::LocalTensor<int8_t> yLocal = inQueueY.AllocTensor<int8_t>();

        uint32_t globalOffset = this->startOffset + progress * this->tileLength;
        AscendC::DataCopy(xLocal, xGm[globalOffset], padded);

        for (uint32_t t = 0; t < padded; t++) {
            uint32_t flatIdx = globalOffset + t;
            if (flatIdx < totalLength) {
                uint32_t bIdx;
                GetBroadcastIndex(flatIdx, bIdx);
                yLocal.SetValue(t, yGm.GetValue(bIdx));
            } else {
                yLocal.SetValue(t, (int8_t)0);
            }
        }

        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void ComputeRemaining(uint32_t progress, uint32_t remaining, uint32_t padded)
    {
        AscendC::LocalTensor<int8_t> xLocal = inQueueX.DeQue<int8_t>();
        AscendC::LocalTensor<int8_t> yLocal = inQueueY.DeQue<int8_t>();
        AscendC::LocalTensor<int8_t> zLocal = outQueueZ.AllocTensor<int8_t>();

        AscendC::And(zLocal, xLocal, yLocal, padded);

        outQueueZ.EnQue<int8_t>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOutRemaining(uint32_t progress, uint32_t remaining, uint32_t padded)
    {
        AscendC::LocalTensor<int8_t> zLocal = outQueueZ.DeQue<int8_t>();
        uint32_t globalOffset = this->startOffset + progress * this->tileLength;
        AscendC::DataCopy(zGm[globalOffset], zLocal, padded);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<int8_t> xGm;
    AscendC::GlobalTensor<int8_t> yGm;
    AscendC::GlobalTensor<int8_t> zGm;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t startOffset;
    uint32_t dim0, dim1, dim2;
    uint32_t bDim0, bDim1, bDim2;
    uint32_t bTotalLength;
};

extern "C" __global__ __aicore__ void logic_and_broadcast_custom(GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelLogicAndBroadcast op;
    op.Init(x, y, z, tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.dim0, tiling_data.dim1, tiling_data.dim2,
            tiling_data.bDim0, tiling_data.bDim1, tiling_data.bDim2);
    op.Process();
}
