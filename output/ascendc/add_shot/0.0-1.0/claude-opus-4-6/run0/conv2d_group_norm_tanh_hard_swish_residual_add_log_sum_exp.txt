

Looking at the architecture, I need to create a custom AscendC kernel that fuses the operations after convolution: GroupNorm → Tanh → HardSwish → Residual Add → LogSumExp. The convolution itself is complex and best left to the built-in operator.

Let me analyze the data flow:
1. Conv2d: (B, in_ch, H, W) → (B, out_ch, H', W') where H'=H-kernel_size+1=126, W'=126
2. GroupNorm on conv output
3. Tanh on normalized output
4. HardSwish on tanh output  
5. Residual add: conv_output + hardswish_output
6. LogSumExp along dim=1 with keepdim=True

I'll fuse steps 2-6 into a single custom kernel.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x_conv, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace,
                                uint32_t batchSize, uint32_t outChannels, uint32_t height, uint32_t width,
                                uint32_t groups, float eps)
    {
        this->batchSize = batchSize;
        this->outChannels = outChannels;
        this->height = height;
        this->width = width;
        this->groups = groups;
        this->eps = eps;
        this->HW = height * width;
        this->channelsPerGroup = outChannels / groups;
        this->spatialPerGroup = this->channelsPerGroup * this->HW;

        uint32_t totalBatches = batchSize;
        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->batchStart = blockIdx * ((totalBatches + numBlocks - 1) / numBlocks);
        this->batchEnd = (blockIdx + 1) * ((totalBatches + numBlocks - 1) / numBlocks);
        if (this->batchEnd > totalBatches) this->batchEnd = totalBatches;

        uint32_t totalElements = batchSize * outChannels * HW;
        uint32_t outElements = batchSize * 1 * HW;

        xConvGm.SetGlobalBuffer((__gm__ float*)x_conv, totalElements);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, outChannels);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, outChannels);
        yGm.SetGlobalBuffer((__gm__ float*)y, outElements);
        workGm.SetGlobalBuffer((__gm__ float*)workspace, totalElements);

        // Allocate buffers - we process one group's spatial data at a time
        uint32_t alignedSpatialPerGroup = (this->spatialPerGroup + 31) / 32 * 32;
        uint32_t alignedHW = (this->HW + 31) / 32 * 32;
        uint32_t alignedChannels = (this->outChannels + 31) / 32 * 32;

        pipe.InitBuffer(inQueue, BUFFER_NUM, alignedSpatialPerGroup * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, alignedSpatialPerGroup * sizeof(float));
        pipe.InitBuffer(tmpBuf1, alignedSpatialPerGroup * sizeof(float));
        pipe.InitBuffer(tmpBuf2, alignedSpatialPerGroup * sizeof(float));
        pipe.InitBuffer(reduceBuf, alignedSpatialPerGroup * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t b = batchStart; b < batchEnd; b++) {
            // Phase 1: GroupNorm + Tanh + HardSwish + ResAdd, store to workspace
            for (uint32_t g = 0; g < groups; g++) {
                ProcessGroup(b, g);
            }
            // Phase 2: LogSumExp across channels for each spatial position
            ProcessLogSumExp(b);
        }
    }

private:
    __aicore__ inline void ProcessGroup(uint32_t b, uint32_t g)
    {
        uint32_t groupOffset = b * outChannels * HW + g * channelsPerGroup * HW;
        uint32_t alignedSPG = (spatialPerGroup + 31) / 32 * 32;

        // Load group data
        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        // Copy channel by channel to handle potential non-contiguous
        for (uint32_t c = 0; c < channelsPerGroup; c++) {
            uint32_t srcOffset = groupOffset + c * HW;
            uint32_t dstOffset = c * HW;
            uint32_t alignedHW_copy = (HW + 7) / 8 * 8;
            AscendC::DataCopy(xLocal[dstOffset], xConvGm[srcOffset], alignedHW_copy);
        }
        AscendC::PipeBarrier<PIPE_ALL>();

        // Compute mean
        AscendC::LocalTensor<float> sumTensor = reduceBuf.Get<float>();
        float mean = 0.0f;
        // Sum all elements
        AscendC::ReduceSum(sumTensor, xLocal, tmpBuf1.Get<float>(), alignedSPG);
        AscendC::PipeBarrier<PIPE_ALL>();
        mean = sumTensor.GetValue(0) / (float)spatialPerGroup;

        // Compute variance: E[(x - mean)^2]
        AscendC::LocalTensor<float> diffLocal = tmpBuf1.Get<float>();
        AscendC::Adds(diffLocal, xLocal, -mean, alignedSPG);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::LocalTensor<float> diff2Local = tmpBuf2.Get<float>();
        AscendC::Mul(diff2Local, diffLocal, diffLocal, alignedSPG);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::ReduceSum(sumTensor, diff2Local, diffLocal, alignedSPG);
        AscendC::PipeBarrier<PIPE_ALL>();
        float variance = sumTensor.GetValue(0) / (float)spatialPerGroup;
        float invStd = 1.0f / sqrtf(variance + eps);

        // Normalize: (x - mean) * invStd
        AscendC::Adds(diffLocal, xLocal, -mean, alignedSPG);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Muls(diffLocal, diffLocal, invStd, alignedSPG);
        AscendC::PipeBarrier<PIPE_ALL>();

        // Apply gamma and beta per channel, then tanh, hardswish, residual add
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();

        for (uint32_t c = 0; c < channelsPerGroup; c++) {
            uint32_t chIdx = g * channelsPerGroup + c;
            float gam = gammaGm.GetValue(chIdx);
            float bet = betaGm.GetValue(chIdx);
            uint32_t offset = c * HW;
            uint32_t alignedHW_val = (HW + 7) / 8 * 8;

            // normVal = diffLocal[offset..] * gamma + beta
            AscendC::Muls(diff2Local[offset], diffLocal[offset], gam, alignedHW_val);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Adds(diff2Local[offset], diff2Local[offset], bet, alignedHW_val);
            AscendC::PipeBarrier<PIPE_ALL>();

            // Tanh: use element-wise computation
            // tanh(x) approximation or use the intrinsic
            // We'll compute tanh element by element for correctness
            for (uint32_t i = 0; i < HW; i++) {
                float val = diff2Local.GetValue(offset + i);
                float tanhVal = tanhf(val);
                // HardSwish: x * relu6(x+3) / 6
                float hs_in = tanhVal + 3.0f;
                if (hs_in < 0.0f) hs_in = 0.0f;
                if (hs_in > 6.0f) hs_in = 6.0f;
                float hardswish = tanhVal * hs_in / 6.0f;
                // Residual add: x_conv + hardswish
                float xc = xLocal.GetValue(offset + i);
                float res = xc + hardswish;
                outLocal.SetValue(offset + i, res);
            }
        }
        AscendC::PipeBarrier<PIPE_ALL>();

        // Store result to workspace
        for (uint32_t c = 0; c < channelsPerGroup; c++) {
            uint32_t dstOffset = groupOffset + c * HW;
            uint32_t srcOffset = c * HW;
            uint32_t alignedHW_copy = (HW + 7) / 8 * 8;
            AscendC::DataCopy(workGm[dstOffset], outLocal[srcOffset], alignedHW_copy);
        }
        AscendC::PipeBarrier<PIPE_ALL>();

        inQueue.FreeTensor(xLocal);
        outQueue.FreeTensor(outLocal);
    }

    __aicore__ inline void ProcessLogSumExp(uint32_t b)
    {
        // LogSumExp along dim=1 (channels): for each spatial position
        // log(sum(exp(x), dim=1))
        uint32_t batchOffset = b * outChannels * HW;
        uint32_t outBatchOffset = b * HW;
        uint32_t alignedHW_val = (HW + 7) / 8 * 8;

        // Process spatial positions
        // First find max across channels for numerical stability
        AscendC::LocalTensor<float> accLocal = inQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> chLocal = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> maxLocal = tmpBuf2.Get<float>();
        AscendC::LocalTensor<float> expLocal = reduceBuf.Get<float>();

        // Load first channel as initial max
        AscendC::DataCopy(maxLocal, workGm[batchOffset], alignedHW_val);
        AscendC::PipeBarrier<PIPE_ALL>();

        // Find max across all channels
        for (uint32_t c = 1; c < outChannels; c++) {
            AscendC::DataCopy(chLocal, workGm[batchOffset + c * HW], alignedHW_val);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Max(maxLocal, maxLocal, chLocal, alignedHW_val);
            AscendC::PipeBarrier<PIPE_ALL>();
        }

        // Compute sum of exp(x - max)
        // Initialize accumulator to 0
        AscendC::Duplicate(accLocal, 0.0f, alignedHW_val);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t c = 0; c < outChannels; c++) {
            AscendC::DataCopy(chLocal, workGm[batchOffset + c * HW], alignedHW_val);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Sub(chLocal, chLocal, maxLocal, alignedHW_val);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Exp(expLocal, chLocal, alignedHW_val);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Add(accLocal, accLocal, expLocal, alignedHW_val);
            AscendC::PipeBarrier<PIPE_ALL>();
        }

        // log(sum) + max
        AscendC::Ln(accLocal, accLocal, alignedHW_val);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Add(accLocal, accLocal, maxLocal, alignedHW_val);
        AscendC::PipeBarrier<PIPE_ALL>();

        // Write output
        AscendC::DataCopy(yGm[outBatchOffset], accLocal, alignedHW_val);
        AscendC::PipeBarrier<PIPE_ALL>();

        inQueue.FreeTensor(accLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2, reduceBuf;
    AscendC::GlobalTensor<float> xConvGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> workGm;
    uint32_t batchSize, outChannels, height, width, groups;
    float eps;
    uint32_t HW, channelsPerGroup, spatialPerGroup;
    uint32_t batchStart, batchEnd;
};

extern "C" __global__ __aicore__ void conv2d_group_norm_tanh_hard_swish_residual_add_log_sum_exp_custom(
    GM_ADDR x_conv, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x_conv, gamma, beta, y, workspace,
            tiling_data.batchSize, tiling_data.outChannels,
            tiling_data.height, tiling_data.width,
            tiling_data.groups, tiling_data.eps);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model that uses custom AscendC kernel for fused 
    GroupNorm + Tanh + HardSwish + Residual Add + LogSumExp after Conv2d.
    """
    def __init__(self, in_channels, out_channels, kernel_size, groups, eps=1e-5):
        super().__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.group_norm = torch.nn.GroupNorm(groups, out_channels, eps=eps)

    def forward(self, x):
        x_conv = self.conv(x)
        gamma = self.group_norm.weight.contiguous()
        beta = self.group_norm.bias.contiguous()
        x_conv_contig = x_conv.contiguous()
        return custom_ops_lib.conv2d_group_norm_tanh_hard_swish_residual_add_log_sum_exp_custom(
            x_conv_contig, gamma, beta)
