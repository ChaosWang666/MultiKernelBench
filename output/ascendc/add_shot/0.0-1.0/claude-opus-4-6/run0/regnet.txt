

Looking at the RegNet architecture, I'll identify a key operation to optimize with a custom AscendC kernel. The Global Average Pooling (`torch.mean(x, dim=[2, 3])`) followed by the linear layer is a good candidate. However, for a more impactful and straightforward optimization, I'll create a custom kernel that fuses the ReLU activation, which is used repeatedly throughout the network.

Actually, looking more carefully, a better target is the Global Average Pooling operation (`torch.mean(x, dim=[2,3])`) which reduces a 4D tensor to 2D by averaging over spatial dimensions. This is a clean, self-contained operation that can benefit from a custom kernel.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGlobalAvgPool {
public:
    __aicore__ inline KernelGlobalAvgPool() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t channels, uint32_t spatialSize, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->tileNum = tileNum;

        uint32_t totalBC = batchSize * channels;
        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        // Each block processes a subset of (batch, channel) pairs
        this->bcPerBlock = (totalBC + numBlocks - 1) / numBlocks;
        this->bcStart = blockIdx * this->bcPerBlock;
        this->bcEnd = this->bcStart + this->bcPerBlock;
        if (this->bcEnd > totalBC) {
            this->bcEnd = totalBC;
        }

        // Align tile length to 32 bytes (8 floats)
        uint32_t alignNum = 8;
        if (spatialSize <= alignNum) {
            this->tileLength = ((spatialSize + alignNum - 1) / alignNum) * alignNum;
            this->actualTileNum = 1;
        } else {
            uint32_t rawTile = (spatialSize + tileNum - 1) / tileNum;
            this->tileLength = ((rawTile + alignNum - 1) / alignNum) * alignNum;
            this->actualTileNum = (spatialSize + this->tileLength - 1) / this->tileLength;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * channels * spatialSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize * channels);

        // Buffer for input tiles and a small work buffer for partial sums
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, alignNum * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t bc = this->bcStart; bc < this->bcEnd; bc++) {
            ProcessOne(bc);
        }
    }

private:
    __aicore__ inline void ProcessOne(uint32_t bcIdx)
    {
        float totalSum = 0.0f;
        uint32_t baseOffset = bcIdx * this->spatialSize;
        uint32_t remaining = this->spatialSize;

        for (uint32_t t = 0; t < this->actualTileNum; t++) {
            uint32_t curLen = this->tileLength;
            uint32_t actualLen = remaining;
            if (actualLen > curLen) {
                actualLen = curLen;
            }

            // CopyIn
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();

            // If last tile might be partial, we need to handle alignment
            uint32_t copyLen = ((actualLen + 7) / 8) * 8;
            if (copyLen > this->tileLength) {
                copyLen = this->tileLength;
            }
            AscendC::DataCopy(xLocal, xGm[baseOffset + t * this->tileLength], copyLen);
            inQueueX.EnQue(xLocal);

            // Compute partial sum
            AscendC::LocalTensor<float> xIn = inQueueX.DeQue<float>();

            // Zero out elements beyond actualLen if partial tile
            if (actualLen < copyLen) {
                for (uint32_t i = actualLen; i < copyLen; i++) {
                    xLocal.SetValue(i, 0.0f);
                }
            }

            // Use ReduceSum to sum elements
            AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
            // ReduceSum: reduce copyLen elements
            AscendC::ReduceSum(zLocal, xIn, copyLen, false);
            float partialSum = zLocal.GetValue(0);
            totalSum += partialSum;

            outQueueZ.FreeTensor(zLocal);
            inQueueX.FreeTensor(xIn);

            remaining -= actualLen;
        }

        // Write result: mean = totalSum / spatialSize
        float mean = totalSum / (float)this->spatialSize;

        // Write single value - need aligned write
        AscendC::LocalTensor<float> zOut = outQueueZ.AllocTensor<float>();
        zOut.SetValue(0, mean);
        // We write just one float but DataCopy needs alignment
        // Use DataCopyPad or write with SetValue to global memory
        outQueueZ.EnQue(zOut);
        AscendC::LocalTensor<float> zResult = outQueueZ.DeQue<float>();
        // Copy 8 floats (minimum aligned), but we only care about first
        // We'll just set the GM value directly
        AscendC::DataCopy(yGm[bcIdx], zResult, 8);
        outQueueZ.FreeTensor(zResult);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t actualTileNum;
    uint32_t bcPerBlock;
    uint32_t bcStart;
    uint32_t bcEnd;
};

extern "C" __global__ __aicore__ void regnet_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGlobalAvgPool op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib
import torch.nn as nn

class ModelNew(nn.Module):
    def __init__(self, input_channels, stages, block_widths, output_classes):
        """
        :param input_channels: int, Number of input channels for the first layer
        :param stages: int, Number of stages in the RegNet architecture
        :param block_widths: List[int], Width (number of channels) for each block in the stages
        :param output_classes: int, Number of output classes for classification
        """
        super(ModelNew, self).__init__()

        self.stages = stages
        self.block_widths = block_widths
        
        layers = []
        current_channels = input_channels
        
        # Construct the stages with their respective blocks
        for i in range(stages):
            layers.append(self._make_stage(current_channels, block_widths[i]))
            current_channels = block_widths[i]
        
        self.feature_extractor = nn.Sequential(*layers)
        
        # Final fully connected layer for classification
        self.fc = nn.Linear(block_widths[-1], output_classes)
    
    def _make_stage(self, in_channels, out_channels):
        """
        Creates a simple block for each stage.
        :param in_channels: int, number of input channels
        :param out_channels: int, number of output channels
        :return: nn.Sequential block with convolutional layers
        """
        return nn.Sequential(
            nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
            nn.BatchNorm2d(out_channels),
            nn.ReLU(),
            nn.Conv2d(out_channels, out_channels, kernel_size=3, padding=1),
            nn.BatchNorm2d(out_channels),
            nn.ReLU(),
            nn.MaxPool2d(kernel_size=2, stride=2)
        )

    def forward(self, x):
        """
        Forward pass through the RegNet model with custom global average pooling.
        :param x: torch.Tensor of shape (batch_size, input_channels, height, width)
        :return: torch.Tensor of shape (batch_size, output_classes)
        """
        x = self.feature_extractor(x)
        # Use custom AscendC kernel for Global Average Pooling instead of torch.mean(x, dim=[2, 3])
        x = custom_ops_lib.regnet_custom(x)
        x = self.fc(x)
        return x
'''
