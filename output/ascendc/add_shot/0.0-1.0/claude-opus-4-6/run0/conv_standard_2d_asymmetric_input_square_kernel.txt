

Looking at this architecture, it performs a standard 2D convolution. Implementing a full conv2d from scratch in AscendC is complex, but I'll create a custom kernel that leverages the Ascend hardware's capabilities for 2D convolution with asymmetric input and square kernel.

Given the complexity of conv2d (im2col + matmul or direct convolution), I'll implement this using a tiled approach that processes the convolution efficiently on the Ascend AI cores.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv2d {
public:
    __aicore__ inline KernelConv2d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t height, uint32_t width,
                                 uint32_t kernelSize, uint32_t stride, uint32_t padding,
                                 uint32_t dilation, uint32_t groups,
                                 uint32_t outHeight, uint32_t outWidth,
                                 uint32_t totalOutputElements, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->height = height;
        this->width = width;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->dilation = dilation;
        this->groups = groups;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->totalOutputElements = totalOutputElements;

        // Each block processes a portion of the total output elements
        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        // Total output positions: batchSize * outChannels * outHeight * outWidth
        this->blockLength = totalOutputElements / numBlocks;
        uint32_t remainder = totalOutputElements % numBlocks;
        if (blockIdx < remainder) {
            this->blockLength += 1;
            this->blockStart = blockIdx * this->blockLength;
        } else {
            this->blockStart = blockIdx * this->blockLength + remainder;
        }

        if (this->blockLength == 0) {
            this->tileLength = 0;
            this->tileNum = 0;
            return;
        }

        // Align tileLength to 8 (32 bytes / 4 bytes per float)
        this->tileNum = tileNum;
        if (this->tileNum > this->blockLength) {
            this->tileNum = 1;
        }
        this->tileLength = this->blockLength / this->tileNum;
        // Ensure tileLength is aligned to 8
        if (this->tileLength < 8) {
            this->tileLength = 8;
            this->tileNum = 1;
        } else {
            this->tileLength = (this->tileLength / 8) * 8;
            this->tileNum = this->blockLength / this->tileLength;
        }
        this->lastTileLength = this->blockLength - (this->tileNum - 1) * this->tileLength;
        if (this->lastTileLength == 0) {
            this->lastTileLength = this->tileLength;
        } else {
            // Align last tile
            this->lastTileLength = ((this->lastTileLength + 7) / 8) * 8;
        }

        uint32_t inSize = batchSize * inChannels * height * width;
        uint32_t weightSize = outChannels * (inChannels / groups) * kernelSize * kernelSize;
        uint32_t outSize = totalOutputElements;

        xGm.SetGlobalBuffer((__gm__ float *)x, inSize);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, weightSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, outSize);

        this->inChannelsPerGroup = inChannels / groups;
        this->outChannelsPerGroup = outChannels / groups;

        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->tileLength == 0 || this->tileNum == 0) return;

        for (int32_t t = 0; t < this->tileNum; t++) {
            uint32_t curTileLen = (t == this->tileNum - 1) ? this->lastTileLength : this->tileLength;
            uint32_t curActualLen = (t == this->tileNum - 1) ? (this->blockLength - t * this->tileLength) : this->tileLength;

            AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();

            // Zero the output tile
            AscendC::Duplicate(outLocal, (float)0.0f, curTileLen);

            for (uint32_t idx = 0; idx < curActualLen; idx++) {
                uint32_t globalIdx = this->blockStart + t * this->tileLength + idx;

                // Decode globalIdx -> (n, oc, oh, ow)
                uint32_t ow = globalIdx % outWidth;
                uint32_t tmp = globalIdx / outWidth;
                uint32_t oh = tmp % outHeight;
                tmp = tmp / outHeight;
                uint32_t oc = tmp % outChannels;
                uint32_t n = tmp / outChannels;

                uint32_t g = oc / outChannelsPerGroup;
                uint32_t oc_in_group = oc % outChannelsPerGroup;

                float sum = 0.0f;

                for (uint32_t ic = 0; ic < inChannelsPerGroup; ic++) {
                    uint32_t abs_ic = g * inChannelsPerGroup + ic;
                    for (uint32_t kh = 0; kh < kernelSize; kh++) {
                        for (uint32_t kw = 0; kw < kernelSize; kw++) {
                            int32_t ih = (int32_t)(oh * stride + kh * dilation) - (int32_t)padding;
                            int32_t iw = (int32_t)(ow * stride + kw * dilation) - (int32_t)padding;

                            if (ih >= 0 && ih < (int32_t)height && iw >= 0 && iw < (int32_t)width) {
                                uint32_t xIdx = n * (inChannels * height * width) +
                                                abs_ic * (height * width) +
                                                (uint32_t)ih * width + (uint32_t)iw;
                                uint32_t wIdx = oc * (inChannelsPerGroup * kernelSize * kernelSize) +
                                                ic * (kernelSize * kernelSize) +
                                                kh * kernelSize + kw;

                                float xVal = xGm.GetValue(xIdx);
                                float wVal = weightGm.GetValue(wIdx);
                                sum += xVal * wVal;
                            }
                        }
                    }
                }
                outLocal.SetValue(idx, sum);
            }

            outQueue.EnQue(outLocal);

            // Copy out
            AscendC::LocalTensor<float> outLocal2 = outQueue.DeQue<float>();
            AscendC::DataCopy(yGm[this->blockStart + t * this->tileLength], outLocal2, curTileLen);
            outQueue.FreeTensor(outLocal2);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t batchSize, inChannels, outChannels, height, width;
    uint32_t kernelSize, stride, padding, dilation, groups;
    uint32_t outHeight, outWidth;
    uint32_t totalOutputElements;
    uint32_t inChannelsPerGroup, outChannelsPerGroup;
    uint32_t blockStart, blockLength;
    uint32_t tileNum, tileLength, lastTileLength;
};

extern "C" __global__ __aicore__ void conv_standard_2d_asymmetric_input_square_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2d op;
    op.Init(x, weight, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.height, tiling_data.width,
            tiling_data.kernelSize, tiling_data.stride, tiling_data.padding,
            tiling_data.dilation, tiling_data.groups,
            tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.totalOutputElements, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Performs a standard 2D convolution operation with an asymmetric input and a square kernel,
    using a custom AscendC kernel.
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, stride: int = 1,
                 padding: int = 0, dilation: int = 1, groups: int = 1, bias: bool = False):
        super(ModelNew, self).__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        self.dilation = dilation
        self.groups = groups
        # Use nn.Conv2d just to get the weight parameter with proper initialization
        conv_tmp = torch.nn.Conv2d(in_channels, out_channels, (kernel_size, kernel_size),
                                    stride=stride, padding=padding, dilation=dilation,
                                    groups=groups, bias=bias)
        self.weight = torch.nn.Parameter(conv_tmp.weight.data.clone())
        if bias:
            self.bias = torch.nn.Parameter(conv_tmp.bias.data.clone())
        else:
            self.bias = None

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        batch_size = x.shape[0]
        height = x.shape[2]
        width = x.shape[3]

        # Ensure contiguous
        x_contig = x.contiguous()
        w_contig = self.weight.contiguous()

        result = custom_ops_lib.conv_standard_2d_asymmetric_input_square_kernel_custom(
            x_contig, w_contig,
            batch_size, self.in_channels, self.out_channels,
            height, width, self.kernel_size,
            self.stride, self.padding, self.dilation, self.groups
        )

        if self.bias is not None:
            result = result + self.bias.view(1, -1, 1, 1)

        return result
'''
