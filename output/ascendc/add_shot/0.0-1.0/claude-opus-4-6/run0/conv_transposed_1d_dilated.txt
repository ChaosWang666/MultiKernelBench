

Looking at this architecture, I need to implement a transposed 1D convolution with dilation using custom AscendC kernels. The transposed 1D convolution computes the output by essentially scattering input values through the kernel weights with appropriate stride, padding, and dilation parameters.

Let me analyze the parameters:
- batch_size=32, in_channels=32, out_channels=64, kernel_size=5, length=131072
- stride=1, padding=0, dilation=3
- Output length = (length - 1) * stride - 2 * padding + dilation * (kernel_size - 1) + 1

The transposed convolution can be viewed as: for each output position, sum over input channels and kernel positions where the input contributes.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

// Transposed 1D convolution with dilation
// x: (batch, inC, inL), weight: (inC, outC, kS), output: (batch, outC, outL)
// outL = (inL-1)*stride - 2*padding + dilation*(kS-1) + 1
// For each output[b][oc][o]: sum over ic,k of x[b][ic][i]*weight[ic][oc][k]
//   where i = (o + padding - k*dilation) / stride, if divisible and 0<=i<inL

class KernelConvTransposed1dDilated {
public:
    __aicore__ inline KernelConvTransposed1dDilated() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels,
                                 uint32_t outChannels, uint32_t inputLength,
                                 uint32_t outputLength, uint32_t kernelSize,
                                 uint32_t stride, uint32_t padding,
                                 uint32_t dilation, uint32_t blockDim,
                                 uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inputLength = inputLength;
        this->outputLength = outputLength;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->dilation = dilation;
        this->blockDim = blockDim;

        // Total tasks = batchSize * outChannels
        this->totalTasks = batchSize * outChannels;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * inputLength);
        wGm.SetGlobalBuffer((__gm__ float *)weight, inChannels * outChannels * kernelSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize * outChannels * outputLength);

        // We need buffers for: input tile, weight tile, output accumulation, temp
        // Allocate based on available UB
        // Process output in tiles along outputLength dimension
        this->tileNum = tileNum;
        uint32_t tileLenRaw = outputLength / tileNum;
        // Align to 8 for float (32 bytes)
        this->tileLength = (tileLenRaw + 7) & ~7;
        if (this->tileLength * tileNum < outputLength) {
            this->tileLength = ((outputLength + tileNum - 1) / tileNum + 7) & ~7;
        }

        pipe.InitBuffer(outBuf, 1, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, 1, this->tileLength * sizeof(float));
        // For input: we need at most (tileLength + dilation*(kS-1)) / stride + 1 elements per ic
        uint32_t maxInputTile = (this->tileLength + dilation * (kernelSize - 1)) / stride + 2;
        maxInputTile = (maxInputTile + 7) & ~7;
        this->maxInputTile = maxInputTile;
        pipe.InitBuffer(inBuf, 1, maxInputTile * sizeof(float));
        pipe.InitBuffer(wBuf, 1, ((kernelSize + 7) & ~7) * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        // Each block processes a subset of tasks
        uint32_t tasksPerBlock = (totalTasks + blockDim - 1) / blockDim;
        uint32_t taskStart = blockIdx * tasksPerBlock;
        uint32_t taskEnd = taskStart + tasksPerBlock;
        if (taskEnd > totalTasks) taskEnd = totalTasks;

        for (uint32_t task = taskStart; task < taskEnd; task++) {
            uint32_t b = task / outChannels;
            uint32_t oc = task % outChannels;
            ProcessOneOutput(b, oc);
        }
    }

private:
    __aicore__ inline void ProcessOneOutput(uint32_t b, uint32_t oc)
    {
        // Process output[b][oc][:] in tiles along output length
        for (uint32_t t = 0; t < outputLength; t += tileLength) {
            uint32_t curTileLen = tileLength;
            if (t + curTileLen > outputLength) {
                curTileLen = outputLength - t;
            }
            uint32_t alignedTileLen = (curTileLen + 7) & ~7;

            // Zero the output accumulator
            AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
            AscendC::Duplicate(outLocal, (float)0.0f, alignedTileLen);

            // For each input channel
            for (uint32_t ic = 0; ic < inChannels; ic++) {
                // For each kernel position
                for (uint32_t k = 0; k < kernelSize; k++) {
                    // output pos o maps to input pos i = (o + padding - k*dilation) / stride
                    // Need: (o + padding - k*dilation) >= 0 and divisible by stride and i < inputLength
                    int32_t kd = (int32_t)(k * dilation);

                    // Load the single weight value weight[ic][oc][k]
                    // weight layout: (inChannels, outChannels, kernelSize)
                    uint32_t wOffset = ic * outChannels * kernelSize + oc * kernelSize + k;
                    AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
                    // We need just one weight value, but DataCopy requires alignment
                    // Copy aligned chunk
                    uint32_t wAlignedOffset = (wOffset / 8) * 8;
                    uint32_t wIntraOffset = wOffset - wAlignedOffset;
                    AscendC::DataCopy(wLocal, wGm[wAlignedOffset], 8);
                    AscendC::PipeBarrier<PIPE_ALL>();
                    float wVal = wLocal.GetValue(wIntraOffset);

                    if (wVal == 0.0f) continue;

                    // For this tile of output positions [t, t+curTileLen):
                    // i = (o + padding - kd) / stride
                    // We need o + padding - kd >= 0 => o >= kd - padding
                    // and (o + padding - kd) % stride == 0
                    // and i < inputLength

                    AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
                    AscendC::Duplicate(tmpLocal, (float)0.0f, alignedTileLen);
                    AscendC::PipeBarrier<PIPE_ALL>();

                    // Process element by element for correctness
                    uint32_t xBaseOffset = b * inChannels * inputLength + ic * inputLength;

                    for (uint32_t idx = 0; idx < curTileLen; idx++) {
                        uint32_t o = t + idx;
                        int32_t numerator = (int32_t)o + (int32_t)padding - kd;
                        if (numerator < 0) continue;
                        if (stride > 1 && (numerator % (int32_t)stride != 0)) continue;
                        uint32_t i = (uint32_t)numerator / stride;
                        if (i >= inputLength) continue;

                        // Read x[b][ic][i]
                        uint32_t xIdx = xBaseOffset + i;
                        uint32_t xAligned = (xIdx / 8) * 8;
                        uint32_t xIntra = xIdx - xAligned;
                        AscendC::LocalTensor<float> inLocal = inBuf.Get<float>();
                        AscendC::DataCopy(inLocal, xGm[xAligned], 8);
                        AscendC::PipeBarrier<PIPE_ALL>();
                        float xVal = inLocal.GetValue(xIntra);

                        float contribution = xVal * wVal;
                        float curVal = outLocal.GetValue(idx);
                        outLocal.SetValue(idx, curVal + contribution);
                    }
                    AscendC::PipeBarrier<PIPE_ALL>();
                }
            }

            // Write output tile
            uint32_t yOffset = b * outChannels * outputLength + oc * outputLength + t;
            AscendC::PipeBarrier<PIPE_ALL>();
            // Need to handle unaligned writes
            uint32_t alignedWrite = (curTileLen + 7) & ~7;
            if (t + alignedWrite <= outputLength) {
                AscendC::DataCopy(yGm[yOffset], outLocal, alignedWrite);
            } else {
                // Last tile might exceed, but we allocated enough in output
                AscendC::DataCopy(yGm[yOffset], outLocal, alignedWrite);
            }
            AscendC::PipeBarrier<PIPE_ALL>();
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf, tmpBuf, inBuf, wBuf;
    AscendC::GlobalTensor<float> xGm, wGm, yGm;
    uint32_t batchSize, inChannels, outChannels, inputLength, outputLength;
    uint32_t kernelSize, stride, padding, dilation;
    uint32_t blockDim, totalTasks;
    uint32_t tileNum, tileLength, maxInputTile;
};

extern "C" __global__ __aicore__ void conv_transposed_1d_dilated_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTransposed1dDilated op;
    op.Init(x, weight, y,
            tiling_data.batchSize, tiling_data.inChannels,
            tiling_data.outChannels, tiling_data.inputLength,
            tiling_data.outputLength, tiling_data.kernelSize,
            tiling_data.stride, tiling_data.padding,
            tiling_data.dilation, tiling_data.blockDim,
            tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized transposed 1D convolution with dilation using custom AscendC kernel.
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int,
                 stride: int = 1, padding: int = 0, dilation: int = 1, bias: bool = False):
        super(ModelNew, self).__init__()
        # Use the original module to get properly initialized weights
        self.conv1d_transpose = torch.nn.ConvTranspose1d(
            in_channels, out_channels, kernel_size,
            stride=stride, padding=padding, dilation=dilation, bias=bias
        )
        self.stride = stride
        self.padding = padding
        self.dilation = dilation
        self.has_bias = bias

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        weight = self.conv1d_transpose.weight  # (in_channels, out_channels, kernel_size)
        result = custom_ops_lib.conv_transposed_1d_dilated_custom(
            x, weight, self.stride, self.padding, self.dilation
        )
        if self.has_bias and self.conv1d_transpose.bias is not None:
            result = result + self.conv1d_transpose.bias.unsqueeze(0).unsqueeze(2)
        return result
