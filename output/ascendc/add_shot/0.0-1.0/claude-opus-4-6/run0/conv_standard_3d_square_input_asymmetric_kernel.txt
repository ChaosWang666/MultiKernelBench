kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv3d {
public:
    __aicore__ inline KernelConv3d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inputD, uint32_t inputH, uint32_t inputW,
                                 uint32_t kernelD, uint32_t kernelH, uint32_t kernelW,
                                 uint32_t strideD, uint32_t strideH, uint32_t strideW,
                                 uint32_t padD, uint32_t padH, uint32_t padW,
                                 uint32_t dilationD, uint32_t dilationH, uint32_t dilationW,
                                 uint32_t groups, uint32_t hasBias,
                                 uint32_t outD, uint32_t outH, uint32_t outW,
                                 uint32_t totalOutputPoints, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inputD = inputD;
        this->inputH = inputH;
        this->inputW = inputW;
        this->kernelD = kernelD;
        this->kernelH = kernelH;
        this->kernelW = kernelW;
        this->strideD = strideD;
        this->strideH = strideH;
        this->strideW = strideW;
        this->padD = padD;
        this->padH = padH;
        this->padW = padW;
        this->dilationD = dilationD;
        this->dilationH = dilationH;
        this->dilationW = dilationW;
        this->groups = groups;
        this->hasBias = hasBias;
        this->outD = outD;
        this->outH = outH;
        this->outW = outW;

        this->inChannelsPerGroup = inChannels / groups;
        this->outChannelsPerGroup = outChannels / groups;
        this->patchLen = inChannelsPerGroup * kernelD * kernelH * kernelW;

        // total output spatial points per sample
        this->outSpatial = outD * outH * outW;

        // total output elements = batch * outChannels * outSpatial
        this->totalOut = totalOutputPoints;

        // Each block handles a portion of (batch, outChannel) combinations
        uint32_t totalBatchOC = batchSize * outChannels;
        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->myStart = (totalBatchOC / numBlocks) * blockIdx;
        uint32_t myEnd;
        if (blockIdx < numBlocks - 1) {
            myEnd = (totalBatchOC / numBlocks) * (blockIdx + 1);
        } else {
            myEnd = totalBatchOC;
        }
        this->myCount = myEnd - this->myStart;

        uint32_t inputSize = batchSize * inChannels * inputD * inputH * inputW;
        uint32_t weightSize = outChannels * inChannelsPerGroup * kernelD * kernelH * kernelW;
        uint32_t outputSize = batchSize * outChannels * outD * outH * outW;

        xGm.SetGlobalBuffer((__gm__ float*)x, inputSize);
        wGm.SetGlobalBuffer((__gm__ float*)weight, weightSize);
        if (hasBias) {
            bGm.SetGlobalBuffer((__gm__ float*)bias, outChannels);
        }
        yGm.SetGlobalBuffer((__gm__ float*)y, outputSize);

        // Allocate UB buffers
        // We process one output spatial point at a time per (batch, oc) pair
        // Need patchLen floats for input patch, patchLen floats for weight patch, and some for accumulation
        uint32_t alignedPatch = ((patchLen + 7) / 8) * 8;  // align to 8 floats (32 bytes)
        this->alignedPatch = alignedPatch;

        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedPatch * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, alignedPatch * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, alignedPatch * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t idx = 0; idx < this->myCount; idx++) {
            uint32_t globalIdx = this->myStart + idx;
            uint32_t n = globalIdx / this->outChannels;
            uint32_t oc = globalIdx % this->outChannels;
            uint32_t g = oc / this->outChannelsPerGroup;
            uint32_t ocInGroup = oc % this->outChannelsPerGroup;

            // Weight offset for this output channel: weight[oc, :, :, :, :]
            // shape = [outChannels, inChannelsPerGroup, kD, kH, kW]
            uint32_t wOffset = oc * this->patchLen;

            // Load weight into UB once for this oc
            AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
            // Copy weight patch
            uint32_t copyLen = this->alignedPatch;
            if (wOffset + copyLen <= this->outChannels * this->patchLen) {
                AscendC::DataCopy(wLocal, wGm[wOffset], copyLen);
            } else {
                // handle boundary - just copy what we can
                AscendC::DataCopy(wLocal, wGm[wOffset], this->patchLen);
            }
            inQueueW.EnQue(wLocal);
            wLocal = inQueueW.DeQue<float>();

            float biasVal = 0.0f;
            if (this->hasBias) {
                // We'll just read bias from GM directly - simple scalar
                // For simplicity, copy bias into a small local tensor
                biasVal = 0.0f; // will accumulate later
            }

            for (uint32_t od = 0; od < this->outD; od++) {
                for (uint32_t oh = 0; oh < this->outH; oh++) {
                    for (uint32_t ow = 0; ow < this->outW; ow++) {
                        // Build input patch and compute dot product
                        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                        // Zero out the buffer
                        AscendC::Duplicate(xLocal, 0.0f, this->alignedPatch);

                        // Fill input patch
                        uint32_t pIdx = 0;
                        for (uint32_t ic = 0; ic < this->inChannelsPerGroup; ic++) {
                            uint32_t actualIC = g * this->inChannelsPerGroup + ic;
                            for (uint32_t kd = 0; kd < this->kernelD; kd++) {
                                int32_t id = (int32_t)(od * this->strideD + kd * this->dilationD) - (int32_t)this->padD;
                                for (uint32_t kh = 0; kh < this->kernelH; kh++) {
                                    int32_t ih = (int32_t)(oh * this->strideH + kh * this->dilationH) - (int32_t)this->padH;
                                    for (uint32_t kw = 0; kw < this->kernelW; kw++) {
                                        int32_t iw = (int32_t)(ow * this->strideW + kw * this->dilationW) - (int32_t)this->padW;
                                        float val = 0.0f;
                                        if (id >= 0 && id < (int32_t)this->inputD &&
                                            ih >= 0 && ih < (int32_t)this->inputH &&
                                            iw >= 0 && iw < (int32_t)this->inputW) {
                                            uint32_t xIdx = ((n * this->inChannels + actualIC) * this->inputD + (uint32_t)id) * this->inputH * this->inputW +
                                                            (uint32_t)ih * this->inputW + (uint32_t)iw;
                                            // set value - scalar set
                                            xLocal.SetValue(pIdx, xGm.GetValue(xIdx));
                                        }
                                        pIdx++;
                                    }
                                }
                            }
                        }

                        inQueueX.EnQue(xLocal);
                        xLocal = inQueueX.DeQue<float>();

                        // Compute element-wise multiply then reduce sum
                        AscendC::LocalTensor<float> tmpLocal = outQueueY.AllocTensor<float>();
                        AscendC::Mul(tmpLocal, xLocal, wLocal, this->alignedPatch);

                        // Reduce sum
                        float sum = 0.0f;
                        for (uint32_t i = 0; i < this->patchLen; i++) {
                            sum += tmpLocal.GetValue(i);
                        }

                        if (this->hasBias) {
                            sum += bGm.GetValue(oc);
                        }

                        // Write output
                        uint32_t yIdx = ((n * this->outChannels + oc) * this->outD + od) * this->outH * this->outW +
                                        oh * this->outW + ow;
                        yGm.SetValue(yIdx, sum);

                        outQueueY.EnQue(tmpLocal);
                        tmpLocal = outQueueY.DeQue<float>();
                        outQueueY.FreeTensor(tmpLocal);
                        inQueueX.FreeTensor(xLocal);
                    }
                }
            }

            inQueueW.FreeTensor(wLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, yGm;

    uint32_t batchSize, inChannels, outChannels;
    uint32_t inputD, inputH, inputW;
    uint32_t kernelD, kernelH, kernelW;
    uint32_t strideD, strideH, strideW;
    uint32_t padD, padH, padW;
    uint32_t dilationD, dilationH, dilationW;
    uint32_t groups, hasBias;
    uint32_t outD, outH, outW;
    uint32_t inChannelsPerGroup, outChannelsPerGroup;
    uint32_t patchLen, alignedPatch;
    uint32_t outSpatial, totalOut;
    uint32_t myStart, myCount;
};

extern "C" __global__ __aicore__ void conv_standard_3d_square_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3d op;
    op.Init(x, weight, bias, y, workspace,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inputD, tiling_data.inputH, tiling_data.inputW,
            tiling_data.kernelD, tiling_data.kernelH, tiling_data.kernelW,
            tiling_data.strideD, tiling_data.strideH, tiling_data.strideW,
            tiling_data.padD, tiling_data.padH, tiling_data.padW,
            tiling_data.dilationD, tiling_data.dilationH, tiling_data.dilationW,
            tiling_data.groups, tiling_data.hasBias,
            tiling_data.outD, tiling_data.outH, tiling_data.outW,
            tiling_data.totalOutputPoints, tiling_data.tileNum);
    op.Process();
}
