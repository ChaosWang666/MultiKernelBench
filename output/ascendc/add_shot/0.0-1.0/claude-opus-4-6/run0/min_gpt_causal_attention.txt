

Looking at this architecture, the key computationally intensive part is the causal self-attention: scaled dot-product attention with causal masking and softmax. I'll create a fused kernel that handles the attention computation: `att = softmax(q @ k^T / sqrt(d) + causal_mask)`, then `y = att @ v`.

Given the complexity of fusing everything into one kernel on AscendC, I'll focus on fusing the scaled dot-product with causal masking and softmax into a custom operator, while keeping the linear projections and dropout as standard PyTorch ops.

However, given the constraints of AscendC programming (limited memory per core, tiling requirements), I'll create a kernel that fuses the mask-fill and softmax operations on the attention scores, as this is a memory-bound operation that benefits from fusion.

Let me implement a fused `masked_fill + softmax` kernel for the attention scores.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMinGptCausalAttention {
public:
    __aicore__ inline KernelMinGptCausalAttention() {}
    __aicore__ inline void Init(GM_ADDR att, GM_ADDR mask, GM_ADDR out, uint32_t totalLength, uint32_t tileNum, uint32_t lastDim)
    {
        this->totalLength = totalLength;
        this->lastDim = lastDim;
        // Each block processes a portion of rows
        // totalLength = numRows * lastDim
        uint32_t numRows = totalLength / lastDim;
        this->blockLength = (numRows / AscendC::GetBlockNum()) * lastDim;
        // If not evenly divisible, last block gets remainder
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowsPerBlock = numRows / AscendC::GetBlockNum();
        uint32_t startRow = blockIdx * rowsPerBlock;
        if (blockIdx == AscendC::GetBlockNum() - 1) {
            rowsPerBlock = numRows - startRow;
            this->blockLength = rowsPerBlock * lastDim;
        }
        this->numRows = rowsPerBlock;

        // Each tile processes one row (lastDim elements)
        // We process row by row for softmax
        this->tileLength = lastDim;
        // Align tile length to 32 bytes (8 floats)
        uint32_t alignNum = 8;
        this->tileLengthAligned = (this->tileLength + alignNum - 1) / alignNum * alignNum;

        uint32_t startOffset = startRow * lastDim;
        attGm.SetGlobalBuffer((__gm__ float *)att + startOffset, this->blockLength);
        maskGm.SetGlobalBuffer((__gm__ float *)mask + startOffset, this->blockLength);
        outGm.SetGlobalBuffer((__gm__ float *)out + startOffset, this->blockLength);

        pipe.InitBuffer(inQueueAtt, BUFFER_NUM, this->tileLengthAligned * sizeof(float));
        pipe.InitBuffer(inQueueMask, BUFFER_NUM, this->tileLengthAligned * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLengthAligned * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->tileLengthAligned * sizeof(float));
        pipe.InitBuffer(maxBuf, alignNum * sizeof(float));
        pipe.InitBuffer(sumBuf, alignNum * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->numRows; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> attLocal = inQueueAtt.AllocTensor<float>();
        AscendC::LocalTensor<float> maskLocal = inQueueMask.AllocTensor<float>();
        AscendC::DataCopy(attLocal, attGm[rowIdx * this->tileLength], this->tileLengthAligned);
        AscendC::DataCopy(maskLocal, maskGm[rowIdx * this->tileLength], this->tileLengthAligned);
        inQueueAtt.EnQue(attLocal);
        inQueueMask.EnQue(maskLocal);
    }

    __aicore__ inline void Compute(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> attLocal = inQueueAtt.DeQue<float>();
        AscendC::LocalTensor<float> maskLocal = inQueueMask.DeQue<float>();
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();

        uint32_t len = this->tileLengthAligned;

        // masked_fill: where mask == 0, set att to -inf
        // mask contains 1.0 for valid, 0.0 for masked
        // We want: att = att * mask + (1 - mask) * (-inf)
        // Equivalent: where mask == 0 => -inf, else att
        // Step 1: Create -inf mask: (1 - mask) * large_negative
        float negInf = -3.4028235e+38f;

        // tmp = 1.0 - mask
        AscendC::Adds(tmpLocal, maskLocal, -1.0f, len);
        AscendC::Muls(tmpLocal, tmpLocal, -1.0f, len);
        // Now tmpLocal = 1.0 - mask (1 where masked, 0 where valid)
        // tmp = tmp * negInf
        AscendC::Muls(tmpLocal, tmpLocal, negInf, len);
        // att = att * mask  (zero out masked positions)
        AscendC::Mul(outLocal, attLocal, maskLocal, len);
        // att = att + tmp (add -inf to masked positions)
        AscendC::Add(outLocal, outLocal, tmpLocal, len);

        // Softmax along the row
        // Step 1: find max
        AscendC::ReduceMax(maxLocal, outLocal, tmpLocal, len);
        float maxVal = maxLocal.GetValue(0);

        // Step 2: subtract max
        AscendC::Adds(outLocal, outLocal, -maxVal, len);

        // Step 3: exp
        AscendC::Exp(outLocal, outLocal, len);

        // Step 4: sum
        AscendC::ReduceSum(sumLocal, outLocal, tmpLocal, len);
        float sumVal = sumLocal.GetValue(0);

        // Step 5: divide by sum
        float invSum;
        if (sumVal > 0.0f) {
            invSum = 1.0f / sumVal;
        } else {
            invSum = 0.0f;
        }
        AscendC::Muls(outLocal, outLocal, invSum, len);

        outQueue.EnQue(outLocal);
        inQueueAtt.FreeTensor(attLocal);
        inQueueMask.FreeTensor(maskLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> outLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(outGm[rowIdx * this->tileLength], outLocal, this->tileLengthAligned);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueAtt, inQueueMask;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::GlobalTensor<float> attGm;
    AscendC::GlobalTensor<float> maskGm;
    AscendC::GlobalTensor<float> outGm;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t lastDim;
    uint32_t tileLength;
    uint32_t tileLengthAligned;
    uint32_t numRows;
};

extern "C" __global__ __aicore__ void min_gpt_causal_attention_custom(GM_ADDR att, GM_ADDR mask, GM_ADDR out, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMinGptCausalAttention op;
    op.Init(att, mask, out, tiling_data.totalLength, tiling_data.tileNum, tiling_data.lastDim);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import math
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    """
    A vanilla multi-head masked self-attention layer with a projection at the end.
    Uses custom AscendC kernel for fused masked_fill + softmax.
    """

    def __init__(self, n_embd, n_head, attn_pdrop, resid_pdrop, max_seqlen):
        super().__init__()
        assert n_embd % n_head == 0
        self.c_attn = nn.Linear(n_embd, 3 * n_embd)
        self.c_proj = nn.Linear(n_embd, n_embd)
        self.attn_dropout = nn.Dropout(attn_pdrop)
        self.resid_dropout = nn.Dropout(resid_pdrop)
        self.register_buffer("bias", torch.tril(torch.ones(max_seqlen, max_seqlen))
                                     .view(1, 1, max_seqlen, max_seqlen))
        self.n_head = n_head
        self.n_embd = n_embd

    def forward(self, x):
        B, T, C = x.size()

        q, k, v = self.c_attn(x).split(self.n_embd, dim=2)
        k = k.view(B, T, self.n_head, C // self.n_head).transpose(1, 2)
        q = q.view(B, T, self.n_head, C // self.n_head).transpose(1, 2)
        v = v.view(B, T, self.n_head, C // self.n_head).transpose(1, 2)

        att = (q @ k.transpose(-2, -1)) * (1.0 / math.sqrt(k.size(-1)))

        # Use custom fused masked_fill + softmax kernel
        mask = self.bias[:, :, :T, :T].expand(B, self.n_head, T, T).contiguous()
        att = att.contiguous()
        att = custom_ops_lib.min_gpt_causal_attention_custom(att, mask)

        att = self.attn_dropout(att)
        y = att @ v
        y = y.transpose(1, 2).contiguous().view(B, T, C)

        y = self.resid_dropout(self.c_proj(y))
        return y
'''
