kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelGemmGroupNormSwishMultiplySwish {
public:
    __aicore__ inline KernelGemmGroupNormSwishMultiplySwish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, 
                                 GM_ADDR gamma, GM_ADDR beta, GM_ADDR multiplyWeight,
                                 GM_ADDR z, uint32_t batchSize, uint32_t outFeatures,
                                 uint32_t numGroups, uint32_t channelsPerGroup, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->channelsPerGroup = channelsPerGroup;
        
        // Each block processes a subset of batch rows
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->rowStart = blockIdx * ((batchSize + blockNum - 1) / blockNum);
        this->rowEnd = (blockIdx + 1) * ((batchSize + blockNum - 1) / blockNum);
        if (this->rowEnd > batchSize) this->rowEnd = batchSize;
        this->rowCount = (this->rowEnd > this->rowStart) ? (this->rowEnd - this->rowStart) : 0;
        
        // Set up global memory pointers - x not used in kernel since GEMM done outside
        // Actually this kernel does post-GEMM operations: GroupNorm + Swish + Multiply + Swish
        // The input 'x' here is already the GEMM result passed in
        // But as per the op definition, we receive x as the original input and weight/bias for GEMM
        // For simplicity in this fused kernel, we'll do the operations after GEMM on device
        // However, GEMM on AscendC for large matrices is complex. 
        // We'll handle GroupNorm+Swish+Multiply+Swish in this kernel, GEMM done via PyTorch
        
        // For this kernel, x is already the GEMM output (batch_size, out_features)
        xGm.SetGlobalBuffer((__gm__ float*)x + this->rowStart * outFeatures, this->rowCount * outFeatures);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, outFeatures);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, outFeatures);
        mulWGm.SetGlobalBuffer((__gm__ float*)multiplyWeight, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z + this->rowStart * outFeatures, this->rowCount * outFeatures);
        
        // Allocate buffers for processing one group at a time
        pipe.InitBuffer(inQueueX, BUFFER_NUM, channelsPerGroup * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, channelsPerGroup * sizeof(float));
        pipe.InitBuffer(tmpBuf1, BUFFER_NUM, channelsPerGroup * sizeof(float));
        pipe.InitBuffer(tmpBuf2, BUFFER_NUM, channelsPerGroup * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        for (uint32_t row = 0; row < this->rowCount; row++) {
            for (uint32_t g = 0; g < this->numGroups; g++) {
                CopyIn(row, g);
                Compute(row, g);
                CopyOut(row, g);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row, uint32_t group)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        uint32_t offset = row * outFeatures + group * channelsPerGroup;
        AscendC::DataCopy(xLocal, xGm[offset], channelsPerGroup);
        inQueueX.EnQue(xLocal);
    }
    
    __aicore__ inline void Compute(uint32_t row, uint32_t group)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.AllocTensor<float>();
        
        uint32_t len = channelsPerGroup;
        
        // Step 1: GroupNorm - compute mean and variance for this group
        // mean = sum(x) / len
        float sumVal = 0.0f;
        // Use ReduceSum
        AscendC::ReduceSum(tmp1, xLocal, tmp2, len);
        float mean = tmp1.GetValue(0) / (float)len;
        
        // x = x - mean
        AscendC::Adds(xLocal, xLocal, -mean, len);
        
        // variance = sum(x^2) / len
        AscendC::Mul(tmp1, xLocal, xLocal, len);
        AscendC::ReduceSum(tmp2, tmp1, zLocal, len);
        float var = tmp2.GetValue(0) / (float)len;
        
        // x = x / sqrt(var + eps)
        float invStd = 1.0f / sqrtf(var + 1e-5f);
        AscendC::Muls(xLocal, xLocal, invStd, len);
        
        // Step 2: Apply gamma and beta (affine transform)
        // Load gamma and beta for this group
        // gamma[group*cpg : (group+1)*cpg], beta[group*cpg : (group+1)*cpg]
        AscendC::DataCopy(tmp1, gammaGm[group * channelsPerGroup], len);
        AscendC::DataCopy(tmp2, betaGm[group * channelsPerGroup], len);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(0);
        
        AscendC::Mul(xLocal, xLocal, tmp1, len);
        AscendC::Add(xLocal, xLocal, tmp2, len);
        
        // Step 3: Swish: x = x * sigmoid(x)
        // sigmoid(x) = 1 / (1 + exp(-x))
        AscendC::Muls(tmp1, xLocal, -1.0f, len);
        AscendC::Exp(tmp1, tmp1, len);
        AscendC::Adds(tmp1, tmp1, 1.0f, len);
        AscendC::Reciprocal(tmp1, tmp1, len);
        AscendC::Mul(xLocal, xLocal, tmp1, len);
        
        // Step 4: Multiply by multiply_weight
        AscendC::DataCopy(tmp1, mulWGm[group * channelsPerGroup], len);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(1);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(1);
        AscendC::Mul(xLocal, xLocal, tmp1, len);
        
        // Step 5: Swish again: x = x * sigmoid(x)
        AscendC::Muls(tmp1, xLocal, -1.0f, len);
        AscendC::Exp(tmp1, tmp1, len);
        AscendC::Adds(tmp1, tmp1, 1.0f, len);
        AscendC::Reciprocal(tmp1, tmp1, len);
        AscendC::Mul(zLocal, xLocal, tmp1, len);
        
        outQueueZ.EnQue(zLocal);
        inQueueX.FreeTensor(xLocal);
        tmpBuf1.FreeTensor(tmp1);
        tmpBuf2.FreeTensor(tmp2);
    }
    
    __aicore__ inline void CopyOut(uint32_t row, uint32_t group)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        uint32_t offset = row * outFeatures + group * channelsPerGroup;
        AscendC::DataCopy(zGm[offset], zLocal, channelsPerGroup);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TQue<AscendC::TPosition::VECCALC, BUFFER_NUM> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    AscendC::GlobalTensor<float> mulWGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t outFeatures;
    uint32_t numGroups;
    uint32_t channelsPerGroup;
    uint32_t rowStart;
    uint32_t rowEnd;
    uint32_t rowCount;
};

extern "C" __global__ __aicore__ void gemm_group_norm_swish_multiply_swish_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, 
    GM_ADDR gamma, GM_ADDR beta, GM_ADDR multiply_weight,
    GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormSwishMultiplySwish op;
    op.Init(x, weight, bias, gamma, beta, multiply_weight, z,
            tiling_data.batchSize, tiling_data.outFeatures,
            tiling_data.numGroups, tiling_data.channelsPerGroup,
            tiling_data.tileNum);
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor gemm_group_norm_swish_multiply_swish_custom_impl_npu(
    const at::Tensor& x, const at::Tensor& weight, const at::Tensor& bias,
    const at::Tensor& gamma, const at::Tensor& beta, const at::Tensor& multiply_weight) {
    
    // First perform GEMM: x @ weight.T + bias
    at::Tensor gemm_out = at::linear(x, weight, bias);
    
    // Then call custom kernel for GroupNorm+Swish+Multiply+Swish
    at::Tensor result = at::empty_like(gemm_out);
    EXEC_NPU_CMD(aclnnGemmGroupNormSwishMultiplySwishCustom, 
                  gemm_out, weight, bias, gamma, beta, multiply_weight, result);
    return result;
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model using custom AscendC kernel for fused GroupNorm+Swish+Multiply+Swish.
    GEMM is performed via PyTorch, then the fused kernel handles the rest.
    """
    def __init__(self, in_features, out_features, num_groups, multiply_weight_shape):
        super(ModelNew, self).__init__()
        self.gemm = torch.nn.Linear(in_features, out_features)
        self.group_norm = torch.nn.GroupNorm(num_groups, out_features)
        self.multiply_weight = torch.nn.Parameter(torch.randn(multiply_weight_shape))

    def forward(self, x):
        return custom_ops_lib.gemm_group_norm_swish_multiply_swish_custom(
            x,
            self.gemm.weight,
            self.gemm.bias,
            self.group_norm.weight,
            self.group_norm.bias,
            self.multiply_weight
        )
