kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelPowerBroadcast {
public:
    __aicore__ inline KernelPowerBroadcast() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR exponent, GM_ADDR z, uint32_t totalLength, uint32_t tileNum, uint32_t rows, uint32_t cols)
    {
        this->cols = cols;
        this->rows = rows;
        // Each block handles a subset of rows
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        this->rowsPerBlock = rows / blockNum;
        this->blockLength = this->rowsPerBlock * cols;
        this->tileNum = tileNum;
        // Each tile processes tileLength elements (must be multiple of cols for correct broadcasting)
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        // Make tileLength a multiple of cols
        if (this->tileLength < cols) {
            this->tileLength = cols;
        } else {
            this->tileLength = (this->tileLength / cols) * cols;
        }
        
        // Recalculate actual loop count
        this->loopCount = this->blockLength / this->tileLength;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->rowsPerBlock * cols * blockIdx, this->blockLength);
        expGm.SetGlobalBuffer((__gm__ float *)exponent, cols);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->rowsPerBlock * cols * blockIdx, this->blockLength);
        
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueExp, BUFFER_NUM, cols * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf1, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (int32_t i = 0; i < this->loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> expLocal = inQueueExp.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(expLocal, expGm[0], this->cols);
        inQueueX.EnQue(xLocal);
        inQueueExp.EnQue(expLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> expLocal = inQueueExp.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> lnLocal = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> broadcastExp = tmpBuf2.Get<float>();
        
        // Broadcast exponent to match tileLength
        // tileLength is a multiple of cols, so we can replicate exponent
        uint32_t numRowsInTile = this->tileLength / this->cols;
        for (uint32_t r = 0; r < numRowsInTile; r++) {
            AscendC::DataCopy(broadcastExp[r * this->cols], expLocal, this->cols);
        }
        
        // Compute x^exponent = exp(exponent * ln(x))
        // ln(x)
        AscendC::Ln(lnLocal, xLocal, this->tileLength);
        // exponent * ln(x)
        AscendC::Mul(lnLocal, broadcastExp, lnLocal, this->tileLength);
        // exp(exponent * ln(x))
        AscendC::Exp(zLocal, lnLocal, this->tileLength);
        
        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueExp.FreeTensor(expLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueExp;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> expGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t loopCount;
    uint32_t rows;
    uint32_t cols;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void power_broadcast_custom(GM_ADDR x, GM_ADDR exponent, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelPowerBroadcast op;
    op.Init(x, exponent, z, tiling_data.totalLength, tiling_data.tileNum, tiling_data.rows, tiling_data.cols);
    op.Process();
}
