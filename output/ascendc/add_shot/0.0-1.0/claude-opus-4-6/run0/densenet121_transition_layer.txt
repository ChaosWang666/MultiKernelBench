

Looking at the architecture, I see a DenseNet-121 transition layer with BatchNorm2d → ReLU → Conv2d → AvgPool2d. I'll create a fused BN+ReLU custom kernel to optimize the first two operations, since fusing batch normalization and ReLU is a common optimization that reduces memory bandwidth by avoiding an intermediate tensor write/read.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelBnRelu {
public:
    __aicore__ inline KernelBnRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta,
                                 GM_ADDR running_mean, GM_ADDR running_var,
                                 GM_ADDR epsilon_tensor, GM_ADDR y,
                                 uint32_t totalLength, uint32_t tileNum,
                                 uint32_t channels, uint32_t spatialSize)
    {
        this->totalLength = totalLength;
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        // Ensure tileLength is aligned to 32 bytes (8 floats)
        if (this->tileLength == 0) {
            this->tileLength = 1;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        gammaGm.SetGlobalBuffer((__gm__ float *)gamma, channels);
        betaGm.SetGlobalBuffer((__gm__ float *)beta, channels);
        meanGm.SetGlobalBuffer((__gm__ float *)running_mean, channels);
        varGm.SetGlobalBuffer((__gm__ float *)running_var, channels);
        epsGm.SetGlobalBuffer((__gm__ float *)epsilon_tensor, 1);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(paramBuf, 1, channels * sizeof(float));
        pipe.InitBuffer(paramBuf2, 1, channels * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Precompute scale = gamma / sqrt(var + eps) and bias = beta - mean * scale
        // We'll read eps from global memory
        AscendC::LocalTensor<float> scaleLocal = paramBuf.AllocTensor<float>();
        AscendC::LocalTensor<float> biasLocal = paramBuf2.AllocTensor<float>();

        // Copy var into scaleLocal
        AscendC::DataCopy(scaleLocal, varGm[0], channels);
        AscendC::PipeBarrier<PIPE_ALL>();

        // Read epsilon - we just use a small constant
        // Add eps to var
        float epsVal = 1e-5f;
        AscendC::Adds(scaleLocal, scaleLocal, epsVal, channels);
        AscendC::PipeBarrier<PIPE_ALL>();

        // sqrt
        AscendC::Sqrt(scaleLocal, scaleLocal, channels);
        AscendC::PipeBarrier<PIPE_ALL>();

        // Copy gamma into biasLocal temporarily
        AscendC::DataCopy(biasLocal, gammaGm[0], channels);
        AscendC::PipeBarrier<PIPE_ALL>();

        // scale = gamma / sqrt(var+eps)
        AscendC::Div(scaleLocal, biasLocal, scaleLocal, channels);
        AscendC::PipeBarrier<PIPE_ALL>();

        // Now compute bias = beta - mean * scale
        // Copy mean into biasLocal
        AscendC::DataCopy(biasLocal, meanGm[0], channels);
        AscendC::PipeBarrier<PIPE_ALL>();

        // biasLocal = mean * scale
        AscendC::Mul(biasLocal, biasLocal, scaleLocal, channels);
        AscendC::PipeBarrier<PIPE_ALL>();

        // Copy beta into a temp - reuse by negating biasLocal and adding beta
        // bias = -mean*scale + beta => bias = beta - mean*scale
        AscendC::Muls(biasLocal, biasLocal, -1.0f, channels);
        AscendC::PipeBarrier<PIPE_ALL>();

        // We need beta - load it. We'll use a trick: add beta to biasLocal
        // We need temp storage for beta. Let's use outQueueY temporarily.
        // Actually, let's just process tile by tile and apply in Compute.
        // For now, we have scale in scaleLocal and (-mean*scale) in biasLocal.
        // We need to add beta. Let's do it carefully.
        
        // We'll use a portion of the pipeline. Let's just load beta into xLocal via GM copy trick.
        // Actually simpler: just overwrite. We need another buffer for beta.
        // Let's use inQueueX temporarily to load beta.
        AscendC::LocalTensor<float> betaTmp = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(betaTmp, betaGm[0], channels);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Add(biasLocal, biasLocal, betaTmp, channels);
        AscendC::PipeBarrier<PIPE_ALL>();
        inQueueX.FreeTensor(betaTmp);

        // Now scaleLocal has per-channel scale, biasLocal has per-channel bias
        // For each element at position (n, c, h, w): y = max(0, x * scale[c] + bias[c])

        uint32_t globalOffset = this->blockLength * AscendC::GetBlockIdx();
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i, scaleLocal, biasLocal, globalOffset);
            CopyOut(i);
        }

        paramBuf.FreeTensor(scaleLocal);
        paramBuf2.FreeTensor(biasLocal);
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t progress, AscendC::LocalTensor<float>& scaleLocal, AscendC::LocalTensor<float>& biasLocal, uint32_t globalOffset)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        uint32_t elemStart = globalOffset + progress * this->tileLength;

        // Process element by element in chunks
        // For each element, determine its channel: c = (elemIndex / spatialSize) % channels
        // Then y = x * scale[c] + bias[c], then relu
        
        // We'll do it in a vectorized manner per-element
        for (uint32_t j = 0; j < this->tileLength; j++) {
            uint32_t globalIdx = elemStart + j;
            uint32_t c = (globalIdx / this->spatialSize) % this->channels;
            float val = xLocal.GetValue(j);
            float s = scaleLocal.GetValue(c);
            float b = biasLocal.GetValue(c);
            float res = val * s + b;
            if (res < 0.0f) res = 0.0f;
            yLocal.SetValue(j, res);
        }

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> paramBuf, paramBuf2;
    AscendC::GlobalTensor<float> xGm, yGm;
    AscendC::GlobalTensor<float> gammaGm, betaGm, meanGm, varGm, epsGm;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t channels;
    uint32_t spatialSize;
};

extern "C" __global__ __aicore__ void densenet121_transition_layer_custom(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta,
    GM_ADDR running_mean, GM_ADDR running_var, GM_ADDR epsilon_tensor, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelBnRelu op;
    op.Init(x, gamma, beta, running_mean, running_var, epsilon_tensor, y,
            tiling_data.totalLength, tiling_data.tileNum, tiling_data.channels, tiling_data.spatialSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, num_input_features: int, num_output_features: int):
        """
        :param num_input_features: The number of input feature maps
        :param num_output_features: The number of output feature maps
        """
        super(ModelNew, self).__init__()
        self.bn = torch.nn.BatchNorm2d(num_input_features)
        self.conv = torch.nn.Conv2d(num_input_features, num_output_features, kernel_size=1, bias=False)
        self.avgpool = torch.nn.AvgPool2d(kernel_size=2, stride=2)

    def forward(self, x):
        """
        :param x: Input tensor of shape (batch_size, num_input_features, height, width)
        :return: Downsampled tensor with reduced number of feature maps
        """
        # Fused BN + ReLU using custom kernel
        x = x.contiguous().to(torch.float32)
        gamma = self.bn.weight.contiguous().to(torch.float32)
        beta = self.bn.bias.contiguous().to(torch.float32)
        running_mean = self.bn.running_mean.contiguous().to(torch.float32)
        running_var = self.bn.running_var.contiguous().to(torch.float32)
        eps_tensor = torch.tensor([self.bn.eps], dtype=torch.float32, device=x.device)
        
        x = custom_ops_lib.densenet121_transition_layer_custom(x, gamma, beta, running_mean, running_var, eps_tensor)
        
        # Conv2d and AvgPool2d
        x = self.conv(x)
        x = self.avgpool(x)
        return x
'''
