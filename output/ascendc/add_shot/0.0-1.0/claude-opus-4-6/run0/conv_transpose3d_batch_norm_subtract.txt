

Looking at the architecture, I need to identify the operation to optimize. The model performs:
1. ConvTranspose3d
2. BatchNorm3d
3. Subtract mean along spatial dimensions (2, 3, 4)

The subtract-mean operation is a good candidate for a custom kernel since it's a relatively simple operation that can benefit from fusion. Let me create a custom kernel that computes `x - mean(x, dim=(2,3,4), keepdim=True)` for a 5D tensor.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSubtractMean {
public:
    __aicore__ inline KernelSubtractMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t channels,
                                 uint32_t spatialSize, uint32_t totalLength, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->totalBC = batchSize * channels;
        this->tileNum = tileNum;

        // Each block processes a subset of (batch, channel) pairs
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->bcStart = blockIdx * ((this->totalBC + blockNum - 1) / blockNum);
        this->bcEnd = (blockIdx + 1) * ((this->totalBC + blockNum - 1) / blockNum);
        if (this->bcEnd > this->totalBC) {
            this->bcEnd = this->totalBC;
        }
        if (this->bcStart > this->totalBC) {
            this->bcStart = this->totalBC;
        }
        this->bcCount = this->bcEnd - this->bcStart;

        // Align tile length to 32 bytes (8 floats)
        uint32_t alignNum = 8;
        this->spatialAligned = (spatialSize + alignNum - 1) / alignNum * alignNum;

        xGm.SetGlobalBuffer((__gm__ float *)x, totalLength);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalLength);

        // We process each (b,c) spatial slice. Use tiling within the spatial dimension.
        // Buffer size for one tile of spatial data
        uint32_t tileSpatial = (this->spatialAligned + BUFFER_NUM * tileNum - 1) / (BUFFER_NUM * tileNum);
        tileSpatial = (tileSpatial + alignNum - 1) / alignNum * alignNum;
        if (tileSpatial < alignNum) tileSpatial = alignNum;
        this->tileSpatialLen = tileSpatial;

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileSpatialLen * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileSpatialLen * sizeof(float));
        pipe.InitBuffer(tmpBuf, 1, this->tileSpatialLen * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t bc = 0; bc < this->bcCount; bc++) {
            uint32_t globalBcIdx = this->bcStart + bc;
            uint32_t baseOffset = globalBcIdx * this->spatialSize;

            // First pass: compute sum
            float totalSum = 0.0f;
            uint32_t remaining = this->spatialSize;
            uint32_t offset = 0;

            while (remaining > 0) {
                uint32_t curLen = this->tileSpatialLen;
                if (curLen > remaining) curLen = remaining;
                // Align curLen for DataCopy
                uint32_t alignNum = 8;
                uint32_t copyLen = (curLen + alignNum - 1) / alignNum * alignNum;

                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[baseOffset + offset], copyLen);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<float> xProc = inQueueX.DeQue<float>();
                // Zero out padding elements if needed
                if (curLen < copyLen) {
                    for (uint32_t i = curLen; i < copyLen; i++) {
                        xProc.SetValue(i, 0.0f);
                    }
                }
                // Partial sum using ReduceSum
                AscendC::LocalTensor<float> workLocal = tmpBuf.Get<float>();
                float partialSum = 0.0f;
                AscendC::ReduceSum(workLocal, xProc, workLocal, curLen);
                partialSum = workLocal.GetValue(0);
                totalSum += partialSum;
                inQueueX.FreeTensor(xProc);

                offset += curLen;
                remaining -= curLen;
            }

            float meanVal = totalSum / (float)this->spatialSize;

            // Second pass: subtract mean and write output
            remaining = this->spatialSize;
            offset = 0;
            while (remaining > 0) {
                uint32_t curLen = this->tileSpatialLen;
                if (curLen > remaining) curLen = remaining;
                uint32_t alignNum = 8;
                uint32_t copyLen = (curLen + alignNum - 1) / alignNum * alignNum;

                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[baseOffset + offset], copyLen);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<float> xProc = inQueueX.DeQue<float>();
                AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
                AscendC::Adds(zLocal, xProc, -meanVal, copyLen);
                outQueueZ.EnQue(zLocal);
                inQueueX.FreeTensor(xProc);

                AscendC::LocalTensor<float> zOut = outQueueZ.DeQue<float>();
                AscendC::DataCopy(yGm[baseOffset + offset], zOut, copyLen);
                outQueueZ.FreeTensor(zOut);

                offset += curLen;
                remaining -= curLen;
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t spatialAligned;
    uint32_t totalBC;
    uint32_t bcStart;
    uint32_t bcEnd;
    uint32_t bcCount;
    uint32_t tileNum;
    uint32_t tileSpatialLen;
};

extern "C" __global__ __aicore__ void conv_transpose3d_batch_norm_subtract_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSubtractMean op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.channels,
            tiling_data.spatialSize, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    A 3D convolutional transpose layer followed by Batch Normalization and subtraction.
    Uses custom AscendC kernel for the mean subtraction step.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, bias=True):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, bias=bias)
        self.batch_norm = torch.nn.BatchNorm3d(out_channels)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.batch_norm(x)
        x = x.contiguous()
        x = custom_ops_lib.conv_transpose3d_batch_norm_subtract_custom(x)
        return x
'''
