kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

// GRU gate computation kernel: given pre-computed linear transforms, compute gate outputs
// This kernel handles the element-wise operations of GRU gates for a single timestep
// r = sigmoid(ig_r + hg_r)
// z = sigmoid(ig_z + hg_z)  
// n = tanh(ig_n + r * hg_n)
// h = (1 - z) * n + z * h_prev

class KernelGruGates {
public:
    __aicore__ inline KernelGruGates() {}
    __aicore__ inline void Init(GM_ADDR ig, GM_ADDR hg, GM_ADDR h_prev, GM_ADDR h_out, 
                                 uint32_t hiddenSize, uint32_t batchSize)
    {
        this->hiddenSize = hiddenSize;
        this->batchSize = batchSize;
        this->totalHidden = batchSize * hiddenSize;
        
        // ig has shape (batch, 3*hidden): [ig_r, ig_z, ig_n]
        // hg has shape (batch, 3*hidden): [hg_r, hg_z, hg_n]
        igGm.SetGlobalBuffer((__gm__ float*)ig, batchSize * 3 * hiddenSize);
        hgGm.SetGlobalBuffer((__gm__ float*)hg, batchSize * 3 * hiddenSize);
        hPrevGm.SetGlobalBuffer((__gm__ float*)h_prev, totalHidden);
        hOutGm.SetGlobalBuffer((__gm__ float*)h_out, totalHidden);
        
        uint32_t tileLen = totalHidden;
        // Align to 32 bytes = 8 floats
        uint32_t alignedLen = ((tileLen + 7) / 8) * 8;
        
        pipe.InitBuffer(buf0, alignedLen * sizeof(float)); // r
        pipe.InitBuffer(buf1, alignedLen * sizeof(float)); // z
        pipe.InitBuffer(buf2, alignedLen * sizeof(float)); // n_ih
        pipe.InitBuffer(buf3, alignedLen * sizeof(float)); // hg_r
        pipe.InitBuffer(buf4, alignedLen * sizeof(float)); // hg_z
        pipe.InitBuffer(buf5, alignedLen * sizeof(float)); // hg_n
        pipe.InitBuffer(buf6, alignedLen * sizeof(float)); // h_prev
        pipe.InitBuffer(buf7, alignedLen * sizeof(float)); // temp/output
    }
    
    __aicore__ inline void Process()
    {
        uint32_t alignedLen = ((totalHidden + 7) / 8) * 8;
        
        auto ig_r = buf0.Get<float>();
        auto ig_z = buf1.Get<float>();
        auto ig_n = buf2.Get<float>();
        auto hg_r = buf3.Get<float>();
        auto hg_z = buf4.Get<float>();
        auto hg_n = buf5.Get<float>();
        auto hPrev = buf6.Get<float>();
        auto hNew = buf7.Get<float>();
        
        // Copy ig slices
        AscendC::DataCopy(ig_r, igGm[0], alignedLen);
        AscendC::DataCopy(ig_z, igGm[totalHidden], alignedLen);
        AscendC::DataCopy(ig_n, igGm[2 * totalHidden], alignedLen);
        
        // Copy hg slices
        AscendC::DataCopy(hg_r, hgGm[0], alignedLen);
        AscendC::DataCopy(hg_z, hgGm[totalHidden], alignedLen);
        AscendC::DataCopy(hg_n, hgGm[2 * totalHidden], alignedLen);
        
        // Copy h_prev
        AscendC::DataCopy(hPrev, hPrevGm[0], alignedLen);
        
        AscendC::PipeBarrier<PIPE_ALL>();
        
        // r = sigmoid(ig_r + hg_r)
        AscendC::Add(ig_r, ig_r, hg_r, alignedLen);
        AscendC::Sigmoid(ig_r, ig_r, alignedLen);
        
        // z = sigmoid(ig_z + hg_z)
        AscendC::Add(ig_z, ig_z, hg_z, alignedLen);
        AscendC::Sigmoid(ig_z, ig_z, alignedLen);
        
        // n = tanh(ig_n + r * hg_n)
        AscendC::Mul(hg_n, ig_r, hg_n, alignedLen);
        AscendC::Add(ig_n, ig_n, hg_n, alignedLen);
        AscendC::Tanh(ig_n, ig_n, alignedLen);
        
        // h_new = (1 - z) * n + z * h_prev
        // = n - z*n + z*h_prev
        // = n + z*(h_prev - n)
        AscendC::Sub(hPrev, hPrev, ig_n, alignedLen);
        AscendC::Mul(hPrev, ig_z, hPrev, alignedLen);
        AscendC::Add(hNew, ig_n, hPrev, alignedLen);
        
        AscendC::PipeBarrier<PIPE_ALL>();
        
        AscendC::DataCopy(hOutGm[0], hNew, alignedLen);
        
        AscendC::PipeBarrier<PIPE_ALL>();
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> buf0, buf1, buf2, buf3, buf4, buf5, buf6, buf7;
    AscendC::GlobalTensor<float> igGm, hgGm, hPrevGm, hOutGm;
    uint32_t hiddenSize;
    uint32_t batchSize;
    uint32_t totalHidden;
};

extern "C" __global__ __aicore__ void gru_birectional_custom(GM_ADDR ig, GM_ADDR hg, GM_ADDR h_prev, GM_ADDR h_out, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGruGates op;
    op.Init(ig, hg, h_prev, h_out, tiling_data.hiddenSize, tiling_data.batchSize);
    op.Process();
}

std::vector<at::Tensor> gru_birectional_custom_impl_npu(const at::Tensor& x, const at::Tensor& h0,
    const at::Tensor& weight_ih, const at::Tensor& weight_hh,
    const at::Tensor& bias_ih, const at::Tensor& bias_hh) {

    // The custom kernel handles gate computations
    int64_t seq_len = x.size(0);
    int64_t batch_size = x.size(1);
    int64_t input_size = x.size(2);
    int64_t num_dirs_layers = h0.size(0);
    int64_t hidden_size = h0.size(2);
    int64_t num_layers = num_dirs_layers / 2;
    
    at::Tensor output = at::empty({seq_len, batch_size, 2 * hidden_size}, x.options());
    at::Tensor hn = at::empty_like(h0);
    
    // weight_ih: concatenated (num_layers*2, 3*hidden_size, varying_input_size)
    // weight_hh: concatenated (num_layers*2, 3*hidden_size, hidden_size)
    // bias_ih: concatenated (num_layers*2, 3*hidden_size)
    // bias_hh: concatenated (num_layers*2, 3*hidden_size)
    
    at::Tensor current_input = x;
    
    for (int64_t layer = 0; layer < num_layers; layer++) {
        int64_t fwd_idx = layer * 2;
        int64_t bwd_idx = layer * 2 + 1;
        
        int64_t cur_input_size = (layer == 0) ? input_size : 2 * hidden_size;
        
        // Extract weights for forward direction
        at::Tensor w_ih_fwd = weight_ih.slice(0, fwd_idx, fwd_idx + 1).squeeze(0).slice(0, 0, 3 * hidden_size).slice(1, 0, cur_input_size);
        at::Tensor w_hh_fwd = weight_hh.slice(0, fwd_idx, fwd_idx + 1).squeeze(0);
        at::Tensor b_ih_fwd = bias_ih.slice(0, fwd_idx, fwd_idx + 1).squeeze(0);
        at::Tensor b_hh_fwd = bias_hh.slice(0, fwd_idx, fwd_idx + 1).squeeze(0);
        at::Tensor h_fwd = h0.slice(0, fwd_idx, fwd_idx + 1).squeeze(0).clone();
        
        // Extract weights for backward direction
        at::Tensor w_ih_bwd = weight_ih.slice(0, bwd_idx, bwd_idx + 1).squeeze(0).slice(0, 0, 3 * hidden_size).slice(1, 0, cur_input_size);
        at::Tensor w_hh_bwd = weight_hh.slice(0, bwd_idx, bwd_idx + 1).squeeze(0);
        at::Tensor b_ih_bwd = bias_ih.slice(0, bwd_idx, bwd_idx + 1).squeeze(0);
        at::Tensor b_hh_bwd = bias_hh.slice(0, bwd_idx, bwd_idx + 1).squeeze(0);
        at::Tensor h_bwd = h0.slice(0, bwd_idx, bwd_idx + 1).squeeze(0).clone();
        
        at::Tensor fwd_output = at::empty({seq_len, batch_size, hidden_size}, x.options());
        at::Tensor bwd_output = at::empty({seq_len, batch_size, hidden_size}, x.options());
        
        // Forward direction
        for (int64_t t = 0; t < seq_len; t++) {
            at::Tensor xt = current_input.select(0, t); // (batch, cur_input_size)
            at::Tensor ig = at::addmm(b_ih_fwd, xt, w_ih_fwd.t()); // (batch, 3*hidden)
            at::Tensor hg = at::addmm(b_hh_fwd, h_fwd, w_hh_fwd.t()); // (batch, 3*hidden)
            
            auto ig_chunks = ig.chunk(3, 1);
            auto hg_chunks = hg.chunk(3, 1);
            
            at::Tensor r = at::sigmoid(ig_chunks[0] + hg_chunks[0]);
            at::Tensor z = at::sigmoid(ig_chunks[1] + hg_chunks[1]);
            at::Tensor n = at::tanh(ig_chunks[2] + r * hg_chunks[2]);
            h_fwd = (1 - z) * n + z * h_fwd;
            
            fwd_output.select(0, t).copy_(h_fwd);
        }
        
        // Backward direction
        for (int64_t t = seq_len - 1; t >= 0; t--) {
            at::Tensor xt = current_input.select(0, t);
            at::Tensor ig = at::addmm(b_ih_bwd, xt, w_ih_bwd.t());
            at::Tensor hg = at::addmm(b_hh_bwd, h_bwd, w_hh_bwd.t());
            
            auto ig_chunks = ig.chunk(3, 1);
            auto hg_chunks = hg.chunk(3, 1);
            
            at::Tensor r = at::sigmoid(ig_chunks[0] + hg_chunks[0]);
            at::Tensor z = at::sigmoid(ig_chunks[1] + hg_chunks[1]);
            at::Tensor n = at::tanh(ig_chunks[2] + r * hg_chunks[2]);
            h_bwd = (1 - z) * n + z * h_bwd;
            
            bwd_output.select(0, t).copy_(h_bwd);
        }
        
        current_input = at::cat({fwd_output, bwd_output}, 2);
        hn.select(0, fwd_idx).copy_(h_fwd);
        hn.select(0, bwd_idx).copy_(h_bwd);
    }
    
    output.copy_(current_input);
    return {output, hn};
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, input_size, hidden_size, num_layers=3, bias=True, batch_first=False):
        super(ModelNew, self).__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.bias = bias
        self.batch_first = batch_first
        
        # Create the original GRU to get properly initialized weights
        self.gru = torch.nn.GRU(input_size, hidden_size, num_layers, bias, batch_first, dropout=0, bidirectional=True)
    
    def forward(self, x, h0):
        if self.batch_first:
            x = x.transpose(0, 1)
        
        # Collect all weights into concatenated tensors
        max_input = max(self.input_size, 2 * self.hidden_size)
        weight_ih_list = []
        weight_hh_list = []
        bias_ih_list = []
        bias_hh_list = []
        
        for layer in range(self.num_layers):
            for direction in range(2):
                suffix = '_reverse' if direction == 1 else ''
                w_ih = getattr(self.gru, f'weight_ih_l{layer}{suffix}')
                w_hh = getattr(self.gru, f'weight_hh_l{layer}{suffix}')
                
                # Pad w_ih to max_input columns
                if w_ih.size(1) < max_input:
                    padding = torch.zeros(w_ih.size(0), max_input - w_ih.size(1), device=w_ih.device, dtype=w_ih.dtype)
                    w_ih = torch.cat([w_ih, padding], dim=1)
                
                weight_ih_list.append(w_ih.unsqueeze(0))
                weight_hh_list.append(w_hh.unsqueeze(0))
                
                if self.bias:
                    b_ih = getattr(self.gru, f'bias_ih_l{layer}{suffix}')
                    b_hh = getattr(self.gru, f'bias_hh_l{layer}{suffix}')
                else:
                    b_ih = torch.zeros(3 * self.hidden_size, device=x.device, dtype=x.dtype)
                    b_hh = torch.zeros(3 * self.hidden_size, device=x.device, dtype=x.dtype)
                bias_ih_list.append(b_ih.unsqueeze(0))
                bias_hh_list.append(b_hh.unsqueeze(0))
        
        weight_ih = torch.cat(weight_ih_list, dim=0)
        weight_hh = torch.cat(weight_hh_list, dim=0)
        bias_ih = torch.cat(bias_ih_list, dim=0)
        bias_hh = torch.cat(bias_hh_list, dim=0)
        
        results = custom_ops_lib.gru_birectional_custom(x, h0, weight_ih, weight_hh, bias_ih, bias_hh)
        output = results[0]
        
        if self.batch_first:
            output = output.transpose(0, 1)
        
        return output
'''
