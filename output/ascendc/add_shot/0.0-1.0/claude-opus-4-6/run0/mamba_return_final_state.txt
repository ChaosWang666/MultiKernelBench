

Looking at this architecture, I need to analyze the `forward` method to identify the key computational bottleneck. The final operation extracts `new_states[:, -1]` from an einsum result. The model computes a complex series of operations involving segment sums, exponentials, einsum operations, and state recurrence.

The most impactful operation to optimize is the final return which extracts the last state from the inter-chunk recurrence computation. Specifically, the operation `torch.einsum("bhzc,bchpn->bzhpn", decay_chunk, states)[:, -1]` computes all states but only returns the final one. We can fuse this to only compute the last slice, which is essentially `torch.einsum("bhc,bchpn->bhpn", decay_chunk[:,:,-1,:], states)` - computing only the last z-index.

Let me create a custom kernel for this fused operation: given `decay_chunk` of shape `(b, h, z, c)` and `states` of shape `(b, c, h, p, n)`, compute `result[b,h,p,n] = sum_c(decay_chunk[b,h,-1,c] * states[b,c,h,p,n])`.

kernel_src="""
#include "kernel_operator.h"

// decay_chunk: (b, h, z, c) - we only need z = last index (z-1)
// states: (b, c, h, p, n)
// output: (b, h, p, n) = sum_c decay_chunk[b,h,z-1,c] * states[b,c,h,p,n]

constexpr int32_t BUFFER_NUM = 2;

class KernelMambaReturnFinalState {
public:
    __aicore__ inline KernelMambaReturnFinalState() {}
    __aicore__ inline void Init(GM_ADDR decay_chunk, GM_ADDR states, GM_ADDR output,
                                 uint32_t batchSize, uint32_t nHeads, uint32_t numC,
                                 uint32_t dHead, uint32_t dState, uint32_t totalOutput, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->nHeads = nHeads;
        this->numC = numC;  // full c dimension (numChunks+1)
        this->dHead = dHead;
        this->dState = dState;
        this->totalOutput = totalOutput;

        // Each block handles a portion of the output elements
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->outputPerBlock = (totalOutput + blockNum - 1) / blockNum;
        this->startIdx = blockIdx * this->outputPerBlock;
        this->endIdx = startIdx + this->outputPerBlock;
        if (this->endIdx > totalOutput) this->endIdx = totalOutput;
        if (this->startIdx >= totalOutput) {
            this->startIdx = 0;
            this->endIdx = 0;
        }
        this->myLength = this->endIdx - this->startIdx;

        // pn = dHead * dState
        this->pn = dHead * dState;

        decayGm.SetGlobalBuffer((__gm__ float*)decay_chunk, batchSize * nHeads * numC * numC);
        statesGm.SetGlobalBuffer((__gm__ float*)states, batchSize * numC * nHeads * dHead * dState);
        outputGm.SetGlobalBuffer((__gm__ float*)output, totalOutput);

        // We'll process in tiles
        this->tileNum = tileNum;
        if (this->myLength == 0) {
            this->tileLength = 0;
        } else {
            // Align tile length to 8 (32 bytes / 4 bytes per float)
            uint32_t totalTiles = tileNum * BUFFER_NUM;
            this->tileLength = (this->myLength + totalTiles - 1) / totalTiles;
            // align to 8
            this->tileLength = ((this->tileLength + 7) / 8) * 8;
        }

        if (this->tileLength > 0 && this->myLength > 0) {
            pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLength * sizeof(float));
            pipe.InitBuffer(tmpBuf, 1, this->tileLength * sizeof(float));
        }
    }

    __aicore__ inline void Process()
    {
        if (this->myLength == 0) return;

        // For each output element [b, h, p, n]:
        //   output = sum_c decay_chunk[b, h, lastZ, c] * states[b, c, h, p, n]
        // where lastZ = numC - 1
        // We process tiles of output elements

        uint32_t processed = 0;
        int32_t loopCount = this->tileNum * BUFFER_NUM;

        for (int32_t tile = 0; tile < loopCount; tile++) {
            uint32_t curLen = this->tileLength;
            if (processed + curLen > this->myLength) {
                curLen = this->myLength - processed;
                if (curLen == 0) break;
            }
            ComputeTile(processed, curLen);
            processed += curLen;
        }
    }

private:
    __aicore__ inline void ComputeTile(uint32_t offset, uint32_t len)
    {
        // Allocate output tensor
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        // Zero out the output
        uint32_t alignedLen = ((len + 7) / 8) * 8;
        AscendC::Duplicate(outLocal, 0.0f, alignedLen);

        uint32_t lastZ = this->numC - 1;

        // For each c in [0, numC):
        //   Load decay_chunk[b, h, lastZ, c] for each element in tile
        //   Load states[b, c, h, p, n] for each element in tile
        //   Multiply and accumulate

        for (uint32_t c = 0; c < this->numC; c++) {
            // For each element in the tile, compute the global output index
            // outputIdx = startIdx + offset + i
            // From outputIdx, derive b, h, p, n:
            // outputIdx = b * (nHeads * dHead * dState) + h * (dHead * dState) + p * dState + n

            for (uint32_t i = 0; i < len; i++) {
                uint32_t gIdx = this->startIdx + offset + i;
                uint32_t rem = gIdx;
                uint32_t b_idx = rem / (nHeads * pn);
                rem = rem % (nHeads * pn);
                uint32_t h_idx = rem / pn;
                rem = rem % pn;
                uint32_t p_idx = rem / dState;
                uint32_t n_idx = rem % dState;

                // decay_chunk index: (b, h, lastZ, c) -> b*h*numC*numC + h*numC*numC + lastZ*numC + c
                uint32_t decayIdx = b_idx * nHeads * numC * numC + h_idx * numC * numC + lastZ * numC + c;
                float decayVal = *(((__gm__ float*)decayGm.GetPhyAddr()) + decayIdx);

                // states index: (b, c, h, p, n) -> b*numC*nHeads*dHead*dState + c*nHeads*dHead*dState + h*dHead*dState + p*dState + n
                uint32_t statesIdx = b_idx * numC * nHeads * dHead * dState +
                                     c * nHeads * dHead * dState +
                                     h_idx * dHead * dState +
                                     p_idx * dState + n_idx;
                float stateVal = *(((__gm__ float*)statesGm.GetPhyAddr()) + statesIdx);

                // outLocal[i] += decayVal * stateVal
                float curVal = outLocal.GetValue(i);
                outLocal.SetValue(i, curVal + decayVal * stateVal);
            }
        }

        outQueue.EnQue(outLocal);

        // Copy out
        AscendC::LocalTensor<float> outResult = outQueue.DeQue<float>();
        AscendC::DataCopy(outputGm[this->startIdx + offset], outResult, alignedLen);
        outQueue.FreeTensor(outResult);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> decayGm;
    AscendC::GlobalTensor<float> statesGm;
    AscendC::GlobalTensor<float> outputGm;
    uint32_t batchSize, nHeads, numC, dHead, dState;
    uint32_t totalOutput;
    uint32_t startIdx, endIdx, myLength;
    uint32_t outputPerBlock;
    uint32_t pn;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void mamba_return_final_state_custom(GM_ADDR decay_chunk, GM_ADDR states, GM_ADDR output, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMambaReturnFinalState op;
    op.Init(decay_chunk, states, output,
            tiling_data.batchSize, tiling_data.nHeads, tiling_data.numChunks,
            tiling_data.dHead, tiling_data.dState, tiling_data.totalOutput, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib
from einops import rearrange

class ModelNew(nn.Module):
    def __init__(self, batch_size, seq_length, n_heads, d_head, d_state, block_len=64):
        super(ModelNew, self).__init__()
        
        assert seq_length % block_len == 0, "Sequence length must be divisible by block length"
        
        self.batch_size = batch_size
        self.seq_length = seq_length
        self.n_heads = n_heads
        self.d_head = d_head
        self.d_state = d_state
        self.block_len = block_len
        
        self.A = nn.Parameter(torch.randn(batch_size, seq_length, n_heads))
        self.B = nn.Parameter(torch.randn(batch_size, seq_length, n_heads, d_state))
        self.C = nn.Parameter(torch.randn(batch_size, seq_length, n_heads, d_state))
        
    def segsum(self, x):
        T = x.size(-1)
        x_cumsum = torch.cumsum(x, dim=-1)
        x_segsum = x_cumsum[..., :, None] - x_cumsum[..., None, :]
        mask = torch.tril(torch.ones(T, T, device=x.device, dtype=bool), diagonal=0)
        x_segsum = x_segsum.masked_fill(~mask, -torch.inf)
        return x_segsum
    
    def forward(self, X, initial_states=None):
        X_blocks, A_blocks, B_blocks, C_blocks = [
            rearrange(x, "b (c l) ... -> b c l ...", l=self.block_len)
            for x in (X, self.A, self.B, self.C)
        ]
        
        A_blocks = rearrange(A_blocks, "b c l h -> b h c l")
        A_cumsum = torch.cumsum(A_blocks, dim=-1)
        
        # 1. Compute diagonal block outputs
        L = torch.exp(self.segsum(A_blocks))
        Y_diag = torch.einsum("bclhn,bcshn,bhcls,bcshp->bclhp", 
                             C_blocks, B_blocks, L, X_blocks)
        
        # 2. Compute intra-chunk states
        decay_states = torch.exp((A_cumsum[:, :, :, -1:] - A_cumsum))
        states = torch.einsum("bclhn,bhcl,bclhp->bchpn", 
                            B_blocks, decay_states, X_blocks)
        
        # 3. Compute inter-chunk recurrence
        if initial_states is None:
            initial_states = torch.zeros_like(states[:, :1])
        states = torch.cat([initial_states, states], dim=1)
        
        decay_chunk = torch.exp(self.segsum(F.pad(A_cumsum[:, :, :, -1], (1, 0))))
        
        # Use custom kernel for the final state computation
        # decay_chunk: (b, h, z, c), states: (b, c, h, p, n) -> output: (b, h, p, n)
        decay_chunk_contig = decay_chunk.contiguous()
        states_contig = states.contiguous()
        result = custom_ops_lib.mamba_return_final_state_custom(decay_chunk_contig, states_contig)
        return result
'''
