

Looking at this architecture, I need to implement a depthwise 2D convolution with a square input and asymmetric kernel (kernel_size x 1). This is a depthwise convolution where each channel is convolved independently with its own filter.

Let me analyze the parameters:
- Input: (batch_size, in_channels, height, width) 
- Kernel: (in_channels, 1, kernel_size, 1) - asymmetric kernel
- Output: (batch_size, in_channels, height_out, width_out)
- height_out = (height + 2*padding - dilation*(kernel_size-1) - 1) / stride + 1
- width_out = width (since kernel width is 1 and padding applies only to height direction... actually padding is symmetric in nn.Conv2d with int padding)

Wait, looking more carefully: `padding` in nn.Conv2d with an int applies to all sides. But the kernel is (kernel_size, 1), so the width dimension effectively has kernel size 1. Let me reconsider - padding as int applies equally to H and W dimensions.

For this depthwise conv: groups=in_channels, so each channel has its own (kernel_size, 1) filter.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConvDw {
public:
    __aicore__ inline KernelConvDw() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels,
                                 uint32_t height, uint32_t width,
                                 uint32_t kernelSize, uint32_t stride,
                                 uint32_t padding, uint32_t dilation,
                                 uint32_t hasBias, uint32_t outHeight,
                                 uint32_t outWidth, uint32_t totalBC)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->height = height;
        this->width = width;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->dilation = dilation;
        this->hasBias = hasBias;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->totalBC = totalBC;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->bcPerBlock = (totalBC + blockNum - 1) / blockNum;
        this->bcStart = blockIdx * this->bcPerBlock;
        this->bcEnd = this->bcStart + this->bcPerBlock;
        if (this->bcEnd > totalBC) this->bcEnd = totalBC;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inChannels * height * width);
        wGm.SetGlobalBuffer((__gm__ float*)weight, inChannels * kernelSize);
        if (hasBias) {
            bGm.SetGlobalBuffer((__gm__ float*)bias, inChannels);
        }
        yGm.SetGlobalBuffer((__gm__ float*)y, batchSize * inChannels * outHeight * outWidth);

        // Allocate buffers for one row of output width
        uint32_t alignedWidth = ((outWidth + 7) / 8) * 8;
        this->alignedWidth = alignedWidth;

        pipe.InitBuffer(inQueue, BUFFER_NUM, alignedWidth * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, alignedWidth * sizeof(float));
        pipe.InitBuffer(tmpBuf, 1, alignedWidth * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t bc = bcStart; bc < bcEnd; bc++) {
            uint32_t b = bc / inChannels;
            uint32_t c = bc % inChannels;
            ProcessOneChannel(b, c);
        }
    }

private:
    __aicore__ inline void ProcessOneChannel(uint32_t b, uint32_t c)
    {
        uint32_t inOffset = (b * inChannels + c) * height * width;
        uint32_t outOffset = (b * inChannels + c) * outHeight * outWidth;

        for (uint32_t oh = 0; oh < outHeight; oh++) {
            // Compute one output row
            AscendC::LocalTensor<float> accLocal = outQueue.AllocTensor<float>();

            // Zero out accumulator
            AscendC::Duplicate(accLocal, (float)0.0f, this->alignedWidth);

            for (uint32_t kh = 0; kh < kernelSize; kh++) {
                int32_t ih = (int32_t)(oh * stride) - (int32_t)padding + (int32_t)(kh * dilation);
                if (ih < 0 || ih >= (int32_t)height) continue;

                float wVal = 0.0f;
                // Read weight value - weight shape is (inChannels, 1, kernelSize, 1)
                // weight[c][0][kh][0]
                uint32_t wIdx = c * kernelSize + kh;

                // We need to copy weight from GM. Use a small trick: copy one element
                AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

                // Copy the input row
                AscendC::LocalTensor<float> inLocal = inQueue.AllocTensor<float>();

                // For the width dimension: kernel width is 1, so output_w depends on padding
                // out_w = (in_w + 2*padding - 1) / stride + 1 when kernel_w=1, dilation doesn't matter for w
                // Each output pixel at (oh, ow): sum over kh of input[ih][ow*stride - padding] * weight[kh]
                // But kernel_w = 1, so for width: ow_start depends on padding
                // Actually for width dim with kernel=1: input_w_idx = ow * stride - padding
                // We need input row at ih, starting from appropriate column

                // Since kernel width is 1, each output (oh, ow) = sum_kh input[ih][iw] * w[kh]
                // where iw = ow * stride - padding (for width dim)
                // But we handle width similarly

                // For stride=1, padding=0 in width (kernel_w=1): iw = ow, outWidth = width
                // For general case with padding: iw = ow * stride - padding

                // Let's handle the general case by copying the relevant input data
                // We need to pick elements from input row ih at positions iw = ow*stride - padding for ow=0..outWidth-1

                // Simple approach: copy the whole input row, then handle indexing
                // Copy input row from GM
                uint32_t rowOffset = inOffset + ih * width;

                // If stride==1 and padding==0 for width (kernel_w=1), input and output widths match
                // Actually let's think: width dim kernel is 1, so:
                //   outWidth = (width + 2*padding - 1) / stride + 1 = (width + 2*padding) / stride  (when dilation doesn't matter for size-1 kernel)
                //   Wait, outWidth = (width + 2*padding - dilation*(1-1) - 1)/stride + 1 = (width + 2*padding - 1)/stride + 1

                // For the default test case: padding=0, stride=1, so outWidth = width

                // Copy input row
                uint32_t copyLen = ((width + 7) / 8) * 8;
                if (copyLen > this->alignedWidth) copyLen = this->alignedWidth;
                AscendC::DataCopy(inLocal, xGm[rowOffset], copyLen);
                AscendC::PipeBarrier<PIPE_ALL>();

                // Read weight
                AscendC::DataCopy(tmpLocal, wGm[wIdx], 8);
                AscendC::PipeBarrier<PIPE_ALL>();
                // tmpLocal[0] has the weight value

                // If stride==1 and padding==0:
                // output[ow] += input[ow] * weight[kh]
                // This is simply: accLocal += inLocal * wVal (element-wise scalar multiply)

                if (stride == 1 && padding == 0) {
                    // Multiply input row by weight scalar and accumulate
                    AscendC::Muls(inLocal, inLocal, tmpLocal.GetValue(0), this->alignedWidth);
                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::Add(accLocal, accLocal, inLocal, this->alignedWidth);
                    AscendC::PipeBarrier<PIPE_V>();
                } else {
                    // General case: need to handle stride and padding for width
                    // output[ow] += input[ow*stride - padding] * weight[kh]  (only if in bounds for width)
                    float w = tmpLocal.GetValue(0);
                    for (uint32_t ow = 0; ow < outWidth; ow++) {
                        int32_t iw = (int32_t)(ow * stride) - (int32_t)padding;
                        if (iw >= 0 && iw < (int32_t)width) {
                            float val = inLocal.GetValue(iw);
                            accLocal.SetValue(ow, accLocal.GetValue(ow) + val * w);
                        }
                    }
                }

                inQueue.FreeTensor(inLocal);
            }

            // Add bias if needed
            if (hasBias) {
                AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
                AscendC::DataCopy(tmpLocal, bGm[c], 8);
                AscendC::PipeBarrier<PIPE_ALL>();
                float biasVal = tmpLocal.GetValue(0);
                AscendC::Adds(accLocal, accLocal, biasVal, this->alignedWidth);
                AscendC::PipeBarrier<PIPE_V>();
            }

            // Copy output row to GM
            uint32_t outRowOffset = outOffset + oh * outWidth;
            uint32_t copyOutLen = ((outWidth + 7) / 8) * 8;
            AscendC::DataCopy(yGm[outRowOffset], accLocal, copyOutLen);
            AscendC::PipeBarrier<PIPE_ALL>();

            outQueue.FreeTensor(accLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, yGm;
    uint32_t batchSize, inChannels, height, width;
    uint32_t kernelSize, stride, padding, dilation;
    uint32_t hasBias, outHeight, outWidth, totalBC;
    uint32_t bcPerBlock, bcStart, bcEnd;
    uint32_t alignedWidth;
};

extern "C" __global__ __aicore__ void conv_depthwise_2d_square_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvDw op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels,
            tiling_data.height, tiling_data.width,
            tiling_data.kernelSize, tiling_data.stride,
            tiling_data.padding, tiling_data.dilation,
            tiling_data.hasBias, tiling_data.outHeight,
            tiling_data.outWidth, tiling_data.totalBC);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized depthwise 2D convolution with square input and asymmetric kernel using custom AscendC kernel.
    """
    def __init__(self, in_channels: int, kernel_size: int, stride: int = 1, padding: int = 0, dilation: int = 1, bias: bool = False):
        super().__init__()
        self.in_channels = in_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        self.dilation = dilation
        self.has_bias = bias
        
        # Initialize weight: shape (in_channels, 1, kernel_size, 1)
        self.weight = torch.nn.Parameter(torch.empty(in_channels, 1, kernel_size, 1))
        torch.nn.init.kaiming_uniform_(self.weight)
        
        if bias:
            self.bias = torch.nn.Parameter(torch.empty(in_channels))
            torch.nn.init.zeros_(self.bias)
        else:
            self.bias = None

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        batch_size = x.shape[0]
        height = x.shape[2]
        width = x.shape[3]
        
        # Reshape weight to (in_channels, kernel_size) for the kernel
        weight_flat = self.weight.reshape(self.in_channels, self.kernel_size).contiguous()
        
        return custom_ops_lib.conv_depthwise_2d_square_input_asymmetric_kernel_custom(
            x.contiguous(), weight_flat, self.bias,
            batch_size, self.in_channels, height, width,
            self.kernel_size, self.stride, self.padding, self.dilation,
            1 if self.has_bias else 0
        )
