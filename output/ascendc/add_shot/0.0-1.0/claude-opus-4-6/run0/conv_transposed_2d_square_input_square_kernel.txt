

Looking at this architecture, it performs a transposed 2D convolution. This is a complex operation that involves significant computation. I'll implement a custom AscendC kernel for transposed 2D convolution with square input and square kernel.

For a transposed 2D convolution, the approach is to use an im2col-like strategy but adapted for the transpose case (col2im). However, given the complexity, I'll implement this using a direct approach where each output element is computed by accumulating contributions from the input and kernel.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConvTranspose2d {
public:
    __aicore__ inline KernelConvTranspose2d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inHeight, uint32_t inWidth,
                                 uint32_t kernelSize, uint32_t stride, uint32_t padding,
                                 uint32_t outHeight, uint32_t outWidth,
                                 uint32_t groups, uint32_t outputPadding)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->groups = groups;

        uint32_t totalOutputElements = batchSize * outChannels * outHeight * outWidth;
        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->elementsPerBlock = totalOutputElements / numBlocks;
        this->startElement = blockIdx * this->elementsPerBlock;
        if (blockIdx == numBlocks - 1) {
            this->elementsPerBlock = totalOutputElements - this->startElement;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inChannels * inHeight * inWidth);
        wGm.SetGlobalBuffer((__gm__ float*)weight, inChannels * outChannels * kernelSize * kernelSize);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalOutputElements);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->elementsPerBlock; i++) {
            uint32_t globalIdx = this->startElement + i;

            uint32_t ow = globalIdx % outWidth;
            uint32_t tmp = globalIdx / outWidth;
            uint32_t oh = tmp % outHeight;
            tmp = tmp / outHeight;
            uint32_t oc = tmp % outChannels;
            uint32_t n = tmp / outChannels;

            float sum = 0.0f;

            for (uint32_t ic = 0; ic < inChannels; ic++) {
                for (uint32_t kh = 0; kh < kernelSize; kh++) {
                    for (uint32_t kw = 0; kw < kernelSize; kw++) {
                        int32_t ih_s = (int32_t)oh + (int32_t)padding - (int32_t)kh;
                        int32_t iw_s = (int32_t)ow + (int32_t)padding - (int32_t)kw;

                        if (ih_s % (int32_t)stride != 0) continue;
                        if (iw_s % (int32_t)stride != 0) continue;

                        int32_t ih = ih_s / (int32_t)stride;
                        int32_t iw = iw_s / (int32_t)stride;

                        if (ih >= 0 && ih < (int32_t)inHeight && iw >= 0 && iw < (int32_t)inWidth) {
                            uint32_t xIdx = n * inChannels * inHeight * inWidth +
                                             ic * inHeight * inWidth +
                                             (uint32_t)ih * inWidth + (uint32_t)iw;
                            uint32_t wIdx = ic * outChannels * kernelSize * kernelSize +
                                             oc * kernelSize * kernelSize +
                                             kh * kernelSize + kw;
                            float xVal = xGm.GetValue(xIdx);
                            float wVal = wGm.GetValue(wIdx);
                            sum += xVal * wVal;
                        }
                    }
                }
            }
            yGm.SetValue(globalIdx, sum);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inHeight;
    uint32_t inWidth;
    uint32_t kernelSize;
    uint32_t stride;
    uint32_t padding;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t groups;
    uint32_t elementsPerBlock;
    uint32_t startElement;
};

extern "C" __global__ __aicore__ void conv_transposed_2d_square_input_square_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose2d op;
    op.Init(x, weight, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inHeight, tiling_data.inWidth,
            tiling_data.kernelSize, tiling_data.stride, tiling_data.padding,
            tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.groups, tiling_data.outputPadding);
    op.Process();
}
