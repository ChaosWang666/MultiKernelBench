        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLength * sizeof(float));

            for (uint32_t i = 0; i < alignedHalf; i += 8) {
                if (i + half < len) {

                }

        AscendC::ReduceSum(meanLocal, accumLocal, accumLocal, 8);
