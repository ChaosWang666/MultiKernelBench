

Looking at this architecture, I need to analyze what operations are performed:
1. 3D Convolution
2. Softmax along channel dimension
3. MaxPool3d
4. MaxPool3d

The convolution is complex with learned weights, so I'll keep it as a PyTorch operation. I'll create a custom fused kernel that combines softmax + two max pooling operations into a single kernel for better performance.

However, given the complexity of fusing softmax with 3D max pooling in a single AscendC kernel, and the constraints of the AscendC programming model, I'll create a custom kernel that fuses the two consecutive max pool operations into one (since they're the same pool_kernel_size=2, two consecutive MaxPool3d with kernel 2 is equivalent to one MaxPool3d with kernel 4), and keep softmax separate.

Actually, let me reconsider. The most practical approach given AscendC constraints is to implement a custom kernel for the softmax + double max pooling pipeline. But implementing 3D spatial operations in AscendC is extremely complex. Let me focus on creating a workable fused kernel.

Given the complexity, I'll implement a custom element-wise softmax kernel along channel dimension fused with the pooling, but realistically, the best approach for correctness is to implement the full pipeline using host-side orchestration with a custom kernel for the compute-intensive parts.

Let me implement a practical solution - a custom kernel for the double max pooling (fusing two MaxPool3d(2) into one MaxPool3d(4)):

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv3dSoftmaxMaxPoolMaxPool {
public:
    __aicore__ inline KernelConv3dSoftmaxMaxPoolMaxPool() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t totalLength, uint32_t tileNum,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t depth, uint32_t height, uint32_t width,
                                 uint32_t outDepth, uint32_t outHeight, uint32_t outWidth)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->depth = depth;
        this->height = height;
        this->width = width;
        this->outDepth = outDepth;
        this->outHeight = outHeight;
        this->outWidth = outWidth;

        uint32_t totalOutElements = batchSize * channels * outDepth * outHeight * outWidth;

        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        // Each block processes a portion of the output spatial locations across all batch*channel
        this->totalOutSpatial = outDepth * outHeight * outWidth;
        uint32_t totalWork = batchSize * channels * totalOutSpatial;

        this->workPerBlock = totalWork / numBlocks;
        this->workStart = blockIdx * this->workPerBlock;
        if (blockIdx == numBlocks - 1) {
            this->workPerBlock = totalWork - this->workStart;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x, totalLength);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalOutElements);

        // We'll process element by element for the fused double max pool
        // Allocate small buffers for tile processing
        uint32_t bufSize = 64 * sizeof(float); // small buffer for intermediate
        pipe.InitBuffer(inQueueX, BUFFER_NUM, bufSize);
        pipe.InitBuffer(outQueueY, BUFFER_NUM, bufSize);
    }

    __aicore__ inline void Process()
    {
        uint32_t inSpatial = depth * height * width;

        for (uint32_t idx = 0; idx < this->workPerBlock; idx++) {
            uint32_t globalIdx = this->workStart + idx;

            uint32_t bc = globalIdx / totalOutSpatial;
            uint32_t spatialIdx = globalIdx % totalOutSpatial;

            uint32_t b = bc / channels;
            uint32_t c = bc % channels;

            uint32_t od = spatialIdx / (outHeight * outWidth);
            uint32_t rem = spatialIdx % (outHeight * outWidth);
            uint32_t oh = rem / outWidth;
            uint32_t ow = rem % outWidth;

            // The fused double maxpool with kernel_size=2 each means
            // effective pool kernel = 4, stride = 4
            uint32_t dStart = od * 4;
            uint32_t hStart = oh * 4;
            uint32_t wStart = ow * 4;

            float maxVal = -3.402823e+38f;

            // Find max over 4x4x4 region in input
            for (uint32_t dd = 0; dd < 4 && (dStart + dd) < depth; dd++) {
                for (uint32_t hh = 0; hh < 4 && (hStart + hh) < height; hh++) {
                    for (uint32_t ww = 0; ww < 4 && (wStart + ww) < width; ww++) {
                        uint32_t inIdx = ((b * channels + c) * depth + (dStart + dd)) * height * width +
                                         (hStart + hh) * width + (wStart + ww);
                        float val = *(((__gm__ float*)xGm.GetPhyAddr()) + inIdx);
                        if (val > maxVal) {
                            maxVal = val;
                        }
                    }
                }
            }

            uint32_t outIdx = ((b * channels + c) * outDepth + od) * outHeight * outWidth +
                              oh * outWidth + ow;
            *((__gm__ float*)yGm.GetPhyAddr() + outIdx) = maxVal;
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t batchSize, channels, depth, height, width;
    uint32_t outDepth, outHeight, outWidth;
    uint32_t totalOutSpatial;
    uint32_t workPerBlock;
    uint32_t workStart;
};

extern "C" __global__ __aicore__ void conv3d_softmax_max_pool_max_pool_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dSoftmaxMaxPoolMaxPool op;
    op.Init(x, y,
            tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.depth, tiling_data.height, tiling_data.width,
            tiling_data.outDepth, tiling_data.outHeight, tiling_data.outWidth);
    op.Process();
}

class ModelNew(torch.nn.Module):
