kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv2dGeluGlobalAvgPool {
public:
    __aicore__ inline KernelConv2dGeluGlobalAvgPool() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t height, uint32_t width, uint32_t kernelSize,
                                 uint32_t outHeight, uint32_t outWidth, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->height = height;
        this->width = width;
        this->kernelSize = kernelSize;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->tileNum = tileNum;
        
        // Total work items: batchSize * outChannels
        uint32_t totalWork = batchSize * outChannels;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        this->workPerBlock = (totalWork + blockNum - 1) / blockNum;
        this->workStart = blockIdx * this->workPerBlock;
        this->workEnd = this->workStart + this->workPerBlock;
        if (this->workEnd > totalWork) {
            this->workEnd = totalWork;
        }
        
        this->spatialSize = outHeight * outWidth;
        // Align spatial size to 32 bytes (8 floats)
        this->alignedSpatialSize = (this->spatialSize + 7) / 8 * 8;
        
        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * height * width);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, outChannels * inChannels * kernelSize * kernelSize);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize * outChannels);
        
        // We need buffers for intermediate computation
        // accBuf: accumulator for conv output spatial map
        // tmpBuf: temporary for loading input patches and kernel weights
        pipe.InitBuffer(accBuf, BUFFER_NUM, this->alignedSpatialSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, BUFFER_NUM, this->alignedSpatialSize * sizeof(float));
        pipe.InitBuffer(sumBuf, BUFFER_NUM, 8 * sizeof(float));  // for reduction
    }
    
    __aicore__ inline void Process()
    {
        for (uint32_t workIdx = this->workStart; workIdx < this->workEnd; workIdx++) {
            uint32_t b = workIdx / this->outChannels;
            uint32_t oc = workIdx % this->outChannels;
            ComputeOne(b, oc);
        }
    }

private:
    __aicore__ inline void ComputeOne(uint32_t b, uint32_t oc)
    {
        // Compute convolution for one (batch, out_channel) pair
        // Result is a spatial map of size outHeight * outWidth
        // Then apply GELU and global average pooling
        
        AscendC::LocalTensor<float> accLocal = accBuf.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.AllocTensor<float>();
        
        // Initialize accumulator with bias
        float biasVal = biasGm.GetValue(oc);
        AscendC::Duplicate(accLocal, biasVal, this->alignedSpatialSize);
        
        // For each input channel and kernel position, accumulate
        for (uint32_t ic = 0; ic < this->inChannels; ic++) {
            for (uint32_t kh = 0; kh < this->kernelSize; kh++) {
                for (uint32_t kw = 0; kw < this->kernelSize; kw++) {
                    float wVal = weightGm.GetValue(oc * inChannels * kernelSize * kernelSize + 
                                                    ic * kernelSize * kernelSize + 
                                                    kh * kernelSize + kw);
                    
                    // Gather input values for all spatial positions
                    // input[b, ic, oh+kh, ow+kw] for all (oh, ow)
                    for (uint32_t oh = 0; oh < this->outHeight; oh++) {
                        for (uint32_t ow = 0; ow < this->outWidth; ow++) {
                            uint32_t ih = oh + kh;
                            uint32_t iw = ow + kw;
                            float inVal = xGm.GetValue(b * inChannels * height * width + 
                                                        ic * height * width + 
                                                        ih * width + iw);
                            uint32_t spatialIdx = oh * outWidth + ow;
                            float curVal = accLocal.GetValue(spatialIdx);
                            accLocal.SetValue(spatialIdx, curVal + inVal * wVal);
                        }
                    }
                }
            }
        }
        
        // Apply GELU: x * 0.5 * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
        // Approximate with: x * 0.5 * (1 + erf(x / sqrt(2)))
        // We'll use the simple approximation available
        // For AscendC we can compute element-wise
        float sum = 0.0f;
        float invSpatial = 1.0f / (float)(this->spatialSize);
        for (uint32_t i = 0; i < this->spatialSize; i++) {
            float val = accLocal.GetValue(i);
            // GELU approximation: 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715*x^3)))
            float x3 = val * val * val;
            float inner = 0.7978845608f * (val + 0.044715f * x3);
            // tanh approximation
            float expPos = 1.0f;
            float expNeg = 1.0f;
            float absInner = inner > 0 ? inner : -inner;
            // Simple tanh: (exp(2x)-1)/(exp(2x)+1), use approximation
            float e2x;
            float twoInner = 2.0f * absInner;
            // Rough exp approximation for tanh computation
            // Use the identity tanh(x) = 1 - 2/(exp(2x)+1)
            if (twoInner > 8.0f) {
                e2x = 1.0f; // tanh saturates
            } else {
                // exp approximation: 1 + x + x^2/2 + x^3/6 + x^4/24 + x^5/120
                float t = twoInner;
                float t2 = t * t;
                float t3 = t2 * t;
                float t4 = t2 * t2;
                float t5 = t4 * t;
                float t6 = t3 * t3;
                float expVal = 1.0f + t + t2 * 0.5f + t3 * 0.166666667f + t4 * 0.041666667f + t5 * 0.008333333f + t6 * 0.001388889f;
                e2x = 1.0f - 2.0f / (expVal + 1.0f);
            }
            float tanhVal = inner > 0 ? e2x : -e2x;
            float gelu = 0.5f * val * (1.0f + tanhVal);
            sum += gelu;
        }
        
        float avgVal = sum * invSpatial;
        yGm.SetValue(b * outChannels + oc, avgVal);
        
        accBuf.FreeTensor(accLocal);
        tmpBuf.FreeTensor(tmpLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECIN> accBuf, tmpBuf, sumBuf;
    AscendC::GlobalTensor<float> xGm, weightGm, biasGm, yGm;
    uint32_t batchSize, inChannels, outChannels, height, width;
    uint32_t kernelSize, outHeight, outWidth;
    uint32_t spatialSize, alignedSpatialSize;
    uint32_t tileNum;
    uint32_t workPerBlock, workStart, workEnd;
};

extern "C" __global__ __aicore__ void conv2d_gelu_global_avg_pool_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dGeluGlobalAvgPool op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.height, tiling_data.width, tiling_data.kernelSize,
            tiling_data.outHeight, tiling_data.outWidth, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size):
        super().__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size

    def forward(self, x):
        batch_size = x.shape[0]
        height = x.shape[2]
        width = x.shape[3]
        weight = self.conv.weight
        bias = self.conv.bias
        return custom_ops_lib.conv2d_gelu_global_avg_pool_custom(
            x, weight, bias,
            batch_size, self.in_channels, self.out_channels,
            height, width, self.kernel_size
        )
