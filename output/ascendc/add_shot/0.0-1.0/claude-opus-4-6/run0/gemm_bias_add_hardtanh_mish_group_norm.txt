

Looking at the architecture, I need to fuse the sequence: GEMM → BiasAdd → Hardtanh → Mish → GroupNorm into a custom AscendC kernel. However, GEMM is a compute-intensive matrix multiplication that's best left to optimized libraries. I'll fuse the post-GEMM operations (BiasAdd + Hardtanh + Mish + GroupNorm) into a single custom kernel, keeping the GEMM as `nn.Linear`.

The fused kernel will:
1. Add bias
2. Apply Hardtanh (clamp to [-1, 1])
3. Apply Mish (x * tanh(softplus(x)))
4. Apply GroupNorm

        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, outFeatures);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z, batchSize * outFeatures);

        // Align tile to 32 bytes = 8 floats
        uint32_t tileLen = channelsPerGroup;
        // Round up to multiple of 8
        uint32_t tileLenAligned = (tileLen + 7) / 8 * 8;
        this->tileLenAligned = tileLenAligned;
        this->tileLen = tileLen;

        // We need: xLocal, biasLocal, gammaLocal, betaLocal, tmpLocal, outLocal
        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLenAligned * sizeof(float));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, tileLenAligned * sizeof(float));
        pipe.InitBuffer(inQueueGamma, BUFFER_NUM, tileLenAligned * sizeof(float));
        pipe.InitBuffer(inQueueBeta, BUFFER_NUM, tileLenAligned * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, tileLenAligned * sizeof(float));
        pipe.InitBuffer(tmpBuf1, BUFFER_NUM, tileLenAligned * sizeof(float));

            for (uint32_t g = 0; g < this->numGroups; g++) {
                ProcessOneGroup(row, g);
            }
        }
    }

        uint32_t lenAligned = this->tileLenAligned;

        AscendC::DataCopy(xLocal, xGm[offset], lenAligned);
        inQueueX.EnQue(xLocal);
        
        // Load bias tile

        AscendC::DataCopy(biasLocal, biasGm[biasOffset], lenAligned);
        inQueueBias.EnQue(biasLocal);
        
        // Load gamma tile
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.AllocTensor<float>();
        AscendC::DataCopy(gammaLocal, gammaGm[biasOffset], lenAligned);
        inQueueGamma.EnQue(gammaLocal);
        
        // Load beta tile
        AscendC::LocalTensor<float> betaLocal = inQueueBeta.AllocTensor<float>();
        AscendC::DataCopy(betaLocal, betaGm[biasOffset], lenAligned);
        inQueueBeta.EnQue(betaLocal);

        biasLocal = inQueueBias.DeQue<float>();
        gammaLocal = inQueueGamma.DeQue<float>();
        betaLocal = inQueueBeta.DeQue<float>();
