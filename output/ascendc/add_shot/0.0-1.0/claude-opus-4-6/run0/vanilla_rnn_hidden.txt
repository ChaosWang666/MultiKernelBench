

Looking at the architecture, the core computation in the RNN loop is:
1. `torch.cat((x[t], hidden), dim=1)` - concatenate input and hidden
2. `self.tanh(self.i2h(combined))` - linear transform + tanh for hidden state update
3. `self.h2o(hidden)` - linear transform for output

The key bottleneck is the repeated hidden state update: `hidden = tanh(combined @ W_i2h.T + b_i2h)`. I'll create a custom kernel `vanilla_rnn_hidden_custom` that fuses the tanh activation with the bias addition after the matrix multiplication, since the matmul itself is best left to optimized BLAS libraries. The custom kernel will compute `z = tanh(x + bias)` element-wise where x is the result of matmul and bias is broadcast along dim=0.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelVanillaRnnHidden {
public:
    __aicore__ inline KernelVanillaRnnHidden() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z, 
                                 uint32_t totalLength, uint32_t tileNum,
                                 uint32_t batchSize, uint32_t hiddenSize)
    {
        this->batchSize = batchSize;
        this->hiddenSize = hiddenSize;
        this->totalLength = totalLength;
        
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        uint32_t blockOffset = this->blockLength * AscendC::GetBlockIdx();
        xGm.SetGlobalBuffer((__gm__ float *)x + blockOffset, this->blockLength);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, hiddenSize);
        zGm.SetGlobalBuffer((__gm__ float *)z + blockOffset, this->blockLength);
        
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
        
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        
        // We need to load the appropriate bias values
        // x is (batchSize, hiddenSize), bias is (hiddenSize,)
        // We need to broadcast bias across batch dimension
        // The global offset for this block is blockLength * blockIdx
        // Within the tile, we need to figure out which bias elements to load
        uint32_t blockOffset = this->blockLength * AscendC::GetBlockIdx();
        uint32_t globalStart = blockOffset + progress * this->tileLength;
        uint32_t biasStart = globalStart % this->hiddenSize;
        
        // If tileLength <= hiddenSize and aligned, we can do a simple copy
        // For general case, we load bias segment
        AscendC::DataCopy(biasLocal, biasGm[biasStart], this->tileLength);
        
        inQueueX.EnQue(xLocal);
        inQueueBias.EnQue(biasLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        
        // z = tanh(x + bias)
        AscendC::Add(zLocal, xLocal, biasLocal, this->tileLength);
        // Apply tanh activation using Tanh intrinsic
        AscendC::Tanh(zLocal, zLocal, this->tileLength);
        
        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueBias.FreeTensor(biasLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t batchSize;
    uint32_t hiddenSize;
    uint32_t totalLength;
};

extern "C" __global__ __aicore__ void vanilla_rnn_hidden_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelVanillaRnnHidden op;
    op.Init(x, bias, z, tiling_data.totalLength, tiling_data.tileNum, tiling_data.batchSize, tiling_data.hiddenSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, input_size: int, hidden_size: int, output_size: int):
        """
        Initialize the Vanilla RNN model with fused tanh+bias kernel.
        """
        super(ModelNew, self).__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.output_size = output_size

        self.i2h = torch.nn.Linear(input_size + hidden_size, hidden_size)
        self.h2o = torch.nn.Linear(hidden_size, output_size)

    def forward(self, x: torch.Tensor, h0: torch.Tensor) -> torch.Tensor:
        seq_len, batch_size, _ = x.size()
        hidden = h0.to(x.device)
        outputs = []

        i2h_weight = self.i2h.weight
        i2h_bias = self.i2h.bias

        for t in range(seq_len):
            combined = torch.cat((x[t], hidden), dim=1)
            # Compute linear without bias: combined @ W^T
            linear_out = torch.mm(combined, i2h_weight.t())
            # Fused bias add + tanh using custom kernel
            hidden = custom_ops_lib.vanilla_rnn_hidden_custom(linear_out, i2h_bias)
            output = self.h2o(hidden)
            outputs.append(output)

        return torch.stack(outputs, dim=0)
'''
