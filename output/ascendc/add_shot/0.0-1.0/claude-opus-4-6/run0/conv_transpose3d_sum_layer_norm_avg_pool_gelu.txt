

Looking at this architecture, I need to analyze what operations are performed:
1. 3D transposed convolution
2. Add a scalar weight
3. Layer normalization
4. 3D average pooling
5. GELU activation

The transposed convolution and layer norm are complex operations best left to the framework. I'll fuse the simpler element-wise operations (sum + GELU) and the avg_pool into a custom kernel where practical. However, given the complexity, I'll focus on fusing the post-layernorm operations: average pooling followed by GELU into a single custom kernel to reduce memory round-trips.

Actually, let me reconsider. The most practical fusion is the **sum + layer_norm + avg_pool + GELU** after the conv_transpose. But layer norm and avg_pool are complex. A more realistic approach: fuse **add_scalar + GELU** or create a kernel for the entire post-conv pipeline.

Given kernel complexity constraints, I'll create a custom kernel that fuses **element-wise add + GELU** (applied after layer norm and avg pool), but looking at the order: sum comes before norm/pool. Let me fuse the **avg_pool + GELU** since they're the last two operations - avg_pool on 3D followed by GELU element-wise.

After more careful analysis, I'll create a fused kernel for the **add (sum_weight) + GELU** at the end, restructuring the model to do: conv_transpose → norm → avg_pool → (add + gelu). Wait, that changes semantics.

Let me just fuse the last operation GELU with avg_pool output, keeping the architecture semantics. Actually, the simplest and most impactful fusion: **element-wise add scalar + GELU** - but they're not adjacent.

I'll create a custom kernel for the **GELU** operation applied element-wise on the final tensor, which avoids extra memory reads/writes.

#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelAddGelu {
public:
    __aicore__ inline KernelAddGelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR sum_weight, GM_ADDR z, uint32_t totalLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        swGm.SetGlobalBuffer((__gm__ float *)sum_weight, 1);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf1, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> t1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> t2 = tmpBuf2.Get<float>();

        // Add sum_weight scalar
        float sw = swGm.GetValue(0);
        AscendC::Adds(zLocal, xLocal, sw, this->tileLength);

        // GELU approximation: x * 0.5 * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
        // Use built-in: compute sigmoid-based GELU = x * sigmoid(1.702 * x) as fast approx
        // Or exact: 0.5 * x * (1 + erf(x / sqrt(2)))

        // Fast GELU: x * 0.5 * (1.0 + tanh(0.7978845608 * (x + 0.044715 * x^3)))
        // Step 1: t1 = x^2
        AscendC::Mul(t1, zLocal, zLocal, this->tileLength);
        // Step 2: t1 = x^3
        AscendC::Mul(t1, t1, zLocal, this->tileLength);
        // Step 3: t1 = 0.044715 * x^3
        AscendC::Muls(t1, t1, (float)0.044715f, this->tileLength);
        // Step 4: t1 = x + 0.044715 * x^3
        AscendC::Add(t1, zLocal, t1, this->tileLength);
        // Step 5: t1 = 0.7978845608 * (x + 0.044715 * x^3)
        AscendC::Muls(t1, t1, (float)0.7978845608f, this->tileLength);
        // Step 6: t1 = tanh(...)
        // Use exp-based tanh: tanh(a) = (exp(2a)-1)/(exp(2a)+1)
        AscendC::Muls(t2, t1, (float)2.0f, this->tileLength);
        AscendC::Exp(t2, t2, this->tileLength);
        AscendC::Adds(t1, t2, (float)1.0f, this->tileLength);  // t1 = exp(2a)+1
        AscendC::Adds(t2, t2, (float)-1.0f, this->tileLength); // t2 = exp(2a)-1
        AscendC::Div(t1, t2, t1, this->tileLength);             // t1 = tanh
        // Step 7: t1 = 1 + tanh(...)
        AscendC::Adds(t1, t1, (float)1.0f, this->tileLength);
        // Step 8: t1 = 0.5 * (1 + tanh(...))
        AscendC::Muls(t1, t1, (float)0.5f, this->tileLength);
        // Step 9: result = x * 0.5 * (1 + tanh(...))
        AscendC::Mul(zLocal, zLocal, t1, this->tileLength);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> swGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv_transpose3d_sum_layer_norm_avg_pool_gelu_custom(GM_ADDR x, GM_ADDR sum_weight, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelAddGelu op;
    op.Init(x, sum_weight, z, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}

import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    """
    Optimized model that performs 3D transposed convolution, fused sum+GELU with custom kernel,
    layer normalization, and average pooling.
    The custom kernel fuses the element-wise add (sum_weight) and GELU activation.
    We reorder: conv_transpose -> layer_norm -> avg_pool -> (add + gelu) fused kernel.
    Note: Since sum_weight is a scalar added element-wise, and layer_norm normalizes per-channel,
    we keep the original order but fuse add+gelu: conv_transpose -> add_sum+gelu is not adjacent.
    Instead we fuse the last two: after avg_pool, apply GELU via custom kernel.
    Actually, to maintain correctness, we do conv_transpose -> then custom_kernel(add+gelu) around norm+pool.
    
    Correct approach: conv_transpose -> custom(add+gelu) wrapping the sum and gelu, with norm and pool in between done by PyTorch.
    We apply: conv_transpose -> (x + sum_weight) via custom -> norm -> avg_pool -> gelu via custom.
    
    Simplest correct fusion: replace (x + sum_weight) and gelu with custom kernel calls.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, sum_weight, norm_shape, pool_kernel_size):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.sum_weight = nn.Parameter(torch.tensor(sum_weight))
        self.norm = nn.LayerNorm(norm_shape)
        self.avg_pool = nn.AvgPool3d(kernel_size=pool_kernel_size)

    def forward(self, x):
        x = self.conv_transpose(x)
        # Fused add scalar + GELU using custom kernel
        # But we need norm and avg_pool in between. So we do add first, then norm, pool, then gelu.
        # Use custom kernel for add:
        x = x + self.sum_weight  # simple add, keep as-is since custom kernel does add+gelu together
        x = self.norm(x)
        x = self.avg_pool(x)
        # Apply fused add(0) + GELU via custom kernel (sum_weight=0 just for GELU)
        zero_weight = torch.zeros(1, dtype=x.dtype, device=x.device)
        x = custom_ops_lib.conv_transpose3d_sum_layer_norm_avg_pool_gelu_custom(x, zero_weight)
        return x
