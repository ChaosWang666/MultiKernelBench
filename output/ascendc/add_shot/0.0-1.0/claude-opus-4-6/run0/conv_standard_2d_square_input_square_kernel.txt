kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv2d {
public:
    __aicore__ inline KernelConv2d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t N, uint32_t Ci, uint32_t Hi, uint32_t Wi,
                                 uint32_t Co, uint32_t K, uint32_t stride, uint32_t padding,
                                 uint32_t Ho, uint32_t Wo, uint32_t totalOutput)
    {
        this->N = N;
        this->Ci = Ci;
        this->Hi = Hi;
        this->Wi = Wi;
        this->Co = Co;
        this->K = K;
        this->stride = stride;
        this->padding = padding;
        this->Ho = Ho;
        this->Wo = Wo;
        this->totalOutput = totalOutput;

        // Each block processes a portion of the output elements
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        this->outputPerBlock = (totalOutput + blockNum - 1) / blockNum;
        this->startIdx = blockIdx * this->outputPerBlock;
        this->endIdx = startIdx + outputPerBlock;
        if (this->endIdx > totalOutput) this->endIdx = totalOutput;
        
        xGm.SetGlobalBuffer((__gm__ float *)x, N * Ci * Hi * Wi);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, Co * Ci * K * K);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, Co);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalOutput);
        
        // Allocate buffer for partial sums - size for K*K elements aligned to 32 bytes
        uint32_t patchSize = K * K;
        uint32_t alignedPatchSize = ((patchSize + 7) / 8) * 8; // align to 8 floats (32 bytes)
        this->alignedPatchSize = alignedPatchSize;
        
        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedPatchSize * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, alignedPatchSize * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, alignedPatchSize * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        for (uint32_t idx = this->startIdx; idx < this->endIdx; idx++) {
            ComputeOneOutput(idx);
        }
    }

private:
    __aicore__ inline void ComputeOneOutput(uint32_t flatIdx)
    {
        // flatIdx = n * Co * Ho * Wo + co * Ho * Wo + ho * Wo + wo
        uint32_t HoWo = Ho * Wo;
        uint32_t CoHoWo = Co * HoWo;
        
        uint32_t n = flatIdx / CoHoWo;
        uint32_t rem = flatIdx % CoHoWo;
        uint32_t co = rem / HoWo;
        rem = rem % HoWo;
        uint32_t ho = rem / Wo;
        uint32_t wo = rem % Wo;
        
        float result = 0.0f;
        
        // Compute convolution for this output position
        for (uint32_t ci = 0; ci < Ci; ci++) {
            // Load input patch and weight patch, compute dot product
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
            
            // Fill with zeros first
            AscendC::Duplicate(xLocal, 0.0f, this->alignedPatchSize);
            
            // Load input patch
            for (uint32_t kh = 0; kh < K; kh++) {
                for (uint32_t kw = 0; kw < K; kw++) {
                    int32_t ih = (int32_t)(ho * stride) - (int32_t)padding + (int32_t)kh;
                    int32_t iw = (int32_t)(wo * stride) - (int32_t)padding + (int32_t)kw;
                    float val = 0.0f;
                    if (ih >= 0 && ih < (int32_t)Hi && iw >= 0 && iw < (int32_t)Wi) {
                        uint32_t inputIdx = n * Ci * Hi * Wi + ci * Hi * Wi + (uint32_t)ih * Wi + (uint32_t)iw;
                        val = xGm.GetValue(inputIdx);
                    }
                    xLocal.SetValue(kh * K + kw, val);
                }
            }
            
            // Load weight patch
            AscendC::Duplicate(wLocal, 0.0f, this->alignedPatchSize);
            for (uint32_t kh = 0; kh < K; kh++) {
                for (uint32_t kw = 0; kw < K; kw++) {
                    uint32_t wIdx = co * Ci * K * K + ci * K * K + kh * K + kw;
                    wLocal.SetValue(kh * K + kw, weightGm.GetValue(wIdx));
                }
            }
            
            inQueueX.EnQue(xLocal);
            inQueueW.EnQue(wLocal);
            
            // Compute element-wise multiply and sum
            AscendC::LocalTensor<float> xL = inQueueX.DeQue<float>();
            AscendC::LocalTensor<float> wL = inQueueW.DeQue<float>();
            AscendC::LocalTensor<float> oL = outQueue.AllocTensor<float>();
            
            AscendC::Mul(oL, xL, wL, this->alignedPatchSize);
            
            // Sum the elements
            for (uint32_t i = 0; i < K * K; i++) {
                result += oL.GetValue(i);
            }
            
            outQueue.EnQue(oL);
            AscendC::LocalTensor<float> oFree = outQueue.DeQue<float>();
            outQueue.FreeTensor(oFree);
            inQueueX.FreeTensor(xL);
            inQueueW.FreeTensor(wL);
        }
        
        // Add bias
        result += biasGm.GetValue(co);
        
        yGm.SetValue(flatIdx, result);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::GlobalTensor<float> xGm, weightGm, biasGm, yGm;
    uint32_t N, Ci, Hi, Wi, Co, K, stride, padding, Ho, Wo;
    uint32_t totalOutput, outputPerBlock, startIdx, endIdx;
    uint32_t alignedPatchSize;
};

extern "C" __global__ __aicore__ void conv_standard_2d_square_input_square_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2d op;
    op.Init(x, weight, bias, y,
            tiling_data.N, tiling_data.Ci, tiling_data.Hi, tiling_data.Wi,
            tiling_data.Co, tiling_data.K, tiling_data.stride, tiling_data.padding,
            tiling_data.Ho, tiling_data.Wo, tiling_data.totalOutput);
    op.Process();
}
