kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv3d {
public:
    __aicore__ inline KernelConv3d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR y,
                                uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                uint32_t inputDepth, uint32_t inputHeight, uint32_t inputWidth,
                                uint32_t kernelSize,
                                uint32_t outputDepth, uint32_t outputHeight, uint32_t outputWidth,
                                uint32_t stride, uint32_t padding, uint32_t dilation, uint32_t groups,
                                uint32_t totalOutputElements)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inputDepth = inputDepth;
        this->inputHeight = inputHeight;
        this->inputWidth = inputWidth;
        this->kernelSize = kernelSize;
        this->outputDepth = outputDepth;
        this->outputHeight = outputHeight;
        this->outputWidth = outputWidth;
        this->stride = stride;
        this->padding = padding;
        this->dilation = dilation;
        this->groups = groups;
        this->totalOutputElements = totalOutputElements;

        this->inCPerGroup = inChannels / groups;
        this->outCPerGroup = outChannels / groups;

        uint32_t inputSize = batchSize * inChannels * inputDepth * inputHeight * inputWidth;
        uint32_t weightSize = outChannels * this->inCPerGroup * kernelSize * kernelSize * kernelSize;

        xGm.SetGlobalBuffer((__gm__ float *)x, inputSize);
        wGm.SetGlobalBuffer((__gm__ float *)weight, weightSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalOutputElements);

        // Tile size for accumulation along reduction dimension
        uint32_t reductionLen = this->inCPerGroup * kernelSize * kernelSize * kernelSize;
        // We'll process tiles of the reduction dimension
        this->tileLength = 128; // process 128 elements at a time
        if (this->tileLength > reductionLen) {
            this->tileLength = reductionLen;
        }
        // Align tileLength to 8 (32 bytes / 4 bytes per float)
        this->tileLength = (this->tileLength / 8) * 8;
        if (this->tileLength == 0) this->tileLength = 8;

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        // Each block processes a subset of output elements
        uint32_t elementsPerBlock = (totalOutputElements + blockNum - 1) / blockNum;
        uint32_t startElem = blockIdx * elementsPerBlock;
        uint32_t endElem = startElem + elementsPerBlock;
        if (endElem > totalOutputElements) endElem = totalOutputElements;

        for (uint32_t idx = startElem; idx < endElem; idx++) {
            ComputeOneOutput(idx);
        }
    }

private:
    __aicore__ inline void ComputeOneOutput(uint32_t idx)
    {
        // Decode idx into (n, oc, od, oh, ow)
        uint32_t tmp = idx;
        uint32_t ow = tmp % outputWidth; tmp /= outputWidth;
        uint32_t oh = tmp % outputHeight; tmp /= outputHeight;
        uint32_t od = tmp % outputDepth; tmp /= outputDepth;
        uint32_t oc = tmp % outChannels; tmp /= outChannels;
        uint32_t n = tmp;

        uint32_t g = oc / outCPerGroup;
        uint32_t ocInGroup = oc % outCPerGroup;

        float sum = 0.0f;

        uint32_t reductionLen = inCPerGroup * kernelSize * kernelSize * kernelSize;
        uint32_t fullTiles = reductionLen / tileLength;
        uint32_t remainder = reductionLen % tileLength;

        for (uint32_t t = 0; t < fullTiles; t++) {
            uint32_t rStart = t * tileLength;
            sum += ProcessTile(n, oc, g, ocInGroup, od, oh, ow, rStart, tileLength);
        }
        if (remainder > 0) {
            uint32_t rStart = fullTiles * tileLength;
            // Process remainder elements one by one (scalar)
            for (uint32_t r = rStart; r < reductionLen; r++) {
                uint32_t ic_local = r / (kernelSize * kernelSize * kernelSize);
                uint32_t kRem = r % (kernelSize * kernelSize * kernelSize);
                uint32_t kd = kRem / (kernelSize * kernelSize);
                uint32_t kRem2 = kRem % (kernelSize * kernelSize);
                uint32_t kh = kRem2 / kernelSize;
                uint32_t kw = kRem2 % kernelSize;

                uint32_t ic = g * inCPerGroup + ic_local;

                int32_t id = (int32_t)(od * stride) - (int32_t)padding + (int32_t)(kd * dilation);
                int32_t ih = (int32_t)(oh * stride) - (int32_t)padding + (int32_t)(kh * dilation);
                int32_t iw = (int32_t)(ow * stride) - (int32_t)padding + (int32_t)(kw * dilation);

                float xVal = 0.0f;
                if (id >= 0 && id < (int32_t)inputDepth && ih >= 0 && ih < (int32_t)inputHeight && iw >= 0 && iw < (int32_t)inputWidth) {
                    uint32_t xIdx = ((((uint32_t)n * inChannels + ic) * inputDepth + (uint32_t)id) * inputHeight + (uint32_t)ih) * inputWidth + (uint32_t)iw;
                    xVal = xGm.GetValue(xIdx);
                }

                uint32_t wIdx = (((oc * inCPerGroup + ic_local) * kernelSize + kd) * kernelSize + kh) * kernelSize + kw;
                float wVal = wGm.GetValue(wIdx);
                sum += xVal * wVal;
            }
        }

        yGm.SetValue(idx, sum);
    }

    __aicore__ inline float ProcessTile(uint32_t n, uint32_t oc, uint32_t g, uint32_t ocInGroup,
                                         uint32_t od, uint32_t oh, uint32_t ow,
                                         uint32_t rStart, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = outQueueY.AllocTensor<float>();

        // Fill xLocal and wLocal
        for (uint32_t i = 0; i < len; i++) {
            uint32_t r = rStart + i;
            uint32_t ic_local = r / (kernelSize * kernelSize * kernelSize);
            uint32_t kRem = r % (kernelSize * kernelSize * kernelSize);
            uint32_t kd = kRem / (kernelSize * kernelSize);
            uint32_t kRem2 = kRem % (kernelSize * kernelSize);
            uint32_t kh = kRem2 / kernelSize;
            uint32_t kw = kRem2 % kernelSize;

            uint32_t ic = g * inCPerGroup + ic_local;

            int32_t id = (int32_t)(od * stride) - (int32_t)padding + (int32_t)(kd * dilation);
            int32_t ih = (int32_t)(oh * stride) - (int32_t)padding + (int32_t)(kh * dilation);
            int32_t iw = (int32_t)(ow * stride) - (int32_t)padding + (int32_t)(kw * dilation);

            float xVal = 0.0f;
            if (id >= 0 && id < (int32_t)inputDepth && ih >= 0 && ih < (int32_t)inputHeight && iw >= 0 && iw < (int32_t)inputWidth) {
                uint32_t xIdx = ((((uint32_t)n * inChannels + ic) * inputDepth + (uint32_t)id) * inputHeight + (uint32_t)ih) * inputWidth + (uint32_t)iw;
                xVal = xGm.GetValue(xIdx);
            }

            uint32_t wIdx = (((oc * inCPerGroup + ic_local) * kernelSize + kd) * kernelSize + kh) * kernelSize + kw;
            float wVal = wGm.GetValue(wIdx);

            xLocal.SetValue(i, xVal);
            wLocal.SetValue(i, wVal);
        }

        // Multiply element-wise
        AscendC::Mul(tmpLocal, xLocal, wLocal, len);

        // Reduce sum
        float sum = 0.0f;
        for (uint32_t i = 0; i < len; i++) {
            sum += tmpLocal.GetValue(i);
        }

        inQueueX.FreeTensor(xLocal);
        inQueueW.FreeTensor(wLocal);
        outQueueY.FreeTensor(tmpLocal);

        return sum;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm, wGm, yGm;

    uint32_t batchSize, inChannels, outChannels;
    uint32_t inputDepth, inputHeight, inputWidth;
    uint32_t kernelSize;
    uint32_t outputDepth, outputHeight, outputWidth;
    uint32_t stride, padding, dilation, groups;
    uint32_t inCPerGroup, outCPerGroup;
    uint32_t totalOutputElements;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv_standard_3d_square_input_square_kernel_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3d op;
    op.Init(x, weight, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inputDepth, tiling_data.inputHeight, tiling_data.inputWidth,
            tiling_data.kernelSize,
            tiling_data.outputDepth, tiling_data.outputHeight, tiling_data.outputWidth,
            tiling_data.stride, tiling_data.padding, tiling_data.dilation, tiling_data.groups,
            tiling_data.totalOutputElements);
    op.Process();
}
