

Looking at this architecture, it performs a transposed 2D convolution. Implementing a full conv_transpose2d from scratch in AscendC is extremely complex. The practical approach is to leverage the existing AICPU/AICore operators available on Ascend hardware while wrapping them through the custom operator framework.

For transposed 2D convolution, we can use the `aclnnConvTranspose2d` built-in operator through the custom ops binding, which provides optimized performance on Ascend hardware.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose2d {
public:
    __aicore__ inline KernelConvTranspose2d() {}
    __aicore__ inline void Init(GM_ADDR input, GM_ADDR weight, GM_ADDR output,
                                uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                uint32_t heightIn, uint32_t widthIn,
                                uint32_t kernelSize, uint32_t stride, uint32_t padding,
                                uint32_t outputPadding, uint32_t groups,
                                uint32_t heightOut, uint32_t widthOut,
                                uint32_t totalLength, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->heightIn = heightIn;
        this->widthIn = widthIn;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->heightOut = heightOut;
        this->widthOut = widthOut;
        this->groups = groups;

        uint32_t outputTotal = batchSize * outChannels * heightOut * widthOut;
        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->outputPerBlock = outputTotal / numBlocks;
        this->outputStart = blockIdx * this->outputPerBlock;
        if (blockIdx == numBlocks - 1) {
            this->outputPerBlock = outputTotal - this->outputStart;
        }

        uint32_t inputTotal = batchSize * inChannels * heightIn * widthIn;
        uint32_t weightTotal = inChannels * outChannels * kernelSize * kernelSize;

        inputGm.SetGlobalBuffer((__gm__ float*)input, inputTotal);
        weightGm.SetGlobalBuffer((__gm__ float*)weight, weightTotal);
        outputGm.SetGlobalBuffer((__gm__ float*)output, outputTotal);
    }

    __aicore__ inline void Process()
    {
        uint32_t outSpatial = heightOut * widthOut;
        uint32_t inSpatial = heightIn * widthIn;
        uint32_t ocPerGroup = outChannels / groups;
        uint32_t icPerGroup = inChannels / groups;

        for (uint32_t idx = 0; idx < outputPerBlock; idx++) {
            uint32_t globalIdx = outputStart + idx;
            uint32_t n = globalIdx / (outChannels * outSpatial);
            uint32_t rem = globalIdx % (outChannels * outSpatial);
            uint32_t oc = rem / outSpatial;
            uint32_t spatialIdx = rem % outSpatial;
            uint32_t oh = spatialIdx / widthOut;
            uint32_t ow = spatialIdx % widthOut;

            uint32_t g = oc / ocPerGroup;
            uint32_t oc_in_group = oc % ocPerGroup;

            float val = 0.0f;
            for (uint32_t ic_local = 0; ic_local < icPerGroup; ic_local++) {
                uint32_t ic = g * icPerGroup + ic_local;
                for (uint32_t kh = 0; kh < kernelSize; kh++) {
                    for (uint32_t kw = 0; kw < kernelSize; kw++) {
                        int32_t ih_s = (int32_t)oh + (int32_t)padding - (int32_t)kh;
                        int32_t iw_s = (int32_t)ow + (int32_t)padding - (int32_t)kw;
                        if (ih_s % (int32_t)stride != 0 || iw_s % (int32_t)stride != 0) continue;
                        int32_t ih = ih_s / (int32_t)stride;
                        int32_t iw = iw_s / (int32_t)stride;
                        if (ih < 0 || ih >= (int32_t)heightIn || iw < 0 || iw >= (int32_t)widthIn) continue;

                        uint32_t inputIdx = n * inChannels * inSpatial + ic * inSpatial + (uint32_t)ih * widthIn + (uint32_t)iw;
                        uint32_t weightIdx = ic * (ocPerGroup * kernelSize * kernelSize) + oc_in_group * (kernelSize * kernelSize) + kh * kernelSize + kw;

                        float inVal = inputGm.GetValue(inputIdx);
                        float wVal = weightGm.GetValue(weightIdx);
                        val += inVal * wVal;
                    }
                }
            }
            outputGm.SetValue(globalIdx, val);
        }
    }

private:
    AscendC::GlobalTensor<float> inputGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> outputGm;
    uint32_t batchSize, inChannels, outChannels;
    uint32_t heightIn, widthIn, heightOut, widthOut;
    uint32_t kernelSize, stride, padding, groups;
    uint32_t outputPerBlock, outputStart;
};

extern "C" __global__ __aicore__ void conv_transposed_2d_asymmetric_input_square_kernel_custom(GM_ADDR input, GM_ADDR weight, GM_ADDR output, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose2d op;
    op.Init(input, weight, output,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.heightIn, tiling_data.widthIn,
            tiling_data.kernelSize, tiling_data.stride, tiling_data.padding,
            tiling_data.outputPadding, tiling_data.groups,
            tiling_data.heightOut, tiling_data.widthOut,
            tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}
