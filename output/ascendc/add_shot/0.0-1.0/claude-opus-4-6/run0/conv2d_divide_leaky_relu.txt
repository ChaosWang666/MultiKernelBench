

Looking at the architecture, I need to fuse the division by a constant and LeakyReLU into a single custom AscendC kernel. The convolution will remain as the standard PyTorch operation since it's complex and already well-optimized. The custom kernel `conv2d_divide_leaky_relu_custom` will take the convolution output, divide by a constant, and apply LeakyReLU.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelDivideLeakyRelu {
public:
    __aicore__ inline KernelDivideLeakyRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalLength, uint32_t tileNum, float divisor, float negativeSlope)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->invDivisor = 1.0f / divisor;
        this->negativeSlope = negativeSlope;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpBuffer, this->tileLength * sizeof(DTYPE_X));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_X> tmpLocal = tmpBuffer.Get<DTYPE_X>();

        // Divide: x = x * (1/divisor)
        AscendC::Muls(zLocal, xLocal, this->invDivisor, this->tileLength);

        // LeakyReLU: max(x, 0) + negativeSlope * min(x, 0)
        // Compute negativeSlope * zLocal into tmpLocal
        AscendC::Muls(tmpLocal, zLocal, this->negativeSlope, this->tileLength);
        // max(zLocal, tmpLocal) gives LeakyReLU for both positive and negative
        // For positive values: zLocal > negativeSlope*zLocal (since negativeSlope=0.01 < 1), so max = zLocal (correct)
        // For negative values: zLocal < negativeSlope*zLocal (since negativeSlope*neg > neg), so max = negativeSlope*zLocal (correct)
        AscendC::Max(zLocal, zLocal, tmpLocal, this->tileLength);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuffer;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    float invDivisor;
    float negativeSlope;
};

extern "C" __global__ __aicore__ void conv2d_divide_leaky_relu_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelDivideLeakyRelu op;
    op.Init(x, z, tiling_data.totalLength, tiling_data.tileNum, tiling_data.divisor, tiling_data.negativeSlope);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model that performs convolution, then uses a fused custom kernel for divide + LeakyReLU.
    """
    def __init__(self, in_channels, out_channels, kernel_size, divisor):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.divisor = divisor

    def forward(self, x):
        x = self.conv(x)
        x = custom_ops_lib.conv2d_divide_leaky_relu_custom(x, self.divisor, 0.01)
        return x
'''
