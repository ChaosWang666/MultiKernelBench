

Looking at this architecture, I need to analyze the operations:
1. Linear (matmul + bias)
2. Scale by factor
3. Add residual (x + x, which is 2*x)
4. Clamp
5. LogSumExp (reduction along dim=1, keepdim=True)
6. Mish activation (x * tanh(softplus(x)))
7. Multiply x * mish(x)

The linear layer is best kept in PyTorch (highly optimized). I'll fuse the post-linear operations: scale, residual add (2x), clamp, logsumexp, and x*mish(x) into a custom kernel.

Actually, let me reconsider. The logsumexp is a reduction operation that reduces dim=1 from hidden_size to 1. This changes the shape. Let me split this into two kernels or handle it carefully.

Let me create a fused kernel that takes the linear output and performs: scale → double (residual add) → clamp → logsumexp → mish multiplication.

Given the complexity, I'll fuse the element-wise ops (scale, residual_add, clamp) into one kernel applied before logsumexp, and then apply logsumexp and mish using PyTorch. Actually, for best optimization on AscendC, let me fuse scale+residual+clamp as an element-wise kernel since they operate element-wise, and keep the reduction (logsumexp) and mish in PyTorch.

Wait - scaling by scale_factor then adding x+x is equivalent to x * scale_factor * 2. So the element-wise part is: x * (scale_factor * 2), then clamp. Let me fuse all post-linear element-wise operations.

#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulScaleResidualAddClampLogSumExpMish {
public:
    __aicore__ inline KernelMatmulScaleResidualAddClampLogSumExpMish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, 
                                 uint32_t totalLength, uint32_t tileNum,
                                 float scaleFactor, float clampMin, float clampMax,
                                 uint32_t hiddenSize, uint32_t batchSize)
    {
        this->hiddenSize = hiddenSize;
        this->batchSize = batchSize;
        this->scaleFactor = scaleFactor;
        this->clampMin = clampMin;
        this->clampMax = clampMax;
        this->totalLength = totalLength;
        
        // Each block processes a chunk of rows
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        
        // Distribute rows across blocks
        this->rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        if (this->startRow + this->rowsPerBlock > batchSize) {
            this->rowsPerBlock = (this->startRow < batchSize) ? (batchSize - this->startRow) : 0;
        }
        
        xGm.SetGlobalBuffer((__gm__ float *)x, totalLength);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize);
        
        // We process one row at a time, each row has hiddenSize elements
        // Tile each row into chunks
        this->tileNum = tileNum;
        this->tileLength = hiddenSize / tileNum;
        // Ensure tileLength is aligned to 8 (32 bytes / 4 bytes per float)
        if (this->tileLength < 8) this->tileLength = 8;
        this->tileNum = hiddenSize / this->tileLength;
        
        // Allocate buffers for processing tiles
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, 1, this->tileLength * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        for (uint32_t row = 0; row < this->rowsPerBlock; row++) {
            uint32_t actualRow = this->startRow + row;
            if (actualRow >= this->batchSize) break;
            ProcessRow(actualRow);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t row)
    {
        // For logsumexp: log(sum(exp(x_i)))
        // We use the numerically stable version: maxVal + log(sum(exp(x_i - maxVal)))
        // First pass: apply scale*2, clamp, find max
        // Second pass: compute sum of exp(x_i - max)
        // Then: result = maxVal + log(sumExp)
        // Then apply x * mish(x) where mish(x) = x * tanh(softplus(x))
        
        uint32_t rowOffset = row * this->hiddenSize;
        
        // First pass: find max after scale+residual+clamp
        float globalMax = -1e30f;
        
        for (uint32_t t = 0; t < this->tileNum; t++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[rowOffset + t * this->tileLength], this->tileLength);
            inQueueX.EnQue(xLocal);
            
            AscendC::LocalTensor<float> xProc = inQueueX.DeQue<float>();
            // scale * 2 (scale then add residual x+x = 2x)
            float combinedScale = this->scaleFactor * 2.0f;
            AscendC::Muls(xProc, xProc, combinedScale, this->tileLength);
            // clamp
            AscendC::Mins(xProc, xProc, this->clampMax, this->tileLength);
            AscendC::Maxs(xProc, xProc, this->clampMin, this->tileLength);
            
            // Find max in this tile
            float tileMax = -1e30f;
            AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
            AscendC::ReduceMax(tmpLocal, xProc, this->tileLength, false);
            tileMax = tmpLocal.GetValue(0);
            if (tileMax > globalMax) globalMax = tileMax;
            
            inQueueX.FreeTensor(xProc);
        }
        
        // Second pass: compute sum(exp(x_i - max))
        float sumExp = 0.0f;
        
        for (uint32_t t = 0; t < this->tileNum; t++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[rowOffset + t * this->tileLength], this->tileLength);
            inQueueX.EnQue(xLocal);
            
            AscendC::LocalTensor<float> xProc = inQueueX.DeQue<float>();
            // Reapply scale*2 and clamp
            float combinedScale = this->scaleFactor * 2.0f;
            AscendC::Muls(xProc, xProc, combinedScale, this->tileLength);
            AscendC::Mins(xProc, xProc, this->clampMax, this->tileLength);
            AscendC::Maxs(xProc, xProc, this->clampMin, this->tileLength);
            
            // Subtract max
            AscendC::Adds(xProc, xProc, -globalMax, this->tileLength);
            // Exp
            AscendC::Exp(xProc, xProc, this->tileLength);
            
            // Sum
            AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
            AscendC::ReduceSum(tmpLocal, xProc, this->tileLength, false);
            sumExp += tmpLocal.GetValue(0);
            
            inQueueX.FreeTensor(xProc);
        }
        
        // logsumexp = max + log(sumExp)
        float logsumexpVal = globalMax + logf(sumExp);
        
        // Apply x * mish(x) where mish(x) = x * tanh(softplus(x)) = x * tanh(log(1+exp(x)))
        // So result = logsumexpVal * (logsumexpVal * tanh(log(1 + exp(logsumexpVal))))
        float sp = logf(1.0f + expf(logsumexpVal));  // softplus
        float th = tanhf(sp);  // tanh(softplus(x))
        float mish = logsumexpVal * th;  // mish(x)
        float result = logsumexpVal * mish;  // x * mish(x)
        
        // Write output - single value per row
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        zLocal.SetValue(0, result);
        // Pad remaining to avoid issues
        for (uint32_t i = 1; i < 8; i++) {
            zLocal.SetValue(i, 0.0f);
        }
        outQueueZ.EnQue(zLocal);
        
        AscendC::LocalTensor<float> zOut = outQueueZ.DeQue<float>();
        // Write single float - but DataCopy needs aligned size, so we copy 8 minimum
        // We need to write just one float per row to yGm[row]
        AscendC::DataCopy(yGm[row], zOut, 8);
        outQueueZ.FreeTensor(zOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t hiddenSize;
    uint32_t batchSize;
    uint32_t totalLength;
    float scaleFactor;
    float clampMin;
    float clampMax;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t rowsPerBlock;
    uint32_t startRow;
};

extern "C" __global__ __aicore__ void matmul_scale_residual_add_clamp_log_sum_exp_mish_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulScaleResidualAddClampLogSumExpMish op;
    op.Init(x, y, 
            tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.scaleFactor, tiling_data.clampMin, tiling_data.clampMax,
            tiling_data.hiddenSize, tiling_data.batchSize);
    op.Process();
}
