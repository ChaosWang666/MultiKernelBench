

Looking at this architecture, I need to analyze what operations can be fused into a custom kernel. The model does:
1. 3D transposed convolution
2. Scaling by scale1
3. Average pooling (kernel_size=2)
4. Bias addition
5. Scaling by scale2

The transposed convolution is complex and best left to the built-in operator. However, steps 2-5 (scale, avg_pool, bias_add, scale) can be fused into a single custom kernel. After the conv_transpose3d, we have a 5D tensor. We can fuse: `((avg_pool(x * scale1)) + bias) * scale2`.

Since `x * scale1` followed by avg_pool is equivalent to `avg_pool(x) * scale1` (avg_pool is linear), we can simplify to `(avg_pool(x) * scale1 + bias) * scale2`.

Let me implement a custom kernel that takes the conv_transpose output and performs scaling, 3D average pooling with kernel_size=2, bias addition, and final scaling.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose3dScalingAvgPoolBiasAddScaling {
public:
    __aicore__ inline KernelConvTranspose3dScalingAvgPoolBiasAddScaling() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale1, GM_ADDR bias, GM_ADDR scale2, GM_ADDR y,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t inDepth, uint32_t inHeight, uint32_t inWidth,
                                 uint32_t outDepth, uint32_t outHeight, uint32_t outWidth,
                                 uint32_t totalOutput, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->inDepth = inDepth;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->outDepth = outDepth;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->totalOutput = totalOutput;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->blockLength = totalOutput / blockNum;
        uint32_t blockStart = blockIdx * this->blockLength;
        if (blockIdx == blockNum - 1) {
            this->blockLength = totalOutput - blockStart;
        }
        this->outputStart = blockStart;

        this->tileNum = tileNum;
        this->tileLength = this->blockLength / (tileNum * BUFFER_NUM);
        if (this->tileLength == 0) {
            this->tileLength = this->blockLength;
            this->tileNum = 1;
        }
        // Align tileLength to 8 (32 bytes / 4 bytes per float)
        uint32_t alignFactor = 8;
        this->tileLength = (this->tileLength / alignFactor) * alignFactor;
        if (this->tileLength == 0) {
            this->tileLength = alignFactor;
        }

        uint32_t inTotal = batchSize * channels * inDepth * inHeight * inWidth;
        xGm.SetGlobalBuffer((__gm__ float*)x, inTotal);
        scale1Gm.SetGlobalBuffer((__gm__ float*)scale1, 1);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, channels);
        scale2Gm.SetGlobalBuffer((__gm__ float*)scale2, 1);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalOutput);

        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Read scalars
        float s1 = scale1Gm.GetValue(0);
        float s2 = scale2Gm.GetValue(0);
        float combinedScale = s1 * s2 / 8.0f; // scale1 * scale2 / (2*2*2) for avg pooling

        uint32_t totalTiles = this->tileNum * BUFFER_NUM;
        uint32_t processed = 0;

        for (uint32_t t = 0; t < totalTiles && processed < this->blockLength; t++) {
            uint32_t curLen = this->tileLength;
            if (processed + curLen > this->blockLength) {
                curLen = this->blockLength - processed;
                curLen = (curLen + 7) / 8 * 8;
                if (curLen > this->tileLength) curLen = this->tileLength;
            }
            ComputeTile(processed, curLen, combinedScale, s2);
            processed += curLen;
        }
    }

private:
    __aicore__ inline void ComputeTile(uint32_t offset, uint32_t len, float combinedScale, float s2)
    {
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        uint32_t spatialOut = outDepth * outHeight * outWidth;
        uint32_t inSpatial = inDepth * inHeight * inWidth;
        uint32_t inHW = inHeight * inWidth;
        uint32_t outHW = outHeight * outWidth;

        for (uint32_t i = 0; i < len; i++) {
            uint32_t globalIdx = this->outputStart + offset + i;
            if (globalIdx >= this->totalOutput) {
                outLocal.SetValue(i, 0.0f);
                continue;
            }

            uint32_t n = globalIdx / (channels * spatialOut);
            uint32_t rem = globalIdx - n * (channels * spatialOut);
            uint32_t c = rem / spatialOut;
            uint32_t spatialIdx = rem - c * spatialOut;
            uint32_t od = spatialIdx / outHW;
            uint32_t rem2 = spatialIdx - od * outHW;
            uint32_t oh = rem2 / outWidth;
            uint32_t ow = rem2 - oh * outWidth;

            uint32_t id_start = od * 2;
            uint32_t ih_start = oh * 2;
            uint32_t iw_start = ow * 2;

            float sum = 0.0f;
            uint32_t baseOffset = n * channels * inSpatial + c * inSpatial;

            for (uint32_t dd = 0; dd < 2; dd++) {
                for (uint32_t dh = 0; dh < 2; dh++) {
                    for (uint32_t dw = 0; dw < 2; dw++) {
                        uint32_t id = id_start + dd;
                        uint32_t ih = ih_start + dh;
                        uint32_t iw = iw_start + dw;
                        uint32_t inIdx = baseOffset + id * inHW + ih * inWidth + iw;
                        sum += xGm.GetValue(inIdx);
                    }
                }
            }

            float biasVal = biasGm.GetValue(c);
            float result = sum * combinedScale + biasVal * s2;
            outLocal.SetValue(i, result);
        }

        outQueue.EnQue(outLocal);
        outLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(yGm[this->outputStart + offset], outLocal, len);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> scale1Gm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> scale2Gm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize, channels;
    uint32_t inDepth, inHeight, inWidth;
    uint32_t outDepth, outHeight, outWidth;
    uint32_t totalOutput;
    uint32_t blockLength;
    uint32_t outputStart;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv_transpose3d_scaling_avg_pool_bias_add_scaling_custom(
    GM_ADDR x, GM_ADDR scale1, GM_ADDR bias, GM_ADDR scale2, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dScalingAvgPoolBiasAddScaling op;
    op.Init(x, scale1, bias, scale2, y,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.inDepth, tiling_data.inHeight, tiling_data.inWidth,
            tiling_data.outDepth, tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.totalOutput, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    """
    Optimized model that performs 3D transposed convolution using PyTorch,
    then fuses scaling, average pooling, bias addition, and scaling into a custom AscendC kernel.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, scale1, scale2, bias_shape):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.scale1 = nn.Parameter(torch.tensor(scale1))
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.scale2 = nn.Parameter(torch.tensor(scale2))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = custom_ops_lib.conv_transpose3d_scaling_avg_pool_bias_add_scaling_custom(
            x, self.scale1, self.bias, self.scale2)
        return x
'''
