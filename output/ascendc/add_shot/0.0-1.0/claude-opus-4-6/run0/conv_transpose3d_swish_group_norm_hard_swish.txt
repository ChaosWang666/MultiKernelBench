kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelSwishGNHardSwish {
public:
    __aicore__ inline KernelSwishGNHardSwish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                 uint32_t batchSize, uint32_t numChannels, 
                                 uint32_t spatialSize, uint32_t numGroups, float eps)
    {
        this->batchSize = batchSize;
        this->numChannels = numChannels;
        this->spatialSize = spatialSize;
        this->numGroups = numGroups;
        this->eps = eps;
        this->channelsPerGroup = numChannels / numGroups;
        this->groupSpatialSize = this->channelsPerGroup * spatialSize;
        this->totalPerBatch = numChannels * spatialSize;
        
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        // Each block handles a portion of batch elements
        this->batchStart = blockIdx * ((batchSize + blockNum - 1) / blockNum);
        this->batchEnd = (blockIdx + 1) * ((batchSize + blockNum - 1) / blockNum);
        if (this->batchEnd > batchSize) this->batchEnd = batchSize;
        
        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * totalPerBatch);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, numChannels);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, numChannels);
        yGm.SetGlobalBuffer((__gm__ float*)y, batchSize * totalPerBatch);
        
        // Allocate buffers - we process one group's spatial data at a time
        // tileLength should be aligned to 32 bytes (8 floats)
        uint32_t alignedGroupSpatial = ((groupSpatialSize + 7) / 8) * 8;
        
        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedGroupSpatial * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, alignedGroupSpatial * sizeof(float));
        pipe.InitBuffer(tmpBuf1, BUFFER_NUM, alignedGroupSpatial * sizeof(float));
        pipe.InitBuffer(tmpBuf2, BUFFER_NUM, alignedGroupSpatial * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        for (uint32_t b = batchStart; b < batchEnd; b++) {
            for (uint32_t g = 0; g < numGroups; g++) {
                ProcessGroup(b, g);
            }
        }
    }

private:
    __aicore__ inline void ProcessGroup(uint32_t batch, uint32_t group)
    {
        uint32_t alignedGroupSpatial = ((groupSpatialSize + 7) / 8) * 8;
        uint32_t offset = batch * totalPerBatch + group * groupSpatialSize;
        
        // Step 1: Load data and apply Swish: x * sigmoid(x)
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], alignedGroupSpatial);
        inQueueX.EnQue(xLocal);
        
        xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> swishLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.AllocTensor<float>();
        
        // sigmoid(x) = 1 / (1 + exp(-x))
        // Use Sigmoid API
        AscendC::Sigmoid(tmp1, xLocal, groupSpatialSize);
        // swish = x * sigmoid(x)
        AscendC::Mul(swishLocal, xLocal, tmp1, groupSpatialSize);
        
        inQueueX.FreeTensor(xLocal);
        tmpBuf1.FreeTensor(tmp1);
        
        // Step 2: Compute mean and variance for group norm
        // We need to compute mean and var over groupSpatialSize elements
        float mean = 0.0f;
        float var = 0.0f;
        
        // Use ReduceSum to compute sum
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1b = tmpBuf1.AllocTensor<float>();
        
        // Compute sum for mean
        // ReduceSum: reduce swishLocal into a scalar sum
        float invN = 1.0f / (float)groupSpatialSize;
        
        // Compute mean: sum all elements
        AscendC::ReduceSum(tmp1b, swishLocal, tmp2, groupSpatialSize);
        mean = tmp1b.GetValue(0) * invN;
        
        // Compute variance: sum((x - mean)^2) / N
        AscendC::Adds(tmp1b, swishLocal, -mean, groupSpatialSize);
        AscendC::Mul(tmp2, tmp1b, tmp1b, groupSpatialSize);
        
        AscendC::LocalTensor<float> xLocal2 = inQueueX.AllocTensor<float>();
        AscendC::ReduceSum(xLocal2, tmp2, tmp1b, groupSpatialSize);
        var = xLocal2.GetValue(0) * invN;
        inQueueX.FreeTensor(xLocal2);
        
        // Step 3: Normalize: (x - mean) / sqrt(var + eps)
        float invStd = 1.0f / sqrtf(var + eps);
        
        // tmp1b already has (swish - mean)
        AscendC::Adds(tmp1b, swishLocal, -mean, groupSpatialSize);
        AscendC::Muls(tmp1b, tmp1b, invStd, groupSpatialSize);
        
        // Step 4: Apply gamma and beta per channel
        // For each channel in the group, scale and shift
        for (uint32_t c = 0; c < channelsPerGroup; c++) {
            uint32_t channelIdx = group * channelsPerGroup + c;
            float g_val = gammaGm.GetValue(channelIdx);
            float b_val = betaGm.GetValue(channelIdx);
            uint32_t startIdx = c * spatialSize;
            uint32_t alignedSpatial = ((spatialSize + 7) / 8) * 8;
            
            // normalized * gamma + beta for this channel's spatial elements
            // We work in-place within tmp1b
            for (uint32_t s = 0; s < spatialSize; s++) {
                float val = tmp1b.GetValue(startIdx + s);
                val = val * g_val + b_val;
                tmp1b.SetValue(startIdx + s, val);
            }
        }
        
        // Step 5: Apply HardSwish: x * min(max(x + 3, 0), 6) / 6
        // hardswish(x) = x * relu6(x + 3) / 6
        AscendC::Adds(tmp2, tmp1b, 3.0f, groupSpatialSize);
        
        // Clamp to [0, 6]: use Mins and Maxs (or manual)
        // relu: max(x, 0)
        AscendC::Maxs(tmp2, tmp2, 0.0f, groupSpatialSize);
        // min(x, 6)
        AscendC::Mins(tmp2, tmp2, 6.0f, groupSpatialSize);
        // x * clamp / 6
        AscendC::Mul(swishLocal, tmp1b, tmp2, groupSpatialSize);
        AscendC::Muls(swishLocal, swishLocal, 1.0f / 6.0f, groupSpatialSize);
        
        tmpBuf1.FreeTensor(tmp1b);
        tmpBuf2.FreeTensor(tmp2);
        
        outQueueY.EnQue(swishLocal);
        
        // Step 6: Write output
        AscendC::LocalTensor<float> outLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[offset], outLocal, alignedGroupSpatial);
        outQueueY.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm, gammaGm, betaGm, yGm;
    
    uint32_t batchSize, numChannels, spatialSize, numGroups;
    uint32_t channelsPerGroup, groupSpatialSize, totalPerBatch;
    uint32_t batchStart, batchEnd;
    float eps;
};

extern "C" __global__ __aicore__ void conv_transpose3d_swish_group_norm_hard_swish_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSwishGNHardSwish op;
    op.Init(x, gamma, beta, y,
            tiling_data.batchSize, tiling_data.numChannels,
            tiling_data.spatialSize, tiling_data.numGroups, tiling_data.eps);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model that performs 3D transposed convolution, then uses a custom
    fused kernel for Swish + GroupNorm + HardSwish.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, groups, eps, bias=True):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, bias=bias)
        self.group_norm = torch.nn.GroupNorm(num_groups=groups, num_channels=out_channels, eps=eps)

    def forward(self, x):
        x = self.conv_transpose(x)
        # Fused Swish + GroupNorm + HardSwish
        # Reshape x to be contiguous in (N, C, D*H*W) layout
        x = x.contiguous()
        gamma = self.group_norm.weight.contiguous()
        beta = self.group_norm.bias.contiguous()
        x = custom_ops_lib.conv_transpose3d_swish_group_norm_hard_swish_custom(x, gamma, beta)
        return x
'''
