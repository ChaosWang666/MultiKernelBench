kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmDivideSumScaling {
public:
    __aicore__ inline KernelGemmDivideSumScaling() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR z,
                                 uint32_t batchSize, uint32_t inputSize,
                                 uint32_t hiddenSize, float scalingFactor)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->scalingFactor = scalingFactor;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        // Each block processes a range of batch rows
        this->rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        this->endRow = startRow + rowsPerBlock;
        if (this->endRow > batchSize) this->endRow = batchSize;

        // Global memory setup
        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inputSize);
        wGm.SetGlobalBuffer((__gm__ float *)weight, hiddenSize * inputSize);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize);

        // We process inputSize elements at a time for dot product
        // Tile the inputSize dimension
        this->tileLength = 1024;  // process 1024 floats at a time
        if (this->tileLength > inputSize) this->tileLength = inputSize;
        // Align tileLength to 8 (32 bytes / 4 bytes per float)
        this->tileLength = (this->tileLength / 8) * 8;
        if (this->tileLength == 0) this->tileLength = 8;

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueue, 1, 8 * sizeof(float)); // small buffer for reduction result
        pipe.InitBuffer(workBuf, 1, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t row = this->startRow; row < this->endRow; row++) {
            ProcessOneRow(row);
        }
    }

private:
    __aicore__ inline void ProcessOneRow(uint32_t row)
    {
        // For each batch row, compute:
        // result = sum_j( sum_k(x[row,k] * weight[j,k]) / 2 ) * scalingFactor
        // = (scalingFactor / 2) * sum_j sum_k (x[row,k] * weight[j,k])
        // = (scalingFactor / 2) * sum_k ( x[row,k] * sum_j(weight[j,k]) )
        // 
        // So we can precompute colSum[k] = sum_j weight[j,k], then dot with x row.
        // But weight is large. Let's just do it directly: accumulate over hidden dims.
        //
        // Actually, the simplest approach:
        // gemm result: g[j] = sum_k x[row,k]*w[j,k] for j in [0, hiddenSize)
        // divide: g[j] /= 2
        // sum: result = sum_j g[j]
        // scale: result *= scalingFactor
        //
        // Equivalently: result = (scalingFactor/2) * sum_j sum_k x[row,k]*w[j,k]
        //             = (scalingFactor/2) * sum_k x[row,k] * (sum_j w[j,k])
        //
        // We can compute columnSums of weight incrementally, then dot with x.
        // This is O(hiddenSize*inputSize + inputSize) vs O(hiddenSize*inputSize).
        // Both are same asymptotically but the second avoids storing hiddenSize results.
        
        // Strategy: accumulate sum over hidden dimension in tiles of inputSize
        // For each tile of k:
        //   load x[row, k_start:k_end]
        //   For each j in hiddenSize:
        //     load w[j, k_start:k_end]
        //     element-wise multiply and accumulate
        // This is still O(H*K) loads of weight which is expensive.
        //
        // Better: use the identity. Compute wColSum[k] = sum_j w[j,k], then dot x with wColSum.
        // For wColSum, iterate over j for each tile of k.
        
        float totalSum = 0.0f;
        uint32_t numTiles = (inputSize + this->tileLength - 1) / this->tileLength;

        for (uint32_t t = 0; t < numTiles; t++) {
            uint32_t kStart = t * this->tileLength;
            uint32_t kLen = this->tileLength;
            if (kStart + kLen > inputSize) kLen = inputSize - kStart;
            // Align kLen up to 8 for the last tile
            uint32_t kLenAligned = ((kLen + 7) / 8) * 8;

            // Load x[row, kStart:kStart+kLen]
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[row * inputSize + kStart], kLenAligned);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            // Compute wColSum for this tile: sum over j of w[j, kStart:kStart+kLen]
            AscendC::LocalTensor<float> colSumLocal = workBuf.AllocTensor<float>();
            // Initialize to zero
            AscendC::Duplicate<float>(colSumLocal, 0.0f, kLenAligned);

            for (uint32_t j = 0; j < hiddenSize; j++) {
                AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                AscendC::DataCopy(wLocal, wGm[j * inputSize + kStart], kLenAligned);
                inQueueW.EnQue(wLocal);
                wLocal = inQueueW.DeQue<float>();
                AscendC::Add(colSumLocal, colSumLocal, wLocal, kLenAligned);
                inQueueW.FreeTensor(wLocal);
            }

            // Multiply x * colSum
            AscendC::Mul(colSumLocal, xLocal, colSumLocal, kLenAligned);
            inQueueX.FreeTensor(xLocal);

            // Sum the products
            // Use ReduceSum
            AscendC::LocalTensor<float> resultLocal = outQueue.AllocTensor<float>();
            AscendC::ReduceSum(resultLocal, colSumLocal, colSumLocal, kLenAligned);
            float tileSum = resultLocal.GetValue(0);
            totalSum += tileSum;
            outQueue.FreeTensor(resultLocal);

            workBuf.FreeTensor(colSumLocal);
        }

        // Apply divide by 2 and scaling
        totalSum = totalSum * (this->scalingFactor / 2.0f);
        
        // Workaround: write the scalar to global memory
        // We need to use pipe to write, but for a single value we can use DataCopy with aligned size
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        outLocal.SetValue(0, totalSum);
        // Need at least 8 elements for DataCopy alignment
        AscendC::DataCopy(zGm[row], outLocal, 8);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t inputSize;
    uint32_t hiddenSize;
    uint32_t startRow, endRow, rowsPerBlock;
    uint32_t tileLength;
    float scalingFactor;
};

extern "C" __global__ __aicore__ void gemm_divide_sum_scaling_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmDivideSumScaling op;
    op.Init(x, weight, z, tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize, tiling_data.scalingFactor);
    op.Process();
}
