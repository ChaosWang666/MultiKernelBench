

Looking at the architecture, I need to create a custom AscendC kernel that fuses the scaling and minimum-along-channel operations. The convolution will remain as the standard PyTorch operation since it's complex and already well-optimized, but I'll fuse `x * scale_factor` followed by `torch.min(x, dim=1, keepdim=True)[0]` into a single custom kernel.

The fused kernel takes the conv output of shape (batch_size, out_channels, height, width), scales it by scale_factor, and computes the minimum along the channel dimension (dim=1), producing output of shape (batch_size, 1, height, width).

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv2dScalingMin {
public:
    __aicore__ inline KernelConv2dScalingMin() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t channels,
                                 uint32_t spatialSize, float scaleFactor, uint32_t totalSpatial, uint32_t tileNum)
    {
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->scaleFactor = scaleFactor;
        this->totalSpatial = totalSpatial;

        // Each block processes a portion of the total spatial elements (batch * H * W)
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->blockSpatialLen = totalSpatial / blockNum;
        uint32_t remainder = totalSpatial % blockNum;
        if (blockIdx < remainder) {
            this->blockSpatialLen += 1;
            this->spatialOffset = blockIdx * this->blockSpatialLen;
        } else {
            this->spatialOffset = blockIdx * this->blockSpatialLen + remainder;
        }

        // Align tileLength to 32 bytes (8 floats)
        this->tileNum = tileNum;
        if (this->blockSpatialLen < this->tileNum) {
            this->tileNum = this->blockSpatialLen > 0 ? this->blockSpatialLen : 1;
        }
        this->tileLength = this->blockSpatialLen / this->tileNum;
        this->tileTail = this->blockSpatialLen % this->tileNum;

        // Align tile length to 8 (32 bytes for float)
        uint32_t alignedTileLength = ((this->tileLength + 7) / 8) * 8;
        if (alignedTileLength == 0) alignedTileLength = 8;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * channels * spatialSize);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->spatialOffset, this->blockSpatialLen);

        // We need one buffer for reading one channel tile and one for accumulating min
        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedTileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, alignedTileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        uint32_t offset = 0;
        for (uint32_t t = 0; t < this->tileNum; t++) {
            uint32_t curLen = this->tileLength;
            if (t < this->tileTail) curLen += 1;
            if (curLen == 0) continue;
            uint32_t alignedLen = ((curLen + 7) / 8) * 8;
            ProcessTile(offset, curLen, alignedLen);
            offset += curLen;
        }
    }

private:
    __aicore__ inline void ProcessTile(uint32_t spatialStart, uint32_t curLen, uint32_t alignedLen)
    {
        // Initialize min buffer with first channel data * scale
        AscendC::LocalTensor<float> minLocal = outQueueZ.AllocTensor<float>();

        // Process first channel
        {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            // For each spatial position, compute batch and spatial index
            // Global spatial index = spatialOffset + spatialStart + i
            // batch = globalSpatialIdx / spatialSize
            // sp = globalSpatialIdx % spatialSize
            // x index = batch * channels * spatialSize + channel * spatialSize + sp
            // We need to gather from non-contiguous memory per channel

            // We'll iterate spatial elements; but they may span multiple batches
            // For efficiency, let's do channel-by-channel, copying spatial tile
            uint32_t globalStart = this->spatialOffset + spatialStart;

            // Check if tile is within single batch for optimization
            uint32_t startBatch = globalStart / this->spatialSize;
            uint32_t endBatch = (globalStart + curLen - 1) / this->spatialSize;

            if (startBatch == endBatch) {
                uint32_t sp = globalStart - startBatch * this->spatialSize;
                uint32_t srcOffset = startBatch * this->channels * this->spatialSize + 0 * this->spatialSize + sp;
                AscendC::DataCopy(xLocal, xGm[srcOffset], alignedLen);
                AscendC::DataCacheCleanAndInvalid<float, AscendC::TPosition::VECIN>(xLocal);
            } else {
                // Handle cross-batch: copy element by element into xLocal
                // For simplicity, copy in sub-segments per batch
                uint32_t copied = 0;
                for (uint32_t b = startBatch; b <= endBatch && copied < curLen; b++) {
                    uint32_t batchStart = b * this->spatialSize;
                    uint32_t segStart = (globalStart + copied > batchStart) ? (globalStart + copied - batchStart) : 0;
                    uint32_t segEnd = this->spatialSize;
                    if (b == endBatch) {
                        segEnd = globalStart + curLen - batchStart;
                    }
                    uint32_t segLen = segEnd - segStart;
                    uint32_t alignedSegLen = ((segLen + 7) / 8) * 8;
                    uint32_t srcOff = b * this->channels * this->spatialSize + 0 * this->spatialSize + segStart;
                    // Use DataCopy for the segment
                    AscendC::DataCopy(xLocal[copied], xGm[srcOff], alignedSegLen);
                    copied += segLen;
                }
                AscendC::DataCacheCleanAndInvalid<float, AscendC::TPosition::VECIN>(xLocal);
            }
            inQueueX.EnQue(xLocal);
            AscendC::LocalTensor<float> xIn = inQueueX.DeQue<float>();
            // Scale and copy to min
            float sf = this->scaleFactor;
            AscendC::Muls(minLocal, xIn, sf, alignedLen);
            inQueueX.FreeTensor(xIn);
        }

        // Process remaining channels
        for (uint32_t c = 1; c < this->channels; c++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            uint32_t globalStart = this->spatialOffset + spatialStart;
            uint32_t startBatch = globalStart / this->spatialSize;
            uint32_t endBatch = (globalStart + curLen - 1) / this->spatialSize;

            if (startBatch == endBatch) {
                uint32_t sp = globalStart - startBatch * this->spatialSize;
                uint32_t srcOffset = startBatch * this->channels * this->spatialSize + c * this->spatialSize + sp;
                AscendC::DataCopy(xLocal, xGm[srcOffset], alignedLen);
                AscendC::DataCacheCleanAndInvalid<float, AscendC::TPosition::VECIN>(xLocal);
            } else {
                uint32_t copied = 0;
                for (uint32_t b = startBatch; b <= endBatch && copied < curLen; b++) {
                    uint32_t batchStart = b * this->spatialSize;
                    uint32_t segStart = (globalStart + copied > batchStart) ? (globalStart + copied - batchStart) : 0;
                    uint32_t segEnd = this->spatialSize;
                    if (b == endBatch) {
                        segEnd = globalStart + curLen - batchStart;
                    }
                    uint32_t segLen = segEnd - segStart;
                    uint32_t alignedSegLen = ((segLen + 7) / 8) * 8;
                    uint32_t srcOff = b * this->channels * this->spatialSize + c * this->spatialSize + segStart;
                    AscendC::DataCopy(xLocal[copied], xGm[srcOff], alignedSegLen);
                    copied += segLen;
                }
                AscendC::DataCacheCleanAndInvalid<float, AscendC::TPosition::VECIN>(xLocal);
            }
            inQueueX.EnQue(xLocal);
            AscendC::LocalTensor<float> xIn = inQueueX.DeQue<float>();
            // Scale
            float sf = this->scaleFactor;
            AscendC::Muls(xIn, xIn, sf, alignedLen);
            // Min with current min
            AscendC::Min(minLocal, minLocal, xIn, alignedLen);
            inQueueX.FreeTensor(xIn);
        }

        // Copy result out
        AscendC::DataCopy(yGm[spatialStart], minLocal, alignedLen);
        outQueueZ.FreeTensor(minLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t channels;
    uint32_t spatialSize;
    float scaleFactor;
    uint32_t totalSpatial;
    uint32_t blockSpatialLen;
    uint32_t spatialOffset;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t tileTail;
};

extern "C" __global__ __aicore__ void conv2d_scaling_min_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dScalingMin op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.channels,
            tiling_data.spatialSize, tiling_data.scaleFactor,
            tiling_data.totalSpatial, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model that performs convolution, then uses a custom fused kernel
    for scaling and min along channel dimension.
    """
    def __init__(self, in_channels, out_channels, kernel_size, scale_factor):
        super().__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.scale_factor = scale_factor

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
        result = custom_ops_lib.conv2d_scaling_min_custom(x, self.scale_factor, x.size(1), x.size(2) * x.size(3))
        return result
'''
