kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSoftmax {
public:
    __aicore__ inline KernelSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t dimSize, uint32_t tileNum)
    {
        this->totalLength = totalLength;
        this->dimSize = dimSize;
        // Total number of rows (each row has dimSize elements, softmax applied per row)
        uint32_t totalRows = totalLength / dimSize;
        // Distribute rows across blocks
        uint32_t rowsPerBlock = totalRows / AscendC::GetBlockNum();
        uint32_t remainRows = totalRows % AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        if (blockIdx < remainRows) {
            this->blockRows = rowsPerBlock + 1;
            this->blockOffset = blockIdx * this->blockRows * dimSize;
        } else {
            this->blockRows = rowsPerBlock;
            this->blockOffset = (remainRows * (rowsPerBlock + 1) + (blockIdx - remainRows) * rowsPerBlock) * dimSize;
        }
        
        this->blockLength = this->blockRows * dimSize;
        
        // Align dimSize to 32 bytes (8 floats)
        this->alignedDimSize = (dimSize + 7) / 8 * 8;
        
        // Process multiple rows per tile if possible
        // Each tile processes some rows; we want tileNum tiles with BUFFER_NUM buffers
        uint32_t totalTiles = tileNum * BUFFER_NUM;
        if (totalTiles > this->blockRows) {
            totalTiles = this->blockRows > 0 ? this->blockRows : 1;
        }
        this->rowsPerTile = this->blockRows / totalTiles;
        this->remainTileRows = this->blockRows % totalTiles;
        this->loopCount = totalTiles;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockOffset, this->blockLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->blockOffset, this->blockLength);
        
        // Buffer size: max rows per tile * alignedDimSize
        uint32_t maxRowsPerTile = this->rowsPerTile + (this->remainTileRows > 0 ? 1 : 0);
        uint32_t bufSize = maxRowsPerTile * this->alignedDimSize * sizeof(float);
        if (bufSize == 0) bufSize = this->alignedDimSize * sizeof(float);
        
        pipe.InitBuffer(inQueueX, BUFFER_NUM, bufSize);
        pipe.InitBuffer(outQueueY, BUFFER_NUM, bufSize);
        // Temp buffers for max, sum, etc.
        pipe.InitBuffer(tmpBuf, 1, this->alignedDimSize * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        if (this->blockRows == 0) return;
        uint32_t rowOffset = 0;
        for (int32_t i = 0; i < this->loopCount; i++) {
            uint32_t curRows = this->rowsPerTile + (i < this->remainTileRows ? 1 : 0);
            CopyIn(rowOffset, curRows);
            Compute(curRows);
            CopyOut(rowOffset, curRows);
            rowOffset += curRows;
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowOffset, uint32_t curRows)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        for (uint32_t r = 0; r < curRows; r++) {
            AscendC::DataCopy(xLocal[r * this->alignedDimSize], xGm[(rowOffset + r) * this->dimSize], this->alignedDimSize);
        }
        inQueueX.EnQue(xLocal);
    }
    
    __aicore__ inline void Compute(uint32_t curRows)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        
        for (uint32_t r = 0; r < curRows; r++) {
            uint32_t off = r * this->alignedDimSize;
            // Compute max
            AscendC::ReduceMax(tmpLocal, xLocal[off], this->alignedDimSize, false);
            float maxVal = tmpLocal.GetValue(0);
            
            // Subtract max: y = x - max
            AscendC::Adds(yLocal[off], xLocal[off], -maxVal, this->alignedDimSize);
            
            // Exp: y = exp(y)
            AscendC::Exp(yLocal[off], yLocal[off], this->alignedDimSize);
            
            // Zero out padding elements if dimSize < alignedDimSize
            for (uint32_t p = this->dimSize; p < this->alignedDimSize; p++) {
                yLocal.SetValue(off + p, 0.0f);
            }
            
            // Sum
            AscendC::ReduceSum(tmpLocal, yLocal[off], this->alignedDimSize, false);
            float sumVal = tmpLocal.GetValue(0);
            
            // Divide
            float invSum = 1.0f / sumVal;
            AscendC::Muls(yLocal[off], yLocal[off], invSum, this->alignedDimSize);
        }
        
        outQueueY.EnQue(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    
    __aicore__ inline void CopyOut(uint32_t rowOffset, uint32_t curRows)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        for (uint32_t r = 0; r < curRows; r++) {
            AscendC::DataCopy(yGm[(rowOffset + r) * this->dimSize], yLocal[r * this->alignedDimSize], this->alignedDimSize);
        }
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t totalLength;
    uint32_t dimSize;
    uint32_t alignedDimSize;
    uint32_t blockLength;
    uint32_t blockRows;
    uint32_t blockOffset;
    uint32_t rowsPerTile;
    uint32_t remainTileRows;
    uint32_t loopCount;
};

extern "C" __global__ __aicore__ void unet_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSoftmax op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.dimSize, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class CustomSoftmax(nn.Module):
    def __init__(self, dim=-1):
        super().__init__()
        self.dim = dim

    def forward(self, x):
        return custom_ops_lib.unet_softmax_custom(x)

class DoubleConv(nn.Module):
    def __init__(self, in_channels, out_channels):
        super().__init__()
        self.double_conv = nn.Sequential(
            nn.Conv2d(in_channels, out_channels, kernel_size=3, padding=1),
            nn.BatchNorm2d(out_channels),
            CustomSoftmax(dim=-1),
            nn.Conv2d(out_channels, out_channels, kernel_size=3, padding=1),
            nn.BatchNorm2d(out_channels),
            CustomSoftmax(dim=-1)
        )

    def forward(self, x):
        return self.double_conv(x)

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, features):
        super(ModelNew, self).__init__()
        self.encoder1 = DoubleConv(in_channels, features)
        self.pool1 = nn.MaxPool2d(kernel_size=2, stride=2)
        self.encoder2 = DoubleConv(features, features * 2)
        self.pool2 = nn.MaxPool2d(kernel_size=2, stride=2)
        self.encoder3 = DoubleConv(features * 2, features * 4)
        self.pool3 = nn.MaxPool2d(kernel_size=2, stride=2)
        self.encoder4 = DoubleConv(features * 4, features * 8)
        self.pool4 = nn.MaxPool2d(kernel_size=2, stride=2)

        self.bottleneck = DoubleConv(features * 8, features * 16)

        self.upconv4 = nn.ConvTranspose2d(features * 16, features * 8, kernel_size=2, stride=2)
        self.decoder4 = DoubleConv(features * 16, features * 8)
        self.upconv3 = nn.ConvTranspose2d(features * 8, features * 4, kernel_size=2, stride=2)
        self.decoder3 = DoubleConv(features * 8, features * 4)
        self.upconv2 = nn.ConvTranspose2d(features * 4, features * 2, kernel_size=2, stride=2)
        self.decoder2 = DoubleConv(features * 4, features * 2)
        self.upconv1 = nn.ConvTranspose2d(features * 2, features, kernel_size=2, stride=2)
        self.decoder1 = DoubleConv(features * 2, features)

        self.final_conv = nn.Conv2d(features, out_channels, kernel_size=1)

    def forward(self, x):
        enc1 = self.encoder1(x)
        enc2 = self.encoder2(self.pool1(enc1))
        enc3 = self.encoder3(self.pool2(enc2))
        enc4 = self.encoder4(self.pool3(enc3))

        bottleneck = self.bottleneck(self.pool4(enc4))

        dec4 = self.upconv4(bottleneck)
        dec4 = torch.cat((dec4, enc4), dim=1)
        dec4 = self.decoder4(dec4)
        dec3 = self.upconv3(dec4)
        dec3 = torch.cat((dec3, enc3), dim=1)
        dec3 = self.decoder3(dec3)
        dec2 = self.upconv2(dec3)
        dec2 = torch.cat((dec2, enc2), dim=1)
        dec2 = self.decoder2(dec2)
        dec1 = self.upconv1(dec2)
        dec1 = torch.cat((dec1, enc1), dim=1)
        dec1 = self.decoder1(dec1)

        return self.final_conv(dec1)
'''
