kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelClampBroadcast {
public:
    __aicore__ inline KernelClampBroadcast() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR minVal, GM_ADDR maxVal, GM_ADDR z,
                                 uint32_t totalLength, uint32_t tileNum, uint32_t rowLength)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->rowLength = rowLength;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        minGm.SetGlobalBuffer((__gm__ float *)minVal, rowLength);
        maxGm.SetGlobalBuffer((__gm__ float *)maxVal, rowLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueMin, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueMax, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> minLocal = inQueueMin.AllocTensor<float>();
        AscendC::LocalTensor<float> maxLocal = inQueueMax.AllocTensor<float>();

        // Copy x tile
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);

        // Broadcast min_val and max_val: they have shape (rowLength,) and need to be tiled
        // The global offset for this block starts at blockLength * blockIdx
        // We need to figure out which elements of min/max correspond to this tile
        uint32_t globalOffset = this->blockLength * AscendC::GetBlockIdx() + progress * this->tileLength;
        // For each element in the tile, the min/max index is (globalOffset + i) % rowLength
        // If tileLength is a multiple of rowLength, we can copy full rows
        // Otherwise we need element-wise handling

        // Since rowLength=2048 and tiles are aligned, we can copy repeated rows
        uint32_t elemsRemaining = this->tileLength;
        uint32_t dstOffset = 0;
        uint32_t srcStart = globalOffset % this->rowLength;

        while (elemsRemaining > 0) {
            uint32_t copyLen = this->rowLength - srcStart;
            if (copyLen > elemsRemaining) {
                copyLen = elemsRemaining;
            }
            AscendC::DataCopy(minLocal[dstOffset], minGm[srcStart], copyLen);
            AscendC::DataCopy(maxLocal[dstOffset], maxGm[srcStart], copyLen);
            dstOffset += copyLen;
            elemsRemaining -= copyLen;
            srcStart = 0;
        }

        inQueueX.EnQue(xLocal);
        inQueueMin.EnQue(minLocal);
        inQueueMax.EnQue(maxLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> minLocal = inQueueMin.DeQue<float>();
        AscendC::LocalTensor<float> maxLocal = inQueueMax.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        // clamp: z = min(max(x, min_val), max_val)
        AscendC::Maxs(zLocal, xLocal, minLocal, this->tileLength);
        AscendC::Mins(zLocal, zLocal, maxLocal, this->tileLength);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueMin.FreeTensor(minLocal);
        inQueueMax.FreeTensor(maxLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueMin, inQueueMax;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> minGm;
    AscendC::GlobalTensor<float> maxGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t rowLength;
};

extern "C" __global__ __aicore__ void clamp_broadcast_custom(GM_ADDR x, GM_ADDR min_val, GM_ADDR max_val, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelClampBroadcast op;
    op.Init(x, min_val, max_val, z, tiling_data.totalLength, tiling_data.tileNum, tiling_data.rowLength);
    op.Process();
}
