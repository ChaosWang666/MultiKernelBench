

Looking at the architecture, I need to fuse the operations: matmul → swish (sigmoid * x) → bias add → group_norm into a custom kernel. Given the complexity, I'll fuse the swish + bias_add + group_norm into a single custom kernel, keeping the matmul (nn.Linear) in PyTorch since matrix multiplication is already highly optimized.

The kernel `matmul_swish_sum_group_norm_custom` will take the output of matmul, apply swish activation, add bias, and perform group normalization.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelMatmulSwishSumGroupNorm {
public:
    __aicore__ inline KernelMatmulSwishSumGroupNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                 uint32_t batchSize, uint32_t outFeatures, uint32_t numGroups, float eps)
    {
        this->batchSize = batchSize;
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->eps = eps;
        this->channelsPerGroup = outFeatures / numGroups;

        uint32_t totalRows = batchSize;
        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->rowsPerBlock = (totalRows + numBlocks - 1) / numBlocks;
        this->startRow = blockIdx * this->rowsPerBlock;
        this->endRow = startRow + rowsPerBlock;
        if (this->endRow > totalRows) this->endRow = totalRows;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * outFeatures);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, outFeatures);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, outFeatures);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, outFeatures);
        yGm.SetGlobalBuffer((__gm__ float*)y, batchSize * outFeatures);

        // Allocate buffers for one row
        pipe.InitBuffer(inQueueX, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(inQueueGamma, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(inQueueBeta, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf, BUFFER_NUM, outFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t row = startRow; row < endRow; row++) {
            CopyIn(row);
            Compute(row);
            CopyOut(row);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.AllocTensor<float>();
        AscendC::LocalTensor<float> betaLocal = inQueueBeta.AllocTensor<float>();

        AscendC::DataCopy(xLocal, xGm[row * outFeatures], outFeatures);
        AscendC::DataCopy(biasLocal, biasGm[0], outFeatures);
        AscendC::DataCopy(gammaLocal, gammaGm[0], outFeatures);
        AscendC::DataCopy(betaLocal, betaGm[0], outFeatures);

        inQueueX.EnQue(xLocal);
        inQueueBias.EnQue(biasLocal);
        inQueueGamma.EnQue(gammaLocal);
        inQueueBeta.EnQue(betaLocal);
    }

    __aicore__ inline void Compute(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.DeQue<float>();
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.DeQue<float>();
        AscendC::LocalTensor<float> betaLocal = inQueueBeta.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.AllocTensor<float>();

        // Swish: sigmoid(x) * x
        // tmpLocal = sigmoid(x)
        AscendC::Sigmoid(tmpLocal, xLocal, outFeatures);
        // yLocal = tmpLocal * xLocal = sigmoid(x) * x
        AscendC::Mul(yLocal, tmpLocal, xLocal, outFeatures);

        // Add bias: yLocal = yLocal + biasLocal
        AscendC::Add(yLocal, yLocal, biasLocal, outFeatures);

        // GroupNorm per group
        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t offset = g * channelsPerGroup;
            // Compute mean
            float sum = 0.0f;
            for (uint32_t c = 0; c < channelsPerGroup; c++) {
                sum += yLocal.GetValue(offset + c);
            }
            float mean = sum / (float)channelsPerGroup;

            // Compute variance
            float varSum = 0.0f;
            for (uint32_t c = 0; c < channelsPerGroup; c++) {
                float diff = yLocal.GetValue(offset + c) - mean;
                varSum += diff * diff;
            }
            float variance = varSum / (float)channelsPerGroup;
            float invStd = 1.0f / sqrtf(variance + eps);

            // Normalize, scale (gamma), shift (beta)
            for (uint32_t c = 0; c < channelsPerGroup; c++) {
                float val = yLocal.GetValue(offset + c);
                float normed = (val - mean) * invStd;
                float result = normed * gammaLocal.GetValue(offset + c) + betaLocal.GetValue(offset + c);
                yLocal.SetValue(offset + c, result);
            }
        }

        outQueue.EnQue(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueBias.FreeTensor(biasLocal);
        inQueueGamma.FreeTensor(gammaLocal);
        inQueueBeta.FreeTensor(betaLocal);
        tmpBuf.FreeTensor(tmpLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row)
    {
        AscendC::LocalTensor<float> yLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(yGm[row * outFeatures], yLocal, outFeatures);
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueBias, inQueueGamma, inQueueBeta;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> tmpBuf;
    AscendC::GlobalTensor<float> xGm, biasGm, gammaGm, betaGm, yGm;
    uint32_t batchSize, outFeatures, numGroups, channelsPerGroup;
    uint32_t rowsPerBlock, startRow, endRow;
    float eps;
};

extern "C" __global__ __aicore__ void matmul_swish_sum_group_norm_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSwishSumGroupNorm op;
    op.Init(x, bias, gamma, beta, y, tiling_data.batchSize, tiling_data.outFeatures, tiling_data.numGroups, tiling_data.eps);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model that uses a custom AscendC kernel to fuse Swish activation,
    bias addition, and GroupNorm after the linear layer.
    """
    def __init__(self, in_features, out_features, num_groups, bias_shape):
        super(ModelNew, self).__init__()
        self.matmul = torch.nn.Linear(in_features, out_features)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.group_norm = torch.nn.GroupNorm(num_groups, out_features)

    def forward(self, x):
        x = self.matmul(x)
        # Fused swish + bias_add + group_norm using custom kernel
        x = custom_ops_lib.matmul_swish_sum_group_norm_custom(
            x, self.bias, self.group_norm.weight, self.group_norm.bias
        )
        return x
