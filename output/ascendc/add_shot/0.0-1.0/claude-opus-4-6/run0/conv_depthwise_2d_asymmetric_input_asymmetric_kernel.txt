kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvDepthwise2d {
public:
    __aicore__ inline KernelConvDepthwise2d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels,
                                 uint32_t inH, uint32_t inW,
                                 uint32_t kernelH, uint32_t kernelW,
                                 uint32_t strideH, uint32_t strideW,
                                 uint32_t paddingH, uint32_t paddingW,
                                 uint32_t dilationH, uint32_t dilationW,
                                 uint32_t outH, uint32_t outW,
                                 uint32_t hasBias,
                                 uint32_t totalOutputChannels, uint32_t blockDim)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->inH = inH;
        this->inW = inW;
        this->kernelH = kernelH;
        this->kernelW = kernelW;
        this->strideH = strideH;
        this->strideW = strideW;
        this->paddingH = paddingH;
        this->paddingW = paddingW;
        this->dilationH = dilationH;
        this->dilationW = dilationW;
        this->outH = outH;
        this->outW = outW;
        this->hasBias = hasBias;
        this->totalOutputChannels = totalOutputChannels;

        this->inSpatial = inH * inW;
        this->outSpatial = outH * outW;
        this->kernelSize = kernelH * kernelW;

        // Each block handles a subset of (batch, channel) pairs
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t numBlocks = blockDim;
        
        // Calculate channels per block
        this->channelsPerBlock = (totalOutputChannels + numBlocks - 1) / numBlocks;
        this->startChannel = blockIdx * this->channelsPerBlock;
        this->endChannel = startChannel + channelsPerBlock;
        if (this->endChannel > totalOutputChannels) {
            this->endChannel = totalOutputChannels;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inChannels * inSpatial);
        weightGm.SetGlobalBuffer((__gm__ float*)weight, inChannels * kernelSize);
        if (hasBias) {
            biasGm.SetGlobalBuffer((__gm__ float*)bias, inChannels);
        }
        yGm.SetGlobalBuffer((__gm__ float*)y, batchSize * inChannels * outSpatial);

        // Allocate buffers for processing output rows
        // We process output one row at a time for each channel
        // Align outW to 8 for vector ops
        uint32_t alignedOutW = ((outW + 7) / 8) * 8;
        this->alignedOutW = alignedOutW;

        pipe.InitBuffer(inQueueX, 1, alignedOutW * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, alignedOutW * sizeof(float));
        pipe.InitBuffer(tmpBuf, 1, alignedOutW * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t bc = startChannel; bc < endChannel; bc++) {
            uint32_t b = bc / inChannels;
            uint32_t c = bc % inChannels;
            ProcessOneChannel(b, c);
        }
    }

private:
    __aicore__ inline void ProcessOneChannel(uint32_t b, uint32_t c)
    {
        uint32_t inOffset = (b * inChannels + c) * inSpatial;
        uint32_t outOffset = (b * inChannels + c) * outSpatial;
        uint32_t weightOffset = c * kernelSize;

        for (uint32_t oh = 0; oh < outH; oh++) {
            // Compute one output row
            AscendC::LocalTensor<float> outLocal = outQueueY.AllocTensor<float>();
            
            // Zero initialize the output row
            AscendC::Duplicate(outLocal, (float)0.0f, alignedOutW);

            for (uint32_t kh = 0; kh < kernelH; kh++) {
                for (uint32_t kw = 0; kw < kernelW; kw++) {
                    int32_t ih_base = (int32_t)(oh * strideH) - (int32_t)paddingH + (int32_t)(kh * dilationH);
                    if (ih_base < 0 || ih_base >= (int32_t)inH) continue;

                    float w_val = weightGm.GetValue(weightOffset + kh * kernelW + kw);

                    // For each output column
                    for (uint32_t ow = 0; ow < outW; ow++) {
                        int32_t iw = (int32_t)(ow * strideW) - (int32_t)paddingW + (int32_t)(kw * dilationW);
                        if (iw < 0 || iw >= (int32_t)inW) continue;

                        float x_val = xGm.GetValue(inOffset + ih_base * inW + iw);
                        float cur = outLocal.GetValue(ow);
                        outLocal.SetValue(ow, cur + x_val * w_val);
                    }
                }
            }

            // Add bias if needed
            if (hasBias) {
                float bias_val = biasGm.GetValue(c);
                for (uint32_t ow = 0; ow < outW; ow++) {
                    float cur = outLocal.GetValue(ow);
                    outLocal.SetValue(ow, cur + bias_val);
                }
            }

            // Copy output row to global memory
            outQueueY.EnQue(outLocal);
            AscendC::LocalTensor<float> outResult = outQueueY.DeQue<float>();
            AscendC::DataCopy(yGm[outOffset + oh * outW], outResult, alignedOutW);
            outQueueY.FreeTensor(outResult);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t batchSize, inChannels, inH, inW;
    uint32_t kernelH, kernelW, strideH, strideW;
    uint32_t paddingH, paddingW, dilationH, dilationW;
    uint32_t outH, outW, hasBias;
    uint32_t totalOutputChannels;
    uint32_t inSpatial, outSpatial, kernelSize;
    uint32_t channelsPerBlock, startChannel, endChannel;
    uint32_t alignedOutW;
};

extern "C" __global__ __aicore__ void conv_depthwise_2d_asymmetric_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvDepthwise2d op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels,
            tiling_data.inH, tiling_data.inW,
            tiling_data.kernelH, tiling_data.kernelW,
            tiling_data.strideH, tiling_data.strideW,
            tiling_data.paddingH, tiling_data.paddingW,
            tiling_data.dilationH, tiling_data.dilationW,
            tiling_data.outH, tiling_data.outW,
            tiling_data.hasBias,
            tiling_data.totalOutputChannels, tiling_data.blockDim);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels: int, out_channels: int, kernel_size_h: int, kernel_size_w: int,
                 stride_h: int = 1, stride_w: int = 1, padding_h: int = 0, padding_w: int = 0,
                 dilation_h: int = 1, dilation_w: int = 1, groups: int = 1, bias: bool = False):
        super(ModelNew, self).__init__()
        self.in_channels = in_channels
        self.kernel_size_h = kernel_size_h
        self.kernel_size_w = kernel_size_w
        self.stride_h = stride_h
        self.stride_w = stride_w
        self.padding_h = padding_h
        self.padding_w = padding_w
        self.dilation_h = dilation_h
        self.dilation_w = dilation_w
        self.has_bias = bias

        # Create the same conv2d to get the weight (and bias) parameters
        self.conv2d = torch.nn.Conv2d(in_channels, in_channels, (kernel_size_h, kernel_size_w),
                                       stride=(stride_h, stride_w), padding=(padding_h, padding_w),
                                       dilation=(dilation_h, dilation_w), groups=in_channels, bias=bias)

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        batch_size = x.shape[0]
        in_h = x.shape[2]
        in_w = x.shape[3]
        out_h = (in_h + 2 * self.padding_h - self.dilation_h * (self.kernel_size_h - 1) - 1) // self.stride_h + 1
        out_w = (in_w + 2 * self.padding_w - self.dilation_w * (self.kernel_size_w - 1) - 1) // self.stride_w + 1

        # Weight shape is (in_channels, 1, kH, kW), reshape to (in_channels, kH*kW)
        weight = self.conv2d.weight.reshape(self.in_channels, self.kernel_size_h * self.kernel_size_w)
        bias = self.conv2d.bias if self.has_bias else None
        has_bias_int = 1 if self.has_bias else 0

        return custom_ops_lib.conv_depthwise_2d_asymmetric_input_asymmetric_kernel_custom(
            x.contiguous(), weight.contiguous(), bias,
            batch_size, self.in_channels, in_h, in_w,
            self.kernel_size_h, self.kernel_size_w,
            self.stride_h, self.stride_w,
            self.padding_h, self.padding_w,
            self.dilation_h, self.dilation_w,
            out_h, out_w, has_bias_int)
'''
