    tiling.set_dim1(dim1);
    tiling.set_dim2(dim2);

    auto condShape = context->GetInputShape(0)->GetOriginShape();
    uint32_t condDim0 = condShape.GetDim(0);
    uint32_t condDim1 = condShape.GetDim(1);
    uint32_t condDim2 = condShape.GetDim(2);

    tiling.set_condDim1(condDim1);
    tiling.set_condDim2(condDim2);

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelWhereBroadcast {
public:
    __aicore__ inline KernelWhereBroadcast() {}
    __aicore__ inline void Init(GM_ADDR cond, GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                uint32_t totalLength, uint32_t tileNum,
                                uint32_t dim0, uint32_t dim1, uint32_t dim2,
                                uint32_t condDim0, uint32_t condDim1, uint32_t condDim2)
    {
        this->totalLength = totalLength;
        this->dim0 = dim0;
        this->dim1 = dim1;
        this->dim2 = dim2;
        this->condDim0 = condDim0;
        this->condDim1 = condDim1;
        this->condDim2 = condDim2;
        this->condTotalLength = condDim0 * condDim1 * condDim2;

        // Each block processes a portion of the total elements
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->blockLength = totalLength / blockNum;
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        // Align tileLength to 32 bytes (8 floats)
        if (this->tileLength == 0) {
            this->tileLength = 8;
        }

        this->startOffset = blockIdx * this->blockLength;

        xGm.SetGlobalBuffer((__gm__ float *)x, totalLength);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalLength);
        zGm.SetGlobalBuffer((__gm__ float *)z, totalLength);
        condGm.SetGlobalBuffer((__gm__ int8_t *)cond, condTotalLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
        // For select mask we need uint8 buffer, and a float buffer for ones/zeros
        pipe.InitBuffer(inQueueCond, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        uint32_t offset = this->startOffset + progress * this->tileLength;
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> yLocal = inQueueY.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], this->tileLength);
        AscendC::DataCopy(yLocal, yGm[offset], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = inQueueY.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> condLocal = inQueueCond.AllocTensor<float>();
        AscendC::LocalTensor<float> onesLocal = tmpBuf.AllocTensor<float>();

        uint32_t offset = this->startOffset + progress * this->tileLength;
        uint32_t stride12 = dim1 * dim2;

        // Build a float mask: 1.0 where cond is true, 0.0 where false
        // For each element in the tile, compute the broadcast cond index
        for (uint32_t i = 0; i < this->tileLength; i++) {
            uint32_t globalIdx = offset + i;
            uint32_t i0 = globalIdx / stride12;
            uint32_t rem = globalIdx % stride12;
            uint32_t i1 = rem / dim2;
            uint32_t i2 = rem % dim2;

            // Broadcast: if condDim is 1, use index 0
            uint32_t c0 = (condDim0 == 1) ? 0 : i0;
            uint32_t c1 = (condDim1 == 1) ? 0 : i1;
            uint32_t c2 = (condDim2 == 1) ? 0 : i2;

            uint32_t condIdx = c0 * condDim1 * condDim2 + c1 * condDim2 + c2;
            int8_t condVal = condGm.GetValue(condIdx);
            condLocal.SetValue(i, condVal ? 1.0f : 0.0f);
        }

        // z = cond * x + (1 - cond) * y
        // onesLocal = 1.0 - condLocal
        AscendC::Duplicate(onesLocal, 1.0f, this->tileLength);
        AscendC::Sub(onesLocal, onesLocal, condLocal, this->tileLength);

        // condLocal = condLocal * x
        AscendC::Mul(condLocal, condLocal, xLocal, this->tileLength);
        // onesLocal = onesLocal * y
        AscendC::Mul(onesLocal, onesLocal, yLocal, this->tileLength);
        // z = condLocal + onesLocal
        AscendC::Add(zLocal, condLocal, onesLocal, this->tileLength);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
        inQueueCond.FreeTensor(condLocal);
        tmpBuf.FreeTensor(onesLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        uint32_t offset = this->startOffset + progress * this->tileLength;
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[offset], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueY, inQueueCond;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> zGm;
    AscendC::GlobalTensor<int8_t> condGm;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t startOffset;
    uint32_t dim0, dim1, dim2;
    uint32_t condDim0, condDim1, condDim2;
    uint32_t condTotalLength;
};

extern "C" __global__ __aicore__ void where_broadcast_custom(GM_ADDR cond, GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelWhereBroadcast op;
    op.Init(cond, x, y, z,
            tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.dim0, tiling_data.dim1, tiling_data.dim2,
            tiling_data.condDim0, tiling_data.condDim1, tiling_data.condDim2);
    op.Process();
}
