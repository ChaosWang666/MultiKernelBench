#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelScaleAvgClamp {
public:
    __aicore__ inline KernelScaleAvgClamp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t channels, 
                                 uint32_t spatialSize, float scaleVal, float clampMin, float clampMax, uint32_t tileNum)
    {
        this->totalBC = batchSize * channels;
        this->spatialSize = spatialSize;
        this->scaleVal = scaleVal;
        this->clampMin = clampMin;
        this->clampMax = clampMax;
        this->tileNum = tileNum;
        
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        this->bcPerBlock = (this->totalBC + blockNum - 1) / blockNum;
        this->bcStart = blockIdx * this->bcPerBlock;
        this->bcEnd = this->bcStart + this->bcPerBlock;
        if (this->bcEnd > this->totalBC) {
            this->bcEnd = this->totalBC;
        }
        
        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * channels * spatialSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize * channels);
        
        uint32_t tileLength = (spatialSize + tileNum - 1) / tileNum;
        // align to 32 bytes = 8 floats
        tileLength = (tileLength + 7) & ~7;
        if (tileLength < 8) tileLength = 8;
        this->tileLength = tileLength;
        
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, 8 * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        for (uint32_t bc = this->bcStart; bc < this->bcEnd; bc++) {
            ProcessOneChannel(bc);
        }
    }

private:
    __aicore__ inline void ProcessOneChannel(uint32_t bc)
    {
        float sum = 0.0f;
        uint32_t offset = bc * this->spatialSize;
        uint32_t remaining = this->spatialSize;
        uint32_t pos = 0;
        
        while (remaining > 0) {
            uint32_t curLen = remaining > this->tileLength ? this->tileLength : remaining;
            uint32_t alignedLen = (curLen + 7) & ~7;
            if (alignedLen < 8) alignedLen = 8;
            
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            
            // Zero out the buffer first if alignedLen > curLen
            if (alignedLen > curLen) {
                AscendC::Duplicate(xLocal, 0.0f, alignedLen);
            }
            
            AscendC::DataCopy(xLocal, xGm[offset + pos], alignedLen);
            inQueueX.EnQue(xLocal);
            
            AscendC::LocalTensor<float> xProc = inQueueX.DeQue<float>();
            
            // Scale
            AscendC::Muls(xProc, xProc, this->scaleVal, alignedLen);
            
            // Sum reduction
            AscendC::LocalTensor<float> workLocal = outQueueZ.AllocTensor<float>();
            AscendC::ReduceSum(workLocal, xProc, alignedLen, false);
            float tileSum = workLocal.GetValue(0);
            sum += tileSum;
            outQueueZ.FreeTensor(workLocal);
            
            inQueueX.FreeTensor(xProc);
            
            remaining -= curLen;
            pos += curLen;
        }
        
        // Compute average
        float avg = sum / (float)this->spatialSize;
        
        // Clamp
        if (avg < this->clampMin) avg = this->clampMin;
        if (avg > this->clampMax) avg = this->clampMax;
        
        // Write output
        AscendC::LocalTensor<float> outLocal = outQueueZ.AllocTensor<float>();
        outLocal.SetValue(0, avg);
        for (int i = 1; i < 8; i++) {
            outLocal.SetValue(i, 0.0f);
        }
        outQueueZ.EnQue(outLocal);
        AscendC::LocalTensor<float> outRes = outQueueZ.DeQue<float>();
        // We only need to write 1 float but DataCopy needs alignment
        AscendC::DataCopy(yGm[bc], outRes, 8);
        outQueueZ.FreeTensor(outRes);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t totalBC;
    uint32_t spatialSize;
    float scaleVal;
    float clampMin;
    float clampMax;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t bcPerBlock;
    uint32_t bcStart;
    uint32_t bcEnd;
};

extern "C" __global__ __aicore__ void conv_transpose3d_multiply_max_global_avg_pool_clamp_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);

    op.Init(x, y, tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize,
            tiling_data.scaleVal, tiling_data.clampMin, tiling_data.clampMax, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model that uses conv_transpose3d and maxpool from PyTorch,
    then fuses scale * x, global_avg_pool, and clamp into a custom AscendC kernel.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, scale, maxpool_kernel_size):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.scale = scale
        self.maxpool = torch.nn.MaxPool3d(kernel_size=maxpool_kernel_size)

    def forward(self, x):
        x = self.conv_transpose(x)
        # maxpool needs to happen before our fused kernel
        # But scale is element-wise, so scale then maxpool = maxpool(scale*x)
        # Since scale > 0, max(scale*xi) = scale * max(xi), so we can do maxpool first then scale inside kernel
        x = self.maxpool(x)
        # Custom kernel: scale * x -> global_avg_pool -> clamp
        x = custom_ops_lib.conv_transpose3d_multiply_max_global_avg_pool_clamp_custom(x)
        return x
