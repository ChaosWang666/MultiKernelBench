

Looking at this architecture, I need to analyze the operations:
1. Conv2d (in_channels=8, out_channels=64, kernel_size=3)
2. AvgPool2d(pool_kernel_size=4)
3. Sigmoid
4. Sum over dims [1,2,3]

The conv2d is a complex operation best left to existing optimized libraries. I'll focus on fusing the post-conv operations: AvgPool2d + Sigmoid + Sum into a single custom kernel. This avoids multiple memory round-trips.

After conv2d, the tensor shape is (128, 64, 382, 382). After avg_pool with kernel 4, it becomes (128, 64, 95, 95). Then sigmoid and sum over dims [1,2,3] gives shape (128,).

I'll create a fused kernel for avg_pool + sigmoid + sum that operates on the conv output.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelAvgPoolSigmoidSum {
public:
    __aicore__ inline KernelAvgPoolSigmoidSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t height, uint32_t width,
                                 uint32_t poolKernel, uint32_t pooledH, uint32_t pooledW)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->height = height;
        this->width = width;
        this->poolKernel = poolKernel;
        this->pooledH = pooledH;
        this->pooledW = pooledW;

        uint32_t batchIdx = AscendC::GetBlockIdx();
        uint32_t singleBatchSize = channels * height * width;
        xGm.SetGlobalBuffer((__gm__ float *)x + batchIdx * singleBatchSize, singleBatchSize);
        yGm.SetGlobalBuffer((__gm__ float *)y + batchIdx, batchSize);

        // We process one row of input at a time per channel
        // Tile length = width (one row), aligned to 32 bytes (8 floats)
        uint32_t alignedWidth = (width + 7) / 8 * 8;
        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedWidth * sizeof(float));
        // Buffer for accumulating pooled row results
        uint32_t alignedPooledW = (pooledW + 7) / 8 * 8;
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, alignedPooledW * sizeof(float));
        // Temp buffer for partial sums
        pipe.InitBuffer(tmpBuf, BUFFER_NUM, alignedPooledW * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        float totalSum = 0.0f;
        uint32_t alignedPooledW = (pooledW + 7) / 8 * 8;
        uint32_t alignedWidth = (width + 7) / 8 * 8;

        for (uint32_t c = 0; c < channels; c++) {
            uint32_t channelOffset = c * height * width;

            for (uint32_t ph = 0; ph < pooledH; ph++) {
                // Initialize accumulator for this pooled row
                AscendC::LocalTensor<float> accLocal = outQueueZ.AllocTensor<float>();
                AscendC::Duplicate(accLocal, 0.0f, alignedPooledW);

                // Sum poolKernel rows
                for (uint32_t kr = 0; kr < poolKernel; kr++) {
                    uint32_t srcRow = ph * poolKernel + kr;
                    uint32_t rowOffset = channelOffset + srcRow * width;

                    AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                    AscendC::DataCopy(xLocal, xGm[rowOffset], alignedWidth);
                    inQueueX.EnQue(xLocal);
                    AscendC::LocalTensor<float> xIn = inQueueX.DeQue<float>();

                    // For each pooled column, sum poolKernel elements horizontally
                    // We do this by iterating over the pool columns
                    AscendC::LocalTensor<float> tmpLocal = tmpBuf.AllocTensor<float>();

                    for (uint32_t pw = 0; pw < pooledW; pw++) {
                        float rowPoolSum = 0.0f;
                        uint32_t colStart = pw * poolKernel;
                        for (uint32_t kc = 0; kc < poolKernel; kc++) {
                            rowPoolSum += xIn.GetValue(colStart + kc);
                        }
                        tmpLocal.SetValue(pw, rowPoolSum);
                    }

                    // Add to accumulator
                    AscendC::Add(accLocal, accLocal, tmpLocal, alignedPooledW);

                    tmpBuf.FreeTensor(tmpLocal);
                    inQueueX.FreeTensor(xIn);
                }

                // accLocal now has sum over poolKernel x poolKernel window
                // Divide by poolKernel*poolKernel for average
                float invPoolArea = 1.0f / (float)(poolKernel * poolKernel);
                AscendC::Muls(accLocal, accLocal, invPoolArea, alignedPooledW);

                // Apply sigmoid: 1/(1+exp(-x))
                // Use AscendC Exp and reciprocal
                // First negate
                AscendC::Muls(accLocal, accLocal, -1.0f, alignedPooledW);
                // Exp
                AscendC::Exp(accLocal, accLocal, alignedPooledW);
                // Add 1
                AscendC::Adds(accLocal, accLocal, 1.0f, alignedPooledW);
                // Reciprocal
                AscendC::Reciprocal(accLocal, accLocal, alignedPooledW);

                // Sum all elements
                for (uint32_t pw = 0; pw < pooledW; pw++) {
                    totalSum += accLocal.GetValue(pw);
                }

                outQueueZ.FreeTensor(accLocal);
            }
        }

        // Write result
        // We need to use DataCopy, so use a small buffer
        AscendC::LocalTensor<float> resultLocal = outQueueZ.AllocTensor<float>();
        resultLocal.SetValue(0, totalSum);
        // Pad remaining to avoid issues
        for (uint32_t i = 1; i < 8; i++) {
            resultLocal.SetValue(i, 0.0f);
        }
        outQueueZ.EnQue(resultLocal);
        AscendC::LocalTensor<float> resOut = outQueueZ.DeQue<float>();
        AscendC::DataCopy(yGm[0], resOut, 8);
        outQueueZ.FreeTensor(resOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t height;
    uint32_t width;
    uint32_t poolKernel;
    uint32_t pooledH;
    uint32_t pooledW;
};

extern "C" __global__ __aicore__ void conv2d_avg_pool_sigmoid_sum_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelAvgPoolSigmoidSum op;
    op.Init(x, y,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.height, tiling_data.width,
            tiling_data.poolKernel, tiling_data.pooledH, tiling_data.pooledW);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model: Conv2d kept as-is, then fused AvgPool+Sigmoid+Sum via custom AscendC kernel.
    """
    def __init__(self, in_channels, out_channels, kernel_size, pool_kernel_size):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)

    def forward(self, x):
        x = self.conv(x)
        # Fused avgpool + sigmoid + sum
        x = x.contiguous()
        x = custom_ops_lib.conv2d_avg_pool_sigmoid_sum_custom(x)
        return x
