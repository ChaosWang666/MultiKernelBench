

Looking at the GoogLeNet Inception V1 architecture, I'll identify the key operation that can be optimized with a custom AscendC kernel. The most impactful custom kernel here is the **concatenation of the four inception branches along the channel dimension** (`torch.cat(outputs, 1)`), which is called repeatedly throughout the network.

I'll create a custom kernel that performs the channel-wise concatenation of 4 tensors (the 4 branches of each inception module), which is a memory-bound operation that can benefit from optimized data movement on Ascend hardware.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelCatChannels {
public:
    __aicore__ inline KernelCatChannels() {}
    __aicore__ inline void Init(GM_ADDR branch1, GM_ADDR branch2, GM_ADDR branch3, GM_ADDR branch4,
                                GM_ADDR output,
                                uint32_t batchSize, uint32_t height, uint32_t width,
                                uint32_t c1, uint32_t c2, uint32_t c3, uint32_t c4,
                                uint32_t totalOutput, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->height = height;
        this->width = width;
        this->c1 = c1;
        this->c2 = c2;
        this->c3 = c3;
        this->c4 = c4;
        this->hw = height * width;
        this->cTotal = c1 + c2 + c3 + c4;

        uint32_t totalElements1 = batchSize * c1 * hw;
        uint32_t totalElements2 = batchSize * c2 * hw;
        uint32_t totalElements3 = batchSize * c3 * hw;
        uint32_t totalElements4 = batchSize * c4 * hw;

        branch1Gm.SetGlobalBuffer((__gm__ float *)branch1, totalElements1);
        branch2Gm.SetGlobalBuffer((__gm__ float *)branch2, totalElements2);
        branch3Gm.SetGlobalBuffer((__gm__ float *)branch3, totalElements3);
        branch4Gm.SetGlobalBuffer((__gm__ float *)branch4, totalElements4);
        outputGm.SetGlobalBuffer((__gm__ float *)output, totalOutput);

        uint32_t maxChannels = c1;
        if (c2 > maxChannels) maxChannels = c2;
        if (c3 > maxChannels) maxChannels = c3;
        if (c4 > maxChannels) maxChannels = c4;

        uint32_t bufSize = maxChannels * hw * sizeof(float);
        if (bufSize < 32) bufSize = 32;
        // Align to 32 bytes
        bufSize = (bufSize + 31) / 32 * 32;

        pipe.InitBuffer(inQueue, BUFFER_NUM, bufSize);
        pipe.InitBuffer(outQueue, BUFFER_NUM, bufSize);
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        for (uint32_t b = blockIdx; b < batchSize; b += blockNum) {
            // Copy branch1 for this batch sample
            CopyBranch(branch1Gm, b, c1, 0);
            // Copy branch2
            CopyBranch(branch2Gm, b, c2, c1);
            // Copy branch3
            CopyBranch(branch3Gm, b, c3, c1 + c2);
            // Copy branch4
            CopyBranch(branch4Gm, b, c4, c1 + c2 + c3);
        }
    }

private:
    __aicore__ inline void CopyBranch(AscendC::GlobalTensor<float>& srcGm,
                                       uint32_t batchIdx, uint32_t channels, uint32_t channelOffset)
    {
        uint32_t copyLen = channels * hw;
        uint32_t alignedLen = (copyLen + 7) / 8 * 8; // align to 8 floats = 32 bytes

        AscendC::LocalTensor<float> localIn = inQueue.AllocTensor<float>();
        uint32_t srcOffset = batchIdx * channels * hw;
        AscendC::DataCopy(localIn, srcGm[srcOffset], alignedLen);
        inQueue.EnQue(localIn);

        AscendC::LocalTensor<float> localOut = inQueue.DeQue<float>();
        uint32_t dstOffset = batchIdx * cTotal * hw + channelOffset * hw;
        AscendC::DataCopy(outputGm[dstOffset], localOut, alignedLen);
        inQueue.FreeTensor(localOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::GlobalTensor<float> branch1Gm, branch2Gm, branch3Gm, branch4Gm, outputGm;
    uint32_t batchSize, height, width, hw;
    uint32_t c1, c2, c3, c4, cTotal;
};

extern "C" __global__ __aicore__ void googlenet_inception_v1_custom(
    GM_ADDR branch1, GM_ADDR branch2, GM_ADDR branch3, GM_ADDR branch4,
    GM_ADDR output, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelCatChannels op;
    op.Init(branch1, branch2, branch3, branch4, output,
            tiling_data.batchSize, tiling_data.height, tiling_data.width,
            tiling_data.c1, tiling_data.c2, tiling_data.c3, tiling_data.c4,
            tiling_data.totalOutput, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

class InceptionModuleNew(nn.Module):
    def __init__(self, in_channels, out_1x1, reduce_3x3, out_3x3, reduce_5x5, out_5x5, pool_proj):
        super(InceptionModuleNew, self).__init__()
        
        # 1x1 convolution branch
        self.branch1x1 = nn.Conv2d(in_channels, out_1x1, kernel_size=1)
        
        # 3x3 convolution branch
        self.branch3x3 = nn.Sequential(
            nn.Conv2d(in_channels, reduce_3x3, kernel_size=1),
            nn.Conv2d(reduce_3x3, out_3x3, kernel_size=3, padding=1)
        )
        
        # 5x5 convolution branch
        self.branch5x5 = nn.Sequential(
            nn.Conv2d(in_channels, reduce_5x5, kernel_size=1),
            nn.Conv2d(reduce_5x5, out_5x5, kernel_size=5, padding=2)
        )
        
        # Max pooling branch
        self.branch_pool = nn.Sequential(
            nn.MaxPool2d(kernel_size=3, stride=1, padding=1),
            nn.Conv2d(in_channels, pool_proj, kernel_size=1)
        )
    
    def forward(self, x):
        branch1x1 = self.branch1x1(x)
        branch3x3 = self.branch3x3(x)
        branch5x5 = self.branch5x5(x)
        branch_pool = self.branch_pool(x)
        
        return custom_ops_lib.googlenet_inception_v1_custom(branch1x1, branch3x3, branch5x5, branch_pool)


class ModelNew(nn.Module):
    def __init__(self, num_classes=1000):
        super(ModelNew, self).__init__()
        
        self.conv1 = nn.Conv2d(3, 64, kernel_size=7, stride=2, padding=3)
        self.maxpool1 = nn.MaxPool2d(3, stride=2, padding=1)
        self.conv2 = nn.Conv2d(64, 64, kernel_size=1)
        self.conv3 = nn.Conv2d(64, 192, kernel_size=3, padding=1)
        self.maxpool2 = nn.MaxPool2d(3, stride=2, padding=1)
        
        self.inception3a = InceptionModuleNew(192, 64, 96, 128, 16, 32, 32)
        self.inception3b = InceptionModuleNew(256, 128, 128, 192, 32, 96, 64)
        self.maxpool3 = nn.MaxPool2d(3, stride=2, padding=1)
        
        self.inception4a = InceptionModuleNew(480, 192, 96, 208, 16, 48, 64)
        self.inception4b = InceptionModuleNew(512, 160, 112, 224, 24, 64, 64)
        self.inception4c = InceptionModuleNew(512, 128, 128, 256, 24, 64, 64)
        self.inception4d = InceptionModuleNew(512, 112, 144, 288, 32, 64, 64)
        self.inception4e = InceptionModuleNew(528, 256, 160, 320, 32, 128, 128)
        self.maxpool4 = nn.MaxPool2d(3, stride=2, padding=1)
        
        self.inception5a = InceptionModuleNew(832, 256, 160, 320, 32, 128, 128)
        self.inception5b = InceptionModuleNew(832, 384, 192, 384, 48, 128, 128)
        
        self.avgpool = nn.AdaptiveAvgPool2d((1, 1))
        self.dropout = nn.Dropout(0.0)
        self.fc = nn.Linear(1024, num_classes)
    
    def forward(self, x):
        x = self.maxpool1(F.relu(self.conv1(x)))
        x = F.relu(self.conv2(x))
        x = self.maxpool2(F.relu(self.conv3(x)))
        
        x = self.inception3a(x)
        x = self.inception3b(x)
        x = self.maxpool3(x)
        
        x = self.inception4a(x)
        x = self.inception4b(x)
        x = self.inception4c(x)
        x = self.inception4d(x)
        x = self.inception4e(x)
        x = self.maxpool4(x)
        
        x = self.inception5a(x)
        x = self.inception5b(x)
        
        x = self.avgpool(x)
        x = torch.flatten(x, 1)
        x = self.dropout(x)
        x = self.fc(x)
        
        return x
'''
