

Looking at this architecture, I need to implement a GRU (Gated Recurrent Unit) operation using custom AscendC kernels. The GRU involves complex operations including matrix multiplications and element-wise operations for gates (reset, update, new gates).

Given the complexity of a full GRU implementation in AscendC (which involves matrix multiplications across multiple layers and timesteps), I'll focus on optimizing the core GRU hidden state computation - the element-wise gate operations that happen at each timestep after the linear projections. The key computation is:

- r_t = sigmoid(W_ir @ x_t + b_ir + W_hr @ h_{t-1} + b_hr)  (reset gate)
- z_t = sigmoid(W_iz @ x_t + b_iz + W_hz @ h_{t-1} + b_hz)  (update gate)  
- n_t = tanh(W_in @ x_t + b_in + r_t * (W_hn @ h_{t-1} + b_hn))  (new gate)
- h_t = (1 - z_t) * n_t + z_t * h_{t-1}

I'll implement the element-wise hidden state update kernel (`gru_hidden_custom`) that computes h_t from the gate intermediates, while using PyTorch for the linear projections.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGruHidden {
public:
    __aicore__ inline KernelGruHidden() {}
    __aicore__ inline void Init(GM_ADDR gates_x, GM_ADDR gates_h, GM_ADDR h_prev, GM_ADDR h_new,
                                 uint32_t batchSize, uint32_t hiddenSize, uint32_t totalLength, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->hiddenSize = hiddenSize;
        this->totalHidden = totalLength;
        uint32_t threeH = 3 * hiddenSize;
        uint32_t totalGates = batchSize * threeH;

        this->blockLength = this->totalHidden / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        if (this->tileLength == 0) {
            this->tileLength = this->blockLength;
            this->tileNum = 1;
        }

        uint32_t blockOffset = this->blockLength * AscendC::GetBlockIdx();

        // gates_x shape: [batch, 3*hidden], gates_h shape: [batch, 3*hidden]
        // We'll treat them as flat arrays and index appropriately
        // For the hidden computation, we need r,z,n gates
        // gates_x: [r_x | z_x | n_x] for each batch element
        // gates_h: [r_h | z_h | n_h] for each batch element

        gatesXGm.SetGlobalBuffer((__gm__ float *)gates_x, totalGates);
        gatesHGm.SetGlobalBuffer((__gm__ float *)gates_h, totalGates);
        hPrevGm.SetGlobalBuffer((__gm__ float *)h_prev + blockOffset, this->blockLength);
        hNewGm.SetGlobalBuffer((__gm__ float *)h_new + blockOffset, this->blockLength);

        pipe.InitBuffer(inQueueRX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueZX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueNX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueRH, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueZH, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueNH, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueH, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf1, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyGateData(int32_t progress, int gateIdx,
                                         AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM>& queue,
                                         AscendC::GlobalTensor<float>& gatesGm)
    {
        AscendC::LocalTensor<float> local = queue.AllocTensor<float>();
        uint32_t blockOffset = this->blockLength * AscendC::GetBlockIdx();
        uint32_t startElem = blockOffset + progress * this->tileLength;

        // We need to gather elements from gate layout [batch, 3*hidden]
        // For element i in the flat hidden array, batch_idx = i / hiddenSize, h_idx = i % hiddenSize
        // Gate offset in gates array = batch_idx * 3 * hiddenSize + gateIdx * hiddenSize + h_idx
        for (uint32_t j = 0; j < this->tileLength; j++) {
            uint32_t flatIdx = startElem + j;
            uint32_t bIdx = flatIdx / this->hiddenSize;
            uint32_t hIdx = flatIdx % this->hiddenSize;
            uint32_t gateOffset = bIdx * 3 * this->hiddenSize + gateIdx * this->hiddenSize + hIdx;
            // We can't do element-wise copy efficiently, so use DataCopy with stride if possible
            // For simplicity, we'll handle it with a block copy approach
        }
        // Actually, let's use a simpler approach: copy contiguous blocks
        // Since tileLength aligns with hiddenSize multiples (when blockLength divides evenly),
        // we can copy gate data as contiguous chunks

        // For each element in tile, compute source index
        // This is complex for scattered access. Let's restructure.
        // We'll copy the full gate slice for the tile's batch elements
        
        // Simpler: if tileLength is a multiple of hiddenSize, each tile covers whole batch rows
        // In that case, for gate g, source = batch_start * 3*H + g*H, length = H, stride = 3*H
        
        // Use DataCopy with stride
        uint32_t numRows = this->tileLength / this->hiddenSize;
        if (numRows > 0 && (this->tileLength % this->hiddenSize == 0)) {
            uint32_t startBatch = startElem / this->hiddenSize;
            for (uint32_t r = 0; r < numRows; r++) {
                uint32_t srcOff = (startBatch + r) * 3 * this->hiddenSize + gateIdx * this->hiddenSize;
                AscendC::DataCopy(local[r * this->hiddenSize], gatesGm[srcOff], this->hiddenSize);
            }
        } else {
            // fallback: element by element using DataCopy of minimum block
            for (uint32_t j = 0; j < this->tileLength; j += 8) {
                uint32_t flatIdx = startElem + j;
                uint32_t bIdx = flatIdx / this->hiddenSize;
                uint32_t hIdx = flatIdx % this->hiddenSize;
                uint32_t gateOffset = bIdx * 3 * this->hiddenSize + gateIdx * this->hiddenSize + hIdx;
                uint32_t copyLen = 8;
                if (j + copyLen > this->tileLength) copyLen = this->tileLength - j;
                if (hIdx + copyLen > this->hiddenSize) copyLen = this->hiddenSize - hIdx;
                if (copyLen < 8) copyLen = 8;
                AscendC::DataCopy(local[j], gatesGm[gateOffset], copyLen);
            }
        }
        queue.EnQue(local);
    }

    __aicore__ inline void CopyIn(int32_t progress)
    {
        // Copy h_prev
        AscendC::LocalTensor<float> hLocal = inQueueH.AllocTensor<float>();
        AscendC::DataCopy(hLocal, hPrevGm[progress * this->tileLength], this->tileLength);
        inQueueH.EnQue(hLocal);

        // Copy gate data for r, z, n from both gates_x and gates_h
        CopyGateData(progress, 0, inQueueRX, gatesXGm);
        CopyGateData(progress, 1, inQueueZX, gatesXGm);
        CopyGateData(progress, 2, inQueueNX, gatesXGm);
        CopyGateData(progress, 0, inQueueRH, gatesHGm);
        CopyGateData(progress, 1, inQueueZH, gatesHGm);
        CopyGateData(progress, 2, inQueueNH, gatesHGm);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> rxLocal = inQueueRX.DeQue<float>();
        AscendC::LocalTensor<float> zxLocal = inQueueZX.DeQue<float>();
        AscendC::LocalTensor<float> nxLocal = inQueueNX.DeQue<float>();
        AscendC::LocalTensor<float> rhLocal = inQueueRH.DeQue<float>();
        AscendC::LocalTensor<float> zhLocal = inQueueZH.DeQue<float>();
        AscendC::LocalTensor<float> nhLocal = inQueueNH.DeQue<float>();
        AscendC::LocalTensor<float> hLocal = inQueueH.DeQue<float>();
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> t1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> t2 = tmpBuf2.Get<float>();

        // r = sigmoid(rx + rh)
        AscendC::Add(rxLocal, rxLocal, rhLocal, this->tileLength);
        AscendC::Sigmoid(rxLocal, rxLocal, this->tileLength);

        // z = sigmoid(zx + zh)
        AscendC::Add(zxLocal, zxLocal, zhLocal, this->tileLength);
        AscendC::Sigmoid(zxLocal, zxLocal, this->tileLength);

        // n = tanh(nx + r * nh)
        AscendC::Mul(nhLocal, rxLocal, nhLocal, this->tileLength);
        AscendC::Add(nxLocal, nxLocal, nhLocal, this->tileLength);
        AscendC::Tanh(nxLocal, nxLocal, this->tileLength);

        // h_new = (1 - z) * n + z * h_prev
        // = n - z*n + z*h = n + z*(h - n)
        AscendC::Sub(t1, hLocal, nxLocal, this->tileLength);
        AscendC::Mul(t1, zxLocal, t1, this->tileLength);
        AscendC::Add(outLocal, nxLocal, t1, this->tileLength);

        outQueue.EnQue(outLocal);
        inQueueRX.FreeTensor(rxLocal);
        inQueueZX.FreeTensor(zxLocal);
        inQueueNX.FreeTensor(nxLocal);
        inQueueRH.FreeTensor(rhLocal);
        inQueueZH.FreeTensor(zhLocal);
        inQueueNH.FreeTensor(nhLocal);
        inQueueH.FreeTensor(hLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> outLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(hNewGm[progress * this->tileLength], outLocal, this->tileLength);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueRX, inQueueZX, inQueueNX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueRH, inQueueZH, inQueueNH;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueH;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> gatesXGm, gatesHGm, hPrevGm, hNewGm;
    uint32_t batchSize;
    uint32_t hiddenSize;
    uint32_t totalHidden;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void gru_hidden_custom(GM_ADDR gates_x, GM_ADDR gates_h, GM_ADDR h_prev, GM_ADDR h_new, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGruHidden op;
    op.Init(gates_x, gates_h, h_prev, h_new, tiling_data.batchSize, tiling_data.hiddenSize, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, input_size, hidden_size, num_layers=3, bias=True, batch_first=False):
        """
        GRU implementation that uses custom AscendC kernel for hidden state computation.
        """
        super(ModelNew, self).__init__()
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.num_layers = num_layers
        self.bias = bias
        self.batch_first = batch_first

        # Use the standard GRU to hold weights - we'll extract them for manual computation
        self.gru = nn.GRU(input_size, hidden_size, num_layers, bias, batch_first, dropout=0, bidirectional=False)

    def forward(self, x, h0):
        if self.batch_first:
            x = x.transpose(0, 1)  # (batch, seq, feat) -> (seq, batch, feat)

        seq_len = x.size(0)
        batch_size = x.size(1)

        h = h0.clone()

        for t in range(seq_len):
            inp = x[t]  # (batch, input_size) or (batch, hidden_size) for layers > 0
            for layer in range(self.num_layers):
                # Get weights for this layer
                w_ih = getattr(self.gru, f'weight_ih_l{layer}')  # (3*hidden, input_size or hidden_size)
                w_hh = getattr(self.gru, f'weight_hh_l{layer}')  # (3*hidden, hidden_size)

                h_prev = h[layer]  # (batch, hidden_size)

                # Compute gates: linear projections
                gates_x = torch.mm(inp, w_ih.t())  # (batch, 3*hidden)
                gates_h = torch.mm(h_prev, w_hh.t())  # (batch, 3*hidden)

                if self.bias:
                    b_ih = getattr(self.gru, f'bias_ih_l{layer}')
                    b_hh = getattr(self.gru, f'bias_hh_l{layer}')
                    gates_x = gates_x + b_ih
                    gates_h = gates_h + b_hh

                # Use custom kernel for hidden state update
                gates_x_contig = gates_x.contiguous()
                gates_h_contig = gates_h.contiguous()
                h_prev_contig = h_prev.contiguous()

                h_new = custom_ops_lib.gru_hidden_custom(gates_x_contig, gates_h_contig, h_prev_contig)
                h[layer] = h_new
                inp = h_new  # input to next layer

        return h
'''
