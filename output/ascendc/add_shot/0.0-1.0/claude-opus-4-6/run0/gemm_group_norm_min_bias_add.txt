#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelGemmGroupNormMinBiasAdd {
public:
    __aicore__ inline KernelGemmGroupNormMinBiasAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR gemmBias, GM_ADDR gnGamma, GM_ADDR gnBeta, GM_ADDR bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t outFeatures, uint32_t numGroups, uint32_t channelsPerGroup, uint32_t biasLength)
    {
        this->batchSize = batchSize;
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->channelsPerGroup = channelsPerGroup;
        this->biasLength = biasLength;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        if (this->startRow + this->rowsPerBlock > batchSize) {
            this->rowsPerBlock = batchSize - this->startRow;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * outFeatures);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, outFeatures * outFeatures);
        gemmBiasGm.SetGlobalBuffer((__gm__ float *)gemmBias, outFeatures);
        gnGammaGm.SetGlobalBuffer((__gm__ float *)gnGamma, outFeatures);
        gnBetaGm.SetGlobalBuffer((__gm__ float *)gnBeta, outFeatures);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, biasLength);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * biasLength);

        uint32_t alignedFeatures = ((outFeatures + 7) / 8) * 8;
        uint32_t alignedCPG = ((channelsPerGroup + 7) / 8) * 8;

        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedFeatures * sizeof(float));
        pipe.InitBuffer(inQueueY, BUFFER_NUM, alignedFeatures * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, alignedFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf1, BUFFER_NUM, alignedFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf2, BUFFER_NUM, alignedCPG * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsPerBlock; i++) {
            uint32_t row = this->startRow + i;
            if (row >= this->batchSize) break;
            ProcessRow(row);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t row)
    {
        uint32_t alignedFeatures = ((outFeatures + 7) / 8) * 8;
        uint32_t alignedCPG = ((channelsPerGroup + 7) / 8) * 8;

        // Load the GEMM result row (x has already been through linear in Python, 
        // but we receive intermediate x here - actually we receive the original x
        // and need to do GEMM. However, GEMM is complex on AscendC, so we'll handle
        // the post-GEMM part: GroupNorm + Min + BiasAdd)
        // Actually: we receive x which is already the GEMM output from Python side.
        // Let's just load x row.
        
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * outFeatures], alignedFeatures);
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        // Load gamma and beta
        AscendC::LocalTensor<float> gammaLocal = inQueueY.AllocTensor<float>();
        AscendC::DataCopy(gammaLocal, gnGammaGm[0], alignedFeatures);
        inQueueY.EnQue(gammaLocal);
        gammaLocal = inQueueY.DeQue<float>();

        AscendC::LocalTensor<float> betaLocal = outQueueZ.AllocTensor<float>();
        AscendC::DataCopy(betaLocal, gnBetaGm[0], alignedFeatures);
        outQueueZ.EnQue(betaLocal);
        betaLocal = outQueueZ.DeQue<float>();

        AscendC::LocalTensor<float> normLocal = tmpBuf1.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf2.AllocTensor<float>();

        // GroupNorm: for each group, compute mean and variance, normalize, scale+shift
        float globalMin = 1e30f;
        
        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t offset = g * channelsPerGroup;
            
            // Compute mean of this group
            float sum = 0.0f;
            for (uint32_t c = 0; c < channelsPerGroup; c++) {
                sum += xLocal.GetValue(offset + c);
            }
            float mean = sum / (float)channelsPerGroup;
            
            // Compute variance
            float varSum = 0.0f;
            for (uint32_t c = 0; c < channelsPerGroup; c++) {
                float diff = xLocal.GetValue(offset + c) - mean;
                varSum += diff * diff;
            }
            float var = varSum / (float)channelsPerGroup;
            float invStd = 1.0f / sqrtf(var + 1e-5f);
            
            // Normalize, scale, shift, and track min
            for (uint32_t c = 0; c < channelsPerGroup; c++) {
                float val = (xLocal.GetValue(offset + c) - mean) * invStd;
                val = val * gammaLocal.GetValue(offset + c) + betaLocal.GetValue(offset + c);
                normLocal.SetValue(offset + c, val);
                if (val < globalMin) {
                    globalMin = val;
                }
            }
        }

        tmpBuf2.FreeTensor(tmpLocal);
        tmpBuf1.FreeTensor(normLocal);
        outQueueZ.FreeTensor(betaLocal);
        inQueueY.FreeTensor(gammaLocal);
        inQueueX.FreeTensor(xLocal);

        // Now write output: globalMin + bias for each element
        // Output shape per row: biasLength elements
        // bias shape is (1, outFeatures, 1, 1) => biasLength = outFeatures
        // Output per row is outFeatures values: globalMin + bias[i]
        
        AscendC::LocalTensor<float> biasLocal = inQueueX.AllocTensor<float>();
        uint32_t alignedBias = ((biasLength + 7) / 8) * 8;
        if (alignedBias <= alignedFeatures) {
            AscendC::DataCopy(biasLocal, biasGm[0], alignedBias);
            inQueueX.EnQue(biasLocal);
            biasLocal = inQueueX.DeQue<float>();
            
            AscendC::LocalTensor<float> outLocal = outQueueZ.AllocTensor<float>();
            // Set all to globalMin
            AscendC::Duplicate(outLocal, globalMin, alignedBias);
            // Add bias
            AscendC::Add(outLocal, outLocal, biasLocal, alignedBias);
            outQueueZ.EnQue(outLocal);
            outLocal = outQueueZ.DeQue<float>();
            
            AscendC::DataCopy(zGm[row * biasLength], outLocal, alignedBias);
            
            outQueueZ.FreeTensor(outLocal);
            inQueueX.FreeTensor(biasLocal);
        } else {
            inQueueX.FreeTensor(biasLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> gemmBiasGm;
    AscendC::GlobalTensor<float> gnGammaGm;
    AscendC::GlobalTensor<float> gnBetaGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t outFeatures;
    uint32_t numGroups;
    uint32_t channelsPerGroup;
    uint32_t biasLength;
    uint32_t rowsPerBlock;
    uint32_t startRow;
};

extern "C" __global__ __aicore__ void gemm_group_norm_min_bias_add_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR gemmBias, GM_ADDR gnGamma, GM_ADDR gnBeta, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormMinBiasAdd op;
    op.Init(x, weight, gemmBias, gnGamma, gnBeta, bias, z,
            tiling_data.batchSize, tiling_data.outFeatures, tiling_data.numGroups, tiling_data.channelsPerGroup, tiling_data.biasLength);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model that performs GEMM, then uses custom AscendC kernel for GroupNorm+Min+BiasAdd.
    """
    def __init__(self, in_features, out_features, num_groups, bias_shape):
        super(ModelNew, self).__init__()
        self.gemm = torch.nn.Linear(in_features, out_features)
        self.group_norm = torch.nn.GroupNorm(num_groups, out_features)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        # Use built-in GEMM (highly optimized already)
        x = self.gemm(x)
        # Use custom fused kernel for GroupNorm + Min + BiasAdd
        result = custom_ops_lib.gemm_group_norm_min_bias_add_custom(
            x.contiguous(),
            self.gemm.weight.contiguous(),
            self.gemm.bias.contiguous(),
            self.group_norm.weight.contiguous(),
            self.group_norm.bias.contiguous(),
            self.bias.contiguous()
        )
