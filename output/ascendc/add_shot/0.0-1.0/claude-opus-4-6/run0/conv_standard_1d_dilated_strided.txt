kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv1d {
public:
    __aicore__ inline KernelConv1d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR y,
                                uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                uint32_t inputLength, uint32_t outputLength,
                                uint32_t kernelSize, uint32_t stride, uint32_t dilation,
                                uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inputLength = inputLength;
        this->outputLength = outputLength;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->dilation = dilation;

        // Total work items: batch * outChannels * outputLength
        // Each block processes a portion of the total work
        uint32_t totalOutElements = batchSize * outChannels * outputLength;
        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        this->blockStart = (uint64_t)blockIdx * totalOutElements / numBlocks;
        this->blockEnd = (uint64_t)(blockIdx + 1) * totalOutElements / numBlocks;
        this->blockLength = (uint32_t)(this->blockEnd - this->blockStart);

        // Align blockLength to 8 for vectorized operations
        // We'll process element by element for the convolution reduction

        xGm.SetGlobalBuffer((__gm__ float *)x, (uint64_t)batchSize * inChannels * inputLength);
        wGm.SetGlobalBuffer((__gm__ float *)weight, (uint64_t)outChannels * inChannels * kernelSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, (uint64_t)batchSize * outChannels * outputLength);

        // We'll tile the output elements
        this->tileNum = tileNum;
        if (this->blockLength == 0) {
            this->tileLength = 0;
            this->lastTileLength = 0;
        } else {
            // Align tile length to 8
            uint32_t rawTile = this->blockLength / this->tileNum;
            this->tileLength = (rawTile / 8) * 8;
            if (this->tileLength == 0) this->tileLength = 8;
            // Recalculate tileNum
            this->tileNum = this->blockLength / this->tileLength;
            this->lastTileLength = this->blockLength - this->tileNum * this->tileLength;
            if (this->lastTileLength > 0) {
                // Round up last tile to multiple of 8
                this->lastTileLength = ((this->lastTileLength + 7) / 8) * 8;
            }
        }

        uint32_t maxTile = this->tileLength > this->lastTileLength ? this->tileLength : this->lastTileLength;
        if (maxTile == 0) maxTile = 8;

        // We need buffers for output tile and a temp buffer for accumulation
        pipe.InitBuffer(outQueue, BUFFER_NUM, maxTile * sizeof(float));
        // Buffer for loading a segment of input for one (batch, ic) pair
        // We need kernelSize elements per output element, spread with dilation
        // For a tile of tileLength output elements, we need input from
        // position startPos*stride to startPos*stride + (tileLength-1)*stride + (kernelSize-1)*dilation
        // Max input segment length for one channel:
        uint32_t maxInputSegLen = (maxTile - 1) * stride + (kernelSize - 1) * dilation + 1;
        maxInputSegLen = ((maxInputSegLen + 7) / 8) * 8;
        pipe.InitBuffer(inQueue, BUFFER_NUM, maxInputSegLen * sizeof(float));
        // Temp buffer for intermediate multiply results
        pipe.InitBuffer(tmpBuf, 1, maxTile * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->blockLength == 0) return;

        uint32_t numFullTiles = this->tileNum;
        for (uint32_t t = 0; t < numFullTiles; t++) {
            ProcessTile(t, this->tileLength);
        }
        if (this->lastTileLength > 0) {
            ProcessTile(numFullTiles, this->lastTileLength);
        }
    }

private:
    __aicore__ inline void ProcessTile(uint32_t tileIdx, uint32_t curTileLength)
    {
        // Determine the global output indices for this tile
        uint64_t globalStart = this->blockStart + (uint64_t)tileIdx * this->tileLength;
        uint32_t actualCount;
        if (tileIdx < this->tileNum) {
            actualCount = this->tileLength;
        } else {
            actualCount = this->blockLength - this->tileNum * this->tileLength;
        }

        // Allocate output tensor and zero it
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        // Zero the output
        AscendC::Duplicate(outLocal, (float)0.0f, curTileLength);

        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        // For each output element in [globalStart, globalStart+actualCount),
        // compute: y[b][oc][ol] = sum_{ic,k} x[b][ic][ol*stride + k*dilation] * w[oc][ic][k]
        // The output elements in this tile may span multiple (b, oc) combinations.
        // We need to handle this carefully.

        // Iterate over the output elements and figure out (b, oc, ol) ranges
        // For efficiency, we try to find contiguous runs of ol for same (b, oc)

        uint64_t pos = globalStart;
        uint32_t outOffset = 0;
        uint32_t ocLen = this->outputLength;
        uint32_t bocLen = (uint64_t)this->outChannels * ocLen;

        while (outOffset < actualCount) {
            uint64_t curPos = pos + outOffset;
            uint32_t b = (uint32_t)(curPos / bocLen);
            uint32_t rem = (uint32_t)(curPos - (uint64_t)b * bocLen);
            uint32_t oc = rem / ocLen;
            uint32_t ol = rem % ocLen;

            // How many contiguous ol's from this (b, oc)?
            uint32_t runLen = ocLen - ol;
            if (runLen > actualCount - outOffset) runLen = actualCount - outOffset;

            // Compute conv for this run: y[b][oc][ol..ol+runLen-1]
            // = sum_{ic, k} x[b][ic][(ol+i)*stride + k*dilation] * w[oc][ic][k]

            // Zero the portion of outLocal for this run
            // (already zeroed above for the full tile, but we need to re-zero for accumulation)
            // Actually we zeroed the whole tile above. But if we're accumulating across ic and k,
            // the initial zero suffices for the first (b,oc) run. For subsequent runs in the same
            // tile, we need to zero again.
            if (outOffset > 0 || true) {
                // Always safe to zero the run
                // But we zeroed the whole tile... Let's just be safe:
                // We'll accumulate into outLocal starting at outOffset
            }

            // Zero the run region
            AscendC::Duplicate(tmpLocal, (float)0.0f, ((runLen + 7) / 8) * 8);

            for (uint32_t ic = 0; ic < this->inChannels; ic++) {
                for (uint32_t k = 0; k < this->kernelSize; k++) {
                    // Load weight w[oc][ic][k]
                    uint64_t wIdx = (uint64_t)oc * this->inChannels * this->kernelSize + (uint64_t)ic * this->kernelSize + k;
                    float wVal = *(((__gm__ float*)wGm.GetPhyAddr()) + wIdx);

                    // Load input segment x[b][ic][ol*stride + k*dilation .. (ol+runLen-1)*stride + k*dilation]
                    // These are at positions: ol*stride + k*dilation + i*stride for i in [0, runLen)
                    // Start: ol*stride + k*dilation
                    // End: (ol+runLen-1)*stride + k*dilation
                    // We need to gather with stride, or load contiguous and then gather

                    uint64_t xBase = (uint64_t)b * this->inChannels * this->inputLength + (uint64_t)ic * this->inputLength;
                    uint32_t startIn = ol * this->stride + k * this->dilation;

                    if (this->stride == 1) {
                        // Contiguous load
                        uint32_t loadLen = runLen;
                        uint32_t loadLenAligned = ((loadLen + 7) / 8) * 8;
                        AscendC::LocalTensor<float> inLocal = inQueue.AllocTensor<float>();
                        AscendC::DataCopy(inLocal, xGm[xBase + startIn], loadLenAligned);
                        inQueue.EnQue(inLocal);
                        AscendC::LocalTensor<float> inLocal2 = inQueue.DeQue<float>();
                        // Multiply by wVal and add to tmpLocal
                        AscendC::Muls(inLocal2, inLocal2, wVal, loadLenAligned);
                        AscendC::Add(tmpLocal, tmpLocal, inLocal2, loadLenAligned);
                        inQueue.FreeTensor(inLocal2);
                    } else {
                        // Strided access: we need elements at startIn, startIn+stride, ..., startIn+(runLen-1)*stride
                        // Load the full range and gather, or do scalar
                        uint32_t totalSpan = (runLen - 1) * this->stride + 1;
                        uint32_t totalSpanAligned = ((totalSpan + 7) / 8) * 8;
                        
                        AscendC::LocalTensor<float> inLocal = inQueue.AllocTensor<float>();
                        AscendC::DataCopy(inLocal, xGm[xBase + startIn], totalSpanAligned);
                        inQueue.EnQue(inLocal);
                        AscendC::LocalTensor<float> inLocal2 = inQueue.DeQue<float>();

                        // Gather strided elements - do scalar extraction
                        uint32_t runLenAligned = ((runLen + 7) / 8) * 8;
                        AscendC::LocalTensor<float> gatherLocal = outQueue.AllocTensor<float>();
                        for (uint32_t i = 0; i < runLen; i++) {
                            float val = inLocal2.GetValue(i * this->stride);
                            gatherLocal.SetValue(i, val);
                        }
                        for (uint32_t i = runLen; i < runLenAligned; i++) {
                            gatherLocal.SetValue(i, 0.0f);
                        }
                        // Multiply by wVal and accumulate
                        AscendC::Muls(gatherLocal, gatherLocal, wVal, runLenAligned);
                        AscendC::Add(tmpLocal, tmpLocal, gatherLocal, runLenAligned);

                        outQueue.FreeTensor(gatherLocal);
                        inQueue.FreeTensor(inLocal2);
                    }
                }
            }

            // Copy tmpLocal to outLocal at the right offset
            uint32_t runLenAligned = ((runLen + 7) / 8) * 8;
            for (uint32_t i = 0; i < runLen; i++) {
                outLocal.SetValue(outOffset + i, tmpLocal.GetValue(i));
            }

            outOffset += runLen;
        }

        // Write output
        uint32_t writeLen = ((actualCount + 7) / 8) * 8;
        outQueue.EnQue(outLocal);
        AscendC::LocalTensor<float> outLocal2 = outQueue.DeQue<float>();
        AscendC::DataCopy(yGm[globalStart], outLocal2, writeLen);
        outQueue.FreeTensor(outLocal2);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inputLength;
    uint32_t outputLength;
    uint32_t kernelSize;
    uint32_t stride;
    uint32_t dilation;
    uint64_t blockStart;
    uint64_t blockEnd;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t lastTileLength;
};

extern "C" __global__ __aicore__ void conv_standard_1d_dilated_strided_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv1d op;
    op.Init(x, weight, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inputLength, tiling_data.outputLength,
            tiling_data.kernelSize, tiling_data.stride, tiling_data.dilation,
            tiling_data.tileNum);
    op.Process();
}
