

Looking at this architecture, I need to implement a depthwise 2D convolution with asymmetric input and square kernel as a custom AscendC kernel. This is a depthwise convolution where `groups=in_channels`, meaning each input channel is convolved with its own filter.

Let me analyze the parameters:
- batch_size=64, in_channels=128, out_channels=128 (must equal in_channels for depthwise)
- kernel_size=3, stride=1, padding=0
- Input: (64, 128, 256, 512), Output: (64, 128, 254, 510)

kernel_src="""
#include "kernel_operator.h"

using namespace AscendC;

class KernelConvDw {
public:
    __aicore__ inline KernelConvDw() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR y,
                                uint32_t batchSize, uint32_t channels,
                                uint32_t heightIn, uint32_t widthIn,
                                uint32_t kernelSize, uint32_t stride,
                                uint32_t padding, uint32_t heightOut,
                                uint32_t widthOut, uint32_t totalTasks)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->heightIn = heightIn;
        this->widthIn = widthIn;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->heightOut = heightOut;
        this->widthOut = widthOut;
        this->totalTasks = totalTasks;

        uint32_t blockNum = GetBlockNum();
        uint32_t blockIdx = GetBlockIdx();

        this->tasksPerBlock = (totalTasks + blockNum - 1) / blockNum;
        this->taskStart = blockIdx * this->tasksPerBlock;
        this->taskEnd = taskStart + tasksPerBlock;
        if (this->taskEnd > totalTasks) this->taskEnd = totalTasks;

        xGm.SetGlobalBuffer((__gm__ float *)x, (uint64_t)batchSize * channels * heightIn * widthIn);
        wGm.SetGlobalBuffer((__gm__ float *)weight, (uint64_t)channels * 1 * kernelSize * kernelSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, (uint64_t)batchSize * channels * heightOut * widthOut);

        // Align widthOut to 8 for vectorized operations
        this->widthOutAligned = ((widthOut + 7) / 8) * 8;

        // Buffer sizes
        uint32_t inputRowBytes = widthIn * sizeof(float);
        uint32_t outputRowBytes = widthOutAligned * sizeof(float);
        uint32_t kernelBytes = ((kernelSize * kernelSize + 7) / 8) * 8 * sizeof(float);

        // We process one output row at a time
        // Need kernelSize input rows + 1 output row + kernel weights
        pipe.InitBuffer(inQueueX, 1, kernelSize * ((widthIn * sizeof(float) + 31) / 32) * 32);
        pipe.InitBuffer(outQueueY, 1, (widthOutAligned * sizeof(float) + 31) / 32 * 32);
        pipe.InitBuffer(inQueueW, 1, kernelBytes > 32 ? kernelBytes : 32);
        pipe.InitBuffer(tmpBuf, 1, (widthOutAligned * sizeof(float) + 31) / 32 * 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t task = taskStart; task < taskEnd; task++) {
            uint32_t b = task / channels;
            uint32_t c = task % channels;
            ProcessOneChannel(b, c);
        }
    }

private:
    __aicore__ inline void ProcessOneChannel(uint32_t b, uint32_t c)
    {
        uint64_t inputOffset = ((uint64_t)b * channels + c) * heightIn * widthIn;
        uint64_t outputOffset = ((uint64_t)b * channels + c) * heightOut * widthOut;
        uint64_t weightOffset = (uint64_t)c * kernelSize * kernelSize;

        // Load kernel weights
        uint32_t kElems = kernelSize * kernelSize;
        uint32_t kElemsAligned = ((kElems + 7) / 8) * 8;
        LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
        // Initialize to zero
        Duplicate(wLocal, (float)0, kElemsAligned);
        DataCopy(wLocal, wGm[weightOffset], ((kElems + 7) / 8) * 8 <= kElemsAligned ? ((kElems + 7) / 8) * 8 : kElemsAligned);
        pipe_barrier(PIPE_ALL);

        uint32_t widthInAligned = ((widthIn + 7) / 8) * 8;

        for (uint32_t oh = 0; oh < heightOut; oh++) {
            LocalTensor<float> outLocal = outQueueY.AllocTensor<float>();
            Duplicate(outLocal, (float)0, widthOutAligned);
            pipe_barrier(PIPE_V);

            for (uint32_t kh = 0; kh < kernelSize; kh++) {
                int32_t ih = (int32_t)(oh * stride) - (int32_t)padding + (int32_t)kh;
                if (ih < 0 || ih >= (int32_t)heightIn) continue;

                LocalTensor<float> inLocal = inQueueX.AllocTensor<float>();
                // Load input row
                DataCopy(inLocal, xGm[inputOffset + ih * widthIn], widthInAligned <= widthIn ? widthIn : widthInAligned);
                pipe_barrier(PIPE_MTE2);

                for (uint32_t kw = 0; kw < kernelSize; kw++) {
                    int32_t iwStart = (int32_t)(0 * stride) - (int32_t)padding + (int32_t)kw;

                    float wVal;
                    // Read weight value
                    uint32_t wIdx = kh * kernelSize + kw;

                    LocalTensor<float> tmpLocal = tmpBuf.AllocTensor<float>();

                    // For each output position ow: out[ow] += in[ow*stride + kw - padding] * w[kh][kw]
                    // With stride=1, padding=0: out[ow] += in[ow + kw] * w[kh][kw]
                    // This is: slice of input starting at kw, multiplied by scalar weight

                    // We need to extract the weight scalar
                    // Use Muls with scalar from weight tensor
                    // First copy the relevant input slice to tmpLocal
                    // input[iwStart .. iwStart + widthOut - 1] where iwStart = kw - padding (for stride=1)

                    if (iwStart >= 0 && iwStart + (int32_t)widthOut <= (int32_t)widthIn) {
                        // All valid
                        // Muls: tmpLocal = inLocal[iwStart:] * wLocal[wIdx]
                        // We need scalar multiplication
                        // Strategy: copy slice, multiply by scalar, add to output
                        
                        // Get weight value by broadcasting
                        Duplicate(tmpLocal, (float)0, widthOutAligned);
                        pipe_barrier(PIPE_V);
                        
                        // Copy from inLocal starting at iwStart
                        // Unfortunately we can't do local-to-local DataCopy easily
                        // Use vector Add with offset
                        // tmpLocal[j] = inLocal[iwStart + j] for j in [0, widthOut)
                        
                        // We'll use Adds to shift: not possible directly
                        // Alternative: just do it element-wise or use gather

                        // Actually, let's use the fact that with AscendC we can do:
                        // DataCopy from GM for each needed row+offset
                        // Then Muls by weight scalar, then Add to output
                        
                        DataCopy(tmpLocal, xGm[inputOffset + ih * widthIn + iwStart], widthOutAligned <= (widthIn - iwStart) ? widthOutAligned : (uint32_t)(((widthIn - iwStart) + 7) / 8) * 8);
                        pipe_barrier(PIPE_MTE2);
                        
                        // Broadcast weight scalar
                        LocalTensor<float> dummy = inLocal; // reuse after freeing
                        
                        // Muls tmpLocal by wLocal[wIdx]
                        // We need scalar. AscendC Muls takes scalar.
                        // But we have weight in local tensor. Let's read it.
                        // Actually we can do element-wise tricks:
                        // We can't easily extract scalar from local tensor in AscendC.
                        // Let's load weight from GM directly as scalar won't work.
                        // Alternative: Use Mul with broadcast.
                        
                        // Simplest: reload weight val from GM each time
                        // Or: precompute on host. But kernel_size is small (3).
                        
                        // Let's just use: create a tensor filled with w[kh][kw] and Mul
                        // We already have wLocal. Let's Duplicate a temp with wLocal[wIdx].
                        // But we can't index wLocal as scalar in AscendC easily.
                        
                        // Actually in AscendC we can do:
                        // float val = wLocal.GetValue(wIdx);
                        // Muls(tmpLocal, tmpLocal, val, widthOutAligned);
                        
                        float wScalar = wLocal.GetValue(wIdx);
                        Muls(tmpLocal, tmpLocal, wScalar, widthOutAligned);
                        pipe_barrier(PIPE_V);
                        
                        Add(outLocal, outLocal, tmpLocal, widthOutAligned);
                        pipe_barrier(PIPE_V);
                    } else {
                        // Handle boundary - shouldn't happen with padding=0 and valid kw
                        // but handle for generality
                        Duplicate(tmpLocal, (float)0, widthOutAligned);
                        pipe_barrier(PIPE_V);
                        
                        for (uint32_t ow = 0; ow < widthOut; ow++) {
                            int32_t iw = (int32_t)(ow * stride) + iwStart;
                            if (iw >= 0 && iw < (int32_t)widthIn) {
                                float inVal = inLocal.GetValue(iw);
                                float wScalar = wLocal.GetValue(wIdx);
                                tmpLocal.SetValue(ow, inVal * wScalar);
                            }
                        }
                        pipe_barrier(PIPE_V);
                        Add(outLocal, outLocal, tmpLocal, widthOutAligned);
                        pipe_barrier(PIPE_V);
                    }
                    
                    tmpBuf.FreeTensor(tmpLocal);
                }
                
                inQueueX.FreeTensor(inLocal);
            }
            
            // Write output row
            DataCopy(yGm[outputOffset + oh * widthOut], outLocal, widthOutAligned <= widthOut ? widthOut : widthOutAligned);
            pipe_barrier(PIPE_MTE3);
            outQueueY.FreeTensor(outLocal);
        }
        
        inQueueW.FreeTensor(wLocal);
    }

private:
    TPipe pipe;
    TQue<TPosition::VECIN, 1> inQueueX;
    TQue<TPosition::VECIN, 1> inQueueW;
    TQue<TPosition::VECOUT, 1> outQueueY;
    TBuf<TPosition::VECCALC> tmpBuf;
    GlobalTensor<float> xGm;
    GlobalTensor<float> wGm;
    GlobalTensor<float> yGm;

    uint32_t batchSize, channels, heightIn, widthIn;
    uint32_t kernelSize, stride, padding;
    uint32_t heightOut, widthOut, widthOutAligned;
    uint32_t totalTasks, tasksPerBlock, taskStart, taskEnd;
};

extern "C" __global__ __aicore__ void conv_depthwise_2d_asymmetric_input_square_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvDw op;
    op.Init(x, weight, y,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.heightIn, tiling_data.widthIn,
            tiling_data.kernelSize, tiling_data.stride,
            tiling_data.padding, tiling_data.heightOut,
            tiling_data.widthOut, tiling_data.totalTasks);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized depthwise 2D convolution using custom AscendC kernel.
    """
    def __init__(self, in_channels: int, out_channels: int, kernel_size: int, stride: int = 1, padding: int = 0, bias: bool = False):
        super(ModelNew, self).__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.kernel_size = kernel_size
        self.stride = stride
        self.padding = padding
        # Create depthwise conv weight: shape (in_channels, 1, kernel_size, kernel_size)
        self.weight = torch.nn.Parameter(torch.empty(in_channels, 1, kernel_size, kernel_size))
        torch.nn.init.kaiming_uniform_(self.weight)
        if bias:
            self.bias = torch.nn.Parameter(torch.empty(out_channels))
        else:
            self.bias = None

    def forward(self, x: torch.Tensor) -> torch.Tensor:
        batch_size = x.shape[0]
        channels = x.shape[1]
        height_in = x.shape[2]
        width_in = x.shape[3]
        
        result = custom_ops_lib.conv_depthwise_2d_asymmetric_input_square_kernel_custom(
            x, self.weight,
            batch_size, channels, height_in, width_in,
            self.kernel_size, self.stride, self.padding
        )
        
        if self.bias is not None:
            result = result + self.bias.reshape(1, -1, 1, 1)
        
        return result
'''
