kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmGroupNormHardtanh {
public:
    __aicore__ inline KernelGemmGroupNormHardtanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z, GM_ADDR workspace,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures, uint32_t numGroups,
                                float minVal, float maxVal, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->minVal = minVal;
        this->maxVal = maxVal;
        this->channelsPerGroup = outFeatures / numGroups;

        // Each block processes a subset of batch rows
        uint32_t totalRows = batchSize;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowStart = blockIdx * ((totalRows + blockNum - 1) / blockNum);
        this->rowEnd = (blockIdx + 1) * ((totalRows + blockNum - 1) / blockNum);
        if (this->rowEnd > totalRows) this->rowEnd = totalRows;
        this->myRows = (this->rowEnd > this->rowStart) ? (this->rowEnd - this->rowStart) : 0;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, outFeatures * inFeatures);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);
        gammaGm.SetGlobalBuffer((__gm__ float *)gamma, outFeatures);
        betaGm.SetGlobalBuffer((__gm__ float *)beta, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * outFeatures);
        workGm.SetGlobalBuffer((__gm__ float *)workspace, batchSize * outFeatures);

        // Allocate buffers for tile processing
        uint32_t bufSize = this->channelsPerGroup * sizeof(float);
        if (bufSize < 32) bufSize = 32;
        pipe.InitBuffer(inQueueX, 1, bufSize);
        pipe.InitBuffer(outQueueZ, 1, bufSize);
        pipe.InitBuffer(tmpBuf1, 1, bufSize);
        pipe.InitBuffer(tmpBuf2, 1, bufSize);
    }

    __aicore__ inline void Process()
    {
        if (this->myRows == 0) return;

        // Step 1: GEMM - compute z = x * weight^T + bias for our rows
        // We do this element by element using workspace as intermediate
        // For simplicity in AscendC, we compute the GEMM row by row, group by group
        for (uint32_t row = 0; row < this->myRows; row++) {
            uint32_t globalRow = this->rowStart + row;
            // For each output element, compute dot product
            // We'll compute per group for efficiency
            for (uint32_t g = 0; g < this->numGroups; g++) {
                uint32_t colStart = g * this->channelsPerGroup;
                
                // Compute GEMM output for this group of columns
                for (uint32_t c = 0; c < this->channelsPerGroup; c++) {
                    uint32_t outCol = colStart + c;
                    float acc = 0.0f;
                    // Dot product: x[globalRow, :] * weight[outCol, :]
                    for (uint32_t k = 0; k < this->inFeatures; k++) {
                        float xVal, wVal;
                        xVal = xGm.GetValue(globalRow * this->inFeatures + k);
                        wVal = weightGm.GetValue(outCol * this->inFeatures + k);
                        acc += xVal * wVal;
                    }
                    acc += biasGm.GetValue(outCol);
                    workGm.SetValue(globalRow * this->outFeatures + outCol, acc);
                }
            }
            
            // Step 2: Group Normalization for this row
            for (uint32_t g = 0; g < this->numGroups; g++) {
                uint32_t colStart = g * this->channelsPerGroup;
                
                // Compute mean
                float sum = 0.0f;
                for (uint32_t c = 0; c < this->channelsPerGroup; c++) {
                    sum += workGm.GetValue(globalRow * this->outFeatures + colStart + c);
                }
                float mean = sum / (float)this->channelsPerGroup;
                
                // Compute variance
                float varSum = 0.0f;
                for (uint32_t c = 0; c < this->channelsPerGroup; c++) {
                    float val = workGm.GetValue(globalRow * this->outFeatures + colStart + c) - mean;
                    varSum += val * val;
                }
                float variance = varSum / (float)this->channelsPerGroup;
                float invStd = 1.0f / sqrtf(variance + 1e-5f);
                
                // Normalize, scale, shift, and HardTanh
                for (uint32_t c = 0; c < this->channelsPerGroup; c++) {
                    uint32_t idx = globalRow * this->outFeatures + colStart + c;
                    float val = workGm.GetValue(idx);
                    val = (val - mean) * invStd;
                    // Apply gamma and beta
                    float g_val = gammaGm.GetValue(colStart + c);
                    float b_val = betaGm.GetValue(colStart + c);
                    val = val * g_val + b_val;
                    // HardTanh
                    if (val < this->minVal) val = this->minVal;
                    if (val > this->maxVal) val = this->maxVal;
                    zGm.SetValue(idx, val);
                }
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm, weightGm, biasGm, gammaGm, betaGm, zGm, workGm;
    uint32_t batchSize, inFeatures, outFeatures, numGroups;
    uint32_t channelsPerGroup;
    float minVal, maxVal;
    uint32_t rowStart, rowEnd, myRows;
};

extern "C" __global__ __aicore__ void gemm_group_norm_hardtanh_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormHardtanh op;
    op.Init(x, weight, bias, gamma, beta, z, workspace,
            tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures,
            tiling_data.numGroups, tiling_data.minVal, tiling_data.maxVal, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, num_groups, hardtanh_min, hardtanh_max):
        super().__init__()
        self.gemm = torch.nn.Linear(in_features, out_features)
        self.group_norm = torch.nn.GroupNorm(num_groups, out_features)
        self.in_features = in_features
        self.out_features = out_features
        self.num_groups = num_groups
        self.hardtanh_min = hardtanh_min
        self.hardtanh_max = hardtanh_max

    def forward(self, x):
        batch_size = x.shape[0]
        weight = self.gemm.weight
        bias = self.gemm.bias
        gamma = self.group_norm.weight
        beta = self.group_norm.bias
        return custom_ops_lib.gemm_group_norm_hardtanh_custom(
            x, weight, bias, gamma, beta,
            batch_size, self.in_features, self.out_features,
            self.num_groups, self.hardtanh_min, self.hardtanh_max
        )
'''
