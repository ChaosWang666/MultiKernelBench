

Looking at this architecture, it's a shallow wide MLP with Linear layers and ReLU activations. The key operation that can be fused is the Linear + ReLU pattern (matrix multiplication + bias addition + ReLU activation). I'll create a custom kernel that fuses the MatMul + BiasAdd + ReLU operation for the hidden layers.

However, given the complexity of matrix multiplication on AscendC (which requires cube core operations), a more practical approach is to fuse the bias addition and ReLU activation into a single kernel, since these are element-wise operations that benefit from fusion. The matmul itself is already highly optimized in the NPU's cube cores.

Let me create a fused `BiasAdd + ReLU` kernel and a final `BiasAdd` kernel (for the output layer), applied after matmul operations.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelBiasAddRelu {
public:
    __aicore__ inline KernelBiasAddRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                 uint32_t totalLength, uint32_t tileNum,
                                 uint32_t batchSize, uint32_t featureSize,
                                 uint32_t applyRelu)
    {
        this->totalLength = totalLength;
        this->batchSize = batchSize;
        this->featureSize = featureSize;
        this->applyRelu = applyRelu;

        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, this->featureSize);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();

        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);

        // Calculate global offset for bias broadcast
        uint32_t globalOffset = this->blockLength * AscendC::GetBlockIdx() + progress * this->tileLength;

        // Copy bias with proper broadcasting - for each element, bias index = globalOffset % featureSize
        // We need to fill biasLocal tile by tile from biasGm
        // If tileLength <= featureSize and tiles align, we can do direct copy
        // Otherwise we handle the general case

        uint32_t startBiasIdx = globalOffset % this->featureSize;

        if (startBiasIdx + this->tileLength <= this->featureSize) {
            // Simple case: tile fits within one row of bias
            AscendC::DataCopy(biasLocal, biasGm[startBiasIdx], this->tileLength);
        } else {
            // Need to handle wrap-around: copy in segments
            uint32_t copied = 0;
            uint32_t biasIdx = startBiasIdx;
            while (copied < this->tileLength) {
                uint32_t chunkLen = this->featureSize - biasIdx;
                if (chunkLen > this->tileLength - copied) {
                    chunkLen = this->tileLength - copied;
                }
                // Ensure chunkLen is aligned to 32 bytes (8 floats) for DataCopy
                uint32_t alignedLen = chunkLen;
                AscendC::DataCopy(biasLocal[copied], biasGm[biasIdx], alignedLen);
                copied += chunkLen;
                biasIdx = 0;
            }
        }

        inQueueX.EnQue(xLocal);
        inQueueBias.EnQue(biasLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        // z = x + bias
        AscendC::Add(zLocal, xLocal, biasLocal, this->tileLength);

        // Apply ReLU if needed
        if (this->applyRelu) {
            // ReLU: max(0, z)
            float zeroVal = 0.0f;
            AscendC::Maxs(zLocal, zLocal, zeroVal, this->tileLength);
        }

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueBias.FreeTensor(biasLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t batchSize;
    uint32_t featureSize;
    uint32_t applyRelu;
};

extern "C" __global__ __aicore__ void shallow_wide_mlp_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelBiasAddRelu op;
    op.Init(x, bias, z, tiling_data.totalLength, tiling_data.tileNum,
            tiling_data.batchSize, tiling_data.featureSize, tiling_data.applyRelu);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, input_size, hidden_layer_sizes, output_size):
        """
        :param input_size: The number of input features
        :param hidden_layer_sizes: A list of ints containing the sizes of each hidden layer
        :param output_size: The number of output features
        """
        super(ModelNew, self).__init__()

        self.weights = torch.nn.ParameterList()
        self.biases = torch.nn.ParameterList()
        self.num_hidden = len(hidden_layer_sizes)

        current_input_size = input_size
        for hidden_size in hidden_layer_sizes:
            linear = torch.nn.Linear(current_input_size, hidden_size)
            self.weights.append(torch.nn.Parameter(linear.weight))
            self.biases.append(torch.nn.Parameter(linear.bias))
            current_input_size = hidden_size

        # Output layer
        linear_out = torch.nn.Linear(current_input_size, output_size)
        self.weights.append(torch.nn.Parameter(linear_out.weight))
        self.biases.append(torch.nn.Parameter(linear_out.bias))

    def forward(self, x):
        """
        :param x: The input tensor, shape (batch_size, input_size)
        :return: The output tensor, shape (batch_size, output_size)
        """
        for i in range(self.num_hidden):
            # MatMul: x @ W^T
            x = torch.mm(x, self.weights[i].t())
            # Fused bias add + ReLU using custom kernel
            x = custom_ops_lib.shallow_wide_mlp_custom(x, self.biases[i], 1)

        # Output layer: matmul + bias (no relu)
        x = torch.mm(x, self.weights[self.num_hidden].t())
        x = custom_ops_lib.shallow_wide_mlp_custom(x, self.biases[self.num_hidden], 0)

        return x
'''
