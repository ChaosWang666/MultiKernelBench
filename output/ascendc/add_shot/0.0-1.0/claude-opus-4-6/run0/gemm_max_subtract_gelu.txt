constexpr int32_t BUFFER_NUM = 2;

class KernelGemmMaxSubtractGelu {
public:
    __aicore__ inline KernelGemmMaxSubtractGelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t batchSize, uint32_t outFeatures)
    {
        this->batchSize = batchSize;
        this->outFeatures = outFeatures;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        // Each block processes a subset of rows
        this->rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        this->endRow = this->startRow + this->rowsPerBlock;
        if (this->endRow > batchSize) {
            this->endRow = batchSize;
        }
        this->myRows = (this->endRow > this->startRow) ? (this->endRow - this->startRow) : 0;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize);

        // Allocate buffer to hold one row of data at a time
        // We process in tiles if outFeatures is large
        // Align tile length to 32 bytes = 8 floats
        uint32_t maxTileLen = 8192; // max elements per tile
        if (outFeatures <= maxTileLen) {
            this->tileLength = outFeatures;
            this->numTiles = 1;
        } else {
            this->numTiles = (outFeatures + maxTileLen - 1) / maxTileLen;
            this->tileLength = (outFeatures + this->numTiles - 1) / this->numTiles;
            // Align to 8
            this->tileLength = (this->tileLength + 7) / 8 * 8;
            this->numTiles = (outFeatures + this->tileLength - 1) / this->tileLength;
        }

        pipe.InitBuffer(inQueueX, 1, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, 8 * sizeof(float)); // small buffer for single output value
        pipe.InitBuffer(tmpBuf, 1, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t row = 0; row < this->myRows; row++) {
            ProcessRow(this->startRow + row);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t rowIdx)
    {
        float maxVal = -3.4028235e+38f; // -FLT_MAX

        // First pass: find max across the row
        for (uint32_t t = 0; t < this->numTiles; t++) {
            uint32_t offset = t * this->tileLength;
            uint32_t curLen = this->tileLength;
            if (offset + curLen > this->outFeatures) {
                curLen = this->outFeatures - offset;
            }
            uint32_t alignedLen = (curLen + 7) / 8 * 8;

            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[rowIdx * this->outFeatures + offset], alignedLen);
            inQueueX.EnQue(xLocal);

            AscendC::LocalTensor<float> xProc = inQueueX.DeQue<float>();
            AscendC::LocalTensor<float> tmpLocal = tmpBuf.AllocTensor<float>();

            // ReduceMax
            AscendC::ReduceMax(tmpLocal, xProc, alignedLen);
            float tileMax = tmpLocal.GetValue(0);
            if (tileMax > maxVal) {
                maxVal = tileMax;
            }

            tmpBuf.FreeTensor(tmpLocal);
            inQueueX.FreeTensor(xProc);
        }

        // maxVal is the max over the entire row
        // After max with keepdim, result is (1,) per row
        // mean of a single value along dim=1 is itself
        // subtract: maxVal - maxVal = 0
        // gelu(0) = 0
        float result = 0.0f;

        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        zLocal.SetValue(0, result);
        outQueueZ.EnQue(zLocal);

        AscendC::LocalTensor<float> zOut = outQueueZ.DeQue<float>();
        // Write single value; need to copy at least 8 floats (32 bytes aligned)
        // But we only need 1 value per row. We'll write to the correct position.
        // For safety with alignment, we handle this carefully.
        // Since output is (batchSize, 1), total = batchSize floats
        // We need to be careful with DataCopy alignment
        zOut.SetValue(0, result);
        // Pad remaining
        for (int i = 1; i < 8; i++) {
            zOut.SetValue(i, 0.0f);
        }

        // Use DataCopy with 8-element alignment
        uint32_t outIdx = rowIdx;
        // If outIdx is aligned to 8
        if (outIdx % 8 == 0 && outIdx + 8 <= this->batchSize) {
            AscendC::DataCopy(zGm[outIdx], zOut, 8);
        } else {
            // For non-aligned or boundary cases, we set value directly
            AscendC::SetAtomicAdd<float>();
            // Actually let's just use SetValue approach
            AscendC::SetAtomicNone<float>();
            // We'll write 8 values but only first matters - need care
            // Simple approach: just store directly
            zGm.SetValue(outIdx, result);
        }

        outQueueZ.FreeTensor(zOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t outFeatures;
    uint32_t rowsPerBlock;
    uint32_t startRow;
    uint32_t endRow;
    uint32_t myRows;
    uint32_t tileLength;
    uint32_t numTiles;
};

extern "C" __global__ __aicore__ void gemm_max_subtract_gelu_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmMaxSubtractGelu op;
    op.Init(x, z, tiling_data.batchSize, tiling_data.outFeatures);
    op.Process();
}
