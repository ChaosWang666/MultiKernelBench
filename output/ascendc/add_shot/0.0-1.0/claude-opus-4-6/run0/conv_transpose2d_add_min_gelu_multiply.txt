

Looking at this architecture, I need to analyze the operations:
1. Transposed convolution (complex, keep as PyTorch op)
2. Add a scalar value
3. Min with 0.0
4. GELU activation
5. Multiply by a scalar value

I'll fuse operations 2-5 (add + min + gelu + multiply) into a single custom AscendC kernel, since these are all element-wise operations that benefit from fusion.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelAddMinGeluMultiply {
public:
    __aicore__ inline KernelAddMinGeluMultiply() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t tileNum, float addValue, float multiplyValue)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->addValue = addValue;
        this->multiplyValue = multiplyValue;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf1, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();

        // x = x + add_value
        AscendC::Adds(xLocal, xLocal, this->addValue, this->tileLength);

        // x = min(x, 0.0)
        AscendC::Mins(xLocal, xLocal, (float)0.0f, this->tileLength);

        // GELU(x) = 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
        // Use approximation: gelu(x) = 0.5 * x * (1 + tanh(0.7978845608 * (x + 0.044715 * x^3)))
        
        // tmp1 = x * x * x
        AscendC::Mul(tmp1, xLocal, xLocal, this->tileLength);
        AscendC::Mul(tmp1, tmp1, xLocal, this->tileLength);
        
        // tmp1 = 0.044715 * x^3
        AscendC::Muls(tmp1, tmp1, (float)0.044715f, this->tileLength);
        
        // tmp1 = x + 0.044715 * x^3
        AscendC::Add(tmp1, xLocal, tmp1, this->tileLength);
        
        // tmp1 = 0.7978845608 * (x + 0.044715 * x^3)
        AscendC::Muls(tmp1, tmp1, (float)0.7978845608f, this->tileLength);
        
        // tmp2 = tanh(tmp1) -- use exp-based: tanh(a) = (exp(2a)-1)/(exp(2a)+1)
        // tmp2 = 2 * tmp1
        AscendC::Muls(tmp2, tmp1, (float)2.0f, this->tileLength);
        // tmp2 = exp(tmp2)
        AscendC::Exp(tmp2, tmp2, this->tileLength);
        // yLocal = tmp2 - 1
        AscendC::Adds(yLocal, tmp2, (float)(-1.0f), this->tileLength);
        // tmp1 = tmp2 + 1
        AscendC::Adds(tmp1, tmp2, (float)(1.0f), this->tileLength);
        // tmp2 = yLocal / tmp1 = tanh
        AscendC::Div(tmp2, yLocal, tmp1, this->tileLength);
        
        // tmp2 = 1 + tanh(...)
        AscendC::Adds(tmp2, tmp2, (float)1.0f, this->tileLength);
        
        // yLocal = 0.5 * x
        AscendC::Muls(yLocal, xLocal, (float)0.5f, this->tileLength);
        
        // yLocal = 0.5 * x * (1 + tanh(...))
        AscendC::Mul(yLocal, yLocal, tmp2, this->tileLength);

        // x = gelu(x) * multiply_value
        AscendC::Muls(yLocal, yLocal, this->multiplyValue, this->tileLength);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    float addValue;
    float multiplyValue;
};

extern "C" __global__ __aicore__ void conv_transpose2d_add_min_gelu_multiply_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelAddMinGeluMultiply op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.tileNum, tiling_data.addValue, tiling_data.multiplyValue);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model that performs transposed convolution using PyTorch,
    then fuses add + min + gelu + multiply into a single custom AscendC kernel.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, add_value, multiply_value):
        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride)
        self.add_value = add_value
        self.multiply_value = multiply_value

    def forward(self, x):
        x = self.conv_transpose(x)
        x = custom_ops_lib.conv_transpose2d_add_min_gelu_multiply_custom(x, self.add_value, self.multiply_value)
        return x
'''
