

Looking at this architecture, I need to fuse convolution, batch normalization, and scaling into a custom AscendC kernel. However, convolution is complex to implement efficiently in AscendC. A more practical approach is to fuse the batch normalization and scaling operations (since during inference, BN is just a linear transform that can be combined with scaling), while keeping the convolution as a standard PyTorch operation.

Actually, let me reconsider. The most beneficial fusion here is combining BatchNorm + Scaling into one kernel, since BN in eval mode is `y = (x - mean) / sqrt(var + eps) * weight + bias`, and then we multiply by `scaling_factor`. This can be fused into a single element-wise operation with per-channel parameters.

Let me implement a fused BN+Scale kernel that takes the conv output and applies the combined batch norm and scaling in one pass.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelBNScale {
public:
    __aicore__ inline KernelBNScale() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                 uint32_t totalLength, uint32_t channelLength,
                                 uint32_t spatialLength, uint32_t tileNum)
    {
        this->totalLength = totalLength;
        this->channelLength = channelLength;
        this->spatialLength = spatialLength;
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        wGm.SetGlobalBuffer((__gm__ float *)weight, this->channelLength);
        bGm.SetGlobalBuffer((__gm__ float *)bias, this->channelLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, 1, this->spatialLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        uint32_t globalOffset = this->blockLength * AscendC::GetBlockIdx() + progress * this->tileLength;
        uint32_t processed = 0;
        
        while (processed < this->tileLength) {
            uint32_t globalPos = globalOffset + processed;
            uint32_t withinSample = globalPos % (this->channelLength * this->spatialLength);
            uint32_t c = withinSample / this->spatialLength;
            uint32_t spatialPos = withinSample % this->spatialLength;
            
            uint32_t remaining = this->spatialLength - spatialPos;
            uint32_t chunk = remaining;
            if (chunk > this->tileLength - processed) {
                chunk = this->tileLength - processed;
            }
            
            // Align chunk to 8 elements for vector ops, handle remainder scalar
            uint32_t alignedChunk = (chunk / 8) * 8;
            
            if (alignedChunk > 0) {
                AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
                // Fill tmpLocal with weight[c]
                AscendC::Duplicate(tmpLocal, wGm.GetValue(c), alignedChunk);
                AscendC::Mul(zLocal[processed], xLocal[processed], tmpLocal, alignedChunk);
                // Fill tmpLocal with bias[c]
                AscendC::Duplicate(tmpLocal, bGm.GetValue(c), alignedChunk);
                AscendC::Add(zLocal[processed], zLocal[processed], tmpLocal, alignedChunk);
            }
            
            // Handle remaining elements
            float w_val = wGm.GetValue(c);
            float b_val = bGm.GetValue(c);
            for (uint32_t j = alignedChunk; j < chunk; j++) {
                float val = xLocal.GetValue(processed + j);
                zLocal.SetValue(processed + j, val * w_val + b_val);
            }
            
            processed += chunk;
        }

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t totalLength;
    uint32_t channelLength;
    uint32_t spatialLength;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv2d_batch_norm_scaling_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelBNScale op;
    op.Init(x, weight, bias, z, tiling_data.totalLength, tiling_data.channelLength, tiling_data.spatialLength, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    """
    Optimized model that performs convolution, then fused Batch Normalization + Scaling
    using a custom AscendC kernel.
    """
    def __init__(self, in_channels, out_channels, kernel_size, scaling_factor):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.bn = nn.BatchNorm2d(out_channels)
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = self.conv(x)
        # Fuse BN and scaling: y = (x - mean)/sqrt(var+eps) * gamma * scale + beta * scale
        # Precompute combined weight and bias
        if self.bn.running_mean is not None and self.bn.running_var is not None:
            mean = self.bn.running_mean
            var = self.bn.running_var
            gamma = self.bn.weight if self.bn.weight is not None else torch.ones_like(mean)
            beta = self.bn.bias if self.bn.bias is not None else torch.zeros_like(mean)
            eps = self.bn.eps
            
            inv_std = 1.0 / torch.sqrt(var + eps)
            # Combined: y = x * (gamma * inv_std * scaling) + (beta - mean * gamma * inv_std) * scaling
            fused_weight = gamma * inv_std * self.scaling_factor
            fused_bias = (beta - mean * gamma * inv_std) * self.scaling_factor
            
            x = custom_ops_lib.conv2d_batch_norm_scaling_custom(x.contiguous(), fused_weight.contiguous(), fused_bias.contiguous())
        else:
            x = self.bn(x)
            x = x * self.scaling_factor
        return x
'''
