

Looking at this architecture, I need to analyze what operations are performed:
1. ConvTranspose2d
2. MaxPool2d
3. Hardtanh (clamp between min and max)
4. Mean over spatial dims (2,3) with keepdim
5. Tanh

The conv_transpose and maxpool are complex operations best left to existing NPU implementations. I'll fuse the post-processing operations (hardtanh + mean + tanh) into a custom kernel, since after maxpool we have a tensor where we need to clamp, reduce, and apply tanh - these are memory-bound operations that benefit from fusion.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelHardtanhMeanTanh {
public:
    __aicore__ inline KernelHardtanhMeanTanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalBC, uint32_t spatialSize,
                                  float hardtanhMin, float hardtanhMax, uint32_t tileNum)
    {
        this->totalBC = totalBC;
        this->spatialSize = spatialSize;
        this->hardtanhMin = hardtanhMin;
        this->hardtanhMax = hardtanhMax;
        this->tileNum = tileNum;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        // Each block handles a range of (batch*channel) slices
        this->bcPerBlock = (totalBC + blockNum - 1) / blockNum;
        this->bcStart = blockIdx * this->bcPerBlock;
        this->bcEnd = this->bcStart + this->bcPerBlock;
        if (this->bcEnd > totalBC) {
            this->bcEnd = totalBC;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x, (uint64_t)totalBC * spatialSize);
        yGm.SetGlobalBuffer((__gm__ float*)y, (uint64_t)totalBC);

        // Align tile length to 32 bytes (8 floats)
        uint32_t alignUnit = 8;
        if (spatialSize <= 4096) {
            this->tileLength = ((spatialSize + alignUnit - 1) / alignUnit) * alignUnit;
            this->numTiles = 1;
        } else {
            uint32_t rawTile = (spatialSize + tileNum - 1) / tileNum;
            this->tileLength = ((rawTile + alignUnit - 1) / alignUnit) * alignUnit;
            this->numTiles = (spatialSize + this->tileLength - 1) / this->tileLength;
        }

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, alignUnit * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t bc = this->bcStart; bc < this->bcEnd; bc++) {
            ProcessOneSlice(bc);
        }
    }

private:
    __aicore__ inline void ProcessOneSlice(uint32_t bc)
    {
        float accumSum = 0.0f;
        uint64_t baseOffset = (uint64_t)bc * this->spatialSize;

        for (uint32_t t = 0; t < this->numTiles; t++) {
            uint32_t offset = t * this->tileLength;
            uint32_t remaining = this->spatialSize - offset;
            uint32_t curLen = remaining < this->tileLength ? remaining : this->tileLength;
            uint32_t alignUnit = 8;
            uint32_t curLenAligned = ((curLen + alignUnit - 1) / alignUnit) * alignUnit;

            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();

            // Copy data
            AscendC::DataCopy(xLocal, xGm[baseOffset + offset], curLenAligned);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            // If we copied more than actual remaining, zero out the extra
            if (curLen < curLenAligned) {
                for (uint32_t i = curLen; i < curLenAligned; i++) {
                    xLocal.SetValue(i, 0.0f);
                }
            }

            // Hardtanh: clamp values
            AscendC::Mins(xLocal, xLocal, (float)this->hardtanhMax, curLenAligned);
            AscendC::Maxs(xLocal, xLocal, (float)this->hardtanhMin, curLenAligned);

            // Reduce sum
            AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
            AscendC::ReduceSum(zLocal, xLocal, curLenAligned, false);
            float tileSum = zLocal.GetValue(0);
            accumSum += tileSum;

            outQueueZ.FreeTensor(zLocal);
            inQueueX.FreeTensor(xLocal);
        }

        // Mean = sum / spatialSize
        float meanVal = accumSum / (float)this->spatialSize;

        // Tanh
        float tanhVal;
        float ep = meanVal + meanVal;  // 2*x
        if (ep > 20.0f) {
            tanhVal = 1.0f;
        } else if (ep < -20.0f) {
            tanhVal = -1.0f;
        } else {
            // tanh(x) = (exp(2x) - 1) / (exp(2x) + 1)
            // Use simple computation
            float e2x;
            // Approximate exp using the device - we'll just compute directly
            // exp(ep) approximation via series won't be accurate enough
            // Use scalar math
            e2x = AscendC::Scalar<float>(ep);
            // We need to use a different approach - let's compute on a small tensor
            AscendC::LocalTensor<float> tmpLocal = outQueueZ.AllocTensor<float>();
            tmpLocal.SetValue(0, meanVal);
            // Pad remaining with zeros
            for (uint32_t i = 1; i < 8; i++) {
                tmpLocal.SetValue(i, 0.0f);
            }
            AscendC::Tanh(tmpLocal, tmpLocal, 8);
            tanhVal = tmpLocal.GetValue(0);
            outQueueZ.FreeTensor(tmpLocal);
        }

        // Write output
        // We need to write a single float to global memory at bc position
        // But DataCopy requires aligned sizes. We'll use a trick.
        yGm.SetValue(bc, tanhVal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t totalBC;
    uint32_t spatialSize;
    float hardtanhMin;
    float hardtanhMax;
    uint32_t tileNum;
    uint32_t numTiles;
    uint32_t tileLength;
    uint32_t bcPerBlock;
    uint32_t bcStart;
    uint32_t bcEnd;
};

extern "C" __global__ __aicore__ void conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelHardtanhMeanTanh op;
    op.Init(x, y, tiling_data.totalBC, tiling_data.spatialSize,
            tiling_data.hardtanhMin, tiling_data.hardtanhMax, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    """
    Optimized model that performs transposed convolution, max pooling using PyTorch,
    then fuses hardtanh + mean + tanh into a custom AscendC kernel.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, maxpool_kernel_size, maxpool_stride, hardtanh_min, hardtanh_max):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.maxpool = nn.MaxPool2d(kernel_size=maxpool_kernel_size, stride=maxpool_stride)
        self.hardtanh_min = hardtanh_min
        self.hardtanh_max = hardtanh_max

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.maxpool(x)
        # Fused hardtanh + mean + tanh custom kernel
        b, c, h, w = x.shape
        x = x.contiguous()
        x = custom_ops_lib.conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(
            x, b, c, h, w, self.hardtanh_min, self.hardtanh_max)
        return x
'''
