

Looking at the architecture, I need to analyze the operations:
1. Conv3d
2. Multiply by learnable parameter
3. InstanceNorm3d
4. Clamp
5. Multiply by same learnable parameter
6. Max over channel dimension

I'll fuse operations 2-6 (multiply, instance norm, clamp, multiply, max) into a custom kernel, keeping Conv3d as the standard PyTorch op since it's complex and already well-optimized.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv3dMultiplyInstanceNormClampMultiplyMax {
public:
    __aicore__ inline KernelConv3dMultiplyInstanceNormClampMultiplyMax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR multiplier, GM_ADDR clampMinAddr, GM_ADDR clampMaxAddr,
                                 GM_ADDR y, uint32_t batchSize, uint32_t channels, uint32_t spatialSize, float epsilon)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->epsilon = epsilon;
        
        uint32_t totalBlocks = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        // Each block processes some batch elements
        this->batchPerBlock = (batchSize + totalBlocks - 1) / totalBlocks;
        this->batchStart = blockIdx * this->batchPerBlock;
        if (this->batchStart >= batchSize) {
            this->batchPerBlock = 0;
        } else if (this->batchStart + this->batchPerBlock > batchSize) {
            this->batchPerBlock = batchSize - this->batchStart;
        }
        
        uint32_t totalInputSize = batchSize * channels * spatialSize;
        uint32_t totalOutputSize = batchSize * spatialSize;
        
        xGm.SetGlobalBuffer((__gm__ float *)x, totalInputSize);
        multiplierGm.SetGlobalBuffer((__gm__ float *)multiplier, channels);
        clampMinGm.SetGlobalBuffer((__gm__ float *)clampMinAddr, 1);
        clampMaxGm.SetGlobalBuffer((__gm__ float *)clampMaxAddr, 1);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalOutputSize);
        
        // Align spatialSize to 8 for float (32 bytes)
        uint32_t alignedSpatial = ((spatialSize + 7) / 8) * 8;
        
        // Buffer for one channel's spatial data
        pipe.InitBuffer(inQueue, BUFFER_NUM, alignedSpatial * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, alignedSpatial * sizeof(float));
        pipe.InitBuffer(tmpBuf1, BUFFER_NUM, alignedSpatial * sizeof(float));
        pipe.InitBuffer(tmpBuf2, BUFFER_NUM, alignedSpatial * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        for (uint32_t b = 0; b < this->batchPerBlock; b++) {
            uint32_t batchIdx = this->batchStart + b;
            ProcessOneBatch(batchIdx);
        }
    }

private:
    __aicore__ inline void ProcessOneBatch(uint32_t batchIdx)
    {
        uint32_t alignedSpatial = ((spatialSize + 7) / 8) * 8;
        
        float clampMinVal = clampMinGm.GetValue(0);
        float clampMaxVal = clampMaxGm.GetValue(0);
        
        // For each channel: multiply by multiplier, compute mean/var for instance norm
        // Then clamp, multiply again, then max across channels
        
        // First pass: for each channel compute mean and variance, normalize, clamp, multiply
        // Then max across channels
        
        bool firstChannel = true;
        
        for (uint32_t c = 0; c < this->channels; c++) {
            float mult = multiplierGm.GetValue(c);
            uint32_t inOffset = batchIdx * channels * spatialSize + c * spatialSize;
            
            // Copy in
            AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[inOffset], alignedSpatial);
            inQueue.EnQue(xLocal);
            
            AscendC::LocalTensor<float> xData = inQueue.DeQue<float>();
            AscendC::LocalTensor<float> workLocal = tmpBuf1.AllocTensor<float>();
            
            // Step 1: multiply by multiplier[c]
            AscendC::Muls(xData, xData, mult, spatialSize);
            
            // Step 2: instance norm - compute mean
            float sum = 0.0f;
            for (uint32_t i = 0; i < spatialSize; i++) {
                sum += xData.GetValue(i);
            }
            float mean = sum / (float)spatialSize;
            
            // Subtract mean
            AscendC::Adds(xData, xData, -mean, spatialSize);
            
            // Compute variance
            AscendC::Mul(workLocal, xData, xData, spatialSize);
            float varSum = 0.0f;
            for (uint32_t i = 0; i < spatialSize; i++) {
                varSum += workLocal.GetValue(i);
            }
            float var = varSum / (float)spatialSize;
            float invStd = 1.0f / sqrtf(var + epsilon);
            
            // Normalize
            AscendC::Muls(xData, xData, invStd, spatialSize);
            
            // Step 3: clamp
            AscendC::Mins(xData, xData, clampMaxVal, spatialSize);
            AscendC::Maxs(xData, xData, clampMinVal, spatialSize);
            
            // Step 4: multiply by multiplier[c] again
            AscendC::Muls(xData, xData, mult, spatialSize);
            
            tmpBuf1.FreeTensor(workLocal);
            
            // Step 5: max across channels
            if (firstChannel) {
                // Copy to output buffer
                AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
                AscendC::DataCopy(outLocal, xData, alignedSpatial);
                outQueue.EnQue(outLocal);
                AscendC::LocalTensor<float> outData = outQueue.DeQue<float>();
                
                // Write to output
                uint32_t outOffset = batchIdx * spatialSize;
                AscendC::DataCopy(yGm[outOffset], outData, alignedSpatial);
                outQueue.FreeTensor(outData);
                firstChannel = false;
            } else {
                // Load current max from output
                AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
                uint32_t outOffset = batchIdx * spatialSize;
                AscendC::DataCopy(outLocal, yGm[outOffset], alignedSpatial);
                outQueue.EnQue(outLocal);
                AscendC::LocalTensor<float> outData = outQueue.DeQue<float>();
                
                // Element-wise max
                AscendC::Max(outData, outData, xData, spatialSize);
                
                // Write back
                AscendC::DataCopy(yGm[outOffset], outData, alignedSpatial);
                outQueue.FreeTensor(outData);
            }
            
            inQueue.FreeTensor(xData);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> multiplierGm;
    AscendC::GlobalTensor<float> clampMinGm;
    AscendC::GlobalTensor<float> clampMaxGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t spatialSize;
    float epsilon;
    uint32_t batchPerBlock;
    uint32_t batchStart;
};

extern "C" __global__ __aicore__ void conv3d_multiply_instance_norm_clamp_multiply_max_custom(
    GM_ADDR x, GM_ADDR multiplier, GM_ADDR clamp_min_val, GM_ADDR clamp_max_val,
    GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dMultiplyInstanceNormClampMultiplyMax op;
    op.Init(x, multiplier, clamp_min_val, clamp_max_val, y,
            tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize, tiling_data.epsilon);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    """
    Optimized model using custom AscendC kernel for fused multiply+instancenorm+clamp+multiply+max.
    Conv3d remains as standard PyTorch op.
    """
    def __init__(self, in_channels, out_channels, kernel_size, multiplier_shape, clamp_min, clamp_max):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        self.multiplier = torch.nn.Parameter(torch.randn(multiplier_shape))
        self.clamp_min = clamp_min
        self.clamp_max = clamp_max

    def forward(self, x):
        x = self.conv(x)
        # Flatten multiplier for the kernel
        multiplier_flat = self.multiplier.view(-1).contiguous()
        clamp_min_t = torch.tensor([self.clamp_min], dtype=x.dtype, device=x.device)
        clamp_max_t = torch.tensor([self.clamp_max], dtype=x.dtype, device=x.device)
        x = custom_ops_lib.conv3d_multiply_instance_norm_clamp_multiply_max_custom(
            x.contiguous(), multiplier_flat, clamp_min_t, clamp_max_t)
        return x
