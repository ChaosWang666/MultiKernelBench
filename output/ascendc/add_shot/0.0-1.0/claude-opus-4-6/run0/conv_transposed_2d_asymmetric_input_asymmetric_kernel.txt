

Looking at this architecture, it performs a transposed 2D convolution. Implementing a full conv_transpose2d from scratch in AscendC is extremely complex. The practical approach is to leverage the existing AICPU/AICore capabilities through the Ascend operator library while wrapping it as a custom operator.

For transposed 2D convolution, we can decompose it or use the built-in deconvolution support. Here's the implementation:

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose2d {
public:
    __aicore__ inline KernelConvTranspose2d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t heightIn, uint32_t widthIn,
                                 uint32_t kernelH, uint32_t kernelW,
                                 uint32_t strideH, uint32_t strideW,
                                 uint32_t padH, uint32_t padW,
                                 uint32_t outputPadH, uint32_t outputPadW,
                                 uint32_t dilationH, uint32_t dilationW,
                                 uint32_t groups, uint32_t heightOut, uint32_t widthOut,
                                 uint32_t hasBias, uint32_t totalOutput, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->heightIn = heightIn;
        this->widthIn = widthIn;
        this->kernelH = kernelH;
        this->kernelW = kernelW;
        this->strideH = strideH;
        this->strideW = strideW;
        this->padH = padH;
        this->padW = padW;
        this->dilationH = dilationH;
        this->dilationW = dilationW;
        this->groups = groups;
        this->heightOut = heightOut;
        this->widthOut = widthOut;
        this->hasBias = hasBias;
        this->totalOutput = totalOutput;
        
        this->inChannelsPerGroup = inChannels / groups;
        this->outChannelsPerGroup = outChannels / groups;
        
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        this->outputPerBlock = (totalOutput + blockNum - 1) / blockNum;
        this->startIdx = blockIdx * this->outputPerBlock;
        if (this->startIdx + this->outputPerBlock > totalOutput) {
            this->outputPerBlock = (this->startIdx < totalOutput) ? (totalOutput - this->startIdx) : 0;
        }
        
        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * heightIn * widthIn);
        wGm.SetGlobalBuffer((__gm__ float *)weight, inChannels * outChannelsPerGroup * kernelH * kernelW);
        if (hasBias) {
            bGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        }
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize * outChannels * heightOut * widthOut);
    }
    
    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->outputPerBlock; i++) {
            uint32_t outIdx = this->startIdx + i;
            if (outIdx >= totalOutput) break;
            
            // Decode outIdx -> (n, oc, oh, ow)
            uint32_t ow = outIdx % widthOut;
            uint32_t tmp = outIdx / widthOut;
            uint32_t oh = tmp % heightOut;
            tmp = tmp / heightOut;
            uint32_t oc = tmp % outChannels;
            uint32_t n = tmp / outChannels;
            
            float sum = 0.0f;
            
            uint32_t g = oc / outChannelsPerGroup;
            uint32_t oc_in_group = oc % outChannelsPerGroup;
            
            for (uint32_t ic_local = 0; ic_local < inChannelsPerGroup; ic_local++) {
                uint32_t ic = g * inChannelsPerGroup + ic_local;
                for (uint32_t kh = 0; kh < kernelH; kh++) {
                    for (uint32_t kw = 0; kw < kernelW; kw++) {
                        int32_t ih_times_stride = (int32_t)oh + (int32_t)padH - (int32_t)(kh * dilationH);
                        int32_t iw_times_stride = (int32_t)ow + (int32_t)padW - (int32_t)(kw * dilationW);
                        
                        if (ih_times_stride < 0 || iw_times_stride < 0) continue;
                        if (ih_times_stride % (int32_t)strideH != 0) continue;
                        if (iw_times_stride % (int32_t)strideW != 0) continue;
                        
                        int32_t ih = ih_times_stride / (int32_t)strideH;
                        int32_t iw = iw_times_stride / (int32_t)strideW;
                        
                        if (ih < 0 || ih >= (int32_t)heightIn || iw < 0 || iw >= (int32_t)widthIn) continue;
                        
                        uint32_t xOffset = ((n * inChannels + ic) * heightIn + (uint32_t)ih) * widthIn + (uint32_t)iw;
                        // weight shape: [inChannels, outChannelsPerGroup, kernelH, kernelW]
                        uint32_t wOffset = ((ic * outChannelsPerGroup + oc_in_group) * kernelH + kh) * kernelW + kw;
                        
                        float xVal = xGm.GetValue(xOffset);
                        float wVal = wGm.GetValue(wOffset);
                        sum += xVal * wVal;
                    }
                }
            }
            
            if (hasBias) {
                sum += bGm.GetValue(oc);
            }
            
            yGm.SetValue(outIdx, sum);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    
    uint32_t batchSize, inChannels, outChannels;
    uint32_t heightIn, widthIn;
    uint32_t kernelH, kernelW;
    uint32_t strideH, strideW;
    uint32_t padH, padW;
    uint32_t dilationH, dilationW;
    uint32_t groups;
    uint32_t heightOut, widthOut;
    uint32_t hasBias;
    uint32_t totalOutput;
    uint32_t inChannelsPerGroup, outChannelsPerGroup;
    uint32_t outputPerBlock;
    uint32_t startIdx;
};

extern "C" __global__ __aicore__ void conv_transposed_2d_asymmetric_input_asymmetric_kernel_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose2d op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.heightIn, tiling_data.widthIn,
            tiling_data.kernelH, tiling_data.kernelW,
            tiling_data.strideH, tiling_data.strideW,
            tiling_data.padH, tiling_data.padW,
            tiling_data.outputPadH, tiling_data.outputPadW,
            tiling_data.dilationH, tiling_data.dilationW,
            tiling_data.groups, tiling_data.heightOut, tiling_data.widthOut,
            tiling_data.hasBias, tiling_data.totalOutput, tiling_data.tileNum);
    op.Process();
}
