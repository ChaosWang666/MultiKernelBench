kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelAddBiasBroadcast {
public:
    __aicore__ inline KernelAddBiasBroadcast() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t height, uint32_t width,
                                 uint32_t totalLength, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->height = height;
        this->width = width;
        this->hw = height * width;
        this->totalLength = totalLength;

        // Each block processes a portion of the total elements
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->blockLength = totalLength / blockNum;
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        // Align tileLength to 32 bytes (8 floats)
        if (this->tileLength % 8 != 0) {
            this->tileLength = (this->tileLength / 8) * 8;
        }
        if (this->tileLength == 0) {
            this->tileLength = 8;
        }

        uint32_t startOffset = blockIdx * this->blockLength;
        xGm.SetGlobalBuffer((__gm__ float *)x + startOffset, this->blockLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + startOffset, this->blockLength);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, channels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));

        this->startOffset = startOffset;
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();

        // Copy x tile
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);

        // Build bias tile: for each element in the tile, determine which channel it belongs to
        // Global index = startOffset + progress * tileLength + i
        // channel = (globalIndex / hw) % channels
        uint32_t baseGlobalIdx = this->startOffset + progress * this->tileLength;
        // We need to fill biasLocal with the correct bias values
        // Use a temporary approach: copy bias values element-wise via duplicate + arithmetic
        // For efficiency, we'll generate the bias tile on device

        // We'll fill biasLocal by iterating through the tile
        // Since we can't easily do per-element indexing on device, we use a simpler approach:
        // Process in chunks of hw elements that share the same channel

        // Simple approach: fill bias values for each position in the tile
        for (uint32_t i = 0; i < this->tileLength; i += 8) {
            uint32_t globalIdx = baseGlobalIdx + i;
            uint32_t c = (globalIdx / this->hw) % this->channels;
            // Check if the next 8 elements all belong to the same channel
            uint32_t endGlobalIdx = globalIdx + 7;
            uint32_t cEnd = (endGlobalIdx / this->hw) % this->channels;
            if (c == cEnd) {
                // All 8 elements in same channel, use Duplicate
                float biasVal = 0.0f;
                // Read bias value - we need to get it from global memory
                // Use DataCopy with size 8 from biasGm, but we need aligned access
                // Instead, let's use a simple scalar broadcast
                AscendC::Duplicate(biasLocal[i], biasGm.GetValue(c), 8);
            } else {
                // Elements span channel boundary, handle element by element in groups of 8
                for (uint32_t j = 0; j < 8; j++) {
                    uint32_t gIdx = globalIdx + j;
                    uint32_t ch = (gIdx / this->hw) % this->channels;
                    float bVal = biasGm.GetValue(ch);
                    biasLocal.SetValue(i + j, bVal);
                }
            }
        }

        inQueueX.EnQue(xLocal);
        inQueueBias.EnQue(biasLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::Add(zLocal, xLocal, biasLocal, this->tileLength);
        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueBias.FreeTensor(biasLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t height;
    uint32_t width;
    uint32_t hw;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t startOffset;
};

extern "C" __global__ __aicore__ void add_bias_four_dim_broadcast_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelAddBiasBroadcast op;
    op.Init(x, bias, z, tiling_data.batchSize, tiling_data.channels,
            tiling_data.height, tiling_data.width, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}
