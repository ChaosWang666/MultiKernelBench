kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelResizeWithAntialias {
public:
    __aicore__ inline KernelResizeWithAntialias() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t inputHeight, uint32_t inputWidth, uint32_t outputHeight, uint32_t outputWidth, uint32_t channel, uint32_t batchSize)
    {
        this->inputHeight = inputHeight;
        this->inputWidth = inputWidth;
        this->outputHeight = outputHeight;
        this->outputWidth = outputWidth;
        this->channel = channel;
        this->batchSize = batchSize;
        this->blockLength = inputHeight * inputWidth * channel * batchSize / AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->blockLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->blockLength * sizeof(DTYPE_Y));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->batchSize * this->channel;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->inputHeight * this->inputWidth], this->inputHeight * this->inputWidth);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        // Simplified bilinear interpolation with antialiasing
        // In real implementation, this would involve more complex logic
        for (uint32_t h = 0; h < this->outputHeight; ++h) {
            for (uint32_t w = 0; w < this->outputWidth; ++w) {
                float ratioH = (float)h / (float)this->outputHeight * (float)this->inputHeight;
                float ratioW = (float)w / (float)this->outputWidth * (float)this->inputWidth;
                uint32_t h0 = (uint32_t)ratioH;
                uint32_t w0 = (uint32_t)ratioW;
                uint32_t h1 = h0 + 1;
                uint32_t w1 = w0 + 1;
                if (h1 >= this->inputHeight) h1 = this->inputHeight - 1;
                if (w1 >= this->inputWidth) w1 = this->inputWidth - 1;
                float dh = ratioH - h0;
                float dw = ratioW - w0;
                float v00 = xLocal[h0 * this->inputWidth + w0];
                float v01 = xLocal[h0 * this->inputWidth + w1];
                float v10 = xLocal[h1 * this->inputWidth + w0];
                float v11 = xLocal[h1 * this->inputWidth + w1];
                float val = (1.0f - dh) * (1.0f - dw) * v00 +
                           (1.0f - dh) * dw * v01 +
                           dh * (1.0f - dw) * v10 +
                           dh * dw * v11;
                yLocal[h * this->outputWidth + w] = val;
            }
        }
        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[progress * this->outputHeight * this->outputWidth], yLocal, this->outputHeight * this->outputWidth);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t inputHeight;
    uint32_t inputWidth;
    uint32_t outputHeight;
    uint32_t outputWidth;
    uint32_t channel;
    uint32_t batchSize;
    uint32_t blockLength;
};

extern "C" __global__ __aicore__ void resize_with_antialias_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelResizeWithAntialias op;
    op.Init(x, y, tiling_data.inputHeight, tiling_data.inputWidth, tiling_data.outputHeight, tiling_data.outputWidth, tiling_data.channel, tiling_data.batchSize);
    op.Process();
}
