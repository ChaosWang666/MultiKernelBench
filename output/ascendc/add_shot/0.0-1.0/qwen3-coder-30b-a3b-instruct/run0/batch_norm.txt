kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelBatchNorm {
public:
    __aicore__ inline KernelBatchNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR bias, GM_ADDR mean, GM_ADDR variance,
                                GM_ADDR y, uint32_t batchSize, uint32_t channelNum, uint32_t height, uint32_t width, float eps)
    {
        this->batchSize = batchSize;
        this->channelNum = channelNum;
        this->height = height;
        this->width = width;
        this->eps = eps;
        this->totalElements = batchSize * channelNum * height * width;
        this->elementsPerChannel = height * width;
        this->elementsPerBlock = this->totalElements / AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, this->totalElements);
        scaleGm.SetGlobalBuffer((__gm__ DTYPE_SCALE *)scale, this->channelNum);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, this->channelNum);
        meanGm.SetGlobalBuffer((__gm__ DTYPE_MEAN *)mean, this->channelNum);
        varianceGm.SetGlobalBuffer((__gm__ DTYPE_VAR *)variance, this->channelNum);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, this->totalElements);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->elementsPerBlock * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueScale, BUFFER_NUM, this->channelNum * sizeof(DTYPE_SCALE));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, this->channelNum * sizeof(DTYPE_BIAS));
        pipe.InitBuffer(inQueueMean, BUFFER_NUM, this->channelNum * sizeof(DTYPE_MEAN));
        pipe.InitBuffer(inQueueVar, BUFFER_NUM, this->channelNum * sizeof(DTYPE_VAR));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->elementsPerBlock * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->batchSize * this->channelNum;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_SCALE> scaleLocal = inQueueScale.AllocTensor<DTYPE_SCALE>();
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = inQueueBias.AllocTensor<DTYPE_BIAS>();
        AscendC::LocalTensor<DTYPE_MEAN> meanLocal = inQueueMean.AllocTensor<DTYPE_MEAN>();
        AscendC::LocalTensor<DTYPE_VAR> varLocal = inQueueVar.AllocTensor<DTYPE_VAR>();

        uint32_t offset = progress * this->elementsPerChannel;
        AscendC::DataCopy(xLocal, xGm[offset], this->elementsPerChannel);
        AscendC::DataCopy(scaleLocal, scaleGm[progress % this->channelNum], 1);
        AscendC::DataCopy(biasLocal, biasGm[progress % this->channelNum], 1);
        AscendC::DataCopy(meanLocal, meanGm[progress % this->channelNum], 1);
        AscendC::DataCopy(varLocal, varianceGm[progress % this->channelNum], 1);

        inQueueX.EnQue(xLocal);
        inQueueScale.EnQue(scaleLocal);
        inQueueBias.EnQue(biasLocal);
        inQueueMean.EnQue(meanLocal);
        inQueueVar.EnQue(varLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_SCALE> scaleLocal = inQueueScale.DeQue<DTYPE_SCALE>();
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = inQueueBias.DeQue<DTYPE_BIAS>();
        AscendC::LocalTensor<DTYPE_MEAN> meanLocal = inQueueMean.DeQue<DTYPE_MEAN>();
        AscendC::LocalTensor<DTYPE_VAR> varLocal = inQueueVar.DeQue<DTYPE_VAR>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        // BatchNorm computation
        uint32_t channel = progress % this->channelNum;
        float scaleVal = scaleLocal[0];
        float biasVal = biasLocal[0];
        float meanVal = meanLocal[0];
        float varVal = varLocal[0];
        float invStd = 1.0f / sqrtf(varVal + this->eps);

        for (uint32_t i = 0; i < this->elementsPerChannel; i++) {
            float val = xLocal[i];
            float normalized = (val - meanVal) * invStd;
            yLocal[i] = normalized * scaleVal + biasVal;
        }

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueScale.FreeTensor(scaleLocal);
        inQueueBias.FreeTensor(biasLocal);
        inQueueMean.FreeTensor(meanLocal);
        inQueueVar.FreeTensor(varLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        uint32_t offset = progress * this->elementsPerChannel;
        AscendC::DataCopy(yGm[offset], yLocal, this->elementsPerChannel);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueScale, inQueueBias, inQueueMean, inQueueVar;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_SCALE> scaleGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_MEAN> meanGm;
    AscendC::GlobalTensor<DTYPE_VAR> varianceGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batchSize;
    uint32_t channelNum;
    uint32_t height;
    uint32_t width;
    uint32_t totalElements;
    uint32_t elementsPerChannel;
    uint32_t elementsPerBlock;
    float eps;
};

extern "C" __global__ __aicore__ void batch_norm_custom(
    GM_ADDR x, GM_ADDR scale, GM_ADDR bias, GM_ADDR mean, GM_ADDR variance, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelBatchNorm op;
    op.Init(x, scale, bias, mean, variance, y, tiling_data.batchSize, tiling_data.channelNum,
            tiling_data.height, tiling_data.width, tiling_data.eps);
    op.Process();
}
