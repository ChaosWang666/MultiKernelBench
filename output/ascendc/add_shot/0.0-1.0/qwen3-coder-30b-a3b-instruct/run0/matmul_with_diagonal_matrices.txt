kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelMatmulWithDiagonalMatrices {
public:
    __aicore__ inline KernelMatmulWithDiagonalMatrices() {}
    __aicore__ inline void Init(GM_ADDR diagonal, GM_ADDR matrix, GM_ADDR result, uint32_t n, uint32_t m, uint32_t tileNum)
    {
        this->n = n;
        this->m = m;
        this->tileNum = tileNum;
        this->blockLength = n / AscendC::GetBlockNum();
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        diagonalGm.SetGlobalBuffer((__gm__ float *)diagonal + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        matrixGm.SetGlobalBuffer((__gm__ float *)matrix + this->blockLength * AscendC::GetBlockIdx() * m, this->blockLength * m);
        resultGm.SetGlobalBuffer((__gm__ float *)result + this->blockLength * AscendC::GetBlockIdx() * m, this->blockLength * m);
        pipe.InitBuffer(inQueueDiag, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueMatrix, BUFFER_NUM, this->tileLength * m * sizeof(float));
        pipe.InitBuffer(outQueueResult, BUFFER_NUM, this->tileLength * m * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> diagLocal = inQueueDiag.AllocTensor<float>();
        AscendC::LocalTensor<float> matrixLocal = inQueueMatrix.AllocTensor<float>();
        AscendC::DataCopy(diagLocal, diagonalGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(matrixLocal, matrixGm[progress * this->tileLength * m], this->tileLength * m);
        inQueueDiag.EnQue(diagLocal);
        inQueueMatrix.EnQue(matrixLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> diagLocal = inQueueDiag.DeQue<float>();
        AscendC::LocalTensor<float> matrixLocal = inQueueMatrix.DeQue<float>();
        AscendC::LocalTensor<float> resultLocal = outQueueResult.AllocTensor<float>();
        // Perform element-wise multiplication: diag[i] * matrix[i][j]
        for (int32_t j = 0; j < m; j++) {
            AscendC::Mul(resultLocal[j], diagLocal, matrixLocal[j], this->tileLength);
        }
        outQueueResult.EnQue<float>(resultLocal);
        inQueueDiag.FreeTensor(diagLocal);
        inQueueMatrix.FreeTensor(matrixLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> resultLocal = outQueueResult.DeQue<float>();
        AscendC::DataCopy(resultGm[progress * this->tileLength * m], resultLocal, this->tileLength * m);
        outQueueResult.FreeTensor(resultLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueDiag, inQueueMatrix;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueResult;
    AscendC::GlobalTensor<float> diagonalGm;
    AscendC::GlobalTensor<float> matrixGm;
    AscendC::GlobalTensor<float> resultGm;
    uint32_t n;
    uint32_t m;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void matmul_with_diagonal_matrices_custom(GM_ADDR diagonal, GM_ADDR matrix, GM_ADDR result, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulWithDiagonalMatrices op;
    op.Init(diagonal, matrix, result, tiling_data.n, tiling_data.m, tiling_data.tileNum);
    op.Process();
}
