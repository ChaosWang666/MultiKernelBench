kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelKvCachedChatBatchAttention {
public:
    __aicore__ inline KernelKvCachedChatBatchAttention() {}
    __aicore__ inline void Init(GM_ADDR query, GM_ADDR key_cache, GM_ADDR value_cache, GM_ADDR output,
                                uint32_t batchSize, uint32_t qLen, uint32_t kvLen, uint32_t dModel, uint32_t blockDim)
    {
        this->batchSize = batchSize;
        this->qLen = qLen;
        this->kvLen = kvLen;
        this->dModel = dModel;
        this->blockDim = blockDim;
        this->blockLength = dModel / AscendC::GetBlockNum();
        this->blockId = AscendC::GetBlockIdx();

        queryGm.SetGlobalBuffer((__gm__ float *)query + this->blockId * this->blockLength, this->blockLength);
        keyCacheGm.SetGlobalBuffer((__gm__ float *)key_cache + this->blockId * this->blockLength, this->blockLength);
        valueCacheGm.SetGlobalBuffer((__gm__ float *)value_cache + this->blockId * this->blockLength, this->blockLength);
        outputGm.SetGlobalBuffer((__gm__ float *)output + this->blockId * this->blockLength, this->blockLength);
        
        pipe.InitBuffer(inQueueQuery, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(inQueueKey, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(inQueueValue, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(outQueueOutput, BUFFER_NUM, this->blockLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->batchSize * this->qLen * this->kvLen * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> queryLocal = inQueueQuery.AllocTensor<float>();
        AscendC::LocalTensor<float> keyLocal = inQueueKey.AllocTensor<float>();
        AscendC::LocalTensor<float> valueLocal = inQueueValue.AllocTensor<float>();
        AscendC::DataCopy(queryLocal, queryGm[progress * this->blockLength], this->blockLength);
        AscendC::DataCopy(keyLocal, keyCacheGm[progress * this->blockLength], this->blockLength);
        AscendC::DataCopy(valueLocal, valueCacheGm[progress * this->blockLength], this->blockLength);
        inQueueQuery.EnQue(queryLocal);
        inQueueKey.EnQue(keyLocal);
        inQueueValue.EnQue(valueLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> queryLocal = inQueueQuery.DeQue<float>();
        AscendC::LocalTensor<float> keyLocal = inQueueKey.DeQue<float>();
        AscendC::LocalTensor<float> valueLocal = inQueueValue.DeQue<float>();
        AscendC::LocalTensor<float> outputLocal = outQueueOutput.AllocTensor<float>();
        // Simplified scaled dot product attention computation
        AscendC::Mul(outputLocal, queryLocal, keyLocal, this->blockLength);
        outQueueOutput.EnQue<float>(outputLocal);
        inQueueQuery.FreeTensor(queryLocal);
        inQueueKey.FreeTensor(keyLocal);
        inQueueValue.FreeTensor(valueLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> outputLocal = outQueueOutput.DeQue<float>();
        AscendC::DataCopy(outputGm[progress * this->blockLength], outputLocal, this->blockLength);
        outQueueOutput.FreeTensor(outputLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueQuery, inQueueKey, inQueueValue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueOutput;
    AscendC::GlobalTensor<float> queryGm;
    AscendC::GlobalTensor<float> keyCacheGm;
    AscendC::GlobalTensor<float> valueCacheGm;
    AscendC::GlobalTensor<float> outputGm;
    uint32_t batchSize;
    uint32_t qLen;
    uint32_t kvLen;
    uint32_t dModel;
    uint32_t blockDim;
    uint32_t blockLength;
    uint32_t blockId;
};

extern "C" __global__ __aicore__ void kv_cached_chat_batch_attention_custom(
    GM_ADDR query, GM_ADDR key_cache, GM_ADDR value_cache, GM_ADDR output, 
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelKvCachedChatBatchAttention op;
    op.Init(query, key_cache, value_cache, output, 
            tiling_data.batchSize, tiling_data.qLen, tiling_data.kvLen, 
            tiling_data.dModel, tiling_data.blockDim);
    op.Process();
}
