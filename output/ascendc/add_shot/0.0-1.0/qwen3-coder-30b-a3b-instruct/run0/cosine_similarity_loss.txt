kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelCosineSimilarityLoss {
public:
    __aicore__ inline KernelCosineSimilarityLoss() {}
    __aicore__ inline void Init(GM_ADDR predictions, GM_ADDR targets, GM_ADDR loss, uint32_t batchSize, uint32_t featureDim)
    {
        this->batchSize = batchSize;
        this->featureDim = featureDim;
        this->blockLength = featureDim / AscendC::GetBlockNum();
        this->tileNum = 128;
        this->tileLength = this->blockLength / this->tileNum / BUFFER_NUM;

        predictionsGm.SetGlobalBuffer((__gm__ float *)predictions + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        targetsGm.SetGlobalBuffer((__gm__ float *)targets + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        lossGm.SetGlobalBuffer((__gm__ float *)loss, 1);
        pipe.InitBuffer(inQueuePred, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueTarget, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueDot, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueNormPred, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueNormTarget, BUFFER_NUM, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        // Initialize accumulators
        AscendC::LocalTensor<float> sumDot = AscendC::LocalTensor<float>::AllocLocalTensor<float>();
        AscendC::LocalTensor<float> sumNormPred = AscendC::LocalTensor<float>::AllocLocalTensor<float>();
        AscendC::LocalTensor<float> sumNormTarget = AscendC::LocalTensor<float>::AllocLocalTensor<float>();
        AscendC::Fill(sumDot, 0.0f);
        AscendC::Fill(sumNormPred, 0.0f);
        AscendC::Fill(sumNormTarget, 0.0f);
        
        int32_t loopCount = this->batchSize * this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i, sumDot, sumNormPred, sumNormTarget);
            CopyOut(i);
        }
        
        // Final reduction
        AscendC::ReduceSum(sumDot, sumDot, 1);
        AscendC::ReduceSum(sumNormPred, sumNormPred, 1);
        AscendC::ReduceSum(sumNormTarget, sumNormTarget, 1);
        
        // Compute final loss
        AscendC::LocalTensor<float> cosineSim = AscendC::LocalTensor<float>::AllocLocalTensor<float>();
        AscendC::Mul(cosineSim, sumDot, 1.0f / (sumNormPred * sumNormTarget + 1e-8f));
        AscendC::Sub(cosineSim, 1.0f, cosineSim);
        AscendC::ReduceSum(cosineSim, cosineSim, 1);
        AscendC::Div(cosineSim, cosineSim, static_cast<float>(this->batchSize));
        AscendC::DataCopy(lossGm, cosineSim, 1);
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> predLocal = inQueuePred.AllocTensor<float>();
        AscendC::LocalTensor<float> targetLocal = inQueueTarget.AllocTensor<float>();
        AscendC::DataCopy(predLocal, predictionsGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(targetLocal, targetsGm[progress * this->tileLength], this->tileLength);
        inQueuePred.EnQue(predLocal);
        inQueueTarget.EnQue(targetLocal);
    }
    __aicore__ inline void Compute(int32_t progress, AscendC::LocalTensor<float>& sumDot, AscendC::LocalTensor<float>& sumNormPred, AscendC::LocalTensor<float>& sumNormTarget)
    {
        AscendC::LocalTensor<float> predLocal = inQueuePred.DeQue<float>();
        AscendC::LocalTensor<float> targetLocal = inQueueTarget.DeQue<float>();
        AscendC::LocalTensor<float> dotLocal = outQueueDot.AllocTensor<float>();
        AscendC::LocalTensor<float> normPredLocal = outQueueNormPred.AllocTensor<float>();
        AscendC::LocalTensor<float> normTargetLocal = outQueueNormTarget.AllocTensor<float>();
        
        // Compute dot product
        AscendC::Mul(dotLocal, predLocal, targetLocal, this->tileLength);
        AscendC::ReduceSum(dotLocal, dotLocal, this->tileLength);
        AscendC::Add(sumDot, sumDot, dotLocal);
        
        // Compute norms
        AscendC::Mul(normPredLocal, predLocal, predLocal, this->tileLength);
        AscendC::ReduceSum(normPredLocal, normPredLocal, this->tileLength);
        AscendC::Add(sumNormPred, sumNormPred, normPredLocal);
        
        AscendC::Mul(normTargetLocal, targetLocal, targetLocal, this->tileLength);
        AscendC::ReduceSum(normTargetLocal, normTargetLocal, this->tileLength);
        AscendC::Add(sumNormTarget, sumNormTarget, normTargetLocal);
        
        inQueuePred.FreeTensor(predLocal);
        inQueueTarget.FreeTensor(targetLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        // No need to copy out intermediate results
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueuePred, inQueueTarget;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueDot, outQueueNormPred, outQueueNormTarget;
    AscendC::GlobalTensor<float> predictionsGm;
    AscendC::GlobalTensor<float> targetsGm;
    AscendC::GlobalTensor<float> lossGm;
    uint32_t batchSize;
    uint32_t featureDim;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void cosine_similarity_loss_custom(GM_ADDR predictions, GM_ADDR targets, GM_ADDR loss, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelCosineSimilarityLoss op;
    op.Init(predictions, targets, loss, tiling_data.batchSize, tiling_data.featureDim);
    op.Process();
}
