kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelSparseAttention {
public:
    __aicore__ inline KernelSparseAttention() {}
    __aicore__ inline void Init(GM_ADDR query, GM_ADDR key, GM_ADDR value, GM_ADDR attn_mask, GM_ADDR output,
                                uint32_t seqLen, uint32_t dModel, uint32_t numHeads, uint32_t batchSize, uint32_t windowSize)
    {
        this->seqLen = seqLen;
        this->dModel = dModel;
        this->numHeads = numHeads;
        this->batchSize = batchSize;
        this->windowSize = windowSize;
        this->headDim = dModel / numHeads;
        this->blockLength = seqLen / AscendC::GetBlockNum();
        this->blockLength = (this->blockLength + 15) / 16 * 16; // align to 16

        queryGm.SetGlobalBuffer((__gm__ float *)query + this->blockLength * AscendC::GetBlockIdx() * dModel, this->blockLength * dModel);
        keyGm.SetGlobalBuffer((__gm__ float *)key + this->blockLength * AscendC::GetBlockIdx() * dModel, this->blockLength * dModel);
        valueGm.SetGlobalBuffer((__gm__ float *)value + this->blockLength * AscendC::GetBlockIdx() * dModel, this->blockLength * dModel);
        attnMaskGm.SetGlobalBuffer((__gm__ bool *)attn_mask, seqLen * seqLen);
        outputGm.SetGlobalBuffer((__gm__ float *)output + this->blockLength * AscendC::GetBlockIdx() * dModel, this->blockLength * dModel);

        pipe.InitBuffer(inQueueQuery, BUFFER_NUM, this->blockLength * dModel * sizeof(float));
        pipe.InitBuffer(inQueueKey, BUFFER_NUM, this->blockLength * dModel * sizeof(float));
        pipe.InitBuffer(inQueueValue, BUFFER_NUM, this->blockLength * dModel * sizeof(float));
        pipe.InitBuffer(inQueueAttnMask, BUFFER_NUM, seqLen * seqLen * sizeof(bool));
        pipe.InitBuffer(outQueueOutput, BUFFER_NUM, this->blockLength * dModel * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t batchId = 0; batchId < batchSize; ++batchId) {
            for (uint32_t headId = 0; headId < numHeads; ++headId) {
                ComputeHead(batchId, headId);
            }
        }
    }

private:
    __aicore__ inline void ComputeHead(uint32_t batchId, uint32_t headId)
    {
        uint32_t headOffset = headId * headDim;
        uint32_t queryOffset = batchId * seqLen * dModel + headOffset;
        uint32_t keyOffset = batchId * seqLen * dModel + headOffset;
        uint32_t valueOffset = batchId * seqLen * dModel + headOffset;
        uint32_t outputOffset = batchId * seqLen * dModel + headOffset;

        for (uint32_t i = 0; i < seqLen; i += 16) {
            uint32_t tileEnd = (i + 16 > seqLen) ? seqLen : i + 16;
            ComputeTile(queryOffset, keyOffset, valueOffset, outputOffset, i, tileEnd);
        }
    }

    __aicore__ inline void ComputeTile(uint32_t queryOffset, uint32_t keyOffset, uint32_t valueOffset, uint32_t outputOffset,
                                       uint32_t startRow, uint32_t endRow)
    {
        AscendC::LocalTensor<float> queryLocal = inQueueQuery.AllocTensor<float>();
        AscendC::LocalTensor<float> keyLocal = inQueueKey.AllocTensor<float>();
        AscendC::LocalTensor<float> valueLocal = inQueueValue.AllocTensor<float>();
        AscendC::LocalTensor<bool> attnMaskLocal = inQueueAttnMask.AllocTensor<bool>();

        AscendC::DataCopy(queryLocal, queryGm[startRow * dModel], (endRow - startRow) * dModel);
        AscendC::DataCopy(keyLocal, keyGm[startRow * dModel], (endRow - startRow) * dModel);
        AscendC::DataCopy(valueLocal, valueGm[startRow * dModel], (endRow - startRow) * dModel);
        AscendC::DataCopy(attnMaskLocal, attnMaskGm[startRow * seqLen], (endRow - startRow) * seqLen);

        inQueueQuery.EnQue(queryLocal);
        inQueueKey.EnQue(keyLocal);
        inQueueValue.EnQue(valueLocal);
        inQueueAttnMask.EnQue(attnMaskLocal);

        AscendC::LocalTensor<float> outputLocal = outQueueOutput.AllocTensor<float>();
        ComputeAttention(queryLocal, keyLocal, valueLocal, attnMaskLocal, outputLocal, startRow, endRow);
        outQueueOutput.EnQue(outputLocal);

        AscendC::DataCopy(outputGm[startRow * dModel], outputLocal, (endRow - startRow) * dModel);

        inQueueQuery.FreeTensor(queryLocal);
        inQueueKey.FreeTensor(keyLocal);
        inQueueValue.FreeTensor(valueLocal);
        inQueueAttnMask.FreeTensor(attnMaskLocal);
        outQueueOutput.FreeTensor(outputLocal);
    }

    __aicore__ inline void ComputeAttention(AscendC::LocalTensor<float>& queryLocal,
                                            AscendC::LocalTensor<float>& keyLocal,
                                            AscendC::LocalTensor<float>& valueLocal,
                                            AscendC::LocalTensor<bool>& attnMaskLocal,
                                            AscendC::LocalTensor<float>& outputLocal,
                                            uint32_t startRow, uint32_t endRow)
    {
        for (uint32_t row = startRow; row < endRow; ++row) {
            for (uint32_t col = 0; col < seqLen; ++col) {
                if (attnMaskLocal[row * seqLen + col]) {
                    // Skip masked positions
                    continue;
                }
                float score = 0.0f;
                for (uint32_t k = 0; k < headDim; ++k) {
                    score += queryLocal[row * headDim + k] * keyLocal[col * headDim + k];
                }
                score /= sqrtf(static_cast<float>(headDim));
                for (uint32_t k = 0; k < headDim; ++k) {
                    outputLocal[row * headDim + k] += score * valueLocal[col * headDim + k];
                }
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueQuery, inQueueKey, inQueueValue, inQueueAttnMask;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueOutput;
    AscendC::GlobalTensor<float> queryGm, keyGm, valueGm, outputGm;
    AscendC::GlobalTensor<bool> attnMaskGm;
    uint32_t seqLen;
    uint32_t dModel;
    uint32_t numHeads;
    uint32_t batchSize;
    uint32_t windowSize;
    uint32_t headDim;
    uint32_t blockLength;
};

extern "C" __global__ __aicore__ void sparse_attention_custom(GM_ADDR query, GM_ADDR key, GM_ADDR value, GM_ADDR attn_mask, GM_ADDR output, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSparseAttention op;
    op.Init(query, key, value, attn_mask, output,
            tiling_data.seqLen, tiling_data.dModel, tiling_data.numHeads, tiling_data.batchSize, tiling_data.windowSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, d_model=256, num_heads=4, window_size=32, max_seq_len=4096):
        super().__init__()
        self.mha = torch.nn.MultiheadAttention(d_model, num_heads, batch_first=True)
        self.window_size = window_size
        self.max_seq_len = max_seq_len

        # Precompute sparse mask
        mask = torch.ones(max_seq_len, max_seq_len)
        for i in range(max_seq_len):
            start = max(0, i - window_size)
            end = min(max_seq_len, i + window_size)
            mask[i, start:end] = 0
        mask = mask.bool()

        # Register as buffer 
        self.register_buffer("attn_mask", mask, persistent=False)

    def forward(self, x):
        """
        Args:
            x: (batch, seq_len, d_model)
        Returns:
            output: (batch, seq_len, d_model)
        """
        seq_len = x.size(1)
        attn_mask = self.attn_mask[:seq_len, :seq_len]
        # Use custom kernel instead of default MHA
        out = custom_ops_lib.sparse_attention_custom(x, x, x, attn_mask)
        return out
'''
