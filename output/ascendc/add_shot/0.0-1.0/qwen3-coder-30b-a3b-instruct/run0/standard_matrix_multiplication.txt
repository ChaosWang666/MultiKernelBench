kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelStandardMatrixMultiplication {
public:
    __aicore__ inline KernelStandardMatrixMultiplication() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR c, uint32_t m, uint32_t k, uint32_t n, uint32_t tileM, uint32_t tileK, uint32_t tileN)
    {
        this->m = m;
        this->k = k;
        this->n = n;
        this->tileM = tileM;
        this->tileK = tileK;
        this->tileN = tileN;
        this->blockM = m / AscendC::GetBlockNum();
        this->blockN = n / AscendC::GetBlockNum();

        aGm.SetGlobalBuffer((__gm__ DTYPE_A *)a, m * k);
        bGm.SetGlobalBuffer((__gm__ DTYPE_B *)b, k * n);
        cGm.SetGlobalBuffer((__gm__ DTYPE_C *)c, m * n);
        pipe.InitBuffer(inQueueA, BUFFER_NUM, tileM * tileK * sizeof(DTYPE_A));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, tileK * tileN * sizeof(DTYPE_B));
        pipe.InitBuffer(outQueueC, BUFFER_NUM, tileM * tileN * sizeof(DTYPE_C));
    }
    __aicore__ inline void Process()
    {
        uint32_t blockId = AscendC::GetBlockIdx();
        uint32_t mStart = blockId * blockM;
        uint32_t nStart = blockId * blockN;
        uint32_t mEnd = (blockId == AscendC::GetBlockNum() - 1) ? m : (blockId + 1) * blockM;
        uint32_t nEnd = (blockId == AscendC::GetBlockNum() - 1) ? n : (blockId + 1) * blockN;
        uint32_t mSize = mEnd - mStart;
        uint32_t nSize = nEnd - nStart;

        for (uint32_t mIter = 0; mIter < mSize; mIter += tileM) {
            for (uint32_t nIter = 0; nIter < nSize; nIter += tileN) {
                for (uint32_t kIter = 0; kIter < k; kIter += tileK) {
                    CopyIn(mIter, nIter, kIter);
                    Compute(mIter, nIter, kIter);
                    CopyOut(mIter, nIter, kIter);
                }
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t mOffset, uint32_t nOffset, uint32_t kOffset)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.AllocTensor<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.AllocTensor<DTYPE_B>();
        AscendC::DataCopy(aLocal, aGm[mOffset * k + kOffset], tileM * tileK);
        AscendC::DataCopy(bLocal, bGm[kOffset * n + nOffset], tileK * tileN);
        inQueueA.EnQue(aLocal);
        inQueueB.EnQue(bLocal);
    }
    __aicore__ inline void Compute(uint32_t mOffset, uint32_t nOffset, uint32_t kOffset)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.DeQue<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.DeQue<DTYPE_B>();
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.AllocTensor<DTYPE_C>();
        AscendC::MatMul(cLocal, aLocal, bLocal, tileM, tileK, tileN);
        outQueueC.EnQue<DTYPE_C>(cLocal);
        inQueueA.FreeTensor(aLocal);
        inQueueB.FreeTensor(bLocal);
    }
    __aicore__ inline void CopyOut(uint32_t mOffset, uint32_t nOffset, uint32_t kOffset)
    {
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.DeQue<DTYPE_C>();
        AscendC::DataCopy(cGm[mOffset * n + nOffset], cLocal, tileM * tileN);
        outQueueC.FreeTensor(cLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::GlobalTensor<DTYPE_A> aGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_C> cGm;
    uint32_t m, k, n;
    uint32_t tileM, tileK, tileN;
    uint32_t blockM, blockN;
};

extern "C" __global__ __aicore__ void standard_matrix_multiplication_custom(GM_ADDR a, GM_ADDR b, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelStandardMatrixMultiplication op;
    op.Init(a, b, c, tiling_data.m, tiling_data.k, tiling_data.n, tiling_data.tileM, tiling_data.tileK, tiling_data.tileN);
    op.Process();
}
