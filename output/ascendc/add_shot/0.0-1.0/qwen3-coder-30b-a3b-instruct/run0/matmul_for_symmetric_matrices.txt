kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelMatmulForSymmetricMatrices {
public:
    __aicore__ inline KernelMatmulForSymmetricMatrices() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR c, uint32_t n, uint32_t tileNum)
    {
        this->n = n;
        this->tileNum = tileNum;
        this->tileLength = this->n / tileNum / BUFFER_NUM;

        aGm.SetGlobalBuffer((__gm__ DTYPE_A *)a, this->n * this->n);
        bGm.SetGlobalBuffer((__gm__ DTYPE_B *)b, this->n * this->n);
        cGm.SetGlobalBuffer((__gm__ DTYPE_C *)c, this->n * this->n);
        pipe.InitBuffer(inQueueA, BUFFER_NUM, this->tileLength * this->n * sizeof(DTYPE_A));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, this->tileLength * this->n * sizeof(DTYPE_B));
        pipe.InitBuffer(outQueueC, BUFFER_NUM, this->tileLength * this->n * sizeof(DTYPE_C));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.AllocTensor<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.AllocTensor<DTYPE_B>();
        AscendC::DataCopy(aLocal, aGm[progress * this->tileLength * this->n], this->tileLength * this->n);
        AscendC::DataCopy(bLocal, bGm[progress * this->tileLength * this->n], this->tileLength * this->n);
        inQueueA.EnQue(aLocal);
        inQueueB.EnQue(bLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.DeQue<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.DeQue<DTYPE_B>();
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.AllocTensor<DTYPE_C>();
        AscendC::MatMul(cLocal, aLocal, bLocal, this->tileLength, this->n, this->n);
        outQueueC.EnQue<DTYPE_C>(cLocal);
        inQueueA.FreeTensor(aLocal);
        inQueueB.FreeTensor(bLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.DeQue<DTYPE_C>();
        AscendC::DataCopy(cGm[progress * this->tileLength * this->n], cLocal, this->tileLength * this->n);
        outQueueC.FreeTensor(cLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::GlobalTensor<DTYPE_A> aGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_C> cGm;
    uint32_t n;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void matmul_for_symmetric_matrices_custom(GM_ADDR a, GM_ADDR b, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulForSymmetricMatrices op;
    op.Init(a, b, c, tiling_data.n, tiling_data.tileNum);
    op.Process();
}
