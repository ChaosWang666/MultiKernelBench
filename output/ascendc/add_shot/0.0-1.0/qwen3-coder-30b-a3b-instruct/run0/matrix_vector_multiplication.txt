kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelMatrixVectorMultiplication {
public:
    __aicore__ inline KernelMatrixVectorMultiplication() {}
    __aicore__ inline void Init(GM_ADDR matrix, GM_ADDR vector, GM_ADDR result, uint32_t m, uint32_t k, uint32_t tileNum)
    {
        this->m = m;
        this->k = k;
        this->tileNum = tileNum;
        this->blockLength = m / AscendC::GetBlockNum();
        this->tileLength = k / tileNum / BUFFER_NUM;

        matrixGm.SetGlobalBuffer((__gm__ DTYPE_MATRIX *)matrix + this->blockLength * AscendC::GetBlockIdx(), this->blockLength * k);
        vectorGm.SetGlobalBuffer((__gm__ DTYPE_VECTOR *)vector, k);
        resultGm.SetGlobalBuffer((__gm__ DTYPE_RESULT *)result + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueMatrix, BUFFER_NUM, this->tileLength * sizeof(DTYPE_MATRIX));
        pipe.InitBuffer(inQueueVector, BUFFER_NUM, this->tileLength * sizeof(DTYPE_VECTOR));
        pipe.InitBuffer(outQueueResult, BUFFER_NUM, this->tileLength * sizeof(DTYPE_RESULT));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_MATRIX> matrixLocal = inQueueMatrix.AllocTensor<DTYPE_MATRIX>();
        AscendC::LocalTensor<DTYPE_VECTOR> vectorLocal = inQueueVector.AllocTensor<DTYPE_VECTOR>();
        AscendC::DataCopy(matrixLocal, matrixGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(vectorLocal, vectorGm[progress * this->tileLength], this->tileLength);
        inQueueMatrix.EnQue(matrixLocal);
        inQueueVector.EnQue(vectorLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_MATRIX> matrixLocal = inQueueMatrix.DeQue<DTYPE_MATRIX>();
        AscendC::LocalTensor<DTYPE_VECTOR> vectorLocal = inQueueVector.DeQue<DTYPE_VECTOR>();
        AscendC::LocalTensor<DTYPE_RESULT> resultLocal = outQueueResult.AllocTensor<DTYPE_RESULT>();
        AscendC::MatMul(resultLocal, matrixLocal, vectorLocal, this->tileLength);
        outQueueResult.EnQue<DTYPE_RESULT>(resultLocal);
        inQueueMatrix.FreeTensor(matrixLocal);
        inQueueVector.FreeTensor(vectorLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_RESULT> resultLocal = outQueueResult.DeQue<DTYPE_RESULT>();
        AscendC::DataCopy(resultGm[progress * this->tileLength], resultLocal, this->tileLength);
        outQueueResult.FreeTensor(resultLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueMatrix, inQueueVector;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueResult;
    AscendC::GlobalTensor<DTYPE_MATRIX> matrixGm;
    AscendC::GlobalTensor<DTYPE_VECTOR> vectorGm;
    AscendC::GlobalTensor<DTYPE_RESULT> resultGm;
    uint32_t m;
    uint32_t k;
    uint32_t tileNum;
    uint32_t blockLength;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void matrix_vector_multiplication_custom(GM_ADDR matrix, GM_ADDR vector, GM_ADDR result, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatrixVectorMultiplication op;
    op.Init(matrix, vector, result, tiling_data.m, tiling_data.k, tiling_data.tileNum);
    op.Process();
}
