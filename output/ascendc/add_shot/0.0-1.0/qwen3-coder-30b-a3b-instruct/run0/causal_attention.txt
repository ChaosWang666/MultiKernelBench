kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelCausalAttention {
public:
    __aicore__ inline KernelCausalAttention() {}
    __aicore__ inline void Init(GM_ADDR query, GM_ADDR key, GM_ADDR value, GM_ADDR attn_mask, GM_ADDR output,
                                uint32_t batchSize, uint32_t seqLen, uint32_t headNum, uint32_t headDim, uint32_t tileSeqLen)
    {
        this->batchSize = batchSize;
        this->seqLen = seqLen;
        this->headNum = headNum;
        this->headDim = headDim;
        this->tileSeqLen = tileSeqLen;
        this->blockLength = seqLen / AscendC::GetBlockNum();
        this->blockOffset = this->blockLength * AscendC::GetBlockIdx();

        queryGm.SetGlobalBuffer((__gm__ float *)query + this->blockOffset, this->blockLength * headNum * headDim);
        keyGm.SetGlobalBuffer((__gm__ float *)key + this->blockOffset, this->blockLength * headNum * headDim);
        valueGm.SetGlobalBuffer((__gm__ float *)value + this->blockOffset, this->blockLength * headNum * headDim);
        attnMaskGm.SetGlobalBuffer((__gm__ bool *)attn_mask, seqLen * seqLen);
        outputGm.SetGlobalBuffer((__gm__ float *)output + this->blockOffset, this->blockLength * headNum * headDim);

        pipe.InitBuffer(inQueueQuery, BUFFER_NUM, this->blockLength * headNum * headDim * sizeof(float));
        pipe.InitBuffer(inQueueKey, BUFFER_NUM, this->blockLength * headNum * headDim * sizeof(float));
        pipe.InitBuffer(inQueueValue, BUFFER_NUM, this->blockLength * headNum * headDim * sizeof(float));
        pipe.InitBuffer(inQueueAttnMask, BUFFER_NUM, seqLen * seqLen * sizeof(bool));
        pipe.InitBuffer(outQueueOutput, BUFFER_NUM, this->blockLength * headNum * headDim * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t head = 0; head < headNum; ++head) {
            for (uint32_t seq = 0; seq < seqLen; seq += tileSeqLen) {
                CopyIn(head, seq);
                Compute(head, seq);
                CopyOut(head, seq);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t head, uint32_t seqStart)
    {
        AscendC::LocalTensor<float> queryLocal = inQueueQuery.AllocTensor<float>();
        AscendC::LocalTensor<float> keyLocal = inQueueKey.AllocTensor<float>();
        AscendC::LocalTensor<float> valueLocal = inQueueValue.AllocTensor<float>();
        AscendC::LocalTensor<bool> attnMaskLocal = inQueueAttnMask.AllocTensor<bool>();

        uint32_t offsetQuery = head * seqLen * headDim + seqStart * headDim;
        uint32_t offsetKey = head * seqLen * headDim + seqStart * headDim;
        uint32_t offsetValue = head * seqLen * headDim + seqStart * headDim;
        uint32_t offsetAttnMask = seqStart * seqLen;

        AscendC::DataCopy(queryLocal, queryGm[offsetQuery], this->tileSeqLen * headDim);
        AscendC::DataCopy(keyLocal, keyGm[offsetKey], this->tileSeqLen * headDim);
        AscendC::DataCopy(valueLocal, valueGm[offsetValue], this->tileSeqLen * headDim);
        AscendC::DataCopy(attnMaskLocal, attnMaskGm[offsetAttnMask], this->tileSeqLen * seqLen);

        inQueueQuery.EnQue(queryLocal);
        inQueueKey.EnQue(keyLocal);
        inQueueValue.EnQue(valueLocal);
        inQueueAttnMask.EnQue(attnMaskLocal);
    }
    __aicore__ inline void Compute(uint32_t head, uint32_t seqStart)
    {
        AscendC::LocalTensor<float> queryLocal = inQueueQuery.DeQue<float>();
        AscendC::LocalTensor<float> keyLocal = inQueueKey.DeQue<float>();
        AscendC::LocalTensor<float> valueLocal = inQueueValue.DeQue<float>();
        AscendC::LocalTensor<bool> attnMaskLocal = inQueueAttnMask.DeQue<bool>();
        AscendC::LocalTensor<float> outputLocal = outQueueOutput.AllocTensor<float>();

        // Simplified attention computation
        uint32_t outputOffset = head * seqLen * headDim + seqStart * headDim;
        for (uint32_t i = 0; i < tileSeqLen; ++i) {
            for (uint32_t j = 0; j < headDim; ++j) {
                float sum = 0.0f;
                for (uint32_t k = 0; k < seqLen; ++k) {
                    if (attnMaskLocal[i * seqLen + k]) {
                        sum += queryLocal[i * headDim + j] * keyLocal[k * headDim + j];
                    }
                }
                outputLocal[i * headDim + j] = sum;
            }
        }

        outQueueOutput.EnQue<float>(outputLocal);
        inQueueQuery.FreeTensor(queryLocal);
        inQueueKey.FreeTensor(keyLocal);
        inQueueValue.FreeTensor(valueLocal);
        inQueueAttnMask.FreeTensor(attnMaskLocal);
    }
    __aicore__ inline void CopyOut(uint32_t head, uint32_t seqStart)
    {
        AscendC::LocalTensor<float> outputLocal = outQueueOutput.DeQue<float>();
        uint32_t outputOffset = head * seqLen * headDim + seqStart * headDim;
        AscendC::DataCopy(outputGm[outputOffset], outputLocal, this->tileSeqLen * headDim);
        outQueueOutput.FreeTensor(outputLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueQuery, inQueueKey, inQueueValue, inQueueAttnMask;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueOutput;
    AscendC::GlobalTensor<float> queryGm, keyGm, valueGm, outputGm;
    AscendC::GlobalTensor<bool> attnMaskGm;
    uint32_t batchSize;
    uint32_t seqLen;
    uint32_t headNum;
    uint32_t headDim;
    uint32_t tileSeqLen;
    uint32_t blockLength;
    uint32_t blockOffset;
};

extern "C" __global__ __aicore__ void causal_attention_custom(
    GM_ADDR query, GM_ADDR key, GM_ADDR value, GM_ADDR attn_mask, GM_ADDR output, 
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelCausalAttention op;
    op.Init(query, key, value, attn_mask, output, 
            tiling_data.batchSize, tiling_data.seqLen, tiling_data.headNum, tiling_data.headDim, tiling_data.tileSeqLen);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, d_model=256, num_heads=4, max_seq_len=4096):
        super().__init__()
        self.d_model = d_model
        self.num_heads = num_heads
        self.max_seq_len = max_seq_len

        # Precompute causal mask (1 for disallowed positions)
        mask = torch.triu(torch.ones(max_seq_len, max_seq_len), diagonal=1).bool()
        self.register_buffer("causal_mask", mask, persistent=False)

    def forward(self, x):
        """
        Args:
            x: (batch, seq_len, d_model)
        Returns:
            output: (batch, seq_len, d_model)
        """
        batch_size, seq_len, _ = x.shape
        attn_mask = self.causal_mask[:seq_len, :seq_len]
        # Use custom kernel instead of MultiheadAttention
        query = x.view(batch_size, seq_len, self.num_heads, self.d_model // self.num_heads).transpose(1, 2)
        key = x.view(batch_size, seq_len, self.num_heads, self.d_model // self.num_heads).transpose(1, 2)
        value = x.view(batch_size, seq_len, self.num_heads, self.d_model // self.num_heads).transpose(1, 2)
        out = custom_ops_lib.causal_attention_custom(query, key, value, attn_mask)
        out = out.transpose(1, 2).contiguous().view(batch_size, seq_len, self.d_model)
        return out
'''
