kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelHingeLoss {
public:
    __aicore__ inline KernelHingeLoss() {}
    __aicore__ inline void Init(GM_ADDR predictions, GM_ADDR targets, GM_ADDR loss, uint32_t totalLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        predictionsGm.SetGlobalBuffer((__gm__ DTYPE_PREDICTIONS *)predictions + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        targetsGm.SetGlobalBuffer((__gm__ DTYPE_TARGETS *)targets + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        lossGm.SetGlobalBuffer((__gm__ DTYPE_LOSS *)loss + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueuePred, BUFFER_NUM, this->tileLength * sizeof(DTYPE_PREDICTIONS));
        pipe.InitBuffer(inQueueTarget, BUFFER_NUM, this->tileLength * sizeof(DTYPE_TARGETS));
        pipe.InitBuffer(outQueueLoss, BUFFER_NUM, this->tileLength * sizeof(DTYPE_LOSS));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_PREDICTIONS> predLocal = inQueuePred.AllocTensor<DTYPE_PREDICTIONS>();
        AscendC::LocalTensor<DTYPE_TARGETS> targetLocal = inQueueTarget.AllocTensor<DTYPE_TARGETS>();
        AscendC::DataCopy(predLocal, predictionsGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(targetLocal, targetsGm[progress * this->tileLength], this->tileLength);
        inQueuePred.EnQue(predLocal);
        inQueueTarget.EnQue(targetLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_PREDICTIONS> predLocal = inQueuePred.DeQue<DTYPE_PREDICTIONS>();
        AscendC::LocalTensor<DTYPE_TARGETS> targetLocal = inQueueTarget.DeQue<DTYPE_TARGETS>();
        AscendC::LocalTensor<DTYPE_LOSS> lossLocal = outQueueLoss.AllocTensor<DTYPE_LOSS>();
        AscendC::Mul(lossLocal, predLocal, targetLocal, this->tileLength);
        AscendC::Sub(lossLocal, lossLocal, 1.0f, this->tileLength);
        AscendC::Neg(lossLocal, lossLocal, this->tileLength);
        AscendC::Max(lossLocal, lossLocal, 0.0f, this->tileLength);
        outQueueLoss.EnQue<DTYPE_LOSS>(lossLocal);
        inQueuePred.FreeTensor(predLocal);
        inQueueTarget.FreeTensor(targetLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_LOSS> lossLocal = outQueueLoss.DeQue<DTYPE_LOSS>();
        AscendC::DataCopy(lossGm[progress * this->tileLength], lossLocal, this->tileLength);
        outQueueLoss.FreeTensor(lossLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueuePred, inQueueTarget;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueLoss;
    AscendC::GlobalTensor<DTYPE_PREDICTIONS> predictionsGm;
    AscendC::GlobalTensor<DTYPE_TARGETS> targetsGm;
    AscendC::GlobalTensor<DTYPE_LOSS> lossGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void hinge_loss_custom(GM_ADDR predictions, GM_ADDR targets, GM_ADDR loss, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelHingeLoss op;
    op.Init(predictions, targets, loss, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}
