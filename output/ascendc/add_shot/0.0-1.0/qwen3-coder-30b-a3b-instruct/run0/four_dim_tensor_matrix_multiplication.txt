kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelFourDimTensorMatrixMultiplication {
public:
    __aicore__ inline KernelFourDimTensorMatrixMultiplication() {}
    __aicore__ inline void Init(GM_ADDR A, GM_ADDR B, GM_ADDR C, uint32_t b, uint32_t i, uint32_t j, uint32_t l, uint32_t k, uint32_t tileL, uint32_t tileK)
    {
        this->b = b;
        this->i = i;
        this->j = j;
        this->l = l;
        this->k = k;
        this->tileL = tileL;
        this->tileK = tileK;

        this->blockLength = b * i * j * l / AscendC::GetBlockNum();
        this->blockLengthK = b * i * j * k / AscendC::GetBlockNum();

        A_Gm.SetGlobalBuffer((__gm__ float *)A + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        B_Gm.SetGlobalBuffer((__gm__ float *)B, l * k);
        C_Gm.SetGlobalBuffer((__gm__ float *)C + this->blockLengthK * AscendC::GetBlockIdx(), this->blockLengthK);

        pipe.InitBuffer(inQueueA, BUFFER_NUM, this->tileL * sizeof(float));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, this->tileK * sizeof(float));
        pipe.InitBuffer(outQueueC, BUFFER_NUM, this->tileK * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t bi = 0; bi < b; ++bi) {
            for (uint32_t ii = 0; ii < i; ++ii) {
                for (uint32_t jj = 0; jj < j; ++jj) {
                    for (uint32_t ll = 0; ll < l; ll += tileL) {
                        for (uint32_t kk = 0; kk < k; kk += tileK) {
                            CopyIn(bi, ii, jj, ll, kk);
                            Compute(bi, ii, jj, ll, kk);
                            CopyOut(bi, ii, jj, ll, kk);
                        }
                    }
                }
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t bi, uint32_t ii, uint32_t jj, uint32_t ll, uint32_t kk)
    {
        uint32_t offsetA = bi * i * j * l + ii * j * l + jj * l + ll;
        uint32_t offsetB = ll * k + kk;
        uint32_t lenA = (ll + tileL > l) ? (l - ll) : tileL;
        uint32_t lenB = (kk + tileK > k) ? (k - kk) : tileK;

        AscendC::LocalTensor<float> ALocal = inQueueA.AllocTensor<float>();
        AscendC::LocalTensor<float> BLocal = inQueueB.AllocTensor<float>();
        AscendC::DataCopy(ALocal, A_Gm[offsetA], lenA);
        AscendC::DataCopy(BLocal, B_Gm[offsetB], lenB);
        inQueueA.EnQue(ALocal);
        inQueueB.EnQue(BLocal);
    }
    __aicore__ inline void Compute(uint32_t bi, uint32_t ii, uint32_t jj, uint32_t ll, uint32_t kk)
    {
        AscendC::LocalTensor<float> ALocal = inQueueA.DeQue<float>();
        AscendC::LocalTensor<float> BLocal = inQueueB.DeQue<float>();
        AscendC::LocalTensor<float> CLocal = outQueueC.AllocTensor<float>();
        AscendC::MatMul(CLocal, ALocal, BLocal, ll, kk, tileL, tileK, false, false);
        outQueueC.EnQue<float>(CLocal);
        inQueueA.FreeTensor(ALocal);
        inQueueB.FreeTensor(BLocal);
    }
    __aicore__ inline void CopyOut(uint32_t bi, uint32_t ii, uint32_t jj, uint32_t ll, uint32_t kk)
    {
        uint32_t offsetC = bi * i * j * k + ii * j * k + jj * k + kk;
        AscendC::LocalTensor<float> CLocal = outQueueC.DeQue<float>();
        AscendC::DataCopy(C_Gm[offsetC], CLocal, tileK);
        outQueueC.FreeTensor(CLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::GlobalTensor<float> A_Gm;
    AscendC::GlobalTensor<float> B_Gm;
    AscendC::GlobalTensor<float> C_Gm;
    uint32_t b;
    uint32_t i;
    uint32_t j;
    uint32_t l;
    uint32_t k;
    uint32_t tileL;
    uint32_t tileK;
    uint32_t blockLength;
    uint32_t blockLengthK;
};

extern "C" __global__ __aicore__ void four_dim_tensor_matrix_multiplication_custom(GM_ADDR A, GM_ADDR B, GM_ADDR C, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFourDimTensorMatrixMultiplication op;
    op.Init(A, B, C, tiling_data.b, tiling_data.i, tiling_data.j, tiling_data.l, tiling_data.k, tiling_data.tileL, tiling_data.tileK);
    op.Process();
}
