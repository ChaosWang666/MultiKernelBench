        this->inChannels = inChannels;
        this->outChannels = outChannels;

        this->channelsPerGroup = inChannels / groups;

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, groups=3):
        super(ModelNew, self).__init__()
        assert out_channels % 4 == 0
        self.mid_channels = out_channels // 4
        self.groups = groups
        
        self.conv1 = torch.nn.Conv2d(in_channels, self.mid_channels, kernel_size=1, stride=1, padding=0, groups=groups, bias=False)
        self.bn1 = torch.nn.BatchNorm2d(self.mid_channels)
        self.conv2 = torch.nn.Conv2d(self.mid_channels, self.mid_channels, kernel_size=3, stride=1, padding=1, groups=self.mid_channels, bias=False)
        self.bn2 = torch.nn.BatchNorm2d(self.mid_channels)
        self.conv3 = torch.nn.Conv2d(self.mid_channels, out_channels, kernel_size=1, stride=1, padding=0, groups=groups, bias=False)
        self.bn3 = torch.nn.BatchNorm2d(out_channels)
        self.shuffle = ChannelShuffle(groups)
        
        if in_channels == out_channels:
            self.shortcut = torch.nn.Sequential()
        else:
            self.shortcut = torch.nn.Sequential(
                torch.nn.Conv2d(in_channels, out_channels, kernel_size=1, stride=1, padding=0, bias=False),
                torch.nn.BatchNorm2d(out_channels)
            )

    def forward(self, x):
        out = torch.nn.functional.relu(self.bn1(self.conv1(x)))
        out = self.bn2(self.conv2(out))
        out = self.shuffle(out)
        out = torch.nn.functional.relu(self.bn3(self.conv3(out)))
        
        out += self.shortcut(x)
        return out

class ChannelShuffle(torch.nn.Module):
    def __init__(self, groups):
        super(ChannelShuffle, self).__init__()
        self.groups = groups
    
    def forward(self, x):
        batch_size, channels, height, width = x.size()
        channels_per_group = channels // self.groups
        
        x = x.view(batch_size, self.groups, channels_per_group, height, width)
        x = x.transpose(1, 2).contiguous()
        x = x.view(batch_size, -1, height, width)
        
        return x

# Replace the original forward pass with the custom op
def forward_with_custom_op(self, x):
    # This is a simplified version that assumes all parameters are available
    # In practice, you'd need to extract weights/biases from the modules
    # and pass them to the custom op
    return custom_ops_lib.shufflenet_unit_custom(
        x,
        self.conv1.weight,
        self.conv1.bias,
        self.bn1.weight,
        self.bn1.bias,
        self.conv2.weight,
        self.conv2.bias,
        self.bn2.weight,
        self.bn2.bias,
        self.conv3.weight,
        self.conv3.bias,
        self.bn3.weight,
        self.bn3.bias,
        self.shortcut[0].weight if len(self.shortcut) > 0 else None,
        self.shortcut[0].bias if len(self.shortcut) > 0 and hasattr(self.shortcut[0], 'bias') else None,
        self.shortcut[1].weight if len(self.shortcut) > 1 else None,
        self.shortcut[1].bias if len(self.shortcut) > 1 and hasattr(self.shortcut[1], 'bias') else None
    )

# Monkey patch the forward method
ModelNew.forward = forward_with_custom_op
'''
