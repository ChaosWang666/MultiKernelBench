kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelGather {
public:
    __aicore__ inline KernelGather() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR index, GM_ADDR z, uint32_t totalLength, uint32_t tileNum, uint32_t dimSize, uint32_t gatherSize)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->dimSize = dimSize;
        this->gatherSize = gatherSize;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        indexGm.SetGlobalBuffer((__gm__ DTYPE_INDEX *)index + this->gatherSize * AscendC::GetBlockIdx(), this->gatherSize);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->gatherSize * AscendC::GetBlockIdx(), this->gatherSize);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueIndex, BUFFER_NUM, this->tileLength * sizeof(DTYPE_INDEX));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_INDEX> indexLocal = inQueueIndex.AllocTensor<DTYPE_INDEX>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(indexLocal, indexGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueIndex.EnQue(indexLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_INDEX> indexLocal = inQueueIndex.DeQue<DTYPE_INDEX>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        // Perform gather operation manually since there's no direct gather instruction
        for (int32_t i = 0; i < this->tileLength; i++) {
            uint32_t idx = indexLocal[i];
            zLocal[i] = xLocal[idx];
        }
        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueIndex.FreeTensor(indexLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueIndex;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_INDEX> indexGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t dimSize;
    uint32_t gatherSize;
};

extern "C" __global__ __aicore__ void gather_custom(GM_ADDR x, GM_ADDR index, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGather op;
    op.Init(x, index, z, tiling_data.totalLength, tiling_data.tileNum, tiling_data.dimSize, tiling_data.gatherSize);
    op.Process();
}
