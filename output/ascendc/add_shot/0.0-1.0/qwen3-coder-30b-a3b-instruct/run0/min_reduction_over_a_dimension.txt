kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelMinReduction {
public:
    __aicore__ inline KernelMinReduction() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalLength, uint32_t reduceDimSize, uint32_t otherDimsProduct, uint32_t blockSize)
    {
        this->totalLength = totalLength;
        this->reduceDimSize = reduceDimSize;
        this->otherDimsProduct = otherDimsProduct;
        this->blockSize = blockSize;
        this->blockLength = totalLength / AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->blockLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->blockLength * sizeof(DTYPE_Z));
    }
    __aicore__ inline void Process()
    {
        // For simplicity, assume we process one element per block
        // In practice, more complex reduction logic would be needed
        int32_t loopCount = this->otherDimsProduct;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->reduceDimSize], this->reduceDimSize);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        // Perform min reduction on the local tensor
        // Simplified implementation - actual implementation may require more complex logic
        DTYPE_X minVal = xLocal[0];
        for (uint32_t j = 1; j < this->reduceDimSize; j++) {
            minVal = AscendC::Min(minVal, xLocal[j]);
        }
        zLocal[0] = minVal;
        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress], zLocal, 1);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t totalLength;
    uint32_t reduceDimSize;
    uint32_t otherDimsProduct;
    uint32_t blockSize;
    uint32_t blockLength;
};

extern "C" __global__ __aicore__ void min_reduction_over_a_dimension_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMinReduction op;
    op.Init(x, z, tiling_data.totalLength, tiling_data.reduceDimSize, tiling_data.otherDimsProduct, tiling_data.blockSize);
    op.Process();
}
