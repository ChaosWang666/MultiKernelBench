kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelGemmSubtractGlobalAvgPoolLogSumExpGeluResidualAdd {
public:
    __aicore__ inline KernelGemmSubtractGlobalAvgPoolLogSumExpGeluResidualAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR subtract, GM_ADDR z,
                                uint32_t batch, uint32_t inFeatures, uint32_t outFeatures, uint32_t tileNum)
    {
        this->batch = batch;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->tileNum = tileNum;
        this->blockLength = outFeatures / AscendC::GetBlockNum();
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ float *)x, batch * inFeatures);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, inFeatures * outFeatures);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);
        subtractGm.SetGlobalBuffer((__gm__ float *)subtract, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z, batch * outFeatures);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueWeight, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueSubtract, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        // Gemm
        for (uint32_t i = 0; i < batch; i++) {
            Gemm(i);
        }
        // Subtract
        for (uint32_t i = 0; i < batch; i++) {
            Subtract(i);
        }
        // GlobalAvgPool
        for (uint32_t i = 0; i < batch; i++) {
            GlobalAvgPool(i);
        }
        // LogSumExp
        for (uint32_t i = 0; i < batch; i++) {
            LogSumExp(i);
        }
        // GELU
        for (uint32_t i = 0; i < batch; i++) {
            GELU(i);
        }
        // ResidualAdd
        for (uint32_t i = 0; i < batch; i++) {
            ResidualAdd(i);
        }
    }

private:
    __aicore__ inline void Gemm(uint32_t batchId)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> weightLocal = inQueueWeight.AllocTensor<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[batchId * inFeatures], inFeatures);
        AscendC::DataCopy(weightLocal, weightGm[0], inFeatures * outFeatures);
        AscendC::DataCopy(biasLocal, biasGm[0], outFeatures);
        AscendC::MatMul(zLocal, xLocal, weightLocal, biasLocal, inFeatures, outFeatures, false, true);
        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueWeight.FreeTensor(weightLocal);
        inQueueBias.FreeTensor(biasLocal);
    }
    __aicore__ inline void Subtract(uint32_t batchId)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::LocalTensor<float> subtractLocal = inQueueSubtract.AllocTensor<float>();
        AscendC::DataCopy(subtractLocal, subtractGm[0], outFeatures);
        AscendC::Sub(zLocal, zLocal, subtractLocal, outFeatures);
        outQueueZ.EnQue<float>(zLocal);
        inQueueSubtract.FreeTensor(subtractLocal);
    }
    __aicore__ inline void GlobalAvgPool(uint32_t batchId)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::LocalTensor<float> avgLocal = outQueueZ.AllocTensor<float>();
        AscendC::ReduceMean(avgLocal, zLocal, outFeatures, 1);
        outQueueZ.EnQue<float>(avgLocal);
    }
    __aicore__ inline void LogSumExp(uint32_t batchId)
    {
        AscendC::LocalTensor<float> avgLocal = outQueueZ.DeQue<float>();
        AscendC::LocalTensor<float> logSumExpLocal = outQueueZ.AllocTensor<float>();
        AscendC::LogSumExp(logSumExpLocal, avgLocal, 1);
        outQueueZ.EnQue<float>(logSumExpLocal);
    }
    __aicore__ inline void GELU(uint32_t batchId)
    {
        AscendC::LocalTensor<float> logSumExpLocal = outQueueZ.DeQue<float>();
        AscendC::LocalTensor<float> geluLocal = outQueueZ.AllocTensor<float>();
        AscendC::Gelu(geluLocal, logSumExpLocal, 1);
        outQueueZ.EnQue<float>(geluLocal);
    }
    __aicore__ inline void ResidualAdd(uint32_t batchId)
    {
        AscendC::LocalTensor<float> geluLocal = outQueueZ.DeQue<float>();
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[batchId * inFeatures], inFeatures);
        AscendC::Add(geluLocal, geluLocal, xLocal, outFeatures);
        outQueueZ.EnQue<float>(geluLocal);
        inQueueX.FreeTensor(xLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueWeight, inQueueBias, inQueueSubtract;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> subtractGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batch;
    uint32_t inFeatures;
    uint32_t outFeatures;
    uint32_t tileNum;
    uint32_t blockLength;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void gemm_subtract_global_avg_pool_log_sum_exp_gelu_residual_add_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR subtract, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmSubtractGlobalAvgPoolLogSumExpGeluResidualAdd op;
    op.Init(x, weight, bias, subtract, z, tiling_data.batch, tiling_data.inFeatures, tiling_data.outFeatures, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, bias=True):
        super(ModelNew, self).__init__()
        self.gemm = torch.nn.Linear(in_features, out_features, bias=bias)
        self.subtract = torch.nn.Parameter(torch.randn(out_features))

    def forward(self, x):
        original_x = x.clone().detach()
        # Gemm
        x = self.gemm(x)

        # Subtract
        x = x - self.subtract

        # GlobalAvgPool
        x = torch.mean(x, dim=1, keepdim=True)

        # LogSumExp
        x = torch.logsumexp(x, dim=1, keepdim=True)

        # GELU
        x = torch.nn.functional.gelu(x)

        # ResidualAdd
        x = x + original_x

        return x
'''
