kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelScatter {
public:
    __aicore__ inline KernelScatter() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR idx, GM_ADDR updates, GM_ADDR tiling, uint32_t totalLength, uint32_t tileNum, uint32_t dimSize, uint32_t innerSize)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->dimSize = dimSize;
        this->innerSize = innerSize;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        idxGm.SetGlobalBuffer((__gm__ DTYPE_IDX *)idx + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        updatesGm.SetGlobalBuffer((__gm__ DTYPE_UPDATES *)updates + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueIdx, BUFFER_NUM, this->tileLength * sizeof(DTYPE_IDX));
        pipe.InitBuffer(inQueueUpdates, BUFFER_NUM, this->tileLength * sizeof(DTYPE_UPDATES));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_IDX> idxLocal = inQueueIdx.AllocTensor<DTYPE_IDX>();
        AscendC::LocalTensor<DTYPE_UPDATES> updatesLocal = inQueueUpdates.AllocTensor<DTYPE_UPDATES>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(idxLocal, idxGm[progress * this->tileLength], this->tileLength);
        AscendC::DataCopy(updatesLocal, updatesGm[progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueIdx.EnQue(idxLocal);
        inQueueUpdates.EnQue(updatesLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_IDX> idxLocal = inQueueIdx.DeQue<DTYPE_IDX>();
        AscendC::LocalTensor<DTYPE_UPDATES> updatesLocal = inQueueUpdates.DeQue<DTYPE_UPDATES>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        // Scatter operation implementation
        for (int32_t j = 0; j < this->tileLength; j++) {
            uint32_t idxVal = idxLocal[j];
            if (idxVal >= 0 && idxVal < this->dimSize) {
                zLocal[idxVal] = updatesLocal[j];
            } else {
                zLocal[idxVal] = xLocal[j];
            }
        }
        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueIdx.FreeTensor(idxLocal);
        inQueueUpdates.FreeTensor(updatesLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueIdx, inQueueUpdates;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_IDX> idxGm;
    AscendC::GlobalTensor<DTYPE_UPDATES> updatesGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t dimSize;
    uint32_t innerSize;
};

extern "C" __global__ __aicore__ void scatter_custom(GM_ADDR x, GM_ADDR idx, GM_ADDR updates, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelScatter op;
    op.Init(x, idx, updates, tiling, tiling_data.totalLength, tiling_data.tileNum, tiling_data.dimSize, tiling_data.innerSize);
    op.Process();
}
