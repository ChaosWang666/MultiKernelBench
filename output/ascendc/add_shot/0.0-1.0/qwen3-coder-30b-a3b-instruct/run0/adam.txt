kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelAdam {
public:
    __aicore__ inline KernelAdam() {}
    __aicore__ inline void Init(GM_ADDR param, GM_ADDR grad, GM_ADDR m, GM_ADDR v, GM_ADDR out_param,
                                uint32_t totalLength, float beta1, float beta2, float lr, float eps, uint32_t step)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->beta1 = beta1;
        this->beta2 = beta2;
        this->lr = lr;
        this->eps = eps;
        this->step = step;

        paramGm.SetGlobalBuffer((__gm__ DTYPE_PARAM *)param + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        gradGm.SetGlobalBuffer((__gm__ DTYPE_GRAD *)grad + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        mGm.SetGlobalBuffer((__gm__ DTYPE_M *)m + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        vGm.SetGlobalBuffer((__gm__ DTYPE_V *)v + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        outParamGm.SetGlobalBuffer((__gm__ DTYPE_OUT_PARAM *)out_param + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        
        pipe.InitBuffer(inQueueParam, BUFFER_NUM, this->blockLength * sizeof(DTYPE_PARAM));
        pipe.InitBuffer(inQueueGrad, BUFFER_NUM, this->blockLength * sizeof(DTYPE_GRAD));
        pipe.InitBuffer(inQueueM, BUFFER_NUM, this->blockLength * sizeof(DTYPE_M));
        pipe.InitBuffer(inQueueV, BUFFER_NUM, this->blockLength * sizeof(DTYPE_V));
        pipe.InitBuffer(outQueueParam, BUFFER_NUM, this->blockLength * sizeof(DTYPE_OUT_PARAM));
    }
    __aicore__ inline void Process()
    {
        // Load data
        AscendC::LocalTensor<DTYPE_PARAM> paramLocal = inQueueParam.AllocTensor<DTYPE_PARAM>();
        AscendC::LocalTensor<DTYPE_GRAD> gradLocal = inQueueGrad.AllocTensor<DTYPE_GRAD>();
        AscendC::LocalTensor<DTYPE_M> mLocal = inQueueM.AllocTensor<DTYPE_M>();
        AscendC::LocalTensor<DTYPE_V> vLocal = inQueueV.AllocTensor<DTYPE_V>();
        
        AscendC::DataCopy(paramLocal, paramGm, this->blockLength);
        AscendC::DataCopy(gradLocal, gradGm, this->blockLength);
        AscendC::DataCopy(mLocal, mGm, this->blockLength);
        AscendC::DataCopy(vLocal, vGm, this->blockLength);
        
        inQueueParam.EnQue(paramLocal);
        inQueueGrad.EnQue(gradLocal);
        inQueueM.EnQue(mLocal);
        inQueueV.EnQue(vLocal);
        
        // Compute m = beta1 * m + (1 - beta1) * grad
        AscendC::LocalTensor<DTYPE_M> mHatLocal = outQueueParam.AllocTensor<DTYPE_M>();
        AscendC::Mul(mHatLocal, mLocal, beta1, this->blockLength);
        AscendC::LocalTensor<DTYPE_GRAD> tempGradLocal = inQueueGrad.DeQue<DTYPE_GRAD>();
        AscendC::Mul(tempGradLocal, tempGradLocal, (1.0f - beta1), this->blockLength);
        AscendC::Add(mHatLocal, mHatLocal, tempGradLocal, this->blockLength);
        
        // Compute v = beta2 * v + (1 - beta2) * grad^2
        AscendC::LocalTensor<DTYPE_GRAD> gradSquareLocal = inQueueGrad.AllocTensor<DTYPE_GRAD>();
        AscendC::Mul(gradSquareLocal, tempGradLocal, tempGradLocal, this->blockLength);
        AscendC::Mul(vLocal, vLocal, beta2, this->blockLength);
        AscendC::Add(vLocal, vLocal, gradSquareLocal, this->blockLength);
        
        // Compute m_hat = m / (1 - beta1 ^ step)
        float beta1Pow = powf(beta1, (float)this->step);
        float oneMinusBeta1Pow = 1.0f - beta1Pow;
        AscendC::Div(mHatLocal, mHatLocal, oneMinusBeta1Pow, this->blockLength);
        
        // Compute v_hat = v / (1 - beta2 ^ step)
        float beta2Pow = powf(beta2, (float)this->step);
        float oneMinusBeta2Pow = 1.0f - beta2Pow;
        AscendC::Div(vLocal, vLocal, oneMinusBeta2Pow, this->blockLength);
        
        // Compute sqrt(v_hat)
        AscendC::Sqrt(vLocal, vLocal, this->blockLength);
        
        // Compute m_hat / (sqrt(v_hat) + eps)
        AscendC::Add(vLocal, vLocal, eps, this->blockLength);
        AscendC::Div(tempGradLocal, mHatLocal, vLocal, this->blockLength);
        
        // Compute param = param - lr * m_hat / (sqrt(v_hat) + eps)
        AscendC::Mul(tempGradLocal, tempGradLocal, lr, this->blockLength);
        AscendC::Sub(outParamGm, paramLocal, tempGradLocal, this->blockLength);
        
        outQueueParam.EnQue<DTYPE_OUT_PARAM>(outParamGm);
        inQueueParam.FreeTensor(paramLocal);
        inQueueGrad.FreeTensor(gradLocal);
        inQueueM.FreeTensor(mLocal);
        inQueueV.FreeTensor(vLocal);
        outQueueParam.FreeTensor(outParamGm);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueParam, inQueueGrad, inQueueM, inQueueV;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueParam;
    AscendC::GlobalTensor<DTYPE_PARAM> paramGm;
    AscendC::GlobalTensor<DTYPE_GRAD> gradGm;
    AscendC::GlobalTensor<DTYPE_M> mGm;
    AscendC::GlobalTensor<DTYPE_V> vGm;
    AscendC::GlobalTensor<DTYPE_OUT_PARAM> outParamGm;
    uint32_t blockLength;
    float beta1;
    float beta2;
    float lr;
    float eps;
    uint32_t step;
};

extern "C" __global__ __aicore__ void adam_custom(GM_ADDR param, GM_ADDR grad, GM_ADDR m, GM_ADDR v, GM_ADDR out_param, 
                                                 GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelAdam op;
    op.Init(param, grad, m, v, out_param, tiling_data.totalLength, 
            tiling_data.beta1, tiling_data.beta2, tiling_data.lr, tiling_data.eps, tiling_data.step);
    op.Process();
}
