kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelMatmulBatchNormBiasAddDivideSwish {
public:
    __aicore__ inline KernelMatmulBatchNormBiasAddDivideSwish() {}
    __aicore__ inline void Init(
        GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
        GM_ADDR bnWeight, GM_ADDR bnBias, GM_ADDR bnRunningMean, GM_ADDR bnRunningVar,
        GM_ADDR y, uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures, float bnEps, float divideValue)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->bnEps = bnEps;
        this->divideValue = divideValue;

        this->blockLength = outFeatures / AscendC::GetBlockNum();
        this->tileNum = 1024;
        this->tileLength = this->blockLength / this->tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, inFeatures * outFeatures);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);
        bnWeightGm.SetGlobalBuffer((__gm__ float *)bnWeight, outFeatures);
        bnBiasGm.SetGlobalBuffer((__gm__ float *)bnBias, outFeatures);
        bnRunningMeanGm.SetGlobalBuffer((__gm__ float *)bnRunningMean, outFeatures);
        bnRunningVarGm.SetGlobalBuffer((__gm__ float *)bnRunningVar, outFeatures);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize * outFeatures);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueWeight, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBnWeight, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBnBias, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBnRunningMean, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueBnRunningVar, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t batchId = 0; batchId < batchSize; ++batchId) {
            uint32_t offset = batchId * outFeatures;
            for (uint32_t i = 0; i < outFeatures; i += blockLength) {
                CopyIn(offset + i);
                Compute(offset + i);
                CopyOut(offset + i);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> weightLocal = inQueueWeight.AllocTensor<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
        AscendC::LocalTensor<float> bnWeightLocal = inQueueBnWeight.AllocTensor<float>();
        AscendC::LocalTensor<float> bnBiasLocal = inQueueBnBias.AllocTensor<float>();
        AscendC::LocalTensor<float> bnRunningMeanLocal = inQueueBnRunningMean.AllocTensor<float>();
        AscendC::LocalTensor<float> bnRunningVarLocal = inQueueBnRunningVar.AllocTensor<float>();

        AscendC::DataCopy(xLocal, xGm[offset], tileLength);
        AscendC::DataCopy(weightLocal, weightGm[offset], tileLength);
        AscendC::DataCopy(biasLocal, biasGm[offset], tileLength);
        AscendC::DataCopy(bnWeightLocal, bnWeightGm[offset], tileLength);
        AscendC::DataCopy(bnBiasLocal, bnBiasGm[offset], tileLength);
        AscendC::DataCopy(bnRunningMeanLocal, bnRunningMeanGm[offset], tileLength);
        AscendC::DataCopy(bnRunningVarLocal, bnRunningVarGm[offset], tileLength);

        inQueueX.EnQue(xLocal);
        inQueueWeight.EnQue(weightLocal);
        inQueueBias.EnQue(biasLocal);
        inQueueBnWeight.EnQue(bnWeightLocal);
        inQueueBnBias.EnQue(bnBiasLocal);
        inQueueBnRunningMean.EnQue(bnRunningMeanLocal);
        inQueueBnRunningVar.EnQue(bnRunningVarLocal);
    }

    __aicore__ inline void Compute(uint32_t offset)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> weightLocal = inQueueWeight.DeQue<float>();
        AscendC::LocalTensor<float> biasLocal = inQueueBias.DeQue<float>();
        AscendC::LocalTensor<float> bnWeightLocal = inQueueBnWeight.DeQue<float>();
        AscendC::LocalTensor<float> bnBiasLocal = inQueueBnBias.DeQue<float>();
        AscendC::LocalTensor<float> bnRunningMeanLocal = inQueueBnRunningMean.DeQue<float>();
        AscendC::LocalTensor<float> bnRunningVarLocal = inQueueBnRunningVar.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // Matrix multiplication
        AscendC::MatMul(yLocal, xLocal, weightLocal, false, true, 1.0f, 0.0f, tileLength);

        // Batch norm
        AscendC::Add(yLocal, yLocal, bnBiasLocal, tileLength);
        AscendC::Sub(yLocal, yLocal, bnRunningMeanLocal, tileLength);
        AscendC::Add(yLocal, yLocal, bnWeightLocal, tileLength);
        AscendC::Div(yLocal, yLocal, bnRunningVarLocal, tileLength);
        AscendC::Add(yLocal, yLocal, biasLocal, tileLength);
        AscendC::Div(yLocal, yLocal, divideValue, tileLength);
        AscendC::Sigmoid(yLocal, yLocal, tileLength);

        outQueueY.EnQue<float>(yLocal);

        inQueueX.FreeTensor(xLocal);
        inQueueWeight.FreeTensor(weightLocal);
        inQueueBias.FreeTensor(biasLocal);
        inQueueBnWeight.FreeTensor(bnWeightLocal);
        inQueueBnBias.FreeTensor(bnBiasLocal);
        inQueueBnRunningMean.FreeTensor(bnRunningMeanLocal);
        inQueueBnRunningVar.FreeTensor(bnRunningVarLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[offset], yLocal, tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueWeight, inQueueBias,
        inQueueBnWeight, inQueueBnBias, inQueueBnRunningMean, inQueueBnRunningVar;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm, weightGm, biasGm, bnWeightGm, bnBiasGm, bnRunningMeanGm, bnRunningVarGm, yGm;
    uint32_t batchSize;
    uint32_t inFeatures;
    uint32_t outFeatures;
    float bnEps;
    float divideValue;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void matmul_batch_norm_bias_add_divide_swish_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
    GM_ADDR bnWeight, GM_ADDR bnBias, GM_ADDR bnRunningMean, GM_ADDR bnRunningVar,
    GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulBatchNormBiasAddDivideSwish op;
    op.Init(x, weight, bias, bnWeight, bnBias, bnRunningMean, bnRunningVar, y,
            tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures,
            tiling_data.bnEps, tiling_data.divideValue);
    op.Process();
}
