kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelTallSkinnyMatMul {
public:
    __aicore__ inline KernelTallSkinnyMatMul() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR c, uint32_t m, uint32_t n, uint32_t k, uint32_t tileM, uint32_t tileN, uint32_t tileK)
    {
        this->m = m;
        this->n = n;
        this->k = k;
        this->tileM = tileM;
        this->tileN = tileN;
        this->tileK = tileK;
        this->blockM = m / AscendC::GetBlockNum();
        this->blockN = n / AscendC::GetBlockNum();

        aGm.SetGlobalBuffer((__gm__ DTYPE_A *)a, m * k);
        bGm.SetGlobalBuffer((__gm__ DTYPE_B *)b, k * n);
        cGm.SetGlobalBuffer((__gm__ DTYPE_C *)c, m * n);
        pipe.InitBuffer(inQueueA, BUFFER_NUM, tileM * tileK * sizeof(DTYPE_A));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, tileK * tileN * sizeof(DTYPE_B));
        pipe.InitBuffer(outQueueC, BUFFER_NUM, tileM * tileN * sizeof(DTYPE_C));
    }
    __aicore__ inline void Process()
    {
        uint32_t mBlocks = (m + tileM - 1) / tileM;
        uint32_t nBlocks = (n + tileN - 1) / tileN;
        uint32_t kBlocks = (k + tileK - 1) / tileK;
        for (uint32_t i = 0; i < mBlocks; i++) {
            for (uint32_t j = 0; j < nBlocks; j++) {
                for (uint32_t l = 0; l < kBlocks; l++) {
                    CopyIn(i, j, l);
                    Compute(i, j, l);
                    CopyOut(i, j, l);
                }
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t i, uint32_t j, uint32_t l)
    {
        uint32_t offsetA = i * tileM * k + l * tileK;
        uint32_t offsetB = l * tileK * n + j * tileN;
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.AllocTensor<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.AllocTensor<DTYPE_B>();
        AscendC::DataCopy(aLocal, aGm[offsetA], tileM * tileK);
        AscendC::DataCopy(bLocal, bGm[offsetB], tileK * tileN);
        inQueueA.EnQue(aLocal);
        inQueueB.EnQue(bLocal);
    }
    __aicore__ inline void Compute(uint32_t i, uint32_t j, uint32_t l)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.DeQue<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.DeQue<DTYPE_B>();
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.AllocTensor<DTYPE_C>();
        AscendC::MatMul(cLocal, aLocal, bLocal, tileM, tileN, tileK);
        outQueueC.EnQue<DTYPE_C>(cLocal);
        inQueueA.FreeTensor(aLocal);
        inQueueB.FreeTensor(bLocal);
    }
    __aicore__ inline void CopyOut(uint32_t i, uint32_t j, uint32_t l)
    {
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.DeQue<DTYPE_C>();
        uint32_t offsetC = i * tileM * n + j * tileN;
        AscendC::DataCopy(cGm[offsetC], cLocal, tileM * tileN);
        outQueueC.FreeTensor(cLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::GlobalTensor<DTYPE_A> aGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_C> cGm;
    uint32_t m, n, k;
    uint32_t tileM, tileN, tileK;
    uint32_t blockM, blockN;
};

extern "C" __global__ __aicore__ void tall_skinny_matrix_multiplication_custom(GM_ADDR a, GM_ADDR b, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelTallSkinnyMatMul op;
    op.Init(a, b, c, tiling_data.m, tiling_data.n, tiling_data.k, tiling_data.tileM, tiling_data.tileN, tiling_data.tileK);
    op.Process();
}
