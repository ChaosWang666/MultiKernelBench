kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelLinearAttention {
public:
    __aicore__ inline KernelLinearAttention() {}
    __aicore__ inline void Init(GM_ADDR q, GM_ADDR k, GM_ADDR v, GM_ADDR out, uint32_t batchSize, uint32_t seqLen, uint32_t dModel, uint32_t totalLength)
    {
        this->batchSize = batchSize;
        this->seqLen = seqLen;
        this->dModel = dModel;
        this->totalLength = totalLength;
        this->blockLength = totalLength / AscendC::GetBlockNum();

        qGm.SetGlobalBuffer((__gm__ float *)q + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        kGm.SetGlobalBuffer((__gm__ float *)k + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        vGm.SetGlobalBuffer((__gm__ float *)v + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        outGm.SetGlobalBuffer((__gm__ float *)out + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        
        pipe.InitBuffer(inQueueQ, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(inQueueK, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(inQueueV, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(outQueueOut, BUFFER_NUM, this->blockLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        // ReLU + epsilon for Q and K
        AscendC::LocalTensor<float> qRelu = inQueueQ.AllocTensor<float>();
        AscendC::LocalTensor<float> kRelu = inQueueK.AllocTensor<float>();
        AscendC::LocalTensor<float> vLocal = inQueueV.AllocTensor<float>();
        AscendC::LocalTensor<float> outLocal = outQueueOut.AllocTensor<float>();
        
        // Copy data
        AscendC::DataCopy(qRelu, qGm, this->blockLength);
        AscendC::DataCopy(kRelu, kGm, this->blockLength);
        AscendC::DataCopy(vLocal, vGm, this->blockLength);
        
        // Apply ReLU + epsilon
        AscendC::ReLU(qRelu, qRelu, this->blockLength);
        AscendC::ReLU(kRelu, kRelu, this->blockLength);
        AscendC::AddScalar(qRelu, qRelu, 1e-6f, this->blockLength);
        AscendC::AddScalar(kRelu, kRelu, 1e-6f, this->blockLength);
        
        // Perform einsum operations
        // KV = einsum('bkd,bkv->bdv', K_, V)
        // out = einsum('bqd,bdv->bqv', Q_, KV)
        // For simplicity, we assume these are done via precomputed indices or simplified logic
        
        // Placeholder for actual computation
        AscendC::DataCopy(outGm, outLocal, this->blockLength);
        
        inQueueQ.FreeTensor(qRelu);
        inQueueK.FreeTensor(kRelu);
        inQueueV.FreeTensor(vLocal);
        outQueueOut.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueQ, inQueueK, inQueueV;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueOut;
    AscendC::GlobalTensor<float> qGm;
    AscendC::GlobalTensor<float> kGm;
    AscendC::GlobalTensor<float> vGm;
    AscendC::GlobalTensor<float> outGm;
    uint32_t batchSize;
    uint32_t seqLen;
    uint32_t dModel;
    uint32_t totalLength;
    uint32_t blockLength;
};

extern "C" __global__ __aicore__ void linear_attention_custom(GM_ADDR q, GM_ADDR k, GM_ADDR v, GM_ADDR out, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelLinearAttention op;
    op.Init(q, k, v, out, tiling_data.batchSize, tiling_data.seqLen, tiling_data.dModel, tiling_data.totalLength);
    op.Process();
}
