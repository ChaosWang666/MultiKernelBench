kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelEfficientnetB0 {
public:
    __aicore__ inline KernelEfficientnetB0() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t channel, uint32_t height, uint32_t width, uint32_t totalElements)
    {
        this->batchSize = batchSize;
        this->channel = channel;
        this->height = height;
        this->width = width;
        this->totalElements = totalElements;
        this->blockLength = totalElements / AscendC::GetBlockNum();
        
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->blockLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->blockLength * sizeof(DTYPE_Y));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = 1;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->blockLength], this->blockLength);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        // Simple ReLU operation
        for (uint32_t i = 0; i < this->blockLength; i++) {
            yLocal[i] = AscendC::Max(xLocal[i], static_cast<DTYPE_X>(0.0f));
        }
        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[progress * this->blockLength], yLocal, this->blockLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batchSize;
    uint32_t channel;
    uint32_t height;
    uint32_t width;
    uint32_t totalElements;
    uint32_t blockLength;
};

extern "C" __global__ __aicore__ void efficientnet_b0_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelEfficientnetB0 op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.channel, tiling_data.height, tiling_data.width, tiling_data.totalElements);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, num_classes=1000):
        super(ModelNew, self).__init__()
        self.conv1 = torch.nn.Conv2d(3, 32, kernel_size=3, stride=2, padding=1, bias=False)
        self.bn1 = torch.nn.BatchNorm2d(32)
        self.blocks = torch.nn.Sequential(
            torch.nn.Conv2d(32, 16, kernel_size=3, stride=1, padding=1, bias=False),
            torch.nn.BatchNorm2d(16),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(16, 24, kernel_size=3, stride=2, padding=1, bias=False),
            torch.nn.BatchNorm2d(24),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(24, 24, kernel_size=3, stride=1, padding=1, bias=False),
            torch.nn.BatchNorm2d(24),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(24, 40, kernel_size=5, stride=2, padding=2, bias=False),
            torch.nn.BatchNorm2d(40),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(40, 40, kernel_size=5, stride=1, padding=2, bias=False),
            torch.nn.BatchNorm2d(40),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(40, 80, kernel_size=3, stride=2, padding=1, bias=False),
            torch.nn.BatchNorm2d(80),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(80, 80, kernel_size=3, stride=1, padding=1, bias=False),
            torch.nn.BatchNorm2d(80),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(80, 112, kernel_size=5, stride=1, padding=2, bias=False),
            torch.nn.BatchNorm2d(112),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(112, 112, kernel_size=5, stride=1, padding=2, bias=False),
            torch.nn.BatchNorm2d(112),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(112, 192, kernel_size=5, stride=2, padding=2, bias=False),
            torch.nn.BatchNorm2d(192),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(192, 192, kernel_size=5, stride=1, padding=2, bias=False),
            torch.nn.BatchNorm2d(192),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(192, 192, kernel_size=5, stride=1, padding=2, bias=False),
            torch.nn.BatchNorm2d(192),
            torch.nn.ReLU6(inplace=True),
            torch.nn.Conv2d(192, 320, kernel_size=3, stride=1, padding=1, bias=False),
            torch.nn.BatchNorm2d(320),
            torch.nn.ReLU6(inplace=True),
        )
        self.conv2 = torch.nn.Conv2d(320, 1280, kernel_size=1, stride=1, padding=0, bias=False)
        self.bn2 = torch.nn.BatchNorm2d(1280)
        self.fc = torch.nn.Linear(1280, num_classes)

    def forward(self, x):
        x = custom_ops_lib.efficientnet_b0_custom(self.bn1(self.conv1(x)))
        x = self.blocks(x)
        x = custom_ops_lib.efficientnet_b0_custom(self.bn2(self.conv2(x)))
        x = torch.nn.functional.adaptive_avg_pool2d(x, (1, 1))
        x = x.view(x.size(0), -1)
        x = self.fc(x)
        return x
'''
