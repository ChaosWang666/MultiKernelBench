kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelInplaceUpdate {
public:
    __aicore__ inline KernelInplaceUpdate() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR idx, GM_ADDR value, GM_ADDR tiling, uint32_t totalLength, uint32_t idxLength, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->idxLength = idxLength;
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        idxGm.SetGlobalBuffer((__gm__ DTYPE_IDX *)idx, this->idxLength);
        valueGm.SetGlobalBuffer((__gm__ DTYPE_VALUE *)value, this->idxLength * this->tileLength);
        pipe.InitBuffer(inQueueIdx, BUFFER_NUM, this->idxLength * sizeof(DTYPE_IDX));
        pipe.InitBuffer(inQueueValue, BUFFER_NUM, this->idxLength * this->tileLength * sizeof(DTYPE_VALUE));
        pipe.InitBuffer(outQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_IDX> idxLocal = inQueueIdx.AllocTensor<DTYPE_IDX>();
        AscendC::LocalTensor<DTYPE_VALUE> valueLocal = inQueueValue.AllocTensor<DTYPE_VALUE>();
        AscendC::DataCopy(idxLocal, idxGm[0], this->idxLength);
        AscendC::DataCopy(valueLocal, valueGm[progress * this->idxLength * this->tileLength], this->idxLength * this->tileLength);
        inQueueIdx.EnQue(idxLocal);
        inQueueValue.EnQue(valueLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_IDX> idxLocal = inQueueIdx.DeQue<DTYPE_IDX>();
        AscendC::LocalTensor<DTYPE_VALUE> valueLocal = inQueueValue.DeQue<DTYPE_VALUE>();
        AscendC::LocalTensor<DTYPE_X> xLocal = outQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], this->tileLength);
        // Perform scatter update operation
        for (int32_t j = 0; j < this->idxLength; j++) {
            uint32_t index = idxLocal[j];
            if (index >= 0 && index < this->blockLength) {
                for (int32_t k = 0; k < this->tileLength; k++) {
                    xLocal[index * this->tileLength + k] = valueLocal[j * this->tileLength + k];
                }
            }
        }
        outQueueX.EnQue<DTYPE_X>(xLocal);
        inQueueIdx.FreeTensor(idxLocal);
        inQueueValue.FreeTensor(valueLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = outQueueX.DeQue<DTYPE_X>();
        AscendC::DataCopy(xGm[progress * this->tileLength], xLocal, this->tileLength);
        outQueueX.FreeTensor(xLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueIdx, inQueueValue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueX;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_IDX> idxGm;
    AscendC::GlobalTensor<DTYPE_VALUE> valueGm;
    uint32_t blockLength;
    uint32_t idxLength;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void inplace_update_custom(GM_ADDR x, GM_ADDR idx, GM_ADDR value, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelInplaceUpdate op;
    op.Init(x, idx, value, tiling, tiling_data.totalLength, tiling_data.idxLength, tiling_data.tileNum);
    op.Process();
}
