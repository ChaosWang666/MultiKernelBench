kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelSgd {
public:
    __aicore__ inline KernelSgd() {}
    __aicore__ inline void Init(GM_ADDR param, GM_ADDR grad, GM_ADDR velocity, GM_ADDR z, uint32_t totalLength, float momentum, float lr)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->momentum = momentum;
        this->lr = lr;

        paramGm.SetGlobalBuffer((__gm__ DTYPE_PARAM *)param + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        gradGm.SetGlobalBuffer((__gm__ DTYPE_GRAD *)grad + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        velocityGm.SetGlobalBuffer((__gm__ DTYPE_VELOCITY *)velocity + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueParam, BUFFER_NUM, this->blockLength * sizeof(DTYPE_PARAM));
        pipe.InitBuffer(inQueueGrad, BUFFER_NUM, this->blockLength * sizeof(DTYPE_GRAD));
        pipe.InitBuffer(inQueueVelocity, BUFFER_NUM, this->blockLength * sizeof(DTYPE_VELOCITY));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->blockLength * sizeof(DTYPE_Z));
    }
    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_PARAM> paramLocal = inQueueParam.AllocTensor<DTYPE_PARAM>();
        AscendC::LocalTensor<DTYPE_GRAD> gradLocal = inQueueGrad.AllocTensor<DTYPE_GRAD>();
        AscendC::LocalTensor<DTYPE_VELOCITY> velocityLocal = inQueueVelocity.AllocTensor<DTYPE_VELOCITY>();
        AscendC::DataCopy(paramLocal, paramGm, this->blockLength);
        AscendC::DataCopy(gradLocal, gradGm, this->blockLength);
        AscendC::DataCopy(velocityLocal, velocityGm, this->blockLength);
        inQueueParam.EnQue(paramLocal);
        inQueueGrad.EnQue(gradLocal);
        inQueueVelocity.EnQue(velocityLocal);
        
        Compute();
        
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm, zLocal, this->blockLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE_PARAM> paramLocal = inQueueParam.DeQue<DTYPE_PARAM>();
        AscendC::LocalTensor<DTYPE_GRAD> gradLocal = inQueueGrad.DeQue<DTYPE_GRAD>();
        AscendC::LocalTensor<DTYPE_VELOCITY> velocityLocal = inQueueVelocity.DeQue<DTYPE_VELOCITY>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        
        // velocity = momentum * velocity + grad
        AscendC::Mul(velocityLocal, velocityLocal, this->momentum, this->blockLength);
        AscendC::Add(velocityLocal, velocityLocal, gradLocal, this->blockLength);
        
        // param = param - lr * velocity
        AscendC::Mul(zLocal, velocityLocal, this->lr, this->blockLength);
        AscendC::Sub(zLocal, paramLocal, zLocal, this->blockLength);
        
        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueParam.FreeTensor(paramLocal);
        inQueueGrad.FreeTensor(gradLocal);
        inQueueVelocity.FreeTensor(velocityLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueParam, inQueueGrad, inQueueVelocity;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_PARAM> paramGm;
    AscendC::GlobalTensor<DTYPE_GRAD> gradGm;
    AscendC::GlobalTensor<DTYPE_VELOCITY> velocityGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t blockLength;
    float momentum;
    float lr;
};

extern "C" __global__ __aicore__ void sgd_custom(GM_ADDR param, GM_ADDR grad, GM_ADDR velocity, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSgd op;
    op.Init(param, grad, velocity, z, tiling_data.totalLength, tiling_data.momentum, tiling_data.lr);
    op.Process();
}
