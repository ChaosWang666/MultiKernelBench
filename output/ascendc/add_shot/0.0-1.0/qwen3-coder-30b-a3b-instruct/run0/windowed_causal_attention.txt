kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelWindowedCausalAttention {
public:
    __aicore__ inline KernelWindowedCausalAttention() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR out, uint32_t batch, uint32_t seqLen, uint32_t dModel,
                                uint32_t numHeads, uint32_t windowSize, uint32_t headDim)
    {
        this->batch = batch;
        this->seqLen = seqLen;
        this->dModel = dModel;
        this->numHeads = numHeads;
        this->windowSize = windowSize;
        this->headDim = headDim;
        this->blockLength = seqLen * dModel / AscendC::GetBlockNum();
        this->blockHeadDim = headDim / AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        outGm.SetGlobalBuffer((__gm__ float *)out + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueQ, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(inQueueK, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(inQueueV, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(outQueueAttn, BUFFER_NUM, this->blockLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < seqLen; i++) {
            ProcessToken(i);
        }
    }

private:
    __aicore__ inline void ProcessToken(uint32_t tokenId)
    {
        uint32_t start = (tokenId > windowSize) ? (tokenId - windowSize) : 0;
        uint32_t end = tokenId + 1;
        uint32_t localSeqLen = end - start;
        uint32_t localHeadDim = headDim;
        uint32_t localBatch = batch;
        uint32_t localNumHeads = numHeads;

        // Allocate local tensors for Q, K, V
        AscendC::LocalTensor<float> qLocal = inQueueQ.AllocTensor<float>();
        AscendC::LocalTensor<float> kLocal = inQueueK.AllocTensor<float>();
        AscendC::LocalTensor<float> vLocal = inQueueV.AllocTensor<float>();

        // Copy Q, K, V from global memory
        AscendC::DataCopy(qLocal, xGm[tokenId * dModel], dModel);
        AscendC::DataCopy(kLocal, xGm[start * dModel], localSeqLen * dModel);
        AscendC::DataCopy(vLocal, xGm[start * dModel], localSeqLen * dModel);

        // Perform attention computation
        AscendC::LocalTensor<float> attnOutLocal = outQueueAttn.AllocTensor<float>();
        ComputeAttention(qLocal, kLocal, vLocal, attnOutLocal, localSeqLen, localHeadDim, localNumHeads);

        // Copy result back to global memory
        AscendC::DataCopy(outGm[tokenId * dModel], attnOutLocal, dModel);

        inQueueQ.FreeTensor(qLocal);
        inQueueK.FreeTensor(kLocal);
        inQueueV.FreeTensor(vLocal);
        outQueueAttn.FreeTensor(attnOutLocal);
    }

    __aicore__ inline void ComputeAttention(AscendC::LocalTensor<float>& q, AscendC::LocalTensor<float>& k,
                                            AscendC::LocalTensor<float>& v, AscendC::LocalTensor<float>& out,
                                            uint32_t localSeqLen, uint32_t localHeadDim, uint32_t localNumHeads)
    {
        // Initialize output tensor
        for (uint32_t i = 0; i < dModel; i++) {
            out[i] = 0.0f;
        }

        // Compute attention for each head
        for (uint32_t head = 0; head < localNumHeads; head++) {
            uint32_t headOffset = head * localHeadDim;
            AscendC::LocalTensor<float> qHead = q.Slice(headOffset, localHeadDim);
            AscendC::LocalTensor<float> kHead = k.Slice(headOffset, localHeadDim * localSeqLen);
            AscendC::LocalTensor<float> vHead = v.Slice(headOffset, localHeadDim * localSeqLen);

            // Compute scores matrix
            AscendC::LocalTensor<float> scores = outQueueAttn.AllocTensor<float>();
            for (uint32_t i = 0; i < localSeqLen; i++) {
                for (uint32_t j = 0; j < 1; j++) {
                    float score = 0.0f;
                    for (uint32_t k_idx = 0; k_idx < localHeadDim; k_idx++) {
                        score += qHead[k_idx] * kHead[i * localHeadDim + k_idx];
                    }
                    score /= sqrtf((float)localHeadDim);
                    scores[i] = score;
                }
            }

            // Apply softmax
            AscendC::LocalTensor<float> weights = outQueueAttn.AllocTensor<float>();
            Softmax(scores, weights, localSeqLen);

            // Compute weighted sum
            for (uint32_t i = 0; i < localHeadDim; i++) {
                float sum = 0.0f;
                for (uint32_t j = 0; j < localSeqLen; j++) {
                    sum += weights[j] * vHead[j * localHeadDim + i];
                }
                out[headOffset + i] += sum;
            }

            outQueueAttn.FreeTensor(scores);
            outQueueAttn.FreeTensor(weights);
        }
    }

    __aicore__ inline void Softmax(AscendC::LocalTensor<float>& input, AscendC::LocalTensor<float>& output, uint32_t len)
    {
        // Simple softmax implementation
        float maxVal = input[0];
        for (uint32_t i = 1; i < len; i++) {
            if (input[i] > maxVal) maxVal = input[i];
        }

        float sum = 0.0f;
        for (uint32_t i = 0; i < len; i++) {
            float expVal = expf(input[i] - maxVal);
            output[i] = expVal;
            sum += expVal;
        }

        for (uint32_t i = 0; i < len; i++) {
            output[i] /= sum;
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueQ, inQueueK, inQueueV;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueAttn;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> outGm;
    uint32_t batch;
    uint32_t seqLen;
    uint32_t dModel;
    uint32_t numHeads;
    uint32_t windowSize;
    uint32_t headDim;
    uint32_t blockLength;
    uint32_t blockHeadDim;
};

extern "C" __global__ __aicore__ void windowed_causal_attention_custom(GM_ADDR x, GM_ADDR out, GM_ADDR workspace, GM_ADDR tiling,
                                                                       int64_t d_model, int64_t num_heads, int64_t window_size) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelWindowedCausalAttention op;
    op.Init(x, out, tiling_data.batch, tiling_data.seqLen, tiling_data.dModel,
            tiling_data.numHeads, tiling_data.windowSize, tiling_data.headDim);
    op.Process();
}
