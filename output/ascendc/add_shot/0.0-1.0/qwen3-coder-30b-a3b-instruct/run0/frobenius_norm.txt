kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelFrobeniusNorm {
public:
    __aicore__ inline KernelFrobeniusNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, GM_ADDR norm, uint32_t totalLength, uint32_t blockSize)
    {
        this->totalLength = totalLength;
        this->blockSize = blockSize;
        this->blockLength = totalLength / AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        normGm.SetGlobalBuffer((__gm__ DTYPE_Z *)norm, 1);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->blockSize * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->blockSize * sizeof(DTYPE_Z));
    }
    __aicore__ inline void Process()
    {
        // First pass: compute squared sum
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> sumLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::Fill(sumLocal, 0.0f, 1);
        uint32_t loopCount = (this->blockLength + this->blockSize - 1) / this->blockSize;
        for (uint32_t i = 0; i < loopCount; i++) {
            if (i * this->blockSize >= this->blockLength) break;
            AscendC::DataCopy(xLocal, xGm[i * this->blockSize], min(this->blockSize, this->blockLength - i * this->blockSize));
            inQueueX.EnQue(xLocal);
            AscendC::LocalTensor<DTYPE_Z> tempSum = outQueueZ.DeQue<DTYPE_Z>();
            AscendC::Mul(tempSum, xLocal, xLocal, min(this->blockSize, this->blockLength - i * this->blockSize));
            AscendC::Add(sumLocal, sumLocal, tempSum, 1);
            inQueueX.FreeTensor(xLocal);
        }
        // Reduce sum across blocks
        AscendC::DataCopy(normGm, sumLocal, 1);
        outQueueZ.FreeTensor(sumLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    AscendC::GlobalTensor<DTYPE_Z> normGm;
    uint32_t totalLength;
    uint32_t blockSize;
    uint32_t blockLength;
};

extern "C" __global__ __aicore__ void frobenius_norm_custom(GM_ADDR x, GM_ADDR z, GM_ADDR norm, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFrobeniusNorm op;
    op.Init(x, z, norm, tiling_data.totalLength, tiling_data.blockSize);
    op.Process();
}
"""
