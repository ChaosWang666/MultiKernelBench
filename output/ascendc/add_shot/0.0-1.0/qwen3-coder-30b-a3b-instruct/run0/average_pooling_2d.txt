kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelAveragePooling2d {
public:
    __aicore__ inline KernelAveragePooling2d() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t channels,
                                uint32_t inputHeight, uint32_t inputWidth,
                                uint32_t outputHeight, uint32_t outputWidth,
                                uint32_t kernelSize, uint32_t stride, uint32_t padding)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->inputHeight = inputHeight;
        this->inputWidth = inputWidth;
        this->outputHeight = outputHeight;
        this->outputWidth = outputWidth;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;

        this->blockLength = batchSize * channels * outputHeight * outputWidth;
        this->tileLength = this->blockLength / AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, batchSize * channels * inputHeight * inputWidth);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, batchSize * channels * outputHeight * outputWidth);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = AscendC::GetBlockNum();
        for (int32_t i = 0; i < loopCount; i++) {
            if (i == AscendC::GetBlockIdx()) {
                Compute();
            }
        }
    }

private:
    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        uint32_t batchStart = AscendC::GetBlockIdx() * (this->blockLength / AscendC::GetBlockNum());
        uint32_t batchEnd = (AscendC::GetBlockIdx() + 1) * (this->blockLength / AscendC::GetBlockNum());

        for (uint32_t batch = batchStart; batch < batchEnd; batch++) {
            uint32_t c = batch / (this->outputHeight * this->outputWidth);
            uint32_t h = (batch % (this->outputHeight * this->outputWidth)) / this->outputWidth;
            uint32_t w = (batch % (this->outputHeight * this->outputWidth)) % this->outputWidth;

            float sum = 0.0f;
            uint32_t count = 0;

            for (uint32_t kh = 0; kh < this->kernelSize; kh++) {
                for (uint32_t kw = 0; kw < this->kernelSize; kw++) {
                    int32_t ih = static_cast<int32_t>(h * this->stride + kh - this->padding);
                    int32_t iw = static_cast<int32_t>(w * this->stride + kw - this->padding);

                    if (ih >= 0 && ih < static_cast<int32_t>(this->inputHeight) &&
                        iw >= 0 && iw < static_cast<int32_t>(this->inputWidth)) {
                        uint32_t idx = c * (this->inputHeight * this->inputWidth) +
                                       ih * this->inputWidth + iw;
                        sum += static_cast<float>(xGm[idx]);
                        count++;
                    }
                }
            }

            if (count > 0) {
                yGm[batch] = static_cast<DTYPE_Y>(sum / static_cast<float>(count));
            } else {
                yGm[batch] = static_cast<DTYPE_Y>(0.0f);
            }
        }

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t inputHeight;
    uint32_t inputWidth;
    uint32_t outputHeight;
    uint32_t outputWidth;
    uint32_t kernelSize;
    uint32_t stride;
    uint32_t padding;
    uint32_t blockLength;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void average_pooling_2d_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelAveragePooling2d op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.channels,
            tiling_data.inputHeight, tiling_data.inputWidth,
            tiling_data.outputHeight, tiling_data.outputWidth,
            tiling_data.kernelSize, tiling_data.stride, tiling_data.padding);
    op.Process();
}
