kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelThreeDimTensorMatrixMultiplication {
public:
    __aicore__ inline KernelThreeDimTensorMatrixMultiplication() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR c, uint32_t n, uint32_t m, uint32_t k, uint32_t l, uint32_t tileK, uint32_t tileL)
    {
        this->n = n;
        this->m = m;
        this->k = k;
        this->l = l;
        this->tileK = tileK;
        this->tileL = tileL;
        this->blockLength = k / AscendC::GetBlockNum();
        this->blockLengthL = l / AscendC::GetBlockNum();

        aGm.SetGlobalBuffer((__gm__ DTYPE_A *)a, n * m * k);
        bGm.SetGlobalBuffer((__gm__ DTYPE_B *)b, k * l);
        cGm.SetGlobalBuffer((__gm__ DTYPE_C *)c, n * m * l);
        pipe.InitBuffer(inQueueA, BUFFER_NUM, this->blockLength * sizeof(DTYPE_A));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, this->blockLength * sizeof(DTYPE_B));
        pipe.InitBuffer(outQueueC, BUFFER_NUM, this->blockLengthL * sizeof(DTYPE_C));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < n; i++) {
            for (uint32_t j = 0; j < m; j++) {
                ProcessInner(i, j);
            }
        }
    }

private:
    __aicore__ inline void ProcessInner(uint32_t i, uint32_t j)
    {
        uint32_t offsetA = i * m * k + j * k;
        uint32_t offsetB = 0;
        uint32_t offsetC = i * m * l + j * l;
        uint32_t loopCount = k / tileK;
        for (uint32_t kIter = 0; kIter < loopCount; kIter++) {
            CopyIn(offsetA + kIter * tileK, offsetB, kIter);
            Compute(offsetA + kIter * tileK, offsetB + kIter * tileK, offsetC, kIter);
            CopyOut(offsetC, kIter);
        }
    }

    __aicore__ inline void CopyIn(uint32_t offsetA, uint32_t offsetB, uint32_t kIter)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.AllocTensor<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.AllocTensor<DTYPE_B>();
        AscendC::DataCopy(aLocal, aGm[offsetA], tileK);
        AscendC::DataCopy(bLocal, bGm[offsetB + kIter * tileK], tileK);
        inQueueA.EnQue(aLocal);
        inQueueB.EnQue(bLocal);
    }
    __aicore__ inline void Compute(uint32_t offsetA, uint32_t offsetB, uint32_t offsetC, uint32_t kIter)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.DeQue<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.DeQue<DTYPE_B>();
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.AllocTensor<DTYPE_C>();
        AscendC::MatMul(cLocal, aLocal, bLocal, tileK, tileL);
        outQueueC.EnQue<DTYPE_C>(cLocal);
        inQueueA.FreeTensor(aLocal);
        inQueueB.FreeTensor(bLocal);
    }
    __aicore__ inline void CopyOut(uint32_t offsetC, uint32_t kIter)
    {
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.DeQue<DTYPE_C>();
        AscendC::DataCopy(cGm[offsetC + kIter * tileL], cLocal, tileL);
        outQueueC.FreeTensor(cLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::GlobalTensor<DTYPE_A> aGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_C> cGm;
    uint32_t n;
    uint32_t m;
    uint32_t k;
    uint32_t l;
    uint32_t tileK;
    uint32_t tileL;
    uint32_t blockLength;
    uint32_t blockLengthL;
};

extern "C" __global__ __aicore__ void three_dim_tensor_matrix_multiplication_custom(GM_ADDR a, GM_ADDR b, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelThreeDimTensorMatrixMultiplication op;
    op.Init(a, b, c, tiling_data.n, tiling_data.m, tiling_data.k, tiling_data.l, tiling_data.tileK, tiling_data.tileL);
    op.Process();
}
