kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelBatchedMatrixMultiplication {
public:
    __aicore__ inline KernelBatchedMatrixMultiplication() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR c, uint32_t batchSize, uint32_t m, uint32_t k, uint32_t n, uint32_t tileM, uint32_t tileK, uint32_t tileN)
    {
        this->batchSize = batchSize;
        this->m = m;
        this->k = k;
        this->n = n;
        this->tileM = tileM;
        this->tileK = tileK;
        this->tileN = tileN;
        this->blockSize = AscendC::GetBlockNum();
        this->blockId = AscendC::GetBlockIdx();
        this->totalBlocks = (m + tileM - 1) / tileM * (n + tileN - 1) / tileN;
        this->loopCount = (k + tileK - 1) / tileK;

        aGm.SetGlobalBuffer((__gm__ float *)a, batchSize * m * k);
        bGm.SetGlobalBuffer((__gm__ float *)b, batchSize * k * n);
        cGm.SetGlobalBuffer((__gm__ float *)c, batchSize * m * n);
        pipe.InitBuffer(inQueueA, BUFFER_NUM, tileM * tileK * sizeof(float));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, tileK * tileN * sizeof(float));
        pipe.InitBuffer(outQueueC, BUFFER_NUM, tileM * tileN * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t batch = 0; batch < batchSize; ++batch) {
            ProcessBatch(batch);
        }
    }

private:
    __aicore__ inline void ProcessBatch(uint32_t batch)
    {
        uint32_t aOffset = batch * m * k;
        uint32_t bOffset = batch * k * n;
        uint32_t cOffset = batch * m * n;
        uint32_t totalBlocks = (m + tileM - 1) / tileM * (n + tileN - 1) / tileN;
        uint32_t blockId = AscendC::GetBlockIdx();
        uint32_t blockSize = AscendC::GetBlockNum();
        for (uint32_t block = blockId; block < totalBlocks; block += blockSize) {
            ProcessBlock(block, aOffset, bOffset, cOffset);
        }
    }

    __aicore__ inline void ProcessBlock(uint32_t block, uint32_t aOffset, uint32_t bOffset, uint32_t cOffset)
    {
        uint32_t mBlock = block / ((n + tileN - 1) / tileN);
        uint32_t nBlock = block % ((n + tileN - 1) / tileN);
        uint32_t mStart = mBlock * tileM;
        uint32_t nStart = nBlock * tileN;
        uint32_t mEnd = (mStart + tileM > m) ? m : mStart + tileM;
        uint32_t nEnd = (nStart + tileN > n) ? n : nStart + tileN;
        uint32_t mSize = mEnd - mStart;
        uint32_t nSize = nEnd - nStart;

        if (mSize <= 0 || nSize <= 0) return;

        AscendC::LocalTensor<float> cLocal = outQueueC.AllocTensor<float>();
        for (uint32_t i = 0; i < mSize * nSize; ++i) {
            cLocal[i] = 0.0f;
        }

        for (uint32_t kBlock = 0; kBlock < loopCount; ++kBlock) {
            uint32_t kStart = kBlock * tileK;
            uint32_t kEnd = (kStart + tileK > k) ? k : kStart + tileK;
            uint32_t kSize = kEnd - kStart;

            AscendC::LocalTensor<float> aLocal = inQueueA.AllocTensor<float>();
            AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();

            AscendC::DataCopy(aLocal, aGm[aOffset + mStart * k + kStart], mSize * kSize);
            AscendC::DataCopy(bLocal, bGm[bOffset + kStart * n + nStart], kSize * nSize);

            inQueueA.EnQue(aLocal);
            inQueueB.EnQue(bLocal);

            AscendC::LocalTensor<float> aLocalDeq = inQueueA.DeQue<float>();
            AscendC::LocalTensor<float> bLocalDeq = inQueueB.DeQue<float>();

            AscendC::MatMul(cLocal, aLocalDeq, bLocalDeq, mSize, kSize, nSize, false, false);

            inQueueA.FreeTensor(aLocalDeq);
            inQueueB.FreeTensor(bLocalDeq);
        }

        AscendC::DataCopy(cGm[cOffset + mStart * n + nStart], cLocal, mSize * nSize);
        outQueueC.FreeTensor(cLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::GlobalTensor<float> aGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> cGm;
    uint32_t batchSize;
    uint32_t m;
    uint32_t k;
    uint32_t n;
    uint32_t tileM;
    uint32_t tileK;
    uint32_t tileN;
    uint32_t blockSize;
    uint32_t blockId;
    uint32_t totalBlocks;
    uint32_t loopCount;
};

extern "C" __global__ __aicore__ void batched_matrix_multiplication_custom(GM_ADDR a, GM_ADDR b, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelBatchedMatrixMultiplication op;
    op.Init(a, b, c, tiling_data.batchSize, tiling_data.m, tiling_data.k, tiling_data.n, tiling_data.tileM, tiling_data.tileK, tiling_data.tileN);
    op.Process();
}
