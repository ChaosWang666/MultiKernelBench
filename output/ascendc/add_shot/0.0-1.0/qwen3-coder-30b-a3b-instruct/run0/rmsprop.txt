kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelRmsprop {
public:
    __aicore__ inline KernelRmsprop() {}
    __aicore__ inline void Init(GM_ADDR param, GM_ADDR grad, GM_ADDR v, GM_ADDR out_param, uint32_t totalLength, float lr, float alpha, float eps)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->lr = lr;
        this->alpha = alpha;
        this->eps = eps;
        this->totalLength = totalLength;

        paramGm.SetGlobalBuffer((__gm__ DTYPE_PARAM *)param + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        gradGm.SetGlobalBuffer((__gm__ DTYPE_GRAD *)grad + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        vGm.SetGlobalBuffer((__gm__ DTYPE_V *)v + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        outParamGm.SetGlobalBuffer((__gm__ DTYPE_OUT_PARAM *)out_param + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueParam, BUFFER_NUM, this->blockLength * sizeof(DTYPE_PARAM));
        pipe.InitBuffer(inQueueGrad, BUFFER_NUM, this->blockLength * sizeof(DTYPE_GRAD));
        pipe.InitBuffer(inQueueV, BUFFER_NUM, this->blockLength * sizeof(DTYPE_V));
        pipe.InitBuffer(outQueueParam, BUFFER_NUM, this->blockLength * sizeof(DTYPE_OUT_PARAM));
    }
    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<DTYPE_PARAM> paramLocal = inQueueParam.AllocTensor<DTYPE_PARAM>();
        AscendC::LocalTensor<DTYPE_GRAD> gradLocal = inQueueGrad.AllocTensor<DTYPE_GRAD>();
        AscendC::LocalTensor<DTYPE_V> vLocal = inQueueV.AllocTensor<DTYPE_V>();
        AscendC::DataCopy(paramLocal, paramGm, this->blockLength);
        AscendC::DataCopy(gradLocal, gradGm, this->blockLength);
        AscendC::DataCopy(vLocal, vGm, this->blockLength);
        inQueueParam.EnQue(paramLocal);
        inQueueGrad.EnQue(gradLocal);
        inQueueV.EnQue(vLocal);
    }
    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE_PARAM> paramLocal = inQueueParam.DeQue<DTYPE_PARAM>();
        AscendC::LocalTensor<DTYPE_GRAD> gradLocal = inQueueGrad.DeQue<DTYPE_GRAD>();
        AscendC::LocalTensor<DTYPE_V> vLocal = inQueueV.DeQue<DTYPE_V>();
        AscendC::LocalTensor<DTYPE_OUT_PARAM> outParamLocal = outQueueParam.AllocTensor<DTYPE_OUT_PARAM>();
        
        // v = alpha * v + (1 - alpha) * grad^2
        AscendC::LocalTensor<DTYPE_V> tempV = AscendC::AllocLocalTensor<DTYPE_V>(this->blockLength);
        AscendC::Mul(tempV, vLocal, this->alpha, this->blockLength);
        AscendC::LocalTensor<DTYPE_GRAD> gradSquare = AscendC::AllocLocalTensor<DTYPE_GRAD>(this->blockLength);
        AscendC::Mul(gradSquare, gradLocal, gradLocal, this->blockLength);
        AscendC::LocalTensor<DTYPE_V> oneMinusAlpha = AscendC::AllocLocalTensor<DTYPE_V>(1);
        oneMinusAlpha.Set(1.0f - this->alpha);
        AscendC::Mul(gradSquare, gradSquare, oneMinusAlpha, this->blockLength);
        AscendC::Add(tempV, tempV, gradSquare, this->blockLength);
        
        // param = param - lr * grad / (sqrt(v) + eps)
        AscendC::LocalTensor<DTYPE_V> sqrtV = AscendC::AllocLocalTensor<DTYPE_V>(this->blockLength);
        AscendC::Sqrt(sqrtV, tempV, this->blockLength);
        AscendC::LocalTensor<DTYPE_V> epsTensor = AscendC::AllocLocalTensor<DTYPE_V>(1);
        epsTensor.Set(this->eps);
        AscendC::Add(sqrtV, sqrtV, epsTensor, this->blockLength);
        AscendC::Div(outParamLocal, gradLocal, sqrtV, this->blockLength);
        AscendC::Mul(outParamLocal, outParamLocal, this->lr, this->blockLength);
        AscendC::Sub(outParamLocal, paramLocal, outParamLocal, this->blockLength);
        
        outQueueParam.EnQue<DTYPE_OUT_PARAM>(outParamLocal);
        inQueueParam.FreeTensor(paramLocal);
        inQueueGrad.FreeTensor(gradLocal);
        inQueueV.FreeTensor(vLocal);
    }
    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<DTYPE_OUT_PARAM> outParamLocal = outQueueParam.DeQue<DTYPE_OUT_PARAM>();
        AscendC::DataCopy(outParamGm, outParamLocal, this->blockLength);
        outQueueParam.FreeTensor(outParamLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueParam, inQueueGrad, inQueueV;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueParam;
    AscendC::GlobalTensor<DTYPE_PARAM> paramGm;
    AscendC::GlobalTensor<DTYPE_GRAD> gradGm;
    AscendC::GlobalTensor<DTYPE_V> vGm;
    AscendC::GlobalTensor<DTYPE_OUT_PARAM> outParamGm;
    uint32_t blockLength;
    uint32_t totalLength;
    float lr;
    float alpha;
    float eps;
};

extern "C" __global__ __aicore__ void rmsprop_custom(GM_ADDR param, GM_ADDR grad, GM_ADDR v, GM_ADDR out_param, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelRmsprop op;
    op.Init(param, grad, v, out_param, tiling_data.totalLength, tiling_data.lr, tiling_data.alpha, tiling_data.eps);
    op.Process();
}
