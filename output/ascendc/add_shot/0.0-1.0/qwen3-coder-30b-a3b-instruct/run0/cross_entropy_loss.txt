kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelCrossEntropyLoss {
public:
    __aicore__ inline KernelCrossEntropyLoss() {}
    __aicore__ inline void Init(GM_ADDR predictions, GM_ADDR targets, GM_ADDR loss, uint32_t batchSize, uint32_t numClasses)
    {
        this->batchSize = batchSize;
        this->numClasses = numClasses;
        this->blockLength = batchSize / AscendC::GetBlockNum();
        this->tileLength = this->blockLength / BUFFER_NUM;

        predictionsGm.SetGlobalBuffer((__gm__ float *)predictions + this->blockLength * AscendC::GetBlockIdx(), this->blockLength * numClasses);
        targetsGm.SetGlobalBuffer((__gm__ int32_t *)targets + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        lossGm.SetGlobalBuffer((__gm__ float *)loss + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueuePred, BUFFER_NUM, this->tileLength * numClasses * sizeof(float));
        pipe.InitBuffer(inQueueTarget, BUFFER_NUM, this->tileLength * sizeof(int32_t));
        pipe.InitBuffer(outQueueLoss, BUFFER_NUM, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->blockLength * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> predLocal = inQueuePred.AllocTensor<float>();
        AscendC::LocalTensor<int32_t> targetLocal = inQueueTarget.AllocTensor<int32_t>();
        AscendC::DataCopy(predLocal, predictionsGm[progress * this->numClasses], this->tileLength * this->numClasses);
        AscendC::DataCopy(targetLocal, targetsGm[progress], this->tileLength);
        inQueuePred.EnQue(predLocal);
        inQueueTarget.EnQue(targetLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> predLocal = inQueuePred.DeQue<float>();
        AscendC::LocalTensor<int32_t> targetLocal = inQueueTarget.DeQue<int32_t>();
        AscendC::LocalTensor<float> lossLocal = outQueueLoss.AllocTensor<float>();
        // Simplified implementation of cross entropy loss computation
        // In practice, this would involve log_softmax and index selection
        for (int32_t i = 0; i < this->tileLength; i++) {
            int32_t target = targetLocal[i];
            float maxVal = predLocal[i * this->numClasses + 0];
            for (int32_t j = 1; j < this->numClasses; j++) {
                if (predLocal[i * this->numClasses + j] > maxVal) {
                    maxVal = predLocal[i * this->numClasses + j];
                }
            }
            float sumExp = 0.0f;
            for (int32_t j = 0; j < this->numClasses; j++) {
                sumExp += expf(predLocal[i * this->numClasses + j] - maxVal);
            }
            float logSumExp = logf(sumExp) + maxVal;
            lossLocal[i] = -predLocal[i * this->numClasses + target] + logSumExp;
        }
        outQueueLoss.EnQue<float>(lossLocal);
        inQueuePred.FreeTensor(predLocal);
        inQueueTarget.FreeTensor(targetLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> lossLocal = outQueueLoss.DeQue<float>();
        AscendC::DataCopy(lossGm[progress], lossLocal, this->tileLength);
        outQueueLoss.FreeTensor(lossLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueuePred, inQueueTarget;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueLoss;
    AscendC::GlobalTensor<float> predictionsGm;
    AscendC::GlobalTensor<int32_t> targetsGm;
    AscendC::GlobalTensor<float> lossGm;
    uint32_t batchSize;
    uint32_t numClasses;
    uint32_t blockLength;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void cross_entropy_loss_custom(GM_ADDR predictions, GM_ADDR targets, GM_ADDR loss, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelCrossEntropyLoss op;
    op.Init(predictions, targets, loss, tiling_data.batchSize, tiling_data.numClasses);
    op.Process();
}
