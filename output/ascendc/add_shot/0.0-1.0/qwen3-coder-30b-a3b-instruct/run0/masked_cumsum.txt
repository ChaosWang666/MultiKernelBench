kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelMaskedCumsum {
public:
    __aicore__ inline KernelMaskedCumsum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mask, GM_ADDR z, uint32_t totalLength, uint32_t dimSize, uint32_t outerSize)
    {
        this->totalLength = totalLength;
        this->dimSize = dimSize;
        this->outerSize = outerSize;
        this->blockLength = totalLength / AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        maskGm.SetGlobalBuffer((__gm__ bool *)mask + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(inQueueMask, BUFFER_NUM, this->blockLength * sizeof(bool));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->blockLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        // For simplicity, we process one element at a time
        // In practice, you would implement a more efficient cumulative sum algorithm
        for (uint32_t i = 0; i < this->outerSize; i++) {
            ProcessOneOuter(i);
        }
    }

private:
    __aicore__ inline void ProcessOneOuter(uint32_t outerId)
    {
        uint32_t offset = outerId * this->dimSize;
        float cumsum = 0.0f;
        for (uint32_t j = 0; j < this->dimSize; j++) {
            uint32_t index = offset + j;
            float xVal = xGm[index];
            bool maskVal = maskGm[index];
            if (maskVal) {
                cumsum += xVal;
            }
            zGm[index] = cumsum;
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueMask;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<bool> maskGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t totalLength;
    uint32_t dimSize;
    uint32_t outerSize;
    uint32_t blockLength;
};

extern "C" __global__ __aicore__ void masked_cumsum_custom(GM_ADDR x, GM_ADDR mask, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMaskedCumsum op;
    op.Init(x, mask, z, tiling_data.totalLength, tiling_data.dimSize, tiling_data.outerSize);
    op.Process();
}
