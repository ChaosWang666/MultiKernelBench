kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelLamb {
public:
    __aicore__ inline KernelLamb() {}
    __aicore__ inline void Init(GM_ADDR param, GM_ADDR m, GM_ADDR v, GM_ADDR out, uint32_t totalLength, float lr, float eps)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->lr = lr;
        this->eps = eps;
        this->totalLength = totalLength;

        paramGm.SetGlobalBuffer((__gm__ DTYPE_PARAM *)param + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        mGm.SetGlobalBuffer((__gm__ DTYPE_M *)m + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        vGm.SetGlobalBuffer((__gm__ DTYPE_V *)v + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        outGm.SetGlobalBuffer((__gm__ DTYPE_OUT *)out + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(inQueueParam, BUFFER_NUM, this->blockLength * sizeof(DTYPE_PARAM));
        pipe.InitBuffer(inQueueM, BUFFER_NUM, this->blockLength * sizeof(DTYPE_M));
        pipe.InitBuffer(inQueueV, BUFFER_NUM, this->blockLength * sizeof(DTYPE_V));
        pipe.InitBuffer(outQueueOut, BUFFER_NUM, this->blockLength * sizeof(DTYPE_OUT));
    }
    __aicore__ inline void Process()
    {
        CopyIn();
        Compute();
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn()
    {
        AscendC::LocalTensor<DTYPE_PARAM> paramLocal = inQueueParam.AllocTensor<DTYPE_PARAM>();
        AscendC::LocalTensor<DTYPE_M> mLocal = inQueueM.AllocTensor<DTYPE_M>();
        AscendC::LocalTensor<DTYPE_V> vLocal = inQueueV.AllocTensor<DTYPE_V>();
        AscendC::DataCopy(paramLocal, paramGm, this->blockLength);
        AscendC::DataCopy(mLocal, mGm, this->blockLength);
        AscendC::DataCopy(vLocal, vGm, this->blockLength);
        inQueueParam.EnQue(paramLocal);
        inQueueM.EnQue(mLocal);
        inQueueV.EnQue(vLocal);
    }
    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE_PARAM> paramLocal = inQueueParam.DeQue<DTYPE_PARAM>();
        AscendC::LocalTensor<DTYPE_M> mLocal = inQueueM.DeQue<DTYPE_M>();
        AscendC::LocalTensor<DTYPE_V> vLocal = inQueueV.DeQue<DTYPE_V>();
        AscendC::LocalTensor<DTYPE_OUT> outLocal = outQueueOut.AllocTensor<DTYPE_OUT>();

        // sqrt(v)
        AscendC::LocalTensor<DTYPE_V> sqrtV = outQueueOut.AllocTensor<DTYPE_V>();
        AscendC::Sqrt(sqrtV, vLocal, this->blockLength);
        
        // add eps
        AscendC::LocalTensor<DTYPE_V> vEps = outQueueOut.AllocTensor<DTYPE_V>();
        AscendC::Add(vEps, sqrtV, this->eps, this->blockLength);
        
        // divide m / (sqrt(v) + eps)
        AscendC::LocalTensor<DTYPE_M> r = outQueueOut.AllocTensor<DTYPE_M>();
        AscendC::Div(r, mLocal, vEps, this->blockLength);
        
        // norm of r
        AscendC::LocalTensor<DTYPE_M> rNorm = outQueueOut.AllocTensor<DTYPE_M>();
        AscendC::ReduceNorm(rNorm, r, this->blockLength, 2); // L2 norm
        
        // norm of param
        AscendC::LocalTensor<DTYPE_PARAM> paramNorm = outQueueOut.AllocTensor<DTYPE_PARAM>();
        AscendC::ReduceNorm(paramNorm, paramLocal, this->blockLength, 2); // L2 norm
        
        // trust ratio
        AscendC::LocalTensor<DTYPE_M> trustRatio = outQueueOut.AllocTensor<DTYPE_M>();
        AscendC::Div(trustRatio, paramNorm, rNorm, 1);
        
        // multiply trust ratio * r
        AscendC::LocalTensor<DTYPE_M> trustRatioR = outQueueOut.AllocTensor<DTYPE_M>();
        AscendC::Mul(trustRatioR, trustRatio, r, this->blockLength);
        
        // multiply lr * trust_ratio * r
        AscendC::LocalTensor<DTYPE_M> lrTrustRatioR = outQueueOut.AllocTensor<DTYPE_M>();
        AscendC::Mul(lrTrustRatioR, this->lr, trustRatioR, this->blockLength);
        
        // subtract from param
        AscendC::Sub(outLocal, paramLocal, lrTrustRatioR, this->blockLength);
        
        outQueueOut.EnQue<DTYPE_OUT>(outLocal);
        inQueueParam.FreeTensor(paramLocal);
        inQueueM.FreeTensor(mLocal);
        inQueueV.FreeTensor(vLocal);
    }
    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<DTYPE_OUT> outLocal = outQueueOut.DeQue<DTYPE_OUT>();
        AscendC::DataCopy(outGm, outLocal, this->blockLength);
        outQueueOut.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueParam, inQueueM, inQueueV;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueOut;
    AscendC::GlobalTensor<DTYPE_PARAM> paramGm;
    AscendC::GlobalTensor<DTYPE_M> mGm;
    AscendC::GlobalTensor<DTYPE_V> vGm;
    AscendC::GlobalTensor<DTYPE_OUT> outGm;
    uint32_t blockLength;
    uint32_t totalLength;
    float lr;
    float eps;
};

extern "C" __global__ __aicore__ void lamb_custom(GM_ADDR param, GM_ADDR m, GM_ADDR v, GM_ADDR out, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelLamb op;
    op.Init(param, m, v, out, tiling_data.totalLength, tiling_data.lr, tiling_data.eps);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib
class ModelNew(torch.nn.Module):
    def __init__(self, lr=1e-3, eps=1e-6):
        super().__init__()
        self.lr = lr
        self.eps = eps

    def forward(self, param, m, v):
        r = m / (v.sqrt() + self.eps)
        trust_ratio = param.norm(p=2) / (r.norm(p=2) + self.eps)
        param = param - self.lr * trust_ratio * r
        return param
'''
