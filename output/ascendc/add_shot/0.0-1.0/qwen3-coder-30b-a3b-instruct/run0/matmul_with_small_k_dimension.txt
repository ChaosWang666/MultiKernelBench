kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelMatmul {
public:
    __aicore__ inline KernelMatmul() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR c, uint32_t m, uint32_t n, uint32_t k, uint32_t tileM, uint32_t tileN, uint32_t tileK)
    {
        this->m = m;
        this->n = n;
        this->k = k;
        this->tileM = tileM;
        this->tileN = tileN;
        this->tileK = tileK;
        this->blockM = m / AscendC::GetBlockNum();
        this->blockN = n / AscendC::GetBlockNum();

        aGm.SetGlobalBuffer((__gm__ DTYPE_A *)a + this->blockM * AscendC::GetBlockIdx() * k, this->blockM * k);
        bGm.SetGlobalBuffer((__gm__ DTYPE_B *)b, k * n);
        cGm.SetGlobalBuffer((__gm__ DTYPE_C *)c + this->blockM * AscendC::GetBlockIdx() * n, this->blockM * n);
        pipe.InitBuffer(inQueueA, BUFFER_NUM, this->tileM * this->tileK * sizeof(DTYPE_A));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, this->tileK * this->tileN * sizeof(DTYPE_B));
        pipe.InitBuffer(outQueueC, BUFFER_NUM, this->tileM * this->tileN * sizeof(DTYPE_C));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCountM = (this->m + this->tileM - 1) / this->tileM;
        int32_t loopCountN = (this->n + this->tileN - 1) / this->tileN;
        int32_t loopCountK = (this->k + this->tileK - 1) / this->tileK;
        for (int32_t i = 0; i < loopCountM; i++) {
            for (int32_t j = 0; j < loopCountN; j++) {
                for (int32_t k = 0; k < loopCountK; k++) {
                    CopyIn(i, j, k);
                    Compute(i, j, k);
                    CopyOut(i, j, k);
                }
            }
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t i, int32_t j, int32_t k)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.AllocTensor<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.AllocTensor<DTYPE_B>();
        uint32_t offsetA = i * this->tileM * this->k + k * this->tileK;
        uint32_t offsetB = k * this->tileK * this->n + j * this->tileN;
        uint32_t lenA = (i * this->tileM + this->tileM > this->m) ? (this->m - i * this->tileM) * this->k : this->tileM * this->k;
        uint32_t lenB = (j * this->tileN + this->tileN > this->n) ? this->k * (this->n - j * this->tileN) : this->k * this->tileN;
        if (k == loopCountK - 1) {
            lenA = (i * this->tileM + this->tileM > this->m) ? (this->m - i * this->tileM) * this->k : this->tileM * this->k;
            lenB = (j * this->tileN + this->tileN > this->n) ? this->k * (this->n - j * this->tileN) : this->k * this->tileN;
        }
        AscendC::DataCopy(aLocal, aGm[offsetA], lenA);
        AscendC::DataCopy(bLocal, bGm[offsetB], lenB);
        inQueueA.EnQue(aLocal);
        inQueueB.EnQue(bLocal);
    }
    __aicore__ inline void Compute(int32_t i, int32_t j, int32_t k)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.DeQue<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.DeQue<DTYPE_B>();
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.AllocTensor<DTYPE_C>();
        AscendC::MatMul(cLocal, aLocal, bLocal, this->tileM, this->tileN, this->tileK);
        outQueueC.EnQue<DTYPE_C>(cLocal);
        inQueueA.FreeTensor(aLocal);
        inQueueB.FreeTensor(bLocal);
    }
    __aicore__ inline void CopyOut(int32_t i, int32_t j, int32_t k)
    {
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.DeQue<DTYPE_C>();
        uint32_t offsetC = i * this->tileM * this->n + j * this->tileN;
        uint32_t lenC = (i * this->tileM + this->tileM > this->m) ? (this->m - i * this->tileM) * this->n : this->tileM * this->n;
        if (j == loopCountN - 1) {
            lenC = (i * this->tileM + this->tileM > this->m) ? (this->m - i * this->tileM) * this->n : this->tileM * this->n;
        }
        AscendC::DataCopy(cGm[offsetC], cLocal, lenC);
        outQueueC.FreeTensor(cLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::GlobalTensor<DTYPE_A> aGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_C> cGm;
    uint32_t m;
    uint32_t n;
    uint32_t k;
    uint32_t tileM;
    uint32_t tileN;
    uint32_t tileK;
    uint32_t blockM;
    uint32_t blockN;
};

extern "C" __global__ __aicore__ void matmul_with_small_k_dimension_custom(GM_ADDR a, GM_ADDR b, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmul op;
    op.Init(a, b, c, tiling_data.m, tiling_data.n, tiling_data.k, tiling_data.tileM, tiling_data.tileN, tiling_data.tileK);
    op.Process();
}
