kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelMatmulWithTransposedB {
public:
    __aicore__ inline KernelMatmulWithTransposedB() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR c, uint32_t m, uint32_t k, uint32_t n, uint32_t tileM, uint32_t tileK, uint32_t tileN)
    {
        this->m = m;
        this->k = k;
        this->n = n;
        this->tileM = tileM;
        this->tileK = tileK;
        this->tileN = tileN;
        this->blockM = m / AscendC::GetBlockNum();
        this->blockK = k;
        this->blockN = n;

        aGm.SetGlobalBuffer((__gm__ DTYPE_A *)a + this->blockM * AscendC::GetBlockIdx() * this->blockK, this->blockM * this->blockK);
        bGm.SetGlobalBuffer((__gm__ DTYPE_B *)b, this->blockK * this->blockN);
        cGm.SetGlobalBuffer((__gm__ DTYPE_C *)c + this->blockM * AscendC::GetBlockIdx() * this->blockN, this->blockM * this->blockN);
        pipe.InitBuffer(inQueueA, BUFFER_NUM, this->tileM * this->tileK * sizeof(DTYPE_A));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, this->tileK * this->tileN * sizeof(DTYPE_B));
        pipe.InitBuffer(outQueueC, BUFFER_NUM, this->tileM * this->tileN * sizeof(DTYPE_C));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = (this->m + this->tileM - 1) / this->tileM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.AllocTensor<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.AllocTensor<DTYPE_B>();
        uint32_t offsetA = progress * this->tileM * this->blockK;
        uint32_t offsetB = 0;
        AscendC::DataCopy(aLocal, aGm[offsetA], this->tileM * this->blockK);
        AscendC::DataCopy(bLocal, bGm[offsetB], this->blockK * this->blockN);
        inQueueA.EnQue(aLocal);
        inQueueB.EnQue(bLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.DeQue<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.DeQue<DTYPE_B>();
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.AllocTensor<DTYPE_C>();
        AscendC::MatMul(cLocal, aLocal, bLocal, this->tileM, this->blockK, this->blockN);
        outQueueC.EnQue<DTYPE_C>(cLocal);
        inQueueA.FreeTensor(aLocal);
        inQueueB.FreeTensor(bLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.DeQue<DTYPE_C>();
        uint32_t offsetC = progress * this->tileM * this->blockN;
        AscendC::DataCopy(cGm[offsetC], cLocal, this->tileM * this->blockN);
        outQueueC.FreeTensor(cLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::GlobalTensor<DTYPE_A> aGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_C> cGm;
    uint32_t m;
    uint32_t k;
    uint32_t n;
    uint32_t tileM;
    uint32_t tileK;
    uint32_t tileN;
    uint32_t blockM;
    uint32_t blockK;
    uint32_t blockN;
};

extern "C" __global__ __aicore__ void matmul_with_transposed_b_custom(GM_ADDR a, GM_ADDR b, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulWithTransposedB op;
    op.Init(a, b, c, tiling_data.m, tiling_data.k, tiling_data.n, tiling_data.tileM, tiling_data.tileK, tiling_data.tileN);
    op.Process();
}
