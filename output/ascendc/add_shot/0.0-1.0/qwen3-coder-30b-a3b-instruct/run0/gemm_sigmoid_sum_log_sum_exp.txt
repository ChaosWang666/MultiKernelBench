kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelGemmSigmoidSumLogSumExp {
public:
    __aicore__ inline KernelGemmSigmoidSumLogSumExp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w1, GM_ADDR b1, GM_ADDR w2, GM_ADDR b2, GM_ADDR z, 
                                uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize, uint32_t outputSize)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->outputSize = outputSize;
        this->blockLength = batchSize / AscendC::GetBlockNum();
        this->tileLength = this->blockLength * inputSize;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx() * inputSize, this->blockLength * inputSize);
        w1Gm.SetGlobalBuffer((__gm__ float *)w1, inputSize * hiddenSize);
        b1Gm.SetGlobalBuffer((__gm__ float *)b1, hiddenSize);
        w2Gm.SetGlobalBuffer((__gm__ float *)w2, hiddenSize * outputSize);
        b2Gm.SetGlobalBuffer((__gm__ float *)b2, outputSize);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueW1, BUFFER_NUM, inputSize * hiddenSize * sizeof(float));
        pipe.InitBuffer(inQueueB1, BUFFER_NUM, hiddenSize * sizeof(float));
        pipe.InitBuffer(inQueueW2, BUFFER_NUM, hiddenSize * outputSize * sizeof(float));
        pipe.InitBuffer(inQueueB2, BUFFER_NUM, outputSize * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->blockLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        // First Gemm: x * w1 + b1
        CopyIn(0);
        ComputeGemm1(0);
        CopyOut(0);

        // Sigmoid
        CopyIn(1);
        ComputeSigmoid(1);
        CopyOut(1);

        // Second Gemm: result * w2 + b2
        CopyIn(2);
        ComputeGemm2(2);
        CopyOut(2);

        // LogSumExp
        CopyIn(3);
        ComputeLogSumExp(3);
        CopyOut(3);
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        if(progress == 0) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[0], this->tileLength);
            inQueueX.EnQue(xLocal);
        } else if(progress == 1) {
            AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
            inQueueX.FreeTensor(xLocal);
        } else if(progress == 2) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[0], this->tileLength);
            inQueueX.EnQue(xLocal);
        } else if(progress == 3) {
            AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
            inQueueX.FreeTensor(xLocal);
        }
    }
    __aicore__ inline void ComputeGemm1(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> w1Local = inQueueW1.AllocTensor<float>();
        AscendC::LocalTensor<float> b1Local = inQueueB1.AllocTensor<float>();
        AscendC::LocalTensor<float> yLocal = outQueueZ.AllocTensor<float>();
        AscendC::MatMul(yLocal, xLocal, w1Local, b1Local, this->blockLength, this->inputSize, this->hiddenSize);
        outQueueZ.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueW1.FreeTensor(w1Local);
        inQueueB1.FreeTensor(b1Local);
    }
    __aicore__ inline void ComputeSigmoid(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueZ.AllocTensor<float>();
        AscendC::Sigmoid(yLocal, xLocal, this->blockLength * this->hiddenSize);
        outQueueZ.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void ComputeGemm2(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> w2Local = inQueueW2.AllocTensor<float>();
        AscendC::LocalTensor<float> b2Local = inQueueB2.AllocTensor<float>();
        AscendC::LocalTensor<float> yLocal = outQueueZ.AllocTensor<float>();
        AscendC::MatMul(yLocal, xLocal, w2Local, b2Local, this->blockLength, this->hiddenSize, this->outputSize);
        outQueueZ.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueW2.FreeTensor(w2Local);
        inQueueB2.FreeTensor(b2Local);
    }
    __aicore__ inline void ComputeLogSumExp(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueZ.AllocTensor<float>();
        AscendC::LogSumExp(yLocal, xLocal, this->blockLength, this->outputSize);
        outQueueZ.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[0], zLocal, this->blockLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueW1, inQueueB1, inQueueW2, inQueueB2;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> w1Gm;
    AscendC::GlobalTensor<float> b1Gm;
    AscendC::GlobalTensor<float> w2Gm;
    AscendC::GlobalTensor<float> b2Gm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t inputSize;
    uint32_t hiddenSize;
    uint32_t outputSize;
    uint32_t blockLength;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void gemm_sigmoid_sum_log_sum_exp_custom(GM_ADDR x, GM_ADDR w1, GM_ADDR b1, GM_ADDR w2, GM_ADDR b2, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmSigmoidSumLogSumExp op;
    op.Init(x, w1, b1, w2, b2, z, tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize, tiling_data.outputSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, input_size, hidden_size, output_size):
        super(ModelNew, self).__init__()
        self.linear1 = torch.nn.Linear(input_size, hidden_size)
        self.linear2 = torch.nn.Linear(hidden_size, output_size)

    def forward(self, x):
        x = self.linear1(x)
        x = torch.sigmoid(x)
        x = self.linear2(x)
        x = torch.logsumexp(x, dim=1)
        return x
