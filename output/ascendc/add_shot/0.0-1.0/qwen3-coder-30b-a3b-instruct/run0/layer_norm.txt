kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelLayerNorm {
public:
    __aicore__ inline KernelLayerNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, uint32_t totalLength, uint32_t featureSize, uint32_t batchSize)
    {
        this->totalLength = totalLength;
        this->featureSize = featureSize;
        this->batchSize = batchSize;
        this->blockLength = totalLength / AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        gammaGm.SetGlobalBuffer((__gm__ float *)gamma, featureSize);
        betaGm.SetGlobalBuffer((__gm__ float *)beta, featureSize);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(inQueueGamma, BUFFER_NUM, featureSize * sizeof(float));
        pipe.InitBuffer(inQueueBeta, BUFFER_NUM, featureSize * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->blockLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->batchSize;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t batchId)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.AllocTensor<float>();
        AscendC::LocalTensor<float> betaLocal = inQueueBeta.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[batchId * this->featureSize], this->featureSize);
        AscendC::DataCopy(gammaLocal, gammaGm, this->featureSize);
        AscendC::DataCopy(betaLocal, betaGm, this->featureSize);
        inQueueX.EnQue(xLocal);
        inQueueGamma.EnQue(gammaLocal);
        inQueueBeta.EnQue(betaLocal);
    }
    __aicore__ inline void Compute(int32_t batchId)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> gammaLocal = inQueueGamma.DeQue<float>();
        AscendC::LocalTensor<float> betaLocal = inQueueBeta.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        // Compute mean
        float sum = 0.0f;
        for (int32_t i = 0; i < this->featureSize; i++) {
            sum += xLocal[i];
        }
        float mean = sum / this->featureSize;

        // Compute variance
        float varSum = 0.0f;
        for (int32_t i = 0; i < this->featureSize; i++) {
            float diff = xLocal[i] - mean;
            varSum += diff * diff;
        }
        float variance = varSum / this->featureSize;

        // Normalize and scale
        const float epsilon = 1e-5f;
        float invStd = 1.0f / sqrtf(variance + epsilon);
        for (int32_t i = 0; i < this->featureSize; i++) {
            yLocal[i] = (xLocal[i] - mean) * invStd * gammaLocal[i] + betaLocal[i];
        }

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueGamma.FreeTensor(gammaLocal);
        inQueueBeta.FreeTensor(betaLocal);
    }
    __aicore__ inline void CopyOut(int32_t batchId)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[batchId * this->featureSize], yLocal, this->featureSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueGamma, inQueueBeta;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t totalLength;
    uint32_t featureSize;
    uint32_t batchSize;
    uint32_t blockLength;
};

extern "C" __global__ __aicore__ void layer_norm_custom(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelLayerNorm op;
    op.Init(x, gamma, beta, y, tiling_data.totalLength, tiling_data.featureSize, tiling_data.batchSize);
    op.Process();
}
