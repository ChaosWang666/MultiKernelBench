kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelMatmulWithTransposedA {
public:
    __aicore__ inline KernelMatmulWithTransposedA() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR c, uint32_t m, uint32_t k, uint32_t n, uint32_t tileM, uint32_t tileK, uint32_t tileN)
    {
        this->m = m;
        this->k = k;
        this->n = n;
        this->tileM = tileM;
        this->tileK = tileK;
        this->tileN = tileN;
        this->loopCount = (m + tileM - 1) / tileM * (n + tileN - 1) / tileN;
        this->aGm.SetGlobalBuffer((__gm__ DTYPE_A *)a, m * k);
        this->bGm.SetGlobalBuffer((__gm__ DTYPE_B *)b, k * n);
        this->cGm.SetGlobalBuffer((__gm__ DTYPE_C *)c, m * n);
        this->pipe.InitBuffer(inQueueA, BUFFER_NUM, tileM * tileK * sizeof(DTYPE_A));
        this->pipe.InitBuffer(inQueueB, BUFFER_NUM, tileK * tileN * sizeof(DTYPE_B));
        this->pipe.InitBuffer(outQueueC, BUFFER_NUM, tileM * tileN * sizeof(DTYPE_C));
    }
    __aicore__ inline void Process()
    {
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        uint32_t tileMId = progress % ((m + tileM - 1) / tileM);
        uint32_t tileNId = progress / ((m + tileM - 1) / tileM);
        uint32_t aOffset = tileMId * tileM * k;
        uint32_t bOffset = tileNId * tileN;
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.AllocTensor<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.AllocTensor<DTYPE_B>();
        AscendC::DataCopy(aLocal, aGm[aOffset], tileM * tileK);
        AscendC::DataCopy(bLocal, bGm[bOffset], tileK * tileN);
        inQueueA.EnQue(aLocal);
        inQueueB.EnQue(bLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = inQueueA.DeQue<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = inQueueB.DeQue<DTYPE_B>();
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.AllocTensor<DTYPE_C>();
        AscendC::MatMul(cLocal, aLocal, bLocal, tileM, tileK, tileN, false, true);
        outQueueC.EnQue<DTYPE_C>(cLocal);
        inQueueA.FreeTensor(aLocal);
        inQueueB.FreeTensor(bLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        uint32_t tileMId = progress % ((m + tileM - 1) / tileM);
        uint32_t tileNId = progress / ((m + tileM - 1) / tileM);
        uint32_t cOffset = tileMId * tileM * n + tileNId * tileN;
        AscendC::LocalTensor<DTYPE_C> cLocal = outQueueC.DeQue<DTYPE_C>();
        AscendC::DataCopy(cGm[cOffset], cLocal, tileM * tileN);
        outQueueC.FreeTensor(cLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueA, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueC;
    AscendC::GlobalTensor<DTYPE_A> aGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_C> cGm;
    uint32_t m;
    uint32_t k;
    uint32_t n;
    uint32_t tileM;
    uint32_t tileK;
    uint32_t tileN;
    uint32_t loopCount;
};

extern "C" __global__ __aicore__ void matmul_with_transposed_a_custom(GM_ADDR a, GM_ADDR b, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulWithTransposedA op;
    op.Init(a, b, c, tiling_data.m, tiling_data.k, tiling_data.n, tiling_data.tileM, tiling_data.tileK, tiling_data.tileN);
    op.Process();
}
