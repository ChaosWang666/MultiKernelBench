kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelTripletMarginLoss {
public:
    __aicore__ inline KernelTripletMarginLoss() {}
    __aicore__ inline void Init(GM_ADDR anchor, GM_ADDR positive, GM_ADDR negative, GM_ADDR margin, GM_ADDR loss, uint32_t batchSize, uint32_t inputDim)
    {
        this->batchSize = batchSize;
        this->inputDim = inputDim;
        this->blockLength = batchSize / AscendC::GetBlockNum();
        this->tileLength = this->blockLength / BUFFER_NUM;

        anchorGm.SetGlobalBuffer((__gm__ float *)anchor + this->blockLength * AscendC::GetBlockIdx(), this->blockLength * this->inputDim);
        positiveGm.SetGlobalBuffer((__gm__ float *)positive + this->blockLength * AscendC::GetBlockIdx(), this->blockLength * this->inputDim);
        negativeGm.SetGlobalBuffer((__gm__ float *)negative + this->blockLength * AscendC::GetBlockIdx(), this->blockLength * this->inputDim);
        marginGm.SetGlobalBuffer((__gm__ float *)margin, 1);
        lossGm.SetGlobalBuffer((__gm__ float *)loss + AscendC::GetBlockIdx(), this->blockLength);

        pipe.InitBuffer(inQueueAnchor, BUFFER_NUM, this->tileLength * this->inputDim * sizeof(float));
        pipe.InitBuffer(inQueuePositive, BUFFER_NUM, this->tileLength * this->inputDim * sizeof(float));
        pipe.InitBuffer(inQueueNegative, BUFFER_NUM, this->tileLength * this->inputDim * sizeof(float));
        pipe.InitBuffer(outQueueLoss, BUFFER_NUM, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->blockLength / this->tileLength;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> anchorLocal = inQueueAnchor.AllocTensor<float>();
        AscendC::LocalTensor<float> positiveLocal = inQueuePositive.AllocTensor<float>();
        AscendC::LocalTensor<float> negativeLocal = inQueueNegative.AllocTensor<float>();
        AscendC::DataCopy(anchorLocal, anchorGm[progress * this->tileLength * this->inputDim], this->tileLength * this->inputDim);
        AscendC::DataCopy(positiveLocal, positiveGm[progress * this->tileLength * this->inputDim], this->tileLength * this->inputDim);
        AscendC::DataCopy(negativeLocal, negativeGm[progress * this->tileLength * this->inputDim], this->tileLength * this->inputDim);
        inQueueAnchor.EnQue(anchorLocal);
        inQueuePositive.EnQue(positiveLocal);
        inQueueNegative.EnQue(negativeLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> anchorLocal = inQueueAnchor.DeQue<float>();
        AscendC::LocalTensor<float> positiveLocal = inQueuePositive.DeQue<float>();
        AscendC::LocalTensor<float> negativeLocal = inQueueNegative.DeQue<float>();
        AscendC::LocalTensor<float> lossLocal = outQueueLoss.AllocTensor<float>();
        float marginValue = marginGm[0];
        for (int32_t i = 0; i < this->tileLength; i++) {
            float dist_pos = 0.0f;
            float dist_neg = 0.0f;
            for (int32_t j = 0; j < this->inputDim; j++) {
                float diff_pos = anchorLocal[i * this->inputDim + j] - positiveLocal[i * this->inputDim + j];
                float diff_neg = anchorLocal[i * this->inputDim + j] - negativeLocal[i * this->inputDim + j];
                dist_pos += diff_pos * diff_pos;
                dist_neg += diff_neg * diff_neg;
            }
            float loss_val = max(0.0f, dist_pos - dist_neg + marginValue);
            lossLocal[i] = loss_val;
        }
        outQueueLoss.EnQue<float>(lossLocal);
        inQueueAnchor.FreeTensor(anchorLocal);
        inQueuePositive.FreeTensor(positiveLocal);
        inQueueNegative.FreeTensor(negativeLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> lossLocal = outQueueLoss.DeQue<float>();
        AscendC::DataCopy(lossGm[progress * this->tileLength], lossLocal, this->tileLength);
        outQueueLoss.FreeTensor(lossLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueAnchor, inQueuePositive, inQueueNegative;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueLoss;
    AscendC::GlobalTensor<float> anchorGm;
    AscendC::GlobalTensor<float> positiveGm;
    AscendC::GlobalTensor<float> negativeGm;
    AscendC::GlobalTensor<float> marginGm;
    AscendC::GlobalTensor<float> lossGm;
    uint32_t blockSize;
    uint32_t batchSize;
    uint32_t inputDim;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void triplet_margin_loss_custom(GM_ADDR anchor, GM_ADDR positive, GM_ADDR negative, GM_ADDR margin, GM_ADDR loss, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelTripletMarginLoss op;
    op.Init(anchor, positive, negative, margin, loss, tiling_data.batchSize, tiling_data.inputDim);
    op.Process();
}
