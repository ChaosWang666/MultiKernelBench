constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelConvTranspose2dMinSumGeluAdd {
public:
    __aicore__ inline KernelConvTranspose2dMinSumGeluAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t batch, uint32_t inChannels, uint32_t outChannels,
                                uint32_t height, uint32_t width, uint32_t kernelSize, uint32_t stride,
                                uint32_t padding, uint32_t outputPadding, uint32_t biasShape)
    {
        this->batch = batch;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->height = height;
        this->width = width;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->outputPadding = outputPadding;
        this->biasShape = biasShape;

        // Initialize global buffers
        xGm.SetGlobalBuffer((__gm__ float *)x, batch * inChannels * height * width);
        zGm.SetGlobalBuffer((__gm__ float *)z, batch * outChannels * height * width);
        
        // Initialize pipeline queues
        pipe.InitBuffer(inQueue, BUFFER_NUM, 1024 * sizeof(float)); // Adjust size as needed
        pipe.InitBuffer(outQueue, BUFFER_NUM, 1024 * sizeof(float)); // Adjust size as needed
    }

    __aicore__ inline void Process()
    {
        // ConvTranspose2d
        ConvTranspose2d();

        // Min operation
        MinOperation();

        // Sum operation
        SumOperation();

        // GELU activation
        GeluActivation();

        // Addition with bias
        AddBias();
    }

private:
    __aicore__ inline void ConvTranspose2d()
    {
        // Placeholder for actual implementation
        // This would involve implementing the transposed convolution logic
        // For now, we just copy data through
        AscendC::LocalTensor<float> localTensor = inQueue.AllocTensor<float>();
        AscendC::DataCopy(localTensor, xGm, batch * inChannels * height * width);
        inQueue.EnQue(localTensor);
        AscendC::LocalTensor<float> outTensor = outQueue.AllocTensor<float>();
        AscendC::DataCopy(outTensor, localTensor, batch * inChannels * height * width);
        outQueue.EnQue(outTensor);
        inQueue.FreeTensor(localTensor);
        outQueue.FreeTensor(outTensor);
    }

    __aicore__ inline void MinOperation()
    {
        // Placeholder for min operation along channel dimension
        AscendC::LocalTensor<float> localTensor = outQueue.DeQue<float>();
        // Perform min operation here
        outQueue.EnQue(localTensor);
        outQueue.FreeTensor(localTensor);
    }

    __aicore__ inline void SumOperation()
    {
        // Placeholder for sum operation along height dimension
        AscendC::LocalTensor<float> localTensor = outQueue.DeQue<float>();
        // Perform sum operation here
        outQueue.EnQue(localTensor);
        outQueue.FreeTensor(localTensor);
    }

    __aicore__ inline void GeluActivation()
    {
        // Placeholder for GELU activation
        AscendC::LocalTensor<float> localTensor = outQueue.DeQue<float>();
        // Apply GELU activation
        outQueue.EnQue(localTensor);
        outQueue.FreeTensor(localTensor);
    }

    __aicore__ inline void AddBias()
    {
        // Placeholder for adding bias
        AscendC::LocalTensor<float> localTensor = outQueue.DeQue<float>();
        // Add bias here
        outQueue.EnQue(localTensor);
        outQueue.FreeTensor(localTensor);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batch;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t height;
