kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelEmbedding {
public:
    __aicore__ inline KernelEmbedding() {}
    __aicore__ inline void Init(GM_ADDR indices, GM_ADDR weight, GM_ADDR output, uint32_t totalLength, uint32_t tileNum, uint32_t vocabSize, uint32_t embeddingDim)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->vocabSize = vocabSize;
        this->embeddingDim = embeddingDim;

        indicesGm.SetGlobalBuffer((__gm__ int32_t *)indices + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, this->vocabSize * this->embeddingDim);
        outputGm.SetGlobalBuffer((__gm__ float *)output + this->blockLength * this->embeddingDim, this->blockLength * this->embeddingDim);
        pipe.InitBuffer(inQueueIndices, BUFFER_NUM, this->tileLength * sizeof(int32_t));
        pipe.InitBuffer(outQueueOutput, BUFFER_NUM, this->tileLength * this->embeddingDim * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<int32_t> indicesLocal = inQueueIndices.AllocTensor<int32_t>();
        AscendC::DataCopy(indicesLocal, indicesGm[progress * this->tileLength], this->tileLength);
        inQueueIndices.EnQue(indicesLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<int32_t> indicesLocal = inQueueIndices.DeQue<int32_t>();
        AscendC::LocalTensor<float> outputLocal = outQueueOutput.AllocTensor<float>();
        AscendC::LocalTensor<float> weightLocal = AscendC::LocalTensor<float>::MallocLocalTensor(this->embeddingDim);
        for (int32_t i = 0; i < this->tileLength; i++) {
            int32_t index = indicesLocal[i];
            if (index >= 0 && index < this->vocabSize) {
                AscendC::DataCopy(weightLocal, weightGm[index * this->embeddingDim], this->embeddingDim);
                for (int32_t j = 0; j < this->embeddingDim; j++) {
                    outputLocal[i * this->embeddingDim + j] = weightLocal[j];
                }
            } else {
                for (int32_t j = 0; j < this->embeddingDim; j++) {
                    outputLocal[i * this->embeddingDim + j] = 0.0f;
                }
            }
        }
        outQueueOutput.EnQue<float>(outputLocal);
        inQueueIndices.FreeTensor(indicesLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> outputLocal = outQueueOutput.DeQue<float>();
        AscendC::DataCopy(outputGm[progress * this->tileLength * this->embeddingDim], outputLocal, this->tileLength * this->embeddingDim);
        outQueueOutput.FreeTensor(outputLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueIndices;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueOutput;
    AscendC::GlobalTensor<int32_t> indicesGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> outputGm;
    uint32_t blockLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t vocabSize;
    uint32_t embeddingDim;
};

extern "C" __global__ __aicore__ void embedding_custom(GM_ADDR indices, GM_ADDR weight, GM_ADDR output, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelEmbedding op;
    op.Init(indices, weight, output, tiling_data.totalLength, tiling_data.tileNum, tiling_data.vocabSize, tiling_data.embeddingDim);
    op.Process();
}
