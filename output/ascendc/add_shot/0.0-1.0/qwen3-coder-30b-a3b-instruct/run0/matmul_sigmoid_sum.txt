kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue
 
class KernelMatmulSigmoidSum {
public:
    __aicore__ inline KernelMatmulSigmoidSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, 
                                uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->tileNum = tileNum;
        this->blockLength = inputSize / AscendC::GetBlockNum();
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, inputSize * hiddenSize);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, hiddenSize);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueWeight, BUFFER_NUM, this->tileLength * hiddenSize * sizeof(float));
        pipe.InitBuffer(inQueueBias, BUFFER_NUM, hiddenSize * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, hiddenSize * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t batch = 0; batch < batchSize; batch++) {
            // Matrix multiplication
            for (int32_t i = 0; i < hiddenSize; i++) {
                CopyInMatmul(batch, i);
                ComputeMatmul(batch, i);
                CopyOutMatmul(batch, i);
            }
            // Sigmoid and sum
            CopyInSigmoidSum(batch);
            ComputeSigmoidSum(batch);
            CopyOutSigmoidSum(batch);
        }
    }

private:
    __aicore__ inline void CopyInMatmul(uint32_t batch, uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> weightLocal = inQueueWeight.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[0], this->tileLength);
        AscendC::DataCopy(weightLocal, weightGm[row * this->inputSize], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueWeight.EnQue(weightLocal);
    }
    __aicore__ inline void ComputeMatmul(uint32_t batch, uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> weightLocal = inQueueWeight.DeQue<float>();
        AscendC::LocalTensor<float> resultLocal = outQueueZ.AllocTensor<float>();
        AscendC::Mul(resultLocal, xLocal, weightLocal, this->tileLength);
        outQueueZ.EnQue<float>(resultLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueWeight.FreeTensor(weightLocal);
    }
    __aicore__ inline void CopyOutMatmul(uint32_t batch, uint32_t row)
    {
        AscendC::LocalTensor<float> resultLocal = outQueueZ.DeQue<float>();
        // For simplicity, we assume direct accumulation here
        outQueueZ.FreeTensor(resultLocal);
    }
    __aicore__ inline void CopyInSigmoidSum(uint32_t batch)
    {
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
        AscendC::DataCopy(biasLocal, biasGm[0], hiddenSize);
        inQueueBias.EnQue(biasLocal);
    }
    __aicore__ inline void ComputeSigmoidSum(uint32_t batch)
    {
        AscendC::LocalTensor<float> biasLocal = inQueueBias.DeQue<float>();
        AscendC::LocalTensor<float> resultLocal = outQueueZ.AllocTensor<float>();
        AscendC::Sigmoid(resultLocal, biasLocal, hiddenSize);
        outQueueZ.EnQue<float>(resultLocal);
        inQueueBias.FreeTensor(biasLocal);
    }
    __aicore__ inline void CopyOutSigmoidSum(uint32_t batch)
    {
        AscendC::LocalTensor<float> resultLocal = outQueueZ.DeQue<float>();
        AscendC::ReduceSum(zGm[batch], resultLocal, hiddenSize, 1);
        outQueueZ.FreeTensor(resultLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueWeight, inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t inputSize;
    uint32_t hiddenSize;
    uint32_t tileNum;
    uint32_t blockLength;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void matmul_sigmoid_sum_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSigmoidSum op;
    op.Init(x, weight, bias, z, tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize, tiling_data.tileNum);
    op.Process();
}
