kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2; // tensor num for each queue

class KernelGroupQueryAttention {
public:
    __aicore__ inline KernelGroupQueryAttention() {}
    __aicore__ inline void Init(GM_ADDR query, GM_ADDR key, GM_ADDR value, GM_ADDR output,
                                uint32_t batchSize, uint32_t seqLen, uint32_t numHeads, uint32_t numKvHeads, uint32_t headDim, uint32_t groupSize)
    {
        this->batchSize = batchSize;
        this->seqLen = seqLen;
        this->numHeads = numHeads;
        this->numKvHeads = numKvHeads;
        this->headDim = headDim;
        this->groupSize = groupSize;
        this->blockLength = seqLen * headDim;
        this->totalLength = batchSize * seqLen * numHeads * headDim;

        queryGm.SetGlobalBuffer((__gm__ float *)query, this->totalLength);
        keyGm.SetGlobalBuffer((__gm__ float *)key, this->batchSize * seqLen * numKvHeads * headDim);
        valueGm.SetGlobalBuffer((__gm__ float *)value, this->batchSize * seqLen * numKvHeads * headDim);
        outputGm.SetGlobalBuffer((__gm__ float *)output, this->totalLength);
        
        pipe.InitBuffer(inQueueQuery, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(inQueueKey, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(inQueueValue, BUFFER_NUM, this->blockLength * sizeof(float));
        pipe.InitBuffer(outQueueAttn, BUFFER_NUM, this->seqLen * this->seqLen * sizeof(float));
        pipe.InitBuffer(outQueueOutput, BUFFER_NUM, this->blockLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t batch = 0; batch < batchSize; ++batch) {
            for (uint32_t head = 0; head < numHeads; head += groupSize) {
                ProcessHead(batch, head);
            }
        }
    }

private:
    __aicore__ inline void ProcessHead(uint32_t batch, uint32_t headStart)
    {
        uint32_t headEnd = headStart + groupSize;
        for (uint32_t head = headStart; head < headEnd; ++head) {
            CopyIn(batch, head);
            ComputeAttn(batch, head);
            ComputeOutput(batch, head);
            CopyOut(batch, head);
        }
    }

    __aicore__ inline void CopyIn(uint32_t batch, uint32_t head)
    {
        uint32_t offsetQ = batch * seqLen * numHeads * headDim + head * seqLen * headDim;
        uint32_t offsetK = batch * seqLen * numKvHeads * headDim + (head / groupSize) * seqLen * headDim;
        uint32_t offsetV = batch * seqLen * numKvHeads * headDim + (head / groupSize) * seqLen * headDim;

        AscendC::LocalTensor<float> queryLocal = inQueueQuery.AllocTensor<float>();
        AscendC::LocalTensor<float> keyLocal = inQueueKey.AllocTensor<float>();
        AscendC::LocalTensor<float> valueLocal = inQueueValue.AllocTensor<float>();

        AscendC::DataCopy(queryLocal, queryGm[offsetQ], seqLen * headDim);
        AscendC::DataCopy(keyLocal, keyGm[offsetK], seqLen * headDim);
        AscendC::DataCopy(valueLocal, valueGm[offsetV], seqLen * headDim);

        inQueueQuery.EnQue(queryLocal);
        inQueueKey.EnQue(keyLocal);
        inQueueValue.EnQue(valueLocal);
    }

    __aicore__ inline void ComputeAttn(uint32_t batch, uint32_t head)
    {
        AscendC::LocalTensor<float> queryLocal = inQueueQuery.DeQue<float>();
        AscendC::LocalTensor<float> keyLocal = inQueueKey.DeQue<float>();
        AscendC::LocalTensor<float> attnLocal = outQueueAttn.AllocTensor<float>();

        // Simple matrix multiplication for attention scores
        for (uint32_t i = 0; i < seqLen; ++i) {
            for (uint32_t j = 0; j < seqLen; ++j) {
                float sum = 0.0f;
                for (uint32_t k = 0; k < headDim; ++k) {
                    sum += queryLocal[i * headDim + k] * keyLocal[j * headDim + k];
                }
                attnLocal[i * seqLen + j] = sum / sqrtf(static_cast<float>(headDim));
            }
        }

        outQueueAttn.EnQue<float>(attnLocal);
        inQueueQuery.FreeTensor(queryLocal);
        inQueueKey.FreeTensor(keyLocal);
    }

    __aicore__ inline void ComputeOutput(uint32_t batch, uint32_t head)
    {
        AscendC::LocalTensor<float> attnLocal = outQueueAttn.DeQue<float>();
        AscendC::LocalTensor<float> valueLocal = inQueueValue.DeQue<float>();
        AscendC::LocalTensor<float> outputLocal = outQueueOutput.AllocTensor<float>();

        // Apply softmax and compute output
        for (uint32_t i = 0; i < seqLen; ++i) {
            float maxVal = -1e9f;
            for (uint32_t j = 0; j < seqLen; ++j) {
                maxVal = fmaxf(maxVal, attnLocal[i * seqLen + j]);
            }
            float sum = 0.0f;
            for (uint32_t j = 0; j < seqLen; ++j) {
                float expVal = expf(attnLocal[i * seqLen + j] - maxVal);
                attnLocal[i * seqLen + j] = expVal;
                sum += expVal;
            }
            for (uint32_t j = 0; j < seqLen; ++j) {
                attnLocal[i * seqLen + j] /= sum;
            }
        }

        for (uint32_t i = 0; i < seqLen; ++i) {
            for (uint32_t k = 0; k < headDim; ++k) {
                float sum = 0.0f;
                for (uint32_t j = 0; j < seqLen; ++j) {
                    sum += attnLocal[i * seqLen + j] * valueLocal[j * headDim + k];
                }
                outputLocal[i * headDim + k] = sum;
            }
        }

        outQueueOutput.EnQue<float>(outputLocal);
        outQueueAttn.FreeTensor(attnLocal);
        inQueueValue.FreeTensor(valueLocal);
    }

    __aicore__ inline void CopyOut(uint32_t batch, uint32_t head)
    {
        uint32_t offsetOut = batch * seqLen * numHeads * headDim + head * seqLen * headDim;
        AscendC::LocalTensor<float> outputLocal = outQueueOutput.DeQue<float>();
        AscendC::DataCopy(outputGm[offsetOut], outputLocal, seqLen * headDim);
        outQueueOutput.FreeTensor(outputLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueQuery, inQueueKey, inQueueValue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueAttn, outQueueOutput;
    AscendC::GlobalTensor<float> queryGm, keyGm, valueGm, outputGm;
    uint32_t batchSize;
    uint32_t seqLen;
    uint32_t numHeads;
    uint32_t numKvHeads;
    uint32_t headDim;
    uint32_t groupSize;
    uint32_t blockLength;
    uint32_t totalLength;
};

extern "C" __global__ __aicore__ void group_query_attention_custom(
    GM_ADDR query, GM_ADDR key, GM_ADDR value, GM_ADDR output, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGroupQueryAttention op;
    op.Init(query, key, value, output,
            tiling_data.batchSize, tiling_data.seqLen, tiling_data.numHeads, tiling_data.numKvHeads, tiling_data.headDim, tiling_data.groupSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, d_model=1024, num_heads=16, num_kv_heads=4):
        super().__init__()
        assert num_heads % num_kv_heads == 0, "num_heads must be divisible by num_kv_heads"
        self.num_heads = num_heads
        self.num_kv_heads = num_kv_heads
        self.group_size = num_heads // num_kv_heads

        self.q_proj = torch.nn.Linear(d_model, d_model)
        self.k_proj = torch.nn.Linear(d_model, d_model * num_kv_heads // num_heads)
        self.v_proj = torch.nn.Linear(d_model, d_model * num_kv_heads // num_heads)
        self.out_proj = torch.nn.Linear(d_model, d_model)

    def forward(self, x):
        B, L, D = x.shape
        H, H_kv = self.num_heads, self.num_kv_heads
        head_dim = D // H

        # projections
        Q = self.q_proj(x).view(B, L, H, head_dim)
        K = self.k_proj(x).view(B, L, H_kv, head_dim)
        V = self.v_proj(x).view(B, L, H_kv, head_dim)

        # Expand K/V to match query groups
        if self.group_size > 1:
            K = K.repeat_interleave(self.group_size, dim=2)
            V = V.repeat_interleave(self.group_size, dim=2)

        # attention
        attn = custom_ops_lib.group_query_attention_custom(Q, K, V)
        out = attn.reshape(B, L, D)
        return self.out_proj(out)
'''
