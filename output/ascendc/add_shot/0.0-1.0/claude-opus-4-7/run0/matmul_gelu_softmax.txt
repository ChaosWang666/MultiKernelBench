kernel_src="""
#include "kernel_operator.h"

class KernelMatmulGeluSoftmax {
public:
    __aicore__ inline KernelMatmulGeluSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                 uint32_t M, uint32_t K, uint32_t N)
    {
        this->M = M;
        this->K = K;
        this->N = N;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (M + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        this->endRow = this->startRow + this->rowsPerBlock;
        if (this->endRow > M) this->endRow = M;

        xGm.SetGlobalBuffer((__gm__ float*)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float*)weight, N * K);
        bGm.SetGlobalBuffer((__gm__ float*)bias, N);
        zGm.SetGlobalBuffer((__gm__ float*)z, M * N);

        pipe.InitBuffer(inQueueX, 1, K * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, K * sizeof(float));
        pipe.InitBuffer(inQueueB, 1, N * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, N * sizeof(float));
        pipe.InitBuffer(tmpBuf, N * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Load bias once
        AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
        AscendC::DataCopy(bLocal, bGm, this->N);
        inQueueB.EnQue(bLocal);
        bLocal = inQueueB.DeQue<float>();

        for (uint32_t row = this->startRow; row < this->endRow; row++) {
            // Load x row
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[row * this->K], this->K);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::LocalTensor<float> outLocal = outQueueZ.AllocTensor<float>();
            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

            // Matmul: out[n] = sum_k x[k] * w[n,k] + b[n]
            for (uint32_t n = 0; n < this->N; n++) {
                AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                AscendC::DataCopy(wLocal, wGm[n * this->K], this->K);
                inQueueW.EnQue(wLocal);
                wLocal = inQueueW.DeQue<float>();

                AscendC::Mul(tmp, xLocal, wLocal, this->K);
                float sum = 0.0f;
                for (uint32_t k = 0; k < this->K; k++) {
                    sum += tmp.GetValue(k);
                }
                outLocal.SetValue(n, sum + bLocal.GetValue(n));
                inQueueW.FreeTensor(wLocal);
            }
            inQueueX.FreeTensor(xLocal);

            // GELU approx: 0.5 * x * (1 + tanh(sqrt(2/pi)*(x + 0.044715*x^3)))
            for (uint32_t n = 0; n < this->N; n++) {
                float v = outLocal.GetValue(n);
                float c = 0.7978845608f * (v + 0.044715f * v * v * v);
                float t = (c > 10.0f) ? 1.0f : ((c < -10.0f) ? -1.0f : ((expf(c) - expf(-c)) / (expf(c) + expf(-c))));
                outLocal.SetValue(n, 0.5f * v * (1.0f + t));
            }

            // Softmax
            float maxv = outLocal.GetValue(0);
            for (uint32_t n = 1; n < this->N; n++) {
                float v = outLocal.GetValue(n);
                if (v > maxv) maxv = v;
            }
            float sumExp = 0.0f;
            for (uint32_t n = 0; n < this->N; n++) {
                float e = expf(outLocal.GetValue(n) - maxv);
                outLocal.SetValue(n, e);
                sumExp += e;
            }
            float inv = 1.0f / sumExp;
            for (uint32_t n = 0; n < this->N; n++) {
                outLocal.SetValue(n, outLocal.GetValue(n) * inv);
            }

            outQueueZ.EnQue<float>(outLocal);
            AscendC::LocalTensor<float> zOut = outQueueZ.DeQue<float>();
            AscendC::DataCopy(zGm[row * this->N], zOut, this->N);
            outQueueZ.FreeTensor(zOut);
        }
        inQueueB.FreeTensor(bLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueW, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, zGm;
    uint32_t M, K, N;
    uint32_t rowsPerBlock, startRow, endRow;
};

extern "C" __global__ __aicore__ void matmul_gelu_softmax_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulGeluSoftmax op;
    op.Init(x, weight, bias, z, tiling_data.M, tiling_data.K, tiling_data.N);
    op.Process();
}
