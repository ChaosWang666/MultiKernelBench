kernel_src="""
#include "kernel_operator.h"

class KernelMMPSS {
public:
    __aicore__ inline KernelMMPSS() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures,
                                 uint32_t kernelSize, float scaleFactor)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->kernelSize = kernelSize;
        this->scaleFactor = scaleFactor;
        this->blockIdx = AscendC::GetBlockIdx();
        this->blockNum = AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outFeatures * inFeatures);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize);

        pipe.InitBuffer(xBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(wBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(bBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(mulBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(reduceBuf, 256 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> bLocal = bBuf.Get<float>();
        AscendC::DataCopy(bLocal, bGm, this->outFeatures);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t b = this->blockIdx; b < this->batchSize; b += this->blockNum) {
            AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
            AscendC::DataCopy(xLocal, xGm[b * this->inFeatures], this->inFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();

            // Compute matmul output[o] = bias[o] + sum_i(x[i]*w[o,i]), then maxpool, then sum, then scale
            // Process pairs for maxpool
            float totalSum = 0.0f;
            uint32_t pairs = this->outFeatures / this->kernelSize;
            
            AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
            AscendC::LocalTensor<float> mulLocal = mulBuf.Get<float>();
            AscendC::LocalTensor<float> redLocal = reduceBuf.Get<float>();

            for (uint32_t p = 0; p < pairs; p++) {
                float vals[2];
                for (uint32_t k = 0; k < this->kernelSize; k++) {
                    uint32_t o = p * this->kernelSize + k;
                    AscendC::DataCopy(wLocal, wGm[o * this->inFeatures], this->inFeatures);
                    AscendC::PipeBarrier<PIPE_ALL>();
                    AscendC::Mul(mulLocal, xLocal, wLocal, this->inFeatures);
                    AscendC::PipeBarrier<PIPE_ALL>();
                    AscendC::ReduceSum(redLocal, mulLocal, redLocal, this->inFeatures);
                    AscendC::PipeBarrier<PIPE_ALL>();
                    float dot = redLocal.GetValue(0);
                    vals[k] = dot + bLocal.GetValue(o);
                }
                float maxVal = vals[0];
                for (uint32_t k = 1; k < this->kernelSize; k++) {
                    if (vals[k] > maxVal) maxVal = vals[k];
                }
                totalSum += maxVal;
            }

            float result = totalSum * this->scaleFactor;
            redLocal.SetValue(0, result);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::DataCopyExtParams copyParams{1, sizeof(float), 0, 0, 0};
            AscendC::DataCopyPad(zGm[b], redLocal, copyParams);
            AscendC::PipeBarrier<PIPE_ALL>();
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, wBuf, bBuf, mulBuf, reduceBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, zGm;
    uint32_t batchSize, inFeatures, outFeatures, kernelSize;
    float scaleFactor;
    uint32_t blockIdx, blockNum;
};

extern "C" __global__ __aicore__ void matmul_max_pool_sum_scale_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMMPSS op;
    op.Init(x, weight, bias, z, tiling_data.batchSize, tiling_data.inFeatures,
            tiling_data.outFeatures, tiling_data.kernelSize, tiling_data.scaleFactor);
    op.Process();
}
