kernel_src="""
#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mult, GM_ADDR z, uint32_t batchSize, uint32_t channels, uint32_t spatialSize, float clampMin, float clampMax)
    {
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->clampMin = clampMin;
        this->clampMax = clampMax;
        uint32_t bIdx = AscendC::GetBlockIdx();
        uint32_t perBatch = channels * spatialSize;
        xGm.SetGlobalBuffer((__gm__ float *)x + bIdx * perBatch, perBatch);
        multGm.SetGlobalBuffer((__gm__ float *)mult, channels);
        zGm.SetGlobalBuffer((__gm__ float *)z + bIdx * spatialSize, spatialSize);

        pipe.InitBuffer(inQueueX, 1, spatialSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, spatialSize * sizeof(float));
        pipe.InitBuffer(maxBuf, spatialSize * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, spatialSize * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::Duplicate(maxLocal, -3.4e38f, spatialSize);

        for (uint32_t c = 0; c < channels; c++) {
            float multVal = multGm.GetValue(c);

            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[c * spatialSize], spatialSize);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
            // x * multiplier
            AscendC::Muls(tmp, xLocal, multVal, spatialSize);

            // Compute mean
            float sum = 0.0f;
            for (uint32_t i = 0; i < spatialSize; i++) {
                sum += tmp.GetValue(i);
            }
            float mean = sum / (float)spatialSize;

            // Compute variance
            AscendC::Adds(xLocal, tmp, -mean, spatialSize);
            AscendC::LocalTensor<float> sq = inQueueX.AllocTensor<float>();
            AscendC::Mul(sq, xLocal, xLocal, spatialSize);
            float varSum = 0.0f;
            for (uint32_t i = 0; i < spatialSize; i++) {
                varSum += sq.GetValue(i);
            }
            inQueueX.FreeTensor(sq);
            float var = varSum / (float)spatialSize;
            float invStd = 1.0f / sqrt(var + 1e-5f);

            // normalize
            AscendC::Muls(tmp, xLocal, invStd, spatialSize);
            // clamp
            AscendC::Mins(tmp, tmp, clampMax, spatialSize);
            AscendC::Maxs(tmp, tmp, clampMin, spatialSize);
            // multiply again
            AscendC::Muls(tmp, tmp, multVal, spatialSize);
            // running max
            AscendC::Max(maxLocal, maxLocal, tmp, spatialSize);

            inQueueX.FreeTensor(xLocal);
        }

        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::DataCopy(zLocal, maxLocal, spatialSize);
        outQueueZ.EnQue<float>(zLocal);
        zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm, zLocal, spatialSize);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, maxBuf;
    AscendC::GlobalTensor<float> xGm, multGm, zGm;
    uint32_t channels;
    uint32_t spatialSize;
    float clampMin;
    float clampMax;
};

extern "C" __global__ __aicore__ void conv3d_multiply_instance_norm_clamp_multiply_max_custom(GM_ADDR x, GM_ADDR mult, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, mult, z, tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize, tiling_data.clampMin, tiling_data.clampMax);
    op.Process();
}
