kernel_src="""
#include "kernel_operator.h"

class KernelGemmScaleBN {
public:
    __aicore__ inline KernelGemmScaleBN() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR scale,
                                GM_ADDR bn_weight, GM_ADDR bn_bias, GM_ADDR z,
                                uint32_t M, uint32_t N, uint32_t K, float eps)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        this->eps = eps;
        xGm.SetGlobalBuffer((__gm__ float*)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float*)weight, N * K);
        bGm.SetGlobalBuffer((__gm__ float*)bias, N);
        sGm.SetGlobalBuffer((__gm__ float*)scale, N);
        bnwGm.SetGlobalBuffer((__gm__ float*)bn_weight, N);
        bnbGm.SetGlobalBuffer((__gm__ float*)bn_bias, N);
        zGm.SetGlobalBuffer((__gm__ float*)z, M * N);
    }

    __aicore__ inline void Process()
    {
        // Compute Y[m,n] = (sum_k x[m,k]*w[n,k] + b[n]) * s[n]
        // Then batchnorm along batch dim for each n: mean, var, normalize
        for (uint32_t m = 0; m < M; m++) {
            for (uint32_t n = 0; n < N; n++) {
                float acc = 0.0f;
                for (uint32_t k = 0; k < K; k++) {
                    acc += xGm.GetValue(m * K + k) * wGm.GetValue(n * K + k);
                }
                acc += bGm.GetValue(n);
                acc *= sGm.GetValue(n);
                zGm.SetValue(m * N + n, acc);
            }
        }
        // BN
        for (uint32_t n = 0; n < N; n++) {
            float mean = 0.0f;
            for (uint32_t m = 0; m < M; m++) {
                mean += zGm.GetValue(m * N + n);
            }
            mean /= (float)M;
            float var = 0.0f;
            for (uint32_t m = 0; m < M; m++) {
                float d = zGm.GetValue(m * N + n) - mean;
                var += d * d;
            }
            var /= (float)M;
            float inv = 1.0f / AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                AscendC::ScalarGetSqrt<float>(var + eps));
            float gamma = bnwGm.GetValue(n);
            float beta = bnbGm.GetValue(n);
            for (uint32_t m = 0; m < M; m++) {
                float v = (zGm.GetValue(m * N + n) - mean) * inv * gamma + beta;
                zGm.SetValue(m * N + n, v);
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm, wGm, bGm, sGm, bnwGm, bnbGm, zGm;
    uint32_t M, N, K;
    float eps;
};

extern "C" __global__ __aicore__ void gemm_scale_batchnorm_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR scale,
    GM_ADDR bn_weight, GM_ADDR bn_bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmScaleBN op;
    op.Init(x, weight, bias, scale, bn_weight, bn_bias, z,
            tiling_data.M, tiling_data.N, tiling_data.K, tiling_data.eps);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, scale_shape, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.scale = nn.Parameter(torch.rand(scale_shape))
        self.bn = nn.BatchNorm1d(out_features, eps=eps, momentum=momentum)
        self.eps = eps

    def forward(self, x):
        return custom_ops_lib.gemm_scale_batchnorm_custom(
            x.contiguous(),
            self.gemm.weight.contiguous(),
            self.gemm.bias.contiguous(),
            self.scale.contiguous(),
            self.bn.weight.contiguous(),
            self.bn.bias.contiguous(),
            float(self.eps)
        )
