kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t TILE_K = 2048;

class KernelMatmulSigmoidSum {
public:
    __aicore__ inline KernelMatmulSigmoidSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->blockIdx = AscendC::GetBlockIdx();
        this->blockNum = AscendC::GetBlockNum();

        // distribute hiddenSize across cores
        this->hPerCore = (hiddenSize + this->blockNum - 1) / this->blockNum;
        this->hStart = this->blockIdx * this->hPerCore;
        this->hEnd = (this->hStart + this->hPerCore) > hiddenSize ? hiddenSize : (this->hStart + this->hPerCore);
        if (this->hStart > hiddenSize) this->hStart = hiddenSize;
        this->hLen = this->hEnd - this->hStart;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inputSize);
        wGm.SetGlobalBuffer((__gm__ float *)weight, hiddenSize * inputSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, hiddenSize);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize);

        pipe.InitBuffer(inQueueX, 1, TILE_K * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, TILE_K * sizeof(float));
        pipe.InitBuffer(tmpBuf, TILE_K * sizeof(float));
        pipe.InitBuffer(accBuf, 32 * sizeof(float));  // single scalar (aligned)
        pipe.InitBuffer(outQueueZ, 1, 32 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->hLen == 0) return;
        for (uint32_t b = 0; b < this->batchSize; b++) {
            float partialSum = 0.0f;
            for (uint32_t h = this->hStart; h < this->hEnd; h++) {
                float dot = 0.0f;
                uint32_t k = 0;
                while (k < this->inputSize) {
                    uint32_t curLen = (this->inputSize - k) > TILE_K ? TILE_K : (this->inputSize - k);
                    AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                    AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                    AscendC::DataCopy(xLocal, xGm[b * this->inputSize + k], curLen);
                    AscendC::DataCopy(wLocal, wGm[h * this->inputSize + k], curLen);
                    inQueueX.EnQue(xLocal);
                    inQueueW.EnQue(wLocal);

                    AscendC::LocalTensor<float> xL = inQueueX.DeQue<float>();
                    AscendC::LocalTensor<float> wL = inQueueW.DeQue<float>();
                    AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
                    AscendC::Mul(tmp, xL, wL, curLen);
                    AscendC::LocalTensor<float> acc = accBuf.Get<float>();
                    AscendC::ReduceSum<float>(acc, tmp, tmp, curLen);
                    float partial = acc.GetValue(0);
                    dot += partial;
                    inQueueX.FreeTensor(xL);
                    inQueueW.FreeTensor(wL);
                    k += curLen;
                }
                float biasVal = bGm.GetValue(h);
                float v = dot + biasVal;
                // sigmoid
                float s = 1.0f / (1.0f + AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(expf(-v)));
                partialSum += s;
            }
            // atomic add to zGm[b]
            AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
            zLocal.SetValue(0, partialSum);
            outQueueZ.EnQue<float>(zLocal);
            AscendC::LocalTensor<float> zOut = outQueueZ.DeQue<float>();
            AscendC::SetAtomicAdd<float>();
            AscendC::DataCopy(zGm[b], zOut, 8);
            AscendC::SetAtomicNone();
            outQueueZ.FreeTensor(zOut);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, accBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, zGm;
    uint32_t batchSize, inputSize, hiddenSize;
    uint32_t blockIdx, blockNum;
    uint32_t hPerCore, hStart, hEnd, hLen;
};

extern "C" __global__ __aicore__ void matmul_sigmoid_sum_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSigmoidSum op;
    op.Init(x, weight, bias, z, tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize);
    op.Process();
}
