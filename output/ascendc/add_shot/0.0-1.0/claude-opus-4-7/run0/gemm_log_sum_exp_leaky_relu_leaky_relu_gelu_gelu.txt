#include "kernel_operator.h"

class KernelGemmFused {
public:
    __aicore__ inline KernelGemmFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->startRow = blockIdx * rowsPerBlock;
        this->endRow = (this->startRow + rowsPerBlock < batchSize) ? (this->startRow + rowsPerBlock) : batchSize;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outFeatures * inFeatures);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize);

        pipe.InitBuffer(xBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(wBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(bBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(outBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf, inFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (startRow >= endRow) return;

        AscendC::LocalTensor<float> bLocal = bBuf.Get<float>();
        AscendC::DataCopy(bLocal, bGm, outFeatures);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t row = startRow; row < endRow; row++) {
            AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
            AscendC::DataCopy(xLocal, xGm[row * inFeatures], inFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();

            AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
            AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

            // Compute GEMM: out[j] = sum_k(x[k] * w[j,k]) + b[j]
            for (uint32_t j = 0; j < outFeatures; j++) {
                AscendC::DataCopy(wLocal, wGm[j * inFeatures], inFeatures);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::Mul(tmp, xLocal, wLocal, inFeatures);
                AscendC::PipeBarrier<PIPE_ALL>();
                float sum = 0.0f;
                for (uint32_t k = 0; k < inFeatures; k++) {
                    sum += tmp.GetValue(k);
                }
                outLocal.SetValue(j, sum + bLocal.GetValue(j));
            }
            AscendC::PipeBarrier<PIPE_ALL>();

            // LogSumExp over outFeatures
            float maxVal = outLocal.GetValue(0);
            for (uint32_t j = 1; j < outFeatures; j++) {
                float v = outLocal.GetValue(j);
                if (v > maxVal) maxVal = v;
            }
            float sumExp = 0.0f;
            for (uint32_t j = 0; j < outFeatures; j++) {
                sumExp += expf(outLocal.GetValue(j) - maxVal);
            }
            float result = maxVal + logf(sumExp);

            // LeakyReLU x2
            if (result < 0) result *= 0.01f;
            if (result < 0) result *= 0.01f;

            // GELU x2 (using tanh approximation)
            for (int g = 0; g < 2; g++) {
                float c = 0.7978845608f * (result + 0.044715f * result * result * result);
                float t = tanhf(c);
                result = 0.5f * result * (1.0f + t);
            }

            zGm.SetValue(row, result);
            AscendC::PipeBarrier<PIPE_ALL>();
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, wBuf, bBuf, outBuf, tmpBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, zGm;
    uint32_t batchSize, inFeatures, outFeatures;
    uint32_t startRow, endRow;
};

extern "C" __global__ __aicore__ void gemm_log_sum_exp_leaky_relu_leaky_relu_gelu_gelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmFused op;
    op.Init(x, weight, bias, z, tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures);
    op.Process();
}
