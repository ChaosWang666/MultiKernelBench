kernel_src="""
#include "kernel_operator.h"

class KernelMMSMR {
public:
    __aicore__ inline KernelMMSMR() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                uint32_t M, uint32_t N, uint32_t K, float subVal, float mulVal)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        this->subVal = subVal;
        this->mulVal = mulVal;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (M + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        this->endRow = this->startRow + this->rowsPerBlock;
        if (this->endRow > M) this->endRow = M;

        xGm.SetGlobalBuffer((__gm__ float*)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float*)weight, N * K);
        bGm.SetGlobalBuffer((__gm__ float*)bias, N);
        yGm.SetGlobalBuffer((__gm__ float*)y, M * N);

        pipe.InitBuffer(inQueueX, 1, K * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, K * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, N * sizeof(float));
        pipe.InitBuffer(tmpBuf, K * sizeof(float));
        pipe.InitBuffer(biasBuf, N * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->startRow >= this->endRow) return;
        // Load bias once
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::DataCopy(biasLocal, bGm, this->N);

        for (uint32_t row = this->startRow; row < this->endRow; row++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[row * this->K], this->K);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

            for (uint32_t n = 0; n < this->N; n++) {
                AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                AscendC::DataCopy(wLocal, wGm[n * this->K], this->K);
                inQueueW.EnQue(wLocal);
                wLocal = inQueueW.DeQue<float>();

                AscendC::Mul(tmp, xLocal, wLocal, this->K);
                float sum = 0.0f;
                // reduce sum via ReduceSum
                AscendC::ReduceSum<float>(tmp, tmp, tmp, this->K);
                float val = tmp.GetValue(0) + biasLocal.GetValue(n);
                val = (val - this->subVal) * this->mulVal;
                if (val < 0.0f) val = 0.0f;
                yLocal.SetValue(n, val);

                inQueueW.FreeTensor(wLocal);
            }

            outQueueY.EnQue<float>(yLocal);
            yLocal = outQueueY.DeQue<float>();
            AscendC::DataCopy(yGm[row * this->N], yLocal, this->N);
            outQueueY.FreeTensor(yLocal);
            inQueueX.FreeTensor(xLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, biasBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, yGm;
    uint32_t M, N, K;
    uint32_t startRow, endRow, rowsPerBlock;
    float subVal, mulVal;
};

extern "C" __global__ __aicore__ void matmul_subtract_multiply_relu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMMSMR op;
    op.Init(x, weight, bias, y, tiling_data.M, tiling_data.N, tiling_data.K,
            tiling_data.subVal, tiling_data.mulVal);
    op.Process();
}
