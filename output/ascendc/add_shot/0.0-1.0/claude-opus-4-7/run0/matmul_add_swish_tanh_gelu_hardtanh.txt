#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR add_value, GM_ADDR z, uint32_t totalLength, uint32_t outFeatures, uint32_t tileNum)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->outFeatures = outFeatures;
        this->tileLength = this->blockLength / tileNum / BUFFER_NUM;
        this->blockOffset = this->blockLength * AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->blockOffset, this->blockLength);
        addGm.SetGlobalBuffer((__gm__ DTYPE_X *)add_value, outFeatures);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + this->blockOffset, this->blockLength);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueA, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, this->tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> aLocal = inQueueA.AllocTensor<DTYPE_X>();
        uint32_t offset = progress * this->tileLength;
        AscendC::DataCopy(xLocal, xGm[offset], this->tileLength);
        uint32_t globalOffset = (this->blockOffset + offset) % this->outFeatures;
        AscendC::DataCopy(aLocal, addGm[globalOffset], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueA.EnQue(aLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> aLocal = inQueueA.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> t1 = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> t2 = tmpBuf2.Get<float>();

        // x + add
        AscendC::Add(xLocal, xLocal, aLocal, this->tileLength);
        // swish: sigmoid(x)*x
        AscendC::Muls(t1, xLocal, (float)-1.0, this->tileLength);
        AscendC::Exp(t1, t1, this->tileLength);
        AscendC::Adds(t1, t1, (float)1.0, this->tileLength);
        AscendC::Div(t2, xLocal, t1, this->tileLength); // sigmoid(x)*x = x/(1+exp(-x))
        // tanh
        AscendC::Tanh(t2, t2, this->tileLength);
        // gelu: 0.5*x*(1+erf(x/sqrt(2)))
        AscendC::Gelu(t2, t2, this->tileLength);
        // hardtanh: clamp [-1,1]
        AscendC::Mins(t2, t2, (float)1.0, this->tileLength);
        AscendC::Maxs(zLocal, t2, (float)-1.0, this->tileLength);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueA.FreeTensor(aLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * this->tileLength], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueA;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, tmpBuf2;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_X> addGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t blockLength;
    uint32_t blockOffset;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t outFeatures;
};

extern "C" __global__ __aicore__ void matmul_add_swish_tanh_gelu_hardtanh_custom(GM_ADDR x, GM_ADDR add_value, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, add_value, z, tiling_data.totalLength, tiling_data.outFeatures, tiling_data.tileNum);
    op.Process();
}
