kernel_src="""
#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR lb, GM_ADDR gnw, GM_ADDR gnb, GM_ADDR bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures, uint32_t numGroups)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->groupSize = outFeatures / numGroups;
        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float*)w, outFeatures * inFeatures);
        lbGm.SetGlobalBuffer((__gm__ float*)lb, outFeatures);
        gnwGm.SetGlobalBuffer((__gm__ float*)gnw, outFeatures);
        gnbGm.SetGlobalBuffer((__gm__ float*)gnb, outFeatures);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z, batchSize * outFeatures);
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        for (uint32_t b = blockIdx; b < batchSize; b += blockNum) {
            // compute linear output on the fly and find min across all outFeatures with group norm
            float minVal = 1e30f;
            for (uint32_t g = 0; g < numGroups; g++) {
                // compute group gemm outputs
                float sum = 0.f, sqsum = 0.f;
                // compute each element in group
                for (uint32_t j = 0; j < groupSize; j++) {
                    uint32_t outIdx = g * groupSize + j;
                    float acc = lbGm.GetValue(outIdx);
                    for (uint32_t k = 0; k < inFeatures; k++) {
                        acc += xGm.GetValue(b * inFeatures + k) * wGm.GetValue(outIdx * inFeatures + k);
                    }
                    // store temporarily in zGm
                    zGm.SetValue(b * outFeatures + outIdx, acc);
                    sum += acc;
                    sqsum += acc * acc;
                }
                float mean = sum / (float)groupSize;
                float var = sqsum / (float)groupSize - mean * mean;
                float invStd = 1.0f / AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(sqrt(var + 1e-5f));
                for (uint32_t j = 0; j < groupSize; j++) {
                    uint32_t outIdx = g * groupSize + j;
                    float val = zGm.GetValue(b * outFeatures + outIdx);
                    val = (val - mean) * invStd * gnwGm.GetValue(outIdx) + gnbGm.GetValue(outIdx);
                    if (val < minVal) minVal = val;
                }
            }
            // now broadcast minVal + bias to zGm[b, :]
            for (uint32_t i = 0; i < outFeatures; i++) {
                zGm.SetValue(b * outFeatures + i, minVal + biasGm.GetValue(i));
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm, wGm, lbGm, gnwGm, gnbGm, biasGm, zGm;
    uint32_t batchSize, inFeatures, outFeatures, numGroups, groupSize;
};

extern "C" __global__ __aicore__ void gemm_group_norm_min_bias_add_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR linear_bias, GM_ADDR gn_weight, GM_ADDR gn_bias, GM_ADDR bias,
    GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, weight, linear_bias, gn_weight, gn_bias, bias, z,
            tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures, tiling_data.numGroups);
    op.Process();
}
