kernel_src="""
#include "kernel_operator.h"

extern "C" __global__ __aicore__ void conv2d_scaling_min_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    uint32_t channels = tiling_data.channels;
    uint32_t spatial = tiling_data.spatial;
    float scale = tiling_data.scale;

    uint32_t blockNum = AscendC::GetBlockNum();
    uint32_t blockIdx = AscendC::GetBlockIdx();
    uint32_t perBlock = (spatial + blockNum - 1) / blockNum;
    uint32_t startIdx = blockIdx * perBlock;
    uint32_t endIdx = startIdx + perBlock;
    if (endIdx > spatial) endIdx = spatial;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    xGm.SetGlobalBuffer((__gm__ float*)x, (uint64_t)spatial * channels);
    zGm.SetGlobalBuffer((__gm__ float*)z, spatial);

    AscendC::TPipe pipe;
    constexpr uint32_t TILE = 64;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQ;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQ;
    pipe.InitBuffer(inQ, 2, TILE * sizeof(float));
    pipe.InitBuffer(outQ, 2, TILE * sizeof(float));

    for (uint32_t s = startIdx; s < endIdx; s += TILE) {
        uint32_t curLen = (s + TILE <= endIdx) ? TILE : (endIdx - s);
        // Need contiguous copy, but data is [N, C, H*W] with s indexing spatial.
        // Compute min per spatial point across channels, one-at-a-time.
        AscendC::LocalTensor<float> outLocal = outQ.AllocTensor<float>();
        for (uint32_t i = 0; i < curLen; i++) {
            uint32_t sp = s + i;
            uint32_t n = sp / (spatial / (spatial)); // placeholder
            // Actually layout is [N,C,H,W] flattened. spatial iterates over N*H*W.
            // idx in flat = (n * C + c) * (H*W) + hw, where sp = n*(H*W) + hw
            // We need HW. But we don't have it directly; spatial = N*H*W.
            // Trick: for each sp, we need to know n and hw. Store HW as spatial/N? we don't have N.
            // Simpler: iterate channel with stride. Let HW_stride = unknown.
            // Fall back: treat layout such that channel stride = spatial (wrong for N>1).
            // For this kernel, we'll compute assuming stride pattern using tiling.
            float minVal = xGm.GetValue(sp) * scale;
            for (uint32_t c = 1; c < channels; c++) {
                float v = xGm.GetValue(sp + c * spatial) * scale;
                if (v < minVal) minVal = v;
            }
            outLocal.SetValue(i, minVal);
        }
        outQ.EnQue(outLocal);
        auto o = outQ.DeQue<float>();
        AscendC::DataCopy(zGm[s], o, curLen);
        outQ.FreeTensor(o);
    }
}
"""

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor conv2d_scaling_min_custom_impl_npu(const at::Tensor& self) {
    auto sizes = self.sizes().vec();
    sizes[1] = 1;
    at::Tensor result = at::empty(sizes, self.options());
    // Reshape view: treat input as [N, C, H*W] but pass as-is; kernel uses tiling dims.
    // We need to pass a tensor where spatial = N*H*W and channel stride = spatial.
    // Permute to [C, N, H, W] so channel is outermost
    at::Tensor permuted = self.permute({1, 0, 2, 3}).contiguous();
    EXEC_NPU_CMD(aclnnConv2dScalingMinCustom, permuted, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("conv2d_scaling_min_custom", &conv2d_scaling_min_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("conv2d_scaling_min_custom", &conv2d_scaling_min_custom_impl_npu, "conv2d scaling min");
}
