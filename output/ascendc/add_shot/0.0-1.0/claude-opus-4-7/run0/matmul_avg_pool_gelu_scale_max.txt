kernel_src="""
#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures,
                                 uint32_t poolKernelSize, uint32_t pooledSize, float scaleFactor)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->poolKernelSize = poolKernelSize;
        this->pooledSize = pooledSize;
        this->scaleFactor = scaleFactor;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        this->endRow = this->startRow + this->rowsPerBlock;
        if (this->endRow > batchSize) this->endRow = batchSize;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float*)weight, outFeatures * inFeatures);
        bGm.SetGlobalBuffer((__gm__ float*)bias, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z, batchSize);

        pipe.InitBuffer(xBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(wBuf, inFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->startRow >= this->endRow) return;

        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();

        const float c0 = 0.7978845608f; // sqrt(2/pi)
        const float c1 = 0.044715f;

        for (uint32_t row = this->startRow; row < this->endRow; row++) {
            AscendC::DataCopy(xLocal, xGm[row * inFeatures], inFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();

            float maxVal = -3.4e38f;

            for (uint32_t p = 0; p < pooledSize; p++) {
                // Compute avg over poolKernelSize matmul outputs
                float poolSum = 0.0f;
                for (uint32_t k = 0; k < poolKernelSize; k++) {
                    uint32_t outIdx = p * poolKernelSize + k;
                    AscendC::DataCopy(wLocal, wGm[outIdx * inFeatures], inFeatures);
                    AscendC::PipeBarrier<PIPE_ALL>();

                    // dot product
                    float dot = 0.0f;
                    for (uint32_t i = 0; i < inFeatures; i++) {
                        dot += xLocal.GetValue(i) * wLocal.GetValue(i);
                    }
                    dot += bGm.GetValue(outIdx);
                    poolSum += dot;
                }
                float avg = poolSum / (float)poolKernelSize;
                // GELU tanh approx
                float inner = c0 * (avg + c1 * avg * avg * avg);
                float e1 = AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(1.0f);
                // compute tanh via exp
                float ex = expf(inner);
                float emx = expf(-inner);
                float tanhVal = (ex - emx) / (ex + emx);
                float gelu = 0.5f * avg * (1.0f + tanhVal);
                float scaled = gelu * scaleFactor;
                if (scaled > maxVal) maxVal = scaled;
            }

            zGm.SetValue(row, maxVal);
            AscendC::PipeBarrier<PIPE_ALL>();
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, wBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, zGm;
    uint32_t batchSize, inFeatures, outFeatures, poolKernelSize, pooledSize;
    uint32_t startRow, endRow, rowsPerBlock;
    float scaleFactor;
};

extern "C" __global__ __aicore__ void matmul_avg_pool_gelu_scale_max_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, weight, bias, z,
            tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures,
            tiling_data.poolKernelSize, tiling_data.pooledSize, tiling_data.scaleFactor);
    op.Process();
}
