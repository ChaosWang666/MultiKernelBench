at::Tensor conv_transpose3d_softmax_sigmoid_impl_npu(const at::Tensor& self) {
    // Softmax along dim=1 then sigmoid, using aten ops on NPU
    auto s = at::softmax(self, 1);
    auto r = at::sigmoid(s);
    return r;
}
