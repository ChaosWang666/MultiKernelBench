kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemm {
public:
    __aicore__ inline KernelGemm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t M, uint32_t N, uint32_t K,
                                 float multiplier, float negative_slope)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        this->multiplier = multiplier;
        this->negative_slope = negative_slope;
        
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        
        // Each block processes a chunk of rows
        this->rowsPerBlock = (M + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        this->endRow = (this->startRow + this->rowsPerBlock < M) ? (this->startRow + this->rowsPerBlock) : M;
        if (this->startRow >= M) {
            this->actualRows = 0;
        } else {
            this->actualRows = this->endRow - this->startRow;
        }
        
        xGm.SetGlobalBuffer((__gm__ float*)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float*)weight, N * K);
        bGm.SetGlobalBuffer((__gm__ float*)bias, N);
        yGm.SetGlobalBuffer((__gm__ float*)y, M * N);
        
        pipe.InitBuffer(inQueueX, 1, K * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, K * sizeof(float));
        pipe.InitBuffer(inQueueB, 1, N * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, N * sizeof(float));
        pipe.InitBuffer(tmpBuf, K * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        if (this->actualRows == 0) return;
        
        // Load bias once
        AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
        AscendC::DataCopy(bLocal, bGm, this->N);
        inQueueB.EnQue(bLocal);
        bLocal = inQueueB.DeQue<float>();
        
        for (uint32_t row = 0; row < this->actualRows; row++) {
            uint32_t globalRow = this->startRow + row;
            
            // Load x row
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[globalRow * this->K], this->K);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();
            
            AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
            
            // For each output column n: y[n] = sum_k x[k] * w[n,k] + b[n]
            for (uint32_t n = 0; n < this->N; n++) {
                AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                AscendC::DataCopy(wLocal, wGm[n * this->K], this->K);
                inQueueW.EnQue(wLocal);
                wLocal = inQueueW.DeQue<float>();
                
                AscendC::Mul(tmp, xLocal, wLocal, this->K);
                inQueueW.FreeTensor(wLocal);
                
                // Reduce sum
                float sum = 0.0f;
                AscendC::ReduceSum<float>(tmp, tmp, tmp, this->K);
                AscendC::PipeBarrier<PIPE_V>();
                
                AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                sum = tmp.GetValue(0);
                yLocal.SetValue(n, sum);
            }
            
            inQueueX.FreeTensor(xLocal);
            
            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            
            // Add bias
            AscendC::Add(yLocal, yLocal, bLocal, this->N);
            AscendC::PipeBarrier<PIPE_V>();
            // Multiply
            AscendC::Muls(yLocal, yLocal, this->multiplier, this->N);
            AscendC::PipeBarrier<PIPE_V>();
            // LeakyReLU
            AscendC::LeakyRelu(yLocal, yLocal, (float)this->negative_slope, this->N);
            AscendC::PipeBarrier<PIPE_V>();
            
            outQueueY.EnQue<float>(yLocal);
            yLocal = outQueueY.DeQue<float>();
            AscendC::DataCopy(yGm[globalRow * this->N], yLocal, this->N);
            outQueueY.FreeTensor(yLocal);
        }
        
        inQueueB.FreeTensor(bLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, yGm;
    uint32_t M, N, K;
    uint32_t rowsPerBlock, startRow, endRow, actualRows;
    float multiplier;
    float negative_slope;
};

extern "C" __global__ __aicore__ void gemm_multiply_leakyrelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemm op;
    op.Init(x, weight, bias, y, tiling_data.M, tiling_data.N, tiling_data.K,
            tiling_data.multiplier, tiling_data.negative_slope);
    op.Process();
}
