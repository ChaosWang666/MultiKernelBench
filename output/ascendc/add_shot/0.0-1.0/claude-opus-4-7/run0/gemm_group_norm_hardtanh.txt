kernel_src="""
#include "kernel_operator.h"

class KernelGemmGNHardtanh {
public:
    __aicore__ inline KernelGemmGNHardtanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
                                GM_ADDR gnWeight, GM_ADDR gnBias, GM_ADDR z,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures,
                                uint32_t numGroups, float hMin, float hMax)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->groupSize = outFeatures / numGroups;
        this->hMin = hMin;
        this->hMax = hMax;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->rowStart = blockIdx * this->rowsPerBlock;
        this->rowEnd = this->rowStart + this->rowsPerBlock;
        if (this->rowEnd > batchSize) this->rowEnd = batchSize;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float*)weight, outFeatures * inFeatures);
        bGm.SetGlobalBuffer((__gm__ float*)bias, outFeatures);
        gnwGm.SetGlobalBuffer((__gm__ float*)gnWeight, outFeatures);
        gnbGm.SetGlobalBuffer((__gm__ float*)gnBias, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z, batchSize * outFeatures);

        pipe.InitBuffer(xBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(yBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(wBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(paramBuf, outFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->rowStart >= this->rowEnd) return;

        AscendC::LocalTensor<float> biasLocal = paramBuf.Get<float>();
        AscendC::DataCopy(biasLocal, bGm, this->outFeatures);

        for (uint32_t row = this->rowStart; row < this->rowEnd; row++) {
            AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
            AscendC::LocalTensor<float> yLocal = yBuf.Get<float>();
            AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();

            AscendC::DataCopy(xLocal, xGm[row * this->inFeatures], this->inFeatures);
            AscendC::DataCopy(yLocal, biasLocal, this->outFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();

            // GEMM: y[o] = bias[o] + sum_i x[i] * w[o,i]
            for (uint32_t o = 0; o < this->outFeatures; o++) {
                AscendC::DataCopy(wLocal, wGm[o * this->inFeatures], this->inFeatures);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::Mul(wLocal, wLocal, xLocal, this->inFeatures);
                AscendC::PipeBarrier<PIPE_ALL>();
                float sum = 0.0f;
                for (uint32_t i = 0; i < this->inFeatures; i++) {
                    sum += wLocal.GetValue(i);
                }
                yLocal.SetValue(o, yLocal.GetValue(o) + sum);
            }
            AscendC::PipeBarrier<PIPE_ALL>();

            // GroupNorm
            for (uint32_t g = 0; g < this->numGroups; g++) {
                float mean = 0.0f;
                for (uint32_t i = 0; i < this->groupSize; i++) {
                    mean += yLocal.GetValue(g * this->groupSize + i);
                }
                mean /= (float)this->groupSize;
                float var = 0.0f;
                for (uint32_t i = 0; i < this->groupSize; i++) {
                    float d = yLocal.GetValue(g * this->groupSize + i) - mean;
                    var += d * d;
                }
                var /= (float)this->groupSize;
                float invstd = 1.0f / sqrt(var + 1e-5f);
                for (uint32_t i = 0; i < this->groupSize; i++) {
                    uint32_t idx = g * this->groupSize + i;
                    float v = (yLocal.GetValue(idx) - mean) * invstd;
                    yLocal.SetValue(idx, v);
                }
            }

            // affine + hardtanh
            AscendC::LocalTensor<float> gnwLocal = wBuf.Get<float>();
            AscendC::DataCopy(gnwLocal, gnwGm, this->outFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Mul(yLocal, yLocal, gnwLocal, this->outFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::DataCopy(gnwLocal, gnbGm, this->outFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Add(yLocal, yLocal, gnwLocal, this->outFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Mins(yLocal, yLocal, this->hMax, this->outFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Maxs(yLocal, yLocal, this->hMin, this->outFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();

            AscendC::DataCopy(zGm[row * this->outFeatures], yLocal, this->outFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();

            // reload bias for next row
            AscendC::DataCopy(biasLocal, bGm, this->outFeatures);
            AscendC::PipeBarrier<PIPE_ALL>();
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, yBuf, wBuf, paramBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, gnwGm, gnbGm, zGm;
    uint32_t batchSize, inFeatures, outFeatures, numGroups, groupSize;
    uint32_t rowsPerBlock, rowStart, rowEnd;
    float hMin, hMax;
};

extern "C" __global__ __aicore__ void gemm_group_norm_hardtanh_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR gn_weight, GM_ADDR gn_bias,
    GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGNHardtanh op;
    op.Init(x, weight, bias, gn_weight, gn_bias, z,
            tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures,
            tiling_data.numGroups, tiling_data.hardtanhMin, tiling_data.hardtanhMax);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib
import torch.nn as nn

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, hardtanh_min, hardtanh_max):
        super(ModelNew, self).__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.num_groups = num_groups
        self.hardtanh_min = float(hardtanh_min)
        self.hardtanh_max = float(hardtanh_max)
        self.gemm = nn.Linear(in_features, out_features)
        self.group_norm = nn.GroupNorm(num_groups, out_features)

    def forward(self, x):
        batch_size = x.shape[0]
        return custom_ops_lib.gemm_group_norm_hardtanh_custom(
            x.contiguous(), self.gemm.weight.contiguous(), self.gemm.bias.contiguous(),
            self.group_norm.weight.contiguous(), self.group_norm.bias.contiguous(),
            batch_size, self.in_features, self.out_features,
            self.num_groups, self.hardtanh_min, self.hardtanh_max)
