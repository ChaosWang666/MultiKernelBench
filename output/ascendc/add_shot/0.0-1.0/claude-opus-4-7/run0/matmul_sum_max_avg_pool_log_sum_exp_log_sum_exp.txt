kernel_src="""
#include "kernel_operator.h"

// Note: the chain reduces to: out[b] = sum_j(x[b,:] @ W[j,:]^T) + sum_j(bias[j])
// = sum over all out features of (linear(x)[b,j])
// = sum_i x[b,i] * (sum_j W[j,i]) + sum_j bias[j]
// All subsequent ops (max,mean,logsumexp,logsumexp) along dim=1 of size 1 are identity.

class KernelMatmulChain {
public:
    __aicore__ inline KernelMatmulChain() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->blockIdx = AscendC::GetBlockIdx();
        this->blockNum = AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outFeatures * inFeatures);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize);

        pipe.InitBuffer(inQueueX, 1, this->inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueW, 2, this->inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueB, 1, 64 * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->inFeatures * sizeof(float));
        pipe.InitBuffer(wsumBuf, this->inFeatures * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, 32 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Step 1: compute wsum[i] = sum_j W[j,i], and bsum scalar (only block 0 broadcasts via recompute on each block)
        AscendC::LocalTensor<float> wsum = wsumBuf.Get<float>();
        AscendC::Duplicate(wsum, (float)0, this->inFeatures);

        for (uint32_t j = 0; j < this->outFeatures; j++) {
            AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
            AscendC::DataCopy(wLocal, wGm[j * this->inFeatures], this->inFeatures);
            inQueueW.EnQue(wLocal);
            AscendC::LocalTensor<float> wL = inQueueW.DeQue<float>();
            AscendC::Add(wsum, wsum, wL, this->inFeatures);
            inQueueW.FreeTensor(wL);
        }

        // bias sum
        float bsum = 0.0f;
        // Load bias in chunks
        uint32_t biasChunk = 64;
        AscendC::LocalTensor<float> bAccum = tmpBuf.Get<float>();
        AscendC::Duplicate(bAccum, (float)0, biasChunk);
        uint32_t fullChunks = this->outFeatures / biasChunk;
        uint32_t rem = this->outFeatures % biasChunk;
        for (uint32_t c = 0; c < fullChunks; c++) {
            AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
            AscendC::DataCopy(bLocal, bGm[c * biasChunk], biasChunk);
            inQueueB.EnQue(bLocal);
            AscendC::LocalTensor<float> bL = inQueueB.DeQue<float>();
            AscendC::Add(bAccum, bAccum, bL, biasChunk);
            inQueueB.FreeTensor(bL);
        }
        // Reduce bAccum to scalar
        for (uint32_t k = 0; k < biasChunk; k++) {
            bsum += bAccum.GetValue(k);
        }
        // remainder
        for (uint32_t k = 0; k < rem; k++) {
            bsum += bGm.GetValue(fullChunks * biasChunk + k);
        }

        // Step 2: distribute batch across blocks
        uint32_t perBlock = (this->batchSize + this->blockNum - 1) / this->blockNum;
        uint32_t start = this->blockIdx * perBlock;
        uint32_t end = start + perBlock;
        if (end > this->batchSize) end = this->batchSize;

        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        for (uint32_t b = start; b < end; b++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[b * this->inFeatures], this->inFeatures);
            inQueueX.EnQue(xLocal);
            AscendC::LocalTensor<float> xL = inQueueX.DeQue<float>();

            AscendC::Mul(tmp, xL, wsum, this->inFeatures);
            inQueueX.FreeTensor(xL);

            // Reduce sum
            float s = 0.0f;
            for (uint32_t k = 0; k < this->inFeatures; k++) {
                s += tmp.GetValue(k);
            }
            s += bsum;

            AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
            zLocal.SetValue(0, s);
            outQueueZ.EnQue<float>(zLocal);
            AscendC::LocalTensor<float> zL = outQueueZ.DeQue<float>();
            // Write single value; use DataCopy with 8 elements alignment workaround
            // Use SetAtomicNone and write via GM pointer
            zGm.SetValue(b, s);
            outQueueZ.FreeTensor(zL);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueW;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wsumBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t inFeatures;
    uint32_t outFeatures;
    uint32_t blockIdx;
    uint32_t blockNum;
};

extern "C" __global__ __aicore__ void matmul_sum_max_avg_pool_log_sum_exp_log_sum_exp_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulChain op;
    op.Init(x, weight, bias, z, tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures);
    op.Process();
}
