kernel_src="""
#include "kernel_operator.h"

class KernelMSRACLM {
public:
    __aicore__ inline KernelMSRACLM() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize,
                                float scaleFactor, float clampMin, float clampMax)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        // residual add x+x => 2x, combine with scale: effective = 2*scale
        this->effScale = 2.0f * scaleFactor;
        this->clampMin = clampMin;
        this->clampMax = clampMax;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        this->endRow = this->startRow + this->rowsPerBlock;
        if (this->endRow > batchSize) this->endRow = batchSize;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inputSize);
        wGm.SetGlobalBuffer((__gm__ float *)weight, hiddenSize * inputSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, hiddenSize);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize);

        pipe.InitBuffer(xBuf, inputSize * sizeof(float));
        pipe.InitBuffer(wBuf, inputSize * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();

        for (uint32_t row = startRow; row < endRow; row++) {
            AscendC::DataCopy(xLocal, xGm[row * inputSize], inputSize);
            AscendC::PipeBarrier<PIPE_ALL>();

            float maxVal = -1e30f;
            // First pass: compute all logits, find max
            // We need to store logits; but hiddenSize may be large (8192). Use a temp buffer.
            // Instead, do two passes: pass1 find max, pass2 compute sum exp. Recompute logits each pass.
            // Pass 1
            for (uint32_t j = 0; j < hiddenSize; j++) {
                AscendC::DataCopy(wLocal, wGm[j * inputSize], inputSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                // dot product
                float dot = 0.0f;
                for (uint32_t k = 0; k < inputSize; k++) {
                    dot += xLocal.GetValue(k) * wLocal.GetValue(k);
                }
                float bias_val = bGm.GetValue(j);
                float logit = (dot + bias_val) * effScale;
                if (logit < clampMin) logit = clampMin;
                if (logit > clampMax) logit = clampMax;
                if (logit > maxVal) maxVal = logit;
            }

            // Pass 2: sum exp
            float sumExp = 0.0f;
            for (uint32_t j = 0; j < hiddenSize; j++) {
                AscendC::DataCopy(wLocal, wGm[j * inputSize], inputSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                float dot = 0.0f;
                for (uint32_t k = 0; k < inputSize; k++) {
                    dot += xLocal.GetValue(k) * wLocal.GetValue(k);
                }
                float bias_val = bGm.GetValue(j);
                float logit = (dot + bias_val) * effScale;
                if (logit < clampMin) logit = clampMin;
                if (logit > clampMax) logit = clampMax;
                sumExp += expf(logit - maxVal);
            }
            float lse = maxVal + logf(sumExp);
            // Mish: x * mish(x) = x * x * tanh(softplus(x))
            float sp = logf(1.0f + expf(lse));
            float t = tanhf(sp);
            float out = lse * lse * t;
            zGm.SetValue(row, out);
            AscendC::PipeBarrier<PIPE_ALL>();
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, wBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, zGm;
    uint32_t batchSize, inputSize, hiddenSize;
    uint32_t startRow, endRow, rowsPerBlock;
    float effScale, clampMin, clampMax;
};

extern "C" __global__ __aicore__ void matmul_scale_residual_add_clamp_log_sum_exp_mish_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMSRACLM op;
    op.Init(x, weight, bias, z,
            tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize,
            tiling_data.scaleFactor, tiling_data.clampMin, tiling_data.clampMax);
    op.Process();
}
