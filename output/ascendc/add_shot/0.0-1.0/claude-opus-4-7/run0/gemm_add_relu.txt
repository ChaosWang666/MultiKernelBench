kernel_src="""
#include "kernel_operator.h"

class KernelGemmAddRelu {
public:
    __aicore__ inline KernelGemmAddRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR bias, GM_ADDR y,
                                 uint32_t M, uint32_t N, uint32_t K)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (M + blockNum - 1) / blockNum;
        this->rowStart = blockIdx * this->rowsPerBlock;
        this->rowEnd = this->rowStart + this->rowsPerBlock;
        if (this->rowEnd > M) this->rowEnd = M;

        xGm.SetGlobalBuffer((__gm__ float *)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float *)w, N * K);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, N);
        yGm.SetGlobalBuffer((__gm__ float *)y, M * N);

        pipe.InitBuffer(inQueueX, 1, K * sizeof(float));
        pipe.InitBuffer(inQueueW, 2, K * sizeof(float));
        pipe.InitBuffer(inQueueB, 1, N * sizeof(float));
        pipe.InitBuffer(outQueueY, 2, N * sizeof(float));
        pipe.InitBuffer(tmpBuf, K * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->rowStart >= this->rowEnd) return;
        // Load bias once
        AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
        AscendC::DataCopy(bLocal, biasGm, this->N);
        inQueueB.EnQue(bLocal);
        bLocal = inQueueB.DeQue<float>();

        for (uint32_t row = this->rowStart; row < this->rowEnd; row++) {
            // Load x row
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[row * this->K], this->K);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

            for (uint32_t n = 0; n < this->N; n++) {
                AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                AscendC::DataCopy(wLocal, wGm[n * this->K], this->K);
                inQueueW.EnQue(wLocal);
                wLocal = inQueueW.DeQue<float>();

                AscendC::Mul(tmp, xLocal, wLocal, this->K);
                inQueueW.FreeTensor(wLocal);

                float sum = 0.0f;
                AscendC::LocalTensor<float> sumTensor = yLocal[n];
                // reduce sum
                AscendC::ReduceSum<float>(sumTensor, tmp, tmp, this->K);
            }

            AscendC::Add(yLocal, yLocal, bLocal, this->N);
            AscendC::Relu(yLocal, yLocal, this->N);

            outQueueY.EnQue<float>(yLocal);
            inQueueX.FreeTensor(xLocal);

            AscendC::LocalTensor<float> yOut = outQueueY.DeQue<float>();
            AscendC::DataCopy(yGm[row * this->N], yOut, this->N);
            outQueueY.FreeTensor(yOut);
        }
        inQueueB.FreeTensor(bLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueW;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, 2> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm, wGm, biasGm, yGm;
    uint32_t M, N, K;
    uint32_t rowsPerBlock, rowStart, rowEnd;
};

extern "C" __global__ __aicore__ void gemm_add_relu_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmAddRelu op;
    op.Init(x, weight, bias, y, tiling_data.M, tiling_data.N, tiling_data.K);
    op.Process();
}
