kernel_src="""
#include "kernel_operator.h"

class KernelLNGelu {
public:
    __aicore__ inline KernelLNGelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z,
                                uint32_t totalRows, uint32_t channels, float eps, float scalingFactor)
    {
        this->channels = channels;
        this->eps = eps;
        this->scalingFactor = scalingFactor;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t rowsPerBlock = (totalRows + blockNum - 1) / blockNum;
        uint32_t startRow = blockIdx * rowsPerBlock;
        uint32_t endRow = startRow + rowsPerBlock;
        if (endRow > totalRows) endRow = totalRows;
        this->startRow = startRow;
        this->rowCount = (endRow > startRow) ? (endRow - startRow) : 0;

        xGm.SetGlobalBuffer((__gm__ float*)x + startRow * channels, this->rowCount * channels);
        zGm.SetGlobalBuffer((__gm__ float*)z + startRow * channels, this->rowCount * channels);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, channels);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, channels);

        pipe.InitBuffer(inQueueX, 1, channels * sizeof(float));
        pipe.InitBuffer(inQueueG, 1, channels * sizeof(float));
        pipe.InitBuffer(inQueueB, 1, channels * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, channels * sizeof(float));
        pipe.InitBuffer(tmpBuf, channels * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Load gamma/beta once
        AscendC::LocalTensor<float> gLocal = inQueueG.AllocTensor<float>();
        AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
        AscendC::DataCopy(gLocal, gammaGm, this->channels);
        AscendC::DataCopy(bLocal, betaGm, this->channels);
        inQueueG.EnQue(gLocal);
        inQueueB.EnQue(bLocal);
        gLocal = inQueueG.DeQue<float>();
        bLocal = inQueueB.DeQue<float>();

        for (uint32_t r = 0; r < this->rowCount; r++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[r * this->channels], this->channels);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
            AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

            // Compute mean
            float sum = 0.0f;
            for (uint32_t i = 0; i < this->channels; i++) {
                sum += xLocal.GetValue(i);
            }
            float mean = sum / (float)this->channels;

            float varSum = 0.0f;
            for (uint32_t i = 0; i < this->channels; i++) {
                float d = xLocal.GetValue(i) - mean;
                varSum += d * d;
            }
            float var = varSum / (float)this->channels;
            float rstd = 1.0f / sqrtf(var + this->eps);

            for (uint32_t i = 0; i < this->channels; i++) {
                float v = (xLocal.GetValue(i) - mean) * rstd;
                v = v * gLocal.GetValue(i) + bLocal.GetValue(i);
                // GELU: 0.5*v*(1+tanh(sqrt(2/pi)*(v+0.044715*v^3)))
                float v3 = v * v * v;
                float inner = 0.7978845608f * (v + 0.044715f * v3);
                float t = tanhf(inner);
                float g = 0.5f * v * (1.0f + t);
                zLocal.SetValue(i, g * this->scalingFactor);
            }

            outQueueZ.EnQue<float>(zLocal);
            zLocal = outQueueZ.DeQue<float>();
            AscendC::DataCopy(zGm[r * this->channels], zLocal, this->channels);
            outQueueZ.FreeTensor(zLocal);
            inQueueX.FreeTensor(xLocal);
        }
        inQueueG.FreeTensor(gLocal);
        inQueueB.FreeTensor(bLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueG, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm, gammaGm, betaGm, zGm;
    uint32_t channels;
    uint32_t startRow;
    uint32_t rowCount;
    float eps;
    float scalingFactor;
};

extern "C" __global__ __aicore__ void conv_transpose3d_layer_norm_gelu_scaling_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelLNGelu op;
    op.Init(x, gamma, beta, z, tiling_data.totalRows, tiling_data.channels, tiling_data.eps, tiling_data.scalingFactor);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, bias=True, eps=1e-5, scaling_factor=1.0):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, bias=bias)
        self.layer_norm = nn.LayerNorm(out_channels, eps=eps)
        self.eps = eps
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = self.conv_transpose(x)
        # x shape: (B, C, D', H', W'); LayerNorm over last dim C => move channel to last
        x = x.permute(0, 2, 3, 4, 1).contiguous()
        out = custom_ops_lib.conv_transpose3d_layer_norm_gelu_scaling_custom(
            x, self.layer_norm.weight, self.layer_norm.bias, float(self.eps), float(self.scaling_factor))
        out = out.permute(0, 4, 1, 2, 3).contiguous()
        return out
'''
