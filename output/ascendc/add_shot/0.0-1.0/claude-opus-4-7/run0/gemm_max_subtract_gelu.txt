kernel_src="""
#include "kernel_operator.h"

// After max over dim=1 with keepdim, shape is (batch, 1)
// Then x - x.mean(dim=1, keepdim=True): mean over single element = x itself, so result = 0
// gelu(0) = 0
// So output is all zeros of shape (batch, 1)

class KernelGMSG {
public:
    __aicore__ inline KernelGMSG() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t batchSize)
    {
        this->batchSize = batchSize;
        uint32_t blockNum = AscendC::GetBlockNum();
        this->blockLength = batchSize / blockNum;
        zGm.SetGlobalBuffer((__gm__ float *)z + this->blockLength * AscendC::GetBlockIdx(), this->blockLength);
        pipe.InitBuffer(outQueueZ, 1, this->blockLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::Duplicate(zLocal, (float)0.0, this->blockLength);
        outQueueZ.EnQue<float>(zLocal);
        AscendC::LocalTensor<float> zOut = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm, zOut, this->blockLength);
        outQueueZ.FreeTensor(zOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t blockLength;
};

extern "C" __global__ __aicore__ void gemm_max_subtract_gelu_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGMSG op;
    op.Init(x, z, tiling_data.batchSize);
    op.Process();
}
"""
