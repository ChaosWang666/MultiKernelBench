kernel_src="""
#include "kernel_operator.h"

// x shape: (B, C, 1, H, W) after mean. spatial = H*W
// For each (b, s) compute softmax over C with x[b,c,s]+bias[c], then tanh, scaling.

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z, uint32_t batch, uint32_t channels, uint32_t spatial, float scaling)
    {
        this->batch = batch;
        this->channels = channels;
        this->spatial = spatial;
        this->scaling = scaling;
        this->totalBS = batch * spatial;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startBS = (this->totalBS * blockIdx) / blockNum;
        this->endBS = (this->totalBS * (blockIdx + 1)) / blockNum;

        xGm.SetGlobalBuffer((__gm__ float *)x, batch * channels * spatial);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, channels);
        zGm.SetGlobalBuffer((__gm__ float *)z, batch * channels * spatial);

        pipe.InitBuffer(biasBuf, ((channels * sizeof(float) + 31) / 32) * 32);
        pipe.InitBuffer(vecBuf, ((channels * sizeof(float) + 31) / 32) * 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> vec = vecBuf.Get<float>();
        
        // Load bias once
        for (uint32_t c = 0; c < channels; c++) {
            biasLocal.SetValue(c, biasGm.GetValue(c));
        }

        for (uint32_t idx = startBS; idx < endBS; idx++) {
            uint32_t b = idx / spatial;
            uint32_t s = idx % spatial;
            
            // Load x[b, :, s] + bias
            float maxVal = -1e30f;
            for (uint32_t c = 0; c < channels; c++) {
                float v = xGm.GetValue(b * channels * spatial + c * spatial + s) + biasLocal.GetValue(c);
                vec.SetValue(c, v);
                if (v > maxVal) maxVal = v;
            }
            
            // softmax
            float sum = 0.0f;
            for (uint32_t c = 0; c < channels; c++) {
                float e = expf(vec.GetValue(c) - maxVal);
                vec.SetValue(c, e);
                sum += e;
            }
            float invSum = 1.0f / sum;
            
            // tanh & scale & store
            for (uint32_t c = 0; c < channels; c++) {
                float p = vec.GetValue(c) * invSum;
                float t = tanhf(p) * scaling;
                zGm.SetValue(b * channels * spatial + c * spatial + s, t);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf, vecBuf;
    AscendC::GlobalTensor<float> xGm, biasGm, zGm;
    uint32_t batch, channels, spatial;
    uint32_t totalBS, startBS, endBS;
    float scaling;
};

extern "C" __global__ __aicore__ void convtranspose3d_mean_add_softmax_tanh_scaling_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, z, tiling_data.batch, tiling_data.channels, tiling_data.spatial, tiling_data.scaling);
    op.Process();
}
