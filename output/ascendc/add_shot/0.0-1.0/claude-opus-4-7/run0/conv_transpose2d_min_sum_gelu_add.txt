#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z, uint32_t totalLength, uint32_t width)
    {
        this->totalLength = totalLength;
        this->width = width;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->blockLength = totalLength / blockNum;
        this->offset = this->blockLength * blockIdx;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->offset, this->blockLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->offset, this->blockLength);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, 1);

        pipe.InitBuffer(inQueueX, 1, this->blockLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, this->blockLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm, this->blockLength);
        inQueueX.EnQue(xLocal);

        AscendC::LocalTensor<float> xL = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        float biasVal = biasGm.GetValue(0);
        const float c0 = 0.5f;
        const float c1 = 0.7978845608f; // sqrt(2/pi)
        const float c2 = 0.044715f;

        for (uint32_t i = 0; i < this->blockLength; i++) {
            float v = xL.GetValue(i);
            float t = c1 * (v + c2 * v * v * v);
            float e2 = AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(expf(2.0f * t));
            float tanh_t = (e2 - 1.0f) / (e2 + 1.0f);
            float gelu = c0 * v * (1.0f + tanh_t);
            zLocal.SetValue(i, gelu + biasVal);
        }

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xL);

        AscendC::LocalTensor<float> zOut = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm, zOut, this->blockLength);
        outQueueZ.FreeTensor(zOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    AscendC::GlobalTensor<float> biasGm;
    uint32_t totalLength;
    uint32_t width;
    uint32_t blockLength;
    uint32_t offset;
};

extern "C" __global__ __aicore__ void conv_transpose2d_min_sum_gelu_add_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, z, tiling_data.totalLength, tiling_data.width);
    op.Process();
}

import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, bias_shape):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride, padding, output_padding)
        self.bias = nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = torch.min(x, dim=1, keepdim=True)[0]
        x = torch.sum(x, dim=2, keepdim=True)
        x_contig = x.contiguous()
        bias_contig = self.bias.contiguous().view(-1)[:1].contiguous()
        return custom_ops_lib.conv_transpose2d_min_sum_gelu_add_custom(x_contig, bias_contig)
