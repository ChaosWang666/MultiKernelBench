kernel_src="""
#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR gn_weight, GM_ADDR gn_bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize, uint32_t numGroups,
                                 float eps, float negativeSlope)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->numGroups = numGroups;
        this->groupSize = hiddenSize / numGroups;
        this->eps = eps;
        this->negativeSlope = negativeSlope;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inputSize);
        wGm.SetGlobalBuffer((__gm__ float *)weight, hiddenSize * inputSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, hiddenSize);
        gnwGm.SetGlobalBuffer((__gm__ float *)gn_weight, hiddenSize);
        gnbGm.SetGlobalBuffer((__gm__ float *)gn_bias, hiddenSize);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * hiddenSize);
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        // each block handles a subset of batch rows
        for (uint32_t b = blockIdx; b < batchSize; b += blockNum) {
            ProcessRow(b);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t b)
    {
        // compute out[b, h] = sum_k x[b,k] * w[h,k] + bias[h], for all h
        // then group norm per group, then leaky relu, then *2
        // use scalar loops (simple correctness-first impl)
        for (uint32_t g = 0; g < numGroups; g++) {
            // compute group outputs into a temporary (scalar) - small groupSize
            float vals[64]; // assume groupSize <= 64 (8192/512=16)
            float mean = 0.0f;
            for (uint32_t i = 0; i < groupSize; i++) {
                uint32_t h = g * groupSize + i;
                float acc = 0.0f;
                for (uint32_t k = 0; k < inputSize; k++) {
                    acc += xGm.GetValue(b * inputSize + k) * wGm.GetValue(h * inputSize + k);
                }
                acc += bGm.GetValue(h);
                vals[i] = acc;
                mean += acc;
            }
            mean /= (float)groupSize;
            float var = 0.0f;
            for (uint32_t i = 0; i < groupSize; i++) {
                float d = vals[i] - mean;
                var += d * d;
            }
            var /= (float)groupSize;
            float invStd = 1.0f / sqrt(var + eps);
            for (uint32_t i = 0; i < groupSize; i++) {
                uint32_t h = g * groupSize + i;
                float normed = (vals[i] - mean) * invStd;
                float y = normed * gnwGm.GetValue(h) + gnbGm.GetValue(h);
                if (y < 0.0f) y = y * negativeSlope;
                y = y + y;
                zGm.SetValue(b * hiddenSize + h, y);
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm, wGm, bGm, gnwGm, gnbGm, zGm;
    uint32_t batchSize;
    uint32_t inputSize;
    uint32_t hiddenSize;
    uint32_t numGroups;
    uint32_t groupSize;
    float eps;
    float negativeSlope;
};

extern "C" __global__ __aicore__ void matmul_group_norm_leaky_relu_sum_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR gn_weight, GM_ADDR gn_bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, weight, bias, gn_weight, gn_bias, z,
            tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize,
            tiling_data.numGroups, tiling_data.eps, tiling_data.negativeSlope);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, input_size, hidden_size, num_groups, eps=1e-5, negative_slope=0.01):
        super(ModelNew, self).__init__()
        self.fc = nn.Linear(input_size, hidden_size)
        self.gn = nn.GroupNorm(num_groups=num_groups, num_channels=hidden_size, eps=eps)
        self.input_size = input_size
        self.hidden_size = hidden_size
        self.num_groups = num_groups
        self.eps = eps
        self.negative_slope = negative_slope

    def forward(self, x):
        batch_size = x.shape[0]
        return custom_ops_lib.matmul_group_norm_leaky_relu_sum_custom(
            x.contiguous(), self.fc.weight.contiguous(), self.fc.bias.contiguous(),
            self.gn.weight.contiguous(), self.gn.bias.contiguous(),
            batch_size, self.input_size, self.hidden_size, self.num_groups,
            float(self.eps), float(self.negative_slope))
