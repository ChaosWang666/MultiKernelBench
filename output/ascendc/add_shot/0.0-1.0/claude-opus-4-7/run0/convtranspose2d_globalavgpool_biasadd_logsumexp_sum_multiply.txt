#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z, uint32_t batchSize, uint32_t outChannels)
    {
        this->batchSize = batchSize;
        this->outChannels = outChannels;
        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * outChannels);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize);
    }
    __aicore__ inline void Process()
    {
        for (uint32_t b = 0; b < batchSize; b++) {
            float maxVal = -3.4e38f;
            for (uint32_t c = 0; c < outChannels; c++) {
                float v = xGm.GetValue(b * outChannels + c) + biasGm.GetValue(c);
                if (v > maxVal) maxVal = v;
            }
            float sumExp = 0.0f;
            for (uint32_t c = 0; c < outChannels; c++) {
                float v = xGm.GetValue(b * outChannels + c) + biasGm.GetValue(c);
                sumExp += AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                    AscendC::ScalarGetExpVal<float>(v - maxVal));
            }
            float lse = maxVal + AscendC::ScalarGetLogVal<float>(sumExp);
            zGm.SetValue(b, lse * 10.0f);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t outChannels;
};

extern "C" __global__ __aicore__ void convtranspose2d_globalavgpool_biasadd_logsumexp_sum_multiply_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
