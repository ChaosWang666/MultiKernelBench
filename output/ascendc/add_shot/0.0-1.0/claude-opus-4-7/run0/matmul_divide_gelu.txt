kernel_src="""
#include "kernel_operator.h"
using namespace AscendC;

constexpr int32_t BUFFER_NUM = 2;

class KernelMMDivGelu {
public:
    __aicore__ inline KernelMMDivGelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                uint32_t M, uint32_t N, uint32_t K, float divisor)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        this->invDivisor = 1.0f / divisor;

        uint32_t blockNum = GetBlockNum();
        uint32_t blockIdx = GetBlockIdx();
        // partition M rows across cores
        uint32_t rowsPerCore = (M + blockNum - 1) / blockNum;
        this->rowStart = blockIdx * rowsPerCore;
        this->rowEnd = (this->rowStart + rowsPerCore) > M ? M : (this->rowStart + rowsPerCore);
        if (this->rowStart >= M) {
            this->rowEnd = this->rowStart;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float*)weight, N * K);
        bGm.SetGlobalBuffer((__gm__ float*)bias, N);
        zGm.SetGlobalBuffer((__gm__ float*)z, M * N);

        pipe.InitBuffer(inQueueX, 1, K * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, K * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, N * sizeof(float));
        pipe.InitBuffer(inQueueB, 1, N * sizeof(float));
        pipe.InitBuffer(tmpBuf, K * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (rowStart >= rowEnd) return;

        // Load bias once
        LocalTensor<float> biasLocal = inQueueB.AllocTensor<float>();
        DataCopy(biasLocal, bGm, N);
        inQueueB.EnQue(biasLocal);
        biasLocal = inQueueB.DeQue<float>();

        for (uint32_t row = rowStart; row < rowEnd; row++) {
            // Load x row
            LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            DataCopy(xLocal, xGm[row * K], K);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
            LocalTensor<float> tmp = tmpBuf.Get<float>();

            // For each output column, compute dot product
            for (uint32_t n = 0; n < N; n++) {
                LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                DataCopy(wLocal, wGm[n * K], K);
                inQueueW.EnQue(wLocal);
                wLocal = inQueueW.DeQue<float>();

                Mul(tmp, xLocal, wLocal, K);
                inQueueW.FreeTensor(wLocal);

                float sum = 0.0f;
                // reduce sum via ReduceSum
                // Use ReduceSum scalar output
                ReduceSum<float>(tmp, tmp, tmp, K);
                // tmp[0] now has sum
                zLocal.SetValue(n, tmp.GetValue(0));
            }
            inQueueX.FreeTensor(xLocal);

            // Add bias and divide
            Add(zLocal, zLocal, biasLocal, N);
            Muls(zLocal, zLocal, invDivisor, N);

            // GELU: 0.5 * x * (1 + erf(x / sqrt(2)))
            // Use builtin Gelu if available
            Gelu(zLocal, zLocal, N);

            outQueueZ.EnQue(zLocal);
            LocalTensor<float> zOut = outQueueZ.DeQue<float>();
            DataCopy(zGm[row * N], zOut, N);
            outQueueZ.FreeTensor(zOut);
        }
        inQueueB.FreeTensor(biasLocal);
    }

private:
    TPipe pipe;
    TQue<TPosition::VECIN, 1> inQueueX, inQueueB;
    TQue<TPosition::VECIN, BUFFER_NUM> inQueueW;
    TQue<TPosition::VECOUT, 1> outQueueZ;
    TBuf<TPosition::VECCALC> tmpBuf;
    GlobalTensor<float> xGm, wGm, bGm, zGm;
    uint32_t M, N, K;
    uint32_t rowStart, rowEnd;
    float invDivisor;
};

extern "C" __global__ __aicore__ void matmul_divide_gelu_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMMDivGelu op;
    op.Init(x, weight, bias, z, tiling_data.M, tiling_data.N, tiling_data.K, tiling_data.divisor);
    op.Process();
}
