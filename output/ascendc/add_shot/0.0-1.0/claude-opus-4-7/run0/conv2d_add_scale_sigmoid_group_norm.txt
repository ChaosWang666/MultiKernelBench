#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR scale, GM_ADDR z,
                                uint32_t channels, uint32_t spatial, uint32_t groups)
    {
        this->channels = channels;
        this->spatial = spatial;
        this->groups = groups;
        this->channelsPerGroup = channels / groups;
        this->groupSize = this->channelsPerGroup * spatial;
        uint32_t bIdx = AscendC::GetBlockIdx();
        uint32_t sampleSize = channels * spatial;

        xGm.SetGlobalBuffer((__gm__ float*)x + bIdx * sampleSize, sampleSize);
        zGm.SetGlobalBuffer((__gm__ float*)z + bIdx * sampleSize, sampleSize);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, channels);
        scaleGm.SetGlobalBuffer((__gm__ float*)scale, channels);

        pipe.InitBuffer(inQueueX, 1, spatial * sizeof(float));
        pipe.InitBuffer(outQueue, 1, spatial * sizeof(float));
        pipe.InitBuffer(tmpBuf, spatial * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Phase 1: compute sigmoid((x+bias)*scale) into zGm, accumulate mean/var per group
        for (uint32_t g = 0; g < groups; g++) {
            float sum = 0.f;
            float sumSq = 0.f;
            for (uint32_t cc = 0; cc < channelsPerGroup; cc++) {
                uint32_t c = g * channelsPerGroup + cc;
                float b = biasGm.GetValue(c);
                float s = scaleGm.GetValue(c);

                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[c * spatial], spatial);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();

                AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
                AscendC::Adds(outLocal, xLocal, b, spatial);
                AscendC::Muls(outLocal, outLocal, s, spatial);
                // sigmoid: 1/(1+exp(-x))
                AscendC::Muls(outLocal, outLocal, -1.0f, spatial);
                AscendC::Exp(outLocal, outLocal, spatial);
                AscendC::Adds(outLocal, outLocal, 1.0f, spatial);
                AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
                AscendC::Duplicate(tmp, 1.0f, spatial);
                AscendC::Div(outLocal, tmp, outLocal, spatial);

                // accumulate
                for (uint32_t i = 0; i < spatial; i++) {
                    float v = outLocal.GetValue(i);
                    sum += v;
                    sumSq += v * v;
                }

                outQueue.EnQue<float>(outLocal);
                outLocal = outQueue.DeQue<float>();
                AscendC::DataCopy(zGm[c * spatial], outLocal, spatial);
                outQueue.FreeTensor(outLocal);
                inQueueX.FreeTensor(xLocal);
            }
            float mean = sum / (float)groupSize;
            float var = sumSq / (float)groupSize - mean * mean;
            float invStd = 1.0f / AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                sqrt(var + 1e-5f));

            // Phase 2: normalize
            for (uint32_t cc = 0; cc < channelsPerGroup; cc++) {
                uint32_t c = g * channelsPerGroup + cc;
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, zGm[c * spatial], spatial);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();

                AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
                AscendC::Adds(outLocal, xLocal, -mean, spatial);
                AscendC::Muls(outLocal, outLocal, invStd, spatial);

                outQueue.EnQue<float>(outLocal);
                outLocal = outQueue.DeQue<float>();
                AscendC::DataCopy(zGm[c * spatial], outLocal, spatial);
                outQueue.FreeTensor(outLocal);
                inQueueX.FreeTensor(xLocal);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm, zGm, biasGm, scaleGm;
    uint32_t channels, spatial, groups, channelsPerGroup, groupSize;
};

extern "C" __global__ __aicore__ void conv2d_add_scale_sigmoid_group_norm_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR scale, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, scale, z, tiling_data.channels, tiling_data.spatial, tiling_data.groups);
    op.Process();
}
