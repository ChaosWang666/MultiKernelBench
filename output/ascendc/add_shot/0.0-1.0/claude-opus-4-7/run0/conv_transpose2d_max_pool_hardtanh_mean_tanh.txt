kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t spatialLength)
    {
        this->spatialLength = spatialLength;
        // Each channel (N*C total) is processed by a block
        uint32_t totalChannels = totalLength / spatialLength;
        uint32_t blockNum = AscendC::GetBlockNum();
        this->channelsPerBlock = (totalChannels + blockNum - 1) / blockNum;
        uint32_t startChannel = this->channelsPerBlock * AscendC::GetBlockIdx();
        uint32_t endChannel = startChannel + this->channelsPerBlock;
        if (endChannel > totalChannels) endChannel = totalChannels;
        this->myChannels = (endChannel > startChannel) ? (endChannel - startChannel) : 0;

        xGm.SetGlobalBuffer((__gm__ float *)x + startChannel * spatialLength, this->myChannels * spatialLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + startChannel * spatialLength, this->myChannels * spatialLength);

        // tile along spatial: process in chunks
        this->tileLength = 2048;
        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t c = 0; c < this->myChannels; c++) {
            float sum = 0.0f;
            uint32_t offset = c * this->spatialLength;
            uint32_t remaining = this->spatialLength;
            uint32_t pos = 0;
            while (remaining > 0) {
                uint32_t curLen = remaining > this->tileLength ? this->tileLength : remaining;
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[offset + pos], curLen);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<float> xL = inQueueX.DeQue<float>();
                AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
                // hardtanh: clamp to [-1, 1]
                AscendC::Mins(tmp, xL, 1.0f, curLen);
                AscendC::Maxs(tmp, tmp, -1.0f, curLen);
                // sum reduce
                for (uint32_t i = 0; i < curLen; i++) {
                    sum += tmp.GetValue(i);
                }
                inQueueX.FreeTensor(xL);
                pos += curLen;
                remaining -= curLen;
            }
            float mean = sum / (float)this->spatialLength;
            // tanh
            float e1 = 1.0f;
            // approximate tanh via formula: (exp(2x)-1)/(exp(2x)+1)
            float x2 = mean * 2.0f;
            // scalar exp using AscendC is not trivial; use math
            float ex = 1.0f;
            // Taylor won't be accurate; use builtin
            ex = expf(x2);
            float tanhv = (ex - 1.0f) / (ex + 1.0f);
            // write to first element, zeros elsewhere
            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
            tmp.SetValue(0, tanhv);
            AscendC::DataCopy(yGm[c * this->spatialLength], tmp, 8);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t spatialLength;
    uint32_t channelsPerBlock;
    uint32_t myChannels;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.spatialLength);
    op.Process();
}
