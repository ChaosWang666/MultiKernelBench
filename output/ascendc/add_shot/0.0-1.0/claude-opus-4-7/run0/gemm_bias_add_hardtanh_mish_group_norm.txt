kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t OUT_FEATURES = 8192;
constexpr int32_t NUM_GROUPS = 256;
constexpr int32_t GROUP_SIZE = OUT_FEATURES / NUM_GROUPS; // 32

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z, uint32_t totalLength, uint32_t tileNum)
    {
        // totalLength = batch * out_features. Each row is one sample of length OUT_FEATURES
        uint32_t totalRows = totalLength / OUT_FEATURES;
        uint32_t rowsPerBlock = totalRows / AscendC::GetBlockNum();
        this->rowsPerBlock = rowsPerBlock;
        this->rowOffset = rowsPerBlock * AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ float *)x + this->rowOffset * OUT_FEATURES, rowsPerBlock * OUT_FEATURES);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, OUT_FEATURES);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->rowOffset * OUT_FEATURES, rowsPerBlock * OUT_FEATURES);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, OUT_FEATURES * sizeof(float));
        pipe.InitBuffer(inQueueBias, 1, OUT_FEATURES * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, OUT_FEATURES * sizeof(float));
        pipe.InitBuffer(tmpBuf1, OUT_FEATURES * sizeof(float));
        pipe.InitBuffer(tmpBuf2, OUT_FEATURES * sizeof(float));

        // load bias once
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
        AscendC::DataCopy(biasLocal, biasGm, OUT_FEATURES);
        inQueueBias.EnQue(biasLocal);
        biasCached = inQueueBias.DeQue<float>();
    }

    __aicore__ inline void Process()
    {
        for (uint32_t r = 0; r < rowsPerBlock; r++) {
            CopyIn(r);
            Compute(r);
            CopyOut(r);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t r)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[r * OUT_FEATURES], OUT_FEATURES);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t r)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();

        // x = x + bias
        AscendC::Add(tmp1, xLocal, biasCached, OUT_FEATURES);
        // hardtanh: clamp [-1, 1]
        AscendC::Mins(tmp1, tmp1, (float)1.0f, OUT_FEATURES);
        AscendC::Maxs(tmp1, tmp1, (float)-1.0f, OUT_FEATURES);
        // mish: x * tanh(softplus(x)) = x * tanh(ln(1+exp(x)))
        AscendC::Exp(tmp2, tmp1, OUT_FEATURES);
        AscendC::Adds(tmp2, tmp2, (float)1.0f, OUT_FEATURES);
        AscendC::Ln(tmp2, tmp2, OUT_FEATURES);
        AscendC::Tanh(tmp2, tmp2, OUT_FEATURES);
        AscendC::Mul(tmp1, tmp1, tmp2, OUT_FEATURES);

        // GroupNorm: per group of GROUP_SIZE elements, compute mean & var, normalize
        // No affine (gamma=1, beta=0) since model uses default GroupNorm with affine=True actually...
        // Default nn.GroupNorm has affine=True. But weights default to 1, bias to 0, so initially identity.
        // We'll compute without affine assuming default initial weights.
        const float invN = 1.0f / (float)GROUP_SIZE;
        const float eps = 1e-5f;
        for (int32_t g = 0; g < NUM_GROUPS; g++) {
            int32_t off = g * GROUP_SIZE;
            float sum = 0.0f;
            float sqsum = 0.0f;
            for (int32_t i = 0; i < GROUP_SIZE; i++) {
                float v = tmp1.GetValue(off + i);
                sum += v;
                sqsum += v * v;
            }
            float mean = sum * invN;
            float var = sqsum * invN - mean * mean;
            float rstd = 1.0f / AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                            (float)sqrt(var + eps));
            for (int32_t i = 0; i < GROUP_SIZE; i++) {
                float v = tmp1.GetValue(off + i);
                zLocal.SetValue(off + i, (v - mean) * rstd);
            }
        }

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t r)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[r * OUT_FEATURES], zLocal, OUT_FEATURES);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::LocalTensor<float> biasCached;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t rowsPerBlock;
    uint32_t rowOffset;
};

extern "C" __global__ __aicore__ void gemm_bias_add_hardtanh_mish_group_norm_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, z, tiling_data.totalLength, tiling_data.tileNum);
    op.Process();
}
