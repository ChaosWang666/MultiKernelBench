kernel_src="""
#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR z, uint32_t batchChannels, uint32_t spatialSize, uint32_t channels)
    {
        this->batchChannels = batchChannels;
        this->spatialSize = spatialSize;
        this->channels = channels;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t perBlock = (batchChannels + blockNum - 1) / blockNum;
        this->startIdx = blockIdx * perBlock;
        this->endIdx = (this->startIdx + perBlock < batchChannels) ? (this->startIdx + perBlock) : batchChannels;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchChannels * spatialSize);
        scaleGm.SetGlobalBuffer((__gm__ float *)scale, channels);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchChannels * spatialSize);

        pipe.InitBuffer(inQueueX, 1, this->spatialSize * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, this->spatialSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->spatialSize * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = this->startIdx; i < this->endIdx; i++) {
            ProcessOne(i);
        }
    }

private:
    __aicore__ inline void ProcessOne(uint32_t idx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[idx * this->spatialSize], this->spatialSize);
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        // clamp to [0, 1]
        AscendC::Mins(tmp, xLocal, 1.0f, this->spatialSize);
        AscendC::Maxs(xLocal, tmp, 0.0f, this->spatialSize);

        // softmax: find max
        float maxVal = -1e30f;
        for (uint32_t i = 0; i < this->spatialSize; i++) {
            float v = xLocal.GetValue(i);
            if (v > maxVal) maxVal = v;
        }
        // exp(x - max)
        AscendC::Adds(tmp, xLocal, -maxVal, this->spatialSize);
        AscendC::Exp(xLocal, tmp, this->spatialSize);
        // sum
        float sumVal = 0.0f;
        for (uint32_t i = 0; i < this->spatialSize; i++) {
            sumVal += xLocal.GetValue(i);
        }
        float inv = 1.0f / sumVal;
        AscendC::Muls(tmp, xLocal, inv, this->spatialSize);

        // multiply by scale (per channel)
        uint32_t channelIdx = idx % this->channels;
        float s = scaleGm.GetValue(channelIdx);
        AscendC::Muls(zLocal, tmp, s, this->spatialSize);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);

        zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[idx * this->spatialSize], zLocal, this->spatialSize);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> scaleGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchChannels;
    uint32_t spatialSize;
    uint32_t channels;
    uint32_t startIdx;
    uint32_t endIdx;
};

extern "C" __global__ __aicore__ void conv_transpose3d_avg_pool_clamp_softmax_multiply_custom(GM_ADDR x, GM_ADDR scale, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    uint32_t channels = 64;
    op.Init(x, scale, z, tiling_data.batchChannels, tiling_data.spatialSize, channels);
    op.Process();
}
