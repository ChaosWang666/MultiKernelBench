kernel_src="""
#include "kernel_operator.h"

class KernelInstanceNormDivide {
public:
    __aicore__ inline KernelInstanceNormDivide() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t spatialSize, uint32_t channelStride, float divideBy)
    {
        this->spatialSize = spatialSize;
        this->channelStride = channelStride;
        this->divideBy = divideBy;
        this->channels = channelStride / spatialSize;
        
        uint32_t blockIdx = AscendC::GetBlockIdx();
        xGm.SetGlobalBuffer((__gm__ float *)x + channelStride * blockIdx, channelStride);
        yGm.SetGlobalBuffer((__gm__ float *)y + channelStride * blockIdx, channelStride);
        
        pipe.InitBuffer(inQueue, 1, this->spatialSize * sizeof(float));
        pipe.InitBuffer(outQueue, 1, this->spatialSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->spatialSize * sizeof(float));
    }
    
    __aicore__ inline void Process()
    {
        for (uint32_t c = 0; c < this->channels; c++) {
            ProcessChannel(c);
        }
    }

private:
    __aicore__ inline void ProcessChannel(uint32_t c)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[c * spatialSize], spatialSize);
        inQueue.EnQue(xLocal);
        
        xLocal = inQueue.DeQue<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        
        // Compute mean
        AscendC::ReduceSum<float>(tmp, xLocal, tmp, spatialSize);
        float sum = tmp.GetValue(0);
        float mean = sum / (float)spatialSize;
        
        // Subtract mean
        AscendC::Adds(tmp, xLocal, -mean, spatialSize);
        
        // Compute variance: square then sum
        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::Mul(outLocal, tmp, tmp, spatialSize);
        
        AscendC::LocalTensor<float> reduceTmp = inQueue.AllocTensor<float>();
        AscendC::ReduceSum<float>(reduceTmp, outLocal, reduceTmp, spatialSize);
        float varSum = reduceTmp.GetValue(0);
        inQueue.FreeTensor(reduceTmp);
        
        float var = varSum / (float)spatialSize;
        float invStd = 1.0f / sqrt(var + 1e-5f);
        float scale = invStd / divideBy;
        
        // Multiply (x - mean) by scale
        AscendC::Muls(outLocal, tmp, scale, spatialSize);
        
        outQueue.EnQue(outLocal);
        inQueue.FreeTensor(xLocal);
        
        outLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(yGm[c * spatialSize], outLocal, spatialSize);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t spatialSize;
    uint32_t channelStride;
    uint32_t channels;
    float divideBy;
};

extern "C" __global__ __aicore__ void conv2d_instance_norm_divide_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelInstanceNormDivide op;
    op.Init(x, y, tiling_data.spatialSize, tiling_data.channelStride, tiling_data.divideBy);
    op.Process();
}
