kernel_src="""
#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gemmW, GM_ADDR gemmB, GM_ADDR gnW, GM_ADDR gnB, GM_ADDR mulW, GM_ADDR z,
                                 uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures, uint32_t numGroups, uint32_t groupSize)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->groupSize = groupSize;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        gemmWGm.SetGlobalBuffer((__gm__ float *)gemmW, outFeatures * inFeatures);
        gemmBGm.SetGlobalBuffer((__gm__ float *)gemmB, outFeatures);
        gnWGm.SetGlobalBuffer((__gm__ float *)gnW, outFeatures);
        gnBGm.SetGlobalBuffer((__gm__ float *)gnB, outFeatures);
        mulWGm.SetGlobalBuffer((__gm__ float *)mulW, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * outFeatures);

        pipe.InitBuffer(inQueueX, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(inQueueParam, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf, inFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        uint32_t rowStart = blockIdx * rowsPerBlock;
        uint32_t rowEnd = rowStart + rowsPerBlock;
        if (rowEnd > batchSize) rowEnd = batchSize;

        for (uint32_t row = rowStart; row < rowEnd; row++) {
            ProcessRow(row);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t row)
    {
        // Load x row
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * inFeatures], inFeatures);
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        // GEMM: y[j] = sum_i x[i] * W[j,i] + b[j]
        for (uint32_t j = 0; j < outFeatures; j++) {
            AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
            AscendC::DataCopy(wLocal, gemmWGm[j * inFeatures], inFeatures);
            inQueueW.EnQue(wLocal);
            wLocal = inQueueW.DeQue<float>();

            AscendC::Mul(tmp, xLocal, wLocal, inFeatures);
            inQueueW.FreeTensor(wLocal);

            float sum = 0.0f;
            for (uint32_t i = 0; i < inFeatures; i++) {
                sum += tmp.GetValue(i);
            }
            yLocal.SetValue(j, sum);
        }
        inQueueX.FreeTensor(xLocal);

        // Add bias
        AscendC::LocalTensor<float> bLocal = inQueueParam.AllocTensor<float>();
        AscendC::DataCopy(bLocal, gemmBGm, outFeatures);
        inQueueParam.EnQue(bLocal);
        bLocal = inQueueParam.DeQue<float>();
        AscendC::Add(yLocal, yLocal, bLocal, outFeatures);
        inQueueParam.FreeTensor(bLocal);

        // GroupNorm per group
        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t off = g * groupSize;
            float mean = 0.0f;
            for (uint32_t i = 0; i < groupSize; i++) mean += yLocal.GetValue(off + i);
            mean /= (float)groupSize;
            float var = 0.0f;
            for (uint32_t i = 0; i < groupSize; i++) {
                float d = yLocal.GetValue(off + i) - mean;
                var += d * d;
            }
            var /= (float)groupSize;
            float invStd = 1.0f / AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(0.0f));
            // Compute invStd via sqrt
            float denom = 1.0f;
            {
                float s = var + 1e-5f;
                // Newton for sqrt
                float r = s;
                for (int k = 0; k < 20; k++) { r = 0.5f * (r + s / r); }
                denom = r;
            }
            invStd = 1.0f / denom;
            for (uint32_t i = 0; i < groupSize; i++) {
                float v = (yLocal.GetValue(off + i) - mean) * invStd;
                yLocal.SetValue(off + i, v);
            }
        }

        // Apply gn_weight and gn_bias
        AscendC::LocalTensor<float> gnwLocal = inQueueParam.AllocTensor<float>();
        AscendC::DataCopy(gnwLocal, gnWGm, outFeatures);
        inQueueParam.EnQue(gnwLocal);
        gnwLocal = inQueueParam.DeQue<float>();
        AscendC::Mul(yLocal, yLocal, gnwLocal, outFeatures);
        inQueueParam.FreeTensor(gnwLocal);

        AscendC::LocalTensor<float> gnbLocal = inQueueParam.AllocTensor<float>();
        AscendC::DataCopy(gnbLocal, gnBGm, outFeatures);
        inQueueParam.EnQue(gnbLocal);
        gnbLocal = inQueueParam.DeQue<float>();
        AscendC::Add(yLocal, yLocal, gnbLocal, outFeatures);
        inQueueParam.FreeTensor(gnbLocal);

        // Swish: y = y * sigmoid(y)
        AscendC::LocalTensor<float> sig = tmpBuf.Get<float>();
        // sigmoid
        for (uint32_t i = 0; i < outFeatures; i++) {
            float v = yLocal.GetValue(i);
            float e = 1.0f;
            // compute exp(-v) crude via expansion -- use AscendC::Exp instead
            sig.SetValue(i, -v);
        }
        AscendC::Exp(sig, sig, outFeatures);
        // sigmoid = 1/(1+exp(-v))
        for (uint32_t i = 0; i < outFeatures; i++) {
            float s = 1.0f / (1.0f + sig.GetValue(i));
            sig.SetValue(i, s);
        }
        AscendC::Mul(yLocal, yLocal, sig, outFeatures);

        // Multiply by multiply_weight
        AscendC::LocalTensor<float> mwLocal = inQueueParam.AllocTensor<float>();
        AscendC::DataCopy(mwLocal, mulWGm, outFeatures);
        inQueueParam.EnQue(mwLocal);
        mwLocal = inQueueParam.DeQue<float>();
        AscendC::Mul(yLocal, yLocal, mwLocal, outFeatures);
        inQueueParam.FreeTensor(mwLocal);

        // Swish again
        for (uint32_t i = 0; i < outFeatures; i++) {
            sig.SetValue(i, -yLocal.GetValue(i));
        }
        AscendC::Exp(sig, sig, outFeatures);
        for (uint32_t i = 0; i < outFeatures; i++) {
            float s = 1.0f / (1.0f + sig.GetValue(i));
            sig.SetValue(i, s);
        }
        AscendC::Mul(yLocal, yLocal, sig, outFeatures);

        outQueueY.EnQue(yLocal);
        yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(zGm[row * outFeatures], yLocal, outFeatures);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueW, inQueueParam;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm, gemmWGm, gemmBGm, gnWGm, gnBGm, mulWGm, zGm;
    uint32_t batchSize, inFeatures, outFeatures, numGroups, groupSize;
};

extern "C" __global__ __aicore__ void gemm_group_norm_swish_multiply_swish_custom(
    GM_ADDR x, GM_ADDR gemm_weight, GM_ADDR gemm_bias, GM_ADDR gn_weight, GM_ADDR gn_bias, GM_ADDR multiply_weight,
    GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, gemm_weight, gemm_bias, gn_weight, gn_bias, multiply_weight, z,
            tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures,
            tiling_data.numGroups, tiling_data.groupSize);
    op.Process();
}
