kernel_src="""
#include "kernel_operator.h"

class KernelGemmSigmoidSumLogSumExp {
public:
    __aicore__ inline KernelGemmSigmoidSumLogSumExp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w1, GM_ADDR b1, GM_ADDR w2, GM_ADDR b2, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize, uint32_t outputSize)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->outputSize = outputSize;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        this->endRow = this->startRow + this->rowsPerBlock;
        if (this->endRow > batchSize) this->endRow = batchSize;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inputSize);
        w1Gm.SetGlobalBuffer((__gm__ float *)w1, hiddenSize * inputSize);
        b1Gm.SetGlobalBuffer((__gm__ float *)b1, hiddenSize);
        w2Gm.SetGlobalBuffer((__gm__ float *)w2, outputSize * hiddenSize);
        b2Gm.SetGlobalBuffer((__gm__ float *)b2, outputSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize);

        pipe.InitBuffer(xBuf, inputSize * sizeof(float));
        pipe.InitBuffer(hBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(oBuf, outputSize * sizeof(float));
        pipe.InitBuffer(wRowBuf, ((inputSize > hiddenSize ? inputSize : hiddenSize)) * sizeof(float));
        pipe.InitBuffer(tmpBuf, ((inputSize > hiddenSize ? inputSize : hiddenSize)) * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> hLocal = hBuf.Get<float>();
        AscendC::LocalTensor<float> oLocal = oBuf.Get<float>();
        AscendC::LocalTensor<float> wRow = wRowBuf.Get<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        for (uint32_t row = startRow; row < endRow; row++) {
            // Load x row
            AscendC::DataCopy(xLocal, xGm[row * inputSize], inputSize);
            AscendC::PipeBarrier<PIPE_ALL>();

            // Compute hidden = sigmoid(x @ w1^T + b1)
            for (uint32_t h = 0; h < hiddenSize; h++) {
                AscendC::DataCopy(wRow, w1Gm[h * inputSize], inputSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::Mul(tmp, xLocal, wRow, inputSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                float sum = 0.0f;
                for (uint32_t i = 0; i < inputSize; i++) {
                    sum += tmp.GetValue(i);
                }
                sum += b1Gm.GetValue(h);
                // sigmoid
                float s = 1.0f / (1.0f + AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(expf(-sum)));
                hLocal.SetValue(h, s);
            }
            AscendC::PipeBarrier<PIPE_ALL>();

            // Compute output = hidden @ w2^T + b2
            float maxVal = -3.4e38f;
            for (uint32_t o = 0; o < outputSize; o++) {
                AscendC::DataCopy(wRow, w2Gm[o * hiddenSize], hiddenSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::Mul(tmp, hLocal, wRow, hiddenSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                float sum = 0.0f;
                for (uint32_t i = 0; i < hiddenSize; i++) {
                    sum += tmp.GetValue(i);
                }
                sum += b2Gm.GetValue(o);
                oLocal.SetValue(o, sum);
                if (sum > maxVal) maxVal = sum;
            }
            AscendC::PipeBarrier<PIPE_ALL>();

            // LogSumExp
            float sumExp = 0.0f;
            for (uint32_t o = 0; o < outputSize; o++) {
                sumExp += expf(oLocal.GetValue(o) - maxVal);
            }
            float result = maxVal + logf(sumExp);
            yGm.SetValue(row, result);
            AscendC::PipeBarrier<PIPE_ALL>();
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, hBuf, oBuf, wRowBuf, tmpBuf;
    AscendC::GlobalTensor<float> xGm, w1Gm, b1Gm, w2Gm, b2Gm, yGm;
    uint32_t batchSize, inputSize, hiddenSize, outputSize;
    uint32_t startRow, endRow, rowsPerBlock;
};

extern "C" __global__ __aicore__ void gemm_sigmoid_sum_log_sum_exp_custom(
    GM_ADDR x, GM_ADDR w1, GM_ADDR b1, GM_ADDR w2, GM_ADDR b2, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmSigmoidSumLogSumExp op;
    op.Init(x, w1, b1, w2, b2, y,
            tiling_data.batchSize, tiling_data.inputSize,
            tiling_data.hiddenSize, tiling_data.outputSize);
    op.Process();
}
