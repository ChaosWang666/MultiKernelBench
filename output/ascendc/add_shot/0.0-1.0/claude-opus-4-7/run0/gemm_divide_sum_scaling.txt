kernel_src="""
#include "kernel_operator.h"

// Key math insight:
// out[b] = scaling_factor * sum_h( sum_i( x[b,i] * w[h,i] ) / 2 )
//        = (scaling_factor / 2) * sum_i( x[b,i] * sum_h(w[h,i]) )
// So precompute wsum[i] = sum_h(w[h,i]) once, then for each batch row compute dot(x[b], wsum) * factor.

class KernelGemmDivSumScale {
public:
    __aicore__ inline KernelGemmDivSumScale() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR z,
                                uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize, float scalingFactor)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->factor = scalingFactor * 0.5f;

        this->blockIdx = AscendC::GetBlockIdx();
        this->blockNum = AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inputSize);
        wGm.SetGlobalBuffer((__gm__ float*)w, hiddenSize * inputSize);
        zGm.SetGlobalBuffer((__gm__ float*)z, batchSize);

        // Tile along inputSize dimension
        this->tileLen = 2048;  // elements per tile
        if (this->tileLen > inputSize) this->tileLen = inputSize;

        pipe.InitBuffer(wsumBuf, this->tileLen * sizeof(float));
        pipe.InitBuffer(wRowBuf, this->tileLen * sizeof(float));
        pipe.InitBuffer(xRowBuf, this->tileLen * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->tileLen * sizeof(float));
        pipe.InitBuffer(redBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Each AI core processes a subset of batch rows
        uint32_t rowsPerCore = (batchSize + blockNum - 1) / blockNum;
        uint32_t startRow = blockIdx * rowsPerCore;
        uint32_t endRow = startRow + rowsPerCore;
        if (endRow > batchSize) endRow = batchSize;

        for (uint32_t b = startRow; b < endRow; b++) {
            ProcessRow(b);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t b)
    {
        AscendC::LocalTensor<float> wsumLocal = wsumBuf.Get<float>();
        AscendC::LocalTensor<float> wRowLocal = wRowBuf.Get<float>();
        AscendC::LocalTensor<float> xRowLocal = xRowBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> redLocal = redBuf.Get<float>();

        float totalSum = 0.0f;

        uint32_t numTiles = (inputSize + tileLen - 1) / tileLen;
        for (uint32_t t = 0; t < numTiles; t++) {
            uint32_t offset = t * tileLen;
            uint32_t curLen = tileLen;
            if (offset + curLen > inputSize) curLen = inputSize - offset;

            // Load x[b, offset:offset+curLen]
            AscendC::DataCopy(xRowLocal, xGm[b * inputSize + offset], curLen);

            // Compute wsum[offset:offset+curLen] = sum over h of w[h, i]
            AscendC::Duplicate(wsumLocal, 0.0f, curLen);
            for (uint32_t h = 0; h < hiddenSize; h++) {
                AscendC::DataCopy(wRowLocal, wGm[h * inputSize + offset], curLen);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::Add(wsumLocal, wsumLocal, wRowLocal, curLen);
            }
            AscendC::PipeBarrier<PIPE_ALL>();

            // tmp = x * wsum
            AscendC::Mul(tmpLocal, xRowLocal, wsumLocal, curLen);
            AscendC::PipeBarrier<PIPE_ALL>();

            // Reduce sum
            AscendC::ReduceSum(redLocal, tmpLocal, redLocal, curLen);
            AscendC::PipeBarrier<PIPE_ALL>();
            totalSum += redLocal.GetValue(0);
        }

        float result = totalSum * factor;
        // Write single scalar result
        AscendC::LocalTensor<float> outLocal = xRowBuf.Get<float>();
        outLocal.SetValue(0, result);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::DataCopyExtParams copyParams{1, (uint32_t)(1 * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPad(zGm[b], outLocal, copyParams);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wsumBuf, wRowBuf, xRowBuf, tmpBuf, redBuf;
    AscendC::GlobalTensor<float> xGm, wGm, zGm;
    uint32_t batchSize, inputSize, hiddenSize;
    uint32_t blockIdx, blockNum;
    uint32_t tileLen;
    float factor;
};

extern "C" __global__ __aicore__ void gemm_divide_sum_scaling_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmDivSumScale op;
    op.Init(x, w, z, tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize, tiling_data.scalingFactor);
    op.Process();
}
