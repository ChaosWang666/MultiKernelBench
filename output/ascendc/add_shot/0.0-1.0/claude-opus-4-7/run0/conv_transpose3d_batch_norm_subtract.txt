kernel_src="""
#include "kernel_operator.h"

class KernelSubMean {
public:
    __aicore__ inline KernelSubMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalLength, uint32_t spatialLength, uint32_t ncCount)
    {
        this->spatialLength = spatialLength;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t perBlock = (ncCount + blockNum - 1) / blockNum;
        uint32_t startNc = blockIdx * perBlock;
        uint32_t endNc = (startNc + perBlock) > ncCount ? ncCount : (startNc + perBlock);
        this->ncStart = startNc;
        this->ncEnd = endNc;
        
        xGm.SetGlobalBuffer((__gm__ float *)x, totalLength);
        zGm.SetGlobalBuffer((__gm__ float *)z, totalLength);
        
        pipe.InitBuffer(inQueueX, 1, spatialLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, spatialLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, spatialLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (uint32_t i = this->ncStart; i < this->ncEnd; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t idx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[idx * this->spatialLength], this->spatialLength);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(uint32_t idx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        
        AscendC::ReduceSum<float>(tmp, xLocal, tmp, this->spatialLength);
        float sumVal = tmp.GetValue(0);
        float meanVal = sumVal / (float)this->spatialLength;
        
        AscendC::Adds(zLocal, xLocal, -meanVal, this->spatialLength);
        
        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(uint32_t idx)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[idx * this->spatialLength], zLocal, this->spatialLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t spatialLength;
    uint32_t ncStart;
    uint32_t ncEnd;
};

extern "C" __global__ __aicore__ void conv_transpose3d_batch_norm_subtract_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSubMean op;
    op.Init(x, z, tiling_data.totalLength, tiling_data.spatialLength, tiling_data.ncCount);
    op.Process();
}
