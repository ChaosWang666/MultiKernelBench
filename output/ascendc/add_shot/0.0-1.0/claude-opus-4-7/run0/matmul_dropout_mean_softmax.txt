kernel_src="""
#include "kernel_operator.h"

// Note: during inference, dropout is effectively identity (eval mode),
// and softmax of a constant vector is uniform 1/N. But we implement matmul + softmax.

class KernelMMSoftmax {
public:
    __aicore__ inline KernelMMSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR z,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;
        this->endRow = this->startRow + this->rowsPerBlock;
        if (this->endRow > batchSize) this->endRow = batchSize;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float *)w, outFeatures * inFeatures);
        bGm.SetGlobalBuffer((__gm__ float *)b, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * outFeatures);

        pipe.InitBuffer(inQueueX, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueB, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf, inFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->startRow >= this->endRow) return;

        // Load bias once
        AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
        AscendC::DataCopy(bLocal, bGm, this->outFeatures);
        inQueueB.EnQue(bLocal);
        bLocal = inQueueB.DeQue<float>();

        for (uint32_t row = this->startRow; row < this->endRow; row++) {
            // Load x row
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[row * this->inFeatures], this->inFeatures);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

            // Compute z[j] = sum_i x[i] * w[j,i] + b[j]
            for (uint32_t j = 0; j < this->outFeatures; j++) {
                AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                AscendC::DataCopy(wLocal, wGm[j * this->inFeatures], this->inFeatures);
                inQueueW.EnQue(wLocal);
                wLocal = inQueueW.DeQue<float>();

                AscendC::Mul(tmp, xLocal, wLocal, this->inFeatures);
                float sum = 0.0f;
                // reduce sum via ReduceSum
                AscendC::LocalTensor<float> redTmp = wLocal; // reuse
                AscendC::ReduceSum<float>(redTmp, tmp, redTmp, this->inFeatures);
                sum = redTmp.GetValue(0);
                zLocal.SetValue(j, sum + bLocal.GetValue(j));

                inQueueW.FreeTensor(wLocal);
            }

            // Softmax over outFeatures
            // find max
            float maxVal = zLocal.GetValue(0);
            for (uint32_t j = 1; j < this->outFeatures; j++) {
                float v = zLocal.GetValue(j);
                if (v > maxVal) maxVal = v;
            }
            // subtract max and exp
            AscendC::Adds(zLocal, zLocal, -maxVal, this->outFeatures);
            AscendC::Exp(zLocal, zLocal, this->outFeatures);
            // sum
            AscendC::LocalTensor<float> redTmp2 = tmp;
            AscendC::ReduceSum<float>(redTmp2, zLocal, redTmp2, this->outFeatures);
            float sumExp = redTmp2.GetValue(0);
            float invSum = 1.0f / sumExp;
            AscendC::Muls(zLocal, zLocal, invSum, this->outFeatures);

            outQueueZ.EnQue<float>(zLocal);
            zLocal = outQueueZ.DeQue<float>();
            AscendC::DataCopy(zGm[row * this->outFeatures], zLocal, this->outFeatures);
            outQueueZ.FreeTensor(zLocal);

            inQueueX.FreeTensor(xLocal);
        }

        inQueueB.FreeTensor(bLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueW, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, zGm;
    uint32_t batchSize, inFeatures, outFeatures;
    uint32_t rowsPerBlock, startRow, endRow;
};

extern "C" __global__ __aicore__ void matmul_dropout_mean_softmax_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMMSoftmax op;
    op.Init(x, weight, bias, z, tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures);
    op.Process();
}
