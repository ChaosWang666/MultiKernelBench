kernel_src="""
#include "kernel_operator.h"

// Input after conv: shape (N, C, D, H, W)
// Min over dim=2 (D): -> (N, C, H, W)
// Softmax over dim=1 (C): -> (N, C, H, W)
// We parallelize over (N, H, W) spatial positions.

class KernelConv3dMinSoftmax {
public:
    __aicore__ inline KernelConv3dMinSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t N, uint32_t C, uint32_t D, uint32_t H, uint32_t W)
    {
        this->N = N;
        this->C = C;
        this->D = D;
        this->H = H;
        this->W = W;
        this->totalSpatial = N * H * W;
        xGm.SetGlobalBuffer((__gm__ float *)x);
        yGm.SetGlobalBuffer((__gm__ float *)y);
    }

    __aicore__ inline void Process()
    {
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t perCore = (totalSpatial + blockNum - 1) / blockNum;
        uint32_t start = blockIdx * perCore;
        uint32_t end = start + perCore;
        if (end > totalSpatial) end = totalSpatial;

        for (uint32_t idx = start; idx < end; idx++) {
            uint32_t n = idx / (H * W);
            uint32_t hw = idx % (H * W);
            uint32_t h = hw / W;
            uint32_t w = hw % W;

            // For each channel, compute min over D
            float minVals[64];
            float maxVal = -3.4e38f;
            for (uint32_t c = 0; c < C; c++) {
                float m = 3.4e38f;
                for (uint32_t d = 0; d < D; d++) {
                    uint64_t offset = ((uint64_t)n * C * D * H * W) + ((uint64_t)c * D * H * W) + ((uint64_t)d * H * W) + h * W + w;
                    float v = xGm.GetValue(offset);
                    if (v < m) m = v;
                }
                minVals[c] = m;
                if (m > maxVal) maxVal = m;
            }

            // Softmax over C
            float sum = 0.0f;
            for (uint32_t c = 0; c < C; c++) {
                float e = AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                    (float)expf(minVals[c] - maxVal));
                minVals[c] = e;
                sum += e;
            }
            float inv = 1.0f / sum;
            for (uint32_t c = 0; c < C; c++) {
                uint64_t outOff = ((uint64_t)n * C * H * W) + ((uint64_t)c * H * W) + h * W + w;
                yGm.SetValue(outOff, minVals[c] * inv);
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t N, C, D, H, W;
    uint32_t totalSpatial;
};

extern "C" __global__ __aicore__ void conv3d_min_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dMinSoftmax op;
    op.Init(x, y, tiling_data.N, tiling_data.C, tiling_data.D, tiling_data.H, tiling_data.W);
    op.Process();
}
