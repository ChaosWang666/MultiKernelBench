kernel_src="""
#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z, uint32_t batch, uint32_t channels, uint32_t spatial, float scalingFactor)
    {
        this->batch = batch;
        this->channels = channels;
        this->spatial = spatial;
        this->scalingFactor = scalingFactor;
        this->total = batch * spatial;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->startIdx = (this->total * blockIdx) / blockNum;
        this->endIdx = (this->total * (blockIdx + 1)) / blockNum;

        xGm.SetGlobalBuffer((__gm__ float *)x, batch * channels * spatial);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, channels);
        zGm.SetGlobalBuffer((__gm__ float *)z, batch * channels * spatial);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t idx = startIdx; idx < endIdx; idx++) {
            uint32_t b = idx / spatial;
            uint32_t s = idx % spatial;
            // Compute softmax across channels, then +bias, *scale, sigmoid
            float maxVal = -1e30f;
            for (uint32_t c = 0; c < channels; c++) {
                float v = xGm.GetValue(b * channels * spatial + c * spatial + s);
                if (v > maxVal) maxVal = v;
            }
            float sumExp = 0.0f;
            for (uint32_t c = 0; c < channels; c++) {
                float v = xGm.GetValue(b * channels * spatial + c * spatial + s);
                sumExp += expf(v - maxVal);
            }
            for (uint32_t c = 0; c < channels; c++) {
                float v = xGm.GetValue(b * channels * spatial + c * spatial + s);
                float sm = expf(v - maxVal) / sumExp;
                float bv = biasGm.GetValue(c);
                float r = (sm + bv) * scalingFactor;
                float sig = 1.0f / (1.0f + expf(-r));
                zGm.SetValue(b * channels * spatial + c * spatial + s, sig);
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batch;
    uint32_t channels;
    uint32_t spatial;
    uint32_t total;
    uint32_t startIdx;
    uint32_t endIdx;
    float scalingFactor;
};

extern "C" __global__ __aicore__ void convtranspose2d_softmax_biasadd_scaling_sigmoid_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, z, tiling_data.batch, tiling_data.channels, tiling_data.spatial, tiling_data.scalingFactor);
    op.Process();
}

import torch
import torch.nn as nn

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, bias_shape, scaling_factor):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)

    def forward(self, x):
        x = self.conv_transpose(x)
        bias_flat = self.bias.view(-1).contiguous()
