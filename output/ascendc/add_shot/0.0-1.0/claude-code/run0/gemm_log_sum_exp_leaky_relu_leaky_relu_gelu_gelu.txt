kernel_src="""
#include "kernel_operator.h"

constexpr float LEAKY_SLOPE_SQ = 0.0001f;
constexpr float GELU_C = 0.7978845608028654f;
constexpr float GELU_K = 0.044715f;
constexpr float NEG_INF = -3.4e38f;
constexpr float TANH_CLAMP = 20.0f;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t totalRows, uint32_t rowLength,
                                 uint32_t tileLength, uint32_t rowsPerBlock)
    {
        this->rowLength = rowLength;
        this->tileLength = tileLength;
        this->rowsPerBlock = rowsPerBlock;
        this->tilesPerRow = rowLength / tileLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowOffset = blockIdx * rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + rowOffset * rowLength, rowsPerBlock * rowLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + rowOffset, rowsPerBlock);

        pipe.InitBuffer(inQueueX, 2, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, rowsPerBlock * sizeof(float));
        pipe.InitBuffer(tmpBuf, tileLength * sizeof(float));
        pipe.InitBuffer(workBuf, tileLength * sizeof(float));
        pipe.InitBuffer(resBuf, 32 * sizeof(float));
        pipe.InitBuffer(maxBuf, rowsPerBlock * sizeof(float));
        pipe.InitBuffer(sumBuf, rowsPerBlock * sizeof(float));
        pipe.InitBuffer(tmpOutBuf, rowsPerBlock * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();

        for (uint32_t r = 0; r < rowsPerBlock; r++) {
            float rowMax = NEG_INF;
            for (uint32_t t = 0; t < tilesPerRow; t++) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[r * rowLength + t * tileLength], tileLength);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();

                AscendC::LocalTensor<float> work = workBuf.Get<float>();
                AscendC::LocalTensor<float> res = resBuf.Get<float>();
                AscendC::ReduceMax<float>(res, xLocal, work, (int32_t)tileLength, false);
                float tileMax = res.GetValue(0);
                if (tileMax > rowMax) rowMax = tileMax;

                inQueueX.FreeTensor(xLocal);
            }
            maxLocal.SetValue(r, rowMax);

            float rowSum = 0.0f;
            for (uint32_t t = 0; t < tilesPerRow; t++) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[r * rowLength + t * tileLength], tileLength);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();

                AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
                AscendC::Adds(tmp, xLocal, -rowMax, tileLength);
                AscendC::Exp(tmp, tmp, tileLength);

                AscendC::LocalTensor<float> work = workBuf.Get<float>();
                AscendC::LocalTensor<float> res = resBuf.Get<float>();
                AscendC::ReduceSum<float>(res, tmp, work, (int32_t)tileLength);
                rowSum += res.GetValue(0);

                inQueueX.FreeTensor(xLocal);
            }
            sumLocal.SetValue(r, rowSum);
        }

        AscendC::LocalTensor<float> tmpOut = tmpOutBuf.Get<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();

        AscendC::Ln(tmpOut, sumLocal, rowsPerBlock);
        AscendC::Add(tmpOut, tmpOut, maxLocal, rowsPerBlock);

        AscendC::Abs(tmp, tmpOut, rowsPerBlock);
        AscendC::Add(work, tmpOut, tmp, rowsPerBlock);
        AscendC::Muls(work, work, 0.5f, rowsPerBlock);
        AscendC::Sub(tmp, tmpOut, tmp, rowsPerBlock);
        AscendC::Muls(tmp, tmp, 0.5f * LEAKY_SLOPE_SQ, rowsPerBlock);
        AscendC::Add(tmpOut, work, tmp, rowsPerBlock);

        GeluVec(tmpOut, tmp, work, rowsPerBlock);
        GeluVec(tmpOut, tmp, work, rowsPerBlock);

        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::Adds(yLocal, tmpOut, 0.0f, rowsPerBlock);
        outQueueY.EnQue<float>(yLocal);
        yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm, yLocal, rowsPerBlock);
        outQueueY.FreeTensor(yLocal);
    }

private:
    __aicore__ inline void GeluVec(AscendC::LocalTensor<float> x,
                                    AscendC::LocalTensor<float> tmp,
                                    AscendC::LocalTensor<float> work,
                                    uint32_t n)
    {
        AscendC::Mul(work, x, x, n);
        AscendC::Mul(work, work, x, n);
        AscendC::Muls(work, work, GELU_K, n);
        AscendC::Add(tmp, x, work, n);
        AscendC::Muls(tmp, tmp, GELU_C, n);
        AscendC::Mins(tmp, tmp, TANH_CLAMP, n);
        AscendC::Maxs(tmp, tmp, -TANH_CLAMP, n);
        AscendC::Muls(work, tmp, 2.0f, n);
        AscendC::Exp(work, work, n);
        AscendC::Adds(tmp, work, -1.0f, n);
        AscendC::Adds(work, work, 1.0f, n);
        AscendC::Div(tmp, tmp, work, n);
        AscendC::Adds(tmp, tmp, 1.0f, n);
        AscendC::Mul(tmp, tmp, x, n);
        AscendC::Muls(x, tmp, 0.5f, n);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> resBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpOutBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t rowLength;
    uint32_t tileLength;
    uint32_t rowsPerBlock;
    uint32_t tilesPerRow;
};

extern "C" __global__ __aicore__ void gemm_log_sum_exp_leaky_relu_leaky_relu_gelu_gelu_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, y, tiling_data.totalRows, tiling_data.rowLength,
            tiling_data.tileLength, tiling_data.rowsPerBlock);
    op.Process();
}
