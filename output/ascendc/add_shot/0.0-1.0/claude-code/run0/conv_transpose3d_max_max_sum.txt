kernel_src="""
#include "kernel_operator.h"

class KernelConvTransposeMaxMaxSum {
public:
    __aicore__ inline KernelConvTransposeMaxMaxSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t batchSize, uint32_t numChannels, uint32_t spatialSize, uint32_t batchesPerBlock)
    {
        this->C = numChannels;
        this->N = spatialSize;
        this->batchesPerBlock = batchesPerBlock;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t batchStart = blockIdx * batchesPerBlock;
        if (batchStart >= batchSize) {
            this->batchesThisBlock = 0;
        } else {
            uint32_t remain = batchSize - batchStart;
            this->batchesThisBlock = (remain < batchesPerBlock) ? remain : batchesPerBlock;
        }

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + batchStart * this->C * this->N, this->batchesThisBlock * this->C * this->N);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + batchStart * this->N, this->batchesThisBlock * this->N);

        pipe.InitBuffer(inQueue, 2, this->N * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueue, 1, this->N * sizeof(DTYPE_Z));
        pipe.InitBuffer(accumBuf, this->N * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        if (this->batchesThisBlock == 0) {
            return;
        }
        AscendC::LocalTensor<DTYPE_Z> accum = accumBuf.Get<DTYPE_Z>();
        for (uint32_t b = 0; b < this->batchesThisBlock; b++) {
            AscendC::Duplicate(accum, (DTYPE_Z)0, this->N);
            for (uint32_t c = 0; c < this->C; c++) {
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueue.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(xLocal, xGm[b * this->C * this->N + c * this->N], this->N);
                inQueue.EnQue(xLocal);
                xLocal = inQueue.DeQue<DTYPE_X>();
                AscendC::Add(accum, accum, xLocal, this->N);
                inQueue.FreeTensor(xLocal);
            }
            AscendC::LocalTensor<DTYPE_Z> zLocal = outQueue.AllocTensor<DTYPE_Z>();
            AscendC::DataCopy(zLocal, accum, this->N);
            outQueue.EnQue<DTYPE_Z>(zLocal);
            zLocal = outQueue.DeQue<DTYPE_Z>();
            AscendC::DataCopy(zGm[b * this->N], zLocal, this->N);
            outQueue.FreeTensor(zLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accumBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t C;
    uint32_t N;
    uint32_t batchesPerBlock;
    uint32_t batchesThisBlock;
};

extern "C" __global__ __aicore__ void conv_transpose3d_max_max_sum_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTransposeMaxMaxSum op;
    op.Init(x, z, tiling_data.batchSize, tiling_data.numChannels, tiling_data.spatialSize, tiling_data.batchesPerBlock);
    op.Process();
}
