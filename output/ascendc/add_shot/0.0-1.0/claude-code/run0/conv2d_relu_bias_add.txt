kernel_src = """
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv2dReluBiasAdd {
public:
    __aicore__ inline KernelConv2dReluBiasAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                  uint32_t totalLength, uint32_t spatialSize,
                                  uint32_t channels)
    {
        this->totalLength = totalLength;
        this->spatialSize = spatialSize;
        this->channels = channels;

        uint32_t blockNum = AscendC::GetBlockNum();
        this->blockLength = totalLength / blockNum;
        this->channelsPerBlock = this->blockLength / spatialSize;
        this->blockOffset = this->blockLength * AscendC::GetBlockIdx();
        this->channelStartIdx = this->blockOffset / spatialSize;

        this->alignedSize = ((spatialSize + 7) / 8) * 8;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x + this->blockOffset, this->blockLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y + this->blockOffset, this->blockLength);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS*)bias, channels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->alignedSize * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->alignedSize * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->channelsPerBlock; i++) {
            uint32_t channelIdx = (this->channelStartIdx + i) % this->channels;
            CopyIn(i);
            Compute(channelIdx);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t i)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = this->spatialSize * sizeof(DTYPE_X);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPadExtParams<DTYPE_X> padParams{false, 0, 0, 0};
        AscendC::DataCopyPad(xLocal, xGm[i * this->spatialSize], copyParams, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t channelIdx)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        DTYPE_BIAS biasVal = biasGm.GetValue(channelIdx);

        AscendC::Maxs(yLocal, xLocal, (DTYPE_X)0, this->spatialSize);
        AscendC::Adds(yLocal, yLocal, biasVal, this->spatialSize);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t i)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = this->spatialSize * sizeof(DTYPE_Y);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPad(yGm[i * this->spatialSize], yLocal, copyParams);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t blockOffset;
    uint32_t spatialSize;
    uint32_t channels;
    uint32_t channelsPerBlock;
    uint32_t channelStartIdx;
    uint32_t alignedSize;
};

extern "C" __global__ __aicore__ void conv2d_relu_bias_add_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dReluBiasAdd op;
    op.Init(x, bias, y, tiling_data.totalLength, tiling_data.spatialSize, tiling_data.channels);
    op.Process();
}

model_src = '''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, bias_shape):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.bias = nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
        bias_flat = self.bias.view(-1).contiguous()
        return custom_ops_lib.conv2d_relu_bias_add_custom(x, bias_flat)
'''
