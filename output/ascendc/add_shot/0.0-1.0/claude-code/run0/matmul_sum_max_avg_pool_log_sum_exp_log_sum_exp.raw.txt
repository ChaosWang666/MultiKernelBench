kernel_src="""
#include "kernel_operator.h"


class KernelFusedOp {
public:
    __aicore__ inline KernelFusedOp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inFeatures, uint32_t rowsPerBlock)
    {
        this->inFeatures = inFeatures;
        this->rowsPerBlock = rowsPerBlock;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerBlock;
        uint32_t actual = rowsPerBlock;
        if (startRow >= batchSize) {
            actual = 0;
        } else if (startRow + actual > batchSize) {
            actual = batchSize - startRow;
        }
        this->actualRows = actual;

        xGm.SetGlobalBuffer((__gm__ float *)x + startRow * inFeatures, rowsPerBlock * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float *)w, inFeatures);
        bGm.SetGlobalBuffer((__gm__ float *)b, rowsPerBlock);
        yGm.SetGlobalBuffer((__gm__ float *)y + startRow, rowsPerBlock);

        pipe.InitBuffer(inQueueX, 2, inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueB, 1, rowsPerBlock * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, rowsPerBlock * sizeof(float));
        pipe.InitBuffer(tmpBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(workBuf, inFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->actualRows == 0) {
            return;
        }

        AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
        AscendC::DataCopy(wLocal, wGm, this->inFeatures);
        inQueueW.EnQue(wLocal);
        wLocal = inQueueW.DeQue<float>();

        AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
        AscendC::DataCopy(bLocal, bGm, this->rowsPerBlock);
        inQueueB.EnQue(bLocal);
        bLocal = inQueueB.DeQue<float>();

        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();

        for (uint32_t i = 0; i < this->actualRows; i++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[i * this->inFeatures], this->inFeatures);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::Mul(tmp, xLocal, wLocal, this->inFeatures);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum<float>(yLocal[i], tmp, work, this->inFeatures);
            AscendC::PipeBarrier<PIPE_V>();

            inQueueX.FreeTensor(xLocal);
        }

        AscendC::Add(yLocal, yLocal, bLocal, this->rowsPerBlock);

        outQueueY.EnQue(yLocal);
        yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm, yLocal, this->rowsPerBlock);
        outQueueY.FreeTensor(yLocal);

        inQueueW.FreeTensor(wLocal);
        inQueueB.FreeTensor(bLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueW;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t inFeatures;
    uint32_t rowsPerBlock;
    uint32_t actualRows;
};


extern "C" __global__ __aicore__ void matmul_sum_max_avg_pool_log_sum_exp_log_sum_exp_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {

    KernelFusedOp op;
    op.Init(x, w, b, y, tiling_data.batchSize, tiling_data.inFeatures, tiling_data.rowsPerBlock);

}
