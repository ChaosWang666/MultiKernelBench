kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGlobalAvgPool3D {
public:
    __aicore__ inline KernelGlobalAvgPool3D() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t numBcPerBlock,
                                uint32_t spatialSize,
                                uint32_t tileLength,
                                uint32_t tileNum,
                                uint32_t tailLength,
                                float invSpatial)
    {
        this->numBcPerBlock = numBcPerBlock;
        this->spatialSize = spatialSize;
        this->tileLength = tileLength;
        this->tileNum = tileNum;
        this->tailLength = tailLength;
        this->invSpatial = invSpatial;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockInOffset = blockIdx * numBcPerBlock * spatialSize;
        uint32_t blockOutOffset = blockIdx * numBcPerBlock;

        xGm.SetGlobalBuffer((__gm__ float *)x + blockInOffset, numBcPerBlock * spatialSize);
        yGm.SetGlobalBuffer((__gm__ float *)y + blockOutOffset, numBcPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, numBcPerBlock * sizeof(float));
        pipe.InitBuffer(sumBuf, tileLength * sizeof(float));
        pipe.InitBuffer(reduceBuf, 64);
        pipe.InitBuffer(tailBuf, 64);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> tailLocal = tailBuf.Get<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        for (uint32_t bc = 0; bc < numBcPerBlock; bc++) {
            AscendC::Duplicate<float>(sumLocal, 0.0f, tileLength);
            AscendC::PipeBarrier<PIPE_V>();

            for (uint32_t t = 0; t < tileNum; t++) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[bc * spatialSize + t * tileLength], tileLength);
                inQueueX.EnQue(xLocal);
                AscendC::LocalTensor<float> xLocalIn = inQueueX.DeQue<float>();
                AscendC::Add(sumLocal, sumLocal, xLocalIn, tileLength);
                AscendC::PipeBarrier<PIPE_V>();
                inQueueX.FreeTensor(xLocalIn);
            }

            float tailScalar = 0.0f;
            if (tailLength > 0) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[bc * spatialSize + tileNum * tileLength], tailLength);
                inQueueX.EnQue(xLocal);
                AscendC::LocalTensor<float> xLocalIn = inQueueX.DeQue<float>();
                AscendC::ReduceSum<float>(tailLocal, xLocalIn, tailLocal, tailLength);
                AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
                AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
                tailScalar = tailLocal.GetValue(0);
                AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID1);
                AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID1);
                inQueueX.FreeTensor(xLocalIn);
            }

            AscendC::ReduceSum<float>(reduceLocal, sumLocal, reduceLocal, tileLength);
            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            float mainSum = reduceLocal.GetValue(0);
            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

            yLocal.SetValue(bc, (mainSum + tailScalar) * invSpatial);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        outQueueY.EnQue(yLocal);
        AscendC::LocalTensor<float> yLocalOut = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm, yLocalOut, numBcPerBlock);
        outQueueY.FreeTensor(yLocalOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tailBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t numBcPerBlock;
    uint32_t spatialSize;
    uint32_t tileLength;
    uint32_t tileNum;
    uint32_t tailLength;
    float invSpatial;
};

extern "C" __global__ __aicore__ void conv_transpose3d_scale_batch_norm_global_avg_pool_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGlobalAvgPool3D op;
    op.Init(x, y,
            tiling_data.numBcPerBlock,
            tiling_data.spatialSize,
            tiling_data.tileLength,
            tiling_data.tileNum,
            tiling_data.tailLength,
            tiling_data.invSpatial);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, scale_factor, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size)
        self.scale_factor = scale_factor
        self.batch_norm = nn.BatchNorm3d(out_channels, eps=eps, momentum=momentum)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x * self.scale_factor
        x = self.batch_norm(x)
        x = custom_ops_lib.conv_transpose3d_scale_batch_norm_global_avg_pool_custom(x.contiguous())
        return x
