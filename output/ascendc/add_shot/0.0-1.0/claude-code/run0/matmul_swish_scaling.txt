kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t TILE_LEN = 2048;

template <typename aT, typename bT, typename cT, typename biasT>
class KernelMatmulSwishScaling {
public:
    __aicore__ inline KernelMatmulSwishScaling() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR workspace,
                                 GM_ADDR y, const TCubeTiling &tiling, float scalingFactor,
                                 uint32_t M, uint32_t N, uint32_t K, AscendC::TPipe *pipe)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        this->scalingFactor = scalingFactor;
        this->tiling = tiling;
        this->pipe = pipe;

        xGm.SetGlobalBuffer((__gm__ aT *)x);
        weightGm.SetGlobalBuffer((__gm__ bT *)weight);
        biasGm.SetGlobalBuffer((__gm__ biasT *)bias);
        yGm.SetGlobalBuffer((__gm__ cT *)y);
        workspaceGm.SetGlobalBuffer((__gm__ cT *)workspace);
    }

    __aicore__ inline void Process(AscendC::TPipe *pipe)
    {
        int32_t blockIdx = GetBlockIdx();
        matmulObj.SetTensorA(xGm);
        matmulObj.SetTensorB(weightGm, true);
        matmulObj.SetBias(biasGm);
        matmulObj.IterateAll(workspaceGm);
        matmulObj.End();

        SyncAll();

        if ASCEND_IS_AIV {
            PostProcess();
        }
    }

    __aicore__ inline void PostProcess()
    {
        uint32_t totalLen = M * N;
        uint32_t blockNum = GetBlockNum();
        uint32_t blockIdx = GetBlockIdx();
        uint32_t lenPerBlock = (totalLen + blockNum - 1) / blockNum;
        uint32_t startOffset = blockIdx * lenPerBlock;
        uint32_t realLen = lenPerBlock;
        if (startOffset >= totalLen) {
            return;
        }
        if (startOffset + realLen > totalLen) {
            realLen = totalLen - startOffset;
        }

        pipe->InitBuffer(inQueue, BUFFER_NUM, TILE_LEN * sizeof(cT));
        pipe->InitBuffer(outQueue, BUFFER_NUM, TILE_LEN * sizeof(cT));
        pipe->InitBuffer(tmpBuf, TILE_LEN * sizeof(cT));

        uint32_t processed = 0;
        while (processed < realLen) {
            uint32_t curLen = realLen - processed;
            if (curLen > TILE_LEN) {
                curLen = TILE_LEN;
            }

            LocalTensor<cT> inLocal = inQueue.AllocTensor<cT>();
            DataCopy(inLocal, workspaceGm[startOffset + processed], curLen);
            inQueue.EnQue(inLocal);

            LocalTensor<cT> xLocal = inQueue.DeQue<cT>();
            LocalTensor<cT> outLocal = outQueue.AllocTensor<cT>();
            LocalTensor<cT> tmpLocal = tmpBuf.Get<cT>();

            // sigmoid(x)
            Muls(tmpLocal, xLocal, (cT)(-1.0f), curLen);
            Exp(tmpLocal, tmpLocal, curLen);
            Adds(tmpLocal, tmpLocal, (cT)(1.0f), curLen);
            Duplicate(outLocal, (cT)(1.0f), curLen);
            Div(tmpLocal, outLocal, tmpLocal, curLen);

            // x * sigmoid(x) * scaling_factor
            Mul(outLocal, xLocal, tmpLocal, curLen);
            Muls(outLocal, outLocal, (cT)scalingFactor, curLen);

            outQueue.EnQue<cT>(outLocal);
            LocalTensor<cT> yLocal = outQueue.DeQue<cT>();
            DataCopy(yGm[startOffset + processed], yLocal, curLen);

            inQueue.FreeTensor(xLocal);
            outQueue.FreeTensor(yLocal);

            processed += curLen;
        }
    }

    matmul::Matmul<matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, aT>,
                   matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, bT, true>,
                   matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, cT>,
                   matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, biasT>>
        matmulObj;

private:
    GlobalTensor<aT> xGm;
    GlobalTensor<bT> weightGm;
    GlobalTensor<biasT> biasGm;
    GlobalTensor<cT> yGm;
    GlobalTensor<cT> workspaceGm;

    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;

    TCubeTiling tiling;
    float scalingFactor;
    uint32_t M;
    uint32_t N;
    uint32_t K;
    AscendC::TPipe *pipe;
};

extern "C" __global__ __aicore__ void matmul_swish_scaling_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    AscendC::TPipe pipe;
    KernelMatmulSwishScaling<DTYPE_X, DTYPE_WEIGHT, DTYPE_Y, DTYPE_BIAS> op;

    GM_ADDR userWS = GetUserWorkspace(workspace);

    op.Init(x, weight, bias, userWS, y,
            tilingData.cubeTilingData, tilingData.scalingFactor,
            tilingData.M, tilingData.N, tilingData.K, &pipe);

    if ASCEND_IS_AIC {
        REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), op.matmulObj, &tilingData.cubeTilingData);
        op.matmulObj.SetTensorA(GlobalTensor<DTYPE_X>());
    }

    op.Process(&pipe);
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor matmul_swish_scaling_custom_impl_npu(const at::Tensor& x, const at::Tensor& weight,
                                                 const at::Tensor& bias, double scaling_factor) {
    auto sizes = x.sizes();
    int64_t M = sizes[0];
    int64_t N = weight.size(0);
    at::Tensor result = at::empty({M, N}, x.options());
    float sf = static_cast<float>(scaling_factor);
    EXEC_NPU_CMD(aclnnMatmulSwishScalingCustom, x, weight, bias, sf, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("matmul_swish_scaling_custom", &matmul_swish_scaling_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("matmul_swish_scaling_custom", &matmul_swish_scaling_custom_impl_npu,
          "matmul + swish + scaling");
}
