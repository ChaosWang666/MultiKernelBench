static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    auto platformInfo = context->GetPlatformInfo();
    auto ascendcPlatform = platform_ascendc::PlatformAscendC(platformInfo);

    auto xShape = context->GetInputShape(0)->GetOriginShape();
    auto wShape = context->GetInputShape(1)->GetOriginShape();
    int32_t M = xShape.GetDim(0);
    int32_t K = xShape.GetDim(1);
    int32_t N = wShape.GetDim(0);

    matmul_tiling::MultiCoreMatmulTiling mm(ascendcPlatform);
    mm.SetDim(ascendcPlatform.GetCoreNumAic());
    mm.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                matmul_tiling::DataType::DT_FLOAT, false);
    mm.SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                matmul_tiling::DataType::DT_FLOAT, true);
    mm.SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                matmul_tiling::DataType::DT_FLOAT);
    mm.SetBiasType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                   matmul_tiling::DataType::DT_FLOAT);
    mm.SetBias(true);
    mm.SetShape(M, N, K);
    mm.SetOrgShape(M, N, K);
    mm.SetBufferSpace(-1, -1, -1);

    MatmulMinSubtractCustomTilingData tiling;
    if (mm.GetTiling(tiling.cubeTilingData) == -1) {
        return ge::GRAPH_FAILED;
    }

    context->SetBlockDim(ascendcPlatform.GetCoreNumAic());
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(),
                        context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t *currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = 16 * 1024 * 1024;
    return ge::GRAPH_SUCCESS;
}
}

namespace ge {
static ge::graphStatus InferShape(gert::InferShapeContext* context)
{
    const gert::Shape* xShape = context->GetInputShape(0);
    const gert::Shape* wShape = context->GetInputShape(1);
    gert::Shape* yShape = context->GetOutputShape(0);
    yShape->SetDimNum(2);
    yShape->SetDim(0, xShape->GetDim(0));
    yShape->SetDim(1, wShape->GetDim(0));
    return GRAPH_SUCCESS;
}

{
    const auto inputDataType = context->GetInputDataType(0);
    context->SetOutputDataType(0, inputDataType);
    return ge::GRAPH_SUCCESS;
}

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;

typedef matmul::MatmulType<TPosition::GM, CubeFormat::ND, float> aType;
typedef matmul::MatmulType<TPosition::GM, CubeFormat::ND, float> bType;
typedef matmul::MatmulType<TPosition::GM, CubeFormat::ND, float> cType;
typedef matmul::MatmulType<TPosition::GM, CubeFormat::ND, float> biasType;

class KernelMatmulMinSubtract {
public:
    __aicore__ inline KernelMatmulMinSubtract() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR bias, GM_ADDR y,
                                const TCubeTiling& tiling, TPipe* pipe)
    {
        xGm.SetGlobalBuffer((__gm__ float*)x);
        wGm.SetGlobalBuffer((__gm__ float*)w);
        biasGm.SetGlobalBuffer((__gm__ float*)bias);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        mm.SetSubBlockIdx(0);
        mm.Init(&tiling, pipe);
    }

    __aicore__ inline void Process()
    {
        mm.SetTensorA(xGm);
        mm.SetTensorB(wGm, true);
        mm.SetBias(biasGm);
        mm.IterateAll(yGm);
        mm.End();
    }

    matmul::Matmul<aType, bType, cType, biasType> mm;

private:
    GlobalTensor<float> xGm;
    GlobalTensor<float> wGm;
    GlobalTensor<float> biasGm;
    GlobalTensor<float> yGm;
};

extern "C" __global__ __aicore__ void matmul_min_subtract_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    TPipe pipe;
    KernelMatmulMinSubtract op;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), op.mm, &tiling_data.cubeTilingData);
    op.Init(x, weight, bias, y, tiling_data.cubeTilingData, &pipe);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, constant):
        super(ModelNew, self).__init__()
        self.linear = torch.nn.Linear(in_features, out_features)
        self.constant = torch.nn.Parameter(torch.tensor(constant))

    def forward(self, x):
        x = custom_ops_lib.matmul_min_subtract_custom(x, self.linear.weight, self.linear.bias)
        x = torch.min(x, self.constant)
        x = x - self.constant
        return x
