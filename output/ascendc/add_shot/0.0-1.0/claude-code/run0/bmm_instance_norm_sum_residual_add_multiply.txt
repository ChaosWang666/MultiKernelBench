kernel_src="""
#include "kernel_operator.h"

class KernelBmmInstanceNorm {
public:
    __aicore__ inline KernelBmmInstanceNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                uint32_t batchSize, uint32_t features,
                                float eps, float invFeatures)
    {
        this->features = features;
        this->eps = eps;
        this->invFeatures = invFeatures;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->rowsPerBlock = batchSize / blockNum;

        uint32_t rowOffset = blockIdx * this->rowsPerBlock * features;
        uint32_t totalSize = this->rowsPerBlock * features;

        xGm.SetGlobalBuffer((__gm__ float *)x + rowOffset, totalSize);
        yGm.SetGlobalBuffer((__gm__ float *)y + rowOffset, totalSize);
        zGm.SetGlobalBuffer((__gm__ float *)z + rowOffset, totalSize);

        pipe.InitBuffer(inQueueX, 1, features * sizeof(float));
        pipe.InitBuffer(inQueueY, 1, features * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, features * sizeof(float));
        pipe.InitBuffer(tmpBuf, features * sizeof(float));
        pipe.InitBuffer(workBuf, 2048);
        pipe.InitBuffer(redBuf, 64);
        pipe.InitBuffer(scalarBuf, 64);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsPerBlock; ++i) {
            ProcessRow(i);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * this->features], this->features);
        inQueueX.EnQue(xLocal);

        AscendC::LocalTensor<float> yLocal = inQueueY.AllocTensor<float>();
        AscendC::DataCopy(yLocal, yGm[rowIdx * this->features], this->features);
        inQueueY.EnQue(yLocal);

        xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> red = redBuf.Get<float>();
        AscendC::LocalTensor<float> scalar = scalarBuf.Get<float>();

        AscendC::ReduceSum<float>(red, xLocal, work, this->features);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float sumVal = red.GetValue(0);
        float mean = sumVal * this->invFeatures;

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::Adds(tmp, xLocal, -mean, this->features);
        AscendC::Mul(xLocal, tmp, tmp, this->features);
        AscendC::ReduceSum<float>(red, xLocal, work, this->features);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float sumSq = red.GetValue(0);
        float var = sumSq * this->invFeatures;
        float varPlusEps = var + this->eps;

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::Duplicate(scalar, varPlusEps, 8);
        AscendC::Sqrt(scalar, scalar, 8);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float stdVal = scalar.GetValue(0);
        float invStd = 1.0f / stdVal;

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::Muls(tmp, tmp, invStd, this->features);

        yLocal = inQueueY.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::Add(zLocal, tmp, yLocal, this->features);
        AscendC::Mul(zLocal, zLocal, yLocal, this->features);

        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);

        outQueueZ.EnQue<float>(zLocal);
        zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[rowIdx * this->features], zLocal, this->features);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, workBuf, redBuf, scalarBuf;
    AscendC::GlobalTensor<float> xGm, yGm, zGm;
    uint32_t features;
    uint32_t rowsPerBlock;
    float eps;
    float invFeatures;
};

extern "C" __global__ __aicore__ void bmm_instance_norm_sum_residual_add_multiply_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelBmmInstanceNorm op;
    op.Init(x, y, z, tiling_data.batchSize, tiling_data.features, tiling_data.eps, tiling_data.invFeatures);
    op.Process();
}
