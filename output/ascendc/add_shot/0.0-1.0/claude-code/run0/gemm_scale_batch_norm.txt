kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmScaleBN {
public:
    __aicore__ inline KernelGemmScaleBN() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t numFeatures)
    {
        this->numFeatures = numFeatures;
        this->rowsPerBlock = batchSize / AscendC::GetBlockNum();

        uint32_t blockOffset = this->rowsPerBlock * numFeatures * AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockOffset, this->rowsPerBlock * numFeatures);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + blockOffset, this->rowsPerBlock * numFeatures);
        scaleGm.SetGlobalBuffer((__gm__ DTYPE_SCALE *)scale, numFeatures);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, numFeatures);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, numFeatures * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, numFeatures * sizeof(DTYPE_Z));
        pipe.InitBuffer(scaleBuf, numFeatures * sizeof(DTYPE_SCALE));
        pipe.InitBuffer(biasBuf, numFeatures * sizeof(DTYPE_BIAS));

        AscendC::LocalTensor<DTYPE_SCALE> scaleLocal = scaleBuf.Get<DTYPE_SCALE>();
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();
        AscendC::DataCopy(scaleLocal, scaleGm, numFeatures);
        AscendC::DataCopy(biasLocal, biasGm, numFeatures);
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->rowsPerBlock;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->numFeatures], this->numFeatures);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_SCALE> scaleLocal = scaleBuf.Get<DTYPE_SCALE>();
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();

        AscendC::Mul(zLocal, xLocal, scaleLocal, this->numFeatures);
        AscendC::Add(zLocal, zLocal, biasLocal, this->numFeatures);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * this->numFeatures], zLocal, this->numFeatures);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECIN> scaleBuf, biasBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    AscendC::GlobalTensor<DTYPE_SCALE> scaleGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    uint32_t numFeatures;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void gemm_scale_batch_norm_custom(GM_ADDR x, GM_ADDR scale, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmScaleBN op;
    op.Init(x, scale, bias, z, tiling_data.batchSize, tiling_data.numFeatures);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, scale_shape, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.scale = nn.Parameter(torch.randn(scale_shape))
        self.bn = nn.BatchNorm1d(out_features, eps=eps, momentum=momentum)

    def forward(self, x):
        x = self.gemm(x)
        if self.training:
            x = x * self.scale
            x = self.bn(x)
            return x
        mean = self.bn.running_mean
        var = self.bn.running_var
        inv_std = torch.rsqrt(var + self.bn.eps)
        combined_scale = (self.scale * self.bn.weight * inv_std).contiguous()
        combined_bias = (self.bn.bias - mean * self.bn.weight * inv_std).contiguous()
        x = x.contiguous()
        return custom_ops_lib.gemm_scale_batch_norm_custom(x, combined_scale, combined_bias)
'''
