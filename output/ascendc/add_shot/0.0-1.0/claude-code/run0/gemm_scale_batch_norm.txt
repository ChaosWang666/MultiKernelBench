kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmScaleBatchNorm {
public:
    __aicore__ inline KernelGemmScaleBatchNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR y, uint32_t totalRows, uint32_t cols, uint32_t tileNum)
    {
        this->cols = cols;
        this->blockRows = totalRows / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileRows = this->blockRows / tileNum / BUFFER_NUM;
        this->tileElems = this->tileRows * cols;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + AscendC::GetBlockIdx() * this->blockRows * cols, this->blockRows * cols);
        scaleGm.SetGlobalBuffer((__gm__ DTYPE_SCALE *)scale, cols);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + AscendC::GetBlockIdx() * this->blockRows * cols, this->blockRows * cols);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileElems * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueScale, 1, cols * sizeof(DTYPE_SCALE));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileElems * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_SCALE> scaleLocal = inQueueScale.AllocTensor<DTYPE_SCALE>();
        AscendC::DataCopy(scaleLocal, scaleGm, this->cols);
        inQueueScale.EnQue(scaleLocal);
        scaleLocal = inQueueScale.DeQue<DTYPE_SCALE>();

        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i, scaleLocal);
            CopyOut(i);
        }

        inQueueScale.FreeTensor(scaleLocal);
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileElems], this->tileElems);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t progress, AscendC::LocalTensor<DTYPE_SCALE> scaleLocal)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        for (uint32_t r = 0; r < this->tileRows; r++) {
            AscendC::Mul(yLocal[r * this->cols], xLocal[r * this->cols], scaleLocal, this->cols);
        }
        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[progress * this->tileElems], yLocal, this->tileElems);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueScale;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_SCALE> scaleGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t blockRows;
    uint32_t cols;
    uint32_t tileNum;
    uint32_t tileRows;
    uint32_t tileElems;
};

extern "C" __global__ __aicore__ void gemm_scale_batch_norm_custom(GM_ADDR x, GM_ADDR scale, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmScaleBatchNorm op;
    op.Init(x, scale, y, tiling_data.totalRows, tiling_data.cols, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, scale_shape, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.scale = nn.Parameter(torch.randn(scale_shape))
        self.bn = nn.BatchNorm1d(out_features, eps=eps, momentum=momentum)

    def forward(self, x):
        x = self.gemm(x)
        x = custom_ops_lib.gemm_scale_batch_norm_custom(x.contiguous(), self.scale.contiguous())
        x = self.bn(x)
        return x
