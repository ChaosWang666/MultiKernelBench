kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv2dGeluGlobalAvgPool {
public:
    __aicore__ inline KernelConv2dGeluGlobalAvgPool() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalPairs, uint32_t hwSize,
                                float hwSizeInv, uint32_t tileLength,
                                uint32_t pairsPerCore)
    {
        this->hwSize = hwSize;
        this->hwSizeInv = hwSizeInv;
        this->tileLength = tileLength;
        this->pairsPerCore = pairsPerCore;

        uint32_t coreIdx = AscendC::GetBlockIdx();
        uint32_t startPair = coreIdx * pairsPerCore;

        xGm.SetGlobalBuffer((__gm__ float*)x + startPair * hwSize, pairsPerCore * hwSize);
        yGm.SetGlobalBuffer((__gm__ float*)y + startPair, pairsPerCore);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, tileLength * sizeof(float));
        pipe.InitBuffer(reduceBuf, tileLength * sizeof(float));

        uint32_t outSizeBytes = ((pairsPerCore * sizeof(float) + 31) / 32) * 32;
        pipe.InitBuffer(outBuf, outSizeBytes);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        AscendC::Duplicate(outLocal, 0.0f, pairsPerCore);

        for (uint32_t p = 0; p < this->pairsPerCore; p++) {
            float acc = ProcessOnePair(p);
            outLocal.SetValue(static_cast<int32_t>(p), acc * this->hwSizeInv);
        }

        AscendC::DataCopyExtParams outParams;
        outParams.blockCount = 1;
        outParams.blockLen = static_cast<uint32_t>(this->pairsPerCore * sizeof(float));
        outParams.srcStride = 0;
        outParams.dstStride = 0;
        AscendC::DataCopyPad(yGm, outLocal, outParams);
    }

private:
    __aicore__ inline float ProcessOnePair(uint32_t p)
    {
        float acc = 0.0f;
        uint32_t baseOffset = p * this->hwSize;
        uint32_t numTiles = (this->hwSize + this->tileLength - 1) / this->tileLength;

        for (uint32_t t = 0; t < numTiles; t++) {
            uint32_t tileOff = t * this->tileLength;
            uint32_t curLen;
            if (tileOff + this->tileLength <= this->hwSize) {
                curLen = this->tileLength;
            } else {
                curLen = this->hwSize - tileOff;
            }
            CopyIn(baseOffset + tileOff, curLen);
            float partial = Compute(curLen);
            acc += partial;
        }
        return acc;
    }

    __aicore__ inline void CopyIn(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();

        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = static_cast<uint32_t>(len * sizeof(float));
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        AscendC::DataCopyPadExtParams<float> padParams{true, 0, 0, 0.0f};

        AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline float Compute(uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        AscendC::Muls(tmpLocal, xLocal, (float)0.70710678118654752f, len);
        AscendC::Erf(tmpLocal, tmpLocal, len);
        AscendC::Adds(tmpLocal, tmpLocal, (float)1.0f, len);
        AscendC::Mul(tmpLocal, xLocal, tmpLocal, len);
        AscendC::Muls(tmpLocal, tmpLocal, (float)0.5f, len);

        AscendC::ReduceSum<float>(reduceLocal, tmpLocal, reduceLocal, static_cast<int32_t>(len));

        float partial = reduceLocal.GetValue(0);

        inQueueX.FreeTensor(xLocal);
        return partial;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECOUT> outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t hwSize;
    float hwSizeInv;
    uint32_t tileLength;
    uint32_t pairsPerCore;
};

extern "C" __global__ __aicore__ void conv2d_gelu_global_avg_pool_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dGeluGlobalAvgPool op;
    op.Init(x, y,
            tiling_data.totalPairs,
            tiling_data.hwSize,
            tiling_data.hwSizeInv,
            tiling_data.tileLength,
            tiling_data.pairsPerCore);
    op.Process();
}
