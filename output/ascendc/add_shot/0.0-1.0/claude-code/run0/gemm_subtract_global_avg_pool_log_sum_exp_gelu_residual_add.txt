#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedOp {
public:
    __aicore__ inline KernelFusedOp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                uint32_t M, uint32_t N, uint32_t tileLength)
    {
        this->N = N;
        this->tileLength = tileLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->rowsPerBlock = M / blockNum;
        uint32_t rowStart = blockIdx * this->rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + rowStart, this->rowsPerBlock);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + rowStart * N, this->rowsPerBlock * N);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + rowStart * N, this->rowsPerBlock * N);

        pipe.InitBuffer(inQueueY, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        uint32_t tilesPerRow = N / tileLength;
        for (uint32_t row = 0; row < rowsPerBlock; row++) {
            DTYPE_X scalar = xGm.GetValue(row);
            for (uint32_t t = 0; t < tilesPerRow; t++) {
                CopyIn(row, t);
                Compute(scalar);
                CopyOut(row, t);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row, uint32_t t)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.AllocTensor<DTYPE_Y>();
        AscendC::DataCopy(yLocal, yGm[row * N + t * tileLength], tileLength);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void Compute(DTYPE_X scalar)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::Adds(zLocal, yLocal, scalar, tileLength);
        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row, uint32_t t)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[row * N + t * tileLength], zLocal, tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t N;
    uint32_t tileLength;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void gemm_subtract_global_avg_pool_log_sum_exp_gelu_residual_add_custom(

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, bias=True):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features, bias=bias)
        self.subtract = nn.Parameter(torch.randn(out_features))

    def forward(self, x):
        original_x = x.clone().detach().contiguous()
        y = self.gemm(x)
        y = y - self.subtract
        y = torch.mean(y, dim=1, keepdim=True)
        y = torch.logsumexp(y, dim=1, keepdim=True)
        y = torch.nn.functional.gelu(y)
        y_flat = y.reshape(-1).contiguous()
