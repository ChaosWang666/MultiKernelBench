        blockDim = N;

    while (blockDim > 1 && (N % blockDim) != 0) {
        blockDim--;

    if (blockDim == 0) {
        blockDim = 1;

    tiling.set_hw(HW);

kernel_src="""
#include "kernel_operator.h"

class KernelConv2dMinTanhTanh {
public:
    __aicore__ inline KernelConv2dMinTanhTanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                 uint32_t channels, uint32_t hw,
                                 uint32_t batchesPerBlock, uint32_t tileLength)
    {
        this->channels = channels;
        this->hw = hw;
        this->batchesPerBlock = batchesPerBlock;
        this->tileLength = tileLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t batchStart = static_cast<uint64_t>(blockIdx) * static_cast<uint64_t>(batchesPerBlock);

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + batchStart * static_cast<uint64_t>(channels) * static_cast<uint64_t>(hw));
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + batchStart * static_cast<uint64_t>(hw));

        pipe.InitBuffer(bufMin, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(bufCur, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(bufT1, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(bufT2, tileLength * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t b = 0; b < this->batchesPerBlock; ++b) {
            uint32_t pos = 0;
            while (pos < this->hw) {
                uint32_t len = this->tileLength;
                if (pos + len > this->hw) {
                    len = this->hw - pos;
                }
                ProcessTile(b, pos, len);
                pos += len;
            }
        }
    }

private:
    __aicore__ inline void ProcessTile(uint32_t b, uint32_t sOffset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> minLocal = bufMin.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> curLocal = bufCur.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> t1Local = bufT1.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> t2Local = bufT2.Get<DTYPE_X>();

        uint64_t xBatchOffset = static_cast<uint64_t>(b) * static_cast<uint64_t>(this->channels) * static_cast<uint64_t>(this->hw);
        uint64_t zBatchOffset = static_cast<uint64_t>(b) * static_cast<uint64_t>(this->hw);

        LoadTile(minLocal, xBatchOffset + static_cast<uint64_t>(sOffset), len);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t c = 1; c < this->channels; ++c) {
            uint64_t srcOff = xBatchOffset + static_cast<uint64_t>(c) * static_cast<uint64_t>(this->hw) + static_cast<uint64_t>(sOffset);
            LoadTile(curLocal, srcOff, len);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Min(minLocal, minLocal, curLocal, len);
            AscendC::PipeBarrier<PIPE_ALL>();
        }

        ComputeTanh(minLocal, minLocal, t1Local, t2Local, len);
        AscendC::PipeBarrier<PIPE_ALL>();
        ComputeTanh(minLocal, minLocal, t1Local, t2Local, len);
        AscendC::PipeBarrier<PIPE_ALL>();

        StoreTile(minLocal, zBatchOffset + static_cast<uint64_t>(sOffset), len);
        AscendC::PipeBarrier<PIPE_ALL>();
    }

    __aicore__ inline void LoadTile(AscendC::LocalTensor<DTYPE_X> dst, uint64_t offset, uint32_t len)
    {
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = len * static_cast<uint32_t>(sizeof(DTYPE_X));
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPadExtParams<DTYPE_X> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;
        AscendC::DataCopyPad(dst, xGm[offset], copyParams, padParams);
    }

    __aicore__ inline void StoreTile(AscendC::LocalTensor<DTYPE_Z> src, uint64_t offset, uint32_t len)
    {
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = len * static_cast<uint32_t>(sizeof(DTYPE_Z));
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPad(zGm[offset], src, copyParams);
    }

    __aicore__ inline void ComputeTanh(AscendC::LocalTensor<DTYPE_X> dst,
                                        AscendC::LocalTensor<DTYPE_X> src,
                                        AscendC::LocalTensor<DTYPE_X> t1,
                                        AscendC::LocalTensor<DTYPE_X> t2,
                                        uint32_t len)
    {
        AscendC::Muls(t1, src, static_cast<DTYPE_X>(2.0f), len);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Exp(t1, t1, len);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Adds(t2, t1, static_cast<DTYPE_X>(1.0f), len);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Adds(t1, t1, static_cast<DTYPE_X>(-1.0f), len);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Div(dst, t1, t2, len);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bufMin;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bufCur;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bufT1;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bufT2;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t channels;
    uint32_t hw;
    uint32_t batchesPerBlock;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv2d_min_tanh_tanh_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dMinTanhTanh op;
    op.Init(x, z, tiling_data.channels, tiling_data.hw, tiling_data.batchesPerBlock, tiling_data.tileLength);
    op.Process();
}
