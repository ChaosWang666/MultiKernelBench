kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using AscendC::GlobalTensor;
using AscendC::LocalTensor;
using AscendC::TPipe;
using AscendC::TQue;

constexpr int32_t BUFFER_NUM = 2;

class KernelMmFused {
public:
    __aicore__ inline KernelMmFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                GM_ADDR workspace, const TCubeTiling &mmTiling,
                                uint32_t totalOutSize, float subVal, float mulVal) {
        this->totalOutSize = totalOutSize;
        this->subVal = subVal;
        this->mulVal = mulVal;
        this->mmTiling = mmTiling;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        wGm.SetGlobalBuffer((__gm__ float*)weight);
        biasGm.SetGlobalBuffer((__gm__ float*)bias);
        yGm.SetGlobalBuffer((__gm__ float*)y);
    }

    using AType = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float>;
    using BType = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float, true>;
    using CType = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float>;
    using BiasType = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float>;
    matmul::Matmul<AType, BType, CType, BiasType> mmObj;

    __aicore__ inline void ProcessMatmul(TPipe *pipe) {
        REGIST_MATMUL_OBJ(pipe, GetSysWorkSpacePtr(), mmObj, &mmTiling);
        mmObj.SetTensorA(xGm);
        mmObj.SetTensorB(wGm);
        mmObj.SetBias(biasGm);
        mmObj.IterateAll(yGm);
        mmObj.End();
    }

    __aicore__ inline void ProcessPost(TPipe *pipe) {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t perBlock = totalOutSize / blockNum;
        uint32_t offset = blockIdx * perBlock;

        constexpr uint32_t TILE = 4096;
        uint32_t numTiles = perBlock / TILE;
        uint32_t tailLen = perBlock - numTiles * TILE;

        TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQ;
        TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQ;
        pipe->InitBuffer(inQ, BUFFER_NUM, TILE * sizeof(float));
        pipe->InitBuffer(outQ, BUFFER_NUM, TILE * sizeof(float));

        for (uint32_t i = 0; i < numTiles; i++) {
            uint32_t tileOffset = offset + i * TILE;
            LocalTensor<float> inL = inQ.AllocTensor<float>();
            AscendC::DataCopy(inL, yGm[tileOffset], TILE);
            inQ.EnQue(inL);

            LocalTensor<float> inL2 = inQ.DeQue<float>();
            LocalTensor<float> outL = outQ.AllocTensor<float>();
            AscendC::Adds(outL, inL2, -subVal, TILE);
            AscendC::Muls(outL, outL, mulVal, TILE);
            AscendC::Relu(outL, outL, TILE);
            outQ.EnQue(outL);
            inQ.FreeTensor(inL2);

            LocalTensor<float> outL2 = outQ.DeQue<float>();
            AscendC::DataCopy(yGm[tileOffset], outL2, TILE);
            outQ.FreeTensor(outL2);
        }
    }

private:
    TCubeTiling mmTiling;
    uint32_t totalOutSize;
    float subVal;
    float mulVal;
    GlobalTensor<float> xGm;
    GlobalTensor<float> wGm;
    GlobalTensor<float> biasGm;
    GlobalTensor<float> yGm;
};

extern "C" __global__ __aicore__ void matmul_subtract_multiply_relu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    TPipe pipe;
    KernelMmFused op;
    op.Init(x, weight, bias, y, workspace,
            tiling_data.mmTiling, tiling_data.totalOutSize,
            tiling_data.subVal, tiling_data.mulVal);

    if ASCEND_IS_AIC {
        op.ProcessMatmul(&pipe);
    }
    AscendC::SyncAll();
    if ASCEND_IS_AIV {
        op.ProcessPost(&pipe);
    }
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor matmul_subtract_multiply_relu_custom_impl_npu(
    const at::Tensor& x, const at::Tensor& weight, const at::Tensor& bias,
    double subtract_value, double multiply_value) {
    int64_t M = x.size(0);
    int64_t N = weight.size(0);
    at::Tensor result = at::empty({M, N}, x.options());
    float subVal = static_cast<float>(subtract_value);
    float mulVal = static_cast<float>(multiply_value);
    EXEC_NPU_CMD(aclnnMatmulSubtractMultiplyReluCustom,
                 x, weight, bias, subVal, mulVal, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("matmul_subtract_multiply_relu_custom", &matmul_subtract_multiply_relu_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("matmul_subtract_multiply_relu_custom",
          &matmul_subtract_multiply_relu_custom_impl_npu,
          "relu((x @ weight.T + bias - sub) * mul)");
