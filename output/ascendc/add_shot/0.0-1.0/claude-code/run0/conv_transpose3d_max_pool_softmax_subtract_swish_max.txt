kernel_src = """
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR sub, GM_ADDR y,
                                 uint32_t totalRows, uint32_t channels, uint32_t tileNum)
    {
        this->channels = channels;
        this->tileNum = tileNum;
        uint32_t blockRows = totalRows / AscendC::GetBlockNum();
        this->blockRows = blockRows;
        this->tileRows = blockRows / tileNum / BUFFER_NUM;
        this->tileElems = this->tileRows * channels;

        uint32_t rowOffset = AscendC::GetBlockIdx() * blockRows;
        xGm.SetGlobalBuffer((__gm__ float *)x + rowOffset * channels, blockRows * channels);
        yGm.SetGlobalBuffer((__gm__ float *)y + rowOffset, blockRows);
        subGm.SetGlobalBuffer((__gm__ float *)sub, channels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileElems * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileRows * sizeof(float));
        pipe.InitBuffer(subBuf, channels * sizeof(float));
        pipe.InitBuffer(workBuf, this->tileElems * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->tileElems * sizeof(float));

        auto subLocal = subBuf.Get<float>();
        AscendC::DataCopy(subLocal, subGm, channels);
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        auto xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileElems], this->tileElems);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        auto xLocal = inQueueX.DeQue<float>();
        auto outLocal = outQueueY.AllocTensor<float>();
        auto subLocal = subBuf.Get<float>();
        auto workLocal = workBuf.Get<float>();
        auto tmpLocal = tmpBuf.Get<float>();

        uint32_t C = this->channels;

        for (int32_t r = 0; r < this->tileRows; r++) {
            uint32_t base = r * C;

            AscendC::PipeBarrier<PIPE_V>();
            float maxv = xLocal.GetValue(base);
            for (uint32_t c = 1; c < C; c++) {
                float v = xLocal.GetValue(base + c);
                if (v > maxv) maxv = v;
            }

            AscendC::Adds(workLocal[base], xLocal[base], -maxv, C);
            AscendC::Exp(workLocal[base], workLocal[base], C);

            AscendC::PipeBarrier<PIPE_V>();
            float sumv = 0.0f;
            for (uint32_t c = 0; c < C; c++) {
                sumv += workLocal.GetValue(base + c);
            }
            float invSum = 1.0f / sumv;

            AscendC::Muls(workLocal[base], workLocal[base], invSum, C);
            AscendC::Sub(workLocal[base], workLocal[base], subLocal, C);

            AscendC::Muls(tmpLocal[base], workLocal[base], -1.0f, C);
            AscendC::Exp(tmpLocal[base], tmpLocal[base], C);
            AscendC::Adds(tmpLocal[base], tmpLocal[base], 1.0f, C);
            AscendC::Div(workLocal[base], workLocal[base], tmpLocal[base], C);

            AscendC::PipeBarrier<PIPE_V>();
            float outmax = workLocal.GetValue(base);
            for (uint32_t c = 1; c < C; c++) {
                float v = workLocal.GetValue(base + c);
                if (v > outmax) outmax = v;
            }
            outLocal.SetValue(r, outmax);
        }

        outQueueY.EnQue(outLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        auto outLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileRows], outLocal, this->tileRows);
        outQueueY.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> subBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> subGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t blockRows;
    uint32_t tileNum;
    uint32_t tileRows;
    uint32_t tileElems;
    uint32_t channels;
};

extern "C" __global__ __aicore__ void conv_transpose3d_max_pool_softmax_subtract_swish_max_custom(
    GM_ADDR x, GM_ADDR sub, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, sub, y, tiling_data.totalRows, tiling_data.channels, tiling_data.tileNum);
    op.Process();
}

model_src = '''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding,
                 pool_kernel_size, pool_stride, pool_padding):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size,
                                                 stride=stride, padding=padding,
                                                 output_padding=output_padding)
        self.max_pool = nn.MaxPool3d(kernel_size=pool_kernel_size, stride=pool_stride,
                                     padding=pool_padding)
        self.subtract = nn.Parameter(torch.randn(out_channels))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.max_pool(x)
        x = x.permute(0, 2, 3, 4, 1).contiguous()
        sub = self.subtract.contiguous()
        x = custom_ops_lib.conv_transpose3d_max_pool_softmax_subtract_swish_max_custom(x, sub)
        return x
