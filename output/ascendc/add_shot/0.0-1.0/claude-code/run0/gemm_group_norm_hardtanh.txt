kernel_src="""
#include "kernel_operator.h"

using namespace AscendC;

class KernelGemmGroupNormHardtanh {
public:
    __aicore__ inline KernelGemmGroupNormHardtanh() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b,
                                 GM_ADDR gn_w, GM_ADDR gn_b, GM_ADDR y,
                                 uint32_t M, uint32_t N, uint32_t K,
                                 uint32_t numGroups,
                                 float hardtanhMin, float hardtanhMax) {
        this->M = M;
        this->N = N;
        this->K = K;
        this->numGroups = numGroups;
        this->groupSize = N / numGroups;
        this->hardtanhMin = hardtanhMin;
        this->hardtanhMax = hardtanhMax;
        this->eps = 1e-5f;

        uint32_t blockIdx = GetBlockIdx();
        uint32_t blockNum = GetBlockNum();
        uint32_t rowsPer = (M + blockNum - 1) / blockNum;
        uint32_t start = blockIdx * rowsPer;
        uint32_t end = start + rowsPer;
        if (end > M) end = M;
        this->startRow = (start < M) ? start : M;
        this->rowsPerBlock = (end > this->startRow) ? (end - this->startRow) : 0;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        wGm.SetGlobalBuffer((__gm__ float*)w);
        bGm.SetGlobalBuffer((__gm__ float*)b);
        gnWGm.SetGlobalBuffer((__gm__ float*)gn_w);
        gnBGm.SetGlobalBuffer((__gm__ float*)gn_b);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        pipe.InitBuffer(xRowBuf, K * sizeof(float));
        pipe.InitBuffer(wRowBuf, K * sizeof(float));
        pipe.InitBuffer(mulBuf, K * sizeof(float));
        pipe.InitBuffer(outRowBuf, N * sizeof(float));
        pipe.InitBuffer(scratchBuf, N * sizeof(float));
        pipe.InitBuffer(reduceTmpBuf, 1024 * sizeof(float));
    }

    __aicore__ inline void Process() {
        if (rowsPerBlock == 0) return;

        LocalTensor<float> xRow = xRowBuf.Get<float>();
        LocalTensor<float> wRow = wRowBuf.Get<float>();
        LocalTensor<float> mulTmp = mulBuf.Get<float>();
        LocalTensor<float> outRow = outRowBuf.Get<float>();
        LocalTensor<float> scratch = scratchBuf.Get<float>();
        LocalTensor<float> reduceTmp = reduceTmpBuf.Get<float>();

        for (uint32_t i = 0; i < rowsPerBlock; ++i) {
            uint32_t row = startRow + i;

            DataCopy(xRow, xGm[row * K], K);
            PipeBarrier<PIPE_ALL>();

            // GEMM: compute N dot products for this row
            for (uint32_t j = 0; j < N; ++j) {
                DataCopy(wRow, wGm[j * K], K);
                PipeBarrier<PIPE_ALL>();
                Mul(mulTmp, xRow, wRow, K);
                PipeBarrier<PIPE_V>();
                ReduceSum<float>(reduceTmp, mulTmp, reduceTmp, K);
                PipeBarrier<PIPE_V>();
                float val = reduceTmp.GetValue(0);
                outRow.SetValue(j, val);
            }
            PipeBarrier<PIPE_ALL>();

            // Add bias
            DataCopy(scratch, bGm, N);
            PipeBarrier<PIPE_ALL>();
            Add(outRow, outRow, scratch, N);
            PipeBarrier<PIPE_V>();

            // Group Norm
            for (uint32_t g = 0; g < numGroups; ++g) {
                uint32_t gStart = g * groupSize;
                LocalTensor<float> groupSlice = outRow[gStart];

                ReduceSum<float>(reduceTmp, groupSlice, reduceTmp, groupSize);
                PipeBarrier<PIPE_V>();
                float mean = reduceTmp.GetValue(0) / (float)groupSize;

                Adds(mulTmp, groupSlice, -mean, groupSize);
                PipeBarrier<PIPE_V>();
                Mul(mulTmp, mulTmp, mulTmp, groupSize);
                PipeBarrier<PIPE_V>();
                ReduceSum<float>(reduceTmp, mulTmp, reduceTmp, groupSize);
                PipeBarrier<PIPE_V>();
                float var = reduceTmp.GetValue(0) / (float)groupSize;
                float invStd = 1.0f / sqrt(var + eps);

                Adds(groupSlice, groupSlice, -mean, groupSize);
                PipeBarrier<PIPE_V>();
                Muls(groupSlice, groupSlice, invStd, groupSize);
                PipeBarrier<PIPE_V>();
            }
            PipeBarrier<PIPE_V>();

            // scale by gn_w
            DataCopy(scratch, gnWGm, N);
            PipeBarrier<PIPE_ALL>();
            Mul(outRow, outRow, scratch, N);
            PipeBarrier<PIPE_V>();

            // add gn_b
            DataCopy(scratch, gnBGm, N);
            PipeBarrier<PIPE_ALL>();
            Add(outRow, outRow, scratch, N);
            PipeBarrier<PIPE_V>();

            // HardTanh: clamp to [hardtanhMin, hardtanhMax]
            Mins(outRow, outRow, hardtanhMax, N);
            PipeBarrier<PIPE_V>();
            Maxs(outRow, outRow, hardtanhMin, N);
            PipeBarrier<PIPE_V>();

            DataCopy(yGm[row * N], outRow, N);
            PipeBarrier<PIPE_ALL>();
        }
    }

private:
    TPipe pipe;
    TBuf<TPosition::VECCALC> xRowBuf, wRowBuf, mulBuf, outRowBuf, scratchBuf, reduceTmpBuf;
    GlobalTensor<float> xGm, wGm, bGm, gnWGm, gnBGm, yGm;
    uint32_t M, N, K, numGroups, groupSize, rowsPerBlock, startRow;
    float hardtanhMin, hardtanhMax, eps;
};

extern "C" __global__ __aicore__ void gemm_group_norm_hardtanh_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR gn_w, GM_ADDR gn_b, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormHardtanh op;
    op.Init(x, w, b, gn_w, gn_b, y,
            tiling_data.M, tiling_data.N, tiling_data.K,
            tiling_data.numGroups, tiling_data.hardtanhMin, tiling_data.hardtanhMax);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, num_groups, hardtanh_min, hardtanh_max):
        super().__init__()
        self.gemm = torch.nn.Linear(in_features, out_features)
        self.group_norm = torch.nn.GroupNorm(num_groups, out_features)
        self.num_groups = num_groups
        self.hardtanh_min = float(hardtanh_min)
        self.hardtanh_max = float(hardtanh_max)

    def forward(self, x):
        return custom_ops_lib.gemm_group_norm_hardtanh_custom(
            x.contiguous(),
            self.gemm.weight.contiguous(),
            self.gemm.bias.contiguous(),
            self.group_norm.weight.contiguous(),
            self.group_norm.bias.contiguous(),
            self.num_groups,
            self.hardtanh_min,
            self.hardtanh_max,
        )
