kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTransposeMinSumGeluAdd {
public:
    __aicore__ inline KernelConvTransposeMinSumGeluAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                uint32_t N, uint32_t C, uint32_t H, uint32_t W)
    {
        this->C = C;
        this->H = H;
        this->W = W;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t inOffset = (uint64_t)blockIdx * (uint64_t)C * (uint64_t)H * (uint64_t)W;
        uint64_t outOffset = (uint64_t)blockIdx * (uint64_t)W;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + inOffset, (uint64_t)C * H * W);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + outOffset, W);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, 8);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->W * sizeof(float));
        pipe.InitBuffer(inQueueBias, 1, 32);
        pipe.InitBuffer(outQueueY, 1, this->W * sizeof(float));

        pipe.InitBuffer(tmpMinBuf, this->W * sizeof(float));
        pipe.InitBuffer(tmpSumBuf, this->W * sizeof(float));
        pipe.InitBuffer(tmpGeluBuf, this->W * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
        AscendC::DataCopy(biasLocal, biasGm, 8);
        inQueueBias.EnQue(biasLocal);
        biasLocal = inQueueBias.DeQue<float>();
        float biasVal = biasLocal.GetValue(0);
        inQueueBias.FreeTensor(biasLocal);

        AscendC::LocalTensor<float> sumLocal = tmpSumBuf.Get<float>();
        AscendC::Duplicate(sumLocal, (float)0.0f, this->W);

        AscendC::LocalTensor<float> minLocal = tmpMinBuf.Get<float>();

        for (uint32_t h = 0; h < this->H; h++) {
            AscendC::Duplicate(minLocal, (float)1.0e30f, this->W);

            for (uint32_t c = 0; c < this->C; c++) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                uint32_t offset = c * this->H * this->W + h * this->W;
                AscendC::DataCopy(xLocal, xGm[offset], this->W);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();

                AscendC::Min(minLocal, minLocal, xLocal, this->W);

                inQueueX.FreeTensor(xLocal);
            }

            AscendC::Add(sumLocal, sumLocal, minLocal, this->W);
        }

        AscendC::LocalTensor<float> tmpGelu = tmpGeluBuf.Get<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        AscendC::Mul(tmpGelu, sumLocal, sumLocal, this->W);
        AscendC::Mul(tmpGelu, tmpGelu, sumLocal, this->W);
        AscendC::Muls(tmpGelu, tmpGelu, (float)0.044715f, this->W);
        AscendC::Add(tmpGelu, tmpGelu, sumLocal, this->W);
        AscendC::Muls(tmpGelu, tmpGelu, (float)0.7978845608028654f, this->W);
        AscendC::Tanh(tmpGelu, tmpGelu, this->W);
        AscendC::Adds(tmpGelu, tmpGelu, (float)1.0f, this->W);
        AscendC::Mul(tmpGelu, tmpGelu, sumLocal, this->W);
        AscendC::Muls(tmpGelu, tmpGelu, (float)0.5f, this->W);

        AscendC::Adds(yLocal, tmpGelu, biasVal, this->W);

        outQueueY.EnQue(yLocal);
        yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm, yLocal, this->W);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpMinBuf, tmpSumBuf, tmpGeluBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t C;
    uint32_t H;
    uint32_t W;
};

extern "C" __global__ __aicore__ void conv_transpose2d_min_sum_gelu_add_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTransposeMinSumGeluAdd op;
    op.Init(x, bias, y, tiling_data.N, tiling_data.C, tiling_data.H, tiling_data.W);
    op.Process();
}
