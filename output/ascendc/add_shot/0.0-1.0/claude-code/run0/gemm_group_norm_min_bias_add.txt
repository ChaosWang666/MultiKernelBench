constexpr int32_t BUFFER_NUM = 1;

class KernelGemmGroupNormMinBiasAdd {
public:
    __aicore__ inline KernelGemmGroupNormMinBiasAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR lin_bias,
                                GM_ADDR gn_weight, GM_ADDR gn_bias, GM_ADDR final_bias,
                                GM_ADDR z, uint32_t batchSize, uint32_t inFeatures,
                                uint32_t outFeatures, uint32_t numGroups, uint32_t groupSize)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->groupSize = groupSize;
        this->blockId = AscendC::GetBlockIdx();
        this->blockNum = AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inFeatures);
        weightGm.SetGlobalBuffer((__gm__ float*)weight, outFeatures * inFeatures);
        linBiasGm.SetGlobalBuffer((__gm__ float*)lin_bias, outFeatures);
        gnWeightGm.SetGlobalBuffer((__gm__ float*)gn_weight, outFeatures);
        gnBiasGm.SetGlobalBuffer((__gm__ float*)gn_bias, outFeatures);
        finalBiasGm.SetGlobalBuffer((__gm__ float*)final_bias, outFeatures);
        // Output shape: (1, outFeatures, batchSize, 1) -> total = outFeatures * batchSize
        zGm.SetGlobalBuffer((__gm__ float*)z, outFeatures * batchSize);
    }

    __aicore__ inline void Process()
    {
        // Distribute batches across blocks
        uint32_t batchesPerBlock = (batchSize + blockNum - 1) / blockNum;
        uint32_t batchStart = blockId * batchesPerBlock;
        uint32_t batchEnd = batchStart + batchesPerBlock;
        if (batchEnd > batchSize) batchEnd = batchSize;

        for (uint32_t b = batchStart; b < batchEnd; b++) {
            ProcessOneBatch(b);
        }
    }

private:
    __aicore__ inline void ProcessOneBatch(uint32_t b)
    {
        // Step 1+2+3: For each batch, compute gemm output, group-normalize, find min
        float minVal = 3.4e38f;

        for (uint32_t g = 0; g < numGroups; g++) {
            // Compute group-sized chunk of gemm output for batch b, group g
            // Group covers features [g*groupSize, (g+1)*groupSize)
            float groupVals[32];  // groupSize = 8192/512 = 16
            float sum = 0.0f;
            float sumSq = 0.0f;

            for (uint32_t k = 0; k < groupSize; k++) {
                uint32_t outIdx = g * groupSize + k;
                float acc = 0.0f;
                for (uint32_t i = 0; i < inFeatures; i++) {
                    acc += xGm.GetValue(b * inFeatures + i) *
                           weightGm.GetValue(outIdx * inFeatures + i);
                }
                acc += linBiasGm.GetValue(outIdx);
                groupVals[k] = acc;
                sum += acc;
                sumSq += acc * acc;
            }

            float mean = sum / (float)groupSize;
            float var = sumSq / (float)groupSize - mean * mean;
            float invStd = 1.0f / AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                              (float)AscendC::ScalarGetSquareRoot((float)(var + 1e-5f)));

            for (uint32_t k = 0; k < groupSize; k++) {
                uint32_t outIdx = g * groupSize + k;
                float normed = (groupVals[k] - mean) * invStd;
                normed = normed * gnWeightGm.GetValue(outIdx) + gnBiasGm.GetValue(outIdx);
                if (normed < minVal) minVal = normed;
            }
        }

        // Step 4: Output shape (1, outFeatures, batchSize, 1). For each outFeature i:
        // out[0, i, b, 0] = minVal + final_bias[i]
        for (uint32_t i = 0; i < outFeatures; i++) {
            float biasV = finalBiasGm.GetValue(i);
            zGm.SetValue(i * batchSize + b, minVal + biasV);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> linBiasGm;
    AscendC::GlobalTensor<float> gnWeightGm;
    AscendC::GlobalTensor<float> gnBiasGm;
    AscendC::GlobalTensor<float> finalBiasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t inFeatures;
    uint32_t outFeatures;
    uint32_t numGroups;
    uint32_t groupSize;
    uint32_t blockId;
    uint32_t blockNum;
};

extern "C" __global__ __aicore__ void gemm_group_norm_min_bias_add_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR lin_bias,
    GM_ADDR gn_weight, GM_ADDR gn_bias, GM_ADDR final_bias,
    GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{

            tiling_data.numGroups, tiling_data.groupSize);
    op.Process();
}
