kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;

template <typename T>
__aicore__ inline T CeilDiv(T a, T b) {
    return (a + b - 1) / b;
}

class KernelMatmulMishMish {
public:
    __aicore__ inline KernelMatmulMishMish() {}

    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR bias, GM_ADDR c,
                                GM_ADDR workspace, const TCubeTiling& tilingIn, TPipe* pipe)
    {
        this->tiling = tilingIn;

        int32_t offsetA = 0;
        int32_t offsetB = 0;
        int32_t offsetC = 0;
        int32_t offsetBias = 0;
        CalcOffset(GetBlockIdx(), tilingIn, offsetA, offsetB, offsetC, offsetBias);

        aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(a) + offsetA);
        bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(b) + offsetB);
        biasGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(bias) + offsetBias);
        cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ float*>(c) + offsetC);

        pipe->InitBuffer(outQueue_, 1, tilingIn.baseM * tilingIn.baseN * sizeof(float));
        pipe->InitBuffer(tmpBuf_, tilingIn.baseM * tilingIn.baseN * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t computeRound = 0;
        matmulObj.SetTensorA(aGlobal);
        matmulObj.SetTensorB(bGlobal);
        matmulObj.SetBias(biasGlobal);
        while (matmulObj.template Iterate<true>()) {
            MishMishCompute();
            CopyOut(computeRound);
            computeRound++;
        }
        matmulObj.End();
    }

    __aicore__ inline void MishMishCompute()
    {
        LocalTensor<float> mmOut = outQueue_.AllocTensor<float>();
        matmulObj.template GetTensorC<false>(mmOut, false, true);

        uint32_t size = static_cast<uint32_t>(tiling.baseM) * static_cast<uint32_t>(tiling.baseN);
        LocalTensor<float> tmp = tmpBuf_.Get<float>();

        ApplyMish(mmOut, tmp, size);
        ApplyMish(mmOut, tmp, size);

        outQueue_.EnQue<float>(mmOut);
    }

    __aicore__ inline void ApplyMish(LocalTensor<float>& x, LocalTensor<float>& tmp, uint32_t size)
    {
        Exp(tmp, x, size);
        Adds(tmp, tmp, static_cast<float>(1.0), size);
        Ln(tmp, tmp, size);
        Tanh(tmp, tmp, size);
        Mul(x, x, tmp, size);
    }

    __aicore__ inline void CopyOut(uint32_t count)
    {
        LocalTensor<float> mmOut = outQueue_.DeQue<float>();
        int32_t roundM = tiling.singleCoreM / tiling.baseM;
        int32_t startOffset = (static_cast<int32_t>(count) % roundM) * tiling.baseM * tiling.N +
                              (static_cast<int32_t>(count) / roundM) * tiling.baseN;
        DataCopyParams copyParam;
        copyParam.blockCount = static_cast<uint16_t>(tiling.baseM);
        copyParam.blockLen = static_cast<uint16_t>(tiling.baseN * sizeof(float) / 32);
        copyParam.srcStride = 0;
        copyParam.dstStride = static_cast<uint16_t>((tiling.N - tiling.baseN) * sizeof(float) / 32);
        DataCopy(cGlobal[startOffset], mmOut, copyParam);
        outQueue_.FreeTensor(mmOut);
    }

    __aicore__ inline void CalcOffset(int32_t blockIdx, const TCubeTiling& tilingIn,
                                      int32_t& offsetA, int32_t& offsetB,
                                      int32_t& offsetC, int32_t& offsetBias)
    {
        int32_t mSingleBlocks = CeilDiv<int32_t>(tilingIn.M, tilingIn.singleCoreM);
        int32_t mCoreIndx = blockIdx % mSingleBlocks;
        int32_t nCoreIndx = blockIdx / mSingleBlocks;
        offsetA = mCoreIndx * tilingIn.Ka * tilingIn.singleCoreM;
        offsetB = nCoreIndx * tilingIn.singleCoreN * tilingIn.Kb;
        offsetC = mCoreIndx * tilingIn.N * tilingIn.singleCoreM + nCoreIndx * tilingIn.singleCoreN;
        offsetBias = nCoreIndx * tilingIn.singleCoreN;
    }

    matmul::Matmul<matmul::MatmulType<TPosition::GM, CubeFormat::ND, float, false>,
                   matmul::MatmulType<TPosition::GM, CubeFormat::ND, float, true>,
                   matmul::MatmulType<TPosition::VECIN, CubeFormat::ND, float>,
                   matmul::MatmulType<TPosition::GM, CubeFormat::ND, float>>
        matmulObj;

private:
    GlobalTensor<float> aGlobal;
    GlobalTensor<float> bGlobal;
    GlobalTensor<float> biasGlobal;
    GlobalTensor<float> cGlobal;
    TQue<TPosition::VECOUT, 1> outQueue_;
    TBuf<TPosition::VECCALC> tmpBuf_;
    TCubeTiling tiling;
};

extern "C" __global__ __aicore__ void matmul_mish_mish_custom(
    GM_ADDR a, GM_ADDR b, GM_ADDR bias, GM_ADDR c, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulMishMish op;
    TPipe pipe;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), op.matmulObj, &tiling_data.cubeTilingData);
    op.Init(a, b, bias, c, workspace, tiling_data.cubeTilingData, &pipe);
    op.Process();
}
