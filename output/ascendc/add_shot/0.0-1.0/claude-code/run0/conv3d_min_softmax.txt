kernel_src="""
#include "kernel_operator.h"

class KernelConv3dMinSoftmax {
public:
    __aicore__ inline KernelConv3dMinSoftmax() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t N, uint32_t C, uint32_t D, uint32_t HW, uint32_t nPerBlock)
    {
        this->N = N;
        this->C = C;
        this->D = D;
        this->HW = HW;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t nStart = blockIdx * nPerBlock;
        uint32_t nEnd = (nStart + nPerBlock < N) ? (nStart + nPerBlock) : N;
        this->nStart = nStart;
        this->nCount = (nEnd > nStart) ? (nEnd - nStart) : 0;

        uint32_t inPerBatch = C * D * HW;
        uint32_t outPerBatch = C * HW;

        xGm.SetGlobalBuffer((__gm__ float *)x + nStart * inPerBatch, nCount * inPerBatch);
        yGm.SetGlobalBuffer((__gm__ float *)y + nStart * outPerBatch, nCount * outPerBatch);

        pipe.InitBuffer(inQueueX, 1, D * HW * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, C * HW * sizeof(float));
        pipe.InitBuffer(maxBuf, HW * sizeof(float));
        pipe.InitBuffer(sumBuf, HW * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t n = 0; n < nCount; n++) {
            ProcessOneBatch(n);
        }
    }

private:
    __aicore__ inline void ProcessOneBatch(uint32_t n)
    {
        AscendC::LocalTensor<float> minTensor = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> maxTensor = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumTensor = sumBuf.Get<float>();

        const float POS_INF = 3.4e38f;
        const float NEG_INF = -3.4e38f;

        AscendC::Duplicate<float>(minTensor, POS_INF, C * HW);

        for (uint32_t c = 0; c < C; c++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            uint32_t inOffset = n * C * D * HW + c * D * HW;
            AscendC::DataCopy(xLocal, xGm[inOffset], D * HW);
            inQueueX.EnQue(xLocal);

            AscendC::LocalTensor<float> xDq = inQueueX.DeQue<float>();
            for (uint32_t d = 0; d < D; d++) {
                AscendC::Min(minTensor[c * HW], minTensor[c * HW], xDq[d * HW], HW);
            }
            inQueueX.FreeTensor(xDq);
        }

        AscendC::Duplicate<float>(maxTensor, NEG_INF, HW);
        for (uint32_t c = 0; c < C; c++) {
            AscendC::Max(maxTensor, maxTensor, minTensor[c * HW], HW);
        }
        for (uint32_t c = 0; c < C; c++) {
            AscendC::Sub(minTensor[c * HW], minTensor[c * HW], maxTensor, HW);
        }
        AscendC::Exp(minTensor, minTensor, C * HW);
        AscendC::Duplicate<float>(sumTensor, 0.0f, HW);
        for (uint32_t c = 0; c < C; c++) {
            AscendC::Add(sumTensor, sumTensor, minTensor[c * HW], HW);
        }
        for (uint32_t c = 0; c < C; c++) {
            AscendC::Div(minTensor[c * HW], minTensor[c * HW], sumTensor, HW);
        }

        outQueueY.EnQue(minTensor);
        AscendC::LocalTensor<float> outDq = outQueueY.DeQue<float>();
        uint32_t outOffset = n * C * HW;
        AscendC::DataCopy(yGm[outOffset], outDq, C * HW);
        outQueueY.FreeTensor(outDq);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf, sumBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t N, C, D, HW;
    uint32_t nStart, nCount;
};

extern "C" __global__ __aicore__ void conv3d_min_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dMinSoftmax op;
    op.Init(x, y, tiling_data.N, tiling_data.C, tiling_data.D, tiling_data.HW, tiling_data.nPerBlock);
    op.Process();
}
