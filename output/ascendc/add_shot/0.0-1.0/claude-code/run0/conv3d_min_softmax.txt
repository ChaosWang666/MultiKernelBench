kernel_src="""
#include "kernel_operator.h"

class KernelConv3dMinSoftmax {
public:
    __aicore__ inline KernelConv3dMinSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                uint32_t B_, uint32_t C_, uint32_t D_,
                                uint32_t H_, uint32_t W_, uint32_t totalPos)
    {
        this->C = C_;
        this->D = D_;
        this->H = H_;
        this->W = W_;
        this->HW = H_ * W_;
        this->DHW = D_ * this->HW;
        this->CDHW = C_ * this->DHW;
        this->CHW = C_ * this->HW;

        int32_t blockId = static_cast<int32_t>(AscendC::GetBlockIdx());
        int32_t blockNum = static_cast<int32_t>(AscendC::GetBlockNum());
        int32_t total = static_cast<int32_t>(totalPos);
        int32_t per = (total + blockNum - 1) / blockNum;
        int32_t start = blockId * per;
        int32_t end = start + per;
        if (end > total) end = total;
        if (start > total) start = total;
        this->startPos = static_cast<uint32_t>(start);
        this->endPos = static_cast<uint32_t>(end);

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z);

        uint32_t alignedC = ((C_ + 7) / 8) * 8;
        if (alignedC < 8) alignedC = 8;
        pipe.InitBuffer(buf1, alignedC * sizeof(float));
        pipe.InitBuffer(buf2, alignedC * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> tA = buf1.Get<float>();
        AscendC::LocalTensor<float> tB = buf2.Get<float>();

        for (uint32_t p = startPos; p < endPos; p++) {
            uint32_t w_idx = p % W;
            uint32_t hw_div = p / W;
            uint32_t h_idx = hw_div % H;
            uint32_t b_idx = hw_div / H;

            uint32_t baseIn = b_idx * CDHW + h_idx * W + w_idx;
            uint32_t baseOut = b_idx * CHW + h_idx * W + w_idx;

            uint32_t baseC0 = baseIn;
            float m0 = xGm.GetValue(baseC0);
            for (uint32_t d = 1; d < D; d++) {
                float v = xGm.GetValue(baseC0 + d * HW);
                if (v < m0) m0 = v;
            }
            tA.SetValue(0, m0);
            float maxV = m0;

            for (uint32_t c = 1; c < C; c++) {
                uint32_t baseC = baseIn + c * DHW;
                float m = xGm.GetValue(baseC);
                for (uint32_t d = 1; d < D; d++) {
                    float v = xGm.GetValue(baseC + d * HW);
                    if (v < m) m = v;
                }
                tA.SetValue(c, m);
                if (m > maxV) maxV = m;
            }

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

            AscendC::Adds(tB, tA, -maxV, C);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Exp(tA, tB, C);

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

            float sum = 0.0f;
            for (uint32_t c = 0; c < C; c++) {
                sum += tA.GetValue(c);
            }
            float invSum = 1.0f / sum;

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

            AscendC::Muls(tB, tA, invSum, C);

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

            for (uint32_t c = 0; c < C; c++) {
                zGm.SetValue(baseOut + c * HW, tB.GetValue(c));
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> buf1, buf2;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t C, D, H, W;
    uint32_t HW, DHW, CDHW, CHW;
    uint32_t startPos, endPos;
};

extern "C" __global__ __aicore__ void conv3d_min_softmax_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dMinSoftmax op;
    op.Init(x, z, tiling_data.B, tiling_data.C, tiling_data.D, tiling_data.H, tiling_data.W, tiling_data.totalPos);
    op.Process();
}
