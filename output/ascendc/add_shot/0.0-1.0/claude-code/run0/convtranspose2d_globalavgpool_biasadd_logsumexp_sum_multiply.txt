kernel_src="""
#include "kernel_operator.h"

constexpr uint32_t TILE_LENGTH = 8192;
constexpr uint32_t MAX_CHANNELS_ALIGNED = 512;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t channels,
                                uint32_t spatialSize, float invSpatial)
    {
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->invSpatial = invSpatial;
        this->batchId = AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ float*)x + this->batchId * channels * spatialSize,
                            channels * spatialSize);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, channels);
        zGm.SetGlobalBuffer((__gm__ float*)z, batchSize);

        pipe.InitBuffer(xBuf, TILE_LENGTH * sizeof(float));
        pipe.InitBuffer(biasBuf, MAX_CHANNELS_ALIGNED * sizeof(float));
        pipe.InitBuffer(meansBuf, MAX_CHANNELS_ALIGNED * sizeof(float));
        pipe.InitBuffer(tmpBuf, 512 * sizeof(float));
        pipe.InitBuffer(outBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> meansLocal = meansBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();

        uint32_t channelsAligned = ((channels + 7) / 8) * 8;

        AscendC::DataCopy(biasLocal, biasGm, channelsAligned);
        AscendC::PipeBarrier<PIPE_ALL>();

        uint32_t fullTiles = spatialSize / TILE_LENGTH;
        uint32_t lastTile = spatialSize - fullTiles * TILE_LENGTH;
        uint32_t lastTileAligned = ((lastTile + 7) / 8) * 8;

        for (uint32_t c = 0; c < channels; c++) {
            float channelSum = 0.0f;
            for (uint32_t t = 0; t < fullTiles; t++) {
                AscendC::DataCopy(xLocal, xGm[c * spatialSize + t * TILE_LENGTH], TILE_LENGTH);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::ReduceSum<float>(tmpLocal, xLocal, tmpLocal[256], TILE_LENGTH);
                AscendC::PipeBarrier<PIPE_ALL>();
                channelSum += tmpLocal.GetValue(0);
            }
            if (lastTile > 0) {
                AscendC::DataCopyExtParams dcParams;
                dcParams.blockCount = 1;
                dcParams.blockLen = lastTile * sizeof(float);
                dcParams.srcStride = 0;
                dcParams.dstStride = 0;
                AscendC::DataCopyPadExtParams<float> padParams;
                padParams.isPad = true;
                padParams.leftPadding = 0;
                padParams.rightPadding = (uint8_t)(lastTileAligned - lastTile);
                padParams.paddingValue = 0.0f;
                AscendC::DataCopyPad(xLocal, xGm[c * spatialSize + fullTiles * TILE_LENGTH],
                                     dcParams, padParams);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::ReduceSum<float>(tmpLocal, xLocal, tmpLocal[256], lastTileAligned);
                AscendC::PipeBarrier<PIPE_ALL>();
                channelSum += tmpLocal.GetValue(0);
            }
            float meanVal = channelSum * invSpatial + biasLocal.GetValue(c);
            meansLocal.SetValue(c, meanVal);
        }

        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::ReduceMax<float>(tmpLocal, meansLocal, tmpLocal[256], channels);
        AscendC::PipeBarrier<PIPE_ALL>();
        float maxVal = tmpLocal.GetValue(0);

        AscendC::Adds<float>(meansLocal, meansLocal, -maxVal, channelsAligned);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Exp<float>(meansLocal, meansLocal, channelsAligned);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::ReduceSum<float>(tmpLocal, meansLocal, tmpLocal[256], channels);
        AscendC::PipeBarrier<PIPE_ALL>();
        float sumExp = tmpLocal.GetValue(0);

        outLocal.SetValue(0, sumExp);
        for (uint32_t i = 1; i < 8; i++) {
            outLocal.SetValue(i, 1.0f);
        }
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Ln<float>(outLocal, outLocal, 8);
        AscendC::PipeBarrier<PIPE_ALL>();
        float lnSum = outLocal.GetValue(0);

        float result = (maxVal + lnSum) * 10.0f;
        outLocal.SetValue(0, result);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::DataCopyExtParams outParams;
        outParams.blockCount = 1;
        outParams.blockLen = sizeof(float);
        outParams.srcStride = 0;
        outParams.dstStride = 0;
        AscendC::DataCopyPad(zGm[batchId], outLocal, outParams);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, biasBuf, meansBuf, tmpBuf, outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t batchId;
    float invSpatial;
};

extern "C" __global__ __aicore__ void convtranspose2d_globalavgpool_biasadd_logsumexp_sum_multiply_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, z, tiling_data.batchSize, tiling_data.channels,
            tiling_data.spatialSize, tiling_data.invSpatial);
    op.Process();
}
