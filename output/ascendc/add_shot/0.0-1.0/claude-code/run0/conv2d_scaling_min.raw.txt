kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv2dScalingMin {
public:
    __aicore__ inline KernelConv2dScalingMin() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t C, uint32_t HW, uint32_t tileLen, float scaleVal)
    {
        this->C = C;
        this->HW = HW;
        this->scale = scaleVal;
        this->tileLength = tileLen;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockIdx * C * HW, C * HW);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + blockIdx * HW, HW);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, 1, this->tileLength * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        uint32_t processed = 0;
        while (processed < HW) {
            uint32_t remain = HW - processed;
            uint32_t curLen = (remain < tileLength) ? remain : tileLength;
            ProcessTile(processed, curLen);
            processed += curLen;
        }
    }

private:
    __aicore__ inline void ProcessTile(uint32_t spatialOffset, uint32_t elementsInTile)
    {
        AscendC::DataCopyExtParams copyInParams;
        copyInParams.blockCount = 1;
        copyInParams.blockLen = elementsInTile * sizeof(DTYPE_X);
        copyInParams.srcStride = 0;
        copyInParams.dstStride = 0;

        AscendC::DataCopyPadExtParams<DTYPE_X> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;

        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        for (uint32_t c = 0; c < C; c++) {
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopyPad(xLocal, xGm[c * HW + spatialOffset], copyInParams, padParams);
            inQueueX.EnQue(xLocal);
            AscendC::LocalTensor<DTYPE_X> xIn = inQueueX.DeQue<DTYPE_X>();

            AscendC::Muls(xIn, xIn, scale, elementsInTile);
            if (c == 0) {
                AscendC::Adds(yLocal, xIn, (float)0.0f, elementsInTile);
            } else {
                AscendC::Min(yLocal, yLocal, xIn, elementsInTile);
            }
            inQueueX.FreeTensor(xIn);
        }

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        AscendC::LocalTensor<DTYPE_Y> yOut = outQueueY.DeQue<DTYPE_Y>();

        AscendC::DataCopyExtParams copyOutParams;
        copyOutParams.blockCount = 1;
        copyOutParams.blockLen = elementsInTile * sizeof(DTYPE_Y);
        copyOutParams.srcStride = 0;
        copyOutParams.dstStride = 0;
        AscendC::DataCopyPad(yGm[spatialOffset], yOut, copyOutParams);
        outQueueY.FreeTensor(yOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t C;
    uint32_t HW;
    uint32_t tileLength;
    float scale;
};

extern "C" __global__ __aicore__ void conv2d_scaling_min_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dScalingMin op;
    op.Init(x, y, tiling_data.C, tiling_data.HW, tiling_data.tileLength, tiling_data.scale);
    op.Process();
}

    auto sizes = x.sizes();
    std::vector<int64_t> out_sizes = {sizes[0], 1, sizes[2], sizes[3]};

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, scale_factor):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.scale_factor = scale_factor

    def forward(self, x):
        x = self.conv(x)
        x = custom_ops_lib.conv2d_scaling_min_custom(x, float(self.scale_factor))
