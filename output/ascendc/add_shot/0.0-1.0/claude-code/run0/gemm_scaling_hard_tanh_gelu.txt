kernel_src = """
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;
using namespace matmul;

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmScalingHardTanhGelu {
public:
    __aicore__ inline KernelGemmScalingHardTanhGelu() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 GM_ADDR workspace,
                                 const GemmScalingHardTanhGeluCustomTilingData* tilingData,
                                 TPipe* pipe)
    {
        this->M = tilingData->M;
        this->N = tilingData->N;
        this->K = tilingData->K;
        this->scalingFactor = tilingData->scalingFactor;
        this->hardtanhMin = tilingData->hardtanhMin;
        this->hardtanhMax = tilingData->hardtanhMax;
        this->pipe_ = pipe;
        this->workspace_ = workspace;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        wGm.SetGlobalBuffer((__gm__ float*)weight);
        bGm.SetGlobalBuffer((__gm__ float*)bias);
        yGm.SetGlobalBuffer((__gm__ float*)y);
        mmOutGm.SetGlobalBuffer((__gm__ float*)workspace, (uint64_t)this->M * this->N);

        matmulObj.SetSubBlockIdx(0);
        matmulObj.Init(&tilingData->cubeTilingData, pipe);
        matmulObj.SetTensorA(xGm, false);
        matmulObj.SetTensorB(wGm, true);
        matmulObj.SetBias(bGm);
    }

    __aicore__ inline void Process()
    {
        matmulObj.IterateAll(mmOutGm);
        matmulObj.End();

        SyncAll();

        PostProcess();
    }

private:
    __aicore__ inline void PostProcess()
    {
        uint64_t total = (uint64_t)this->M * this->N;
        uint32_t blockNum = GetBlockNum();
        uint32_t blockIdx = GetBlockIdx();

        uint64_t perCore = (total + blockNum - 1) / blockNum;
        uint64_t startOffset = (uint64_t)blockIdx * perCore;
        uint64_t endOffset = startOffset + perCore;
        if (endOffset > total) endOffset = total;
        if (startOffset >= total) return;

        uint64_t coreLen = endOffset - startOffset;

        constexpr uint32_t TILE = 8192;
        TQue<TPosition::VECIN, BUFFER_NUM> inQ;
        TQue<TPosition::VECOUT, BUFFER_NUM> outQ;
        TBuf<TPosition::VECCALC> tmpBuf1, tmpBuf2, tmpBuf3;

        pipe_->InitBuffer(inQ, BUFFER_NUM, TILE * sizeof(float));
        pipe_->InitBuffer(outQ, BUFFER_NUM, TILE * sizeof(float));
        pipe_->InitBuffer(tmpBuf1, TILE * sizeof(float));
        pipe_->InitBuffer(tmpBuf2, TILE * sizeof(float));
        pipe_->InitBuffer(tmpBuf3, TILE * sizeof(float));

        const float kSqrt2OverPi = 0.7978845608028654f;
        const float kGeluCoef = 0.044715f;

        uint64_t processed = 0;
        while (processed < coreLen) {
            uint32_t curLen = (coreLen - processed) > TILE ? TILE : (uint32_t)(coreLen - processed);

            LocalTensor<float> inLocal = inQ.AllocTensor<float>();
            DataCopy(inLocal, mmOutGm[startOffset + processed], curLen);
            inQ.EnQue(inLocal);

            LocalTensor<float> xLocal = inQ.DeQue<float>();
            LocalTensor<float> outLocal = outQ.AllocTensor<float>();
            LocalTensor<float> t1 = tmpBuf1.Get<float>();
            LocalTensor<float> t2 = tmpBuf2.Get<float>();
            LocalTensor<float> t3 = tmpBuf3.Get<float>();

            // scaling
            Muls(xLocal, xLocal, this->scalingFactor, curLen);

            // hardtanh: clamp to [min, max]
            Mins(xLocal, xLocal, this->hardtanhMax, curLen);
            Maxs(xLocal, xLocal, this->hardtanhMin, curLen);

            // GELU (tanh approximation):
            // 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
            Mul(t1, xLocal, xLocal, curLen);          // x^2
            Mul(t1, t1, xLocal, curLen);              // x^3
            Muls(t1, t1, kGeluCoef, curLen);          // 0.044715 * x^3
            Add(t1, t1, xLocal, curLen);              // x + 0.044715 * x^3
            Muls(t1, t1, kSqrt2OverPi, curLen);       // sqrt(2/pi) * (...)

            // tanh(t1) = (exp(2*t1) - 1) / (exp(2*t1) + 1)
            Muls(t2, t1, 2.0f, curLen);
            Exp(t2, t2, curLen);
            Adds(t3, t2, -1.0f, curLen);
            Adds(t2, t2, 1.0f, curLen);
            Div(t1, t3, t2, curLen);                  // tanh(...)

            Adds(t1, t1, 1.0f, curLen);               // 1 + tanh(...)
            Mul(t1, t1, xLocal, curLen);              // x * (1 + tanh(...))
            Muls(outLocal, t1, 0.5f, curLen);         // 0.5 * x * (1 + tanh(...))

            outQ.EnQue<float>(outLocal);
            inQ.FreeTensor(xLocal);

            LocalTensor<float> zLocal = outQ.DeQue<float>();
            DataCopy(yGm[startOffset + processed], zLocal, curLen);
            outQ.FreeTensor(zLocal);

            processed += curLen;
        }
    }

    Matmul<MatmulType<TPosition::GM, CubeFormat::ND, float>,
           MatmulType<TPosition::GM, CubeFormat::ND, float>,
           MatmulType<TPosition::GM, CubeFormat::ND, float>,
           MatmulType<TPosition::GM, CubeFormat::ND, float>> matmulObj;

    GlobalTensor<float> xGm;
    GlobalTensor<float> wGm;
    GlobalTensor<float> bGm;
    GlobalTensor<float> yGm;
    GlobalTensor<float> mmOutGm;

    uint32_t M;
    uint32_t N;
    uint32_t K;
    float scalingFactor;
    float hardtanhMin;
    float hardtanhMax;

    TPipe* pipe_;
    GM_ADDR workspace_;
};

extern "C" __global__ __aicore__ void gemm_scaling_hard_tanh_gelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    TPipe pipe;
    KernelGemmScalingHardTanhGelu op;
    GM_ADDR userWS = GetUserWorkspace(workspace);
    op.Init(x, weight, bias, y, userWS, &tilingData, &pipe);
    op.Process();
}
