kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;

constexpr int32_t MISH_TILE = 2048;

__aicore__ inline void MishInplace(LocalTensor<float>& data, LocalTensor<float>& tmp1,
                                    LocalTensor<float>& tmp2, uint32_t count)
{
    // softplus(x) = ln(1 + exp(x))
    Exp(tmp1, data, count);
    Adds(tmp1, tmp1, (float)1.0, count);
    Ln(tmp1, tmp1, count);
    // tanh(softplus(x))
    // tanh(v) = (exp(2v)-1)/(exp(2v)+1)
    Muls(tmp2, tmp1, (float)2.0, count);
    Exp(tmp2, tmp2, count);
    Adds(tmp1, tmp2, (float)-1.0, count);
    Adds(tmp2, tmp2, (float)1.0, count);
    Div(tmp1, tmp1, tmp2, count);
    // data = x * tanh(softplus(x))
    Mul(data, data, tmp1, count);
}

class KernelMatmulMishMish {
public:
    __aicore__ inline KernelMatmulMishMish() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
                                  GM_ADDR y, GM_ADDR workspace,
                                  const TCubeTiling& tiling,
                                  uint32_t M, uint32_t N, uint32_t K)
    {
        this->tiling = tiling;
        this->M = M;
        this->N = N;
        this->K = K;

        xGm.SetGlobalBuffer((__gm__ float*)x, M * K);
        weightGm.SetGlobalBuffer((__gm__ float*)weight, N * K);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, N);
        yGm.SetGlobalBuffer((__gm__ float*)y, M * N);

        mm.SetSubBlockIdx(0);
        mm.Init(&this->tiling, &pipe);

        pipe.InitBuffer(inQueueX, 2, MISH_TILE * sizeof(float));
        pipe.InitBuffer(outQueueY, 2, MISH_TILE * sizeof(float));
        pipe.InitBuffer(tmpBuf1, MISH_TILE * sizeof(float));
        pipe.InitBuffer(tmpBuf2, MISH_TILE * sizeof(float));
    }

    __aicore__ inline void ProcessMatmul()
    {
        mm.SetTensorA(xGm);
        mm.SetTensorB(weightGm, true);
        mm.SetBias(biasGm);
        mm.IterateAll(yGm);
        mm.End();
    }

    __aicore__ inline void ProcessMish()
    {
        uint32_t total = this->M * this->N;
        uint32_t blockNum = GetBlockNum();
        uint32_t blockIdx = GetBlockIdx();
        uint32_t perBlock = (total + blockNum - 1) / blockNum;
        perBlock = (perBlock + 63) / 64 * 64;
        uint32_t start = perBlock * blockIdx;
        uint32_t end = start + perBlock;
        if (end > total) end = total;
        if (start >= total) return;

        uint32_t offset = start;
        while (offset < end) {
            uint32_t cur = (end - offset) > MISH_TILE ? MISH_TILE : (end - offset);
            LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            DataCopy(xLocal, yGm[offset], (cur + 7) / 8 * 8);
            inQueueX.EnQue(xLocal);

            LocalTensor<float> xIn = inQueueX.DeQue<float>();
            LocalTensor<float> t1 = tmpBuf1.Get<float>();
            LocalTensor<float> t2 = tmpBuf2.Get<float>();
            MishInplace(xIn, t1, t2, cur);
            MishInplace(xIn, t1, t2, cur);

            LocalTensor<float> yOut = outQueueY.AllocTensor<float>();
            DataCopy(yOut, xIn, (cur + 7) / 8 * 8);
            outQueueY.EnQue(yOut);
            inQueueX.FreeTensor(xIn);

            LocalTensor<float> yLocal = outQueueY.DeQue<float>();
            DataCopy(yGm[offset], yLocal, (cur + 7) / 8 * 8);
            outQueueY.FreeTensor(yLocal);

            offset += cur;
        }
    }

private:
    TPipe pipe;
    TCubeTiling tiling;
    uint32_t M;
    uint32_t N;
    uint32_t K;
    matmul::Matmul<matmul::MatmulType<TPosition::GM, CubeFormat::ND, float>,
                   matmul::MatmulType<TPosition::GM, CubeFormat::ND, float>,
                   matmul::MatmulType<TPosition::GM, CubeFormat::ND, float>,
                   matmul::MatmulType<TPosition::GM, CubeFormat::ND, float>> mm;
    GlobalTensor<float> xGm;
    GlobalTensor<float> weightGm;
    GlobalTensor<float> biasGm;
    GlobalTensor<float> yGm;
    TQue<QuePosition::VECIN, 2> inQueueX;
    TQue<QuePosition::VECOUT, 2> outQueueY;
    TBuf<QuePosition::VECCALC> tmpBuf1;
    TBuf<QuePosition::VECCALC> tmpBuf2;
};

extern "C" __global__ __aicore__ void matmul_mish_mish_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulMishMish op;
    op.Init(x, weight, bias, y, workspace, tiling_data.cubeTilingData,
            tiling_data.M, tiling_data.N, tiling_data.K);
    op.ProcessMatmul();
    SyncAll();
    op.ProcessMish();
}
