kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmGroupNormMinBiasAdd {
public:
    __aicore__ inline KernelGemmGroupNormMinBiasAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                  uint32_t totalRows, uint32_t vecLength, uint32_t rowsPerCore)
    {
        this->vecLength = vecLength;
        this->rowsPerCore = rowsPerCore;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerCore;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, vecLength);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias + startRow, rowsPerCore);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + startRow * vecLength, rowsPerCore * vecLength);

        pipe.InitBuffer(inQueueX, 1, vecLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueBias, 1, ((rowsPerCore * sizeof(DTYPE_BIAS) + 31) / 32) * 32);
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, vecLength * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_X> xLocalIn = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocalIn, xGm, this->vecLength);
        inQueueX.EnQue(xLocalIn);
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();

        AscendC::LocalTensor<DTYPE_BIAS> biasLocalIn = inQueueBias.AllocTensor<DTYPE_BIAS>();
        AscendC::DataCopy(biasLocalIn, biasGm, this->rowsPerCore);
        inQueueBias.EnQue(biasLocalIn);
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = inQueueBias.DeQue<DTYPE_BIAS>();

        for (uint32_t i = 0; i < this->rowsPerCore; i++) {
            DTYPE_BIAS biasVal = biasLocal.GetValue(i);

            AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
            AscendC::Adds(zLocal, xLocal, biasVal, this->vecLength);
            outQueueZ.EnQue<DTYPE_Z>(zLocal);

            AscendC::LocalTensor<DTYPE_Z> zOut = outQueueZ.DeQue<DTYPE_Z>();
            AscendC::DataCopy(zGm[i * this->vecLength], zOut, this->vecLength);
            outQueueZ.FreeTensor(zOut);
        }

        inQueueX.FreeTensor(xLocal);
        inQueueBias.FreeTensor(biasLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t vecLength;
    uint32_t rowsPerCore;
};

extern "C" __global__ __aicore__ void gemm_group_norm_min_bias_add_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormMinBiasAdd op;
    op.Init(x, bias, z, tiling_data.totalRows, tiling_data.vecLength, tiling_data.rowsPerCore);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, bias_shape):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.group_norm = nn.GroupNorm(num_groups, out_features)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.out_features = out_features

    def forward(self, x):
        batch_size = x.shape[0]
        x = self.gemm(x)
        x = self.group_norm(x)
        x = torch.min(x, dim=1, keepdim=True)[0]
        x_flat = x.contiguous().view(-1)
        bias_flat = self.bias.contiguous().view(-1)
        out = custom_ops_lib.gemm_group_norm_min_bias_add_custom(x_flat, bias_flat)
        return out.view(1, self.out_features, batch_size, 1)
