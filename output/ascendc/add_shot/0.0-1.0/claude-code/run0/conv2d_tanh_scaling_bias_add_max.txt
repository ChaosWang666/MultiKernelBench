kernel_src="""
#include "kernel_operator.h"

class KernelConv2dTanhScalingBiasAddMax {
public:
    __aicore__ inline KernelConv2dTanhScalingBiasAddMax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t inH, uint32_t inW,
                                 uint32_t outH, uint32_t outW,
                                 uint32_t poolK, float scalingFactor)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->inH = inH;
        this->inW = inW;
        this->outH = outH;
        this->outW = outW;
        this->poolK = poolK;
        this->scalingFactor = scalingFactor;
        this->tileSize = poolK * inW;

        uint32_t totalNC = batchSize * channels;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->ncPerBlock = totalNC / blockNum;
        this->ncStart = blockIdx * this->ncPerBlock;

        uint64_t xTotalSize = (uint64_t)totalNC * inH * inW;
        uint64_t zTotalSize = (uint64_t)totalNC * outH * outW;

        xGm.SetGlobalBuffer((__gm__ float *)x, xTotalSize);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, channels);
        zGm.SetGlobalBuffer((__gm__ float *)z, zTotalSize);

        uint32_t tileBufBytes = ((this->tileSize * (uint32_t)sizeof(float) + 31u) / 32u) * 32u;
        pipe.InitBuffer(xBuf, tileBufBytes);

        uint32_t outAccumBytes = ((outH * outW * (uint32_t)sizeof(float) + 31u) / 32u) * 32u;
        pipe.InitBuffer(outAccumBuf, outAccumBytes);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> outAccum = outAccumBuf.Get<float>();

        AscendC::DataCopyExtParams copyInParams;
        copyInParams.blockCount = 1;
        copyInParams.blockLen = this->tileSize * (uint32_t)sizeof(float);
        copyInParams.srcStride = 0;
        copyInParams.dstStride = 0;

        AscendC::DataCopyPadExtParams<float> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;

        AscendC::DataCopyExtParams copyOutParams;
        copyOutParams.blockCount = 1;
        copyOutParams.blockLen = this->outH * this->outW * (uint32_t)sizeof(float);
        copyOutParams.srcStride = 0;
        copyOutParams.dstStride = 0;

        for (uint32_t i = 0; i < this->ncPerBlock; i++) {
            uint32_t ncIdx = this->ncStart + i;
            uint32_t cIdx = ncIdx % this->channels;
            float biasVal = biasGm.GetValue(cIdx);

            for (uint32_t rowGroup = 0; rowGroup < this->outH; rowGroup++) {
                uint64_t xOffset = (uint64_t)ncIdx * this->inH * this->inW
                                 + (uint64_t)rowGroup * this->poolK * this->inW;
                AscendC::DataCopyPad(xLocal, xGm[xOffset], copyInParams, padParams);

                AscendC::TEventID evMTE2V = AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::MTE2_V);
                AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(evMTE2V);
                AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(evMTE2V);

                AscendC::Tanh(xLocal, xLocal, this->tileSize);
                AscendC::PipeBarrier<PIPE_V>();
                AscendC::Muls(xLocal, xLocal, this->scalingFactor, this->tileSize);
                AscendC::PipeBarrier<PIPE_V>();
                AscendC::Adds(xLocal, xLocal, biasVal, this->tileSize);

                AscendC::TEventID evVS = AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S);
                AscendC::SetFlag<AscendC::HardEvent::V_S>(evVS);
                AscendC::WaitFlag<AscendC::HardEvent::V_S>(evVS);

                for (uint32_t colOut = 0; colOut < this->outW; colOut++) {
                    uint32_t baseOffset = colOut * this->poolK;
                    float maxVal = xLocal.GetValue(baseOffset);
                    for (uint32_t ki = 0; ki < this->poolK; ki++) {
                        for (uint32_t kj = 0; kj < this->poolK; kj++) {
                            float v = xLocal.GetValue(ki * this->inW + baseOffset + kj);
                            if (v > maxVal) {
                                maxVal = v;
                            }
                        }
                    }
                    outAccum.SetValue(rowGroup * this->outW + colOut, maxVal);
                }

                if (rowGroup < this->outH - 1) {
                    AscendC::TEventID evSMTE2 = AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_MTE2);
                    AscendC::SetFlag<AscendC::HardEvent::S_MTE2>(evSMTE2);
                    AscendC::WaitFlag<AscendC::HardEvent::S_MTE2>(evSMTE2);
                }
            }

            AscendC::TEventID evSMTE3 = AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_MTE3);
            AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(evSMTE3);
            AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(evSMTE3);

            uint64_t zOffset = (uint64_t)ncIdx * this->outH * this->outW;
            AscendC::DataCopyPad(zGm[zOffset], outAccum, copyOutParams);

            if (i < this->ncPerBlock - 1) {
                AscendC::TEventID evMTE3S = AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::MTE3_S);
                AscendC::SetFlag<AscendC::HardEvent::MTE3_S>(evMTE3S);
                AscendC::WaitFlag<AscendC::HardEvent::MTE3_S>(evMTE3S);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outAccumBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t batchSize;
    uint32_t channels;
    uint32_t inH;
    uint32_t inW;
    uint32_t outH;
    uint32_t outW;
    uint32_t poolK;
    uint32_t tileSize;
    uint32_t ncPerBlock;
    uint32_t ncStart;
    float scalingFactor;
};

extern "C" __global__ __aicore__ void conv2d_tanh_scaling_bias_add_max_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dTanhScalingBiasAddMax op;
    op.Init(x, bias, z,
            tiling_data.batchSize, tiling_data.channels,
            tiling_data.inH, tiling_data.inW,
            tiling_data.outH, tiling_data.outW,
            tiling_data.poolK, tiling_data.scalingFactor);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, scaling_factor, bias_shape, pool_kernel_size):
        super().__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.scaling_factor = float(scaling_factor)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.pool_kernel_size = int(pool_kernel_size)

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
        bias_flat = self.bias.view(-1).contiguous()
        return custom_ops_lib.conv2d_tanh_scaling_bias_add_max_custom(
            x, bias_flat, self.scaling_factor, self.pool_kernel_size
