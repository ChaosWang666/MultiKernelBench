import torch
import torch.nn as nn
import torch.nn.functional as F


class Model(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, scaling_factor, bias_shape, pool_kernel_size):
        super(Model, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.scaling_factor = scaling_factor
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.max_pool = nn.MaxPool2d(pool_kernel_size)

    def forward(self, x):
        x = self.conv(x)
        x = torch.tanh(x)
        x = x * self.scaling_factor
        x = x + self.bias
        x = self.max_pool(x)
        return x

class KernelConvTanhScalingBiasAddMax {
public:
    __aicore__ inline KernelConvTanhScalingBiasAddMax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR cb, GM_ADDR b, GM_ADDR y,
                                 uint32_t batch, uint32_t inC, uint32_t outC,
                                 uint32_t H, uint32_t W, uint32_t K,
                                 uint32_t P, uint32_t oH, uint32_t oW,
                                 uint32_t pH, uint32_t pW, float scale)
    {
        this->batch = batch;
        this->inC = inC;
        this->outC = outC;
        this->H = H; this->W = W; this->K = K;
        this->P = P;
        this->oH = oH; this->oW = oW;
        this->pH = pH; this->pW = pW;
        this->scale = scale;

        uint64_t xSize = (uint64_t)batch * inC * H * W;
        uint64_t wSize = (uint64_t)outC * inC * K * K;
        uint64_t ySize = (uint64_t)batch * outC * pH * pW;

        xGm.SetGlobalBuffer((__gm__ float*)x, xSize);
        wGm.SetGlobalBuffer((__gm__ float*)w, wSize);
        cbGm.SetGlobalBuffer((__gm__ float*)cb, outC);
        bGm.SetGlobalBuffer((__gm__ float*)b, outC);
        yGm.SetGlobalBuffer((__gm__ float*)y, ySize);
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t total = this->batch * this->outC;

        for (uint32_t idx = blockIdx; idx < total; idx += blockNum) {
            uint32_t n = idx / this->outC;
            uint32_t oc = idx % this->outC;
            ProcessOne(n, oc);
        }
    }

private:
    __aicore__ inline void ProcessOne(uint32_t n, uint32_t oc)
    {
        float cbias = cbGm.GetValue(oc);
        float pbias = bGm.GetValue(oc);
        uint64_t xBase = (uint64_t)n * this->inC * this->H * this->W;
        uint64_t wBase = (uint64_t)oc * this->inC * this->K * this->K;
        uint64_t yBase = (uint64_t)n * this->outC * this->pH * this->pW + (uint64_t)oc * this->pH * this->pW;

        for (uint32_t ph = 0; ph < this->pH; ph++) {
            for (uint32_t pw = 0; pw < this->pW; pw++) {
                float maxVal = -3.4e38f;
                for (uint32_t dh = 0; dh < this->P; dh++) {
                    for (uint32_t dw = 0; dw < this->P; dw++) {
                        uint32_t oh = ph * this->P + dh;
                        uint32_t ow = pw * this->P + dw;
                        float acc = cbias;
                        for (uint32_t ic = 0; ic < this->inC; ic++) {
                            for (uint32_t kh = 0; kh < this->K; kh++) {
                                for (uint32_t kw = 0; kw < this->K; kw++) {
                                    uint32_t ih = oh + kh;
                                    uint32_t iw = ow + kw;
                                    uint64_t xIdx = xBase + (uint64_t)ic * this->H * this->W + (uint64_t)ih * this->W + iw;
                                    uint64_t wIdx = wBase + (uint64_t)ic * this->K * this->K + (uint64_t)kh * this->K + kw;
                                    acc += xGm.GetValue(xIdx) * wGm.GetValue(wIdx);
                                }
                            }
                        }
                        float t = Tanhf(acc);
                        float v = t * this->scale + pbias;
                        if (v > maxVal) maxVal = v;
                    }
                }
                yGm.SetValue(yBase + ph * this->pW + pw, maxVal);
            }
        }
    }

    __aicore__ inline float Tanhf(float x)
    {
        if (x > 20.0f) return 1.0f;
        if (x < -20.0f) return -1.0f;
        float e2x = Expf(2.0f * x);
        return (e2x - 1.0f) / (e2x + 1.0f);
    }

    __aicore__ inline float Expf(float x)
    {
        float result = 1.0f;
        float term = 1.0f;
        for (int i = 1; i < 20; i++) {
            term *= x / (float)i;
            result += term;
        }
        return result;
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> cbGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batch, inC, outC, H, W, K, P, oH, oW, pH, pW;
    float scale;
};

            tiling_data.poolSize, tiling_data.outH, tiling_data.outW,
            tiling_data.pooledH, tiling_data.pooledW, tiling_data.scalingFactor);
    op.Process();
}
