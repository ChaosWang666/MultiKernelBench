kernel_src = """
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;
using namespace matmul;

template <typename aType, typename bType, typename cType, typename biasType>
class MatmulMulLeakyKernel {
public:
    __aicore__ inline MatmulMulLeakyKernel() {}
    __aicore__ inline void Init(GM_ADDR a, GM_ADDR b, GM_ADDR bias, GM_ADDR c,
                                GM_ADDR workspace, const TCubeTiling &tilingIn,
                                float multiplier, float negSlope, TPipe *pipe);
    template <bool setTmpSpace = false>
    __aicore__ inline void Process(TPipe *pipe);
    __aicore__ inline void MatmulCompute();
    __aicore__ inline void EpilogueCompute();
    __aicore__ inline void CopyOut(uint32_t count);
    __aicore__ inline void CalcOffset(int32_t blockIdx, const TCubeTiling &tiling,
                                      int32_t &offsetA, int32_t &offsetB,
                                      int32_t &offsetC, int32_t &offsetBias);

    Matmul<aType, bType, cType, biasType> matmulObj;
    GlobalTensor<typename aType::T> aGlobal;
    GlobalTensor<typename bType::T> bGlobal;
    GlobalTensor<typename cType::T> cGlobal;
    GlobalTensor<typename biasType::T> biasGlobal;
    LocalTensor<typename cType::T> outLocal;
    TCubeTiling tiling;
    TQue<QuePosition::VECOUT, 1> outQueue_;
    float multiplier;
    float negSlope;
};

template <typename aType, typename bType, typename cType, typename biasType>
__aicore__ inline void MatmulMulLeakyKernel<aType, bType, cType, biasType>::Init(
    GM_ADDR a, GM_ADDR b, GM_ADDR bias, GM_ADDR c, GM_ADDR workspace,
    const TCubeTiling &tilingIn, float multiplier, float negSlope, TPipe *pipe)
{
    this->tiling = tilingIn;
    this->multiplier = multiplier;
    this->negSlope = negSlope;
    aGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ typename aType::T *>(a),
                            tiling.M * tiling.Ka);
    bGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ typename bType::T *>(b),
                            tiling.Kb * tiling.N);
    cGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ typename cType::T *>(c),
                            tiling.M * tiling.N);
    biasGlobal.SetGlobalBuffer(reinterpret_cast<__gm__ typename biasType::T *>(bias),
                               tiling.N);

    int offsetA = 0, offsetB = 0, offsetC = 0, offsetBias = 0;
    CalcOffset(GetBlockIdx(), tiling, offsetA, offsetB, offsetC, offsetBias);
    aGlobal = aGlobal[offsetA];
    bGlobal = bGlobal[offsetB];
    cGlobal = cGlobal[offsetC];
    biasGlobal = biasGlobal[offsetBias];
    pipe->InitBuffer(outQueue_, 1,
                     tiling.baseM * tiling.baseN * sizeof(typename cType::T));
    SetSysWorkspace(workspace);
    if (GetSysWorkSpacePtr() == nullptr) {
        return;
    }
}

template <typename aType, typename bType, typename cType, typename biasType>
template <bool setTmpSpace>
__aicore__ inline void MatmulMulLeakyKernel<aType, bType, cType, biasType>::Process(TPipe *pipe)
{
    uint32_t computeRound = 0;

    if constexpr (setTmpSpace) {
        TBuf<TPosition::VECCALC> tmpMMFormatUb;
        LocalTensor<uint8_t> mmformatUb;
        pipe->InitBuffer(tmpMMFormatUb,
                         tiling.baseM * tiling.baseN * sizeof(typename cType::T));
        mmformatUb = tmpMMFormatUb.Get<uint8_t>(
            tiling.baseM * tiling.baseN * sizeof(typename cType::T));
        matmulObj.SetLocalWorkspace(mmformatUb);
    }

    matmulObj.SetTensorA(aGlobal);
    matmulObj.SetTensorB(bGlobal);
    matmulObj.SetBias(biasGlobal);
    while (matmulObj.template Iterate<true>()) {
        MatmulCompute();
        EpilogueCompute();
        CopyOut(computeRound);
        computeRound++;
    }
    matmulObj.End();
}

template <typename aType, typename bType, typename cType, typename biasType>
__aicore__ inline void MatmulMulLeakyKernel<aType, bType, cType, biasType>::MatmulCompute()
{
    outLocal = outQueue_.AllocTensor<typename cType::T>();
    matmulObj.template GetTensorC<true>(outLocal, false, true);
}

template <typename aType, typename bType, typename cType, typename biasType>
__aicore__ inline void MatmulMulLeakyKernel<aType, bType, cType, biasType>::EpilogueCompute()
{
    uint32_t len = tiling.baseM * tiling.baseN;
    Muls(outLocal, outLocal, static_cast<typename cType::T>(multiplier), len);
    PipeBarrier<PIPE_V>();
    LeakyRelu(outLocal, outLocal, static_cast<typename cType::T>(negSlope), len);
    outQueue_.EnQue(outLocal);
}

template <typename aType, typename bType, typename cType, typename biasType>
__aicore__ inline void MatmulMulLeakyKernel<aType, bType, cType, biasType>::CopyOut(uint32_t count)
{
    outQueue_.DeQue<typename cType::T>();
    const uint32_t roundM = tiling.singleCoreM / tiling.baseM;
    const uint32_t roundN = tiling.singleCoreN / tiling.baseN;
    uint32_t startOffset = (count % roundM * tiling.baseM * tiling.N +
                            count / roundM * tiling.baseN);
    DataCopyParams copyParam;
    copyParam.blockCount = (uint16_t)tiling.baseM;
    copyParam.blockLen = (uint16_t)(tiling.baseN * sizeof(typename cType::T) / 32);
    copyParam.srcStride = 0;
    copyParam.dstStride = (uint16_t)((tiling.N - tiling.baseN) * sizeof(typename cType::T) / 32);
    DataCopy(cGlobal[startOffset], outLocal, copyParam);
    outQueue_.FreeTensor(outLocal);
}

template <typename aType, typename bType, typename cType, typename biasType>
__aicore__ inline void MatmulMulLeakyKernel<aType, bType, cType, biasType>::CalcOffset(
    int32_t blockIdx, const TCubeTiling &tiling,
    int32_t &offsetA, int32_t &offsetB, int32_t &offsetC, int32_t &offsetBias)
{
    auto mSingleBlocks = Ceil(tiling.M, tiling.singleCoreM);
    auto mCoreIndex = blockIdx % mSingleBlocks;
    auto nCoreIndex = blockIdx / mSingleBlocks;
    offsetA = mCoreIndex * tiling.Ka * tiling.singleCoreM;
    offsetB = nCoreIndex * tiling.singleCoreN * tiling.Kb;
    offsetC = mCoreIndex * tiling.N * tiling.singleCoreM + nCoreIndex * tiling.singleCoreN;
    offsetBias = nCoreIndex * tiling.singleCoreN;
}

extern "C" __global__ __aicore__ void gemm_multiply_leakyrelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    using A_T = MatmulType<TPosition::GM, CubeFormat::ND, float>;
    using B_T = MatmulType<TPosition::GM, CubeFormat::ND, float, true>;
    using C_T = MatmulType<TPosition::GM, CubeFormat::ND, float>;
    using BIAS_T = MatmulType<TPosition::GM, CubeFormat::ND, float>;

    TPipe pipe;
    MatmulMulLeakyKernel<A_T, B_T, C_T, BIAS_T> op;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), op.matmulObj, &tiling_data.cubeTilingData);
    op.Init(x, weight, bias, y, workspace, tiling_data.cubeTilingData,
            tiling_data.multiplier, tiling_data.negativeSlope, &pipe);
    op.Process(&pipe);
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor gemm_multiply_leakyrelu_custom_impl_npu(
    const at::Tensor& x, const at::Tensor& weight, const at::Tensor& bias,
    double multiplier, double negative_slope)
{
    auto M = x.size(0);
    auto N = weight.size(0);
    at::Tensor result = at::empty({M, N}, x.options());
    float mult_f = static_cast<float>(multiplier);
    float ns_f = static_cast<float>(negative_slope);
    EXEC_NPU_CMD(aclnnGemmMultiplyLeakyreluCustom, x, weight, bias,
                 mult_f, ns_f, result);
    return result;
}

    m.impl("gemm_multiply_leakyrelu_custom", &gemm_multiply_leakyrelu_custom_impl_npu);

    m.def("gemm_multiply_leakyrelu_custom", &gemm_multiply_leakyrelu_custom_impl_npu,

model_src = '''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, multiplier, negative_slope):
        super().__init__()
        self.gemm = torch.nn.Linear(in_features, out_features)
        self.multiplier = float(multiplier)
        self.negative_slope = float(negative_slope)

    def forward(self, x):
        weight = self.gemm.weight.contiguous()
        bias = self.gemm.bias.contiguous()
        return custom_ops_lib.gemm_multiply_leakyrelu_custom(
            x.contiguous(), weight, bias,
            self.multiplier, self.negative_slope
        )
