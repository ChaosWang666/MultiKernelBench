kernel_src="""
#include "kernel_operator.h"

class KernelGemmSigmoidScalingResidualAdd {
public:
    __aicore__ inline KernelGemmSigmoidScalingResidualAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize,
                                uint32_t rowsPerCore, float scalingFactor)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->scalingFactor = scalingFactor;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowStart = blockIdx * rowsPerCore;
        uint32_t rowEnd = this->rowStart + rowsPerCore;
        if (rowEnd > batchSize) {
            rowEnd = batchSize;
        }
        this->rowsThisCore = (rowEnd > this->rowStart) ? (rowEnd - this->rowStart) : 0;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inputSize);
        wGm.SetGlobalBuffer((__gm__ float *)weight, hiddenSize * inputSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, hiddenSize);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * hiddenSize);

        pipe.InitBuffer(xBuf, inputSize * sizeof(float));
        pipe.InitBuffer(wBuf, inputSize * sizeof(float));
        pipe.InitBuffer(mulBuf, inputSize * sizeof(float));
        pipe.InitBuffer(outBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(sigBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(bBuf, hiddenSize * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->rowsThisCore == 0) {
            return;
        }

        AscendC::LocalTensor<float> bLocal = bBuf.Get<float>();
        AscendC::DataCopy(bLocal, bGm, this->hiddenSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t r = 0; r < this->rowsThisCore; r++) {
            uint32_t rowIdx = this->rowStart + r;
            ProcessRow(rowIdx, bLocal);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t rowIdx, AscendC::LocalTensor<float>& bLocal)
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> mulLocal = mulBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        AscendC::LocalTensor<float> sigLocal = sigBuf.Get<float>();

        AscendC::DataCopy(xLocal, xGm[rowIdx * this->inputSize], this->inputSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t h = 0; h < this->hiddenSize; h++) {
            AscendC::DataCopy(wLocal, wGm[h * this->inputSize], this->inputSize);
            AscendC::PipeBarrier<PIPE_ALL>();

            AscendC::Mul(mulLocal, xLocal, wLocal, this->inputSize);
            AscendC::PipeBarrier<PIPE_ALL>();

            float dotResult = 0.0f;
            for (uint32_t k = 0; k < this->inputSize; k++) {
                dotResult += mulLocal.GetValue(k);
            }
            outLocal.SetValue(h, dotResult);
        }
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Add(outLocal, outLocal, bLocal, this->hiddenSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Sigmoid(sigLocal, outLocal, this->hiddenSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Muls(sigLocal, sigLocal, this->scalingFactor, this->hiddenSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Add(outLocal, sigLocal, outLocal, this->hiddenSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::DataCopy(zGm[rowIdx * this->hiddenSize], outLocal, this->hiddenSize);
        AscendC::PipeBarrier<PIPE_ALL>();
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mulBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sigBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t inputSize;
    uint32_t hiddenSize;
    uint32_t rowStart;
    uint32_t rowsThisCore;
    float scalingFactor;
};

extern "C" __global__ __aicore__ void gemm_sigmoid_scaling_residual_add_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmSigmoidScalingResidualAdd op;
    op.Init(x, weight, bias, z,
            tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize,
            tiling_data.rowsPerCore, tiling_data.scalingFactor);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):

        super(ModelNew, self).__init__()

        self.scaling_factor = float(scaling_factor)
