kernel_src="""
#include "kernel_operator.h"

class KernelConv2dHardSwishRelu {
public:
    __aicore__ inline KernelConv2dHardSwishRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inHeight, uint32_t inWidth,
                                 uint32_t outHeight, uint32_t outWidth,
                                 uint32_t kernelSize, uint32_t totalOutputElements,
                                 uint32_t blockNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->kernelSize = kernelSize;
        this->totalOutputElements = totalOutputElements;
        this->blockNum = blockNum;

        int32_t blockIdx = static_cast<int32_t>(AscendC::GetBlockIdx());
        int32_t total = static_cast<int32_t>(totalOutputElements);
        int32_t bn = static_cast<int32_t>(blockNum);
        int32_t base = total / bn;
        int32_t rem = total - base * bn;
        if (blockIdx < rem) {
            this->elemPerCore = base + 1;
            this->elemOffset = blockIdx * this->elemPerCore;
        } else {
            this->elemPerCore = base;
            this->elemOffset = blockIdx * base + rem;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * inHeight * inWidth);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outChannels * inChannels * kernelSize * kernelSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalOutputElements);
    }

    __aicore__ inline void Process()
    {
        int32_t inH = static_cast<int32_t>(inHeight);
        int32_t inW = static_cast<int32_t>(inWidth);
        int32_t outH = static_cast<int32_t>(outHeight);
        int32_t outW = static_cast<int32_t>(outWidth);
        int32_t kS = static_cast<int32_t>(kernelSize);
        int32_t inC = static_cast<int32_t>(inChannels);
        int32_t outC = static_cast<int32_t>(outChannels);
        int32_t outHW = outH * outW;
        int32_t outCHW = outC * outHW;
        int32_t inHW = inH * inW;
        int32_t inCHW = inC * inHW;
        int32_t kSS = kS * kS;
        int32_t inCkSS = inC * kSS;

        for (int32_t idx = 0; idx < this->elemPerCore; idx++) {
            int32_t globalIdx = this->elemOffset + idx;
            int32_t b = globalIdx / outCHW;
            int32_t rem1 = globalIdx - b * outCHW;
            int32_t oc = rem1 / outHW;
            int32_t rem2 = rem1 - oc * outHW;
            int32_t oh = rem2 / outW;
            int32_t ow = rem2 - oh * outW;

            float sum = bGm.GetValue(oc);

            int32_t xBaseB = b * inCHW;
            int32_t wBaseOc = oc * inCkSS;

            for (int32_t ic = 0; ic < inC; ic++) {
                int32_t xBaseC = xBaseB + ic * inHW;
                int32_t wBaseC = wBaseOc + ic * kSS;
                for (int32_t kh = 0; kh < kS; kh++) {
                    int32_t ih = oh + kh;
                    int32_t xRowBase = xBaseC + ih * inW;
                    int32_t wRowBase = wBaseC + kh * kS;
                    for (int32_t kw = 0; kw < kS; kw++) {
                        int32_t iw = ow + kw;
                        float xv = xGm.GetValue(xRowBase + iw);
                        float wv = wGm.GetValue(wRowBase + kw);
                        sum += xv * wv;
                    }
                }
            }

            float hs_in = sum + 3.0f;
            if (hs_in < 0.0f) hs_in = 0.0f;
            if (hs_in > 6.0f) hs_in = 6.0f;
            float hs = sum * hs_in / 6.0f;
            float result = hs > 0.0f ? hs : 0.0f;

            yGm.SetValue(globalIdx, result);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inHeight;
    uint32_t inWidth;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t kernelSize;
    uint32_t totalOutputElements;
    uint32_t blockNum;
    int32_t elemPerCore;
    int32_t elemOffset;
};

extern "C" __global__ __aicore__ void conv2d_hard_swish_relu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dHardSwishRelu op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inHeight, tiling_data.inWidth,
            tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.kernelSize, tiling_data.totalOutputElements,
            tiling_data.blockNum);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib
class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)

    def forward(self, x):
        weight = self.conv.weight
        bias = self.conv.bias
