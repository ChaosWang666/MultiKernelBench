kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t TILE_LEN = 2048;
constexpr float EPS_VAL = 1e-5f;
constexpr float DIVIDE_BY_VAL = 2.0f;

class KernelInstanceNormDivide {
public:
    __aicore__ inline KernelInstanceNormDivide() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalChannels, uint32_t hw)
    {
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t chPerBlock = totalChannels / blockNum;
        uint32_t extra = totalChannels % blockNum;
        if (blockIdx < extra) {
            this->channelStart = blockIdx * (chPerBlock + 1);
            this->channelCount = chPerBlock + 1;
        } else {
            this->channelStart = extra * (chPerBlock + 1) + (blockIdx - extra) * chPerBlock;
            this->channelCount = chPerBlock;
        }
        this->hw = hw;
        this->numFullTiles = hw / TILE_LEN;
        this->remainder = hw % TILE_LEN;
        this->invHw = 1.0f / (float)hw;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, (uint64_t)totalChannels * hw);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, (uint64_t)totalChannels * hw);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, TILE_LEN * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, TILE_LEN * sizeof(DTYPE_Y));
        pipe.InitBuffer(tmpBuf, TILE_LEN * sizeof(float));
        pipe.InitBuffer(reduceBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t c = 0; c < channelCount; c++) {
            ProcessOneChannel(channelStart + c);
        }
    }

private:
    __aicore__ inline void ProcessOneChannel(uint32_t chIdx)
    {
        uint64_t base = (uint64_t)chIdx * hw;

        float sum = 0.0f;
        float sumSq = 0.0f;

        for (uint32_t t = 0; t < numFullTiles; t++) {
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xLocal, xGm[base + t * TILE_LEN], TILE_LEN);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<DTYPE_X>();

            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
            AscendC::LocalTensor<float> reduce = reduceBuf.Get<float>();

            AscendC::ReduceSum<float>(reduce, xLocal, tmp, TILE_LEN);
            int32_t evSum = static_cast<int32_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(evSum);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(evSum);
            sum += reduce.GetValue(0);

            AscendC::Mul(tmp, xLocal, xLocal, TILE_LEN);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum<float>(reduce, tmp, tmp, TILE_LEN);
            int32_t evSq = static_cast<int32_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(evSq);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(evSq);
            sumSq += reduce.GetValue(0);

            inQueueX.FreeTensor(xLocal);
        }

        if (remainder > 0) {
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::Duplicate<DTYPE_X>(xLocal, (DTYPE_X)0, TILE_LEN);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::DataCopyExtParams extParams{1, remainder * (uint32_t)sizeof(DTYPE_X), 0, 0, 0};
            AscendC::DataCopyPadExtParams<DTYPE_X> padParams{false, 0, 0, 0};
            AscendC::DataCopyPad(xLocal, xGm[base + numFullTiles * TILE_LEN], extParams, padParams);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<DTYPE_X>();

            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
            AscendC::LocalTensor<float> reduce = reduceBuf.Get<float>();

            AscendC::ReduceSum<float>(reduce, xLocal, tmp, remainder);
            int32_t evR1 = static_cast<int32_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(evR1);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(evR1);
            sum += reduce.GetValue(0);

            AscendC::Mul(tmp, xLocal, xLocal, remainder);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum<float>(reduce, tmp, tmp, remainder);
            int32_t evR2 = static_cast<int32_t>(GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(evR2);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(evR2);
            sumSq += reduce.GetValue(0);

            inQueueX.FreeTensor(xLocal);
        }

        float mean = sum * invHw;
        float var = sumSq * invHw - mean * mean;
        if (var < 0.0f) var = 0.0f;
        float invStd = 1.0f / sqrt(var + EPS_VAL);
        float scale = invStd / DIVIDE_BY_VAL;
        float bias = -mean * scale;

        for (uint32_t t = 0; t < numFullTiles; t++) {
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xLocal, xGm[base + t * TILE_LEN], TILE_LEN);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<DTYPE_X>();

            AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
            AscendC::Muls(yLocal, xLocal, (DTYPE_Y)scale, TILE_LEN);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Adds(yLocal, yLocal, (DTYPE_Y)bias, TILE_LEN);
            outQueueY.EnQue<DTYPE_Y>(yLocal);
            inQueueX.FreeTensor(xLocal);

            yLocal = outQueueY.DeQue<DTYPE_Y>();
            AscendC::DataCopy(yGm[base + t * TILE_LEN], yLocal, TILE_LEN);
            outQueueY.FreeTensor(yLocal);
        }

        if (remainder > 0) {
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::Duplicate<DTYPE_X>(xLocal, (DTYPE_X)0, TILE_LEN);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::DataCopyExtParams extParamsIn{1, remainder * (uint32_t)sizeof(DTYPE_X), 0, 0, 0};
            AscendC::DataCopyPadExtParams<DTYPE_X> padParamsIn{false, 0, 0, 0};
            AscendC::DataCopyPad(xLocal, xGm[base + numFullTiles * TILE_LEN], extParamsIn, padParamsIn);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<DTYPE_X>();

            AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
            AscendC::Muls(yLocal, xLocal, (DTYPE_Y)scale, remainder);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Adds(yLocal, yLocal, (DTYPE_Y)bias, remainder);
            outQueueY.EnQue<DTYPE_Y>(yLocal);
            inQueueX.FreeTensor(xLocal);

            yLocal = outQueueY.DeQue<DTYPE_Y>();
            AscendC::DataCopyExtParams extParamsOut{1, remainder * (uint32_t)sizeof(DTYPE_Y), 0, 0, 0};
            AscendC::DataCopyPad(yGm[base + numFullTiles * TILE_LEN], yLocal, extParamsOut);
            outQueueY.FreeTensor(yLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t channelStart;
    uint32_t channelCount;
    uint32_t hw;
    uint32_t numFullTiles;
    uint32_t remainder;
    float invHw;
};

extern "C" __global__ __aicore__ void conv2d_instance_norm_divide_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelInstanceNormDivide op;
    op.Init(x, y, tiling_data.totalChannels, tiling_data.hw);
    op.Process();
}
