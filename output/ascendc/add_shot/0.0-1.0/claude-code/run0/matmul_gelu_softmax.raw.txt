kernel_src="""
#include "kernel_operator.h"

class KernelMatmulGeluSoftmax {
public:
    __aicore__ inline KernelMatmulGeluSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                uint32_t M, uint32_t N, uint32_t K, uint32_t rowsPerBlock)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        this->rowsPerBlock = rowsPerBlock;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowStart = blockIdx * rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ float*)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float*)weight, N * K);
        bGm.SetGlobalBuffer((__gm__ float*)bias, N);
        zGm.SetGlobalBuffer((__gm__ float*)z, M * N);

        pipe.InitBuffer(xBuf, K * sizeof(float));
        pipe.InitBuffer(wBuf, K * sizeof(float));
        pipe.InitBuffer(biasBuf, N * sizeof(float));
        pipe.InitBuffer(outBuf, N * sizeof(float));
        pipe.InitBuffer(tmpBuf, K * sizeof(float));
        pipe.InitBuffer(reduceBuf, 1024 * sizeof(float));
        pipe.InitBuffer(scalarBuf, 32 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();

        AscendC::DataCopy(biasLocal, bGm, this->N);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

        for (uint32_t r = 0; r < this->rowsPerBlock; r++) {
            uint32_t row = this->rowStart + r;

            AscendC::DataCopy(xLocal, xGm[row * this->K], this->K);
            AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID1);
            AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID1);

            for (uint32_t j = 0; j < this->N; j++) {
                AscendC::DataCopy(wLocal, wGm[j * this->K], this->K);
                AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID2);
                AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID2);

                AscendC::Mul(tmpLocal, xLocal, wLocal, this->K);
                AscendC::PipeBarrier<PIPE_V>();
                AscendC::ReduceSum<float>(scalarLocal, tmpLocal, reduceLocal, this->K);

                AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

                float val = scalarLocal.GetValue(0);
                outLocal.SetValue(j, val);

                AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
                AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            }

            AscendC::Add(outLocal, outLocal, biasLocal, this->N);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::Gelu<float, false>(outLocal, outLocal, this->N);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceMax<float>(scalarLocal, outLocal, reduceLocal, this->N, false);
            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID1);

            float maxVal = scalarLocal.GetValue(0);
            float negMax = -maxVal;

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID1);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID1);

            AscendC::Adds(outLocal, outLocal, negMax, this->N);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Exp(outLocal, outLocal, this->N);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum<float>(scalarLocal, outLocal, reduceLocal, this->N);

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID2);

            float sumVal = scalarLocal.GetValue(0);
            float invSum = 1.0f / sumVal;

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID2);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID2);

            AscendC::Muls(outLocal, outLocal, invSum, this->N);

            AscendC::SetFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);

            AscendC::DataCopy(zGm[row * this->N], outLocal, this->N);

            AscendC::SetFlag<AscendC::HardEvent::MTE3_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE3_V>(EVENT_ID0);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, wBuf, outBuf, tmpBuf, biasBuf, reduceBuf, scalarBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, zGm;
    uint32_t M;
    uint32_t N;
    uint32_t K;
    uint32_t rowsPerBlock;
    uint32_t rowStart;
};

extern "C" __global__ __aicore__ void matmul_gelu_softmax_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulGeluSoftmax op;
    op.Init(x, weight, bias, z, tiling_data.M, tiling_data.N, tiling_data.K, tiling_data.rowsPerBlock);
    op.Process();
}
