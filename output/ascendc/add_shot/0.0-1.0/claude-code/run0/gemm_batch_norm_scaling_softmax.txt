kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelGemmBNScaleSoftmax {
public:
    __aicore__ inline KernelGemmBNScaleSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batchSize,
                                uint32_t featureSize,
                                uint32_t rowsPerBlock)
    {
        this->batchSize = batchSize;
        this->featureSize = featureSize;
        this->rowsPerBlock = rowsPerBlock;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startRow * featureSize,
                            rowsPerBlock * featureSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + startRow * featureSize,
                            rowsPerBlock * featureSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, featureSize * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, featureSize * sizeof(float));
        pipe.InitBuffer(workBuf, featureSize * sizeof(float));
        pipe.InitBuffer(reduceBuf, 32 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < rowsPerBlock; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * featureSize], featureSize);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        int32_t count = static_cast<int32_t>(featureSize);

        AscendC::ReduceMax<float>(reduceLocal, xLocal, workLocal, count, false);

        event_t eventVS1 = static_cast<event_t>(
            AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(eventVS1);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(eventVS1);

        float maxVal = reduceLocal.GetValue(0);
        float negMax = -maxVal;

        event_t eventSV1 = static_cast<event_t>(
            AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_V));
        AscendC::SetFlag<AscendC::HardEvent::S_V>(eventSV1);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(eventSV1);

        AscendC::Adds(yLocal, xLocal, negMax, count);
        AscendC::Exp(yLocal, yLocal, count);

        AscendC::ReduceSum<float>(reduceLocal, yLocal, workLocal, count);

        event_t eventVS2 = static_cast<event_t>(
            AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(eventVS2);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(eventVS2);

        float sumVal = reduceLocal.GetValue(0);
        float invSum = 1.0f / sumVal;

        event_t eventSV2 = static_cast<event_t>(
            AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_V));
        AscendC::SetFlag<AscendC::HardEvent::S_V>(eventSV2);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(eventSV2);

        AscendC::Muls(yLocal, yLocal, invSum, count);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * featureSize], yLocal, featureSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t featureSize;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void gemm_batch_norm_scaling_softmax_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmBNScaleSoftmax op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.featureSize, tiling_data.rowsPerBlock);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, bn_eps=1e-5, bn_momentum=0.1, scale_shape=(1,)):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.bn = nn.BatchNorm1d(out_features, eps=bn_eps, momentum=bn_momentum)
        self.scale = nn.Parameter(torch.ones(scale_shape))

    def forward(self, x):
        x = self.gemm(x)
        x = self.bn(x)
        x = self.scale * x
        x = x.contiguous()
        x = custom_ops_lib.gemm_batch_norm_scaling_softmax_custom(x)
        return x
