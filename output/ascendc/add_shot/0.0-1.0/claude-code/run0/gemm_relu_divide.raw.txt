kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

extern "C" __global__ __aicore__ void gemm_relu_divide_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);

    AscendC::TPipe pipe;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> yGm;
    xGm.SetGlobalBuffer((__gm__ float *)x);
    weightGm.SetGlobalBuffer((__gm__ float *)weight);
    biasGm.SetGlobalBuffer((__gm__ float *)bias);
    yGm.SetGlobalBuffer((__gm__ float *)y);

    using aType = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float, false>;
    using bType = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float, true>;
    using cType = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float, false>;
    using biasType = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float, false>;

    matmul::Matmul<aType, bType, cType, biasType> mm;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm, &tilingData.cubeTilingData);
    mm.Init(&tilingData.cubeTilingData, &pipe);

    if ASCEND_IS_AIC {
        mm.SetTensorA(xGm);
        mm.SetTensorB(weightGm);
        mm.SetBias(biasGm);
        mm.IterateAll(yGm);
        mm.End();
    }

    AscendC::SyncAll();

    if ASCEND_IS_AIV {
        const int32_t totalLen = static_cast<int32_t>(tilingData.totalLength);
        const int32_t blkNum = static_cast<int32_t>(AscendC::GetBlockNum());
        const int32_t blkIdx = static_cast<int32_t>(AscendC::GetBlockIdx());
        const int32_t perBlk = totalLen / blkNum;
        const int32_t offset = blkIdx * perBlk;

        constexpr int32_t TILE_LEN = 8192;
        const int32_t loops = perBlk / TILE_LEN;
        const float invDiv = tilingData.invDivisor;

        AscendC::TQue<AscendC::TPosition::VECIN, 2> inQ;
        AscendC::TQue<AscendC::TPosition::VECOUT, 2> outQ;
        pipe.InitBuffer(inQ, 2, TILE_LEN * sizeof(float));
        pipe.InitBuffer(outQ, 2, TILE_LEN * sizeof(float));

        for (int32_t i = 0; i < loops; i++) {
            AscendC::LocalTensor<float> inL = inQ.AllocTensor<float>();
            AscendC::DataCopy(inL, yGm[offset + i * TILE_LEN], TILE_LEN);
            inQ.EnQue(inL);

            AscendC::LocalTensor<float> inD = inQ.DeQue<float>();
            AscendC::LocalTensor<float> outL = outQ.AllocTensor<float>();
            AscendC::Relu(outL, inD, TILE_LEN);
            AscendC::Muls(outL, outL, invDiv, TILE_LEN);
            outQ.EnQue(outL);
            inQ.FreeTensor(inD);

            AscendC::LocalTensor<float> outD = outQ.DeQue<float>();
            AscendC::DataCopy(yGm[offset + i * TILE_LEN], outD, TILE_LEN);
            outQ.FreeTensor(outD);
        }
    }
}
