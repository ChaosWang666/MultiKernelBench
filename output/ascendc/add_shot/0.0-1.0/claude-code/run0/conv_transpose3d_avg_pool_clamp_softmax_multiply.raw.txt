kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelCTASCSM {
public:
    __aicore__ inline KernelCTASCSM() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR y,
                                 uint32_t totalRows, uint32_t rowLength,
                                 uint32_t channels, uint32_t tileLength,
                                 float clampMin, float clampMax)
    {
        this->rowLength = rowLength;
        this->channels = channels;
        this->tileLength = tileLength;
        this->numTiles = rowLength / tileLength;
        this->clampMin = clampMin;
        this->clampMax = clampMax;
        this->negClampMax = -clampMax;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerBlock = totalRows / blockNum;
        this->startRow = blockIdx * this->rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ float*)x + (uint64_t)this->startRow * rowLength,
                            (uint64_t)this->rowsPerBlock * rowLength);
        yGm.SetGlobalBuffer((__gm__ float*)y + (uint64_t)this->startRow * rowLength,
                            (uint64_t)this->rowsPerBlock * rowLength);
        scaleGm.SetGlobalBuffer((__gm__ float*)scale, channels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, tileLength * sizeof(float));
        pipe.InitBuffer(reduceBuf, 2048 * sizeof(float));
        pipe.InitBuffer(scaleBuf, 256 * sizeof(float));

        AscendC::LocalTensor<float> scaleLocal = scaleBuf.Get<float>();
        AscendC::DataCopy(scaleLocal, scaleGm, channels);
        AscendC::event_t evtMte2S = static_cast<AscendC::event_t>(pipe.FetchEventID(AscendC::HardEvent::MTE2_S));
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(evtMte2S);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(evtMte2S);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> scaleLocal = scaleBuf.Get<float>();
        for (uint32_t r = 0; r < this->rowsPerBlock; r++) {
            uint32_t globalRow = this->startRow + r;
            uint32_t channelIdx = globalRow % this->channels;
            float scaleVal = scaleLocal.GetValue(channelIdx);
            ProcessRow(r, scaleVal);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t rowIdx, float scaleVal)
    {
        uint64_t rowOffset = (uint64_t)rowIdx * this->rowLength;

        float sumExp = 0.0f;
        for (uint32_t t = 0; t < this->numTiles; t++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[rowOffset + (uint64_t)t * this->tileLength], this->tileLength);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
            AscendC::Maxs(tmp, xLocal, this->clampMin, this->tileLength);
            AscendC::Mins(tmp, tmp, this->clampMax, this->tileLength);
            AscendC::Adds(tmp, tmp, this->negClampMax, this->tileLength);
            AscendC::Exp(tmp, tmp, this->tileLength);

            AscendC::LocalTensor<float> red = reduceBuf.Get<float>();
            AscendC::ReduceSum<float>(red, tmp, red, this->tileLength);

            AscendC::event_t evtVS = static_cast<AscendC::event_t>(AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(evtVS);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(evtVS);
            sumExp += red.GetValue(0);

            inQueueX.FreeTensor(xLocal);
        }

        float invSumScale = scaleVal / sumExp;

        for (uint32_t t = 0; t < this->numTiles; t++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[rowOffset + (uint64_t)t * this->tileLength], this->tileLength);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
            AscendC::Maxs(yLocal, xLocal, this->clampMin, this->tileLength);
            AscendC::Mins(yLocal, yLocal, this->clampMax, this->tileLength);
            AscendC::Adds(yLocal, yLocal, this->negClampMax, this->tileLength);
            AscendC::Exp(yLocal, yLocal, this->tileLength);
            AscendC::Muls(yLocal, yLocal, invSumScale, this->tileLength);

            outQueueY.EnQue<float>(yLocal);
            inQueueX.FreeTensor(xLocal);

            yLocal = outQueueY.DeQue<float>();
            AscendC::DataCopy(yGm[rowOffset + (uint64_t)t * this->tileLength], yLocal, this->tileLength);
            outQueueY.FreeTensor(yLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scaleBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> scaleGm;

    uint32_t rowLength;
    uint32_t channels;
    uint32_t tileLength;
    uint32_t numTiles;
    uint32_t rowsPerBlock;
    uint32_t startRow;
    float clampMin;
    float clampMax;
    float negClampMax;
};

extern "C" __global__ __aicore__ void conv_transpose3d_avg_pool_clamp_softmax_multiply_custom(
    GM_ADDR x, GM_ADDR scale, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelCTASCSM op;
    op.Init(x, scale, y,
            tiling_data.totalRows, tiling_data.rowLength,
            tiling_data.channels, tiling_data.tileLength,
            tiling_data.clampMin, tiling_data.clampMax);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, pool_kernel_size, clamp_min, clamp_max):
        super(ModelNew, self).__init__()
        self.avg_pool = torch.nn.AvgPool3d(pool_kernel_size)
        self.conv_transpose = torch.nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.clamp_min = clamp_min
        self.clamp_max = clamp_max
        self.scale = torch.nn.Parameter(torch.ones(1, out_channels, 1, 1, 1))

    def forward(self, x):
        x = self.avg_pool(x)
        x = self.conv_transpose(x)
        x = x.contiguous()
        scale_flat = self.scale.reshape(-1).contiguous()
        return custom_ops_lib.conv_transpose3d_avg_pool_clamp_softmax_multiply_custom(x, scale_flat)
