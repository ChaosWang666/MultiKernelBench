kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                 uint32_t totalLength, uint32_t featureSize, uint32_t tileLength)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->featureSize = featureSize;
        this->tileLength = tileLength;
        this->coreLength = totalLength / blockNum;
        this->tileNum = this->coreLength / tileLength / BUFFER_NUM;

        uint32_t coreOffset = this->coreLength * blockIdx;
        this->startFeatureOffset = coreOffset % featureSize;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + coreOffset, this->coreLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, featureSize);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + coreOffset, this->coreLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueY, BUFFER_NUM, tileLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpBuf1, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t loopCount = this->tileNum * BUFFER_NUM;
        for (int32_t i = 0; i < loopCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        uint32_t offsetInCore = (uint32_t)progress * this->tileLength;
        uint32_t featureOffset = (this->startFeatureOffset + offsetInCore) % this->featureSize;

        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.AllocTensor<DTYPE_Y>();
        AscendC::DataCopy(xLocal, xGm[offsetInCore], this->tileLength);
        AscendC::DataCopy(yLocal, yGm[featureOffset], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();

        uint32_t n = this->tileLength;

        AscendC::Add(xLocal, xLocal, yLocal, n);

        AscendC::Muls(tmp1, xLocal, -1.0f, n);
        AscendC::Exp(tmp1, tmp1, n);
        AscendC::Adds(tmp1, tmp1, 1.0f, n);
        AscendC::Div(tmp2, xLocal, tmp1, n);

        AscendC::Muls(tmp1, tmp2, 2.0f, n);
        AscendC::Exp(tmp1, tmp1, n);
        AscendC::Adds(tmp2, tmp1, -1.0f, n);
        AscendC::Adds(tmp1, tmp1, 1.0f, n);
        AscendC::Div(tmp2, tmp2, tmp1, n);

        AscendC::Mul(tmp1, tmp2, tmp2, n);
        AscendC::Mul(tmp1, tmp1, tmp2, n);
        AscendC::Muls(tmp1, tmp1, 0.044715f, n);
        AscendC::Add(tmp1, tmp1, tmp2, n);
        AscendC::Muls(tmp1, tmp1, 0.7978845608f, n);

        AscendC::Muls(xLocal, tmp1, 2.0f, n);
        AscendC::Exp(xLocal, xLocal, n);
        AscendC::Adds(tmp1, xLocal, -1.0f, n);
        AscendC::Adds(xLocal, xLocal, 1.0f, n);
        AscendC::Div(tmp1, tmp1, xLocal, n);

        AscendC::Adds(tmp1, tmp1, 1.0f, n);
        AscendC::Mul(tmp1, tmp1, tmp2, n);
        AscendC::Muls(tmp1, tmp1, 0.5f, n);

        AscendC::Mins(tmp1, tmp1, 1.0f, n);
        AscendC::Maxs(zLocal, tmp1, -1.0f, n);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        uint32_t offsetInCore = (uint32_t)progress * this->tileLength;
        AscendC::DataCopy(zGm[offsetInCore], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1, tmpBuf2;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t coreLength;
    uint32_t featureSize;
    uint32_t tileLength;
    uint32_t tileNum;
    uint32_t startFeatureOffset;
};

extern "C" __global__ __aicore__ void matmul_add_swish_tanh_gelu_hardtanh_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, y, z, tiling_data.totalLength, tiling_data.featureSize, tiling_data.tileLength);
    op.Process();
}
