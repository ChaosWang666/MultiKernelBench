kernel_src="""
#include "kernel_operator.h"

class KernelMatmulSoftmax {
public:
    __aicore__ inline KernelMatmulSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->rowsPerCore = batchSize / blockNum;
        uint32_t remainder = batchSize % blockNum;
        uint32_t startRow;
        if (blockIdx < remainder) {
            this->rowsPerCore += 1;
            startRow = blockIdx * this->rowsPerCore;
        } else {
            startRow = blockIdx * this->rowsPerCore + remainder;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x + startRow * inFeatures, this->rowsPerCore * inFeatures);
        weightGm.SetGlobalBuffer((__gm__ float*)weight, outFeatures * inFeatures);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, outFeatures);
        yGm.SetGlobalBuffer((__gm__ float*)y + startRow * outFeatures, this->rowsPerCore * outFeatures);

        pipe.InitBuffer(inQueueX, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf, 256);
    }

    __aicore__ inline void Process()
    {
        if (this->rowsPerCore == 0) {
            return;
        }
        for (uint32_t b = 0; b < this->rowsPerCore; b++) {
            ProcessRow(b);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t b)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[b * inFeatures], inFeatures);
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> accLocal = outQueueY.AllocTensor<float>();
        AscendC::DataCopy(accLocal, biasGm, outFeatures);
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        for (uint32_t j = 0; j < outFeatures; j++) {
            AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
            AscendC::DataCopy(wLocal, weightGm[j * inFeatures], inFeatures);
            inQueueW.EnQue(wLocal);
            wLocal = inQueueW.DeQue<float>();

            AscendC::Mul(wLocal, xLocal, wLocal, inFeatures);
            AscendC::ReduceSum<float>(tmpLocal, wLocal, wLocal, inFeatures);

            float dot = tmpLocal.GetValue(0);
            float b_val = accLocal.GetValue(j);
            accLocal.SetValue(j, b_val + dot);

            inQueueW.FreeTensor(wLocal);
        }

        inQueueX.FreeTensor(xLocal);

        AscendC::ReduceMax<float>(tmpLocal, accLocal, accLocal, outFeatures, false);
        float maxVal = tmpLocal.GetValue(0);

        AscendC::Adds(accLocal, accLocal, -maxVal, outFeatures);
        AscendC::Exp(accLocal, accLocal, outFeatures);

        AscendC::LocalTensor<float> xLocalReuse = inQueueX.AllocTensor<float>();
        AscendC::ReduceSum<float>(tmpLocal, accLocal, xLocalReuse, outFeatures);
        float sumVal = tmpLocal.GetValue(0);
        inQueueX.FreeTensor(xLocalReuse);

        float inv = 1.0f / sumVal;
        AscendC::Muls(accLocal, accLocal, inv, outFeatures);

        outQueueY.EnQue(accLocal);
        accLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[b * outFeatures], accLocal, outFeatures);
        outQueueY.FreeTensor(accLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t inFeatures;
    uint32_t outFeatures;
    uint32_t rowsPerCore;
};

extern "C" __global__ __aicore__ void matmul_dropout_mean_softmax_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSoftmax op;
    op.Init(x, weight, bias, y, tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures);
    op.Process();
}

        self.dropout = nn.Dropout(dropout_p)

    def forward(self, x):
        weight = self.matmul.weight.contiguous()
