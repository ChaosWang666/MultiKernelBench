kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTransposeBNSub {
public:
    __aicore__ inline KernelConvTransposeBNSub() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalRows, uint32_t spatialSize, uint32_t tileLength)
    {
        this->spatialSize = spatialSize;
        this->tileLength = tileLength;
        this->numTiles = (spatialSize + tileLength - 1) / tileLength;
        this->lastTileLen = spatialSize - (this->numTiles - 1) * tileLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t rowsPerBlock = (totalRows + blockNum - 1) / blockNum;
        this->startRow = blockIdx * rowsPerBlock;
        this->endRow = this->startRow + rowsPerBlock;
        if (this->endRow > totalRows) {
            this->endRow = totalRows;
        }

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(reduceWorkBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(reduceOutBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t row = this->startRow; row < this->endRow; row++) {
            float mean = ComputeMean(row);
            SubtractMean(row, mean);
        }
    }

private:
    __aicore__ inline float ComputeMean(uint32_t row)
    {
        float totalSum = 0.0f;
        uint64_t rowBase = (uint64_t)row * (uint64_t)this->spatialSize;
        for (uint32_t t = 0; t < this->numTiles; t++) {
            uint32_t curLen = (t == this->numTiles - 1) ? this->lastTileLen : this->tileLength;
            uint64_t offset = rowBase + (uint64_t)t * (uint64_t)this->tileLength;

            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopyExtParams copyParams{1, (uint32_t)(curLen * sizeof(DTYPE_X)), 0, 0, 0};
            AscendC::DataCopyPadExtParams<DTYPE_X> padParams{false, 0, 0, 0};
            AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
            inQueueX.EnQue(xLocal);

            xLocal = inQueueX.DeQue<DTYPE_X>();
            AscendC::LocalTensor<float> work = reduceWorkBuf.Get<float>();
            AscendC::LocalTensor<float> outTmp = reduceOutBuf.Get<float>();
            AscendC::ReduceSum<float>(outTmp, xLocal, work, curLen);
            AscendC::PipeBarrier<PIPE_V>();
            totalSum += outTmp.GetValue(0);
            inQueueX.FreeTensor(xLocal);
        }
        return totalSum / (float)this->spatialSize;
    }

    __aicore__ inline void SubtractMean(uint32_t row, float mean)
    {
        uint64_t rowBase = (uint64_t)row * (uint64_t)this->spatialSize;
        float negMean = -mean;
        for (uint32_t t = 0; t < this->numTiles; t++) {
            uint32_t curLen = (t == this->numTiles - 1) ? this->lastTileLen : this->tileLength;
            uint64_t offset = rowBase + (uint64_t)t * (uint64_t)this->tileLength;

            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopyExtParams copyInParams{1, (uint32_t)(curLen * sizeof(DTYPE_X)), 0, 0, 0};
            AscendC::DataCopyPadExtParams<DTYPE_X> padParams{false, 0, 0, 0};
            AscendC::DataCopyPad(xLocal, xGm[offset], copyInParams, padParams);
            inQueueX.EnQue(xLocal);

            xLocal = inQueueX.DeQue<DTYPE_X>();
            AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
            AscendC::Adds(zLocal, xLocal, (DTYPE_X)negMean, curLen);
            outQueueZ.EnQue<DTYPE_Z>(zLocal);
            inQueueX.FreeTensor(xLocal);

            zLocal = outQueueZ.DeQue<DTYPE_Z>();
            AscendC::DataCopyExtParams copyOutParams{1, (uint32_t)(curLen * sizeof(DTYPE_Z)), 0, 0, 0};
            AscendC::DataCopyPad(zGm[offset], zLocal, copyOutParams);
            outQueueZ.FreeTensor(zLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceWorkBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceOutBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t spatialSize;
    uint32_t tileLength;
    uint32_t numTiles;
    uint32_t lastTileLen;
    uint32_t startRow;
    uint32_t endRow;
};

extern "C" __global__ __aicore__ void conv_transpose3d_batch_norm_subtract_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTransposeBNSub op;
    op.Init(x, z, tiling_data.totalRows, tiling_data.spatialSize, tiling_data.tileLength);
    op.Process();
}

import torch
import torch.nn as nn

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, bias=True):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, bias=bias)
        self.batch_norm = nn.BatchNorm3d(out_channels)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.batch_norm(x)
        x = x.contiguous()
