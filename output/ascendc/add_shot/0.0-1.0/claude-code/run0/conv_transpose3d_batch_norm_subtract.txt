kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSubMean {
public:
    __aicore__ inline KernelSubMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalGroups,
                                uint32_t spatialSize,
                                uint32_t tileLength,
                                float invSpatial)
    {
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t gpc = (totalGroups + blockNum - 1) / blockNum;
        uint32_t s = blockIdx * gpc;
        uint32_t e = s + gpc;
        if (s > totalGroups) s = totalGroups;
        if (e > totalGroups) e = totalGroups;
        this->startGroup = s;
        this->endGroup = e;
        this->spatialSize = spatialSize;
        this->tileLength = tileLength;
        this->invSpatial = invSpatial;
        this->numFullTiles = spatialSize / tileLength;
        this->tailLen = spatialSize - this->numFullTiles * tileLength;

        uint64_t totalElems = static_cast<uint64_t>(totalGroups) * static_cast<uint64_t>(spatialSize);
        xGm.SetGlobalBuffer((__gm__ float*)x, totalElems);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalElems);

        pipe.InitBuffer(inQueue, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(workBuf, 2048);
        pipe.InitBuffer(dstBuf, 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t g = startGroup; g < endGroup; g++) {
            uint64_t base = static_cast<uint64_t>(g) * static_cast<uint64_t>(spatialSize);
            float sum = 0.0f;

            for (uint32_t t = 0; t < numFullTiles; t++) {
                sum += SumTile(base + static_cast<uint64_t>(t) * tileLength, tileLength);
            }
            if (tailLen > 0) {
                sum += SumTile(base + static_cast<uint64_t>(numFullTiles) * tileLength, tailLen);
            }

            float mean = sum * invSpatial;
            float negMean = -mean;

            for (uint32_t t = 0; t < numFullTiles; t++) {
                SubTile(base + static_cast<uint64_t>(t) * tileLength, tileLength, negMean);
            }
            if (tailLen > 0) {
                SubTile(base + static_cast<uint64_t>(numFullTiles) * tileLength, tailLen, negMean);
            }
        }
    }

private:
    __aicore__ inline float SumTile(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        AscendC::DataCopyExtParams cp;
        cp.blockCount = 1;
        cp.blockLen = len * sizeof(float);
        cp.srcStride = 0;
        cp.dstStride = 0;
        AscendC::DataCopyPadExtParams<float> pp;
        pp.isPad = true;
        pp.leftPadding = 0;
        pp.rightPadding = 0;
        pp.paddingValue = 0.0f;
        AscendC::DataCopyPad(xLocal, xGm[offset], cp, pp);
        inQueue.EnQue(xLocal);
        AscendC::LocalTensor<float> xIn = inQueue.DeQue<float>();

        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> dst = dstBuf.Get<float>();
        uint32_t alignedLen = (len + 7) / 8 * 8;
        AscendC::ReduceSum<float>(dst, xIn, work, alignedLen);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

        float v = dst.GetValue(0);
        inQueue.FreeTensor(xIn);
        return v;
    }

    __aicore__ inline void SubTile(uint64_t offset, uint32_t len, float negMean)
    {
        AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
        AscendC::DataCopyExtParams cp;
        cp.blockCount = 1;
        cp.blockLen = len * sizeof(float);
        cp.srcStride = 0;
        cp.dstStride = 0;
        AscendC::DataCopyPadExtParams<float> pp;
        pp.isPad = true;
        pp.leftPadding = 0;
        pp.rightPadding = 0;
        pp.paddingValue = 0.0f;
        AscendC::DataCopyPad(xLocal, xGm[offset], cp, pp);
        inQueue.EnQue(xLocal);
        AscendC::LocalTensor<float> xIn = inQueue.DeQue<float>();

        AscendC::LocalTensor<float> yLocal = outQueue.AllocTensor<float>();
        uint32_t alignedLen = (len + 7) / 8 * 8;
        AscendC::Adds<float>(yLocal, xIn, negMean, alignedLen);
        inQueue.FreeTensor(xIn);
        outQueue.EnQue(yLocal);

        AscendC::LocalTensor<float> yOut = outQueue.DeQue<float>();
        AscendC::DataCopyExtParams cp2;
        cp2.blockCount = 1;
        cp2.blockLen = len * sizeof(float);
        cp2.srcStride = 0;
        cp2.dstStride = 0;
        AscendC::DataCopyPad(yGm[offset], yOut, cp2);
        outQueue.FreeTensor(yOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> dstBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t startGroup;
    uint32_t endGroup;
    uint32_t spatialSize;
    uint32_t tileLength;
    uint32_t numFullTiles;
    uint32_t tailLen;
    float invSpatial;
};

extern "C" __global__ __aicore__ void conv_transpose3d_batch_norm_subtract_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSubMean op;
    op.Init(x, y, tiling_data.totalGroups, tiling_data.spatialSize,
            tiling_data.tileLength, tiling_data.invSpatial);
    op.Process();
}
