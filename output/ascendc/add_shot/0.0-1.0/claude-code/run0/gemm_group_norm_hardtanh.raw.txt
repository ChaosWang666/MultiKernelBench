kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmGroupNormHardtanh {
public:
    __aicore__ inline KernelGemmGroupNormHardtanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t numFeatures, uint32_t numGroups,
                                 uint32_t groupSize, uint32_t rowsPerBlock,
                                 float hardtanhMin, float hardtanhMax, float eps, float invGroupSize)
    {
        this->batchSize = batchSize;
        this->numFeatures = numFeatures;
        this->numGroups = numGroups;
        this->groupSize = groupSize;
        this->rowsPerBlock = rowsPerBlock;
        this->hardtanhMin = hardtanhMin;
        this->hardtanhMax = hardtanhMax;
        this->eps = eps;
        this->invGroupSize = invGroupSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowOffset = blockIdx * rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ float*)x + rowOffset * numFeatures, rowsPerBlock * numFeatures);
        yGm.SetGlobalBuffer((__gm__ float*)y + rowOffset * numFeatures, rowsPerBlock * numFeatures);
        weightGm.SetGlobalBuffer((__gm__ float*)weight, numFeatures);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, numFeatures);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, groupSize * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, groupSize * sizeof(float));
        pipe.InitBuffer(inQueueB, 1, groupSize * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, groupSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, groupSize * sizeof(float));
        pipe.InitBuffer(workBuf, groupSize * sizeof(float));
        pipe.InitBuffer(sumBuf, 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t row = 0; row < rowsPerBlock; row++) {
            for (uint32_t g = 0; g < numGroups; g++) {
                ProcessGroup(row, g);
            }
        }
    }

private:
    __aicore__ inline void ProcessGroup(uint32_t row, uint32_t g)
    {
        uint32_t offset = row * numFeatures + g * groupSize;
        uint32_t wOffset = g * groupSize;

        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], groupSize);
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();

        AscendC::ReduceSum<float>(sumLocal, xLocal, work, groupSize);
        float sumVal = sumLocal.GetValue(0);
        float meanVal = sumVal * invGroupSize;
        float negMean = -meanVal;

        AscendC::Adds<float>(tmp, xLocal, negMean, groupSize);
        AscendC::Mul<float>(work, tmp, tmp, groupSize);
        AscendC::ReduceSum<float>(sumLocal, work, tmp, groupSize);
        float varSum = sumLocal.GetValue(0);
        float varVal = varSum * invGroupSize;
        float rstd = 1.0f / sqrt(varVal + eps);

        AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
        AscendC::DataCopy(wLocal, weightGm[wOffset], groupSize);
        inQueueW.EnQue(wLocal);
        wLocal = inQueueW.DeQue<float>();

        AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
        AscendC::DataCopy(bLocal, biasGm[wOffset], groupSize);
        inQueueB.EnQue(bLocal);
        bLocal = inQueueB.DeQue<float>();

        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::Adds<float>(yLocal, xLocal, negMean, groupSize);
        AscendC::Muls<float>(yLocal, yLocal, rstd, groupSize);
        AscendC::Mul<float>(yLocal, yLocal, wLocal, groupSize);
        AscendC::Add<float>(yLocal, yLocal, bLocal, groupSize);
        AscendC::Maxs<float>(yLocal, yLocal, hardtanhMin, groupSize);
        AscendC::Mins<float>(yLocal, yLocal, hardtanhMax, groupSize);

        outQueueY.EnQue<float>(yLocal);
        yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[offset], yLocal, groupSize);
        outQueueY.FreeTensor(yLocal);

        inQueueX.FreeTensor(xLocal);
        inQueueW.FreeTensor(wLocal);
        inQueueB.FreeTensor(bLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueW;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> biasGm;
    uint32_t batchSize;
    uint32_t numFeatures;
    uint32_t numGroups;
    uint32_t groupSize;
    uint32_t rowsPerBlock;
    float hardtanhMin;
    float hardtanhMax;
    float eps;
    float invGroupSize;
};

extern "C" __global__ __aicore__ void gemm_group_norm_hardtanh_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormHardtanh op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.numFeatures, tiling_data.numGroups,
            tiling_data.groupSize, tiling_data.rowsPerBlock,
            tiling_data.hardtanhMin, tiling_data.hardtanhMax, tiling_data.eps, tiling_data.invGroupSize);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, hardtanh_min, hardtanh_max):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.group_norm = nn.GroupNorm(num_groups, out_features)
        self.hardtanh_min = hardtanh_min
        self.hardtanh_max = hardtanh_max

    def forward(self, x):
        x = self.gemm(x)
        x = custom_ops_lib.gemm_group_norm_hardtanh_custom(x.contiguous(), self.group_norm.weight.contiguous(), self.group_norm.bias.contiguous())
        return x
