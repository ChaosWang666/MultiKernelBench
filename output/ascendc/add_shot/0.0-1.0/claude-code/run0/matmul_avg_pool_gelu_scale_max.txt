kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batchSize, uint32_t featureSize, float scale)
    {
        this->featureSize = featureSize;
        this->scale = scale;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->rowsPerBlock = batchSize / blockNum;
        uint32_t rowOffset = blockIdx * this->rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + rowOffset * featureSize,
                            this->rowsPerBlock * featureSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + rowOffset, this->rowsPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, featureSize * sizeof(float));
        pipe.InitBuffer(outBuf, this->rowsPerBlock * sizeof(float));
        pipe.InitBuffer(workBuf, featureSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, featureSize * sizeof(float));
        pipe.InitBuffer(singleBuf, 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        for (uint32_t i = 0; i < this->rowsPerBlock; i++) {
            CopyIn(i);
            ComputeRow(i, outLocal);
        }
        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::DataCopy(yGm, outLocal, this->rowsPerBlock);
    }

private:
    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * featureSize], featureSize);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void ComputeRow(int32_t i, AscendC::LocalTensor<float>& outLocal)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> single = singleBuf.Get<float>();

        AscendC::Gelu<float>(work, xLocal, featureSize);
        AscendC::Muls(work, work, this->scale, featureSize);
        AscendC::ReduceMax<float>(single, work, tmp, featureSize, false);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

        float val = single.GetValue(0);
        outLocal.SetValue(i, val);

        inQueueX.FreeTensor(xLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf, workBuf, tmpBuf, singleBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t featureSize;
    uint32_t rowsPerBlock;
    float scale;
};

extern "C" __global__ __aicore__ void matmul_avg_pool_gelu_scale_max_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.featureSize, tiling_data.scaleFactor);
    op.Process();
}
