kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelMatmulAvgPoolGeluScaleMax {
public:
    __aicore__ inline KernelMatmulAvgPoolGeluScaleMax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures,
                                uint32_t poolKernelSize, uint32_t pooledLength,
                                float scaleFactor, float invPoolKernelSize)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->poolKernelSize = poolKernelSize;
        this->pooledLength = pooledLength;
        this->scaleFactor = scaleFactor;
        this->invPoolKernelSize = invPoolKernelSize;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        int32_t bsInt = static_cast<int32_t>(batchSize);
        int32_t bnInt = static_cast<int32_t>(blockNum);
        int32_t biInt = static_cast<int32_t>(blockIdx);

        int32_t perBlock = (bsInt + bnInt - 1) / bnInt;
        int32_t startRow = biInt * perBlock;
        int32_t endRow = startRow + perBlock;
        if (endRow > bsInt) endRow = bsInt;
        if (startRow > bsInt) startRow = bsInt;

        this->startRow = static_cast<uint32_t>(startRow);
        this->rowCount = static_cast<uint32_t>(endRow - startRow);

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outFeatures * inFeatures);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, 32);
        pipe.InitBuffer(tmpBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(pooledBuf, pooledLength * sizeof(float));
        pipe.InitBuffer(workBuf, pooledLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (rowCount == 0) return;

        AscendC::LocalTensor<float> biasLocal = inQueueB.AllocTensor<float>();
        AscendC::DataCopy(biasLocal, bGm, outFeatures);
        inQueueB.EnQue(biasLocal);
        biasLocal = inQueueB.DeQue<float>();

        for (uint32_t r = 0; r < rowCount; r++) {
            uint32_t row = startRow + r;
            ComputeRow(row, biasLocal);
        }

        inQueueB.FreeTensor(biasLocal);
    }

private:
    __aicore__ inline void ComputeRow(uint32_t row, AscendC::LocalTensor<float>& biasLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * inFeatures], inFeatures);
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> matmulOut = tmpBuf.Get<float>();

        for (uint32_t o = 0; o < outFeatures; o++) {
            AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
            AscendC::DataCopy(wLocal, wGm[o * inFeatures], inFeatures);
            inQueueW.EnQue(wLocal);
            wLocal = inQueueW.DeQue<float>();

            AscendC::Mul(wLocal, wLocal, xLocal, inFeatures);

            float sum = 0.0f;
            for (uint32_t i = 0; i < inFeatures; i++) {
                sum += wLocal.GetValue(i);
            }
            matmulOut.SetValue(o, sum + biasLocal.GetValue(o));

            inQueueW.FreeTensor(wLocal);
        }

        inQueueX.FreeTensor(xLocal);

        AscendC::LocalTensor<float> pooled = pooledBuf.Get<float>();
        for (uint32_t p = 0; p < pooledLength; p++) {
            float acc = 0.0f;
            uint32_t base = p * poolKernelSize;
            for (uint32_t k = 0; k < poolKernelSize; k++) {
                acc += matmulOut.GetValue(base + k);
            }
            pooled.SetValue(p, acc * invPoolKernelSize);
        }

        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        const float c0 = 0.7978845608028654f;
        const float c1 = 0.044715f;
        for (uint32_t p = 0; p < pooledLength; p++) {
            float v = pooled.GetValue(p);
            float inner = c0 * (v + c1 * v * v * v);
            float e1 = AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                        (float)0.0f);
            (void)e1;
            float ex = expf(inner);
            float enx = expf(-inner);
            float tanh_val = (ex - enx) / (ex + enx);
            float gelu_v = 0.5f * v * (1.0f + tanh_val);
            workLocal.SetValue(p, gelu_v * scaleFactor);
        }

        float maxVal = workLocal.GetValue(0);
        for (uint32_t p = 1; p < pooledLength; p++) {
            float v = workLocal.GetValue(p);
            if (v > maxVal) maxVal = v;
        }

        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        yLocal.SetValue(0, maxVal);
        outQueueY.EnQue<float>(yLocal);
        yLocal = outQueueY.DeQue<float>();

        AscendC::DataCopyExtParams copyParams{1, sizeof(float), 0, 0, 0};
        AscendC::DataCopyPad(yGm[row], yLocal, copyParams);
        outQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline float expf(float x)
    {
        if (x > 80.0f) x = 80.0f;
        if (x < -80.0f) x = -80.0f;
        float result = 1.0f + x / 256.0f;
        for (int i = 0; i < 8; i++) {
            result = result * result;
        }
        return result;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueW, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, pooledBuf, workBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, yGm;
    uint32_t batchSize;
    uint32_t inFeatures;
    uint32_t outFeatures;
    uint32_t poolKernelSize;
    uint32_t pooledLength;
    uint32_t startRow;
    uint32_t rowCount;
    float scaleFactor;
    float invPoolKernelSize;
};

extern "C" __global__ __aicore__ void matmul_avg_pool_gelu_scale_max_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulAvgPoolGeluScaleMax op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures,
            tiling_data.poolKernelSize, tiling_data.pooledLength,
            tiling_data.scaleFactor, tiling_data.invPoolKernelSize);
    op.Process();
}
