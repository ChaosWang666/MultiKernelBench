kernel_src="""
#include "kernel_operator.h"

class KernelMeanReduce {
public:
    __aicore__ inline KernelMeanReduce() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalRows, uint32_t elementsPerRow,
                                uint32_t tileLength, uint32_t tilesPerRow, uint32_t rowsPerCore, float invSize)
    {
        this->elementsPerRow = elementsPerRow;
        this->tileLength = tileLength;
        this->tilesPerRow = tilesPerRow;
        this->rowsPerCore = rowsPerCore;
        this->invSize = invSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = rowsPerCore * blockIdx;

        uint64_t xOffset = static_cast<uint64_t>(startRow) * static_cast<uint64_t>(elementsPerRow);
        xGm.SetGlobalBuffer((__gm__ float *)x + xOffset, static_cast<uint64_t>(rowsPerCore) * static_cast<uint64_t>(elementsPerRow));
        yGm.SetGlobalBuffer((__gm__ float *)y + startRow, rowsPerCore);

        pipe.InitBuffer(inQueueX, 2, tileLength * sizeof(float));
        pipe.InitBuffer(yBuf, ((rowsPerCore * sizeof(float) + 31) / 32) * 32);
        pipe.InitBuffer(workBuf, tileLength * sizeof(float));
        pipe.InitBuffer(sumBuf, 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> yLocal = yBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();

        for (uint32_t r = 0; r < this->rowsPerCore; r++) {
            float acc = 0.0f;
            for (uint32_t t = 0; t < this->tilesPerRow; t++) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                uint64_t gmOffset = static_cast<uint64_t>(r) * static_cast<uint64_t>(this->elementsPerRow)
                                  + static_cast<uint64_t>(t) * static_cast<uint64_t>(this->tileLength);
                AscendC::DataCopy(xLocal, xGm[gmOffset], this->tileLength);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();
                AscendC::ReduceSum<float>(sumLocal, xLocal, workLocal, this->tileLength);
                AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                acc += sumLocal.GetValue(0);
                inQueueX.FreeTensor(xLocal);
            }
            yLocal.SetValue(r, acc * this->invSize);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::DataCopy(yGm, yLocal, this->rowsPerCore);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> yBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t elementsPerRow;
    uint32_t tileLength;
    uint32_t tilesPerRow;
    uint32_t rowsPerCore;
    float invSize;
};

extern "C" __global__ __aicore__ void conv_transpose2d_multiply_global_avg_pool_global_avg_pool_mean_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);

            tiling_data.tileLength, tiling_data.tilesPerRow,
            tiling_data.rowsPerCore, tiling_data.invSize);
    op.Process();
}
