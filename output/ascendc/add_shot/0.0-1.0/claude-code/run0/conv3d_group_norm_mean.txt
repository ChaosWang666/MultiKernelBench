kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t TILE_LENGTH = 4096;
constexpr int32_t OUT_ALIGN = 8;

class KernelConv3dGroupNormMean {
public:
    __aicore__ inline KernelConv3dGroupNormMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t innerSize)
    {
        this->innerSize = innerSize;
        uint32_t blockNum = AscendC::GetBlockNum();
        this->batchPerBlock = batchSize / blockNum;
        uint32_t blockId = AscendC::GetBlockIdx();
        uint32_t batchOffset = blockId * this->batchPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + batchOffset * innerSize,
                            this->batchPerBlock * innerSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + batchOffset,
                            this->batchPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, TILE_LENGTH * sizeof(float));
        pipe.InitBuffer(accumBuf, TILE_LENGTH * sizeof(float));
        pipe.InitBuffer(workBuf, TILE_LENGTH * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, OUT_ALIGN * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> outLocal = outQueueY.AllocTensor<float>();
        AscendC::Duplicate<float>(outLocal, 0.0f, OUT_ALIGN);

        for (uint32_t b = 0; b < this->batchPerBlock; b++) {
            float result = ComputeBatchMean(b);
            outLocal.SetValue(b, result);
        }

        outQueueY.EnQue<float>(outLocal);
        AscendC::LocalTensor<float> out = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm, out, OUT_ALIGN);
        outQueueY.FreeTensor(out);
    }

private:
    __aicore__ inline float ComputeBatchMean(uint32_t batchIdx)
    {
        AscendC::LocalTensor<float> accum = accumBuf.Get<float>();
        AscendC::Duplicate<float>(accum, 0.0f, TILE_LENGTH);

        uint32_t numTiles = (this->innerSize + TILE_LENGTH - 1) / TILE_LENGTH;
        uint32_t baseOffset = batchIdx * this->innerSize;

        for (uint32_t i = 0; i < numTiles; i++) {
            uint32_t curLen = TILE_LENGTH;
            if (i == numTiles - 1) {
                uint32_t rem = this->innerSize - i * TILE_LENGTH;
                if (rem < TILE_LENGTH) {
                    curLen = rem;
                }
            }

            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[baseOffset + i * TILE_LENGTH], curLen);
            inQueueX.EnQue(xLocal);

            xLocal = inQueueX.DeQue<float>();
            AscendC::Add(accum, accum, xLocal, curLen);
            inQueueX.FreeTensor(xLocal);
        }

        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::ReduceSum<float>(work, accum, work, TILE_LENGTH);

        AscendC::TEventID eventId = pipe.AllocEventID<AscendC::HardEvent::V_S>();
        AscendC::SetFlag<AscendC::HardEvent::V_S>(eventId);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(eventId);
        pipe.ReleaseEventID<AscendC::HardEvent::V_S>(eventId);

        float totalSum = work.GetValue(0);
        return totalSum / (float)this->innerSize;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t innerSize;
    uint32_t batchPerBlock;
};

extern "C" __global__ __aicore__ void conv3d_group_norm_mean_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dGroupNormMean op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.innerSize);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, num_groups):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.group_norm = nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        x = self.conv(x)
        x = self.group_norm(x)
        x = x.contiguous()
        return custom_ops_lib.conv3d_group_norm_mean_custom(x)
