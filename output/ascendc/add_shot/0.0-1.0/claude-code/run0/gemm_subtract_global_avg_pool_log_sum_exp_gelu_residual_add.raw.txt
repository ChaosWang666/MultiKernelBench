#include "kernel_operator.h"

constexpr uint32_t ALIGN_N = 8;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR bias, GM_ADDR y,
                                uint32_t totalRows, uint32_t inFeatures)
    {
        this->inFeatures = inFeatures;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t numBlocks = AscendC::GetBlockNum();
        uint32_t rowsPerBlock = (totalRows + numBlocks - 1) / numBlocks;
        uint32_t startRow = blockIdx * rowsPerBlock;
        uint32_t endRow = startRow + rowsPerBlock;
        if (endRow > totalRows) {
            endRow = totalRows;
        }
        this->numRows = (endRow > startRow) ? (endRow - startRow) : 0;

        if (this->numRows > 0) {
            xGm.SetGlobalBuffer((__gm__ float*)x + startRow * inFeatures, this->numRows * inFeatures);
            yGm.SetGlobalBuffer((__gm__ float*)y + startRow * inFeatures, this->numRows * inFeatures);
        }
        wGm.SetGlobalBuffer((__gm__ float*)w, inFeatures);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, ALIGN_N);

        pipe.InitBuffer(wBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(biasBuf, 32);
        pipe.InitBuffer(inQueueX, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(reduceBuf, 32);
        pipe.InitBuffer(scalarBuf, 32);
    }

    __aicore__ inline void Process()
    {
        if (this->numRows == 0) {
            return;
        }

        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();

        AscendC::DataCopy(wLocal, wGm, this->inFeatures);
        AscendC::DataCopy(biasLocal, biasGm, ALIGN_N);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        float effectiveBias = biasLocal.GetValue(0);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

        for (uint32_t i = 0; i < this->numRows; i++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[i * this->inFeatures], this->inFeatures);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::Mul(tmpLocal, xLocal, wLocal, this->inFeatures);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum<float>(reduceLocal, tmpLocal, tmpLocal, this->inFeatures);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            float dotResult = reduceLocal.GetValue(0);
            float preGelu = dotResult + effectiveBias;

            AscendC::Duplicate<float>(scalarLocal, preGelu, ALIGN_N);

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

            AscendC::Muls<float>(reduceLocal, scalarLocal, 0.7071067811865475f, ALIGN_N);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Erf<float>(reduceLocal, reduceLocal, ALIGN_N);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Adds<float>(reduceLocal, reduceLocal, 1.0f, ALIGN_N);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Mul<float>(reduceLocal, reduceLocal, scalarLocal, ALIGN_N);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Muls<float>(reduceLocal, reduceLocal, 0.5f, ALIGN_N);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            float geluResult = reduceLocal.GetValue(0);

            AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
            AscendC::Adds<float>(yLocal, xLocal, geluResult, this->inFeatures);
            outQueueY.EnQue<float>(yLocal);
            inQueueX.FreeTensor(xLocal);

            yLocal = outQueueY.DeQue<float>();
            AscendC::DataCopy(yGm[i * this->inFeatures], yLocal, this->inFeatures);
            outQueueY.FreeTensor(yLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t inFeatures;
    uint32_t numRows;
};

extern "C" __global__ __aicore__ void gemm_subtract_global_avg_pool_log_sum_exp_gelu_residual_add_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {

model_src='''
import torch
import torch_npu
import custom_ops_lib


class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, bias=True):
        super().__init__()
        self.gemm = torch.nn.Linear(in_features, out_features, bias=bias)
        self.subtract = torch.nn.Parameter(torch.randn(out_features))
        self.in_features = in_features
        self.out_features = out_features
        self._cache_device = None
        self._effective_w = None
        self._effective_bias = None

    def _prepare(self, device):
        with torch.no_grad():
            w_col_sum = self.gemm.weight.sum(dim=0)
            if self.gemm.bias is not None:
                bias_scalar = (self.gemm.bias - self.subtract).sum()
            else:
                bias_scalar = -self.subtract.sum()
            out_features_f = float(self.out_features)
            self._effective_w = (w_col_sum / out_features_f).contiguous()
            self._effective_bias = (bias_scalar / out_features_f).flatten().repeat(8).contiguous()
            self._cache_device = device

    def forward(self, x):
        if self._effective_w is None or self._cache_device != x.device:
            self._prepare(x.device)
        return custom_ops_lib.gemm_subtract_global_avg_pool_log_sum_exp_gelu_residual_add_custom(
            x, self._effective_w, self._effective_bias)
