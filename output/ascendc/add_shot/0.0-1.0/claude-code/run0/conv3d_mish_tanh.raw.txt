kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv3dMishTanh {
public:
    __aicore__ inline KernelConv3dMishTanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t blockLength, uint32_t tileLength)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t offset = blockIdx * blockLength;

        uint32_t actualLength = 0;
        if (offset < totalLength) {
            uint32_t remaining = totalLength - offset;
            actualLength = (remaining < blockLength) ? remaining : blockLength;
        }

        this->actualLength = actualLength;
        this->tileLength = tileLength;

        xGm.SetGlobalBuffer((__gm__ float *)x + offset, (actualLength == 0) ? 1 : actualLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + offset, (actualLength == 0) ? 1 : actualLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf1, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->actualLength == 0) {
            return;
        }

        uint32_t loopCount = this->actualLength / this->tileLength;
        uint32_t tailLength = this->actualLength - loopCount * this->tileLength;

        for (uint32_t i = 0; i < loopCount; i++) {
            CopyIn(i, this->tileLength);
            Compute(this->tileLength);
            CopyOut(i, this->tileLength);
        }
        if (tailLength > 0) {
            CopyIn(loopCount, tailLength);
            Compute(tailLength);
            CopyOut(loopCount, tailLength);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * this->tileLength], len);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();

        AscendC::Exp(tmp1, xLocal, len);
        AscendC::Adds(tmp1, tmp1, (float)1.0, len);
        AscendC::Ln(tmp1, tmp1, len);
        AscendC::Tanh(tmp2, tmp1, len);
        AscendC::Mul(tmp1, xLocal, tmp2, len);
        AscendC::Tanh(yLocal, tmp1, len);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress, uint32_t len)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * this->tileLength], yLocal, len);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t actualLength;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv3d_mish_tanh_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dMishTanh op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.blockLength, tiling_data.tileLength);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride=1, padding=0):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
