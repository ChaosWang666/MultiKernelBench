kernel_src="""
#include "kernel_operator.h"

class KernelConv2dAvgPoolSigmoidSum {
public:
    __aicore__ inline KernelConv2dAvgPoolSigmoidSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                uint32_t inH, uint32_t inW, uint32_t kH, uint32_t kW,
                                uint32_t convH, uint32_t convW, uint32_t poolK,
                                uint32_t outH, uint32_t outW, uint32_t totalBatches,
                                float invPoolArea)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inH = inH;
        this->inW = inW;
        this->kH = kH;
        this->kW = kW;
        this->convH = convH;
        this->convW = convW;
        this->poolK = poolK;
        this->outH = outH;
        this->outW = outW;
        this->totalBatches = totalBatches;
        this->invPoolArea = invPoolArea;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        int32_t total = static_cast<int32_t>(totalBatches);
        int32_t bNum = static_cast<int32_t>(blockNum);
        int32_t bIdx = static_cast<int32_t>(blockIdx);
        int32_t perCore = (total + bNum - 1) / bNum;
        int32_t startB = bIdx * perCore;
        int32_t endB = startB + perCore;
        if (endB > total) endB = total;
        if (startB > total) startB = total;
        this->startBatch = static_cast<uint32_t>(startB);
        this->endBatch = static_cast<uint32_t>(endB);

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * inH * inW);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outChannels * inChannels * kH * kW);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t b = this->startBatch; b < this->endBatch; b++) {
            float batchSum = 0.0f;
            for (uint32_t oc = 0; oc < this->outChannels; oc++) {
                float biasVal = bGm.GetValue(oc);
                for (uint32_t oh = 0; oh < this->outH; oh++) {
                    for (uint32_t ow = 0; ow < this->outW; ow++) {
                        float poolSum = 0.0f;
                        for (uint32_t ph = 0; ph < this->poolK; ph++) {
                            for (uint32_t pw = 0; pw < this->poolK; pw++) {
                                uint32_t ch = oh * this->poolK + ph;
                                uint32_t cw = ow * this->poolK + pw;
                                float convVal = biasVal;
                                for (uint32_t ic = 0; ic < this->inChannels; ic++) {
                                    for (uint32_t kh = 0; kh < this->kH; kh++) {
                                        for (uint32_t kw = 0; kw < this->kW; kw++) {
                                            uint32_t ih = ch + kh;
                                            uint32_t iw = cw + kw;
                                            uint32_t xIdx = ((b * this->inChannels + ic) * this->inH + ih) * this->inW + iw;
                                            uint32_t wIdx = ((oc * this->inChannels + ic) * this->kH + kh) * this->kW + kw;
                                            float xv = xGm.GetValue(xIdx);
                                            float wv = wGm.GetValue(wIdx);
                                            convVal += xv * wv;
                                        }
                                    }
                                }
                                poolSum += convVal;
                            }
                        }
                        float avgVal = poolSum * this->invPoolArea;
                        float negVal = -avgVal;
                        float expVal = AscendC::ScalarGetExpf(negVal);
                        float sigVal = 1.0f / (1.0f + expVal);
                        batchSum += sigVal;
                    }
                }
            }
            yGm.SetValue(b, batchSum);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inH;
    uint32_t inW;
    uint32_t kH;
    uint32_t kW;
    uint32_t convH;
    uint32_t convW;
    uint32_t poolK;
    uint32_t outH;
    uint32_t outW;
    uint32_t totalBatches;
    float invPoolArea;
    uint32_t startBatch;
    uint32_t endBatch;
};

extern "C" __global__ __aicore__ void conv2d_avg_pool_sigmoid_sum_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dAvgPoolSigmoidSum op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inH, tiling_data.inW, tiling_data.kH, tiling_data.kW,
            tiling_data.convH, tiling_data.convW, tiling_data.poolK,
            tiling_data.outH, tiling_data.outW, tiling_data.totalBatches,
            tiling_data.invPoolArea);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, pool_kernel_size):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.avg_pool = torch.nn.AvgPool2d(pool_kernel_size)

    def forward(self, x):
        weight = self.conv.weight.contiguous()
        bias = self.conv.bias.contiguous()
        x = x.contiguous()
        return custom_ops_lib.conv2d_avg_pool_sigmoid_sum_custom(x, weight, bias)
