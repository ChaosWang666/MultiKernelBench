kernel_src="""
#include "kernel_operator.h"

class KernelConv2dAvgPoolSigmoidSum {
public:
    __aicore__ inline KernelConv2dAvgPoolSigmoidSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inHeight, uint32_t inWidth, uint32_t kernelSize,
                                 uint32_t poolKernelSize, uint32_t convHeight, uint32_t convWidth,
                                 uint32_t poolHeight, uint32_t poolWidth)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->kernelSize = kernelSize;
        this->poolKernelSize = poolKernelSize;
        this->convHeight = convHeight;
        this->convWidth = convWidth;
        this->poolHeight = poolHeight;
        this->poolWidth = poolWidth;

        this->blockIdx = AscendC::GetBlockIdx();
        this->blockNum = AscendC::GetBlockNum();

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * inHeight * inWidth);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outChannels * inChannels * kernelSize * kernelSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize);
    }

    __aicore__ inline void Process()
    {
        uint32_t batchesPerBlock = (batchSize + blockNum - 1) / blockNum;
        uint32_t startBatch = blockIdx * batchesPerBlock;
        uint32_t endBatch = startBatch + batchesPerBlock;
        if (endBatch > batchSize) endBatch = batchSize;

        float invPoolArea = 1.0f / (float)(poolKernelSize * poolKernelSize);

        for (uint32_t b = startBatch; b < endBatch; b++) {
            float sumTotal = 0.0f;
            for (uint32_t oc = 0; oc < outChannels; oc++) {
                float biasVal = bGm.GetValue(oc);
                for (uint32_t ph = 0; ph < poolHeight; ph++) {
                    for (uint32_t pw = 0; pw < poolWidth; pw++) {
                        float poolSum = 0.0f;
                        for (uint32_t py = 0; py < poolKernelSize; py++) {
                            for (uint32_t px = 0; px < poolKernelSize; px++) {
                                uint32_t oh = ph * poolKernelSize + py;
                                uint32_t ow = pw * poolKernelSize + px;
                                float convVal = biasVal;
                                for (uint32_t ic = 0; ic < inChannels; ic++) {
                                    for (uint32_t kh = 0; kh < kernelSize; kh++) {
                                        for (uint32_t kw = 0; kw < kernelSize; kw++) {
                                            uint32_t ih = oh + kh;
                                            uint32_t iw = ow + kw;
                                            uint64_t xIdx = ((uint64_t)b * inChannels + ic) * inHeight * inWidth + ih * inWidth + iw;
                                            uint64_t wIdx = ((uint64_t)oc * inChannels + ic) * kernelSize * kernelSize + kh * kernelSize + kw;
                                            convVal += xGm.GetValue(xIdx) * wGm.GetValue(wIdx);
                                        }
                                    }
                                }
                                poolSum += convVal;
                            }
                        }
                        float avgVal = poolSum * invPoolArea;
                        float sigVal = 1.0f / (1.0f + AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                            (float)__builtin_exp(-avgVal)));
                        sumTotal += sigVal;
                    }
                }
            }
            yGm.SetValue(b, sumTotal);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inHeight;
    uint32_t inWidth;
    uint32_t kernelSize;
    uint32_t poolKernelSize;
    uint32_t convHeight;
    uint32_t convWidth;
    uint32_t poolHeight;
    uint32_t poolWidth;
    uint32_t blockIdx;
    uint32_t blockNum;
};

extern "C" __global__ __aicore__ void conv2d_avg_pool_sigmoid_sum_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dAvgPoolSigmoidSum op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inHeight, tiling_data.inWidth, tiling_data.kernelSize,
            tiling_data.poolKernelSize, tiling_data.convHeight, tiling_data.convWidth,
            tiling_data.poolHeight, tiling_data.poolWidth);
    op.Process();
}

import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, pool_kernel_size):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.avg_pool = nn.AvgPool2d(pool_kernel_size)
        self.pool_kernel_size = pool_kernel_size

    def forward(self, x):
        weight = self.conv.weight
        bias = self.conv.bias
        return custom_ops_lib.conv2d_avg_pool_sigmoid_sum_custom(x, weight, bias)
