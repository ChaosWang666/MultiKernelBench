kernel_src="""
#include "kernel_operator.h"

class KernelConv2dReluHardSwish {
public:
    __aicore__ inline KernelConv2dReluHardSwish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
        uint32_t batch, uint32_t inCh, uint32_t outCh, uint32_t inH, uint32_t inW,
        uint32_t kH, uint32_t kW, uint32_t outH, uint32_t outW)
    {
        this->batch = batch;
        this->inCh = inCh;
        this->outCh = outCh;
        this->inH = inH;
        this->inW = inW;
        this->kH = kH;
        this->kW = kW;
        this->outH = outH;
        this->outW = outW;

        uint32_t totalNC = batch * outCh;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t base = totalNC / blockNum;
        uint32_t rem = totalNC % blockNum;
        this->ncPerBlock = base + (blockIdx < rem ? 1u : 0u);
        this->ncStart = blockIdx * base + (blockIdx < rem ? blockIdx : rem);

        xGm.SetGlobalBuffer((__gm__ float *)x, batch * inCh * inH * inW);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outCh * inCh * kH * kW);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outCh);
        yGm.SetGlobalBuffer((__gm__ float *)y, batch * outCh * outH * outW);
    }

    __aicore__ inline void Process()
    {
        uint32_t inCHW = this->inCh * this->inH * this->inW;
        uint32_t inHW = this->inH * this->inW;
        uint32_t kCHW = this->inCh * this->kH * this->kW;
        uint32_t kHW = this->kH * this->kW;
        uint32_t outCHW = this->outCh * this->outH * this->outW;
        uint32_t outHW = this->outH * this->outW;

        for (uint32_t ncIdx = 0; ncIdx < this->ncPerBlock; ncIdx++) {
            uint32_t globalNC = this->ncStart + ncIdx;
            uint32_t n = globalNC / this->outCh;
            uint32_t co = globalNC % this->outCh;

            float biasVal = bGm.GetValue(co);

            for (uint32_t ho = 0; ho < this->outH; ho++) {
                for (uint32_t wo = 0; wo < this->outW; wo++) {
                    float sum = biasVal;
                    for (uint32_t ci = 0; ci < this->inCh; ci++) {
                        uint32_t xBase = n * inCHW + ci * inHW;
                        uint32_t wBase = co * kCHW + ci * kHW;
                        for (uint32_t kh = 0; kh < this->kH; kh++) {
                            uint32_t xRow = xBase + (ho + kh) * this->inW + wo;
                            uint32_t wRow = wBase + kh * this->kW;
                            for (uint32_t kw = 0; kw < this->kW; kw++) {
                                float xv = xGm.GetValue(xRow + kw);
                                float wv = wGm.GetValue(wRow + kw);
                                sum = sum + xv * wv;
                            }
                        }
                    }
                    float relu = sum > 0.0f ? sum : 0.0f;
                    float hs = (relu + 3.0f) / 6.0f;
                    if (hs < 0.0f) hs = 0.0f;
                    if (hs > 1.0f) hs = 1.0f;
                    float result = relu * hs;

                    uint32_t yIdx = n * outCHW + co * outHW + ho * this->outW + wo;
                    yGm.SetValue(yIdx, result);
                }
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batch;
    uint32_t inCh;
    uint32_t outCh;
    uint32_t inH;
    uint32_t inW;
    uint32_t kH;
    uint32_t kW;
    uint32_t outH;
    uint32_t outW;
    uint32_t ncPerBlock;
    uint32_t ncStart;
};

extern "C" __global__ __aicore__ void conv2d_relu_hard_swish_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dReluHardSwish op;
    op.Init(x, weight, bias, y,
            tiling_data.batch, tiling_data.inCh, tiling_data.outCh,
            tiling_data.inH, tiling_data.inW,
            tiling_data.kH, tiling_data.kW,
            tiling_data.outH, tiling_data.outW);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)

    def forward(self, x):
        w = self.conv.weight.contiguous()
        b = self.conv.bias.contiguous()
        return custom_ops_lib.conv2d_relu_hard_swish_custom(x.contiguous(), w, b)
