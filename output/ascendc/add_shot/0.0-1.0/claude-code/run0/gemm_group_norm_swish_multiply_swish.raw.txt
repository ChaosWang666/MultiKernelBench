kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelGemmGroupNormSwishMultiplySwish {
public:
    __aicore__ inline KernelGemmGroupNormSwishMultiplySwish() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mw, GM_ADDR y,
                                 uint32_t batchSize, uint32_t outFeatures)
    {
        this->outFeatures = outFeatures;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t rowsPerCoreVal = batchSize / blockNum;
        this->rowsPerCore = rowsPerCoreVal;
        uint32_t rowStart = blockIdx * rowsPerCoreVal;

        xGm.SetGlobalBuffer((__gm__ float *)x + rowStart * outFeatures, rowsPerCoreVal * outFeatures);
        yGm.SetGlobalBuffer((__gm__ float *)y + rowStart * outFeatures, rowsPerCoreVal * outFeatures);
        mwGm.SetGlobalBuffer((__gm__ float *)mw, outFeatures);

        pipe.InitBuffer(xQue, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(yQue, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(mwBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(sigBuf, outFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> mwL = mwBuf.Get<float>();
        AscendC::DataCopy(mwL, mwGm, outFeatures);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

        for (uint32_t r = 0; r < rowsPerCore; r++) {
            CopyIn(r);
            Compute(mwL);
            CopyOut(r);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t r)
    {
        AscendC::LocalTensor<float> xL = xQue.AllocTensor<float>();
        AscendC::DataCopy(xL, xGm[r * outFeatures], outFeatures);
        xQue.EnQue(xL);
    }

    __aicore__ inline void Compute(AscendC::LocalTensor<float>& mwL)
    {
        AscendC::LocalTensor<float> xL = xQue.DeQue<float>();
        AscendC::LocalTensor<float> yL = yQue.AllocTensor<float>();
        AscendC::LocalTensor<float> sigL = sigBuf.Get<float>();

        // Swish #1: y = x * sigmoid(x) = x / (1 + exp(-x))
        AscendC::Muls(sigL, xL, -1.0f, outFeatures);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Exp(sigL, sigL, outFeatures);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Adds(sigL, sigL, 1.0f, outFeatures);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Div(yL, xL, sigL, outFeatures);
        AscendC::PipeBarrier<PIPE_V>();

        // Multiply by mw element-wise
        AscendC::Mul(yL, yL, mwL, outFeatures);
        AscendC::PipeBarrier<PIPE_V>();

        // Swish #2: y = y * sigmoid(y) = y / (1 + exp(-y))
        AscendC::Muls(sigL, yL, -1.0f, outFeatures);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Exp(sigL, sigL, outFeatures);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Adds(sigL, sigL, 1.0f, outFeatures);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Div(yL, yL, sigL, outFeatures);
        AscendC::PipeBarrier<PIPE_V>();

        xQue.FreeTensor(xL);
        yQue.EnQue<float>(yL);
    }

    __aicore__ inline void CopyOut(uint32_t r)
    {
        AscendC::LocalTensor<float> yL = yQue.DeQue<float>();
        AscendC::DataCopy(yGm[r * outFeatures], yL, outFeatures);
        yQue.FreeTensor(yL);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> xQue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> yQue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mwBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sigBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> mwGm;
    uint32_t outFeatures;
    uint32_t rowsPerCore;
};

extern "C" __global__ __aicore__ void gemm_group_norm_swish_multiply_swish_custom(
    GM_ADDR x, GM_ADDR mw, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormSwishMultiplySwish op;
    op.Init(x, mw, y, tiling_data.batchSize, tiling_data.outFeatures);
    op.Process();
}

import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, multiply_weight_shape):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.group_norm = nn.GroupNorm(num_groups, out_features)
        self.multiply_weight = nn.Parameter(torch.randn(multiply_weight_shape))

    def forward(self, x):
        x = self.gemm(x)
        x = self.group_norm(x)
        x = custom_ops_lib.gemm_group_norm_swish_multiply_swish_custom(
            x.contiguous(), self.multiply_weight.contiguous())
        return x
