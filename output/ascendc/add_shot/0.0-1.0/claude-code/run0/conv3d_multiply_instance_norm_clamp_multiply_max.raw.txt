#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR multiplier, GM_ADDR y,
        uint32_t batchSize, uint32_t channels, uint32_t spatial,
        uint32_t nPerBlock, uint32_t spatialTile, uint32_t spatialLoops,
        float invSpatial, float eps)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->spatial = spatial;
        this->spatialTile = spatialTile;
        this->spatialLoops = spatialLoops;
        this->invSpatial = invSpatial;
        this->eps = eps;
        this->clampMin = -1.0f;
        this->clampMax = 1.0f;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t perSample = channels * spatial;
        uint32_t ns = blockIdx * nPerBlock;
        uint32_t ne = ns + nPerBlock;
        if (ne > batchSize) ne = batchSize;
        this->nStart = ns;
        this->nEnd = ne;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * perSample);
        yGm.SetGlobalBuffer((__gm__ float*)y, batchSize * spatial);
        mGm.SetGlobalBuffer((__gm__ float*)multiplier, channels);

        pipe.InitBuffer(inQueX, 1, spatialTile * sizeof(float));
        pipe.InitBuffer(outQueY, 1, spatialTile * sizeof(float));
        pipe.InitBuffer(sqBuf, spatialTile * sizeof(float));
        pipe.InitBuffer(reduceBuf, 512 * sizeof(float));
        pipe.InitBuffer(mBuf, 64 * sizeof(float));
        pipe.InitBuffer(sumBuf, 64 * sizeof(float));
        pipe.InitBuffer(sumSqBuf, 64 * sizeof(float));
        pipe.InitBuffer(meanBuf, 64 * sizeof(float));
        pipe.InitBuffer(varBuf, 64 * sizeof(float));
        pipe.InitBuffer(invStdBuf, 64 * sizeof(float));
        pipe.InitBuffer(oneBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process() {
        AscendC::LocalTensor<float> mLocal = mBuf.Get<float>();
        uint32_t mCount = ((channels + 7u) / 8u) * 8u;
        if (mCount < 8u) mCount = 8u;
        if (mCount > 64u) mCount = 64u;
        AscendC::DataCopy(mLocal, mGm, mCount);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(static_cast<event_t>(0));
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(static_cast<event_t>(0));
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(static_cast<event_t>(0));
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(static_cast<event_t>(0));

        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> sumSqLocal = sumSqBuf.Get<float>();
        AscendC::LocalTensor<float> meanLocal = meanBuf.Get<float>();
        AscendC::LocalTensor<float> varLocal = varBuf.Get<float>();
        AscendC::LocalTensor<float> invStdLocal = invStdBuf.Get<float>();
        AscendC::LocalTensor<float> oneLocal = oneBuf.Get<float>();
        AscendC::LocalTensor<float> sqLocal = sqBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        AscendC::Duplicate(oneLocal, 1.0f, 64);

        for (uint32_t n = nStart; n < nEnd; n++) {
            for (uint32_t c = 0; c < channels; c++) {
                float mc = mLocal.GetValue(c);
                float sumC = 0.0f;
                float sumSqC = 0.0f;

                for (uint32_t t = 0; t < spatialLoops; t++) {
                    AscendC::LocalTensor<float> xLocal = inQueX.AllocTensor<float>();
                    AscendC::DataCopy(xLocal, xGm[(n * channels + c) * spatial + t * spatialTile], spatialTile);
                    inQueX.EnQue(xLocal);
                    xLocal = inQueX.DeQue<float>();

                    AscendC::Muls(xLocal, xLocal, mc, spatialTile);
                    AscendC::PipeBarrier<PIPE_V>();

                    AscendC::Mul(sqLocal, xLocal, xLocal, spatialTile);
                    AscendC::PipeBarrier<PIPE_V>();

                    AscendC::ReduceSum<float>(reduceLocal, xLocal, reduceLocal[256], spatialTile);
                    AscendC::SetFlag<AscendC::HardEvent::V_S>(static_cast<event_t>(0));
                    AscendC::WaitFlag<AscendC::HardEvent::V_S>(static_cast<event_t>(0));
                    sumC += reduceLocal.GetValue(0);

                    AscendC::ReduceSum<float>(reduceLocal, sqLocal, reduceLocal[256], spatialTile);
                    AscendC::SetFlag<AscendC::HardEvent::V_S>(static_cast<event_t>(1));
                    AscendC::WaitFlag<AscendC::HardEvent::V_S>(static_cast<event_t>(1));
                    sumSqC += reduceLocal.GetValue(0);

                    inQueX.FreeTensor(xLocal);
                }

                sumLocal.SetValue(c, sumC);
                sumSqLocal.SetValue(c, sumSqC);
            }

            AscendC::SetFlag<AscendC::HardEvent::S_V>(static_cast<event_t>(0));
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(static_cast<event_t>(0));

            AscendC::Muls(meanLocal, sumLocal, invSpatial, 64);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Mul(varLocal, meanLocal, meanLocal, 64);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Muls(sumSqLocal, sumSqLocal, invSpatial, 64);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Sub(varLocal, sumSqLocal, varLocal, 64);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Adds(varLocal, varLocal, eps, 64);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Sqrt(varLocal, varLocal, 64);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Div(invStdLocal, oneLocal, varLocal, 64);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::SetFlag<AscendC::HardEvent::V_S>(static_cast<event_t>(2));
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(static_cast<event_t>(2));

            for (uint32_t t = 0; t < spatialLoops; t++) {
                AscendC::LocalTensor<float> outLocal = outQueY.AllocTensor<float>();
                AscendC::Duplicate(outLocal, -1.0e30f, spatialTile);
                AscendC::PipeBarrier<PIPE_V>();

                for (uint32_t c = 0; c < channels; c++) {
                    float mc = mLocal.GetValue(c);
                    float negMean = -meanLocal.GetValue(c);
                    float invStdC = invStdLocal.GetValue(c);

                    AscendC::LocalTensor<float> xLocal = inQueX.AllocTensor<float>();
                    AscendC::DataCopy(xLocal, xGm[(n * channels + c) * spatial + t * spatialTile], spatialTile);
                    inQueX.EnQue(xLocal);
                    xLocal = inQueX.DeQue<float>();

                    AscendC::Muls(xLocal, xLocal, mc, spatialTile);
                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::Adds(xLocal, xLocal, negMean, spatialTile);
                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::Muls(xLocal, xLocal, invStdC, spatialTile);
                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::Mins(xLocal, xLocal, clampMax, spatialTile);
                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::Maxs(xLocal, xLocal, clampMin, spatialTile);
                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::Muls(xLocal, xLocal, mc, spatialTile);
                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::Max(outLocal, outLocal, xLocal, spatialTile);
                    AscendC::PipeBarrier<PIPE_V>();

                    inQueX.FreeTensor(xLocal);
                }

                outQueY.EnQue<float>(outLocal);
                AscendC::LocalTensor<float> yOut = outQueY.DeQue<float>();
                AscendC::DataCopy(yGm[n * spatial + t * spatialTile], yOut, spatialTile);
                outQueY.FreeTensor(yOut);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sqBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumSqBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> meanBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> varBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> invStdBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> oneBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> mGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t spatial;
    uint32_t spatialTile;
    uint32_t spatialLoops;
    uint32_t nStart;
    uint32_t nEnd;
    float invSpatial;
    float eps;
    float clampMin;
    float clampMax;
};
