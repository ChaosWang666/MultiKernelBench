kernel_src="""
#include "kernel_operator.h"

constexpr int32_t OUT_GROUP = 8;

class KernelConvTrMaxPoolHardtanhMeanTanh {
public:
    __aicore__ inline KernelConvTrMaxPoolHardtanhMeanTanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                 uint32_t totalNC, uint32_t ncPerCore,
                                 uint32_t hIn, uint32_t wIn,
                                 float hardtanhMin, float hardtanhMax,
                                 float poolNorm)
    {
        this->ncPerCore = ncPerCore;
        this->hIn = hIn;
        this->wIn = wIn;
        this->hardtanhMin = hardtanhMin;
        this->hardtanhMax = hardtanhMax;
        this->poolNorm = poolNorm;
        this->hOut = hIn / 2;
        this->wOut = wIn / 2;
        this->elemsPerNC = hIn * wIn;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startNC = blockIdx * ncPerCore;

        xGm.SetGlobalBuffer((__gm__ float*)x + startNC * this->elemsPerNC, ncPerCore * this->elemsPerNC);
        zGm.SetGlobalBuffer((__gm__ float*)z + startNC, ncPerCore);

        pipe.InitBuffer(inQueueX, 1, 2 * wIn * sizeof(float));
        pipe.InitBuffer(tmpHBuf, wIn * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, OUT_GROUP * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        uint32_t numGroups = ncPerCore / OUT_GROUP;
        for (uint32_t g = 0; g < numGroups; g++) {
            ProcessGroup(g);
        }
    }

private:
    __aicore__ inline void ProcessGroup(uint32_t group)
    {
        AscendC::LocalTensor<float> outLocal = outQueueZ.AllocTensor<float>();

        for (int32_t k = 0; k < OUT_GROUP; k++) {
            uint32_t ncIdx = group * OUT_GROUP + static_cast<uint32_t>(k);
            float runningSum = 0.0f;

            for (uint32_t i = 0; i < hOut; i++) {
                AscendC::LocalTensor<float> rowLocal = inQueueX.AllocTensor<float>();
                uint32_t offset = ncIdx * elemsPerNC + (2u * i) * wIn;
                AscendC::DataCopy(rowLocal, xGm[offset], 2 * wIn);
                inQueueX.EnQue(rowLocal);
                rowLocal = inQueueX.DeQue<float>();

                AscendC::LocalTensor<float> tmpHLocal = tmpHBuf.Get<float>();
                AscendC::Max(tmpHLocal, rowLocal, rowLocal[wIn], wIn);

                AscendC::PipeBarrier<PIPE_ALL>();

                int32_t wOutInt = static_cast<int32_t>(wOut);
                for (int32_t j = 0; j < wOutInt; j++) {
                    int32_t idx = 2 * j;
                    float a = tmpHLocal.GetValue(idx);
                    float b = tmpHLocal.GetValue(idx + 1);
                    float v = (a > b) ? a : b;
                    if (v > hardtanhMax) v = hardtanhMax;
                    if (v < hardtanhMin) v = hardtanhMin;
                    runningSum += v;
                }

                AscendC::PipeBarrier<PIPE_ALL>();
                inQueueX.FreeTensor(rowLocal);
            }

            outLocal.SetValue(k, runningSum);
        }

        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Muls(outLocal, outLocal, poolNorm, OUT_GROUP);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Tanh(outLocal, outLocal, OUT_GROUP);
        AscendC::PipeBarrier<PIPE_ALL>();

        outQueueZ.EnQue(outLocal);
        outLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[group * OUT_GROUP], outLocal, OUT_GROUP);
        outQueueZ.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpHBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t ncPerCore;
    uint32_t hIn;
    uint32_t wIn;
    uint32_t hOut;
    uint32_t wOut;
    uint32_t elemsPerNC;
    float hardtanhMin;
    float hardtanhMax;
    float poolNorm;
};

extern "C" __global__ __aicore__ void conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTrMaxPoolHardtanhMeanTanh op;
    op.Init(x, z, tiling_data.totalNC, tiling_data.ncPerCore,
            tiling_data.hIn, tiling_data.wIn,
            tiling_data.hardtanhMin, tiling_data.hardtanhMax,
            tiling_data.poolNorm);
    op.Process();
}
