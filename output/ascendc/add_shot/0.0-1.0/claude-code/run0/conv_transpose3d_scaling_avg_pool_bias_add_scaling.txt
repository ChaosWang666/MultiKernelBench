kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedBiasScale {
public:
    __aicore__ inline KernelFusedBiasScale() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR scale, GM_ADDR z,
                                 uint32_t groupCount, uint32_t spatialSize,
                                 uint32_t channelSize, uint32_t tileLength)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        this->groupsPerBlock = groupCount / blockNum;
        this->startGroup = blockIdx * this->groupsPerBlock;
        this->spatialSize = spatialSize;
        this->channelSize = channelSize;
        this->tileLength = tileLength;
        this->numTilesPerGroup = (spatialSize + tileLength - 1) / tileLength;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, (uint64_t)groupCount * spatialSize);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z, (uint64_t)groupCount * spatialSize);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias, channelSize);
        scaleGm.SetGlobalBuffer((__gm__ DTYPE_SCALE *)scale, 1);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, tileLength * sizeof(DTYPE_Z));

        uint32_t biasBufBytes = ((channelSize * sizeof(DTYPE_BIAS) + 31) / 32) * 32;
        if (biasBufBytes < 32) biasBufBytes = 32;
        pipe.InitBuffer(biasBuf, biasBufBytes);
        pipe.InitBuffer(scaleBuf, 32);

        LoadParams();
    }

    __aicore__ inline void LoadParams()
    {
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();
        AscendC::LocalTensor<DTYPE_SCALE> scaleLocal = scaleBuf.Get<DTYPE_SCALE>();

        AscendC::DataCopyExtParams biasParams{1, (uint32_t)(channelSize * sizeof(DTYPE_BIAS)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<DTYPE_BIAS> biasPad{false, 0, 0, 0};
        AscendC::DataCopyPad(biasLocal, biasGm, biasParams, biasPad);

        AscendC::DataCopyExtParams scaleParams{1, (uint32_t)sizeof(DTYPE_SCALE), 0, 0, 0};
        AscendC::DataCopyPadExtParams<DTYPE_SCALE> scalePad{false, 0, 0, 0};
        AscendC::DataCopyPad(scaleLocal, scaleGm, scaleParams, scalePad);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);

        this->scale2 = scaleLocal.GetValue(0);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = biasBuf.Get<DTYPE_BIAS>();

        for (uint32_t g = 0; g < this->groupsPerBlock; g++) {
            uint32_t globalGroup = this->startGroup + g;
            uint32_t channelIdx = globalGroup % this->channelSize;
            float biasVal = biasLocal.GetValue(channelIdx);
            float mulVal = this->scale2;
            float addVal = biasVal * this->scale2;

            uint64_t groupOffset = (uint64_t)globalGroup * spatialSize;

            for (uint32_t t = 0; t < this->numTilesPerGroup; t++) {
                uint32_t curLen = this->tileLength;
                if (t == this->numTilesPerGroup - 1) {
                    curLen = this->spatialSize - t * this->tileLength;
                }
                uint64_t offset = groupOffset + (uint64_t)t * this->tileLength;
                CopyIn(offset, curLen);
                Compute(curLen, mulVal, addVal);
                CopyOut(offset, curLen);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopyExtParams copyParams{1, (uint32_t)(len * sizeof(DTYPE_X)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<DTYPE_X> padParams{false, 0, 0, 0};
        AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t len, float mulVal, float addVal)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();

        AscendC::Muls(zLocal, xLocal, (DTYPE_X)mulVal, len);
        AscendC::Adds(zLocal, zLocal, (DTYPE_X)addVal, len);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopyExtParams copyParams{1, (uint32_t)(len * sizeof(DTYPE_Z)), 0, 0, 0};
        AscendC::DataCopyPad(zGm[offset], zLocal, copyParams);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scaleBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_SCALE> scaleGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t groupsPerBlock;
    uint32_t startGroup;
    uint32_t spatialSize;
    uint32_t channelSize;
    uint32_t tileLength;
    uint32_t numTilesPerGroup;
    float scale2;
};

extern "C" __global__ __aicore__ void conv_transpose3d_scaling_avg_pool_bias_add_scaling_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR scale, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedBiasScale op;
    op.Init(x, bias, scale, z,
            tiling_data.groupCount, tiling_data.spatialSize,
            tiling_data.channelSize, tiling_data.tileLength);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, scale1, scale2, bias_shape):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.scale1 = nn.Parameter(torch.tensor(scale1))
        self.avg_pool = nn.AvgPool3d(kernel_size=2)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.scale2 = nn.Parameter(torch.tensor(scale2))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x * self.scale1
        x = self.avg_pool(x)
        x = x.contiguous()
        bias_flat = self.bias.view(-1).contiguous()
        scale_tensor = self.scale2.reshape(1).contiguous()
        x = custom_ops_lib.conv_transpose3d_scaling_avg_pool_bias_add_scaling_custom(x, bias_flat, scale_tensor)
        return x
