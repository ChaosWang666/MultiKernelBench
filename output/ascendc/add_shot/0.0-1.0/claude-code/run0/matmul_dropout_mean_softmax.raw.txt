kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelMatmulDropoutMeanSoftmax {
public:
    __aicore__ inline KernelMatmulDropoutMeanSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures, uint32_t tileNum)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->tileNum = tileNum;

        int32_t blockIdx = static_cast<int32_t>(AscendC::GetBlockIdx());
        int32_t blockNum = static_cast<int32_t>(AscendC::GetBlockNum());

        int32_t rowsPerBlock = (static_cast<int32_t>(batchSize) + blockNum - 1) / blockNum;
        this->startRow = blockIdx * rowsPerBlock;
        int32_t endRow = this->startRow + rowsPerBlock;
        if (endRow > static_cast<int32_t>(batchSize)) {
            endRow = static_cast<int32_t>(batchSize);
        }
        this->rowCount = endRow - this->startRow;
        if (this->rowCount < 0) {
            this->rowCount = 0;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outFeatures * inFeatures);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize * outFeatures);

        this->tileLen = outFeatures / tileNum;

        pipe.InitBuffer(inQueueX, BUFFER_NUM, inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, this->tileLen * inFeatures * sizeof(float));
        pipe.InitBuffer(inQueueB, BUFFER_NUM, this->tileLen * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLen * sizeof(float));

        pipe.InitBuffer(tmpBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(rowBuf, outFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (int32_t r = 0; r < this->rowCount; r++) {
            int32_t globalRow = this->startRow + r;
            ProcessRow(globalRow);
        }
    }

private:
    __aicore__ inline void ProcessRow(int32_t globalRow)
    {
        // Step 1: Load x row
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[globalRow * inFeatures], inFeatures);
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        // rowBuf holds the full output row (outFeatures)
        AscendC::LocalTensor<float> rowLocal = rowBuf.Get<float>();

        // Step 2: For each tile of output features, compute matmul + bias
        for (int32_t t = 0; t < static_cast<int32_t>(tileNum); t++) {
            AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
            AscendC::DataCopy(wLocal, wGm[t * tileLen * inFeatures], tileLen * inFeatures);
            inQueueW.EnQue(wLocal);
            wLocal = inQueueW.DeQue<float>();

            AscendC::LocalTensor<float> bLocal = inQueueB.AllocTensor<float>();
            AscendC::DataCopy(bLocal, bGm[t * tileLen], tileLen);
            inQueueB.EnQue(bLocal);
            bLocal = inQueueB.DeQue<float>();

            AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

            for (int32_t i = 0; i < static_cast<int32_t>(tileLen); i++) {
                // dot product between xLocal and wLocal[i, :]
                AscendC::Mul(tmpLocal, xLocal, wLocal[i * inFeatures], inFeatures);
                float sum = 0.0f;
                // Use ReduceSum via work-path: simple scalar loop fallback on UB
                AscendC::LocalTensor<float> redLocal = outQueueZ.AllocTensor<float>();
                AscendC::ReduceSum<float>(redLocal, tmpLocal, redLocal, inFeatures);
                float val = redLocal.GetValue(0);
                outQueueZ.FreeTensor(redLocal);
                float biasVal = bLocal.GetValue(i);
                rowLocal.SetValue(t * tileLen + i, val + biasVal);
            }

            inQueueW.FreeTensor(wLocal);
            inQueueB.FreeTensor(bLocal);
        }

        inQueueX.FreeTensor(xLocal);

        // Step 3: During inference dropout is identity; softmax over outFeatures
        // Compute max
        float maxVal = rowLocal.GetValue(0);
        for (int32_t i = 1; i < static_cast<int32_t>(outFeatures); i++) {
            float v = rowLocal.GetValue(i);
            if (v > maxVal) {
                maxVal = v;
            }
        }

        // subtract max and exp, accumulate sum
        AscendC::LocalTensor<float> expLocal = rowLocal;
        float sumExp = 0.0f;
        for (int32_t i = 0; i < static_cast<int32_t>(outFeatures); i++) {
            float v = rowLocal.GetValue(i) - maxVal;
            float e = AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(0.0f);
            // exp via Taylor is inefficient; use AscendC::Exp on the whole tensor instead
            // We'll handle exp by writing back and then using vector exp below
            rowLocal.SetValue(i, v);
        }

        // Vectorized exp on rowLocal
        AscendC::Exp(rowLocal, rowLocal, outFeatures);

        // Sum of exps
        for (int32_t i = 0; i < static_cast<int32_t>(outFeatures); i++) {
            sumExp += rowLocal.GetValue(i);
        }

        float invSum = 1.0f / sumExp;

        // Divide & write out in tiles
        for (int32_t t = 0; t < static_cast<int32_t>(tileNum); t++) {
            AscendC::LocalTensor<float> outLocal = outQueueZ.AllocTensor<float>();
            for (int32_t i = 0; i < static_cast<int32_t>(tileLen); i++) {
                outLocal.SetValue(i, rowLocal.GetValue(t * tileLen + i) * invSum);
            }
            outQueueZ.EnQue<float>(outLocal);
            AscendC::LocalTensor<float> outDeq = outQueueZ.DeQue<float>();
            AscendC::DataCopy(zGm[globalRow * outFeatures + t * tileLen], outDeq, tileLen);
            outQueueZ.FreeTensor(outDeq);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueW, inQueueB;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, rowBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t inFeatures;
    uint32_t outFeatures;
    uint32_t tileNum;
    uint32_t tileLen;
    int32_t startRow;
    int32_t rowCount;
};

extern "C" __global__ __aicore__ void matmul_dropout_mean_softmax_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulDropoutMeanSoftmax op;
    op.Init(x, weight, bias, z,
            tiling_data.batchSize, tiling_data.inFeatures,
            tiling_data.outFeatures, tiling_data.tileNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, dropout_p):
        super(ModelNew, self).__init__()
        self.in_features = in_features
        self.out_features = out_features
        self.dropout_p = dropout_p
        self.weight = torch.nn.Parameter(torch.empty(out_features, in_features))
        self.bias = torch.nn.Parameter(torch.empty(out_features))
        torch.nn.init.kaiming_uniform_(self.weight, a=5 ** 0.5)
        fan_in = in_features
        bound = 1.0 / (fan_in ** 0.5)
        torch.nn.init.uniform_(self.bias, -bound, bound)

    def forward(self, x):
        return custom_ops_lib.matmul_dropout_mean_softmax_custom(x, self.weight, self.bias)
