kernel_src="""
#include "kernel_operator.h"

class KernelMatmulSwishSumGroupNorm {
public:
    __aicore__ inline KernelMatmulSwishSumGroupNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR linear_bias,
                                 GM_ADDR bias, GM_ADDR gn_weight, GM_ADDR gn_bias,
                                 GM_ADDR z,
                                 uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures,
                                 uint32_t numGroups, uint32_t groupSize, uint32_t rowsPerBlock,
                                 float invGroupSize, float epsilon)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->groupSize = groupSize;
        this->rowsPerBlock = rowsPerBlock;
        this->invGroupSize = invGroupSize;
        this->epsilon = epsilon;

        int32_t blockIdx = static_cast<int32_t>(AscendC::GetBlockIdx());
        this->rowStart = static_cast<uint32_t>(blockIdx) * rowsPerBlock;
        this->rowEnd = this->rowStart + rowsPerBlock;
        if (this->rowEnd > batchSize) {
            this->rowEnd = batchSize;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float*)weight, outFeatures * inFeatures);
        lbGm.SetGlobalBuffer((__gm__ float*)linear_bias, outFeatures);
        bGm.SetGlobalBuffer((__gm__ float*)bias, outFeatures);
        gnwGm.SetGlobalBuffer((__gm__ float*)gn_weight, outFeatures);
        gnbGm.SetGlobalBuffer((__gm__ float*)gn_bias, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z, batchSize * outFeatures);

        pipe.InitBuffer(xQueue, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(wQueue, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(lbQueue, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(bQueue, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(gnwQueue, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(gnbQueue, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(outBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(zQueue, 1, outFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->rowStart >= this->rowEnd) {
            return;
        }

        AscendC::LocalTensor<float> lbLocal = lbQueue.AllocTensor<float>();
        AscendC::DataCopy(lbLocal, lbGm, this->outFeatures);
        lbQueue.EnQue(lbLocal);
        lbLocal = lbQueue.DeQue<float>();

        AscendC::LocalTensor<float> bLocal = bQueue.AllocTensor<float>();
        AscendC::DataCopy(bLocal, bGm, this->outFeatures);
        bQueue.EnQue(bLocal);
        bLocal = bQueue.DeQue<float>();

        AscendC::LocalTensor<float> gnwLocal = gnwQueue.AllocTensor<float>();
        AscendC::DataCopy(gnwLocal, gnwGm, this->outFeatures);
        gnwQueue.EnQue(gnwLocal);
        gnwLocal = gnwQueue.DeQue<float>();

        AscendC::LocalTensor<float> gnbLocal = gnbQueue.AllocTensor<float>();
        AscendC::DataCopy(gnbLocal, gnbGm, this->outFeatures);
        gnbQueue.EnQue(gnbLocal);
        gnbLocal = gnbQueue.DeQue<float>();

        for (uint32_t row = this->rowStart; row < this->rowEnd; ++row) {
            ProcessRow(row, lbLocal, bLocal, gnwLocal, gnbLocal);
        }

        lbQueue.FreeTensor(lbLocal);
        bQueue.FreeTensor(bLocal);
        gnwQueue.FreeTensor(gnwLocal);
        gnbQueue.FreeTensor(gnbLocal);
    }

private:
    __aicore__ inline void ProcessRow(uint32_t row,
                                       AscendC::LocalTensor<float>& lbLocal,
                                       AscendC::LocalTensor<float>& bLocal,
                                       AscendC::LocalTensor<float>& gnwLocal,
                                       AscendC::LocalTensor<float>& gnbLocal)
    {
        AscendC::LocalTensor<float> xLocal = xQueue.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * this->inFeatures], this->inFeatures);
        xQueue.EnQue(xLocal);
        xLocal = xQueue.DeQue<float>();

        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();

        for (uint32_t oc = 0; oc < this->outFeatures; ++oc) {
            AscendC::LocalTensor<float> wLocal = wQueue.AllocTensor<float>();
            AscendC::DataCopy(wLocal, wGm[oc * this->inFeatures], this->inFeatures);
            wQueue.EnQue(wLocal);
            wLocal = wQueue.DeQue<float>();

            AscendC::Mul(wLocal, wLocal, xLocal, this->inFeatures);
            float acc = 0.0f;
            for (uint32_t i = 0; i < this->inFeatures; ++i) {
                acc += wLocal.GetValue(i);
            }
            wQueue.FreeTensor(wLocal);

            float val = acc + lbLocal.GetValue(oc);
            float sig = 1.0f / (1.0f + AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                                          __expf(-val)));
            float swish = val * sig;
            float summed = swish + bLocal.GetValue(oc);
            outLocal.SetValue(oc, summed);
        }

        xQueue.FreeTensor(xLocal);

        AscendC::LocalTensor<float> zLocal = zQueue.AllocTensor<float>();

        for (uint32_t g = 0; g < this->numGroups; ++g) {
            uint32_t base = g * this->groupSize;
            float mean = 0.0f;
            for (uint32_t i = 0; i < this->groupSize; ++i) {
                mean += outLocal.GetValue(base + i);
            }
            mean *= this->invGroupSize;
            float var = 0.0f;
            for (uint32_t i = 0; i < this->groupSize; ++i) {
                float d = outLocal.GetValue(base + i) - mean;
                var += d * d;
            }
            var *= this->invGroupSize;
            float invStd = 1.0f / AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                                      __builtin_sqrtf(var + this->epsilon));
            for (uint32_t i = 0; i < this->groupSize; ++i) {
                float v = outLocal.GetValue(base + i);
                float norm = (v - mean) * invStd;
                float res = norm * gnwLocal.GetValue(base + i) + gnbLocal.GetValue(base + i);
                zLocal.SetValue(base + i, res);
            }
        }

        zQueue.EnQue(zLocal);
        zLocal = zQueue.DeQue<float>();
        AscendC::DataCopy(zGm[row * this->outFeatures], zLocal, this->outFeatures);
        zQueue.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> xQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> wQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> lbQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> bQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> gnwQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> gnbQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> zQueue;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> lbGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> gnwGm;
    AscendC::GlobalTensor<float> gnbGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t batchSize;
    uint32_t inFeatures;
    uint32_t outFeatures;
    uint32_t numGroups;
    uint32_t groupSize;
    uint32_t rowsPerBlock;
    uint32_t rowStart;
    uint32_t rowEnd;
    float invGroupSize;
    float epsilon;
};

extern "C" __global__ __aicore__ void matmul_swish_sum_group_norm_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR linear_bias,
    GM_ADDR bias, GM_ADDR gn_weight, GM_ADDR gn_bias,
    GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSwishSumGroupNorm op;
    op.Init(x, weight, linear_bias, bias, gn_weight, gn_bias, z,
            tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures,
            tiling_data.numGroups, tiling_data.groupSize, tiling_data.rowsPerBlock,
            tiling_data.invGroupSize, tiling_data.epsilon);
    op.Process();
}
