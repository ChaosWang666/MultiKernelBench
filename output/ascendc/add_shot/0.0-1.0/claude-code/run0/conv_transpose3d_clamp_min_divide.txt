#include "kernel_operator.h"

class KernelConvTranspose3dClampMinDivide {
public:
    __aicore__ inline KernelConvTranspose3dClampMinDivide() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inDepth, uint32_t inHeight, uint32_t inWidth,
                                 uint32_t outDepth, uint32_t outHeight, uint32_t outWidth,
                                 uint32_t kernelSize, uint32_t stride, uint32_t padding,
                                 float minValue, float invDivisor,
                                 uint32_t totalOutElements, uint32_t coreNum)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inDepth = inDepth;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->outDepth = outDepth;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->kernelSize = kernelSize;
        this->stride = stride;
        this->padding = padding;
        this->minValue = minValue;
        this->invDivisor = invDivisor;
        this->totalOutElements = totalOutElements;
        this->coreNum = coreNum;

        xGm.SetGlobalBuffer((__gm__ float *)x);
        wGm.SetGlobalBuffer((__gm__ float *)weight);
        bGm.SetGlobalBuffer((__gm__ float *)bias);
        yGm.SetGlobalBuffer((__gm__ float *)y);
    }

    __aicore__ inline void Process()
    {
        int32_t blockIdx = static_cast<int32_t>(AscendC::GetBlockIdx());
        int32_t coreNumInt = static_cast<int32_t>(this->coreNum);
        int32_t totalInt = static_cast<int32_t>(this->totalOutElements);
        int32_t perCore = (totalInt + coreNumInt - 1) / coreNumInt;
        int32_t startIdx = blockIdx * perCore;
        int32_t endIdx = startIdx + perCore;
        if (endIdx > totalInt) {
            endIdx = totalInt;
        }

        int32_t outChannelsInt = static_cast<int32_t>(this->outChannels);
        int32_t inChannelsInt = static_cast<int32_t>(this->inChannels);
        int32_t outDepthInt = static_cast<int32_t>(this->outDepth);
        int32_t outHeightInt = static_cast<int32_t>(this->outHeight);
        int32_t outWidthInt = static_cast<int32_t>(this->outWidth);
        int32_t inDepthInt = static_cast<int32_t>(this->inDepth);
        int32_t inHeightInt = static_cast<int32_t>(this->inHeight);
        int32_t inWidthInt = static_cast<int32_t>(this->inWidth);
        int32_t kernelSizeInt = static_cast<int32_t>(this->kernelSize);
        int32_t strideInt = static_cast<int32_t>(this->stride);
        int32_t paddingInt = static_cast<int32_t>(this->padding);

        int32_t outHW = outHeightInt * outWidthInt;
        int32_t outDHW = outDepthInt * outHW;
        int32_t outCDHW = outChannelsInt * outDHW;

        int32_t inHW = inHeightInt * inWidthInt;
        int32_t inDHW = inDepthInt * inHW;
        int32_t inCDHW = inChannelsInt * inDHW;

        int32_t kernelSq = kernelSizeInt * kernelSizeInt;
        int32_t kernelCube = kernelSizeInt * kernelSq;

        for (int32_t idx = startIdx; idx < endIdx; idx++) {
            int32_t ow = idx % outWidthInt;
            int32_t tmp = idx / outWidthInt;
            int32_t oh = tmp % outHeightInt;
            tmp = tmp / outHeightInt;
            int32_t od = tmp % outDepthInt;
            tmp = tmp / outDepthInt;
            int32_t oc = tmp % outChannelsInt;
            int32_t n = tmp / outChannelsInt;

            float sum = bGm.GetValue(oc);

            int32_t idBase = od + paddingInt;
            int32_t ihBase = oh + paddingInt;
            int32_t iwBase = ow + paddingInt;

            for (int32_t ic = 0; ic < inChannelsInt; ic++) {
                for (int32_t kd = 0; kd < kernelSizeInt; kd++) {
                    int32_t idNum = idBase - kd;
                    if (idNum < 0) continue;
                    if (idNum % strideInt != 0) continue;
                    int32_t id = idNum / strideInt;
                    if (id >= inDepthInt) continue;

                    for (int32_t kh = 0; kh < kernelSizeInt; kh++) {
                        int32_t ihNum = ihBase - kh;
                        if (ihNum < 0) continue;
                        if (ihNum % strideInt != 0) continue;
                        int32_t ih = ihNum / strideInt;
                        if (ih >= inHeightInt) continue;

                        for (int32_t kw = 0; kw < kernelSizeInt; kw++) {
                            int32_t iwNum = iwBase - kw;
                            if (iwNum < 0) continue;
                            if (iwNum % strideInt != 0) continue;
                            int32_t iw = iwNum / strideInt;
                            if (iw >= inWidthInt) continue;

                            int32_t xOffset = n * inCDHW + ic * inDHW + id * inHW + ih * inWidthInt + iw;
                            int32_t wOffset = ic * outChannelsInt * kernelCube + oc * kernelCube + kd * kernelSq + kh * kernelSizeInt + kw;

                            float xVal = xGm.GetValue(xOffset);
                            float wVal = wGm.GetValue(wOffset);
                            sum += xVal * wVal;
                        }
                    }
                }
            }

            if (sum < this->minValue) {
                sum = this->minValue;
            }
            sum = sum * this->invDivisor;

            yGm.SetValue(idx, sum);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inDepth;
    uint32_t inHeight;
    uint32_t inWidth;
    uint32_t outDepth;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t kernelSize;
    uint32_t stride;
    uint32_t padding;
    float minValue;
    float invDivisor;
    uint32_t totalOutElements;
    uint32_t coreNum;
};

extern "C" __global__ __aicore__ void conv_transpose3d_clamp_min_divide_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dClampMinDivide op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inDepth, tiling_data.inHeight, tiling_data.inWidth,
            tiling_data.outDepth, tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.kernelSize, tiling_data.stride, tiling_data.padding,
            tiling_data.minValue, tiling_data.invDivisor,
            tiling_data.totalOutElements, tiling_data.coreNum);
    op.Process();
}
