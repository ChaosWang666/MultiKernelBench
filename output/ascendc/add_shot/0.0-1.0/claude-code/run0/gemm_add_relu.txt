kernel_src = """
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;
using namespace matmul;

template <typename aType, typename bType, typename cType, typename biasType>
class KernelGemmAddRelu {
public:
    __aicore__ inline KernelGemmAddRelu() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                GM_ADDR workspace, const TCubeTiling &tiling,
                                uint32_t M, uint32_t N, uint32_t K, TPipe *pipe)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        xGm.SetGlobalBuffer((__gm__ aType *)x);
        wGm.SetGlobalBuffer((__gm__ bType *)weight);
        biasGm.SetGlobalBuffer((__gm__ biasType *)bias);
        yGm.SetGlobalBuffer((__gm__ cType *)y);

        int32_t offsetA = 0;
        int32_t offsetB = 0;
        int32_t offsetC = 0;
        int32_t offsetBias = 0;
        int32_t tailM = 0;
        int32_t tailN = 0;
        CalcOffset(GetBlockIdx(), tiling, offsetA, offsetB, offsetC, offsetBias, tailM, tailN);

        xGmLocal = xGm[offsetA];
        wGmLocal = wGm[offsetB];
        yGmLocal = yGm[offsetC];
        biasGmLocal = biasGm[offsetBias];

        this->tailM = (tailM == 0) ? tiling.singleCoreM : tailM;
        this->tailN = (tailN == 0) ? tiling.singleCoreN : tailN;

        mm.SetSubBlockIdx(0);
        mm.Init(&tiling, pipe);
    }

    __aicore__ inline void Process(TPipe *pipe)
    {
        if (GetBlockIdx() >= mm.GetTiling().usedCoreNum) {
            return;
        }
        mm.SetTensorA(xGmLocal);
        mm.SetTensorB(wGmLocal, true);
        mm.SetBias(biasGmLocal);
        mm.SetTail(tailM, tailN);
        mm.IterateAll(yGmLocal, 0, false, false);
        mm.End();

        PostProcessRelu();
    }

    Matmul<MatmulType<TPosition::GM, CubeFormat::ND, aType>,
           MatmulType<TPosition::GM, CubeFormat::ND, bType, true>,
           MatmulType<TPosition::GM, CubeFormat::ND, cType>,
           MatmulType<TPosition::GM, CubeFormat::ND, biasType>> mm;

private:
    __aicore__ inline void CalcOffset(int32_t blockIdx, const TCubeTiling &tiling,
                                       int32_t &offsetA, int32_t &offsetB,
                                       int32_t &offsetC, int32_t &offsetBias,
                                       int32_t &tailM, int32_t &tailN)
    {
        auto mSingleBlocks = Ceil(tiling.M, tiling.singleCoreM);
        auto mCoreIndx = blockIdx % mSingleBlocks;
        auto nCoreIndx = blockIdx / mSingleBlocks;

        offsetA = mCoreIndx * tiling.Ka * tiling.singleCoreM;
        offsetB = nCoreIndx * tiling.singleCoreN * tiling.Kb;
        offsetC = mCoreIndx * tiling.N * tiling.singleCoreM + nCoreIndx * tiling.singleCoreN;
        offsetBias = nCoreIndx * tiling.singleCoreN;

        int gmUseM = tiling.M - mCoreIndx * tiling.singleCoreM;
        tailM = gmUseM < tiling.singleCoreM ? gmUseM : tiling.singleCoreM;
        int gmUseN = tiling.N - nCoreIndx * tiling.singleCoreN;
        tailN = gmUseN < tiling.singleCoreN ? gmUseN : tiling.singleCoreN;
    }

    __aicore__ inline uint32_t Ceil(uint32_t a, uint32_t b) {
        return (a + b - 1) / b;
    }

    __aicore__ inline void PostProcessRelu()
    {
    }

    GlobalTensor<aType> xGm;
    GlobalTensor<bType> wGm;
    GlobalTensor<biasType> biasGm;
    GlobalTensor<cType> yGm;
    GlobalTensor<aType> xGmLocal;
    GlobalTensor<bType> wGmLocal;
    GlobalTensor<biasType> biasGmLocal;
    GlobalTensor<cType> yGmLocal;
    uint32_t M, N, K;
    int32_t tailM, tailN;
};

class KernelReluPost {
public:
    __aicore__ inline KernelReluPost() {}
    __aicore__ inline void Init(GM_ADDR y, uint32_t totalLength, TPipe *pipe)
    {
        this->totalLength = totalLength;
        uint32_t blockNum = GetBlockNum();
        this->blockLength = totalLength / blockNum;
        uint32_t tail = totalLength % blockNum;
        uint32_t offset;
        if (GetBlockIdx() < tail) {
            this->blockLength += 1;
            offset = GetBlockIdx() * this->blockLength;
        } else {
            offset = GetBlockIdx() * this->blockLength + tail;
        }
        yGm.SetGlobalBuffer((__gm__ float *)y + offset, this->blockLength);

        this->tileLength = 8192;
        this->tileNum = (this->blockLength + this->tileLength - 1) / this->tileLength;

        pipe->InitBuffer(inQueue, 2, this->tileLength * sizeof(float));
        pipe->InitBuffer(outQueue, 2, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->tileNum; i++) {
            uint32_t curLen = (i == this->tileNum - 1) ?
                (this->blockLength - i * this->tileLength) : this->tileLength;
            if (curLen == 0) break;
            LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
            DataCopy(xLocal, yGm[i * this->tileLength], (curLen + 7) / 8 * 8);
            inQueue.EnQue(xLocal);

            xLocal = inQueue.DeQue<float>();
            LocalTensor<float> yLocal = outQueue.AllocTensor<float>();
            Relu(yLocal, xLocal, curLen);
            outQueue.EnQue<float>(yLocal);
            inQueue.FreeTensor(xLocal);

            yLocal = outQueue.DeQue<float>();
            DataCopy(yGm[i * this->tileLength], yLocal, (curLen + 7) / 8 * 8);
            outQueue.FreeTensor(yLocal);
        }
    }

private:
    GlobalTensor<float> yGm;
    TQue<TPosition::VECIN, 2> inQueue;
    TQue<TPosition::VECOUT, 2> outQueue;
    uint32_t totalLength;
    uint32_t blockLength;
    uint32_t tileLength;
    uint32_t tileNum;
};

extern "C" __global__ __aicore__ void gemm_add_relu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    if ASCEND_IS_AIC {
        TPipe pipe;
        KernelGemmAddRelu<float, float, float, float> op;
        op.Init(x, weight, bias, y, workspace, tiling_data.cubeTilingData,
                tiling_data.M, tiling_data.N, tiling_data.K, &pipe);
        op.Process(&pipe);
    }
    SyncAll();
    if ASCEND_IS_AIV {
        TPipe pipe2;
        KernelReluPost relu;
        relu.Init(y, tiling_data.M * tiling_data.N, &pipe2);
        relu.Process();
    }
}

model_src = '''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, bias_shape):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features, bias=False)
        self.bias = nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        weight = self.gemm.weight
        bias = self.bias.view(-1).contiguous()
        return custom_ops_lib.gemm_add_relu_custom(x, weight, bias)
