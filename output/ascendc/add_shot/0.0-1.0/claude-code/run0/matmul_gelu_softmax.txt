kernel_src="""
#include "kernel_operator.h"

class KernelMatmulGeluSoftmax {
public:
    __aicore__ inline KernelMatmulGeluSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                uint32_t M, uint32_t N, uint32_t K)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t rowsPerBlock = (M + blockNum - 1) / blockNum;
        this->rowStart = blockIdx * rowsPerBlock;
        uint32_t rowEndCand = this->rowStart + rowsPerBlock;
        this->rowEnd = rowEndCand > M ? M : rowEndCand;

        xGm.SetGlobalBuffer((__gm__ float*)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float*)weight, N * K);
        bGm.SetGlobalBuffer((__gm__ float*)bias, N);
        yGm.SetGlobalBuffer((__gm__ float*)y, M * N);

        pipe.InitBuffer(xBuf, K * sizeof(float));
        pipe.InitBuffer(wBuf, K * sizeof(float));
        pipe.InitBuffer(bBuf, N * sizeof(float));
        pipe.InitBuffer(outBuf, N * sizeof(float));
        pipe.InitBuffer(tmpBuf, K * sizeof(float));
        pipe.InitBuffer(redBuf, 64 * sizeof(float));
        pipe.InitBuffer(auxBuf, N * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> bLocal = bBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> red = redBuf.Get<float>();
        AscendC::LocalTensor<float> aux = auxBuf.Get<float>();

        AscendC::DataCopy(bLocal, bGm, this->N);

        for (uint32_t row = this->rowStart; row < this->rowEnd; row++) {
            AscendC::DataCopy(xLocal, xGm[row * this->K], this->K);

            for (uint32_t j = 0; j < this->N; j++) {
                AscendC::DataCopy(wLocal, wGm[j * this->K], this->K);
                AscendC::Mul(tmp, xLocal, wLocal, this->K);
                AscendC::ReduceSum(red, tmp, red, this->K);
                float dot = red.GetValue(0);
                float bv = bLocal.GetValue(j);
                outLocal.SetValue(j, dot + bv);
            }

            ComputeGelu(outLocal, tmp, aux, this->N);
            ComputeSoftmax(outLocal, tmp, aux, red, this->N);

            AscendC::DataCopy(yGm[row * this->N], outLocal, this->N);
        }
    }

private:
    __aicore__ inline void ComputeGelu(AscendC::LocalTensor<float>& x,
                                        AscendC::LocalTensor<float>& t1,
                                        AscendC::LocalTensor<float>& t2,
                                        uint32_t n)
    {
        // GELU approx: 0.5 * x * (1 + tanh(0.7978845608 * (x + 0.044715 * x^3)))
        AscendC::Mul(t1, x, x, n);
        AscendC::Mul(t1, t1, x, n);
        AscendC::Muls(t1, t1, (float)0.044715, n);
        AscendC::Add(t1, t1, x, n);
        AscendC::Muls(t1, t1, (float)1.5957691216, n); // 2 * 0.7978845608
        AscendC::Exp(t1, t1, n);
        AscendC::Adds(t2, t1, (float)1.0, n);
        AscendC::Adds(t1, t1, (float)-1.0, n);
        AscendC::Div(t1, t1, t2, n);
        AscendC::Adds(t1, t1, (float)1.0, n);
        AscendC::Muls(t1, t1, (float)0.5, n);
        AscendC::Mul(x, x, t1, n);
    }

    __aicore__ inline void ComputeSoftmax(AscendC::LocalTensor<float>& x,
                                           AscendC::LocalTensor<float>& t1,
                                           AscendC::LocalTensor<float>& t2,
                                           AscendC::LocalTensor<float>& red,
                                           uint32_t n)
    {
        AscendC::ReduceMax(red, x, red, n, false);
        float mx = red.GetValue(0);
        AscendC::Adds(x, x, -mx, n);
        AscendC::Exp(x, x, n);
        AscendC::ReduceSum(red, x, red, n);
        float sm = red.GetValue(0);
        float inv = (float)1.0 / sm;
        AscendC::Muls(x, x, inv, n);
    }

    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, wBuf, bBuf, outBuf, tmpBuf, redBuf, auxBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, yGm;
    uint32_t M, N, K;
    uint32_t rowStart, rowEnd;
};

extern "C" __global__ __aicore__ void matmul_gelu_softmax_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulGeluSoftmax op;
    op.Init(x, weight, bias, y, tiling_data.M, tiling_data.N, tiling_data.K);
    op.Process();
}
