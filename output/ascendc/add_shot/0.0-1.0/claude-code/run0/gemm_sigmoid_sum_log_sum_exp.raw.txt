kernel_src="""
#include "kernel_operator.h"

class KernelLogSumExp {
public:
    __aicore__ inline KernelLogSumExp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t batchSize, uint32_t features, uint32_t rowsPerCore)
    {
        this->features = features;
        this->rowsPerCore = rowsPerCore;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerCore;

        xGm.SetGlobalBuffer((__gm__ float *)x + startRow * features, rowsPerCore * features);
        zGm.SetGlobalBuffer((__gm__ float *)z + startRow, rowsPerCore);

        pipe.InitBuffer(xBuf, features * sizeof(float));
        pipe.InitBuffer(tBuf, features * sizeof(float));
        pipe.InitBuffer(wBuf, features * sizeof(float));
        pipe.InitBuffer(mBuf, rowsPerCore * sizeof(float));
        pipe.InitBuffer(sBuf, rowsPerCore * sizeof(float));
        pipe.InitBuffer(rBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> tLocal = tBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> mLocal = mBuf.Get<float>();
        AscendC::LocalTensor<float> sLocal = sBuf.Get<float>();
        AscendC::LocalTensor<float> rLocal = rBuf.Get<float>();

        for (uint32_t i = 0; i < rowsPerCore; i++) {
            AscendC::DataCopy(xLocal, xGm[i * features], features);
            AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

            AscendC::ReduceMax<float>(rLocal, xLocal, wLocal, features, false);
            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            float maxVal = rLocal.GetValue(0);
            mLocal.SetValue(i, maxVal);
            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

            AscendC::Adds(tLocal, xLocal, -maxVal, features);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Exp(tLocal, tLocal, features);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum<float>(rLocal, tLocal, wLocal, features);
            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            float sumVal = rLocal.GetValue(0);
            sLocal.SetValue(i, sumVal);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        AscendC::Ln(sLocal, sLocal, rowsPerCore);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Add(mLocal, mLocal, sLocal, rowsPerCore);
        AscendC::SetFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);

        AscendC::DataCopy(zGm, mLocal, rowsPerCore);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, tBuf, wBuf, mBuf, sBuf, rBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t features;
    uint32_t rowsPerCore;
};

extern "C" __global__ __aicore__ void gemm_sigmoid_sum_log_sum_exp_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelLogSumExp op;
    op.Init(x, z, tiling_data.batchSize, tiling_data.features, tiling_data.rowsPerCore);
    op.Process();
}
