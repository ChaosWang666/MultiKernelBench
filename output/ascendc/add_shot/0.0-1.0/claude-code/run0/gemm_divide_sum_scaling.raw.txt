kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t TILE_SIZE = 4096;

class KernelGemmDivideSumScaling {
public:
    __aicore__ inline KernelGemmDivideSumScaling() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR z,
                                  uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize,
                                  uint32_t batchPerBlock, uint32_t tilesPerRow, float scaleHalf)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->batchPerBlock = batchPerBlock;
        this->tilesPerRow = tilesPerRow;
        this->scaleHalf = scaleHalf;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->batchStart = blockIdx * batchPerBlock;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inputSize);
        weightGm.SetGlobalBuffer((__gm__ float*)weight, hiddenSize * inputSize);
        zGm.SetGlobalBuffer((__gm__ float*)z, batchSize);

        pipe.InitBuffer(colSumBuf, inputSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, TILE_SIZE * sizeof(float));
        pipe.InitBuffer(accBuf, TILE_SIZE * sizeof(float));
        pipe.InitBuffer(workBuf, TILE_SIZE * sizeof(float));
        pipe.InitBuffer(reduceBuf, 32 * sizeof(float));
        pipe.InitBuffer(outBuf, batchPerBlock * sizeof(float));
        pipe.InitBuffer(inQueue, BUFFER_NUM, TILE_SIZE * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> colSum = colSumBuf.Get<float>();
        AscendC::Duplicate(colSum, 0.0f, static_cast<int32_t>(inputSize));

        for (uint32_t h = 0; h < hiddenSize; h++) {
            for (uint32_t t = 0; t < tilesPerRow; t++) {
                AscendC::LocalTensor<float> weightTile = inQueue.AllocTensor<float>();
                AscendC::DataCopy(weightTile, weightGm[h * inputSize + t * TILE_SIZE], TILE_SIZE);
                inQueue.EnQue(weightTile);

                weightTile = inQueue.DeQue<float>();
                AscendC::Add(colSum[t * TILE_SIZE], colSum[t * TILE_SIZE], weightTile, TILE_SIZE);
                inQueue.FreeTensor(weightTile);
            }
        }

        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> accVec = accBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduced = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> outTile = outBuf.Get<float>();

        for (uint32_t b = 0; b < batchPerBlock; b++) {
            uint32_t batchIdx = batchStart + b;

            AscendC::Duplicate(accVec, 0.0f, static_cast<int32_t>(TILE_SIZE));

            for (uint32_t t = 0; t < tilesPerRow; t++) {
                AscendC::LocalTensor<float> xTile = inQueue.AllocTensor<float>();
                AscendC::DataCopy(xTile, xGm[batchIdx * inputSize + t * TILE_SIZE], TILE_SIZE);
                inQueue.EnQue(xTile);

                xTile = inQueue.DeQue<float>();
                AscendC::Mul(tmp, xTile, colSum[t * TILE_SIZE], TILE_SIZE);
                inQueue.FreeTensor(xTile);
                AscendC::Add(accVec, accVec, tmp, TILE_SIZE);
            }

            AscendC::ReduceSum<float>(reduced, accVec, work, static_cast<int32_t>(TILE_SIZE));

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

            float val = reduced.GetValue(0);
            outTile.SetValue(b, val * scaleHalf);

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        AscendC::DataCopy(zGm[batchStart], outTile, batchPerBlock);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> colSumBuf, tmpBuf, accBuf, workBuf, reduceBuf, outBuf;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::GlobalTensor<float> xGm, weightGm, zGm;
    uint32_t batchSize, inputSize, hiddenSize;
    uint32_t batchStart, batchPerBlock, tilesPerRow;
    float scaleHalf;
};

extern "C" __global__ __aicore__ void gemm_divide_sum_scaling_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmDivideSumScaling op;
    op.Init(x, weight, z,
             tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize,
             tiling_data.batchPerBlock, tiling_data.tilesPerRow, tiling_data.scaleHalf);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, input_size, hidden_size, scaling_factor):
        super().__init__()
        self.weight = torch.nn.Parameter(torch.randn(hidden_size, input_size))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        y = custom_ops_lib.gemm_divide_sum_scaling_custom(x, self.weight)
        return y * self.scaling_factor
