kernel_src="""
#include "kernel_operator.h"

class KernelConvTranspose3dReluGroupNorm {
public:
    __aicore__ inline KernelConvTranspose3dReluGroupNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR gamma, GM_ADDR beta,
                                GM_ADDR y, GM_ADDR workspace,
                                uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                uint32_t inD, uint32_t inH, uint32_t inW,
                                uint32_t outD, uint32_t outH, uint32_t outW,
                                uint32_t kernelSize, uint32_t groups,
                                uint32_t channelsPerGroup, uint32_t groupSize, float eps)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inD = inD; this->inH = inH; this->inW = inW;
        this->outD = outD; this->outH = outH; this->outW = outW;
        this->kernelSize = kernelSize;
        this->groups = groups;
        this->channelsPerGroup = channelsPerGroup;
        this->groupSize = groupSize;
        this->eps = eps;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->batchIdx = blockIdx / groups;
        this->groupIdx = blockIdx % groups;

        uint64_t inBatchStride = (uint64_t)inChannels * inD * inH * inW;
        uint64_t outBatchStride = (uint64_t)outChannels * outD * outH * outW;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->batchIdx * inBatchStride, inBatchStride);
        weightGm.SetGlobalBuffer((__gm__ float *)weight, (uint64_t)inChannels * outChannels * kernelSize * kernelSize * kernelSize);
        gammaGm.SetGlobalBuffer((__gm__ float *)gamma, outChannels);
        betaGm.SetGlobalBuffer((__gm__ float *)beta, outChannels);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->batchIdx * outBatchStride + this->groupIdx * groupSize, groupSize);

        workspaceGm.SetGlobalBuffer((__gm__ float *)workspace + (uint64_t)blockIdx * groupSize, groupSize);
    }

    __aicore__ inline void Process()
    {
        uint32_t kS = kernelSize;
        uint32_t kVol = kS * kS * kS;
        uint32_t spatialOut = outD * outH * outW;

        for (uint32_t idx = 0; idx < groupSize; idx++) {
            workspaceGm.SetValue(idx, 0.0f);
        }
        AscendC::PipeBarrier<PIPE_ALL>();

        uint32_t ocStart = groupIdx * channelsPerGroup;
        uint32_t ocEnd = ocStart + channelsPerGroup;

        for (uint32_t oc = ocStart; oc < ocEnd; oc++) {
            uint32_t ocLocal = oc - ocStart;
            uint64_t ocOutBase = (uint64_t)ocLocal * spatialOut;

            for (uint32_t ic = 0; ic < inChannels; ic++) {
                uint64_t wBase = ((uint64_t)ic * outChannels + oc) * kVol;
                uint64_t xBase = (uint64_t)ic * inD * inH * inW;

                for (uint32_t id = 0; id < inD; id++) {
                    for (uint32_t ih = 0; ih < inH; ih++) {
                        for (uint32_t iw = 0; iw < inW; iw++) {
                            float xVal = xGm.GetValue(xBase + (uint64_t)id * inH * inW + (uint64_t)ih * inW + iw);
                            for (uint32_t kd = 0; kd < kS; kd++) {
                                for (uint32_t kh = 0; kh < kS; kh++) {
                                    for (uint32_t kw = 0; kw < kS; kw++) {
                                        uint32_t od = id + kd;
                                        uint32_t oh = ih + kh;
                                        uint32_t ow = iw + kw;
                                        float wVal = weightGm.GetValue(wBase + (uint64_t)kd * kS * kS + (uint64_t)kh * kS + kw);
                                        uint64_t outIdx = ocOutBase + (uint64_t)od * outH * outW + (uint64_t)oh * outW + ow;
                                        float cur = workspaceGm.GetValue(outIdx);
                                        workspaceGm.SetValue(outIdx, cur + xVal * wVal);
                                    }
                                }
                            }
                        }
                    }
                }
            }
        }
        AscendC::PipeBarrier<PIPE_ALL>();

        float sum = 0.0f;
        float sumSq = 0.0f;
        for (uint32_t i = 0; i < groupSize; i++) {
            float v = workspaceGm.GetValue(i);
            if (v < 0.0f) v = 0.0f;
            workspaceGm.SetValue(i, v);
            sum += v;
            sumSq += v * v;
        }
        float mean = sum / (float)groupSize;
        float var = sumSq / (float)groupSize - mean * mean;
        float invStd = 1.0f / AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
            (float)AscendC::ScalarGetSqrtVal<float>(var + eps));

        for (uint32_t oc = ocStart; oc < ocEnd; oc++) {
            uint32_t ocLocal = oc - ocStart;
            float g = gammaGm.GetValue(oc);
            float b = betaGm.GetValue(oc);
            uint64_t base = (uint64_t)ocLocal * spatialOut;
            for (uint32_t i = 0; i < spatialOut; i++) {
                float v = workspaceGm.GetValue(base + i);
                float normed = (v - mean) * invStd * g + b;
                yGm.SetValue(base + i, normed);
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> workspaceGm;
    uint32_t batchSize, inChannels, outChannels;
    uint32_t inD, inH, inW, outD, outH, outW;
    uint32_t kernelSize, groups, channelsPerGroup, groupSize;
    uint32_t batchIdx, groupIdx;
    float eps;
};

extern "C" __global__ __aicore__ void convtranspose3d_relu_groupnorm_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR gamma, GM_ADDR beta,
    GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dReluGroupNorm op;
    op.Init(x, weight, gamma, beta, y, workspace,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inD, tiling_data.inH, tiling_data.inW,
            tiling_data.outD, tiling_data.outH, tiling_data.outW,
            tiling_data.kernelSize, tiling_data.groups,
            tiling_data.channelsPerGroup, tiling_data.groupSize, tiling_data.eps);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, groups, bias=False):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, bias=bias)
        self.group_norm = nn.GroupNorm(num_groups=groups, num_channels=out_channels)
        self.groups = groups

    def forward(self, x):
        weight = self.conv_transpose.weight.contiguous()
        gamma = self.group_norm.weight.contiguous()
        beta = self.group_norm.bias.contiguous()
        return custom_ops_lib.convtranspose3d_relu_groupnorm_custom(x, weight, gamma, beta)
