#include "kernel_operator.h"

class KernelConv2dSubtractHardSwishMaxPoolMish {
public:
    __aicore__ inline KernelConv2dSubtractHardSwishMaxPoolMish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                uint32_t height, uint32_t width, uint32_t kernelSize,
                                uint32_t poolKernelSize, uint32_t outHeight, uint32_t outWidth,
                                uint32_t pooledHeight, uint32_t pooledWidth,
                                float subtractValue, uint32_t blockDim, uint32_t totalOut)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->height = height;
        this->width = width;
        this->kernelSize = kernelSize;
        this->poolKernelSize = poolKernelSize;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->pooledHeight = pooledHeight;
        this->pooledWidth = pooledWidth;
        this->subtractValue = subtractValue;
        this->totalOut = totalOut;
        this->blockIdx = AscendC::GetBlockIdx();
        this->blockDim = blockDim;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * height * width);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outChannels * inChannels * kernelSize * kernelSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalOut);
    }

    __aicore__ inline void Process()
    {
        uint32_t chunk = (totalOut + blockDim - 1) / blockDim;
        uint32_t start = blockIdx * chunk;
        uint32_t end = start + chunk;
        if (end > totalOut) end = totalOut;

        for (uint32_t idx = start; idx < end; idx++) {
            uint32_t ow = idx % pooledWidth;
            uint32_t oh = (idx / pooledWidth) % pooledHeight;
            uint32_t oc = (idx / (pooledWidth * pooledHeight)) % outChannels;
            uint32_t n = idx / (pooledWidth * pooledHeight * outChannels);

            float maxv = -3.4e38f;
            for (uint32_t ph = 0; ph < poolKernelSize; ph++) {
                for (uint32_t pw = 0; pw < poolKernelSize; pw++) {
                    uint32_t rh = oh * poolKernelSize + ph;
                    uint32_t rw = ow * poolKernelSize + pw;

                    float acc = bGm.GetValue(oc);
                    for (uint32_t ic = 0; ic < inChannels; ic++) {
                        for (uint32_t kh = 0; kh < kernelSize; kh++) {
                            for (uint32_t kw = 0; kw < kernelSize; kw++) {
                                uint32_t ih = rh + kh;
                                uint32_t iw = rw + kw;
                                uint64_t xOff = ((uint64_t)n * inChannels + ic) * height * width +
                                                (uint64_t)ih * width + iw;
                                uint64_t wOff = ((uint64_t)oc * inChannels + ic) * kernelSize * kernelSize +
                                                (uint64_t)kh * kernelSize + kw;
                                acc += xGm.GetValue(xOff) * wGm.GetValue(wOff);
                            }
                        }
                    }
                    acc = acc - subtractValue;
                    float hs_in = acc + 3.0f;
                    if (hs_in < 0.0f) hs_in = 0.0f;
                    if (hs_in > 6.0f) hs_in = 6.0f;
                    float hs = acc * hs_in / 6.0f;
                    if (hs > maxv) maxv = hs;
                }
            }
            float sp = log_approx(1.0f + exp_approx(maxv));
            float th = tanh_approx(sp);
            float out = maxv * th;
            yGm.SetValue(idx, out);
        }
    }

private:
    __aicore__ inline float exp_approx(float x) {
        if (x > 88.0f) x = 88.0f;
        if (x < -88.0f) x = -88.0f;
        float result = 1.0f;
        float term = 1.0f;
        for (int i = 1; i < 16; i++) {
            term = term * x / (float)i;
            result += term;
        }
        return result;
    }
    __aicore__ inline float log_approx(float x) {
        if (x <= 0.0f) return -88.0f;
        float y = (x - 1.0f) / (x + 1.0f);
        float y2 = y * y;
        float result = 0.0f;
        float term = y;
        for (int i = 0; i < 10; i++) {
            result += term / (float)(2 * i + 1);
            term *= y2;
        }
        return 2.0f * result;
    }
    __aicore__ inline float tanh_approx(float x) {
        float e1 = exp_approx(x);
        float e2 = exp_approx(-x);
        return (e1 - e2) / (e1 + e2);
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t height;
    uint32_t width;
    uint32_t kernelSize;
    uint32_t poolKernelSize;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t pooledHeight;
    uint32_t pooledWidth;
    float subtractValue;
    uint32_t totalOut;
    uint32_t blockIdx;
    uint32_t blockDim;
};

extern "C" __global__ __aicore__ void conv2d_subtract_hard_swish_max_pool_mish_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dSubtractHardSwishMaxPoolMish op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.height, tiling_data.width, tiling_data.kernelSize,
            tiling_data.poolKernelSize, tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.pooledHeight, tiling_data.pooledWidth,
            tiling_data.subtractValue, tiling_data.blockDim, tiling_data.totalOut);
    op.Process();
}
