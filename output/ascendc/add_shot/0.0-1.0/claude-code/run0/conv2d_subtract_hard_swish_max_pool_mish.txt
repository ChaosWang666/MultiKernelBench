kernel_src="""
#include "kernel_operator.h"

constexpr int32_t MAX_PLANE_IN = 128 * 128;
constexpr int32_t MAX_PLANE_OUT = 64 * 64;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t totalPlanes, uint32_t planesPerBlock,
                                 uint32_t heightIn, uint32_t widthIn,
                                 uint32_t heightOut, uint32_t widthOut,
                                 float subtractValue)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startP = blockIdx * planesPerBlock;
        uint32_t endP = startP + planesPerBlock;
        if (endP > totalPlanes) endP = totalPlanes;
        if (startP > totalPlanes) startP = totalPlanes;
        this->startPlane = startP;
        this->endPlane = endP;

        this->heightIn = heightIn;
        this->widthIn = widthIn;
        this->heightOut = heightOut;
        this->widthOut = widthOut;
        this->planeSizeIn = heightIn * widthIn;
        this->planeSizeOut = heightOut * widthOut;
        this->subtractValue = subtractValue;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        pipe.InitBuffer(inBuf, MAX_PLANE_IN * sizeof(float));
        pipe.InitBuffer(outBuf, MAX_PLANE_OUT * sizeof(float));
        pipe.InitBuffer(tmpBuf, MAX_PLANE_IN * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (startPlane >= endPlane) return;

        AscendC::LocalTensor<float> inLocal = inBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        for (uint32_t p = startPlane; p < endPlane; p++) {
            AscendC::DataCopyExtParams copyInParams;
            copyInParams.blockCount = 1;
            copyInParams.blockLen = planeSizeIn * sizeof(float);
            copyInParams.srcStride = 0;
            copyInParams.dstStride = 0;
            AscendC::DataCopyPadExtParams<float> padParams;
            padParams.isPad = false;
            padParams.leftPadding = 0;
            padParams.rightPadding = 0;
            padParams.paddingValue = 0;
            uint64_t inOffset = (uint64_t)p * (uint64_t)planeSizeIn;
            AscendC::DataCopyPad(inLocal, xGm[inOffset], copyInParams, padParams);

            AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

            AscendC::Adds(inLocal, inLocal, -subtractValue, planeSizeIn);
            AscendC::Adds(tmpLocal, inLocal, 3.0f, planeSizeIn);
            AscendC::Maxs(tmpLocal, tmpLocal, 0.0f, planeSizeIn);
            AscendC::Mins(tmpLocal, tmpLocal, 6.0f, planeSizeIn);
            AscendC::Muls(tmpLocal, tmpLocal, (float)(1.0f / 6.0f), planeSizeIn);
            AscendC::Mul(inLocal, inLocal, tmpLocal, planeSizeIn);

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

            for (uint32_t oh = 0; oh < heightOut; oh++) {
                for (uint32_t ow = 0; ow < widthOut; ow++) {
                    uint32_t ih = 2 * oh;
                    uint32_t iw = 2 * ow;
                    float a = inLocal.GetValue(ih * widthIn + iw);
                    float b = inLocal.GetValue(ih * widthIn + iw + 1);
                    float c = inLocal.GetValue((ih + 1) * widthIn + iw);
                    float d = inLocal.GetValue((ih + 1) * widthIn + iw + 1);
                    float m1 = a > b ? a : b;
                    float m2 = c > d ? c : d;
                    float mx = m1 > m2 ? m1 : m2;
                    outLocal.SetValue(oh * widthOut + ow, mx);
                }
            }

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

            AscendC::Exp(tmpLocal, outLocal, planeSizeOut);
            AscendC::Adds(tmpLocal, tmpLocal, 1.0f, planeSizeOut);
            AscendC::Ln(tmpLocal, tmpLocal, planeSizeOut);
            AscendC::Tanh(tmpLocal, tmpLocal, planeSizeOut);
            AscendC::Mul(outLocal, outLocal, tmpLocal, planeSizeOut);

            AscendC::SetFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);

            AscendC::DataCopyExtParams copyOutParams;
            copyOutParams.blockCount = 1;
            copyOutParams.blockLen = planeSizeOut * sizeof(float);
            copyOutParams.srcStride = 0;
            copyOutParams.dstStride = 0;
            uint64_t outOffset = (uint64_t)p * (uint64_t)planeSizeOut;
            AscendC::DataCopyPad(yGm[outOffset], outLocal, copyOutParams);

            AscendC::SetFlag<AscendC::HardEvent::MTE3_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE3_V>(EVENT_ID0);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> inBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t startPlane;
    uint32_t endPlane;
    uint32_t heightIn;
    uint32_t widthIn;
    uint32_t heightOut;
    uint32_t widthOut;
    uint32_t planeSizeIn;
    uint32_t planeSizeOut;
    float subtractValue;
};

extern "C" __global__ __aicore__ void conv2d_subtract_hard_swish_max_pool_mish_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, y,
            tiling_data.totalPlanes, tiling_data.planesPerBlock,
            tiling_data.heightIn, tiling_data.widthIn,
            tiling_data.heightOut, tiling_data.widthOut,
            tiling_data.subtractValue);
    op.Process();
}
