kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;
using namespace matmul;

constexpr int32_t BUFFER_NUM = 1;

class KernelGemmReluDiv {
public:
    __aicore__ inline KernelGemmReluDiv() {}

    Matmul<MatmulType<TPosition::GM, CubeFormat::ND, float>,
           MatmulType<TPosition::GM, CubeFormat::ND, float, true>,
           MatmulType<TPosition::VECIN, CubeFormat::ND, float>,
           MatmulType<TPosition::GM, CubeFormat::ND, float>> matmulObj;

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR bias, GM_ADDR y,
                                const TCubeTiling& tiling, float divisor, TPipe* pipe)
    {
        this->tilingData = tiling;
        this->divisor = divisor;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        wGm.SetGlobalBuffer((__gm__ float*)w);
        biasGm.SetGlobalBuffer((__gm__ float*)bias);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        pipe->InitBuffer(outQueue, BUFFER_NUM,
                         tiling.baseM * tiling.baseN * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t M = tilingData.M;
        int32_t N = tilingData.N;
        int32_t K = tilingData.Ka;
        int32_t baseM = tilingData.baseM;
        int32_t baseN = tilingData.baseN;
        int32_t mBlocks = (M + baseM - 1) / baseM;
        int32_t nBlocks = (N + baseN - 1) / baseN;
        int32_t totalBlocks = mBlocks * nBlocks;
        int32_t coreIdx = GetBlockIdx();
        int32_t coreNum = GetBlockNum();

        float scale = 1.0f / divisor;

        for (int32_t bid = coreIdx; bid < totalBlocks; bid += coreNum) {
            int32_t mIdx = bid / nBlocks;
            int32_t nIdx = bid % nBlocks;
            int32_t mOff = mIdx * baseM;
            int32_t nOff = nIdx * baseN;
            int32_t curM = (mOff + baseM > M) ? (M - mOff) : baseM;
            int32_t curN = (nOff + baseN > N) ? (N - nOff) : baseN;

            matmulObj.SetTensorA(xGm[mOff * K]);
            matmulObj.SetTensorB(wGm[nOff * K]);
            matmulObj.SetBias(biasGm[nOff]);
            matmulObj.SetTail(curM, curN, K);

            LocalTensor<float> cLocal = outQueue.AllocTensor<float>();
            matmulObj.IterateAll(cLocal);
            matmulObj.End();

            int32_t tileSize = curM * curN;
            Maxs(cLocal, cLocal, 0.0f, tileSize);
            Muls(cLocal, cLocal, scale, tileSize);

            outQueue.EnQue(cLocal);
            LocalTensor<float> cOut = outQueue.DeQue<float>();

            DataCopyParams copyParams;
            copyParams.blockCount = (uint16_t)curM;
            copyParams.blockLen = (uint16_t)(curN * sizeof(float) / 32);
            copyParams.srcStride = 0;
            copyParams.dstStride = (uint16_t)((N - curN) * sizeof(float) / 32);
            DataCopy(yGm[mOff * N + nOff], cOut, copyParams);

            outQueue.FreeTensor(cOut);
        }
    }

private:
    GlobalTensor<float> xGm;
    GlobalTensor<float> wGm;
    GlobalTensor<float> biasGm;
    GlobalTensor<float> yGm;
    TCubeTiling tilingData;
    float divisor;
    TQue<QuePosition::VECOUT, BUFFER_NUM> outQueue;
};

extern "C" __global__ __aicore__ void gemm_relu_divide_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    TPipe pipe;
    KernelGemmReluDiv op;
    REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), op.matmulObj,
                      &tilingData.cubeTilingData);
    op.Init(x, weight, bias, y, tilingData.cubeTilingData,
            tilingData.divisor, &pipe);
    op.Process();
}

#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor gemm_relu_divide_custom_impl_npu(const at::Tensor& x,
                                             const at::Tensor& weight,
                                             const at::Tensor& bias,
                                             double divisor) {
    at::Tensor xC = x.contiguous();
    at::Tensor wC = weight.contiguous();
    at::Tensor bC = bias.contiguous();
    int64_t M = xC.size(0);
    int64_t N = wC.size(0);
    at::Tensor result = at::empty({M, N}, xC.options());
    float divisorF = static_cast<float>(divisor);
    EXEC_NPU_CMD(aclnnGemmReluDivideCustom, xC, wC, bC, divisorF, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("gemm_relu_divide_custom", &gemm_relu_divide_custom_impl_npu);
