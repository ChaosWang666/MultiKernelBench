kernel_src="""
#include "kernel_operator.h"

class KernelConv3dFused {
public:
    __aicore__ inline KernelConv3dFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR sum_tensor, GM_ADDR z,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inDepth, uint32_t inHeight, uint32_t inWidth,
                                 uint32_t outDepth, uint32_t outHeight, uint32_t outWidth,
                                 uint32_t kernelSize, uint32_t totalOutputElements,
                                 uint32_t elementsPerCore, uint32_t tailElements)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inDepth = inDepth;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->outDepth = outDepth;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->kernelSize = kernelSize;
        this->totalOutputElements = totalOutputElements;
        this->elementsPerCore = elementsPerCore;
        this->tailElements = tailElements;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        this->startIdx = blockIdx * elementsPerCore;
        if (blockIdx == blockNum - 1) {
            this->myElements = tailElements;
        } else {
            this->myElements = elementsPerCore;
        }
        if (this->startIdx >= totalOutputElements) {
            this->myElements = 0;
        } else if (this->startIdx + this->myElements > totalOutputElements) {
            this->myElements = totalOutputElements - this->startIdx;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * inDepth * inHeight * inWidth);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outChannels * inChannels * kernelSize * kernelSize * kernelSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        sGm.SetGlobalBuffer((__gm__ float *)sum_tensor, outChannels);
        zGm.SetGlobalBuffer((__gm__ float *)z, totalOutputElements);
    }

    __aicore__ inline void Process()
    {
        if (this->myElements == 0) {
            return;
        }

        uint32_t outHW = this->outHeight * this->outWidth;
        uint32_t outDHW = this->outDepth * outHW;
        uint32_t outCDHW = this->outChannels * outDHW;

        uint32_t inHW = this->inHeight * this->inWidth;
        uint32_t inDHW = this->inDepth * inHW;
        uint32_t inCDHW = this->inChannels * inDHW;

        uint32_t kSize = this->kernelSize;
        uint32_t kSq = kSize * kSize;
        uint32_t kCube = kSq * kSize;
        uint32_t wStride = this->inChannels * kCube;

        for (uint32_t i = 0; i < this->myElements; i++) {
            uint32_t outIdx = this->startIdx + i;

            uint32_t ow = outIdx % this->outWidth;
            uint32_t oh = (outIdx / this->outWidth) % this->outHeight;
            uint32_t od = (outIdx / outHW) % this->outDepth;
            uint32_t oc = (outIdx / outDHW) % this->outChannels;
            uint32_t n = outIdx / outCDHW;

            float bias_val = bGm.GetValue(oc);
            float sum_val = sGm.GetValue(oc);

            float acc = bias_val;

            uint32_t x_n_base = n * inCDHW;
            uint32_t w_oc_base = oc * wStride;

            for (uint32_t ic = 0; ic < this->inChannels; ic++) {
                uint32_t x_c_base = x_n_base + ic * inDHW;
                uint32_t w_ic_base = w_oc_base + ic * kCube;

                for (uint32_t kd = 0; kd < kSize; kd++) {
                    uint32_t id = od + kd;
                    uint32_t x_d_base = x_c_base + id * inHW;
                    uint32_t w_kd_base = w_ic_base + kd * kSq;

                    for (uint32_t kh = 0; kh < kSize; kh++) {
                        uint32_t ih = oh + kh;
                        uint32_t x_h_base = x_d_base + ih * this->inWidth;
                        uint32_t w_kh_base = w_kd_base + kh * kSize;

                        for (uint32_t kw = 0; kw < kSize; kw++) {
                            uint32_t iw = ow + kw;
                            float x_val = xGm.GetValue(x_h_base + iw);
                            float w_val = wGm.GetValue(w_kh_base + kw);
                            acc += x_val * w_val;
                        }
                    }
                }
            }

            float v = acc;
            if (v < 0.0f) {
                v = v * 0.2f;
            }
            v = v + sum_val;
            if (v < -1.0f) v = -1.0f;
            if (v > 1.0f) v = 1.0f;

            float c = 0.7978845608028654f;
            float k = 0.044715f;
            float v3 = v * v * v;
            float inner = c * (v + k * v3);
            float e_pos = AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(0.0f);
            float ex = 0.0f;
            {
                float tx = inner;
                float ax = tx < 0.0f ? -tx : tx;
                float e = 1.0f;
                float term = 1.0f;
                for (int32_t it = 1; it < 12; it++) {
                    term = term * ax / (float)it;
                    e += term;
                }
                float e_neg = 1.0f / e;
                float tanh_val;
                if (tx >= 0.0f) {
                    tanh_val = (e - e_neg) / (e + e_neg);
                } else {
                    tanh_val = -((e - e_neg) / (e + e_neg));
                }
                ex = tanh_val;
            }
            float gelu_out = 0.5f * v * (1.0f + ex);

            zGm.SetValue(this->startIdx + i, gelu_out);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> sGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inDepth;
    uint32_t inHeight;
    uint32_t inWidth;
    uint32_t outDepth;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t kernelSize;
    uint32_t totalOutputElements;
    uint32_t elementsPerCore;
    uint32_t tailElements;
    uint32_t startIdx;
    uint32_t myElements;
};

extern "C" __global__ __aicore__ void conv3d_leaky_relu_sum_clamp_gelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR sum_tensor, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dFused op;
    op.Init(x, weight, bias, sum_tensor, z,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inDepth, tiling_data.inHeight, tiling_data.inWidth,
            tiling_data.outDepth, tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.kernelSize, tiling_data.totalOutputElements,
            tiling_data.elementsPerCore, tiling_data.tailElements);
    op.Process();
}
