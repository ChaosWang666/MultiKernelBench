kernel_src="""
#include "kernel_operator.h"

class KernelConv2dGeluGlobalAvgPool {
public:
    __aicore__ inline KernelConv2dGeluGlobalAvgPool() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                uint32_t height, uint32_t width, uint32_t kernelSize)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->height = height;
        this->width = width;
        this->kernelSize = kernelSize;
        this->outHeight = height - kernelSize + 1;
        this->outWidth = width - kernelSize + 1;
        this->outSize = this->outHeight * this->outWidth;

        this->totalOutputs = batchSize * outChannels;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t perBlock = (this->totalOutputs + blockNum - 1) / blockNum;
        this->startOutput = blockIdx * perBlock;
        this->endOutput = this->startOutput + perBlock;
        if (this->endOutput > this->totalOutputs) {
            this->endOutput = this->totalOutputs;
        }

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x);
        weightGm.SetGlobalBuffer((__gm__ DTYPE_WEIGHT *)weight);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y);
    }

    __aicore__ inline float GeluApprox(float v)
    {
        const float c0 = 0.7978845608f;
        const float c1 = 0.044715f;
        float inner = c0 * (v + c1 * v * v * v);
        float e = AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(exp(2.0f * inner));
        float tanhVal = (e - 1.0f) / (e + 1.0f);
        return 0.5f * v * (1.0f + tanhVal);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t idx = this->startOutput; idx < this->endOutput; idx++) {
            uint32_t b = idx / this->outChannels;
            uint32_t oc = idx % this->outChannels;
            float biasVal = biasGm.GetValue(oc);
            float sum = 0.0f;
            for (uint32_t oh = 0; oh < this->outHeight; oh++) {
                for (uint32_t ow = 0; ow < this->outWidth; ow++) {
                    float v = biasVal;
                    for (uint32_t ic = 0; ic < this->inChannels; ic++) {
                        for (uint32_t kh = 0; kh < this->kernelSize; kh++) {
                            for (uint32_t kw = 0; kw < this->kernelSize; kw++) {
                                uint32_t inIdx = ((b * this->inChannels + ic) * this->height + oh + kh) * this->width + ow + kw;
                                uint32_t wIdx = ((oc * this->inChannels + ic) * this->kernelSize + kh) * this->kernelSize + kw;
                                v += xGm.GetValue(inIdx) * weightGm.GetValue(wIdx);
                            }
                        }
                    }
                    sum += GeluApprox(v);
                }
            }
            yGm.SetValue(idx, sum / (float)this->outSize);
        }
    }

private:
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_WEIGHT> weightGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t height;
    uint32_t width;
    uint32_t kernelSize;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t outSize;
    uint32_t totalOutputs;
    uint32_t startOutput;
    uint32_t endOutput;
};

extern "C" __global__ __aicore__ void conv2d_gelu_global_avg_pool_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dGeluGlobalAvgPool op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.height, tiling_data.width, tiling_data.kernelSize);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size):
        super().__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)

    def forward(self, x):
        weight = self.conv.weight.contiguous()
        bias = self.conv.bias.contiguous()
        return custom_ops_lib.conv2d_gelu_global_avg_pool_custom(x, weight, bias)
