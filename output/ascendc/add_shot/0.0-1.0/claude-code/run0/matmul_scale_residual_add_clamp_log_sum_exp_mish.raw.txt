kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedLSE {
public:
    __aicore__ inline KernelFusedLSE() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batchSize, uint32_t hiddenSize,
                                uint32_t rowsPerBlock,
                                float mulFactor, float clampMin, float clampMax)
    {
        this->hiddenSize = hiddenSize;
        this->rowsPerBlock = rowsPerBlock;
        this->mulFactor = mulFactor;
        this->clampMin = clampMin;
        this->clampMax = clampMax;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowOffset = blockIdx * rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ float*)x + rowOffset * hiddenSize, rowsPerBlock * hiddenSize);
        yGm.SetGlobalBuffer((__gm__ float*)y + rowOffset, rowsPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, hiddenSize * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, 64 * sizeof(float));
        pipe.InitBuffer(sumBuf, 64 * sizeof(float));
        pipe.InitBuffer(maxBuf, 64 * sizeof(float));
        pipe.InitBuffer(reduceBuf, 32 * sizeof(float));
        pipe.InitBuffer(workBuf, 2048 * sizeof(float));
        pipe.InitBuffer(tmpBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        for (uint32_t r = 0; r < rowsPerBlock; r++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[r * hiddenSize], hiddenSize);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::Muls(xLocal, xLocal, mulFactor, hiddenSize);
            AscendC::Maxs(xLocal, xLocal, clampMin, hiddenSize);
            AscendC::Mins(xLocal, xLocal, clampMax, hiddenSize);

            AscendC::ReduceMax<float>(reduceLocal, xLocal, workLocal, hiddenSize, false);
            float rowMax = reduceLocal.GetValue(0);
            maxLocal.SetValue(r, rowMax);

            AscendC::Adds(xLocal, xLocal, -rowMax, hiddenSize);
            AscendC::Exp(xLocal, xLocal, hiddenSize);

            AscendC::ReduceSum<float>(reduceLocal, xLocal, workLocal, hiddenSize);
            float rowSum = reduceLocal.GetValue(0);
            sumLocal.SetValue(r, rowSum);

            inQueueX.FreeTensor(xLocal);
        }

        AscendC::LocalTensor<float> outLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        uint32_t n = rowsPerBlock;

        AscendC::Ln(tmp, sumLocal, n);
        AscendC::Add(outLocal, tmp, maxLocal, n);

        AscendC::Exp(tmp, outLocal, n);
        AscendC::Adds(tmp, tmp, 1.0f, n);
        AscendC::Ln(tmp, tmp, n);
        AscendC::Tanh(tmp, tmp, n);
        AscendC::Mul(tmp, tmp, outLocal, n);
        AscendC::Mul(outLocal, tmp, outLocal, n);

        outQueueY.EnQue<float>(outLocal);
        outLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm, outLocal, n);
        outQueueY.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf, maxBuf, reduceBuf, workBuf, tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t hiddenSize;
    uint32_t rowsPerBlock;
    float mulFactor;
    float clampMin;
    float clampMax;
};

extern "C" __global__ __aicore__ void matmul_scale_residual_add_clamp_log_sum_exp_mish_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedLSE op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.hiddenSize, tiling_data.rowsPerBlock,
            tiling_data.mulFactor, tiling_data.clampMin, tiling_data.clampMax);
    op.Process();
}
