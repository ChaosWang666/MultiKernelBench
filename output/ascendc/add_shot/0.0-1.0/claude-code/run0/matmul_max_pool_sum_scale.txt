kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulMaxPoolSumScaleCustom {
public:
    __aicore__ inline KernelMatmulMaxPoolSumScaleCustom() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                 uint32_t batchSize, uint32_t innerSize,
                                 uint32_t tileLength, uint32_t tileNum,
                                 uint32_t batchPerBlock)
    {
        this->innerSize = innerSize;
        this->tileLength = tileLength;
        this->tileNum = tileNum;
        this->batchPerBlock = batchPerBlock;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startBatch = blockIdx * batchPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startBatch * innerSize,
                            batchPerBlock * innerSize);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + startBatch, batchPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(accBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(workBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, this->batchPerBlock * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> accLocal = accBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();

        for (uint32_t b = 0; b < this->batchPerBlock; b++) {
            AscendC::Duplicate<float>(accLocal, 0.0f, this->tileLength);

            for (uint32_t t = 0; t < this->tileNum; t++) {
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(xLocal,
                                  xGm[b * this->innerSize + t * this->tileLength],
                                  this->tileLength);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<DTYPE_X> x = inQueueX.DeQue<DTYPE_X>();
                AscendC::Add(accLocal, accLocal, x, this->tileLength);
                inQueueX.FreeTensor(x);
            }

            AscendC::ReduceSum<float>(workLocal, accLocal, workLocal, this->tileLength);
            float batchSum = workLocal.GetValue(0);
            zLocal.SetValue(b, batchSum);
        }

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        AscendC::LocalTensor<DTYPE_Z> z = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm, z, this->batchPerBlock);
        outQueueZ.FreeTensor(z);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t innerSize;
    uint32_t tileLength;
    uint32_t tileNum;
    uint32_t batchPerBlock;
};

extern "C" __global__ __aicore__ void matmul_max_pool_sum_scale_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulMaxPoolSumScaleCustom op;
    op.Init(x, z,
            tiling_data.batchSize, tiling_data.innerSize,
            tiling_data.tileLength, tiling_data.tileNum,
            tiling_data.batchPerBlock);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, kernel_size, scale_factor):
        super(ModelNew, self).__init__()
        self.matmul = nn.Linear(in_features, out_features)
        self.max_pool = nn.MaxPool1d(kernel_size)
        self.scale_factor = scale_factor

    def forward(self, x):
        x = self.matmul(x)
        x = self.max_pool(x.unsqueeze(1)).squeeze(1).contiguous()
        x = custom_ops_lib.matmul_max_pool_sum_scale_custom(x)
        x = x * self.scale_factor
        return x
