kernel_src="""
#include "kernel_operator.h"

constexpr int32_t IN_TILE = 4096;
constexpr int32_t BIAS_CHUNK = 8;
constexpr int32_t MAX_BATCH_PER_CORE = 16;

class KernelMMMaxPoolSumScale {
public:
    __aicore__ inline KernelMMMaxPoolSumScale() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR out,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t outFeatures,
                                float scaleFactor)
    {
        this->batchSize = batchSize;
        this->inFeatures = inFeatures;
        this->outFeatures = outFeatures;
        this->scaleFactor = scaleFactor;
        this->numTiles = inFeatures / IN_TILE;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->batchPerCore = batchSize / blockNum;
        uint32_t startBatch = blockIdx * this->batchPerCore;

        xGm.SetGlobalBuffer((__gm__ float*)x + startBatch * inFeatures, this->batchPerCore * inFeatures);
        weightGm.SetGlobalBuffer((__gm__ float*)weight, outFeatures * inFeatures);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, outFeatures);
        outGm.SetGlobalBuffer((__gm__ float*)out + startBatch, this->batchPerCore);

        pipe.InitBuffer(xBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(wBuf, IN_TILE * sizeof(float));
        pipe.InitBuffer(tmpBuf, IN_TILE * sizeof(float));
        pipe.InitBuffer(biasBuf, BIAS_CHUNK * sizeof(float));
        pipe.InitBuffer(outBuf, MAX_BATCH_PER_CORE * sizeof(float));
        pipe.InitBuffer(reduceBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        for (uint32_t b = 0; b < this->batchPerCore; b++) {
            AscendC::DataCopy(xLocal, xGm[b * this->inFeatures], this->inFeatures);
            AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

            float batchSum = 0.0f;
            float prevDot = 0.0f;

            for (uint32_t j = 0; j < this->outFeatures; j++) {
                if (j % BIAS_CHUNK == 0) {
                    AscendC::DataCopy(biasLocal, biasGm[j], BIAS_CHUNK);
                    AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID1);
                    AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID1);
                }

                float dot = 0.0f;
                for (uint32_t t = 0; t < this->numTiles; t++) {
                    AscendC::DataCopy(wLocal, weightGm[j * this->inFeatures + t * IN_TILE], IN_TILE);
                    AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
                    AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

                    AscendC::Mul(tmp, xLocal[t * IN_TILE], wLocal, IN_TILE);
                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::ReduceSum<float>(reduceLocal, tmp, tmp, IN_TILE);

                    AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
                    AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID2);
                    dot += reduceLocal.GetValue(0);
                }

                dot += biasLocal.GetValue(j % BIAS_CHUNK);

                if ((j & 1u) == 1u) {
                    float pooled = (prevDot > dot) ? prevDot : dot;
                    batchSum += pooled;
                } else {
                    prevDot = dot;
                }
            }

            outLocal.SetValue(b, batchSum * this->scaleFactor);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID3);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID3);
        AscendC::DataCopy(outGm, outLocal, this->batchPerCore);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, wBuf, tmpBuf, biasBuf, outBuf, reduceBuf;
    AscendC::GlobalTensor<float> xGm, weightGm, biasGm, outGm;
    uint32_t batchSize, inFeatures, outFeatures;
    uint32_t numTiles;
    uint32_t batchPerCore;
    float scaleFactor;
};

extern "C" __global__ __aicore__ void matmul_max_pool_sum_scale_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR out,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMMMaxPoolSumScale op;
    op.Init(x, weight, bias, out,
            tiling_data.batchSize, tiling_data.inFeatures,
            tiling_data.outFeatures, tiling_data.scaleFactor);
    op.Process();
}
