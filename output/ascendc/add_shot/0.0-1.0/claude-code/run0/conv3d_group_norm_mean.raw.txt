kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv3dGroupNormMean {
public:
    __aicore__ inline KernelConv3dGroupNormMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z,
                                uint32_t batchPerBlock, uint32_t elemPerBatch,
                                uint32_t tileLength, uint32_t numTiles,
                                float invElemPerBatch)
    {
        this->batchPerBlock = batchPerBlock;
        this->elemPerBatch = elemPerBatch;
        this->tileLength = tileLength;
        this->numTiles = numTiles;
        this->invElemPerBatch = invElemPerBatch;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockIdx * batchPerBlock * elemPerBatch,
                            batchPerBlock * elemPerBatch);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + blockIdx * batchPerBlock,
                            batchPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, 1, this->batchPerBlock * sizeof(DTYPE_Z));
        pipe.InitBuffer(accumBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(workBuf, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> accum = accumBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();

        for (uint32_t b = 0; b < this->batchPerBlock; b++) {
            AscendC::Duplicate<float>(accum, 0.0f, this->tileLength);

            for (uint32_t t = 0; t < this->numTiles; t++) {
                CopyIn(b, t);
                ComputeAccum(accum);
            }

            AscendC::ReduceSum<float>(work, accum, work, this->tileLength);
            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            float sum = work.GetValue(0);
            outLocal.SetValue(b, sum * this->invElemPerBatch);
            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        }

        outQueueZ.EnQue(outLocal);
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn(uint32_t b, uint32_t t)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[b * this->elemPerBatch + t * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void ComputeAccum(AscendC::LocalTensor<float>& accum)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::Add(accum, accum, xLocal, this->tileLength);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<DTYPE_Z> outLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm, outLocal, this->batchPerBlock);
        outQueueZ.FreeTensor(outLocal);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t batchPerBlock;
    uint32_t elemPerBatch;
    uint32_t tileLength;
    uint32_t numTiles;
    float invElemPerBatch;
};

extern "C" __global__ __aicore__ void conv3d_group_norm_mean_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dGroupNormMean op;
    op.Init(x, z,
            tiling_data.batchPerBlock, tiling_data.elemPerBatch,
            tiling_data.tileLength, tiling_data.numTiles,
            tiling_data.invElemPerBatch);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, num_groups):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.group_norm = nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        x = self.conv(x)
        x = self.group_norm(x)
        x = x.contiguous()
        x = custom_ops_lib.conv3d_group_norm_mean_custom(x)
