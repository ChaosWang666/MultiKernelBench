kernel_src = """
#include "kernel_operator.h"

class KernelSwishMulSwish {
public:
    __aicore__ inline KernelSwishMulSwish() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mul_w, GM_ADDR z,
                                 uint32_t totalRows, uint32_t outFeatures, uint32_t rowsPerCore)
    {
        this->outFeatures = outFeatures;
        uint32_t blockId = AscendC::GetBlockIdx();
        uint32_t startRow = blockId * rowsPerCore;
        if (startRow >= totalRows) {
            this->actualRows = 0;
        } else if (startRow + rowsPerCore > totalRows) {
            this->actualRows = totalRows - startRow;
        } else {
            this->actualRows = rowsPerCore;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x + startRow * outFeatures,
                            this->actualRows * outFeatures);
        mulWGm.SetGlobalBuffer((__gm__ float*)mul_w, outFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z + startRow * outFeatures,
                            this->actualRows * outFeatures);

        pipe.InitBuffer(inXQueue,   1, outFeatures * sizeof(float));
        pipe.InitBuffer(inMulQueue, 1, outFeatures * sizeof(float));
        pipe.InitBuffer(outZQueue,  1, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf1, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf2, outFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (actualRows == 0) {
            return;
        }
        for (uint32_t i = 0; i < actualRows; i++) {
            CopyIn(i);
            Compute();
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal   = inXQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> mulLocal = inMulQueue.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * outFeatures], outFeatures);
        AscendC::DataCopy(mulLocal, mulWGm, outFeatures);
        inXQueue.EnQue(xLocal);
        inMulQueue.EnQue(mulLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal   = inXQueue.DeQue<float>();
        AscendC::LocalTensor<float> mulLocal = inMulQueue.DeQue<float>();
        AscendC::LocalTensor<float> tmp1     = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> tmp2     = tmpBuf2.Get<float>();
        AscendC::LocalTensor<float> outLocal = outZQueue.AllocTensor<float>();

        // tmp1 = sigmoid(x)
        AscendC::Sigmoid(tmp1, xLocal, outFeatures);
        // tmp2 = x * sigmoid(x)  (= swish(x))
        AscendC::Mul(tmp2, xLocal, tmp1, outFeatures);
        // tmp1 = swish(x) * mul_w
        AscendC::Mul(tmp1, tmp2, mulLocal, outFeatures);
        // tmp2 = sigmoid(tmp1)
        AscendC::Sigmoid(tmp2, tmp1, outFeatures);
        // out = tmp1 * sigmoid(tmp1)  (= swish(tmp1))
        AscendC::Mul(outLocal, tmp1, tmp2, outFeatures);

        inXQueue.FreeTensor(xLocal);
        inMulQueue.FreeTensor(mulLocal);
        outZQueue.EnQue<float>(outLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> outLocal = outZQueue.DeQue<float>();
        AscendC::DataCopy(zGm[rowIdx * outFeatures], outLocal, outFeatures);
        outZQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  inXQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1>  inMulQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outZQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC>   tmpBuf1;
    AscendC::TBuf<AscendC::TPosition::VECCALC>   tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> mulWGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t outFeatures;
    uint32_t actualRows;
};

extern "C" __global__ __aicore__ void gemm_group_norm_swish_multiply_swish_custom(
    GM_ADDR x, GM_ADDR mul_w, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelSwishMulSwish op;
    op.Init(x, mul_w, z,
            tiling_data.totalRows,
            tiling_data.outFeatures,
            tiling_data.rowsPerCore);
    op.Process();
}

        return custom_ops_lib.gemm_group_norm_swish_multiply_swish_custom(

            self.gemm.weight,
            self.gemm.bias,
