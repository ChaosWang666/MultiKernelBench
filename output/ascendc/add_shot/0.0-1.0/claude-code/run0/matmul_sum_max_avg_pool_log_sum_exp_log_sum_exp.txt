kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulSumCustom {
public:
    __aicore__ inline KernelMatmulSumCustom() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR y,
                                 uint32_t rowsPerCore, uint32_t inFeatures,
                                 uint32_t tileLength)
    {
        this->rowsPerCore = rowsPerCore;
        this->inFeatures = inFeatures;
        this->tileLength = tileLength;
        this->numTiles = inFeatures / tileLength;

        uint32_t coreIdx = AscendC::GetBlockIdx();
        uint32_t rowStart = coreIdx * rowsPerCore;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + rowStart * inFeatures, rowsPerCore * inFeatures);
        wGm.SetGlobalBuffer((__gm__ DTYPE_W *)w, inFeatures);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + rowStart, rowsPerCore);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, this->tileLength * sizeof(DTYPE_W));
        pipe.InitBuffer(mulBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(workBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(outBuf, ((this->rowsPerCore * sizeof(DTYPE_Y) + 31) / 32) * 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> outLocal = outBuf.Get<DTYPE_Y>();
        AscendC::LocalTensor<float> mulLocal = mulBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();

        for (uint32_t r = 0; r < this->rowsPerCore; r++) {
            float acc = 0.0f;
            for (uint32_t t = 0; t < this->numTiles; t++) {
                CopyIn(r, t);
                acc += ComputeTile(mulLocal, workLocal);
            }
            outLocal.SetValue(r, (DTYPE_Y)acc);
        }
        AscendC::DataCopy(yGm, outLocal, this->rowsPerCore);
    }

private:
    __aicore__ inline void CopyIn(uint32_t r, uint32_t t)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_W> wLocal = inQueueW.AllocTensor<DTYPE_W>();
        AscendC::DataCopy(xLocal, xGm[r * this->inFeatures + t * this->tileLength], this->tileLength);
        AscendC::DataCopy(wLocal, wGm[t * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueW.EnQue(wLocal);
    }

    __aicore__ inline float ComputeTile(AscendC::LocalTensor<float> &mulLocal,
                                         AscendC::LocalTensor<float> &workLocal)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_W> wLocal = inQueueW.DeQue<DTYPE_W>();

        AscendC::Mul(mulLocal, xLocal, wLocal, this->tileLength);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::ReduceSum<float>(mulLocal, mulLocal, workLocal, this->tileLength);
        AscendC::PipeBarrier<PIPE_V>();

        float v = mulLocal.GetValue(0);

        inQueueX.FreeTensor(xLocal);
        inQueueW.FreeTensor(wLocal);
        return v;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueW;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mulBuf, workBuf, outBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_W> wGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t rowsPerCore;
    uint32_t inFeatures;
    uint32_t tileLength;
    uint32_t numTiles;
};

extern "C" __global__ __aicore__ void matmul_sum_max_avg_pool_log_sum_exp_log_sum_exp_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{

    KernelMatmulSumCustom op;
    op.Init(x, w, y, tiling_data.rowsPerCore, tiling_data.inFeatures, tiling_data.tileLength);
    op.Process();
}
