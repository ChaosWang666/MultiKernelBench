kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv2dReluBiasAdd {
public:
    __aicore__ inline KernelConv2dReluBiasAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                 uint32_t totalLength, uint32_t slicesPerBlock,
                                 uint32_t sliceSize, uint32_t numChannels,
                                 uint32_t biasLen)
    {
        this->slicesPerBlock = static_cast<int32_t>(slicesPerBlock);
        this->sliceSize = static_cast<int32_t>(sliceSize);
        this->numChannels = static_cast<int32_t>(numChannels);
        this->biasLen = static_cast<int32_t>(biasLen);
        this->blockIdx = static_cast<int32_t>(AscendC::GetBlockIdx());

        int64_t blockLength = static_cast<int64_t>(this->slicesPerBlock) * static_cast<int64_t>(this->sliceSize);
        int64_t blockOffset = blockLength * static_cast<int64_t>(this->blockIdx);

        xGm.SetGlobalBuffer((__gm__ float *)x + blockOffset, blockLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + blockOffset, blockLength);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, this->biasLen);

        uint32_t sliceBufBytes = ((static_cast<uint32_t>(this->sliceSize) * sizeof(float) + 31) / 32) * 32;
        pipe.InitBuffer(inQueueX, BUFFER_NUM, sliceBufBytes);
        pipe.InitBuffer(outQueueY, BUFFER_NUM, sliceBufBytes);
        uint32_t biasBufBytes = ((static_cast<uint32_t>(this->biasLen) * sizeof(float) + 31) / 32) * 32;
        pipe.InitBuffer(biasBuf, biasBufBytes);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::DataCopy(biasLocal, biasGm, biasLen);

        int32_t alignedSliceSize = ((sliceSize + 7) / 8) * 8;

        for (int32_t s = 0; s < slicesPerBlock; s++) {
            int32_t globalSlice = blockIdx * slicesPerBlock + s;
            int32_t ch = globalSlice % numChannels;
            float biasVal = biasLocal.GetValue(ch);
            CopyIn(s);
            Compute(biasVal, alignedSliceSize);
            CopyOut(s);
        }
    }

private:
    __aicore__ inline void CopyIn(int32_t s)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = static_cast<uint32_t>(sliceSize) * sizeof(float);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        AscendC::DataCopyPadExtParams<float> padParams;
        padParams.isPad = true;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;
        int64_t offset = static_cast<int64_t>(s) * static_cast<int64_t>(sliceSize);
        AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float biasVal, int32_t alignedSliceSize)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::Relu(yLocal, xLocal, alignedSliceSize);
        AscendC::Adds(yLocal, yLocal, biasVal, alignedSliceSize);
        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t s)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = static_cast<uint32_t>(sliceSize) * sizeof(float);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        int64_t offset = static_cast<int64_t>(s) * static_cast<int64_t>(sliceSize);
        AscendC::DataCopyPad(yGm[offset], yLocal, copyParams);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> biasGm;
    int32_t slicesPerBlock;
    int32_t sliceSize;
    int32_t numChannels;
    int32_t biasLen;
    int32_t blockIdx;
};

extern "C" __global__ __aicore__ void conv2d_relu_bias_add_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dReluBiasAdd op;
    op.Init(x, bias, y,
            tiling_data.totalLength, tiling_data.slicesPerBlock,
            tiling_data.sliceSize, tiling_data.numChannels,
            tiling_data.biasLen);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, bias_shape):
        super().__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.bias = nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
        bias_flat = self.bias.contiguous().view(-1).contiguous()
        return custom_ops_lib.conv2d_relu_bias_add_custom(x, bias_flat)
