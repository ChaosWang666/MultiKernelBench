kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulSigmoidSum {
public:
    __aicore__ inline KernelMatmulSigmoidSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize, uint32_t tileLength)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->tileLength = tileLength;

        int32_t blockNum = static_cast<int32_t>(AscendC::GetBlockNum());
        int32_t blockIdx = static_cast<int32_t>(AscendC::GetBlockIdx());
        int32_t bs = static_cast<int32_t>(batchSize);
        this->rowsPerCore = (bs + blockNum - 1) / blockNum;
        this->startRow = blockIdx * this->rowsPerCore;
        this->endRow = this->startRow + this->rowsPerCore;
        if (this->endRow > bs) {
            this->endRow = bs;
        }
        if (this->startRow > bs) {
            this->startRow = bs;
        }

        this->tileNum = static_cast<int32_t>(inputSize) / static_cast<int32_t>(tileLength);

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inputSize);
        wGm.SetGlobalBuffer((__gm__ float *)weight, hiddenSize * inputSize);
        bGm.SetGlobalBuffer((__gm__ float *)bias, hiddenSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(accBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        int32_t hidden = static_cast<int32_t>(this->hiddenSize);
        int32_t input = static_cast<int32_t>(this->inputSize);

        for (int32_t row = this->startRow; row < this->endRow; row++) {
            float rowSum = 0.0f;
            for (int32_t h = 0; h < hidden; h++) {
                float biasVal = bGm.GetValue(h);
                float dot = 0.0f;
                for (int32_t t = 0; t < this->tileNum; t++) {
                    AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                    AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                    AscendC::DataCopy(xLocal, xGm[row * input + t * this->tileLength], this->tileLength);
                    AscendC::DataCopy(wLocal, wGm[h * input + t * this->tileLength], this->tileLength);
                    inQueueX.EnQue(xLocal);
                    inQueueW.EnQue(wLocal);

                    AscendC::LocalTensor<float> xIn = inQueueX.DeQue<float>();
                    AscendC::LocalTensor<float> wIn = inQueueW.DeQue<float>();
                    AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
                    AscendC::Mul(tmp, xIn, wIn, this->tileLength);

                    AscendC::LocalTensor<float> acc = accBuf.Get<float>();
                    AscendC::ReduceSum(acc, tmp, tmp, this->tileLength);
                    float partial = acc.GetValue(0);
                    dot += partial;

                    inQueueX.FreeTensor(xIn);
                    inQueueW.FreeTensor(wIn);
                }
                float preSig = dot + biasVal;
                float sigv = 1.0f / (1.0f + AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                    static_cast<float>(AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                        1.0f / (1.0f + 0.0f)))));
                // compute sigmoid in scalar
                float negv = -preSig;
                float expv = 1.0f;
                // scalar exp via Taylor-free: use AscendC scalar exp if available, otherwise rely on math
                expv = expf(negv);
                sigv = 1.0f / (1.0f + expv);
                rowSum += sigv;
            }
            if (AscendC::GetBlockIdx() < AscendC::GetBlockNum()) {
                yGm.SetValue(row, rowSum);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueW;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t inputSize;
    uint32_t hiddenSize;
    uint32_t tileLength;
    int32_t rowsPerCore;
    int32_t startRow;
    int32_t endRow;
    int32_t tileNum;
};

extern "C" __global__ __aicore__ void matmul_sigmoid_sum_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSigmoidSum op;
    op.Init(x, weight, bias, y, tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize, tiling_data.tileLength);
    op.Process();
}
