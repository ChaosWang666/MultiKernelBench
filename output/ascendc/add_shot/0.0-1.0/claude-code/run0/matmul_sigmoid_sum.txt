kernel_src="""
#include "kernel_operator.h"

class KernelMatmulSigmoidSum {
public:
    __aicore__ inline KernelMatmulSigmoidSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inputSize, uint32_t hiddenSize,
                                 uint32_t batchPerCore, uint32_t kTileLen, uint32_t jTileLen)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->batchPerCore = batchPerCore;
        this->kTileLen = kTileLen;
        this->jTileLen = jTileLen;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startBatch = blockIdx * batchPerCore;
        this->endBatch = this->startBatch + batchPerCore;
        if (this->endBatch > batchSize) { this->endBatch = batchSize; }
        this->blockIdx = blockIdx;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inputSize);
        wGm.SetGlobalBuffer((__gm__ float *)w, hiddenSize * inputSize);
        bGm.SetGlobalBuffer((__gm__ float *)b, hiddenSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize);

        pipe.InitBuffer(xBuf, kTileLen * sizeof(float));
        pipe.InitBuffer(wBuf, kTileLen * sizeof(float));
        pipe.InitBuffer(tmpBuf, kTileLen * sizeof(float));
        pipe.InitBuffer(bBuf, jTileLen * sizeof(float));
        pipe.InitBuffer(yChunkBuf, jTileLen * sizeof(float));
        pipe.InitBuffer(onesBuf, jTileLen * sizeof(float));
        pipe.InitBuffer(redBuf, 64 * sizeof(float));
        pipe.InitBuffer(outBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        AscendC::LocalTensor<float> onesLocal = onesBuf.Get<float>();
        AscendC::Duplicate(onesLocal, (float)1.0f, this->jTileLen);
        for (uint32_t idx = 0; idx < this->batchPerCore; idx++) {
            uint32_t i = this->startBatch + idx;
            if (i >= this->batchSize) {
                outLocal.SetValue(idx, (float)0.0f);
                continue;
            }
            float s = ProcessOneBatch(i);
            outLocal.SetValue(idx, s);
        }
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::DataCopy(yGm[this->blockIdx * this->batchPerCore], outLocal, this->batchPerCore);
    }

private:
    __aicore__ inline float ProcessOneBatch(uint32_t i)
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> bLocal = bBuf.Get<float>();
        AscendC::LocalTensor<float> yChunk = yChunkBuf.Get<float>();
        AscendC::LocalTensor<float> onesLocal = onesBuf.Get<float>();
        AscendC::LocalTensor<float> redLocal = redBuf.Get<float>();

        float totalSum = 0.0f;
        uint32_t jTiles = this->hiddenSize / this->jTileLen;
        uint32_t kTiles = this->inputSize / this->kTileLen;

        for (uint32_t jt = 0; jt < jTiles; jt++) {
            AscendC::DataCopy(bLocal, bGm[jt * this->jTileLen], this->jTileLen);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Adds(yChunk, bLocal, (float)0.0f, this->jTileLen);
            AscendC::PipeBarrier<PIPE_ALL>();

            for (uint32_t jj = 0; jj < this->jTileLen; jj++) {
                float dotSum = 0.0f;
                uint32_t rowBase = (jt * this->jTileLen + jj) * this->inputSize;
                for (uint32_t kt = 0; kt < kTiles; kt++) {
                    AscendC::DataCopy(xLocal, xGm[i * this->inputSize + kt * this->kTileLen], this->kTileLen);
                    AscendC::DataCopy(wLocal, wGm[rowBase + kt * this->kTileLen], this->kTileLen);
                    AscendC::PipeBarrier<PIPE_ALL>();
                    AscendC::Mul(tmpLocal, xLocal, wLocal, this->kTileLen);
                    AscendC::PipeBarrier<PIPE_ALL>();
                    AscendC::ReduceSum<float>(redLocal, tmpLocal, tmpLocal, this->kTileLen);
                    AscendC::PipeBarrier<PIPE_ALL>();
                    dotSum += redLocal.GetValue(0);
                }
                float cur = yChunk.GetValue(jj);
                yChunk.SetValue(jj, cur + dotSum);
            }

            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Muls(yChunk, yChunk, (float)-1.0f, this->jTileLen);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Exp(yChunk, yChunk, this->jTileLen);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Adds(yChunk, yChunk, (float)1.0f, this->jTileLen);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Div(yChunk, onesLocal, yChunk, this->jTileLen);
            AscendC::PipeBarrier<PIPE_ALL>();

            AscendC::ReduceSum<float>(redLocal, yChunk, tmpLocal, this->jTileLen);
            AscendC::PipeBarrier<PIPE_ALL>();
            totalSum += redLocal.GetValue(0);
        }
        return totalSum;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, wBuf, tmpBuf, bBuf, yChunkBuf, onesBuf, redBuf, outBuf;
    AscendC::GlobalTensor<float> xGm, wGm, bGm, yGm;
    uint32_t batchSize;
    uint32_t inputSize;
    uint32_t hiddenSize;
    uint32_t batchPerCore;
    uint32_t kTileLen;
    uint32_t jTileLen;
    uint32_t startBatch;
    uint32_t endBatch;
    uint32_t blockIdx;
};

extern "C" __global__ __aicore__ void matmul_sigmoid_sum_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSigmoidSum op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inputSize, tiling_data.hiddenSize,
            tiling_data.batchPerCore, tiling_data.kTileLen, tiling_data.jTileLen);
    op.Process();
}
