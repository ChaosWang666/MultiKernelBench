#include "kernel_operator.h"

using AscendC::HardEvent;

class KernelConv3dFused {
public:
    __aicore__ inline KernelConv3dFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR convBias, GM_ADDR bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inDepth, uint32_t inHeight, uint32_t inWidth,
                                 uint32_t outDepth, uint32_t outHeight, uint32_t outWidth,
                                 uint32_t kernelSize, uint32_t totalOutputElements,
                                 uint32_t elementsPerCore)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inDepth = inDepth;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->outDepth = outDepth;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->kernelSize = kernelSize;
        this->totalOutputElements = totalOutputElements;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startIdx = blockIdx * elementsPerCore;
        uint32_t endIdx = this->startIdx + elementsPerCore;
        if (endIdx > totalOutputElements) {
            endIdx = totalOutputElements;
        }
        if (this->startIdx > totalOutputElements) {
            this->startIdx = totalOutputElements;
        }
        this->coreLength = endIdx - this->startIdx;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inChannels * inDepth * inHeight * inWidth);
        wGm.SetGlobalBuffer((__gm__ float *)weight, outChannels * inChannels * kernelSize * kernelSize * kernelSize);
        cbGm.SetGlobalBuffer((__gm__ float *)convBias, outChannels);
        bGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);
        zGm.SetGlobalBuffer((__gm__ float *)z, totalOutputElements);
    }

    __aicore__ inline float ComputeOneOutput(uint32_t outIdx)
    {
        uint32_t tmp = outIdx;
        uint32_t ow = tmp % this->outWidth; tmp /= this->outWidth;
        uint32_t oh = tmp % this->outHeight; tmp /= this->outHeight;
        uint32_t od = tmp % this->outDepth; tmp /= this->outDepth;
        uint32_t oc = tmp % this->outChannels; tmp /= this->outChannels;
        uint32_t n = tmp;

        float acc = 0.0f;
        uint32_t K = this->kernelSize;
        uint32_t IC = this->inChannels;
        uint32_t ID = this->inDepth;
        uint32_t IH = this->inHeight;
        uint32_t IW = this->inWidth;

        for (uint32_t ic = 0; ic < IC; ic++) {
            for (uint32_t kd = 0; kd < K; kd++) {
                for (uint32_t kh = 0; kh < K; kh++) {
                    for (uint32_t kw = 0; kw < K; kw++) {
                        uint32_t id = od + kd;
                        uint32_t ih = oh + kh;
                        uint32_t iw = ow + kw;
                        uint32_t xOffset = ((((n * IC + ic) * ID + id) * IH + ih) * IW + iw);
                        uint32_t wOffset = ((((oc * IC + ic) * K + kd) * K + kh) * K + kw);
                        float xVal = xGm.GetValue(xOffset);
                        float wVal = wGm.GetValue(wOffset);
                        acc += xVal * wVal;
                    }
                }
            }
        }

        float cbVal = cbGm.GetValue(oc);
        acc += cbVal;

        // ReLU
        if (acc < 0.0f) acc = 0.0f;
        // LeakyReLU (no-op since acc >= 0 after ReLU)
        // GELU approximation: 0.5 * x * (1 + tanh(sqrt(2/pi) * (x + 0.044715 * x^3)))
        float x3 = acc * acc * acc;
        float inner = 0.7978845608f * (acc + 0.044715f * x3);
        // tanh via exp
        float e1 = AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(1.0f);
        float ex = 0.0f;
        {
            float v = inner * 2.0f;
            // exp approximation using std-like: use fast path
            float ev;
            // Use a polynomial/exp via series won't be stable; use math expf substitute
            // On aicore scalar float, use __expf-like by manual: leverage identity tanh(y) = (e^{2y}-1)/(e^{2y}+1)
            // We'll compute exp via repeated squaring from e^(v/n). For simplicity, use builtin expf.
            ev = expf(v);
            ex = (ev - 1.0f) / (ev + 1.0f);
        }
        float gelu = 0.5f * acc * (1.0f + ex);

        // Sigmoid
        float sig = 1.0f / (1.0f + expf(-gelu));

        float bVal = bGm.GetValue(oc);
        float out = sig + bVal;
        return out;
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->coreLength; i++) {
            uint32_t outIdx = this->startIdx + i;
            float val = ComputeOneOutput(outIdx);
            zGm.SetValue(outIdx, val);
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> cbGm;
    AscendC::GlobalTensor<float> bGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inDepth;
    uint32_t inHeight;
    uint32_t inWidth;
    uint32_t outDepth;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t kernelSize;
    uint32_t totalOutputElements;
    uint32_t startIdx;
    uint32_t coreLength;
};

extern "C" __global__ __aicore__ void conv3d_relu_leaky_relu_gelu_sigmoid_bias_add_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR conv_bias, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dFused op;
    op.Init(x, weight, conv_bias, bias, z,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inDepth, tiling_data.inHeight, tiling_data.inWidth,
            tiling_data.outDepth, tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.kernelSize, tiling_data.totalOutputElements,
            tiling_data.elementsPerCore);
    op.Process();
}
