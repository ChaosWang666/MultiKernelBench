kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulSwishScaling {
public:
    __aicore__ inline KernelMatmulSwishScaling() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                uint32_t M, uint32_t N, uint32_t K, float scalingFactor)
    {
        this->M = M;
        this->N = N;
        this->K = K;
        this->scalingFactor = scalingFactor;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        // Each core handles a chunk of N (output columns)
        uint32_t nPerCore = N / blockNum;
        this->nStart = blockIdx * nPerCore;
        this->nLength = nPerCore;

        xGm.SetGlobalBuffer((__gm__ float *)x, M * K);
        weightGm.SetGlobalBuffer((__gm__ float *)weight + this->nStart * K, this->nLength * K);
        biasGm.SetGlobalBuffer((__gm__ float *)bias + this->nStart, this->nLength);
        zGm.SetGlobalBuffer((__gm__ float *)z, M * N);

        pipe.InitBuffer(inQueueX, 1, K * sizeof(float));
        pipe.InitBuffer(inQueueW, 1, K * sizeof(float));
        pipe.InitBuffer(inQueueBias, 1, this->nLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, K * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, this->nLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Load bias once for this core's N segment
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
        AscendC::DataCopy(biasLocal, biasGm, this->nLength);
        inQueueBias.EnQue(biasLocal);
        biasLocal = inQueueBias.DeQue<float>();

        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        for (uint32_t m = 0; m < this->M; m++) {
            // Load one row of x
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[m * this->K], this->K);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<float>();

            AscendC::LocalTensor<float> outLocal = outQueueZ.AllocTensor<float>();

            // For each output column in this core's range
            for (uint32_t n = 0; n < this->nLength; n++) {
                AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
                AscendC::DataCopy(wLocal, weightGm[n * this->K], this->K);
                inQueueW.EnQue(wLocal);
                wLocal = inQueueW.DeQue<float>();

                // Element-wise multiply x * w into tmp
                AscendC::Mul(tmpLocal, xLocal, wLocal, this->K);

                // Reduce sum
                AscendC::LocalTensor<float> reduceOut = wLocal;
                float sumVal = 0.0f;
                // Use a simple loop-less reduction via WholeReduceSum-like approach:
                // fallback: manual accumulate via ReduceSum
                AscendC::ReduceSum<float>(reduceOut, tmpLocal, reduceOut, this->K);
                sumVal = reduceOut.GetValue(0);

                inQueueW.FreeTensor(wLocal);

                // Add bias
                float biasVal = biasLocal.GetValue(n);
                float val = sumVal + biasVal;

                // Swish: val * sigmoid(val)
                float sig = 1.0f / (1.0f + AscendC::ScalarCast<float, AscendC::RoundMode::CAST_NONE>(
                    static_cast<float>(0.0f)) + 0.0f);
                // Compute sigmoid manually via exp
                float negVal = -val;
                float expNeg;
                // Use AscendC scalar exp through simple approximation:
                // We compute via standard math on scalar: use __expf-like through fast approximation
                // Fallback to series? Use a simple approach: cast through tensor op.
                // Simpler approach: write to local tensor, apply Exp, read back.
                {
                    AscendC::LocalTensor<float> oneElem = tmpLocal;
                    oneElem.SetValue(0, negVal);
                    AscendC::Exp(oneElem, oneElem, 1);
                    expNeg = oneElem.GetValue(0);
                }
                sig = 1.0f / (1.0f + expNeg);

                float outVal = val * sig * this->scalingFactor;
                outLocal.SetValue(n, outVal);
            }

            outQueueZ.EnQue<float>(outLocal);
            AscendC::LocalTensor<float> outDeq = outQueueZ.DeQue<float>();
            AscendC::DataCopy(zGm[m * this->N + this->nStart], outDeq, this->nLength);
            outQueueZ.FreeTensor(outDeq);

            inQueueX.FreeTensor(xLocal);
        }

        inQueueBias.FreeTensor(biasLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX, inQueueW, inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t M;
    uint32_t N;
    uint32_t K;
    uint32_t nStart;
    uint32_t nLength;
    float scalingFactor;
};

extern "C" __global__ __aicore__ void matmul_swish_scaling_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSwishScaling op;
    op.Init(x, weight, bias, z, tiling_data.M, tiling_data.N, tiling_data.K, tiling_data.scalingFactor);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, scaling_factor):
        super(ModelNew, self).__init__()
        self.matmul = torch.nn.Linear(in_features, out_features)
        self.scaling_factor = float(scaling_factor)
        self.in_features = in_features
        self.out_features = out_features

    def forward(self, x):
        x = x.contiguous()
        weight = self.matmul.weight.contiguous()
        bias = self.matmul.bias.contiguous()
        return custom_ops_lib.matmul_swish_scaling_custom(x, weight, bias, self.scaling_factor)
