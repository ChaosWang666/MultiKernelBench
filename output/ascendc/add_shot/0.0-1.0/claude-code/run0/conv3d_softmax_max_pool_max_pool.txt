kernel_src="""
#include "kernel_operator.h"

class KernelConv3dSoftmaxMaxPoolMaxPool {
public:
    __aicore__ inline KernelConv3dSoftmaxMaxPoolMaxPool() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace,
                                 uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                 uint32_t inDepth, uint32_t inHeight, uint32_t inWidth,
                                 uint32_t kernelSize, uint32_t poolKernelSize,
                                 uint32_t convDepth, uint32_t convHeight, uint32_t convWidth,
                                 uint32_t outDepth, uint32_t outHeight, uint32_t outWidth,
                                 uint32_t totalTasks, uint32_t tasksPerBlock)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inDepth = inDepth;
        this->inHeight = inHeight;
        this->inWidth = inWidth;
        this->kernelSize = kernelSize;
        this->poolKernelSize = poolKernelSize;
        this->convDepth = convDepth;
        this->convHeight = convHeight;
        this->convWidth = convWidth;
        this->outDepth = outDepth;
        this->outHeight = outHeight;
        this->outWidth = outWidth;
        this->totalTasks = totalTasks;
        this->tasksPerBlock = tasksPerBlock;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startTask = blockIdx * tasksPerBlock;
        this->endTask = (this->startTask + tasksPerBlock) < totalTasks ? (this->startTask + tasksPerBlock) : totalTasks;

        xGm.SetGlobalBuffer((__gm__ float *)x);
        weightGm.SetGlobalBuffer((__gm__ float *)weight);
        biasGm.SetGlobalBuffer((__gm__ float *)bias);
        zGm.SetGlobalBuffer((__gm__ float *)z);
        workspaceGm.SetGlobalBuffer((__gm__ float *)workspace);
    }

    __aicore__ inline void Process()
    {
        uint32_t convSpatial = convDepth * convHeight * convWidth;
        uint32_t convPerBatch = outChannels * convSpatial;
        uint32_t inPerBatch = inChannels * inDepth * inHeight * inWidth;
        uint32_t outSpatial = outDepth * outHeight * outWidth;
        uint32_t outPerBatch = outChannels * outSpatial;

        for (uint32_t task = startTask; task < endTask; task++) {
            uint32_t b = task / outChannels;
            uint32_t oc = task % outChannels;

            // Step 1: compute conv3d for this (b, oc) over all spatial positions
            for (uint32_t od = 0; od < convDepth; od++) {
                for (uint32_t oh = 0; oh < convHeight; oh++) {
                    for (uint32_t ow = 0; ow < convWidth; ow++) {
                        float val = biasGm.GetValue(oc);
                        for (uint32_t ic = 0; ic < inChannels; ic++) {
                            for (uint32_t kd = 0; kd < kernelSize; kd++) {
                                for (uint32_t kh = 0; kh < kernelSize; kh++) {
                                    for (uint32_t kw = 0; kw < kernelSize; kw++) {
                                        uint32_t id = od + kd;
                                        uint32_t ih = oh + kh;
                                        uint32_t iw = ow + kw;
                                        uint32_t xIdx = b * inPerBatch
                                                      + ic * inDepth * inHeight * inWidth
                                                      + id * inHeight * inWidth
                                                      + ih * inWidth + iw;
                                        uint32_t wIdx = oc * inChannels * kernelSize * kernelSize * kernelSize
                                                      + ic * kernelSize * kernelSize * kernelSize
                                                      + kd * kernelSize * kernelSize
                                                      + kh * kernelSize + kw;
                                        val += xGm.GetValue(xIdx) * weightGm.GetValue(wIdx);
                                    }
                                }
                            }
                        }
                        uint32_t wsIdx = b * convPerBatch + oc * convSpatial
                                       + od * convHeight * convWidth
                                       + oh * convWidth + ow;
                        workspaceGm.SetValue(wsIdx, val);
                    }
                }
            }
        }

        AscendC::SyncAll();

        // Step 2: Softmax across outChannels dim + double maxpool for each b and output spatial position
        // Redistribute work: each block handles a subset of (b, od_out, oh_out, ow_out) tuples
        uint32_t totalOutSpatial = batchSize * outDepth * outHeight * outWidth;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t perBlock = (totalOutSpatial + blockNum - 1) / blockNum;
        uint32_t startIdx = blockIdx * perBlock;
        uint32_t endIdx = (startIdx + perBlock) < totalOutSpatial ? (startIdx + perBlock) : totalOutSpatial;

        uint32_t pool1Depth = convDepth / poolKernelSize;
        uint32_t pool1Height = convHeight / poolKernelSize;
        uint32_t pool1Width = convWidth / poolKernelSize;

        for (uint32_t idx = startIdx; idx < endIdx; idx++) {
            uint32_t b = idx / outSpatial;
            uint32_t rem = idx % outSpatial;
            uint32_t od_out = rem / (outHeight * outWidth);
            uint32_t rem2 = rem % (outHeight * outWidth);
            uint32_t oh_out = rem2 / outWidth;
            uint32_t ow_out = rem2 % outWidth;

            // The 4x4x4 region in conv output (poolKernelSize^2 = 4 per dim)
            uint32_t convBase_d = od_out * poolKernelSize * poolKernelSize;
            uint32_t convBase_h = oh_out * poolKernelSize * poolKernelSize;
            uint32_t convBase_w = ow_out * poolKernelSize * poolKernelSize;
            uint32_t poolRange = poolKernelSize * poolKernelSize; // 4

            float finalMax = -3.4e38f;
            uint32_t finalMaxOc = 0;
            // Iterate through the 4x4x4 positions and compute softmax then pooling
            // Since softmax+max_dim(channel) requires all channels per position, but max_pool then
            // picks max across spatial, we can note: the pooled result at (b,oc,od_out,oh_out,ow_out)
            // equals max over 4x4x4 conv region of softmax(conv)[b,oc,d,h,w].
            // Since softmax is monotonic in its input at fixed channel with same denominator,
            // but denominator differs per spatial position, we must compute per position.

            for (uint32_t oc = 0; oc < outChannels; oc++) {
                float maxVal = -3.4e38f;
                for (uint32_t dd = 0; dd < poolRange; dd++) {
                    for (uint32_t hh = 0; hh < poolRange; hh++) {
                        for (uint32_t ww = 0; ww < poolRange; ww++) {
                            uint32_t cd = convBase_d + dd;
                            uint32_t ch = convBase_h + hh;
                            uint32_t cw = convBase_w + ww;
                            if (cd >= convDepth || ch >= convHeight || cw >= convWidth) continue;

                            // Compute softmax denominator at this spatial position
                            float maxLogit = -3.4e38f;
                            for (uint32_t c = 0; c < outChannels; c++) {
                                uint32_t wsIdx = b * convPerBatch + c * convSpatial
                                               + cd * convHeight * convWidth
                                               + ch * convWidth + cw;
                                float v = workspaceGm.GetValue(wsIdx);
                                if (v > maxLogit) maxLogit = v;
                            }
                            float sumExp = 0.0f;
                            for (uint32_t c = 0; c < outChannels; c++) {
                                uint32_t wsIdx = b * convPerBatch + c * convSpatial
                                               + cd * convHeight * convWidth
                                               + ch * convWidth + cw;
                                float v = workspaceGm.GetValue(wsIdx);
                                float e = AscendC::ScalarCast<float, float, AscendC::RoundMode::CAST_NONE>(
                                    (float)expf(v - maxLogit));
                                sumExp += e;
                            }
                            uint32_t wsIdxOc = b * convPerBatch + oc * convSpatial
                                             + cd * convHeight * convWidth
                                             + ch * convWidth + cw;
                            float ocLogit = workspaceGm.GetValue(wsIdxOc);
                            float smVal = (float)expf(ocLogit - maxLogit) / sumExp;
                            if (smVal > maxVal) maxVal = smVal;
                        }
                    }
                }
                uint32_t outIdx = b * outPerBatch + oc * outSpatial
                                + od_out * outHeight * outWidth
                                + oh_out * outWidth + ow_out;
                zGm.SetValue(outIdx, maxVal);
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> weightGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    AscendC::GlobalTensor<float> workspaceGm;

    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inDepth;
    uint32_t inHeight;
    uint32_t inWidth;
    uint32_t kernelSize;
    uint32_t poolKernelSize;
    uint32_t convDepth;
    uint32_t convHeight;
    uint32_t convWidth;
    uint32_t outDepth;
    uint32_t outHeight;
    uint32_t outWidth;
    uint32_t totalTasks;
    uint32_t tasksPerBlock;
    uint32_t startTask;
    uint32_t endTask;
};

extern "C" __global__ __aicore__ void conv3d_softmax_max_pool_max_pool_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dSoftmaxMaxPoolMaxPool op;
    op.Init(x, weight, bias, z, workspace,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inDepth, tiling_data.inHeight, tiling_data.inWidth,
            tiling_data.kernelSize, tiling_data.poolKernelSize,
            tiling_data.convDepth, tiling_data.convHeight, tiling_data.convWidth,
            tiling_data.outDepth, tiling_data.outHeight, tiling_data.outWidth,
            tiling_data.totalTasks, tiling_data.tasksPerBlock);
    op.Process();
}
