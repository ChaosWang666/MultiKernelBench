kernel_src="""
#include "kernel_operator.h"

class KernelGemmSigmoidGemmLSE {
public:
    __aicore__ inline KernelGemmSigmoidGemmLSE() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w1, GM_ADDR b1,
                                GM_ADDR w2, GM_ADDR b2, GM_ADDR z,
                                uint32_t batchSize, uint32_t inputSize,
                                uint32_t hiddenSize, uint32_t outputSize)
    {
        this->batchSize = batchSize;
        this->inputSize = inputSize;
        this->hiddenSize = hiddenSize;
        this->outputSize = outputSize;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowPerBlock = (batchSize + blockNum - 1) / blockNum;
        this->startRow = blockIdx * rowPerBlock;
        uint32_t endRow = this->startRow + rowPerBlock;
        if (endRow > batchSize) endRow = batchSize;
        this->endRow = endRow;

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * inputSize);
        w1Gm.SetGlobalBuffer((__gm__ float *)w1, hiddenSize * inputSize);
        b1Gm.SetGlobalBuffer((__gm__ float *)b1, hiddenSize);
        w2Gm.SetGlobalBuffer((__gm__ float *)w2, outputSize * hiddenSize);
        b2Gm.SetGlobalBuffer((__gm__ float *)b2, outputSize);
        zGm.SetGlobalBuffer((__gm__ float *)z, batchSize);

        pipe.InitBuffer(xBuf, inputSize * sizeof(float));
        pipe.InitBuffer(hiddenBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(outBuf, outputSize * sizeof(float));
        pipe.InitBuffer(w1RowBuf, inputSize * sizeof(float));
        pipe.InitBuffer(w2RowBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(b1Buf, hiddenSize * sizeof(float));
        pipe.InitBuffer(b2Buf, outputSize * sizeof(float));
        pipe.InitBuffer(redBuf, 256);
        pipe.InitBuffer(sclBuf, 64);
    }

    __aicore__ inline void Process()
    {
        if (startRow >= endRow) return;

        AscendC::LocalTensor<float> xLocal   = xBuf.Get<float>();
        AscendC::LocalTensor<float> hidden   = hiddenBuf.Get<float>();
        AscendC::LocalTensor<float> out      = outBuf.Get<float>();
        AscendC::LocalTensor<float> w1Row    = w1RowBuf.Get<float>();
        AscendC::LocalTensor<float> w2Row    = w2RowBuf.Get<float>();
        AscendC::LocalTensor<float> tmp      = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> b1Local  = b1Buf.Get<float>();
        AscendC::LocalTensor<float> b2Local  = b2Buf.Get<float>();
        AscendC::LocalTensor<float> redTmp   = redBuf.Get<float>();
        AscendC::LocalTensor<float> scl      = sclBuf.Get<float>();

        AscendC::DataCopy(b1Local, b1Gm, hiddenSize);
        AscendC::DataCopy(b2Local, b2Gm, outputSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t i = startRow; i < endRow; i++) {
            AscendC::DataCopy(xLocal, xGm[i * inputSize], inputSize);
            AscendC::PipeBarrier<PIPE_ALL>();

            // hidden[j] = dot(x, w1[j])
            for (uint32_t j = 0; j < hiddenSize; j++) {
                AscendC::DataCopy(w1Row, w1Gm[j * inputSize], inputSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::Mul(tmp, xLocal, w1Row, inputSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::ReduceSum<float>(scl, tmp, redTmp, inputSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                float v = scl.GetValue(0);
                hidden.SetValue(j, v);
            }
            AscendC::PipeBarrier<PIPE_ALL>();

            // hidden = sigmoid(hidden + b1)
            AscendC::Add(hidden, hidden, b1Local, hiddenSize);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Muls(tmp, hidden, (float)-1.0f, hiddenSize);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Exp(tmp, tmp, hiddenSize);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Adds(tmp, tmp, (float)1.0f, hiddenSize);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Duplicate(hidden, (float)1.0f, hiddenSize);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Div(hidden, hidden, tmp, hiddenSize);
            AscendC::PipeBarrier<PIPE_ALL>();

            // out[m] = dot(hidden, w2[m])
            for (uint32_t m = 0; m < outputSize; m++) {
                AscendC::DataCopy(w2Row, w2Gm[m * hiddenSize], hiddenSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::Mul(tmp, hidden, w2Row, hiddenSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::ReduceSum<float>(scl, tmp, redTmp, hiddenSize);
                AscendC::PipeBarrier<PIPE_ALL>();
                float v = scl.GetValue(0);
                out.SetValue(m, v);
            }
            AscendC::PipeBarrier<PIPE_ALL>();

            // out = out + b2
            AscendC::Add(out, out, b2Local, outputSize);
            AscendC::PipeBarrier<PIPE_ALL>();

            // LogSumExp: log(sum(exp(out - max))) + max
            AscendC::ReduceMax<float>(scl, out, redTmp, outputSize, false);
            AscendC::PipeBarrier<PIPE_ALL>();
            float maxVal = scl.GetValue(0);
            AscendC::Adds(out, out, -maxVal, outputSize);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Exp(out, out, outputSize);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::ReduceSum<float>(scl, out, redTmp, outputSize);
            AscendC::PipeBarrier<PIPE_ALL>();
            float sumVal = scl.GetValue(0);
            scl.SetValue(0, sumVal);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Ln(scl, scl, 8);
            AscendC::PipeBarrier<PIPE_ALL>();
            float logSum = scl.GetValue(0);
            scl.SetValue(0, logSum + maxVal);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::DataCopy(zGm[i], scl, 1);
            AscendC::PipeBarrier<PIPE_ALL>();
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, hiddenBuf, outBuf, w1RowBuf, w2RowBuf, tmpBuf, b1Buf, b2Buf, redBuf, sclBuf;
    AscendC::GlobalTensor<float> xGm, w1Gm, b1Gm, w2Gm, b2Gm, zGm;
    uint32_t batchSize;
    uint32_t inputSize;
    uint32_t hiddenSize;
    uint32_t outputSize;
    uint32_t startRow;
    uint32_t endRow;
};

extern "C" __global__ __aicore__ void gemm_sigmoid_sum_log_sum_exp_custom(
    GM_ADDR x, GM_ADDR w1, GM_ADDR b1, GM_ADDR w2, GM_ADDR b2, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmSigmoidGemmLSE op;
    op.Init(x, w1, b1, w2, b2, z,
            tiling_data.batchSize, tiling_data.inputSize,
            tiling_data.hiddenSize, tiling_data.outputSize);
    op.Process();
}
