kernel_src="""
#include "kernel_operator.h"

class KernelConv2dMishMish {
public:
    __aicore__ inline KernelConv2dMishMish() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                uint32_t batchSize, uint32_t inChannels, uint32_t outChannels,
                                uint32_t inH, uint32_t inW, uint32_t kernelSize,
                                uint32_t outH, uint32_t outW)
    {
        this->batchSize = batchSize;
        this->inChannels = inChannels;
        this->outChannels = outChannels;
        this->inH = inH;
        this->inW = inW;
        this->kernelSize = kernelSize;
        this->outH = outH;
        this->outW = outW;

        this->inWAligned = ((inW + 7) / 8) * 8;
        this->outWAligned = ((outW + 7) / 8) * 8;
        this->weightSize = inChannels * kernelSize * kernelSize;
        this->weightSizeAligned = ((this->weightSize + 7) / 8) * 8;
        this->biasSizeAligned = ((outChannels + 7) / 8) * 8;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t totalNC = batchSize * outChannels;
        uint32_t ncPerBlock = (totalNC + blockNum - 1) / blockNum;
        this->ncStart = blockIdx * ncPerBlock;
        uint32_t ncEndCandidate = this->ncStart + ncPerBlock;
        this->ncEnd = (ncEndCandidate < totalNC) ? ncEndCandidate : totalNC;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * inChannels * inH * inW);
        weightGm.SetGlobalBuffer((__gm__ float*)weight, outChannels * inChannels * kernelSize * kernelSize);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, outChannels);
        yGm.SetGlobalBuffer((__gm__ float*)y, batchSize * outChannels * outH * outW);

        pipe.InitBuffer(xRowQue, 1, this->inWAligned * sizeof(float));
        pipe.InitBuffer(weightQue, 1, this->weightSizeAligned * sizeof(float));
        pipe.InitBuffer(biasQue, 1, this->biasSizeAligned * sizeof(float));
        pipe.InitBuffer(yRowQue, 1, this->outWAligned * sizeof(float));
        pipe.InitBuffer(accBuf, this->outWAligned * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->outWAligned * sizeof(float));
        pipe.InitBuffer(tmp2Buf, this->outWAligned * sizeof(float));
        pipe.InitBuffer(tmp3Buf, this->outWAligned * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->ncStart >= this->ncEnd) {
            return;
        }

        AscendC::LocalTensor<float> biasAlloc = biasQue.AllocTensor<float>();
        AscendC::DataCopy(biasAlloc, biasGm, this->biasSizeAligned);
        biasQue.EnQue(biasAlloc);
        AscendC::LocalTensor<float> biasLocal = biasQue.DeQue<float>();

        AscendC::LocalTensor<float> acc = accBuf.Get<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmp2Buf.Get<float>();
        AscendC::LocalTensor<float> tmp3 = tmp3Buf.Get<float>();

        uint32_t kSize = this->kernelSize;
        uint32_t kSquared = kSize * kSize;
        uint32_t inHW = this->inH * this->inW;
        uint32_t inChwSize = this->inChannels * inHW;
        uint32_t outHW = this->outH * this->outW;
        uint32_t outChwSize = this->outChannels * outHW;
        uint32_t weightCInSize = this->weightSize;

        for (uint32_t nc = this->ncStart; nc < this->ncEnd; nc++) {
            uint32_t n = nc / this->outChannels;
            uint32_t c_out = nc % this->outChannels;

            float biasVal = biasLocal.GetValue(c_out);

            AscendC::LocalTensor<float> wAlloc = weightQue.AllocTensor<float>();
            AscendC::DataCopy(wAlloc, weightGm[c_out * weightCInSize], this->weightSizeAligned);
            weightQue.EnQue(wAlloc);
            AscendC::LocalTensor<float> wLocal = weightQue.DeQue<float>();

            for (uint32_t h_out = 0; h_out < this->outH; h_out++) {
                AscendC::Duplicate(acc, biasVal, this->outW);

                for (uint32_t c_in = 0; c_in < this->inChannels; c_in++) {
                    for (uint32_t kh = 0; kh < kSize; kh++) {
                        uint32_t inRow = h_out + kh;
                        uint32_t xOffset = n * inChwSize + c_in * inHW + inRow * this->inW;

                        AscendC::LocalTensor<float> xRowAlloc = xRowQue.AllocTensor<float>();
                        AscendC::DataCopy(xRowAlloc, xGm[xOffset], this->inWAligned);
                        xRowQue.EnQue(xRowAlloc);
                        AscendC::LocalTensor<float> xRow = xRowQue.DeQue<float>();

                        for (uint32_t kw = 0; kw < kSize; kw++) {
                            float w = wLocal.GetValue(c_in * kSquared + kh * kSize + kw);
                            AscendC::Muls(tmp, xRow[kw], w, this->outW);
                            AscendC::Add(acc, acc, tmp, this->outW);
                        }

                        xRowQue.FreeTensor(xRow);
                    }
                }

                ApplyMish(acc, tmp, tmp2, tmp3, this->outW);
                ApplyMish(acc, tmp, tmp2, tmp3, this->outW);

                AscendC::LocalTensor<float> yAlloc = yRowQue.AllocTensor<float>();
                AscendC::Adds(yAlloc, acc, 0.0f, this->outW);
                yRowQue.EnQue(yAlloc);
                AscendC::LocalTensor<float> yOut = yRowQue.DeQue<float>();

                uint32_t yOffset = n * outChwSize + c_out * outHW + h_out * this->outW;
                AscendC::DataCopyExtParams copyParams;
                copyParams.blockCount = 1;
                copyParams.blockLen = this->outW * sizeof(float);
                copyParams.srcStride = 0;
                copyParams.dstStride = 0;
                AscendC::DataCopyPad(yGm[yOffset], yOut, copyParams);

                yRowQue.FreeTensor(yOut);
            }

            weightQue.FreeTensor(wLocal);
        }

        biasQue.FreeTensor(biasLocal);
    }

private:
    __aicore__ inline void ApplyMish(AscendC::LocalTensor<float>& x,
                                     AscendC::LocalTensor<float>& tmp,
                                     AscendC::LocalTensor<float>& tmp2,
                                     AscendC::LocalTensor<float>& tmp3,
                                     uint32_t len)
    {
        AscendC::Abs(tmp, x, len);
        AscendC::Muls(tmp, tmp, -1.0f, len);
        AscendC::Exp(tmp, tmp, len);
        AscendC::Adds(tmp, tmp, 1.0f, len);
        AscendC::Ln(tmp, tmp, len);
        AscendC::Maxs(tmp2, x, 0.0f, len);
        AscendC::Add(tmp, tmp, tmp2, len);

        AscendC::Muls(tmp2, tmp, -2.0f, len);
        AscendC::Exp(tmp2, tmp2, len);
        AscendC::Muls(tmp3, tmp2, -1.0f, len);
        AscendC::Adds(tmp3, tmp3, 1.0f, len);
        AscendC::Adds(tmp2, tmp2, 1.0f, len);
        AscendC::Div(tmp3, tmp3, tmp2, len);
        AscendC::Mul(x, x, tmp3, len);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> xRowQue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> weightQue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> biasQue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> yRowQue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accBuf, tmpBuf, tmp2Buf, tmp3Buf;
    AscendC::GlobalTensor<float> xGm, weightGm, biasGm, yGm;

    uint32_t batchSize;
    uint32_t inChannels;
    uint32_t outChannels;
    uint32_t inH;
    uint32_t inW;
    uint32_t kernelSize;
    uint32_t outH;
    uint32_t outW;
    uint32_t inWAligned;
    uint32_t outWAligned;
    uint32_t weightSize;
    uint32_t weightSizeAligned;
    uint32_t biasSizeAligned;
    uint32_t ncStart;
    uint32_t ncEnd;
};

extern "C" __global__ __aicore__ void conv2d_mish_mish_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dMishMish op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inChannels, tiling_data.outChannels,
            tiling_data.inH, tiling_data.inW, tiling_data.kernelSize,
            tiling_data.outH, tiling_data.outW);
    op.Process();
}
