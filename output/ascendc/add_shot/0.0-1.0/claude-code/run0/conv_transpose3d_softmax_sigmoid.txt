kernel_src="""
#include "kernel_operator.h"

class KernelConvTranspose3dSoftmaxSigmoid {
public:
    __aicore__ inline KernelConvTranspose3dSoftmaxSigmoid() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t channels, uint32_t spatialSize)
    {
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->batchSize = batchSize;
        this->alignedChannels = (channels + 63) / 64 * 64;

        uint32_t totalSpatial = batchSize * spatialSize;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t spatialPerBlock = (totalSpatial + blockNum - 1) / blockNum;
        this->startSpatial = blockIdx * spatialPerBlock;
        uint32_t endSpatial = this->startSpatial + spatialPerBlock;
        if (endSpatial > totalSpatial) endSpatial = totalSpatial;
        this->mySpatial = (endSpatial > this->startSpatial) ? (endSpatial - this->startSpatial) : 0;

        xGm.SetGlobalBuffer((__gm__ float*)x, totalSpatial * channels);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalSpatial * channels);

        pipe.InitBuffer(xBuf, this->alignedChannels * sizeof(float));
        pipe.InitBuffer(yBuf, this->alignedChannels * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->alignedChannels * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> yLocal = yBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        for (uint32_t i = 0; i < this->mySpatial; i++) {
            uint32_t globalSpatialIdx = this->startSpatial + i;
            uint32_t batchIdx = globalSpatialIdx / this->spatialSize;
            uint32_t spatialIdx = globalSpatialIdx % this->spatialSize;
            uint32_t baseOffset = batchIdx * this->channels * this->spatialSize + spatialIdx;

            // Load C channel values (strided)
            for (uint32_t c = 0; c < this->channels; c++) {
                xLocal.SetValue(c, xGm.GetValue(baseOffset + c * this->spatialSize));
            }
            for (uint32_t c = this->channels; c < this->alignedChannels; c++) {
                xLocal.SetValue(c, -1.0e30f);
            }

            // max reduction (scalar)
            float maxVal = xLocal.GetValue(0);
            for (uint32_t c = 1; c < this->channels; c++) {
                float v = xLocal.GetValue(c);
                if (v > maxVal) maxVal = v;
            }

            // x = x - max
            AscendC::Adds(xLocal, xLocal, -maxVal, this->alignedChannels);

            // exp
            AscendC::Exp(xLocal, xLocal, this->alignedChannels);

            // zero-out padding so it doesn't contribute to sum
            for (uint32_t c = this->channels; c < this->alignedChannels; c++) {
                xLocal.SetValue(c, 0.0f);
            }

            // sum reduction (scalar)
            float sumVal = 0.0f;
            for (uint32_t c = 0; c < this->channels; c++) {
                sumVal += xLocal.GetValue(c);
            }
            float invSum = (sumVal > 0.0f) ? (1.0f / sumVal) : 0.0f;

            // softmax = exp / sum
            AscendC::Muls(xLocal, xLocal, invSum, this->alignedChannels);

            // sigmoid(s) = 1 / (1 + exp(-s))
            AscendC::Muls(yLocal, xLocal, -1.0f, this->alignedChannels);
            AscendC::Exp(yLocal, yLocal, this->alignedChannels);
            AscendC::Adds(yLocal, yLocal, 1.0f, this->alignedChannels);
            AscendC::Duplicate(tmpLocal, 1.0f, this->alignedChannels);
            AscendC::Div(yLocal, tmpLocal, yLocal, this->alignedChannels);

            // Store back (strided)
            for (uint32_t c = 0; c < this->channels; c++) {
                yGm.SetValue(baseOffset + c * this->spatialSize, yLocal.GetValue(c));
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> yBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t batchSize;
    uint32_t startSpatial;
    uint32_t mySpatial;
    uint32_t alignedChannels;
};

extern "C" __global__ __aicore__ void conv_transpose3d_softmax_sigmoid_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dSoftmaxSigmoid op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize);
    op.Process();
}

class ModelNew(nn.Module):

        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding, bias=bias)
