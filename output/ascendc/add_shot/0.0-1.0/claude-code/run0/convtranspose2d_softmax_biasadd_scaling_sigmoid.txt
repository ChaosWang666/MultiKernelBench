#include "kernel_operator.h"

class KernelFusedOp {
public:
    __aicore__ inline KernelFusedOp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                  uint32_t totalRows, uint32_t rowSize, float scaling)
    {
        this->rowSize = rowSize;
        this->scaling = scaling;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t base = totalRows / blockNum;
        uint32_t rem = totalRows % blockNum;
        uint32_t rowStart;
        if (blockIdx < rem) {
            this->rowsThisBlock = base + 1;
            rowStart = blockIdx * this->rowsThisBlock;
        } else {
            this->rowsThisBlock = base;
            rowStart = rem * (base + 1) + (blockIdx - rem) * base;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x + rowStart * rowSize, this->rowsThisBlock * rowSize);
        yGm.SetGlobalBuffer((__gm__ float*)y + rowStart * rowSize, this->rowsThisBlock * rowSize);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, rowSize);

        pipe.InitBuffer(inQueueX, 2, rowSize * sizeof(float));
        pipe.InitBuffer(outQueueY, 2, rowSize * sizeof(float));
        pipe.InitBuffer(biasBuf, rowSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, rowSize * sizeof(float));
        pipe.InitBuffer(reduceBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::DataCopy(biasLocal, biasGm, rowSize);

        for (uint32_t r = 0; r < rowsThisBlock; r++) {
            CopyIn(r);
            Compute(biasLocal);
            CopyOut(r);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t r)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[r * rowSize], rowSize);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(AscendC::LocalTensor<float>& biasLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        // softmax: max
        AscendC::ReduceMax<float>(reduceLocal, xLocal, tmpLocal, rowSize, false);
        float maxVal = reduceLocal.GetValue(0);

        // x - max
        AscendC::Adds(tmpLocal, xLocal, -maxVal, rowSize);
        // exp
        AscendC::Exp(tmpLocal, tmpLocal, rowSize);
        // sum
        AscendC::ReduceSum<float>(reduceLocal, tmpLocal, yLocal, rowSize);
        float sumVal = reduceLocal.GetValue(0);
        float invSum = 1.0f / sumVal;

        // softmax
        AscendC::Muls(tmpLocal, tmpLocal, invSum, rowSize);
        // + bias
        AscendC::Add(tmpLocal, tmpLocal, biasLocal, rowSize);
        // * scale
        AscendC::Muls(tmpLocal, tmpLocal, scaling, rowSize);
        // sigmoid = 1 / (1 + exp(-x))
        AscendC::Muls(yLocal, tmpLocal, -1.0f, rowSize);
        AscendC::Exp(yLocal, yLocal, rowSize);
        AscendC::Adds(yLocal, yLocal, 1.0f, rowSize);
        AscendC::Duplicate(tmpLocal, 1.0f, rowSize);
        AscendC::Div(yLocal, tmpLocal, yLocal, rowSize);

        inQueueX.FreeTensor(xLocal);
        outQueueY.EnQue(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t r)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[r * rowSize], yLocal, rowSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 2> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf, tmpBuf, reduceBuf;
    AscendC::GlobalTensor<float> xGm, yGm, biasGm;
    uint32_t rowSize;
    uint32_t rowsThisBlock;
    float scaling;
};

extern "C" __global__ __aicore__ void convtranspose2d_softmax_biasadd_scaling_sigmoid_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{

    KernelFusedOp op;
    op.Init(x, bias, z, tiling_data.totalRows, tiling_data.rowSize, tiling_data.scalingFactor);
    op.Process();
}

import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, bias_shape, scaling_factor):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = self.conv_transpose(x)
        N, C, H, W = x.shape
        x_t = x.permute(0, 2, 3, 1).contiguous().view(-1, C)
        bias_flat = self.bias.view(-1).contiguous()
        y_t = custom_ops_lib.convtranspose2d_softmax_biasadd_scaling_sigmoid_custom(x_t, bias_flat)
        y = y_t.view(N, H, W, C).permute(0, 3, 1, 2).contiguous()
        return y
