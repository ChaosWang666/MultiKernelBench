kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

extern "C" __global__ __aicore__ void gemm_add_relu_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);

    if ASCEND_IS_AIC {
        AscendC::TPipe pipe;
        using aT = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float>;
        using bT = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float, true>;
        using cT = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float>;
        using biasT = matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float>;

        matmul::Matmul<aT, bT, cT, biasT> mm;
        REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm, &tiling_data.cubeTilingData);

        AscendC::GlobalTensor<float> xGm;
        AscendC::GlobalTensor<float> wGm;
        AscendC::GlobalTensor<float> biasGm;
        AscendC::GlobalTensor<float> yGm;
        xGm.SetGlobalBuffer((__gm__ float*)x, tiling_data.M * tiling_data.K);
        wGm.SetGlobalBuffer((__gm__ float*)w, tiling_data.N * tiling_data.K);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, tiling_data.N);
        yGm.SetGlobalBuffer((__gm__ float*)y, tiling_data.M * tiling_data.N);

        mm.SetTensorA(xGm);
        mm.SetTensorB(wGm);
        mm.SetBias(biasGm);
        mm.IterateAll(yGm);
        mm.End();

        AscendC::CrossCoreSetFlag<0x2, PIPE_FIX>(0x8);
    }

    if ASCEND_IS_AIV {
        AscendC::CrossCoreWaitFlag(0x8);

        AscendC::TPipe pipe;
        AscendC::GlobalTensor<float> yGm;
        yGm.SetGlobalBuffer((__gm__ float*)y, tiling_data.M * tiling_data.N);

        uint32_t totalElements = tiling_data.M * tiling_data.N;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t elemsPerBlock = totalElements / blockNum;
        uint32_t baseOffset = blockIdx * elemsPerBlock;

        const uint32_t TILE_SIZE = 8192;
        uint32_t numTiles = elemsPerBlock / TILE_SIZE;

        AscendC::TQue<AscendC::TPosition::VECIN, 2> inQue;
        AscendC::TQue<AscendC::TPosition::VECOUT, 2> outQue;
        pipe.InitBuffer(inQue, 2, TILE_SIZE * sizeof(float));
        pipe.InitBuffer(outQue, 2, TILE_SIZE * sizeof(float));

        for (uint32_t t = 0; t < numTiles; t++) {
            AscendC::LocalTensor<float> inLocal = inQue.AllocTensor<float>();
            AscendC::DataCopy(inLocal, yGm[baseOffset + t * TILE_SIZE], TILE_SIZE);
            inQue.EnQue(inLocal);

            AscendC::LocalTensor<float> inLocal2 = inQue.DeQue<float>();
            AscendC::LocalTensor<float> outLocal = outQue.AllocTensor<float>();
            AscendC::Maxs(outLocal, inLocal2, static_cast<float>(0), TILE_SIZE);
            outQue.EnQue(outLocal);
            inQue.FreeTensor(inLocal2);

            AscendC::LocalTensor<float> outLocal2 = outQue.DeQue<float>();
            AscendC::DataCopy(yGm[baseOffset + t * TILE_SIZE], outLocal2, TILE_SIZE);
            outQue.FreeTensor(outLocal2);
        }
    }
}
