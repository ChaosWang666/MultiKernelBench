kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t TILE_SIZE = 16384;
constexpr float CLAMP_MIN = 0.0f;
constexpr float CLAMP_MAX = 1.0f;

class KernelAvgClamp {
public:
    __aicore__ inline KernelAvgClamp() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t ncCount, uint32_t reduceSize)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t ncPerCore = (ncCount + blockNum - 1) / blockNum;
        uint32_t startNC = blockIdx * ncPerCore;
        uint32_t endNC = startNC + ncPerCore;
        if (endNC > ncCount) endNC = ncCount;
        uint32_t myCount = (endNC > startNC) ? (endNC - startNC) : 0;

        this->startNC = startNC;
        this->myCount = myCount;
        this->reduceSize = reduceSize;
        this->invReduceSize = (reduceSize > 0) ? (1.0f / (float)reduceSize) : 0.0f;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x, (uint64_t)ncCount * reduceSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y, ncCount);

        pipe.InitBuffer(inQueue, BUFFER_NUM, TILE_SIZE * sizeof(DTYPE_X));
        pipe.InitBuffer(reduceBuf, 32 * 1024);
        pipe.InitBuffer(scalarBuf, 64);
        pipe.InitBuffer(outBuf, 4096);
    }

    __aicore__ inline void Process()
    {
        if (myCount == 0) return;

        AscendC::LocalTensor<DTYPE_Y> outLocal = outBuf.Get<DTYPE_Y>();

        for (uint32_t i = 0; i < myCount; i++) {
            uint32_t nc = startNC + i;
            uint64_t gmOffset = (uint64_t)nc * reduceSize;

            float total = 0.0f;
            uint32_t processed = 0;
            while (processed < reduceSize) {
                uint32_t remaining = reduceSize - processed;
                uint32_t curTile = (remaining < TILE_SIZE) ? remaining : TILE_SIZE;

                total += ReduceTile(gmOffset + processed, curTile);
                processed += curTile;
            }

            float avg = total * invReduceSize;
            float clamped = avg;
            if (clamped < CLAMP_MIN) clamped = CLAMP_MIN;
            if (clamped > CLAMP_MAX) clamped = CLAMP_MAX;

            outLocal.SetValue(i, clamped);
        }

        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = myCount * sizeof(DTYPE_Y);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPad(yGm[startNC], outLocal, copyParams);
    }

private:
    __aicore__ inline float ReduceTile(uint64_t gmOffset, uint32_t tileLen)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueue.AllocTensor<DTYPE_X>();

        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = tileLen * sizeof(DTYPE_X);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPadExtParams<DTYPE_X> padParams;
        padParams.isPad = true;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;

        AscendC::DataCopyPad(xLocal, xGm[gmOffset], copyParams, padParams);
        inQueue.EnQue(xLocal);

        AscendC::LocalTensor<DTYPE_X> xIn = inQueue.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> tmpLocal = reduceBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> resLocal = scalarBuf.Get<DTYPE_X>();

        AscendC::ReduceSum<DTYPE_X, true>(resLocal, xIn, tmpLocal, (int32_t)tileLen);
        AscendC::PipeBarrier<PIPE_ALL>();
        float partial = (float)resLocal.GetValue(0);
        inQueue.FreeTensor(xIn);

        return partial;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t startNC;
    uint32_t myCount;
    uint32_t reduceSize;
    float invReduceSize;
};

extern "C" __global__ __aicore__ void conv_transpose3d_multiply_max_global_avg_pool_clamp_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelAvgClamp op;
    op.Init(x, y, tiling_data.ncCount, tiling_data.reduceSize);
    op.Process();
}

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, scale, maxpool_kernel_size):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)

        x = self.conv_transpose(x)
        x = x * self.scale
        x = self.maxpool(x)
        N, C = x.shape[0], x.shape[1]
        D, H, W = x.shape[2], x.shape[3], x.shape[4]
