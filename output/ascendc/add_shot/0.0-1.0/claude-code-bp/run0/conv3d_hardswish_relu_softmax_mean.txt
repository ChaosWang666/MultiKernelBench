kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;
constexpr uint32_t NUM_GROUPS = 4;
constexpr uint32_t CHANNELS_PER_GROUP = 4;
constexpr uint32_t MAX_CHANNELS = 64;

class KernelConv3dPostProc {
public:
    __aicore__ inline KernelConv3dPostProc() {}

    __aicore__ inline void Init(GM_ADDR conv_out, GM_ADDR gamma, GM_ADDR beta, GM_ADDR output,
                                uint32_t B, uint32_t C, uint32_t spatialSize,
                                uint32_t batchesPerBlock, uint32_t blockDim)
    {
        this->B = B;
        this->C = C;
        this->spatialSize = spatialSize;
        this->spatialSizeAlign = (spatialSize + 7) / 8 * 8;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->batchStart = blockIdx * batchesPerBlock;
        this->batchEnd = this->batchStart + batchesPerBlock;
        if (this->batchEnd > B) this->batchEnd = B;

        convOutGm.SetGlobalBuffer((__gm__ float*)conv_out, (uint64_t)B * C * spatialSize);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, C);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, C);
        outputGm.SetGlobalBuffer((__gm__ float*)output, (uint64_t)B * C);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, spatialSizeAlign * sizeof(float));
        pipe.InitBuffer(workBuf, spatialSizeAlign * sizeof(float));
        pipe.InitBuffer(reduceTmpBuf, 4096);
        pipe.InitBuffer(reduceResBuf, 64);
        pipe.InitBuffer(gammaBuf, 256);
        pipe.InitBuffer(betaBuf, 256);
        pipe.InitBuffer(outBuf, 256);
        pipe.InitBuffer(sumBuf, 256);
        pipe.InitBuffer(sumSqBuf, 256);
        pipe.InitBuffer(stdBuf, 64);
    }

    __aicore__ inline void Process()
    {
        if (batchStart >= batchEnd) return;

        LoadParams();

        for (uint32_t b = batchStart; b < batchEnd; b++) {
            ProcessBatch(b);
        }
    }

private:
    __aicore__ inline void LoadParams() {
        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();

        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = (uint32_t)(C * sizeof(float));
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;

        AscendC::DataCopyPadExtParams<float> padParams;
        padParams.isPad = true;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0.0f;

        AscendC::DataCopyPad(gammaLocal, gammaGm, copyParams, padParams);
        AscendC::DataCopyPad(betaLocal, betaGm, copyParams, padParams);

        AscendC::PipeBarrier<PIPE_ALL>();
    }

    __aicore__ inline void ProcessBatch(uint32_t b)
    {
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> sumSqLocal = sumSqBuf.Get<float>();

        for (uint32_t c = 0; c < C; c++) {
            ProcessChannel(b, c, sumLocal, sumSqLocal);
        }

        ComputeOutput(b, sumLocal, sumSqLocal);
    }

    __aicore__ inline void ProcessChannel(uint32_t b, uint32_t c,
                                           AscendC::LocalTensor<float> sumLocal,
                                           AscendC::LocalTensor<float> sumSqLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();

        uint64_t offset = ((uint64_t)b * C + c) * spatialSize;

        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = (uint32_t)(spatialSize * sizeof(float));
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;

        AscendC::DataCopyPadExtParams<float> padParams;
        padParams.isPad = true;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0.0f;

        AscendC::DataCopyPad(xLocal, convOutGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);

        AscendC::LocalTensor<float> x = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceTmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceRes = reduceResBuf.Get<float>();

        // HardSwish: y = x * clamp(x+3, 0, 6) / 6
        AscendC::Adds<float>(work, x, 3.0f, spatialSize);
        AscendC::Maxs<float>(work, work, 0.0f, spatialSize);
        AscendC::Mins<float>(work, work, 6.0f, spatialSize);
        AscendC::Mul<float>(work, work, x, spatialSize);
        AscendC::Muls<float>(work, work, 1.0f / 6.0f, spatialSize);

        AscendC::PipeBarrier<PIPE_V>();

        // sum
        AscendC::ReduceSum<float, true>(reduceRes, work, reduceTmp, (int32_t)spatialSize);
        AscendC::PipeBarrier<PIPE_ALL>();
        float sumVal = reduceRes.GetValue(0);
        sumLocal.SetValue(c, sumVal);

        // sum of squares: square work into x (reuse)
        AscendC::Mul<float>(x, work, work, spatialSize);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::ReduceSum<float, true>(reduceRes, x, reduceTmp, (int32_t)spatialSize);
        AscendC::PipeBarrier<PIPE_ALL>();
        float sumSqVal = reduceRes.GetValue(0);
        sumSqLocal.SetValue(c, sumSqVal);

        inQueueX.FreeTensor(x);
    }

    __aicore__ inline void ComputeOutput(uint32_t b,
                                          AscendC::LocalTensor<float> sumLocal,
                                          AscendC::LocalTensor<float> sumSqLocal)
    {
        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();
        AscendC::LocalTensor<float> stdLocal = stdBuf.Get<float>();

        float eps = 1e-5f;
        float spatialF = (float)spatialSize;
        float groupN = (float)(CHANNELS_PER_GROUP * spatialSize);
        float invSpatial = 1.0f / spatialF;
        float invGroupN = 1.0f / groupN;

        float grpMean[NUM_GROUPS];

        for (uint32_t g = 0; g < NUM_GROUPS; g++) {
            float s = 0.0f;
            float ss = 0.0f;
            for (uint32_t i = 0; i < CHANNELS_PER_GROUP; i++) {
                uint32_t c = g * CHANNELS_PER_GROUP + i;
                s += sumLocal.GetValue(c);
                ss += sumSqLocal.GetValue(c);
            }
            float mean = s * invGroupN;
            float var = ss * invGroupN - mean * mean;
            if (var < 0.0f) var = 0.0f;
            grpMean[g] = mean;
            stdLocal.SetValue(g, var + eps);
        }

        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Sqrt<float>(stdLocal, stdLocal, 8);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t c = 0; c < C; c++) {
            uint32_t g = c / CHANNELS_PER_GROUP;
            float channelMean = sumLocal.GetValue(c) * invSpatial;
            float stdVal = stdLocal.GetValue(g);
            float invStd = 1.0f / stdVal;
            float normVal = (channelMean - grpMean[g]) * invStd;
            float result = normVal * gammaLocal.GetValue(c) + betaLocal.GetValue(c);
            outLocal.SetValue(c, result);
        }

        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::DataCopyExtParams outCopyParams;
        outCopyParams.blockCount = 1;
        outCopyParams.blockLen = (uint32_t)(C * sizeof(float));
        outCopyParams.srcStride = 0;
        outCopyParams.dstStride = 0;
        outCopyParams.rsv = 0;

        AscendC::DataCopyPad(outputGm[(uint64_t)b * C], outLocal, outCopyParams);
        AscendC::PipeBarrier<PIPE_ALL>();
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceTmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceResBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> betaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumSqBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> stdBuf;
    AscendC::GlobalTensor<float> convOutGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    AscendC::GlobalTensor<float> outputGm;
    uint32_t B;
    uint32_t C;
    uint32_t spatialSize;
    uint32_t spatialSizeAlign;
    uint32_t batchStart;
    uint32_t batchEnd;
};

extern "C" __global__ __aicore__ void conv3d_hardswish_relu_softmax_mean_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dPostProc op;
    op.Init(x, gamma, beta, z,
            tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize,
            tiling_data.batchesPerBlock, tiling_data.blockDim);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import torch.nn.functional as F
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, num_groups=4, bias=True):
        super().__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size, bias=bias)
        self.group_norm = nn.GroupNorm(num_groups, out_channels)
        self.num_groups = num_groups

    def forward(self, x):
        conv_out = self.conv(x).contiguous()
        gamma = self.group_norm.weight.contiguous()
        beta = self.group_norm.bias.contiguous()
        return custom_ops_lib.conv3d_hardswish_relu_softmax_mean_custom(conv_out, gamma, beta)
