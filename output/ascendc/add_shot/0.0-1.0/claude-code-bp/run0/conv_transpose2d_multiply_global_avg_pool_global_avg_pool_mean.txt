kernel_src = """
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t TILE_LENGTH = 8192;
constexpr uint32_t MAX_ROWS_PER_CORE = 256;

class KernelConvTransposeMultiplyMean {
public:
    __aicore__ inline KernelConvTransposeMultiplyMean() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t totalRows, uint32_t rLength,
                                 uint32_t usedCoreNum, uint32_t rowsPerCore,
                                 float multiplier)
    {
        this->totalRows = totalRows;
        this->rLength = rLength;
        this->invCountMul = multiplier / (float)rLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->hasWork = (blockIdx < usedCoreNum) ? 1 : 0;

        uint32_t startRow = blockIdx * rowsPerCore;
        uint32_t endRow = startRow + rowsPerCore;
        if (endRow > totalRows) endRow = totalRows;
        if (startRow >= totalRows) {
            this->hasWork = 0;
            startRow = 0;
            endRow = 0;
        }
        this->startRow = startRow;
        this->endRow = endRow;
        this->numLocalRows = endRow - startRow;

        if (this->hasWork) {
            xGm.SetGlobalBuffer((__gm__ float*)x + startRow * rLength, this->numLocalRows * rLength);
            yGm.SetGlobalBuffer((__gm__ float*)y + startRow, this->numLocalRows);

            pipe.InitBuffer(inQueueX, BUFFER_NUM, TILE_LENGTH * sizeof(float));
            pipe.InitBuffer(reduceBuf, 4096);
            pipe.InitBuffer(partialBuf, 32);
            pipe.InitBuffer(outBuf, MAX_ROWS_PER_CORE * sizeof(float));
        }
    }

    __aicore__ inline void Process() {
        if (!this->hasWork) return;

        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();

        for (uint32_t localRow = 0; localRow < this->numLocalRows; localRow++) {
            float result = ProcessRow(localRow);
            outLocal.SetValue(localRow, result);
        }

        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::DataCopyExtParams outCopyParams;
        outCopyParams.blockCount = 1;
        outCopyParams.blockLen = this->numLocalRows * sizeof(float);
        outCopyParams.srcStride = 0;
        outCopyParams.dstStride = 0;
        outCopyParams.rsv = 0;
        AscendC::DataCopyPad(yGm, outLocal, outCopyParams);
    }

private:
    __aicore__ inline float ProcessRow(uint32_t localRow) {
        uint32_t rowOffset = localRow * rLength;
        uint32_t numTiles = (rLength + TILE_LENGTH - 1) / TILE_LENGTH;

        float sumAccum = 0.0f;

        for (uint32_t t = 0; t < numTiles; t++) {
            uint32_t tileStart = t * TILE_LENGTH;
            uint32_t tileLen = (tileStart + TILE_LENGTH <= rLength) ? TILE_LENGTH : (rLength - tileStart);

            CopyInTile(rowOffset + tileStart, tileLen);
            sumAccum += ComputeTile(tileLen);
        }

        return sumAccum * invCountMul;
    }

    __aicore__ inline void CopyInTile(uint32_t gmOffset, uint32_t tileLen) {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = tileLen * sizeof(float);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPadExtParams<float> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0.0f;
        AscendC::DataCopyPad(xLocal, xGm[gmOffset], copyParams, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline float ComputeTile(uint32_t tileLen) {
        AscendC::LocalTensor<float> xIn = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> partialSumLocal = partialBuf.Get<float>();

        AscendC::ReduceSum<float, true>(partialSumLocal, xIn, reduceTmp, (int32_t)tileLen);

        float partialSum = partialSumLocal.GetValue(0);

        inQueueX.FreeTensor(xIn);

        return partialSum;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> partialBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t totalRows;
    uint32_t rLength;
    float invCountMul;
    uint32_t startRow;
    uint32_t endRow;
    uint32_t numLocalRows;
    uint32_t hasWork;
};

extern "C" __global__ __aicore__ void conv_transpose2d_multiply_global_avg_pool_global_avg_pool_mean_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTransposeMultiplyMean op;
    op.Init(x, y, tiling_data.totalRows, tiling_data.rLength,
            tiling_data.usedCoreNum, tiling_data.rowsPerCore, tiling_data.multiplier);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, multiplier):
        super().__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)

        x = x.contiguous()

        C = x.shape[1]
        out = custom_ops_lib.conv_transpose2d_multiply_global_avg_pool_global_avg_pool_mean_custom(x, self.multiplier)
