kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr float HARDTANH_MIN = -1.0f;
constexpr float HARDTANH_MAX = 1.0f;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t totalGroups, uint32_t elementsPerGroup,
                                 uint32_t tileLength)
    {
        this->totalGroups = totalGroups;
        this->elementsPerGroup = elementsPerGroup;
        this->tileLength = tileLength;
        this->invCount = 1.0f / (float)elementsPerGroup;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t groupsPerBlock = totalGroups / blockNum;
        uint32_t remainder = totalGroups % blockNum;
        if (blockIdx < remainder) {
            this->startGroup = blockIdx * (groupsPerBlock + 1);
            this->myGroups = groupsPerBlock + 1;
        } else {
            this->startGroup = blockIdx * groupsPerBlock + remainder;
            this->myGroups = groupsPerBlock;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x);
        yGm.SetGlobalBuffer((__gm__ float *)y);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(reduceBuf, 2048);
        pipe.InitBuffer(scalarBuf, 32);
        pipe.InitBuffer(outBuf, 4096);
    }

    __aicore__ inline void Process()
    {
        if (myGroups == 0) return;

        AscendC::LocalTensor<float> reduceTmp = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> scalarTmp = scalarBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();

        uint32_t alignedGroups = ((myGroups + 7) / 8) * 8;
        AscendC::Duplicate<float>(outLocal, 0.0f, alignedGroups);

        for (uint32_t g = 0; g < myGroups; g++) {
            uint32_t groupIdx = startGroup + g;
            uint64_t baseOffset = (uint64_t)groupIdx * (uint64_t)elementsPerGroup;

            float accum = 0.0f;
            uint32_t numTiles = (elementsPerGroup + tileLength - 1) / tileLength;

            for (uint32_t t = 0; t < numTiles; t++) {
                uint32_t offset = t * tileLength;
                uint32_t currentLen = elementsPerGroup - offset;
                if (currentLen > tileLength) currentLen = tileLength;

                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();

                AscendC::DataCopyExtParams copyInParams;
                copyInParams.blockCount = 1;
                copyInParams.blockLen = currentLen * sizeof(float);
                copyInParams.srcStride = 0;
                copyInParams.dstStride = 0;

                AscendC::DataCopyPadExtParams<float> padParams;
                padParams.isPad = true;
                padParams.leftPadding = 0;
                padParams.rightPadding = 0;
                padParams.paddingValue = 0.0f;

                AscendC::DataCopyPad(xLocal, xGm[baseOffset + offset], copyInParams, padParams);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<float> xIn = inQueueX.DeQue<float>();
                AscendC::Maxs<float>(xIn, xIn, HARDTANH_MIN, currentLen);
                AscendC::Mins<float>(xIn, xIn, HARDTANH_MAX, currentLen);
                AscendC::ReduceSum<float, true>(scalarTmp, xIn, reduceTmp, currentLen);
                accum += scalarTmp.GetValue(0);

                inQueueX.FreeTensor(xIn);
            }

            outLocal.SetValue(g, accum * invCount);
        }

        AscendC::Tanh<float>(outLocal, outLocal, alignedGroups);

        AscendC::DataCopyExtParams copyOutParams;
        copyOutParams.blockCount = 1;
        copyOutParams.blockLen = myGroups * sizeof(float);
        copyOutParams.srcStride = 0;
        copyOutParams.dstStride = 0;
        AscendC::DataCopyPad(yGm[startGroup], outLocal, copyOutParams);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t totalGroups;
    uint32_t elementsPerGroup;
    uint32_t tileLength;
    float invCount;
    uint32_t startGroup;
    uint32_t myGroups;
};

extern "C" __global__ __aicore__ void conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, y, tiling_data.totalGroups, tiling_data.elementsPerGroup, tiling_data.tileLength);
    op.Process();
}
