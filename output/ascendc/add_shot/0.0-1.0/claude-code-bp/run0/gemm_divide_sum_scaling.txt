我先分析下数学优化机会：

`y = sum(x @ W.T / 2, dim=1) * scale`
   `= sum_i(x[b,i] * (scale/2 * sum_h(W[h,i])))`
   `= x @ v`，其中 `v = (scale/2) * W.sum(dim=0)`

这把 O(B·H·I) 的 GEMM 降为 O(H·I) + O(B·I) 的 matvec，巨大优化。

kernel_src = """
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmDivideSumScaling {
public:
    __aicore__ inline KernelGemmDivideSumScaling() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR v, GM_ADDR y,
                                 uint32_t totalRows, uint32_t inputSize) {
        uint32_t coreIdx = AscendC::GetBlockIdx();
        uint32_t numCores = AscendC::GetBlockNum();
        uint32_t rowsPerCore = (totalRows + numCores - 1) / numCores;
        uint32_t startRow = coreIdx * rowsPerCore;

        this->inputSize = inputSize;

        if (startRow >= totalRows) {
            this->myRows = 0;
        } else {
            uint32_t endRow = (startRow + rowsPerCore < totalRows) ? (startRow + rowsPerCore) : totalRows;
            this->myRows = endRow - startRow;
        }

        pipe.InitBuffer(vQueue, 1, inputSize * sizeof(float));
        pipe.InitBuffer(xQueue, BUFFER_NUM, inputSize * sizeof(float));
        pipe.InitBuffer(productBuf, inputSize * sizeof(float));
        pipe.InitBuffer(reduceBuf, 8192);

        uint32_t outAligned = ((rowsPerCore + 7) / 8) * 8;
        if (outAligned == 0) outAligned = 8;
        pipe.InitBuffer(outQueue, 1, outAligned * sizeof(float));

        if (this->myRows > 0) {
            xGm.SetGlobalBuffer((__gm__ float*)x + startRow * inputSize, this->myRows * inputSize);
            yGm.SetGlobalBuffer((__gm__ float*)y + startRow, this->myRows);
        }
        vGm.SetGlobalBuffer((__gm__ float*)v, inputSize);
    }

    __aicore__ inline void Process() {
        if (this->myRows == 0) return;

        AscendC::LocalTensor<float> vLocal = vQueue.AllocTensor<float>();
        AscendC::DataCopy(vLocal, vGm, this->inputSize);
        vQueue.EnQue(vLocal);
        vLocal = vQueue.DeQue<float>();

        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> productLocal = productBuf.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceBuf.Get<float>();

        for (uint32_t row = 0; row < this->myRows; row++) {
            AscendC::LocalTensor<float> xLocal = xQueue.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[row * this->inputSize], this->inputSize);
            xQueue.EnQue(xLocal);

            xLocal = xQueue.DeQue<float>();
            AscendC::Mul(productLocal, xLocal, vLocal, this->inputSize);
            AscendC::ReduceSum<float, true>(outLocal[row], productLocal, reduceTmp,
                                            (int32_t)this->inputSize);
            xQueue.FreeTensor(xLocal);
        }

        outQueue.EnQue(outLocal);
        outLocal = outQueue.DeQue<float>();

        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = this->myRows * (uint32_t)sizeof(float);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPad(yGm, outLocal, copyParams);

        outQueue.FreeTensor(outLocal);
        vQueue.FreeTensor(vLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> vQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> xQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> productBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> vGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t myRows;
    uint32_t inputSize;
};

extern "C" __global__ __aicore__ void gemm_divide_sum_scaling_custom(
    GM_ADDR x, GM_ADDR v, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmDivideSumScaling op;
    op.Init(x, v, y, tiling_data.totalRows, tiling_data.inputSize);
    op.Process();
}
"""

model_src = '''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(torch.nn.Module):
    def __init__(self, input_size, hidden_size, scaling_factor):
        super().__init__()
        self.weight = nn.Parameter(torch.randn(hidden_size, input_size))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        v = (self.weight.sum(dim=0) * (self.scaling_factor / 2.0)).contiguous()
        return custom_ops_lib.gemm_divide_sum_scaling_custom(x.contiguous(), v)
