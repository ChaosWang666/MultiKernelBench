kernel_src = """
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvtr3dReluGn {
public:
    __aicore__ inline KernelConvtr3dReluGn() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
        uint32_t batchSize, uint32_t numChannels, uint32_t spatialSize,
        uint32_t numGroups, uint32_t channelsPerGroup, uint32_t totalGroups,
        uint32_t groupsPerCore, uint32_t alignedSpatial, uint32_t alignedChannels, float eps)
    {
        this->batchSize = batchSize;
        this->numChannels = numChannels;
        this->spatialSize = spatialSize;
        this->numGroups = numGroups;
        this->channelsPerGroup = channelsPerGroup;
        this->totalGroups = totalGroups;
        this->alignedSpatial = alignedSpatial;
        this->alignedChannels = alignedChannels;
        this->eps = eps;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startG = blockIdx * groupsPerCore;
        uint32_t endG = startG + groupsPerCore;
        if (endG > totalGroups) endG = totalGroups;
        this->startGroup = startG;
        this->endGroup = endG;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        yGm.SetGlobalBuffer((__gm__ float*)y);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma);
        betaGm.SetGlobalBuffer((__gm__ float*)beta);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedSpatial * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, alignedSpatial * sizeof(float));
        pipe.InitBuffer(tmpBuf, alignedSpatial * sizeof(float));
        pipe.InitBuffer(zeroBuf, alignedSpatial * sizeof(float));
        pipe.InitBuffer(scalarBuf, 32);
        pipe.InitBuffer(reduceBuf, 8 * 1024);
        pipe.InitBuffer(gammaBuf, alignedChannels * sizeof(float));
        pipe.InitBuffer(betaBuf, alignedChannels * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        // Load gamma and beta to UB once
        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();

        AscendC::DataCopyExtParams gCopy;
        gCopy.blockCount = 1;
        gCopy.blockLen = static_cast<uint32_t>(numChannels * sizeof(float));
        gCopy.srcStride = 0;
        gCopy.dstStride = 0;
        gCopy.rsv = 0;
        AscendC::DataCopyPadExtParams<float> gPad;
        gPad.isPad = false;
        gPad.leftPadding = 0;
        gPad.rightPadding = 0;
        gPad.paddingValue = 0.0f;

        AscendC::DataCopyPad(gammaLocal, gammaGm, gCopy, gPad);
        AscendC::DataCopyPad(betaLocal, betaGm, gCopy, gPad);

        // Prepare zero tensor for ReLU via Max
        AscendC::LocalTensor<float> zeroLocal = zeroBuf.Get<float>();
        AscendC::Duplicate<float>(zeroLocal, 0.0f, alignedSpatial);

        AscendC::PipeBarrier<PIPE_ALL>();

        if (startGroup >= endGroup) return;
        for (uint32_t g = startGroup; g < endGroup; g++) {
            ProcessGroup(g, gammaLocal, betaLocal, zeroLocal);
        }
    }

private:
    __aicore__ inline void ProcessGroup(uint32_t groupIdx,
        AscendC::LocalTensor<float>& gammaLocal,
        AscendC::LocalTensor<float>& betaLocal,
        AscendC::LocalTensor<float>& zeroLocal)
    {
        uint32_t batchIdx = groupIdx / numGroups;
        uint32_t grpInBatch = groupIdx % numGroups;
        uint32_t channelStart = grpInBatch * channelsPerGroup;

        uint64_t elementsInGroup = (uint64_t)channelsPerGroup * spatialSize;
        float invN = 1.0f / (float)elementsInGroup;

        float sumVal = 0.0f;
        float sumSqVal = 0.0f;

        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceBuf.Get<float>();

        // ========== Pass 1: Compute sum and sum_sq with ReLU ==========
        for (uint32_t c = 0; c < channelsPerGroup; c++) {
            uint32_t chIdx = channelStart + c;
            uint64_t gmOffset = (uint64_t)batchIdx * numChannels * spatialSize
                              + (uint64_t)chIdx * spatialSize;

            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopyExtParams copyParams;
            copyParams.blockCount = 1;
            copyParams.blockLen = static_cast<uint32_t>(spatialSize * sizeof(float));
            copyParams.srcStride = 0;
            copyParams.dstStride = 0;
            copyParams.rsv = 0;
            AscendC::DataCopyPadExtParams<float> padParams;
            padParams.isPad = false;
            padParams.leftPadding = 0;
            padParams.rightPadding = 0;
            padParams.paddingValue = 0.0f;
            AscendC::DataCopyPad(xLocal, xGm[gmOffset], copyParams, padParams);
            inQueueX.EnQue(xLocal);

            AscendC::LocalTensor<float> xIn = inQueueX.DeQue<float>();

            // ReLU: tmp = max(xIn, 0)
            AscendC::Max<float>(tmp, xIn, zeroLocal, spatialSize);

            // Sum of ReLU values
            AscendC::ReduceSum<float, true>(scalarLocal, tmp, reduceTmp, spatialSize);
            sumVal += scalarLocal.GetValue(0);

            // Sum of squares of ReLU values
            AscendC::Mul<float>(tmp, tmp, tmp, spatialSize);
            AscendC::ReduceSum<float, true>(scalarLocal, tmp, reduceTmp, spatialSize);
            sumSqVal += scalarLocal.GetValue(0);

            inQueueX.FreeTensor(xIn);
        }

        float meanVal = sumVal * invN;
        float varVal = sumSqVal * invN - meanVal * meanVal;
        if (varVal < 0.0f) varVal = 0.0f;

        // Compute inv_std using Sqrt on vector
        AscendC::Duplicate<float>(scalarLocal, varVal + eps, 8);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Sqrt<float>(scalarLocal, scalarLocal, 8);
        AscendC::PipeBarrier<PIPE_ALL>();
        float stdVal = scalarLocal.GetValue(0);
        float invStd = 1.0f / stdVal;

        // ========== Pass 2: ReLU + Normalize + Affine ==========
        for (uint32_t c = 0; c < channelsPerGroup; c++) {
            uint32_t chIdx = channelStart + c;
            uint64_t gmOffset = (uint64_t)batchIdx * numChannels * spatialSize
                              + (uint64_t)chIdx * spatialSize;

            float gammaVal = gammaLocal.GetValue(chIdx);
            float betaVal = betaLocal.GetValue(chIdx);
            float scale = invStd * gammaVal;
            float shift = betaVal - meanVal * scale;

            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopyExtParams copyParams;
            copyParams.blockCount = 1;
            copyParams.blockLen = static_cast<uint32_t>(spatialSize * sizeof(float));
            copyParams.srcStride = 0;
            copyParams.dstStride = 0;
            copyParams.rsv = 0;
            AscendC::DataCopyPadExtParams<float> padParams;
            padParams.isPad = false;
            padParams.leftPadding = 0;
            padParams.rightPadding = 0;
            padParams.paddingValue = 0.0f;
            AscendC::DataCopyPad(xLocal, xGm[gmOffset], copyParams, padParams);
            inQueueX.EnQue(xLocal);

            AscendC::LocalTensor<float> xIn = inQueueX.DeQue<float>();
            AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

            // ReLU: yLocal = max(xIn, 0)
            AscendC::Max<float>(yLocal, xIn, zeroLocal, spatialSize);
            // Scale and shift
            AscendC::Muls<float>(yLocal, yLocal, scale, spatialSize);
            AscendC::Adds<float>(yLocal, yLocal, shift, spatialSize);

            outQueueY.EnQue(yLocal);
            inQueueX.FreeTensor(xIn);

            AscendC::LocalTensor<float> yOut = outQueueY.DeQue<float>();
            AscendC::DataCopyExtParams copyOutParams;
            copyOutParams.blockCount = 1;
            copyOutParams.blockLen = static_cast<uint32_t>(spatialSize * sizeof(float));
            copyOutParams.srcStride = 0;
            copyOutParams.dstStride = 0;
            copyOutParams.rsv = 0;
            AscendC::DataCopyPad(yGm[gmOffset], yOut, copyOutParams);
            outQueueY.FreeTensor(yOut);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> zeroBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> betaBuf;
    AscendC::GlobalTensor<float> xGm, yGm, gammaGm, betaGm;

    uint32_t batchSize, numChannels, spatialSize, numGroups, channelsPerGroup;
    uint32_t totalGroups, startGroup, endGroup, alignedSpatial, alignedChannels;
    float eps;
};

extern "C" __global__ __aicore__ void convtranspose3d_relu_groupnorm_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvtr3dReluGn op;
    op.Init(x, gamma, beta, z,
        tiling_data.batchSize, tiling_data.numChannels, tiling_data.spatialSize,
        tiling_data.numGroups, tiling_data.channelsPerGroup, tiling_data.totalGroups,
        tiling_data.groupsPerCore, tiling_data.alignedSpatial, tiling_data.alignedChannels,
        tiling_data.eps);
    op.Process();
}

model_src = '''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, groups, bias=False):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, bias=bias)
        self.group_norm = nn.GroupNorm(num_groups=groups, num_channels=out_channels)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = custom_ops_lib.convtranspose3d_relu_groupnorm_custom(
            x.contiguous(),
            self.group_norm.weight.contiguous(),
            self.group_norm.bias.contiguous()
        )
        return x
'''
