kernel_src = """
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTransPost {
public:
    __aicore__ inline KernelConvTransPost() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                 uint32_t totalSlices, uint32_t spatialSize,
                                 uint32_t channels, uint32_t tileSize,
                                 float scalingFactor, float invScalingFactor)
    {
        this->spatialSize = spatialSize;
        this->channels = channels;
        this->tileSize = tileSize;
        this->scalingFactor = scalingFactor;
        this->invScalingFactor = invScalingFactor;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t slicesPerCore = totalSlices / blockNum;
        uint32_t extra = totalSlices % blockNum;

        if (blockIdx < extra) {
            this->startSlice = blockIdx * (slicesPerCore + 1);
            this->numSlices = slicesPerCore + 1;
        } else {
            this->startSlice = extra * (slicesPerCore + 1) + (blockIdx - extra) * slicesPerCore;
            this->numSlices = slicesPerCore;
        }

        uint64_t totalElements = (uint64_t)totalSlices * (uint64_t)spatialSize;
        xGm.SetGlobalBuffer((__gm__ float*)x, totalElements);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalElements);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, channels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileSize * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileSize * sizeof(float));

        uint32_t biasBufSize = ((channels * sizeof(float) + 31) / 32) * 32;
        if (biasBufSize < 32) {
            biasBufSize = 32;
        }
        pipe.InitBuffer(biasBuf, biasBufSize);

        biasLocal = biasBuf.Get<float>();
        uint32_t biasCopyCount = ((channels + 7) / 8) * 8;
        AscendC::DataCopy(biasLocal, biasGm, biasCopyCount);
        AscendC::PipeBarrier<PIPE_ALL>();
    }

    __aicore__ inline void Process()
    {
        if (numSlices == 0) return;
        for (uint32_t s = 0; s < numSlices; s++) {
            uint32_t sliceIdx = startSlice + s;
            uint32_t channelIdx = sliceIdx % channels;
            uint64_t offset = (uint64_t)sliceIdx * (uint64_t)spatialSize;
            float biasVal = biasLocal.GetValue(channelIdx);

            uint32_t numTiles = (spatialSize + tileSize - 1) / tileSize;
            for (uint32_t t = 0; t < numTiles; t++) {
                uint32_t start = t * tileSize;
                uint32_t curLen = (start + tileSize <= spatialSize) ? tileSize : (spatialSize - start);

                CopyIn(offset + start, curLen);
                Compute(biasVal, curLen);
                CopyOut(offset + start, curLen);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], len);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float biasVal, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        AscendC::Adds(yLocal, xLocal, biasVal, len);
        AscendC::Mins(yLocal, yLocal, (float)1.0f, len);
        AscendC::Maxs(yLocal, yLocal, (float)0.0f, len);
        AscendC::Muls(yLocal, yLocal, scalingFactor, len);
        AscendC::Mins(yLocal, yLocal, (float)1.0f, len);
        AscendC::Maxs(yLocal, yLocal, (float)0.0f, len);
        AscendC::Muls(yLocal, yLocal, invScalingFactor, len);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[offset], yLocal, len);
        outQueueY.FreeTensor(yLocal);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::LocalTensor<float> biasLocal;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> biasGm;
    uint32_t spatialSize;
    uint32_t channels;
    uint32_t tileSize;
    float scalingFactor;
    float invScalingFactor;
    uint32_t startSlice;
    uint32_t numSlices;
};

extern "C" __global__ __aicore__ void conv_transpose2d_bias_add_clamp_scaling_clamp_divide_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTransPost op;
    op.Init(x, bias, y,
            tiling_data.totalSlices, tiling_data.spatialSize,
            tiling_data.channels, tiling_data.tileSize,
            tiling_data.scalingFactor, tiling_data.invScalingFactor);
    op.Process();
}

model_src = '''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    """
    Optimized model: conv_transpose kept in PyTorch, remaining elementwise
    chain (bias add + clamp + scale + clamp + divide) fused into a single
    custom AscendC kernel.
    """
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, bias_shape, scaling_factor):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose2d(
            in_channels, out_channels, kernel_size,
            stride=stride, padding=padding, output_padding=output_padding)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.scaling_factor = float(scaling_factor)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x.contiguous()
        bias_flat = self.bias.contiguous().view(-1)
        x = custom_ops_lib.conv_transpose2d_bias_add_clamp_scaling_clamp_divide_custom(
            x, bias_flat, self.scaling_factor)
        return x
