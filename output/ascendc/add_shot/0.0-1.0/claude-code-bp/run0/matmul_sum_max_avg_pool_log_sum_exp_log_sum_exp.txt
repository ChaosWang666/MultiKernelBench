kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulReduce {
public:
    __aicore__ inline KernelMatmulReduce() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w_sum, GM_ADDR b_sum, GM_ADDR z,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t rowsPerCore)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t sr = blockIdx * rowsPerCore;
        uint32_t er = sr + rowsPerCore;
        if (sr > batchSize) sr = batchSize;
        if (er > batchSize) er = batchSize;
        startRow_ = sr;
        rowsThisCore_ = er - sr;
        inFeatures_ = inFeatures;

        xGm_.SetGlobalBuffer((__gm__ float *)x, batchSize * inFeatures);
        wSumGm_.SetGlobalBuffer((__gm__ float *)w_sum, inFeatures);
        bSumGm_.SetGlobalBuffer((__gm__ float *)b_sum, 1);
        zGm_.SetGlobalBuffer((__gm__ float *)z, batchSize);

        pipe_.InitBuffer(inQueueX_, BUFFER_NUM, inFeatures * sizeof(float));
        pipe_.InitBuffer(outQueueZ_, BUFFER_NUM, 32);
        pipe_.InitBuffer(wSumBuf_, inFeatures * sizeof(float));
        pipe_.InitBuffer(bSumBuf_, 32);
        pipe_.InitBuffer(tmpBuf_, inFeatures * sizeof(float));
        pipe_.InitBuffer(reduceBuf_, 8192);
    }

    __aicore__ inline void Process()
    {
        if (rowsThisCore_ == 0) return;

        AscendC::LocalTensor<float> wSumLocal = wSumBuf_.Get<float>();
        AscendC::DataCopy(wSumLocal, wSumGm_, inFeatures_);

        AscendC::LocalTensor<float> bSumLocal = bSumBuf_.Get<float>();
        AscendC::DataCopyExtParams bCopyParams{1, static_cast<uint32_t>(sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<float> bPadParams{false, 0, 0, 0.0f};
        AscendC::DataCopyPad(bSumLocal, bSumGm_, bCopyParams, bPadParams);

        AscendC::PipeBarrier<PIPE_ALL>();
        float bSumVal = bSumLocal.GetValue(0);

        AscendC::LocalTensor<float> tmpLocal = tmpBuf_.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceBuf_.Get<float>();

        for (uint32_t r = 0; r < rowsThisCore_; r++) {
            uint32_t rowIdx = startRow_ + r;

            AscendC::LocalTensor<float> xLocal = inQueueX_.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm_[rowIdx * inFeatures_], inFeatures_);
            inQueueX_.EnQue(xLocal);

            AscendC::LocalTensor<float> xIn = inQueueX_.DeQue<float>();
            AscendC::Mul(tmpLocal, xIn, wSumLocal, inFeatures_);

            AscendC::LocalTensor<float> zLocal = outQueueZ_.AllocTensor<float>();
            AscendC::ReduceSum<float, true>(zLocal, tmpLocal, reduceTmp, static_cast<int32_t>(inFeatures_));

            float val = zLocal.GetValue(0) + bSumVal;
            zLocal.SetValue(0, val);

            outQueueZ_.EnQue<float>(zLocal);
            inQueueX_.FreeTensor(xIn);

            AscendC::LocalTensor<float> zOut = outQueueZ_.DeQue<float>();
            AscendC::DataCopyExtParams zCopyParams{1, static_cast<uint32_t>(sizeof(float)), 0, 0, 0};
            AscendC::DataCopyPad(zGm_[rowIdx], zOut, zCopyParams);
            outQueueZ_.FreeTensor(zOut);
        }
    }

private:
    AscendC::TPipe pipe_;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX_;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ_;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wSumBuf_, bSumBuf_, tmpBuf_, reduceBuf_;
    AscendC::GlobalTensor<float> xGm_, wSumGm_, bSumGm_, zGm_;
    uint32_t startRow_;
    uint32_t rowsThisCore_;
    uint32_t inFeatures_;
};

extern "C" __global__ __aicore__ void matmul_sum_max_avg_pool_log_sum_exp_log_sum_exp_custom(
    GM_ADDR x, GM_ADDR w_sum, GM_ADDR b_sum, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulReduce op;
    op.Init(x, w_sum, b_sum, z, tiling_data.batchSize, tiling_data.inFeatures, tiling_data.rowsPerCore);
    op.Process();
}
