kernel_src="""
#include "kernel_operator.h"

constexpr uint32_t TOTAL_CHANNELS = 8192;
constexpr uint32_t NUM_GROUPS = 256;
constexpr uint32_t CHANNELS_PER_GROUP = 32;

class KernelGemmBiasAddHardtanhMishGroupNorm {
public:
    __aicore__ inline KernelGemmBiasAddHardtanhMishGroupNorm() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                 uint32_t batchSize, float eps)
    {
        this->batchSize = batchSize;
        this->eps = eps;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t rowsPerCore = (batchSize + blockNum - 1) / blockNum;
        uint32_t startRow = blockIdx * rowsPerCore;
        uint32_t endRow = startRow + rowsPerCore;
        if (endRow > batchSize) endRow = batchSize;
        this->actualRows = (startRow < batchSize) ? (endRow - startRow) : 0;

        xGm.SetGlobalBuffer((__gm__ float*)x + startRow * TOTAL_CHANNELS, actualRows * TOTAL_CHANNELS);
        yGm.SetGlobalBuffer((__gm__ float*)y + startRow * TOTAL_CHANNELS, actualRows * TOTAL_CHANNELS);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, TOTAL_CHANNELS);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, TOTAL_CHANNELS);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, TOTAL_CHANNELS);

        pipe.InitBuffer(biasBuf, TOTAL_CHANNELS * sizeof(float));
        pipe.InitBuffer(gammaBuf, TOTAL_CHANNELS * sizeof(float));
        pipe.InitBuffer(betaBuf, TOTAL_CHANNELS * sizeof(float));
        pipe.InitBuffer(inQueueX, 1, TOTAL_CHANNELS * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, TOTAL_CHANNELS * sizeof(float));
        pipe.InitBuffer(meansBuf, NUM_GROUPS * sizeof(float));
        pipe.InitBuffer(varsBuf, NUM_GROUPS * sizeof(float));
        pipe.InitBuffer(onesBuf, NUM_GROUPS * sizeof(float));
        pipe.InitBuffer(reduceTmpBuf, 16 * 1024);
    }

    __aicore__ inline void Process()
    {
        if (actualRows == 0) {
            return;
        }

        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();
        AscendC::LocalTensor<float> onesLocal = onesBuf.Get<float>();

        AscendC::DataCopy(biasLocal, biasGm, TOTAL_CHANNELS);
        AscendC::DataCopy(gammaLocal, gammaGm, TOTAL_CHANNELS);
        AscendC::DataCopy(betaLocal, betaGm, TOTAL_CHANNELS);
        AscendC::Duplicate<float>(onesLocal, 1.0f, NUM_GROUPS);

        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t row = 0; row < actualRows; row++) {
            CopyIn(row);
            Compute(row);
            CopyOut(row);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * TOTAL_CHANNELS], TOTAL_CHANNELS);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();
        AscendC::LocalTensor<float> meansLocal = meansBuf.Get<float>();
        AscendC::LocalTensor<float> varsLocal = varsBuf.Get<float>();
        AscendC::LocalTensor<float> onesLocal = onesBuf.Get<float>();
        AscendC::LocalTensor<uint8_t> reduceTmp = reduceTmpBuf.Get<uint8_t>();

        // 1. BiasAdd: x = x + bias
        AscendC::Add(xLocal, xLocal, biasLocal, TOTAL_CHANNELS);

        // 2. Hardtanh: x = clamp(x, -1, 1)
        AscendC::Mins(xLocal, xLocal, 1.0f, TOTAL_CHANNELS);
        AscendC::Maxs(xLocal, xLocal, -1.0f, TOTAL_CHANNELS);

        // 3. Mish: x = x * tanh(softplus(x)) = x * tanh(log(1+exp(x)))
        AscendC::Exp(yLocal, xLocal, TOTAL_CHANNELS);
        AscendC::Adds(yLocal, yLocal, 1.0f, TOTAL_CHANNELS);
        AscendC::Log(yLocal, yLocal, TOTAL_CHANNELS);
        AscendC::Tanh(yLocal, yLocal, TOTAL_CHANNELS);
        AscendC::Mul(xLocal, xLocal, yLocal, TOTAL_CHANNELS);

        // 4. GroupNorm: 256 groups of 32 elements each
        uint32_t srcShape[2] = {NUM_GROUPS, CHANNELS_PER_GROUP};

        // 4a. Compute means via Pattern::AR
        AscendC::ReduceSum<float, AscendC::Pattern::Reduce::AR, false>(
            meansLocal, xLocal, reduceTmp, srcShape, true);
        AscendC::Muls(meansLocal, meansLocal, 1.0f / static_cast<float>(CHANNELS_PER_GROUP), NUM_GROUPS);

        // 4b. Subtract mean per group: x = x - mean
        for (uint32_t g = 0; g < NUM_GROUPS; g++) {
            float mean = meansLocal.GetValue(g);
            AscendC::Adds(xLocal[g * CHANNELS_PER_GROUP],
                          xLocal[g * CHANNELS_PER_GROUP],
                          -mean, CHANNELS_PER_GROUP);
        }

        // 4c. Compute (x-mean)^2 into yLocal, then sum per group → variance*32
        AscendC::Mul(yLocal, xLocal, xLocal, TOTAL_CHANNELS);
        AscendC::ReduceSum<float, AscendC::Pattern::Reduce::AR, false>(
            varsLocal, yLocal, reduceTmp, srcShape, true);

        // 4d. var = sum/32; std = sqrt(var + eps); invStd = 1/std
        AscendC::Muls(varsLocal, varsLocal, 1.0f / static_cast<float>(CHANNELS_PER_GROUP), NUM_GROUPS);
        AscendC::Adds(varsLocal, varsLocal, eps, NUM_GROUPS);
        AscendC::Sqrt(varsLocal, varsLocal, NUM_GROUPS);
        AscendC::Div(varsLocal, onesLocal, varsLocal, NUM_GROUPS);

        // 4e. Multiply (x-mean) by invStd per group
        for (uint32_t g = 0; g < NUM_GROUPS; g++) {
            float invStd = varsLocal.GetValue(g);
            AscendC::Muls(xLocal[g * CHANNELS_PER_GROUP],
                          xLocal[g * CHANNELS_PER_GROUP],
                          invStd, CHANNELS_PER_GROUP);
        }

        // 4f. y = x * gamma + beta
        AscendC::Mul(yLocal, xLocal, gammaLocal, TOTAL_CHANNELS);
        AscendC::Add(yLocal, yLocal, betaLocal, TOTAL_CHANNELS);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[row * TOTAL_CHANNELS], yLocal, TOTAL_CHANNELS);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> betaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> meansBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> varsBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> onesBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceTmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;

    uint32_t batchSize;
    uint32_t actualRows;
    float eps;
};

extern "C" __global__ __aicore__ void gemm_bias_add_hardtanh_mish_group_norm_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmBiasAddHardtanhMishGroupNorm op;
    op.Init(x, bias, gamma, beta, y, tiling_data.batchSize, tiling_data.eps);
    op.Process();
}
