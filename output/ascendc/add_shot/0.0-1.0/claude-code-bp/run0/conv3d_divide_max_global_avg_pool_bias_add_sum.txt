Looking at this problem, I need to fuse the tail operations (global_avg_pool + bias_add + sum_over_channel) into a custom AscendC kernel, keeping conv3d, divide, and maxpool in PyTorch since they're complex.

**Key insight**: For each batch `n`, the result simplifies to:
```
output[n] = (1/(D*H*W)) * sum_{c,d,h,w}(x[n,c,d,h,w]) + sum_c(bias[c])
```

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t MAX_BATCHES_PER_CORE = 32;

class KernelFusedOp {
public:
    __aicore__ inline KernelFusedOp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                 uint32_t numBatches, uint32_t elementsPerBatch,
                                 uint32_t biasElements, uint32_t tileLength,
                                 uint32_t spatialElements)
    {
        this->elementsPerBatch = elementsPerBatch;
        this->biasElements = biasElements;
        this->tileLength = tileLength;
        this->scale = 1.0f / static_cast<float>(spatialElements);

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t batchesPerCore = numBatches / blockNum;
        uint32_t remainder = numBatches % blockNum;

        if (blockIdx < remainder) {
            this->startBatch = blockIdx * (batchesPerCore + 1);
            this->numBatchesThisCore = batchesPerCore + 1;
        } else {
            this->startBatch = remainder * (batchesPerCore + 1) + (blockIdx - remainder) * batchesPerCore;
            this->numBatchesThisCore = batchesPerCore;
        }

        this->numFullTiles = elementsPerBatch / tileLength;
        this->tailLength = elementsPerBatch % tileLength;

        xGm.SetGlobalBuffer((__gm__ float *)x, numBatches * elementsPerBatch);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, biasElements);
        yGm.SetGlobalBuffer((__gm__ float *)y, numBatches);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(biasBuf, 1024);
        pipe.InitBuffer(reduceBuf, 32 * 1024);
        pipe.InitBuffer(scalarBuf, 64);
        pipe.InitBuffer(outputBuf, 256);
    }

    __aicore__ inline void Process()
    {
        if (numBatchesThisCore == 0) {
            return;
        }

        float biasSum = ComputeBiasSum();

        AscendC::LocalTensor<float> outLocal = outputBuf.Get<float>();

        for (uint32_t b = 0; b < numBatchesThisCore; b++) {
            uint32_t batchIdx = startBatch + b;
            float batchSum = ComputeBatchSum(batchIdx);
            float result = batchSum * scale + biasSum;
            outLocal.SetValue(b, result);
        }

        WriteOutput(outLocal);
    }

private:
    __aicore__ inline float ComputeBiasSum()
    {
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();

        AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(biasElements * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<float> padParams{true, 0, 0, 0.0f};

        AscendC::DataCopyPad(biasLocal, biasGm, copyParams, padParams);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::ReduceSum<float, true>(scalarLocal, biasLocal, reduceTmp, static_cast<int32_t>(biasElements));
        AscendC::PipeBarrier<PIPE_ALL>();
        return scalarLocal.GetValue(0);
    }

    __aicore__ inline float ComputeBatchSum(uint32_t batchIdx)
    {
        float total = 0.0f;
        uint32_t baseOffset = batchIdx * elementsPerBatch;

        for (uint32_t t = 0; t < numFullTiles; t++) {
            total += ProcessTile(baseOffset + t * tileLength, tileLength);
        }
        if (tailLength > 0) {
            total += ProcessTile(baseOffset + numFullTiles * tileLength, tailLength);
        }
        return total;
    }

    __aicore__ inline float ProcessTile(uint32_t offset, uint32_t length)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();

        AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(length * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<float> padParams{true, 0, 0, 0.0f};

        AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        inQueueX.EnQue(xLocal);

        AscendC::LocalTensor<float> xIn = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();

        AscendC::ReduceSum<float, true>(scalarLocal, xIn, reduceTmp, static_cast<int32_t>(length));
        AscendC::PipeBarrier<PIPE_ALL>();
        float val = scalarLocal.GetValue(0);
        inQueueX.FreeTensor(xIn);
        return val;
    }

    __aicore__ inline void WriteOutput(AscendC::LocalTensor<float>& outLocal)
    {
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::DataCopyExtParams copyParams{1, static_cast<uint32_t>(numBatchesThisCore * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPad(yGm[startBatch], outLocal, copyParams);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outputBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t elementsPerBatch;
    uint32_t biasElements;
    uint32_t tileLength;
    uint32_t startBatch;
    uint32_t numBatchesThisCore;
    uint32_t numFullTiles;
    uint32_t tailLength;
    float scale;
};

extern "C" __global__ __aicore__ void conv3d_divide_max_global_avg_pool_bias_add_sum_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedOp op;
    op.Init(x, bias, y,
            tiling_data.numBatches,
            tiling_data.elementsPerBatch,
            tiling_data.biasElements,
            tiling_data.tileLength,
            tiling_data.spatialElements);
    op.Process();
}

import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, divisor, pool_size, bias_shape, sum_dim):
        super(ModelNew, self).__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        self.divisor = divisor
        self.max_pool = torch.nn.MaxPool3d(pool_size)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.sum_dim = sum_dim

    def forward(self, x):
        x = self.conv(x)
        x = x / self.divisor
        x = self.max_pool(x)
        x_cont = x.contiguous()
        bias_flat = self.bias.contiguous().view(-1)
        return custom_ops_lib.conv3d_divide_max_global_avg_pool_bias_add_sum_custom(x_cont, bias_flat)
