kernel_src="""
#include "kernel_operator.h"

class KernelGemmGroupNormSwishMulSwish {
public:
    __aicore__ inline KernelGemmGroupNormSwishMulSwish() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR mw, GM_ADDR y,
                                uint32_t totalRows, uint32_t rowsPerBlock,
                                uint32_t features, uint32_t numGroups, uint32_t groupSize)
    {
        this->totalRows = totalRows;
        this->rowsPerBlock = rowsPerBlock;
        this->features = features;
        this->numGroups = numGroups;
        this->groupSize = groupSize;
        this->eps = 1e-5f;
        this->invGroupSize = 1.0f / (float)groupSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerBlock;
        if (startRow >= totalRows) {
            this->myRows = 0;
            this->startRow = 0;
        } else {
            uint32_t endRow = startRow + rowsPerBlock;
            if (endRow > totalRows) endRow = totalRows;
            this->myRows = endRow - startRow;
            this->startRow = startRow;
        }

        uint32_t totalElems = this->myRows * features;
        if (totalElems == 0) totalElems = features;

        xGm.SetGlobalBuffer((__gm__ float*)x + this->startRow * features, totalElems);
        yGm.SetGlobalBuffer((__gm__ float*)y + this->startRow * features, totalElems);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, features);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, features);
        mwGm.SetGlobalBuffer((__gm__ float*)mw, features);

        pipe.InitBuffer(inQueueX, 1, features * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, features * sizeof(float));
        pipe.InitBuffer(gammaBuf, features * sizeof(float));
        pipe.InitBuffer(betaBuf, features * sizeof(float));
        pipe.InitBuffer(mwBuf, features * sizeof(float));
        pipe.InitBuffer(reduceBuf, 1024);
        pipe.InitBuffer(scalarBuf, 32);
    }

    __aicore__ inline void Process()
    {
        if (this->myRows == 0) return;

        LoadConstants();

        for (uint32_t i = 0; i < this->myRows; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void LoadConstants()
    {
        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();
        AscendC::LocalTensor<float> mwLocal = mwBuf.Get<float>();

        AscendC::DataCopy(gammaLocal, gammaGm, features);
        AscendC::DataCopy(betaLocal, betaGm, features);
        AscendC::DataCopy(mwLocal, mwGm, features);

        event_t eventId = static_cast<event_t>(0);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(eventId);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(eventId);
    }

    __aicore__ inline void CopyIn(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * features], features);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(int32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();
        AscendC::LocalTensor<float> mwLocal = mwBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();

        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t offset = g * groupSize;

            AscendC::ReduceSum<float, true>(scalarLocal, xLocal[offset], reduceLocal, (int32_t)groupSize);
            AscendC::PipeBarrier<PIPE_V>();
            float mean = scalarLocal.GetValue(0) * invGroupSize;

            AscendC::Adds<float>(xLocal[offset], xLocal[offset], -mean, (int32_t)groupSize);

            AscendC::Mul<float>(yLocal[offset], xLocal[offset], xLocal[offset], (int32_t)groupSize);
            AscendC::ReduceSum<float, true>(scalarLocal, yLocal[offset], reduceLocal, (int32_t)groupSize);
            AscendC::PipeBarrier<PIPE_V>();
            float var = scalarLocal.GetValue(0) * invGroupSize;

            AscendC::Duplicate<float>(scalarLocal, var + eps, 8);
            AscendC::Sqrt<float>(scalarLocal, scalarLocal, 8);
            AscendC::PipeBarrier<PIPE_V>();
            float stdev = scalarLocal.GetValue(0);
            float invstd = 1.0f / stdev;

            AscendC::Muls<float>(xLocal[offset], xLocal[offset], invstd, (int32_t)groupSize);
        }

        AscendC::Mul<float>(xLocal, xLocal, gammaLocal, (int32_t)features);
        AscendC::Add<float>(xLocal, xLocal, betaLocal, (int32_t)features);

        AscendC::Muls<float>(yLocal, xLocal, -1.0f, (int32_t)features);
        AscendC::Exp<float>(yLocal, yLocal, (int32_t)features);
        AscendC::Adds<float>(yLocal, yLocal, 1.0f, (int32_t)features);
        AscendC::Div<float>(xLocal, xLocal, yLocal, (int32_t)features);

        AscendC::Mul<float>(xLocal, xLocal, mwLocal, (int32_t)features);

        AscendC::Muls<float>(yLocal, xLocal, -1.0f, (int32_t)features);
        AscendC::Exp<float>(yLocal, yLocal, (int32_t)features);
        AscendC::Adds<float>(yLocal, yLocal, 1.0f, (int32_t)features);
        AscendC::Div<float>(yLocal, xLocal, yLocal, (int32_t)features);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(int32_t progress)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[progress * features], yLocal, features);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaBuf, betaBuf, mwBuf, reduceBuf, scalarBuf;
    AscendC::GlobalTensor<float> xGm, yGm, gammaGm, betaGm, mwGm;
    uint32_t totalRows;
    uint32_t rowsPerBlock;
    uint32_t features;
    uint32_t numGroups;
    uint32_t groupSize;
    uint32_t myRows;
    uint32_t startRow;
    float eps;
    float invGroupSize;
};

extern "C" __global__ __aicore__ void gemm_group_norm_swish_multiply_swish_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR mw, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormSwishMulSwish op;
    op.Init(x, gamma, beta, mw, y,
            tiling_data.totalRows, tiling_data.rowsPerBlock,
            tiling_data.features, tiling_data.numGroups, tiling_data.groupSize);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, multiply_weight_shape):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.group_norm = nn.GroupNorm(num_groups, out_features)
        self.multiply_weight = nn.Parameter(torch.randn(multiply_weight_shape))

    def forward(self, x):
        x = self.gemm(x)
        x = x.contiguous()
        gamma = self.group_norm.weight.contiguous()
        beta = self.group_norm.bias.contiguous()
        mw = self.multiply_weight.contiguous()
        return custom_ops_lib.gemm_group_norm_swish_multiply_swish_custom(x, gamma, beta, mw)
