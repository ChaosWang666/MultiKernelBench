kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;
constexpr float NEG_INF_F = -3.4e38f;
constexpr float EPS = 1e-5f;

class KernelConv3dMimcmm {
public:
    __aicore__ inline KernelConv3dMimcmm() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mult, GM_ADDR z,
                                uint32_t N, uint32_t C, uint32_t spatial,
                                uint32_t nsPerBlock, uint32_t remainderNs,
                                uint32_t usedBlocks, float clampMin, float clampMax)
    {
        this->N_ = N;
        this->C_ = C;
        this->spatial_ = spatial;
        this->clampMin_ = clampMin;
        this->clampMax_ = clampMax;
        this->invSpatial_ = 1.0f / (float)spatial;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        if (blockIdx >= usedBlocks) {
            this->blockNumNs_ = 0;
            this->blockStartN_ = 0;
            return;
        }

        if (blockIdx < remainderNs) {
            this->blockNumNs_ = nsPerBlock + 1;
            this->blockStartN_ = blockIdx * (nsPerBlock + 1);
        } else {
            this->blockNumNs_ = nsPerBlock;
            this->blockStartN_ = remainderNs * (nsPerBlock + 1) + (blockIdx - remainderNs) * nsPerBlock;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x, (uint64_t)N * C * spatial);
        multGm.SetGlobalBuffer((__gm__ float*)mult, C);
        zGm.SetGlobalBuffer((__gm__ float*)z, (uint64_t)N * spatial);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, spatial * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, spatial * sizeof(float));
        pipe.InitBuffer(tempBuf, spatial * sizeof(float));
        pipe.InitBuffer(multBuf, 512);
        pipe.InitBuffer(reduceBuf, 8 * 1024);
        pipe.InitBuffer(scalarBuf, 64);
    }

    __aicore__ inline void Process()
    {
        if (blockNumNs_ == 0) return;

        auto multLocal = multBuf.Get<float>();
        AscendC::DataCopyExtParams mParams;
        mParams.blockCount = 1;
        mParams.blockLen = (uint32_t)(C_ * sizeof(float));
        mParams.srcStride = 0;
        mParams.dstStride = 0;
        AscendC::DataCopyPadExtParams<float> mPadParams;
        mPadParams.isPad = false;
        mPadParams.leftPadding = 0;
        mPadParams.rightPadding = 0;
        mPadParams.paddingValue = 0.0f;
        AscendC::DataCopyPad(multLocal, multGm, mParams, mPadParams);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t localN = 0; localN < blockNumNs_; localN++) {
            uint32_t n = blockStartN_ + localN;
            ProcessOneN(n, multLocal);
        }
    }

private:
    __aicore__ inline void ProcessOneN(uint32_t n, AscendC::LocalTensor<float>& multLocal)
    {
        auto resLocal = outQueueZ.AllocTensor<float>();
        AscendC::Duplicate<float>(resLocal, NEG_INF_F, spatial_);

        auto tempLocal = tempBuf.Get<float>();
        auto reduceTmp = reduceBuf.Get<float>();
        auto scalarLocal = scalarBuf.Get<float>();

        for (uint32_t c = 0; c < C_; c++) {
            auto xLocal = inQueueX.AllocTensor<float>();
            uint64_t xOffset = (uint64_t)n * C_ * spatial_ + (uint64_t)c * spatial_;

            AscendC::DataCopyExtParams params;
            params.blockCount = 1;
            params.blockLen = (uint32_t)(spatial_ * sizeof(float));
            params.srcStride = 0;
            params.dstStride = 0;
            AscendC::DataCopyPadExtParams<float> padParams;
            padParams.isPad = false;
            padParams.leftPadding = 0;
            padParams.rightPadding = 0;
            padParams.paddingValue = 0.0f;

            AscendC::DataCopyPad(xLocal, xGm[xOffset], params, padParams);
            inQueueX.EnQue(xLocal);

            auto xIn = inQueueX.DeQue<float>();
            float multVal = multLocal.GetValue(c);

            AscendC::Muls<float>(xIn, xIn, multVal, spatial_);

            AscendC::ReduceSum<float, true>(scalarLocal, xIn, reduceTmp, (int32_t)spatial_);
            float sumVal = scalarLocal.GetValue(0);
            float meanVal = sumVal * invSpatial_;

            AscendC::Adds<float>(xIn, xIn, -meanVal, spatial_);

            AscendC::Mul<float>(tempLocal, xIn, xIn, spatial_);

            AscendC::ReduceSum<float, true>(scalarLocal, tempLocal, reduceTmp, (int32_t)spatial_);
            float varVal = scalarLocal.GetValue(0) * invSpatial_;

            AscendC::Duplicate<float>(scalarLocal, varVal + EPS, 8);
            AscendC::Sqrt<float>(scalarLocal, scalarLocal, 8);
            float stdVal = scalarLocal.GetValue(0);
            float invstd = 1.0f / stdVal;

            AscendC::Muls<float>(xIn, xIn, invstd, spatial_);

            AscendC::Duplicate<float>(tempLocal, clampMax_, spatial_);
            AscendC::Min<float>(xIn, xIn, tempLocal, spatial_);
            AscendC::Duplicate<float>(tempLocal, clampMin_, spatial_);
            AscendC::Max<float>(xIn, xIn, tempLocal, spatial_);

            AscendC::Muls<float>(xIn, xIn, multVal, spatial_);

            AscendC::Max<float>(resLocal, resLocal, xIn, spatial_);

            inQueueX.FreeTensor(xIn);
        }

        outQueueZ.EnQue(resLocal);
        auto resOut = outQueueZ.DeQue<float>();

        uint64_t zOffset = (uint64_t)n * spatial_;
        AscendC::DataCopyExtParams outParams;
        outParams.blockCount = 1;
        outParams.blockLen = (uint32_t)(spatial_ * sizeof(float));
        outParams.srcStride = 0;
        outParams.dstStride = 0;
        AscendC::DataCopyPad(zGm[zOffset], resOut, outParams);
        outQueueZ.FreeTensor(resOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tempBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> multBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> multGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t N_;
    uint32_t C_;
    uint32_t spatial_;
    uint32_t blockStartN_;
    uint32_t blockNumNs_;
    float clampMin_;
    float clampMax_;
    float invSpatial_;
};

extern "C" __global__ __aicore__ void conv3d_multiply_instance_norm_clamp_multiply_max_custom(
    GM_ADDR x, GM_ADDR mult, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dMimcmm op;
    op.Init(x, mult, z, tiling_data.N, tiling_data.C, tiling_data.spatial,
            tiling_data.nsPerBlock, tiling_data.remainderNs, tiling_data.usedBlocks,
            tiling_data.clampMin, tiling_data.clampMax);
    op.Process();
}
