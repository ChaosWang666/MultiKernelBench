kernel_src="""
#include "kernel_operator.h"

constexpr uint32_t K_TILE = 4096;
constexpr uint32_t O_TILE = 32;

class KernelMatmulSwishScaling {
public:
    __aicore__ inline KernelMatmulSwishScaling() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                uint32_t batch, uint32_t inF, uint32_t outF, float sf)
    {
        batchSize = batch;
        inFeatures = inF;
        outFeatures = outF;
        scalingFactor = sf;

        uint32_t coreId = AscendC::GetBlockIdx();
        uint32_t numCores = AscendC::GetBlockNum();

        uint32_t bPerCore = (batch + numCores - 1) / numCores;
        bStart = coreId * bPerCore;
        uint32_t bEndRaw = bStart + bPerCore;
        bEnd = (bEndRaw > batch) ? batch : bEndRaw;
        if (bStart > batch) bStart = batch;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        wGm.SetGlobalBuffer((__gm__ float*)weight);
        biasGm.SetGlobalBuffer((__gm__ float*)bias);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        pipe.InitBuffer(xBuf, K_TILE * sizeof(float));
        pipe.InitBuffer(wBuf, K_TILE * sizeof(float));
        pipe.InitBuffer(biasBuf, O_TILE * sizeof(float));
        pipe.InitBuffer(accumBuf, O_TILE * sizeof(float));
        pipe.InitBuffer(tmpBuf, K_TILE * sizeof(float));
        pipe.InitBuffer(reduceBuf, K_TILE * sizeof(float));
        pipe.InitBuffer(sigBuf, O_TILE * sizeof(float));
        pipe.InitBuffer(scalarBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (bStart >= bEnd) return;

        uint32_t kTiles = (inFeatures + K_TILE - 1) / K_TILE;
        uint32_t oTiles = (outFeatures + O_TILE - 1) / O_TILE;

        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> accumLocal = accumBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> sigLocal = sigBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();

        AscendC::DataCopyPadExtParams<float> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0.0f;

        for (uint32_t b = bStart; b < bEnd; b++) {
            for (uint32_t ot = 0; ot < oTiles; ot++) {
                uint32_t oStart = ot * O_TILE;
                uint32_t oCount = outFeatures - oStart;
                if (oCount > O_TILE) oCount = O_TILE;

                AscendC::DataCopyExtParams biasCp;
                biasCp.blockCount = 1;
                biasCp.blockLen = oCount * sizeof(float);
                biasCp.srcStride = 0;
                biasCp.dstStride = 0;
                biasCp.rsv = 0;
                AscendC::DataCopyPad(biasLocal, biasGm[oStart], biasCp, padParams);
                AscendC::PipeBarrier<PIPE_ALL>();

                for (uint32_t oi = 0; oi < oCount; oi++) {
                    uint32_t o = oStart + oi;
                    float dotSum = biasLocal.GetValue(oi);

                    for (uint32_t kt = 0; kt < kTiles; kt++) {
                        uint32_t kStart = kt * K_TILE;
                        uint32_t kCount = inFeatures - kStart;
                        if (kCount > K_TILE) kCount = K_TILE;

                        AscendC::DataCopyExtParams xCp;
                        xCp.blockCount = 1;
                        xCp.blockLen = kCount * sizeof(float);
                        xCp.srcStride = 0;
                        xCp.dstStride = 0;
                        xCp.rsv = 0;
                        AscendC::DataCopyPad(xLocal, xGm[b * inFeatures + kStart], xCp, padParams);

                        AscendC::DataCopyExtParams wCp;
                        wCp.blockCount = 1;
                        wCp.blockLen = kCount * sizeof(float);
                        wCp.srcStride = 0;
                        wCp.dstStride = 0;
                        wCp.rsv = 0;
                        AscendC::DataCopyPad(wLocal, wGm[o * inFeatures + kStart], wCp, padParams);
                        AscendC::PipeBarrier<PIPE_ALL>();

                        AscendC::Mul(tmpLocal, xLocal, wLocal, kCount);
                        AscendC::PipeBarrier<PIPE_ALL>();

                        AscendC::ReduceSum<float>(scalarLocal, tmpLocal, reduceLocal, static_cast<int32_t>(kCount));
                        AscendC::PipeBarrier<PIPE_ALL>();

                        dotSum += scalarLocal.GetValue(0);
                    }

                    accumLocal.SetValue(oi, dotSum);
                }

                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::Sigmoid(sigLocal, accumLocal, oCount);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::Mul(accumLocal, accumLocal, sigLocal, oCount);
                AscendC::PipeBarrier<PIPE_ALL>();
                AscendC::Muls(accumLocal, accumLocal, scalingFactor, oCount);
                AscendC::PipeBarrier<PIPE_ALL>();

                AscendC::DataCopyExtParams yCp;
                yCp.blockCount = 1;
                yCp.blockLen = oCount * sizeof(float);
                yCp.srcStride = 0;
                yCp.dstStride = 0;
                yCp.rsv = 0;
                AscendC::DataCopyPad(yGm[b * outFeatures + oStart], accumLocal, yCp);
                AscendC::PipeBarrier<PIPE_ALL>();
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, wBuf, biasBuf, accumBuf, tmpBuf, reduceBuf, sigBuf, scalarBuf;
    AscendC::GlobalTensor<float> xGm, wGm, biasGm, yGm;
    uint32_t batchSize, inFeatures, outFeatures;
    uint32_t bStart, bEnd;
    float scalingFactor;
};

extern "C" __global__ __aicore__ void matmul_swish_scaling_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSwishScaling op;
    op.Init(x, weight, bias, y,
            tiling_data.batchSize, tiling_data.inFeatures, tiling_data.outFeatures,
            tiling_data.scalingFactor);
    op.Process();
}
