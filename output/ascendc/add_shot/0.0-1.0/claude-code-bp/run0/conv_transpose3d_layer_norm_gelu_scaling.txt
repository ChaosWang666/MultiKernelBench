Looking at the model, I need to:
1. Keep ConvTranspose3d in PyTorch (complex op, hard to beat cuDNN/Ascend's impl)
2. Fuse LayerNorm + GELU + Scaling into a custom AscendC kernel

The LayerNorm normalizes over the last dim (W'=64 after transpose conv). Each row of 64 FP32 elements (256 bytes) is naturally 32-byte aligned.

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr float EPS = 1e-5f;
constexpr float SCALING_FACTOR = 1.0f;

class KernelLayerNormGeluScale {
public:
    __aicore__ inline KernelLayerNormGeluScale() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                 uint32_t totalRows, uint32_t rowLength, uint32_t tileRows)
    {
        this->totalRows = totalRows;
        this->rowLength = rowLength;
        this->tileRows = tileRows;
        this->invRowLen = 1.0f / (float)rowLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t rowsPerBlock = (totalRows + blockNum - 1) / blockNum;
        uint32_t startRow = blockIdx * rowsPerBlock;
        uint32_t endRow = startRow + rowsPerBlock;
        if (endRow > totalRows) endRow = totalRows;

        this->rowsThisCore = (endRow > startRow) ? (endRow - startRow) : 0;

        xGm.SetGlobalBuffer((__gm__ float *)x + startRow * rowLength, this->rowsThisCore * rowLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + startRow * rowLength, this->rowsThisCore * rowLength);
        gammaGm.SetGlobalBuffer((__gm__ float *)gamma, rowLength);
        betaGm.SetGlobalBuffer((__gm__ float *)beta, rowLength);

        uint32_t tileBytes = tileRows * rowLength * sizeof(float);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileBytes);
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileBytes);
        pipe.InitBuffer(gammaBuf, rowLength * sizeof(float));
        pipe.InitBuffer(betaBuf, rowLength * sizeof(float));
        pipe.InitBuffer(tmpRowBuf, rowLength * sizeof(float));
        pipe.InitBuffer(scalarBuf, 64);
        pipe.InitBuffer(reduceTmpBuf, 2048);
    }

    __aicore__ inline void Process()
    {
        if (rowsThisCore == 0) return;

        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();
        AscendC::DataCopy(gammaLocal, gammaGm, rowLength);
        AscendC::DataCopy(betaLocal, betaGm, rowLength);

        auto evt = pipe.FetchEventID(AscendC::HardEvent::MTE2_V);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(evt);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(evt);

        uint32_t numTiles = (rowsThisCore + tileRows - 1) / tileRows;
        for (uint32_t i = 0; i < numTiles; i++) {
            uint32_t startLocalRow = i * tileRows;
            uint32_t actualRows = (startLocalRow + tileRows <= rowsThisCore) ?
                                   tileRows : (rowsThisCore - startLocalRow);
            CopyIn(startLocalRow, actualRows);
            Compute(actualRows, gammaLocal, betaLocal);
            CopyOut(startLocalRow, actualRows);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t startLocalRow, uint32_t actualRows)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[startLocalRow * rowLength], actualRows * rowLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t actualRows,
                                    AscendC::LocalTensor<float>& gammaLocal,
                                    AscendC::LocalTensor<float>& betaLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmpRow = tmpRowBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceTmpBuf.Get<float>();

        const float invSqrt2 = 0.70710678118654752440f;

        for (uint32_t r = 0; r < actualRows; r++) {
            uint32_t offset = r * rowLength;

            // 1. mean = sum(x) / N
            AscendC::ReduceSum<float, true>(scalarLocal, xLocal[offset], reduceTmp, (int32_t)rowLength);
            float meanVal = scalarLocal.GetValue(0) * invRowLen;

            // 2. tmpRow = x - mean
            AscendC::Adds<float>(tmpRow, xLocal[offset], -meanVal, rowLength);

            // 3. yLocal[offset] = (x - mean)^2 (temp use)
            AscendC::Mul<float>(yLocal[offset], tmpRow, tmpRow, rowLength);

            // 4. var = sum((x-mean)^2)/N + eps
            AscendC::ReduceSum<float, true>(scalarLocal, yLocal[offset], reduceTmp, (int32_t)rowLength);
            float varVal = scalarLocal.GetValue(0) * invRowLen + EPS;

            // 5. std via tensor Sqrt on a 16-element slab
            AscendC::Duplicate<float>(scalarLocal, varVal, 16);
            AscendC::Sqrt<float>(scalarLocal, scalarLocal, 16);
            float stdVal = scalarLocal.GetValue(0);
            float invStd = 1.0f / stdVal;

            // 6. normalize: tmpRow = (x-mean) * invStd
            AscendC::Muls<float>(tmpRow, tmpRow, invStd, rowLength);

            // 7. affine: yLocal = tmpRow * gamma + beta
            AscendC::Mul<float>(yLocal[offset], tmpRow, gammaLocal, rowLength);
            AscendC::Add<float>(yLocal[offset], yLocal[offset], betaLocal, rowLength);

            // 8. GELU (exact): 0.5 * x * (1 + erf(x / sqrt(2)))
            AscendC::Muls<float>(tmpRow, yLocal[offset], invSqrt2, rowLength);
            AscendC::Erf<float>(tmpRow, tmpRow, rowLength);
            AscendC::Adds<float>(tmpRow, tmpRow, 1.0f, rowLength);
            AscendC::Muls<float>(tmpRow, tmpRow, 0.5f, rowLength);
            AscendC::Mul<float>(yLocal[offset], yLocal[offset], tmpRow, rowLength);

            // 9. scaling
            if (SCALING_FACTOR != 1.0f) {
                AscendC::Muls<float>(yLocal[offset], yLocal[offset], SCALING_FACTOR, rowLength);
            }
        }

        outQueueY.EnQue(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t startLocalRow, uint32_t actualRows)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[startLocalRow * rowLength], yLocal, actualRows * rowLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaBuf, betaBuf, tmpRowBuf, scalarBuf, reduceTmpBuf;
    AscendC::GlobalTensor<float> xGm, yGm, gammaGm, betaGm;
    uint32_t totalRows;
    uint32_t rowLength;
    uint32_t tileRows;
    uint32_t rowsThisCore;
    float invRowLen;
};

extern "C" __global__ __aicore__ void conv_transpose3d_layer_norm_gelu_scaling_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelLayerNormGeluScale op;
    op.Init(x, gamma, beta, y, tiling_data.totalRows, tiling_data.rowLength, tiling_data.tileRows);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, bias=True, eps=1e-5, scaling_factor=1.0):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, bias=bias)
        self.layer_norm = nn.LayerNorm(out_channels, eps=eps)
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = self.conv_transpose(x)
        x = custom_ops_lib.conv_transpose3d_layer_norm_gelu_scaling_custom(
            x.contiguous(), self.layer_norm.weight, self.layer_norm.bias
        )
        return x
