kernel_src = """
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSubHardSwish {
public:
    __aicore__ inline KernelSubHardSwish() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t tileLen, float subtractValue)
    {
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t baseLen = totalLength / blockNum;
        uint32_t remainder = totalLength % blockNum;

        uint32_t myLen;
        uint32_t myOffset;
        if (blockIdx < remainder) {
            myLen = baseLen + 1;
            myOffset = blockIdx * myLen;
        } else {
            myLen = baseLen;
            myOffset = remainder * (baseLen + 1) + (blockIdx - remainder) * baseLen;
        }

        this->blockLength = myLen;
        this->tileLength = tileLen;
        this->subtractValue = subtractValue;
        this->loopCount = (myLen + tileLen - 1) / tileLen;

        xGm.SetGlobalBuffer((__gm__ float*)x + myOffset, myLen);
        yGm.SetGlobalBuffer((__gm__ float*)y + myOffset, myLen);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLen * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLen * sizeof(float));
        pipe.InitBuffer(tmp1Buf, tileLen * sizeof(float));
        pipe.InitBuffer(tmp2Buf, tileLen * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->loopCount; i++) {
            uint32_t curLen;
            if (i == this->loopCount - 1) {
                curLen = this->blockLength - i * this->tileLength;
            } else {
                curLen = this->tileLength;
            }
            CopyIn(i, curLen);
            Compute(curLen);
            CopyOut(i, curLen);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress, uint32_t curLen)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopyExtParams copyParams{1, (uint32_t)(curLen * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPadExtParams<float> padParams{false, 0, 0, 0.0f};
        AscendC::DataCopyPad(xLocal, xGm[progress * this->tileLength], copyParams, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t curLen)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp1Local = tmp1Buf.Get<float>();
        AscendC::LocalTensor<float> tmp2Local = tmp2Buf.Get<float>();

        // x = x - subtractValue
        AscendC::Adds<float>(xLocal, xLocal, -this->subtractValue, curLen);

        // HardSwish: y = x * relu6(x+3) / 6
        // relu6(x+3) = relu(x+3) - relu(x-3)

        // tmp1 = relu(x+3)
        AscendC::Adds<float>(tmp1Local, xLocal, 3.0f, curLen);
        AscendC::Relu<float>(tmp1Local, tmp1Local, curLen);

        // tmp2 = relu(x-3)
        AscendC::Adds<float>(tmp2Local, xLocal, -3.0f, curLen);
        AscendC::Relu<float>(tmp2Local, tmp2Local, curLen);

        // tmp1 = (tmp1 - tmp2) / 6
        AscendC::Sub<float>(tmp1Local, tmp1Local, tmp2Local, curLen);
        AscendC::Muls<float>(tmp1Local, tmp1Local, 1.0f / 6.0f, curLen);

        // y = x * tmp1
        AscendC::Mul<float>(yLocal, xLocal, tmp1Local, curLen);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress, uint32_t curLen)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopyExtParams copyParams{1, (uint32_t)(curLen * sizeof(float)), 0, 0, 0};
        AscendC::DataCopyPad(yGm[progress * this->tileLength], yLocal, copyParams);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmp1Buf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmp2Buf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t blockLength;
    uint32_t tileLength;
    uint32_t loopCount;
    float subtractValue;
};

extern "C" __global__ __aicore__ void conv2d_subtract_hard_swish_max_pool_mish_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSubHardSwish op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.tileLength, tiling_data.subtractValue);
    op.Process();
}

model_src = '''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, subtract_value, pool_kernel_size):
        super().__init__()
        self.conv = torch.nn.Conv2d(in_channels, out_channels, kernel_size)
        self.subtract_value = float(subtract_value)
        self.pool = torch.nn.MaxPool2d(pool_kernel_size)

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
        x = custom_ops_lib.conv2d_subtract_hard_swish_max_pool_mish_custom(x, self.subtract_value)
        x = self.pool(x)
        x = torch.nn.functional.mish(x)
        return x
