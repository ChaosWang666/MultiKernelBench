kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulSwishSumGroupNorm {
public:
    __aicore__ inline KernelMatmulSwishSumGroupNorm() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t outFeatures,
                                uint32_t rowsPerBlock)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerBlock;
        uint32_t endRow = startRow + rowsPerBlock;
        if (endRow > batchSize) endRow = batchSize;
        this->actualRows = (startRow >= batchSize) ? 0 : (endRow - startRow);
        this->outFeatures = outFeatures;

        if (this->actualRows == 0) {
            return;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x + startRow * outFeatures, this->actualRows * outFeatures);
        zGm.SetGlobalBuffer((__gm__ float *)z + startRow * outFeatures, this->actualRows * outFeatures);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, outFeatures);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(biasBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf, outFeatures * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (actualRows == 0) {
            return;
        }

        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::DataCopy(biasLocal, biasGm, outFeatures);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        for (uint32_t i = 0; i < actualRows; i++) {
            CopyIn(i);
            Compute(i, biasLocal, tmpLocal);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[progress * outFeatures], outFeatures);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t progress,
                                   AscendC::LocalTensor<float>& biasLocal,
                                   AscendC::LocalTensor<float>& tmpLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        // swish(x) = x * sigmoid(x) = x / (1 + exp(-x))
        AscendC::Muls<float>(tmpLocal, xLocal, (float)-1.0f, outFeatures);
        AscendC::Exp<float>(tmpLocal, tmpLocal, outFeatures);
        AscendC::Adds<float>(tmpLocal, tmpLocal, (float)1.0f, outFeatures);
        AscendC::Div<float>(zLocal, xLocal, tmpLocal, outFeatures);

        // Add bias (broadcast across rows handled by per-row call)
        AscendC::Add<float>(zLocal, zLocal, biasLocal, outFeatures);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[progress * outFeatures], zLocal, outFeatures);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    AscendC::GlobalTensor<float> biasGm;
    uint32_t actualRows;
    uint32_t outFeatures;
};

extern "C" __global__ __aicore__ void matmul_swish_sum_group_norm_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSwishSumGroupNorm op;
    op.Init(x, bias, z, tiling_data.batchSize, tiling_data.outFeatures,
            tiling_data.rowsPerBlock);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, bias_shape):
        super(ModelNew, self).__init__()
        self.matmul = nn.Linear(in_features, out_features)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.group_norm = nn.GroupNorm(num_groups, out_features)

    def forward(self, x):
        x = self.matmul(x)
        x = custom_ops_lib.matmul_swish_sum_group_norm_custom(x.contiguous(), self.bias.contiguous())
        x = self.group_norm(x)
        return x
