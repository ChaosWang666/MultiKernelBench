                                uint32_t totalSlices, uint32_t sliceSize,
                                uint32_t numChannels)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->slicesPerBlock = totalSlices / blockNum;
        this->startSlice = blockIdx * this->slicesPerBlock;
        this->sliceSize = sliceSize;
        this->numChannels = numChannels;
        this->numFullTilesPerSlice = sliceSize / TILE_ELEMS;
        this->tailElemsPerSlice = sliceSize - this->numFullTilesPerSlice * TILE_ELEMS;
        this->tilesPerSlice = this->numFullTilesPerSlice + (this->tailElemsPerSlice > 0 ? 1 : 0);

        uint64_t totalElems = (uint64_t)totalSlices * (uint64_t)sliceSize;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, totalElems);
        multGm.SetGlobalBuffer((__gm__ DTYPE_MULTIPLIER *)multiplier, numChannels);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z, totalElems);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, TILE_ELEMS * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, TILE_ELEMS * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpBuf, TILE_ELEMS * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t s = 0; s < slicesPerBlock; s++) {
            uint32_t sliceIdx = startSlice + s;
            uint32_t channelIdx = sliceIdx % numChannels;
            DTYPE_MULTIPLIER multScalar = multGm.GetValue(channelIdx);
            uint64_t sliceOffset = (uint64_t)sliceIdx * (uint64_t)sliceSize;

            for (uint32_t t = 0; t < tilesPerSlice; t++) {
                uint32_t len = (t < numFullTilesPerSlice) ? TILE_ELEMS : tailElemsPerSlice;
                uint64_t offset = sliceOffset + (uint64_t)t * (uint64_t)TILE_ELEMS;
                CopyIn(offset, len);
                Compute(len, multScalar);
                CopyOut(offset, len);
            }
        }
    }

private:

    {

        AscendC::DataCopyExtParams copyInParams{1, len * (uint32_t)sizeof(DTYPE_X), 0, 0, 0};
        AscendC::DataCopyPadExtParams<DTYPE_X> padParams{false, 0, 0, 0};
        AscendC::DataCopyPad(xLocal, xGm[gmOffset], copyInParams, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t len, DTYPE_MULTIPLIER multScalar)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_X> tmpLocal = tmpBuf.Get<DTYPE_X>();

        AscendC::Adds(tmpLocal, tmpLocal, (DTYPE_X)1.0f, len);
        AscendC::Mul(zLocal, zLocal, tmpLocal, len);
        AscendC::Muls(zLocal, zLocal, (DTYPE_X)0.5f, len);

        inQueueX.FreeTensor(xLocal);
    }
