kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv2dAvgPoolSigmoidSum {
public:
    __aicore__ inline KernelConv2dAvgPoolSigmoidSum() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t elemsPerBatch,
                                 uint32_t batchesPerCore,
                                 uint32_t tileLength,
                                 uint32_t tilesPerBatch)
    {
        this->elemsPerBatch = elemsPerBatch;
        this->batchesPerCore = batchesPerCore;
        this->tileLength = tileLength;
        this->tilesPerBatch = tilesPerBatch;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t batchOffset = (uint64_t)blockIdx * (uint64_t)batchesPerCore;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + batchOffset * (uint64_t)elemsPerBatch,
                            (uint64_t)batchesPerCore * (uint64_t)elemsPerBatch);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + batchOffset, batchesPerCore);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, 1, batchesPerCore * sizeof(DTYPE_Y));
        pipe.InitBuffer(accumBuf, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(sigBuf, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(workBuf, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(reduceBuf, 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_X> accumLocal = accumBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> sigLocal = sigBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> workLocal = workBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> reduceLocal = reduceBuf.Get<DTYPE_X>();

        for (uint32_t b = 0; b < this->batchesPerCore; b++) {
            AscendC::Duplicate(accumLocal, (DTYPE_X)0, this->tileLength);

            for (uint32_t t = 0; t < this->tilesPerBatch; t++) {
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                uint64_t offset = (uint64_t)b * (uint64_t)this->elemsPerBatch
                                  + (uint64_t)t * (uint64_t)this->tileLength;
                AscendC::DataCopy(xLocal, xGm[offset], this->tileLength);
                inQueueX.EnQue(xLocal);

                xLocal = inQueueX.DeQue<DTYPE_X>();
                AscendC::Sigmoid(sigLocal, xLocal, this->tileLength);
                AscendC::Add(accumLocal, accumLocal, sigLocal, this->tileLength);
                inQueueX.FreeTensor(xLocal);
            }

            AscendC::ReduceSum<DTYPE_X>(reduceLocal, accumLocal, workLocal, this->tileLength);
            DTYPE_X partial = reduceLocal.GetValue(0);
            yLocal.SetValue(b, partial);
        }

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        AscendC::LocalTensor<DTYPE_Y> yLocalOut = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm, yLocalOut, this->batchesPerCore);
        outQueueY.FreeTensor(yLocalOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sigBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t elemsPerBatch;
    uint32_t batchesPerCore;
    uint32_t tileLength;
    uint32_t tilesPerBatch;
};

extern "C" __global__ __aicore__ void conv2d_avg_pool_sigmoid_sum_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dAvgPoolSigmoidSum op;
    op.Init(x, y,
            tiling_data.elemsPerBatch,
            tiling_data.batchesPerCore,
            tiling_data.tileLength,
            tiling_data.tilesPerBatch);
    op.Process();
}

import torch

        super(ModelNew, self).__init__()
