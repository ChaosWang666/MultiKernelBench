kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;

constexpr int32_t BUFFER_NUM = 2;

template <typename aT, typename bT, typename cT, typename biasT>
class KernelMatmulDivideGelu {
public:
    __aicore__ inline KernelMatmulDivideGelu() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
                                GM_ADDR workspace, const TCubeTiling& tilingIn,
                                float divisorIn, uint32_t M_, uint32_t N_, uint32_t K_)
    {
        this->tiling = tilingIn;
        this->divisor = divisorIn;
        this->M = M_;
        this->N = N_;
        this->K = K_;

        xGm.SetGlobalBuffer((__gm__ aT*)x, M * K);
        wGm.SetGlobalBuffer((__gm__ bT*)weight, N * K);
        biasGm.SetGlobalBuffer((__gm__ biasT*)bias, N);
        mmOutGm.SetGlobalBuffer((__gm__ cT*)workspace, M * N);
        zGm.SetGlobalBuffer((__gm__ cT*)z, M * N);
    }

    __aicore__ inline void Process(TPipe* pipe)
    {
        if ASCEND_IS_AIC {
            matmul::Matmul<matmul::MatmulType<TPosition::GM, CubeFormat::ND, aT>,
                           matmul::MatmulType<TPosition::GM, CubeFormat::ND, bT, true>,
                           matmul::MatmulType<TPosition::GM, CubeFormat::ND, cT>,
                           matmul::MatmulType<TPosition::GM, CubeFormat::ND, biasT>> mm;
            REGIST_MATMUL_OBJ(pipe, GetSysWorkSpacePtr(), mm, &tiling);
            mm.SetTensorA(xGm);
            mm.SetTensorB(wGm);
            mm.SetBias(biasGm);
            mm.IterateAll(mmOutGm);
            mm.End();
        }

        SyncAll();

        if ASCEND_IS_AIV {
            int32_t blockIdx = GetBlockIdx();
            uint32_t totalElements = M * N;
            uint32_t blockNum = GetBlockNum();
            uint32_t perCore = (totalElements + blockNum - 1) / blockNum;
            perCore = (perCore + 7) / 8 * 8;
            uint32_t startIdx = blockIdx * perCore;
            if (startIdx >= totalElements) return;
            uint32_t endIdx = startIdx + perCore;
            if (endIdx > totalElements) endIdx = totalElements;
            uint32_t curLen = endIdx - startIdx;

            constexpr uint32_t TILE_SIZE = 8192;
            uint32_t tileNum = (curLen + TILE_SIZE - 1) / TILE_SIZE;

            TQue<QuePosition::VECIN, BUFFER_NUM> inQ;
            TQue<QuePosition::VECOUT, BUFFER_NUM> outQ;
            pipe->InitBuffer(inQ, BUFFER_NUM, TILE_SIZE * sizeof(cT));
            pipe->InitBuffer(outQ, BUFFER_NUM, TILE_SIZE * sizeof(cT));

            float invDivisor = 1.0f / divisor;
            float geluCoef = 0.7978845608028654f;
            float geluConst = 0.044715f;

            for (uint32_t t = 0; t < tileNum; t++) {
                uint32_t offset = startIdx + t * TILE_SIZE;
                uint32_t len = (t == tileNum - 1) ? (curLen - t * TILE_SIZE) : TILE_SIZE;
                uint32_t alignedLen = (len + 7) / 8 * 8;

                LocalTensor<cT> inLocal = inQ.AllocTensor<cT>();
                DataCopy(inLocal, mmOutGm[offset], alignedLen);
                inQ.EnQue(inLocal);

                LocalTensor<cT> xLocal = inQ.DeQue<cT>();
                LocalTensor<cT> yLocal = outQ.AllocTensor<cT>();

                Muls(xLocal, xLocal, (cT)invDivisor, len);
                PipeBarrier<PIPE_V>();

                Mul(yLocal, xLocal, xLocal, len);
                PipeBarrier<PIPE_V>();
                Mul(yLocal, yLocal, xLocal, len);
                PipeBarrier<PIPE_V>();
                Muls(yLocal, yLocal, (cT)geluConst, len);
                PipeBarrier<PIPE_V>();
                Add(yLocal, yLocal, xLocal, len);
                PipeBarrier<PIPE_V>();
                Muls(yLocal, yLocal, (cT)geluCoef, len);
                PipeBarrier<PIPE_V>();

                Tanh(yLocal, yLocal, len);
                PipeBarrier<PIPE_V>();

                Adds(yLocal, yLocal, (cT)1.0f, len);
                PipeBarrier<PIPE_V>();
                Mul(yLocal, yLocal, xLocal, len);
                PipeBarrier<PIPE_V>();
                Muls(yLocal, yLocal, (cT)0.5f, len);
                PipeBarrier<PIPE_V>();

                outQ.EnQue<cT>(yLocal);
                inQ.FreeTensor(xLocal);

                LocalTensor<cT> zLocal = outQ.DeQue<cT>();
                DataCopy(zGm[offset], zLocal, alignedLen);
                outQ.FreeTensor(zLocal);
            }
        }
    }

private:
    GlobalTensor<aT> xGm;
    GlobalTensor<bT> wGm;
    GlobalTensor<biasT> biasGm;
    GlobalTensor<cT> mmOutGm;
    GlobalTensor<cT> zGm;
    TCubeTiling tiling;
    float divisor;
    uint32_t M, N, K;
};

extern "C" __global__ __aicore__ void matmul_divide_gelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    TPipe pipe;
    KernelMatmulDivideGelu<DTYPE_X, DTYPE_WEIGHT, DTYPE_Z, DTYPE_BIAS> op;
    op.Init(x, weight, bias, z, workspace, tiling_data.cubeTiling,
            tiling_data.divisor, tiling_data.M, tiling_data.N, tiling_data.K);
    op.Process(&pipe);
}
