kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;

constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t ACT_TILE_LEN = 8192;

class KernelMatmulAddSwishTanhGeluHardtanh {
public:
    __aicore__ inline KernelMatmulAddSwishTanhGeluHardtanh() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                GM_ADDR workspace, const TCubeTiling& tilingIn,
                                uint32_t M_, uint32_t N_, uint32_t K_)
    {
        this->tiling = tilingIn;
        this->M = M_;
        this->N = N_;
        this->K = K_;

        xGm.SetGlobalBuffer((__gm__ float*)x, M * K);
        wGm.SetGlobalBuffer((__gm__ float*)weight, N * K);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, N);
        yGm.SetGlobalBuffer((__gm__ float*)y, M * N);
    }

    __aicore__ inline void Process(TPipe* pipe)
    {
        if ASCEND_IS_AIC {
            matmul::Matmul<matmul::MatmulType<TPosition::GM, CubeFormat::ND, float>,
                           matmul::MatmulType<TPosition::GM, CubeFormat::ND, float, true>,
                           matmul::MatmulType<TPosition::GM, CubeFormat::ND, float>,
                           matmul::MatmulType<TPosition::GM, CubeFormat::ND, float>> mm;
            REGIST_MATMUL_OBJ(pipe, GetSysWorkSpacePtr(), mm, &tiling);
            mm.SetTensorA(xGm);
            mm.SetTensorB(wGm);
            mm.SetBias(biasGm);
            mm.IterateAll(yGm);
            mm.End();
        }

        SyncAll();

        if ASCEND_IS_AIV {
            uint32_t blockIdx = GetBlockIdx();
            uint32_t numBlocks = GetBlockNum();
            uint32_t totalSize = M * N;

            uint32_t totalTiles = (totalSize + ACT_TILE_LEN - 1) / ACT_TILE_LEN;
            uint32_t tilesPerCore = totalTiles / numBlocks;
            uint32_t extraTiles = totalTiles % numBlocks;

            uint32_t startTile, myTiles;
            if (blockIdx < extraTiles) {
                startTile = blockIdx * (tilesPerCore + 1);
                myTiles = tilesPerCore + 1;
            } else {
                startTile = extraTiles * (tilesPerCore + 1) + (blockIdx - extraTiles) * tilesPerCore;
                myTiles = tilesPerCore;
            }

            if (myTiles == 0) return;

            TQue<TPosition::VECIN, BUFFER_NUM> inQ;
            TQue<TPosition::VECOUT, BUFFER_NUM> outQ;
            pipe->InitBuffer(inQ, BUFFER_NUM, ACT_TILE_LEN * sizeof(float));
            pipe->InitBuffer(outQ, BUFFER_NUM, ACT_TILE_LEN * sizeof(float));

            for (uint32_t t = 0; t < myTiles; t++) {
                uint32_t offset = (startTile + t) * ACT_TILE_LEN;
                uint32_t len = ACT_TILE_LEN;
                if (offset + len > totalSize) {
                    len = totalSize - offset;
                }
                uint32_t alignedLen = (len + 7) / 8 * 8;

                LocalTensor<float> xLocal = inQ.AllocTensor<float>();
                DataCopy(xLocal, yGm[offset], alignedLen);
                inQ.EnQue(xLocal);

                LocalTensor<float> xIn = inQ.DeQue<float>();
                LocalTensor<float> yLocal = outQ.AllocTensor<float>();

                Sigmoid(yLocal, xIn, len);
                PipeBarrier<PIPE_V>();
                Mul(yLocal, xIn, yLocal, len);
                PipeBarrier<PIPE_V>();

                Tanh(yLocal, yLocal, len);
                PipeBarrier<PIPE_V>();

                Gelu(yLocal, yLocal, len);
                PipeBarrier<PIPE_V>();

                Maxs(yLocal, yLocal, (float)-1.0f, len);
                PipeBarrier<PIPE_V>();
                Mins(yLocal, yLocal, (float)1.0f, len);

                outQ.EnQue<float>(yLocal);
                inQ.FreeTensor(xIn);

                LocalTensor<float> yOut = outQ.DeQue<float>();
                DataCopy(yGm[offset], yOut, alignedLen);
                outQ.FreeTensor(yOut);
            }
        }
    }

private:
    GlobalTensor<float> xGm;
    GlobalTensor<float> wGm;
    GlobalTensor<float> biasGm;
    GlobalTensor<float> yGm;
    TCubeTiling tiling;
    uint32_t M, N, K;
};

extern "C" __global__ __aicore__ void matmul_add_swish_tanh_gelu_hardtanh_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    TPipe pipe;
    KernelMatmulAddSwishTanhGeluHardtanh op;
    op.Init(x, weight, bias, y, workspace, tiling_data.cubeTiling,
            tiling_data.M, tiling_data.N, tiling_data.K);
    op.Process(&pipe);
}

import torch
import torch.nn as nn

import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, add_value_shape):
        super(ModelNew, self).__init__()
