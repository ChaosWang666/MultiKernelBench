kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelGemmBNScaleSoftmax {
public:
    __aicore__ inline KernelGemmBNScaleSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t rowLength, uint32_t rowsPerCore)
    {
        this->rowLength = rowLength;
        this->rowsPerCore = rowsPerCore;

        uint64_t perCore = static_cast<uint64_t>(rowsPerCore) * static_cast<uint64_t>(rowLength);
        uint64_t offset = static_cast<uint64_t>(AscendC::GetBlockIdx()) * perCore;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + offset, perCore);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + offset, perCore);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, rowLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, rowLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(workBuf, rowLength * sizeof(float));
        pipe.InitBuffer(scalarBuf, 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t r = 0; r < this->rowsPerCore; r++) {
            CopyIn(r);
            Compute(r);
            CopyOut(r);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t r)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[r * this->rowLength], this->rowLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t r)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> scalar = scalarBuf.Get<float>();

        int32_t count = static_cast<int32_t>(this->rowLength);

        AscendC::ReduceMax<float>(scalar, xLocal, work, count, false);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        float maxVal = scalar.GetValue(0);

        AscendC::Adds(yLocal, xLocal, -maxVal, count);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Exp(yLocal, yLocal, count);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::ReduceSum<float>(scalar, yLocal, work, count);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID1);
        float sumVal = scalar.GetValue(0);
        float invSum = 1.0f / sumVal;

        AscendC::Muls(yLocal, yLocal, invSum, count);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t r)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm[r * this->rowLength], yLocal, this->rowLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t rowLength;
    uint32_t rowsPerCore;
};

extern "C" __global__ __aicore__ void gemm_batch_norm_scaling_softmax_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmBNScaleSoftmax op;
    op.Init(x, y, tiling_data.rowLength, tiling_data.rowsPerCore);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, bn_eps=1e-5, bn_momentum=0.1, scale_shape=(1,)):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.bn = nn.BatchNorm1d(out_features, eps=bn_eps, momentum=bn_momentum)
        self.scale = nn.Parameter(torch.ones(scale_shape))

    def forward(self, x):
        x = self.gemm(x)
        x = self.bn(x)
        x = self.scale * x
        x = x.contiguous()
        return custom_ops_lib.gemm_batch_norm_scaling_softmax_custom(x)
'''
