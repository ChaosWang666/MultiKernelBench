kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmBatchNormScalingSoftmax {
public:
    __aicore__ inline KernelGemmBatchNormScalingSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t dimSize)
    {
        this->batchSize = batchSize;
        this->dimSize = dimSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t rowsPerBlock = batchSize / blockNum;
        uint32_t remainder = batchSize % blockNum;

        if (blockIdx < remainder) {
            this->startRow = blockIdx * (rowsPerBlock + 1);
            this->endRow = this->startRow + rowsPerBlock + 1;
        } else {
            this->startRow = remainder * (rowsPerBlock + 1) + (blockIdx - remainder) * rowsPerBlock;
            this->endRow = this->startRow + rowsPerBlock;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * dimSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize * dimSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, dimSize * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, dimSize * sizeof(float));
        pipe.InitBuffer(reduceBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t row = this->startRow; row < this->endRow; row++) {
            CopyIn(row);
            Compute();
            CopyOut(row);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * this->dimSize], this->dimSize);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceBuf.Get<float>();

        AscendC::ReduceMax<float>(reduceTmp, xLocal, xLocal, this->dimSize, false);
        float maxVal = reduceTmp.GetValue(0);

        AscendC::Adds(yLocal, xLocal, -maxVal, this->dimSize);

        AscendC::Exp(yLocal, yLocal, this->dimSize);

        AscendC::ReduceSum<float>(reduceTmp, yLocal, yLocal, this->dimSize);
        float sumVal = reduceTmp.GetValue(0);

        float invSum = 1.0f / sumVal;
        AscendC::Muls(yLocal, yLocal, invSum, this->dimSize);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[row * this->dimSize], yLocal, this->dimSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t dimSize;
    uint32_t startRow;
    uint32_t endRow;
};

extern "C" __global__ __aicore__ void gemm_batch_norm_scaling_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmBatchNormScalingSoftmax op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.dimSize);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, bn_eps=1e-5, bn_momentum=0.1, scale_shape=(1,)):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.bn = nn.BatchNorm1d(out_features, eps=bn_eps, momentum=bn_momentum)
        self.scale = nn.Parameter(torch.ones(scale_shape))

    def forward(self, x):
        x = self.gemm(x)
        x = self.bn(x)
        x = self.scale * x
        x = x.contiguous()
        return custom_ops_lib.gemm_batch_norm_scaling_softmax_custom(x)
