kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv2dMinTanhTanh {
public:
    __aicore__ inline KernelConv2dMinTanhTanh() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t N_, uint32_t C_, uint32_t H_, uint32_t W_,
                                 uint32_t rowsPerCore_)
    {
        this->N = N_;
        this->C = C_;
        this->H = H_;
        this->W = W_;
        this->rowsPerCore = rowsPerCore_;

        uint32_t bytesPerRow = W_ * sizeof(DTYPE_X);
        uint32_t alignedBytes = (bytesPerRow + 31) / 32 * 32;
        this->alignedW = alignedBytes / sizeof(DTYPE_X);

        uint32_t coreIdx = AscendC::GetBlockIdx();
        this->startRow = coreIdx * rowsPerCore_;

        uint64_t totalX = (uint64_t)N_ * C_ * H_ * W_;
        uint64_t totalY = (uint64_t)N_ * H_ * W_;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X*)x, totalX);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y*)y, totalY);

        pipe.InitBuffer(inQueue, BUFFER_NUM, this->alignedW * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueue, 1, this->alignedW * sizeof(DTYPE_Y));
        pipe.InitBuffer(minBuf, this->alignedW * sizeof(DTYPE_X));
        pipe.InitBuffer(tmpBuf, this->alignedW * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        uint32_t totalRows = this->N * this->H;
        for (uint32_t r = 0; r < this->rowsPerCore; r++) {
            uint32_t globalRow = this->startRow + r;
            if (globalRow >= totalRows) {
                break;
            }
            uint32_t n = globalRow / this->H;
            uint32_t h = globalRow - n * this->H;
            ProcessRow(n, h);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t n, uint32_t h)
    {
        AscendC::LocalTensor<DTYPE_X> minT = minBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> tmpT = tmpBuf.Get<DTYPE_X>();

        uint64_t channelStride = (uint64_t)this->H * this->W;
        uint64_t rowOffsetIn = (uint64_t)n * this->C * channelStride + (uint64_t)h * this->W;
        uint64_t rowOffsetOut = (uint64_t)n * this->H * this->W + (uint64_t)h * this->W;

        AscendC::DataCopyExtParams inCopyParams;
        inCopyParams.blockCount = 1;
        inCopyParams.blockLen = this->W * sizeof(DTYPE_X);
        inCopyParams.srcStride = 0;
        inCopyParams.dstStride = 0;

        AscendC::DataCopyPadExtParams<DTYPE_X> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;

        {
            AscendC::LocalTensor<DTYPE_X> in0 = inQueue.AllocTensor<DTYPE_X>();
            AscendC::DataCopyPad(in0, xGm[rowOffsetIn], inCopyParams, padParams);
            inQueue.EnQue(in0);
            in0 = inQueue.DeQue<DTYPE_X>();
            AscendC::Adds(minT, in0, (DTYPE_X)0.0f, this->W);
            inQueue.FreeTensor(in0);
        }

        for (uint32_t c = 1; c < this->C; c++) {
            AscendC::LocalTensor<DTYPE_X> inC = inQueue.AllocTensor<DTYPE_X>();
            AscendC::DataCopyPad(inC, xGm[rowOffsetIn + (uint64_t)c * channelStride], inCopyParams, padParams);
            inQueue.EnQue(inC);
            inC = inQueue.DeQue<DTYPE_X>();
            AscendC::Min(minT, minT, inC, this->W);
            inQueue.FreeTensor(inC);
        }

        AscendC::Muls(tmpT, minT, (DTYPE_X)2.0f, this->W);
        AscendC::Exp(tmpT, tmpT, this->W);
        AscendC::Adds(minT, tmpT, (DTYPE_X)(-1.0f), this->W);
        AscendC::Adds(tmpT, tmpT, (DTYPE_X)1.0f, this->W);
        AscendC::Div(minT, minT, tmpT, this->W);

        AscendC::LocalTensor<DTYPE_Y> outLocal = outQueue.AllocTensor<DTYPE_Y>();
        AscendC::Muls(tmpT, minT, (DTYPE_X)2.0f, this->W);
        AscendC::Exp(tmpT, tmpT, this->W);
        AscendC::Adds(outLocal, tmpT, (DTYPE_X)(-1.0f), this->W);
        AscendC::Adds(tmpT, tmpT, (DTYPE_X)1.0f, this->W);
        AscendC::Div(outLocal, outLocal, tmpT, this->W);
        outQueue.EnQue<DTYPE_Y>(outLocal);

        outLocal = outQueue.DeQue<DTYPE_Y>();
        AscendC::DataCopyExtParams outCopyParams;
        outCopyParams.blockCount = 1;
        outCopyParams.blockLen = this->W * sizeof(DTYPE_Y);
        outCopyParams.srcStride = 0;
        outCopyParams.dstStride = 0;
        AscendC::DataCopyPad(yGm[rowOffsetOut], outLocal, outCopyParams);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> minBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;

    uint32_t N;
    uint32_t C;
    uint32_t H;
    uint32_t W;
    uint32_t alignedW;
    uint32_t rowsPerCore;
    uint32_t startRow;
};

extern "C" __global__ __aicore__ void conv2d_min_tanh_tanh_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dMinTanhTanh op;
    op.Init(x, y,
            tiling_data.N, tiling_data.C, tiling_data.H, tiling_data.W,
            tiling_data.rowsPerCore);
    op.Process();
}
"""
