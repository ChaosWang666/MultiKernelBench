kernel_src="""
#include "kernel_operator.h"

class KernelConv3dSoftmaxMaxPoolMaxPool {
public:
    static constexpr uint32_t C = 16;
    static constexpr uint32_t D = 14;
    static constexpr uint32_t H = 30;
    static constexpr uint32_t W = 30;
    static constexpr uint32_t OD = 3;
    static constexpr uint32_t OH = 7;
    static constexpr uint32_t OW = 7;
    static constexpr uint32_t PD = 4;
    static constexpr uint32_t PH = 4;
    static constexpr uint32_t PW = 4;
    static constexpr uint32_t TILE_SPATIAL = PD * PH * W;
    static constexpr uint32_t TILE_SIZE = C * TILE_SPATIAL;

    __aicore__ inline KernelConv3dSoftmaxMaxPoolMaxPool() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchesPerCore)
    {
        this->batchesPerCore = batchesPerCore;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t inBatchStride = (uint64_t)C * D * H * W;
        uint64_t outBatchStride = (uint64_t)C * OD * OH * OW;
        uint64_t inStart = (uint64_t)blockIdx * batchesPerCore * inBatchStride;
        uint64_t outStart = (uint64_t)blockIdx * batchesPerCore * outBatchStride;

        xGm.SetGlobalBuffer((__gm__ float*)x + inStart, (uint64_t)batchesPerCore * inBatchStride);
        yGm.SetGlobalBuffer((__gm__ float*)y + outStart, (uint64_t)batchesPerCore * outBatchStride);

        pipe.InitBuffer(tileBuf, TILE_SIZE * sizeof(float));
        pipe.InitBuffer(redBuf, TILE_SPATIAL * sizeof(float));
        pipe.InitBuffer(ptAllBuf, C * PH * W * sizeof(float));
        pipe.InitBuffer(outBuf, C * OW * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> tile = tileBuf.Get<float>();
        AscendC::LocalTensor<float> red = redBuf.Get<float>();
        AscendC::LocalTensor<float> pt = ptAllBuf.Get<float>();
        AscendC::LocalTensor<float> out = outBuf.Get<float>();

        for (uint32_t b = 0; b < batchesPerCore; b++) {
            for (uint32_t od = 0; od < OD; od++) {
                for (uint32_t oh = 0; oh < OH; oh++) {
                    // Phase 1: Load tile (C=16, PD=4, PH*W=120) into UB
                    for (uint32_t c = 0; c < C; c++) {
                        for (uint32_t dd = 0; dd < PD; dd++) {
                            uint64_t srcOffset = (uint64_t)b * C * D * H * W
                                               + (uint64_t)c * D * H * W
                                               + (uint64_t)(od * 4 + dd) * H * W
                                               + (uint64_t)(oh * 4) * W;
                            uint32_t dstOffset = c * TILE_SPATIAL + dd * PH * W;
                            AscendC::DataCopyExtParams copyParams{1, (uint32_t)(PH * W * sizeof(float)), 0, 0, 0};
                            AscendC::DataCopyPadExtParams<float> padParams{false, 0, 0, 0.0f};
                            AscendC::DataCopyPad(tile[dstOffset], xGm[srcOffset], copyParams, padParams);
                        }
                    }

                    AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
                    AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

                    // Phase 2a: Softmax along channel dim
                    AscendC::Max(red, tile[0], tile[TILE_SPATIAL], TILE_SPATIAL);
                    for (uint32_t c = 2; c < C; c++) {
                        AscendC::Max(red, red, tile[c * TILE_SPATIAL], TILE_SPATIAL);
                    }
                    for (uint32_t c = 0; c < C; c++) {
                        AscendC::Sub(tile[c * TILE_SPATIAL], tile[c * TILE_SPATIAL], red, TILE_SPATIAL);
                        AscendC::Exp(tile[c * TILE_SPATIAL], tile[c * TILE_SPATIAL], TILE_SPATIAL);
                    }
                    AscendC::Add(red, tile[0], tile[TILE_SPATIAL], TILE_SPATIAL);
                    for (uint32_t c = 2; c < C; c++) {
                        AscendC::Add(red, red, tile[c * TILE_SPATIAL], TILE_SPATIAL);
                    }
                    AscendC::Reciprocal(red, red, TILE_SPATIAL);
                    for (uint32_t c = 0; c < C; c++) {
                        AscendC::Mul(tile[c * TILE_SPATIAL], tile[c * TILE_SPATIAL], red, TILE_SPATIAL);
                    }

                    // Phase 2b: D-dim max pool (reduce 4 D-slices per channel)
                    for (uint32_t c = 0; c < C; c++) {
                        AscendC::Max(pt[c * PH * W],
                                     tile[c * TILE_SPATIAL + 0 * PH * W],
                                     tile[c * TILE_SPATIAL + 1 * PH * W],
                                     PH * W);
                        AscendC::Max(pt[c * PH * W],
                                     pt[c * PH * W],
                                     tile[c * TILE_SPATIAL + 2 * PH * W],
                                     PH * W);
                        AscendC::Max(pt[c * PH * W],
                                     pt[c * PH * W],
                                     tile[c * TILE_SPATIAL + 3 * PH * W],
                                     PH * W);
                    }

                    AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                    AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

                    // Phase 3: Scalar H,W max-pool (4x4 window over pt[c*120..c*120+120])
                    for (uint32_t c = 0; c < C; c++) {
                        for (uint32_t ow = 0; ow < OW; ow++) {
                            float maxVal = pt.GetValue(c * PH * W + 0 * W + ow * 4);
                            for (uint32_t hh = 0; hh < PH; hh++) {
                                for (uint32_t ww = 0; ww < PW; ww++) {
                                    uint32_t idx = c * PH * W + hh * W + ow * 4 + ww;
                                    float v = pt.GetValue(idx);
                                    maxVal = (v > maxVal) ? v : maxVal;
                                }
                            }
                            out.SetValue(c * OW + ow, maxVal);
                        }
                    }

                    AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
                    AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

                    // Phase 4: Store output strips
                    for (uint32_t c = 0; c < C; c++) {
                        uint64_t dstOffset = (uint64_t)b * C * OD * OH * OW
                                           + (uint64_t)c * OD * OH * OW
                                           + (uint64_t)od * OH * OW
                                           + (uint64_t)oh * OW;
                        AscendC::DataCopyExtParams outParams{1, (uint32_t)(OW * sizeof(float)), 0, 0, 0};
                        AscendC::DataCopyPad(yGm[dstOffset], out[c * OW], outParams);
                    }

                    AscendC::SetFlag<AscendC::HardEvent::MTE3_S>(EVENT_ID0);
                    AscendC::WaitFlag<AscendC::HardEvent::MTE3_S>(EVENT_ID0);
                    AscendC::SetFlag<AscendC::HardEvent::MTE3_MTE2>(EVENT_ID0);
                    AscendC::WaitFlag<AscendC::HardEvent::MTE3_MTE2>(EVENT_ID0);
                }
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tileBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> redBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> ptAllBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchesPerCore;
};

extern "C" __global__ __aicore__ void conv3d_softmax_max_pool_max_pool_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dSoftmaxMaxPoolMaxPool op;
    op.Init(x, y, tiling_data.batchesPerCore);
    op.Process();
}
