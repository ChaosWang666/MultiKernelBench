kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                 uint32_t totalNC, uint32_t HW, uint32_t channels,
                                 uint32_t tileLength, float scalingFactor)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t ncPerBlock = totalNC / AscendC::GetBlockNum();
        uint32_t ncStart = blockIdx * ncPerBlock;

        this->HW = HW;
        this->C = channels;
        this->ncPerBlock = ncPerBlock;
        this->ncStart = ncStart;
        this->tileLength = tileLength;
        this->scalingFactor = scalingFactor;
        this->invScaling = 1.0f / scalingFactor;
        this->numTilesPerNC = HW / tileLength;

        xGm.SetGlobalBuffer((__gm__ float *)x + ncStart * HW, ncPerBlock * HW);
        yGm.SetGlobalBuffer((__gm__ float *)y + ncStart * HW, ncPerBlock * HW);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, channels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(biasQueue, 1, ((channels * sizeof(float) + 31) / 32) * 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasLocal = biasQueue.AllocTensor<float>();
        uint32_t alignedC = ((C + 7) / 8) * 8;
        AscendC::DataCopy(biasLocal, biasGm, alignedC);
        biasQueue.EnQue(biasLocal);
        biasLocal = biasQueue.DeQue<float>();

        for (uint32_t i = 0; i < ncPerBlock; i++) {
            uint32_t ncIdx = ncStart + i;
            uint32_t c = ncIdx % C;
            float biasVal = biasLocal.GetValue(c);
            for (uint32_t t = 0; t < numTilesPerNC; t++) {
                CopyIn(i, t);
                Compute(biasVal);
                CopyOut(i, t);
            }
        }

        biasQueue.FreeTensor(biasLocal);
    }

private:
    __aicore__ inline void CopyIn(uint32_t i, uint32_t t)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[i * HW + t * tileLength], tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float biasVal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::Adds(yLocal, xLocal, biasVal, tileLength);
        AscendC::Mins(yLocal, yLocal, (float)1.0f, tileLength);
        AscendC::Maxs(yLocal, yLocal, (float)0.0f, tileLength);
        AscendC::Muls(yLocal, yLocal, scalingFactor, tileLength);
        AscendC::Mins(yLocal, yLocal, (float)1.0f, tileLength);
        AscendC::Maxs(yLocal, yLocal, (float)0.0f, tileLength);
        AscendC::Muls(yLocal, yLocal, invScaling, tileLength);
        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t i, uint32_t t)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[i * HW + t * tileLength], yLocal, tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> biasQueue;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> biasGm;
    uint32_t HW;
    uint32_t C;
    uint32_t ncPerBlock;
    uint32_t ncStart;
    uint32_t tileLength;
    uint32_t numTilesPerNC;
    float scalingFactor;
    float invScaling;
};

extern "C" __global__ __aicore__ void conv_transpose2d_bias_add_clamp_scaling_clamp_divide_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, y, tiling_data.totalNC, tiling_data.HW, tiling_data.channels,
            tiling_data.tileLength, tiling_data.scalingFactor);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, bias_shape, scaling_factor):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x.contiguous()
        bias_flat = self.bias.view(-1).contiguous()
        return custom_ops_lib.conv_transpose2d_bias_add_clamp_scaling_clamp_divide_custom(
            x, bias_flat, float(self.scaling_factor)
        )
