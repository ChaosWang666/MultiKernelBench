kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusion {
public:
    __aicore__ inline KernelFusion() {}
    __aicore__ inline void Init(GM_ADDR xConv, GM_ADDR xNorm, GM_ADDR y,
                                uint32_t N, uint32_t C, uint32_t H, uint32_t W,
                                uint32_t wAligned, uint32_t rowsPerBlock, uint32_t totalRows)
    {
        this->C = C;
        this->H = H;
        this->W = W;
        this->wAligned = wAligned;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startRow = blockIdx * rowsPerBlock;
        uint32_t maxRow = this->startRow + rowsPerBlock;
        this->endRow = (maxRow < totalRows) ? maxRow : totalRows;

        xConvGm.SetGlobalBuffer((__gm__ float*)xConv);
        xNormGm.SetGlobalBuffer((__gm__ float*)xNorm);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        uint32_t chunkSize = C * wAligned;
        pipe.InitBuffer(inQueueXConv, BUFFER_NUM, chunkSize * sizeof(float));
        pipe.InitBuffer(inQueueXNorm, BUFFER_NUM, chunkSize * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, wAligned * sizeof(float));
        pipe.InitBuffer(resBuf, chunkSize * sizeof(float));
        pipe.InitBuffer(maxBuf, wAligned * sizeof(float));
        pipe.InitBuffer(sumBuf, wAligned * sizeof(float));
        pipe.InitBuffer(tmpBuf, wAligned * sizeof(float));
        pipe.InitBuffer(tmp2Buf, wAligned * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t row = startRow; row < endRow; row++) {
            CopyIn(row);
            Compute();
            CopyOut(row);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row)
    {
        AscendC::LocalTensor<float> xConvLocal = inQueueXConv.AllocTensor<float>();
        AscendC::LocalTensor<float> xNormLocal = inQueueXNorm.AllocTensor<float>();

        uint32_t n = row / H;
        uint32_t h = row - n * H;

        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = W * sizeof(float);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPadExtParams<float> padParams;
        padParams.isPad = false;
        padParams.leftPadding = 0;
        padParams.rightPadding = 0;
        padParams.paddingValue = 0;

        uint64_t baseNOffset = (uint64_t)n * C * H * W + (uint64_t)h * W;
        uint64_t chwStride = (uint64_t)H * W;
        for (uint32_t c = 0; c < C; c++) {
            uint64_t gmOffset = baseNOffset + (uint64_t)c * chwStride;
            AscendC::DataCopyPad(xConvLocal[c * wAligned], xConvGm[gmOffset], copyParams, padParams);
            AscendC::DataCopyPad(xNormLocal[c * wAligned], xNormGm[gmOffset], copyParams, padParams);
        }

        inQueueXConv.EnQue(xConvLocal);
        inQueueXNorm.EnQue(xNormLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xConvLocal = inQueueXConv.DeQue<float>();
        AscendC::LocalTensor<float> xNormLocal = inQueueXNorm.DeQue<float>();
        AscendC::LocalTensor<float> resLocal = resBuf.Get<float>();
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> tmp2Local = tmp2Buf.Get<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        const float threeF = 3.0f;
        const float sixF = 6.0f;
        const float zeroF = 0.0f;
        const float invSixF = 1.0f / 6.0f;
        const float negOneF = -1.0f;
        const float oneF = 1.0f;
        const float twoF = 2.0f;
        const float negInfF = -3.4e38f;

        for (uint32_t c = 0; c < C; c++) {
            AscendC::LocalTensor<float> xNormSlice = xNormLocal[c * wAligned];
            AscendC::LocalTensor<float> xConvSlice = xConvLocal[c * wAligned];
            AscendC::Muls(tmpLocal, xNormSlice, twoF, wAligned);
            AscendC::Exp(tmpLocal, tmpLocal, wAligned);
            AscendC::Adds(tmp2Local, tmpLocal, negOneF, wAligned);
            AscendC::Adds(tmpLocal, tmpLocal, oneF, wAligned);
            AscendC::Div(tmpLocal, tmp2Local, tmpLocal, wAligned);

            AscendC::Adds(tmp2Local, tmpLocal, threeF, wAligned);
            AscendC::Mins(tmp2Local, tmp2Local, sixF, wAligned);
            AscendC::Maxs(tmp2Local, tmp2Local, zeroF, wAligned);
            AscendC::Muls(tmp2Local, tmp2Local, invSixF, wAligned);
            AscendC::Mul(tmpLocal, tmpLocal, tmp2Local, wAligned);
            AscendC::Add(resLocal[c * wAligned], xConvSlice, tmpLocal, wAligned);
        }

        inQueueXConv.FreeTensor(xConvLocal);
        inQueueXNorm.FreeTensor(xNormLocal);

        AscendC::Duplicate(maxLocal, negInfF, wAligned);
        for (uint32_t c = 0; c < C; c++) {
            AscendC::Max(maxLocal, maxLocal, resLocal[c * wAligned], wAligned);
        }

        AscendC::Duplicate(sumLocal, zeroF, wAligned);
        for (uint32_t c = 0; c < C; c++) {
            AscendC::Sub(tmpLocal, resLocal[c * wAligned], maxLocal, wAligned);
            AscendC::Exp(tmpLocal, tmpLocal, wAligned);
            AscendC::Add(sumLocal, sumLocal, tmpLocal, wAligned);
        }

        AscendC::Ln(sumLocal, sumLocal, wAligned);
        AscendC::Add(yLocal, sumLocal, maxLocal, wAligned);

        outQueueY.EnQue<float>(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        uint32_t n = row / H;
        uint32_t h = row - n * H;
        uint64_t gmOffset = (uint64_t)n * H * W + (uint64_t)h * W;
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = W * sizeof(float);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        copyParams.rsv = 0;
        AscendC::DataCopyPad(yGm[gmOffset], yLocal, copyParams);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueXConv;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueXNorm;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> resBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmp2Buf;
    AscendC::GlobalTensor<float> xConvGm;
    AscendC::GlobalTensor<float> xNormGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t C;
    uint32_t H;
    uint32_t W;
    uint32_t wAligned;
    uint32_t startRow;
    uint32_t endRow;
};

extern "C" __global__ __aicore__ void conv2d_group_norm_tanh_hard_swish_residual_add_log_sum_exp_custom(
    GM_ADDR x_conv, GM_ADDR x_norm, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusion op;
    op.Init(x_conv, x_norm, y,
            tiling_data.N, tiling_data.C, tiling_data.H, tiling_data.W,
            tiling_data.wAligned, tiling_data.rowsPerBlock, tiling_data.totalRows);
    op.Process();
}
