kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose3dMultiplyMaxGlobalAvgPoolClampCustom {
public:
    __aicore__ inline KernelConvTranspose3dMultiplyMaxGlobalAvgPoolClampCustom() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t outerLen, uint32_t reduceLen,
                                uint32_t tileLen, uint32_t tileNum,
                                uint32_t tailLen, uint32_t outerPerCore,
                                float invReduce)
    {
        this->outerLen = outerLen;
        this->reduceLen = reduceLen;
        this->tileLen = tileLen;
        this->tileNum = tileNum;
        this->tailLen = tailLen;
        this->outerPerCore = outerPerCore;
        this->invReduce = invReduce;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startOuter = blockIdx * outerPerCore;

        uint64_t gmInOffset = static_cast<uint64_t>(this->startOuter) * static_cast<uint64_t>(reduceLen);
        uint64_t gmInSize = static_cast<uint64_t>(outerPerCore) * static_cast<uint64_t>(reduceLen);
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + gmInOffset, gmInSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + this->startOuter, outerPerCore);

        uint32_t outBufBytes = outerPerCore * sizeof(DTYPE_Y);
        outBufBytes = ((outBufBytes + 31) / 32) * 32;
        if (outBufBytes < 32) { outBufBytes = 32; }

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, 1, outBufBytes);
        pipe.InitBuffer(accumBuf, tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(reduceWorkBuf, tileLen * sizeof(DTYPE_X));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> outLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_X> accum = accumBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> reduceWork = reduceWorkBuf.Get<DTYPE_X>();

        for (uint32_t i = 0; i < this->outerPerCore; i++) {
            AscendC::Duplicate<DTYPE_X>(accum, static_cast<DTYPE_X>(0), this->tileLen);

            for (uint32_t t = 0; t < this->tileNum; t++) {
                uint32_t curLen;
                if (t == this->tileNum - 1 && this->tailLen > 0) {
                    curLen = this->tailLen;
                } else {
                    curLen = this->tileLen;
                }
                CopyIn(i, t, curLen);
                Accumulate(curLen, accum);
            }

            AscendC::ReduceSum<DTYPE_X>(outLocal[i], accum, reduceWork, this->tileLen);
        }

        AscendC::Muls<DTYPE_Y>(outLocal, outLocal,
                               static_cast<DTYPE_Y>(this->invReduce),
                               this->outerPerCore);

        AscendC::LocalTensor<DTYPE_Y> scratch = accumBuf.Get<DTYPE_Y>();
        AscendC::Duplicate<DTYPE_Y>(scratch, static_cast<DTYPE_Y>(1.0f), this->outerPerCore);
        AscendC::Min<DTYPE_Y>(outLocal, outLocal, scratch, this->outerPerCore);
        AscendC::Duplicate<DTYPE_Y>(scratch, static_cast<DTYPE_Y>(0.0f), this->outerPerCore);
        AscendC::Max<DTYPE_Y>(outLocal, outLocal, scratch, this->outerPerCore);

        outQueueY.EnQue<DTYPE_Y>(outLocal);
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn(uint32_t outerIdx, uint32_t tileIdx, uint32_t curLen)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        uint64_t offset = static_cast<uint64_t>(outerIdx) * static_cast<uint64_t>(this->reduceLen) +
                          static_cast<uint64_t>(tileIdx) * static_cast<uint64_t>(this->tileLen);

        if ((curLen * sizeof(DTYPE_X)) % 32 == 0) {
            AscendC::DataCopy(xLocal, xGm[offset], curLen);
        } else {
            AscendC::DataCopyParams copyParams;
            copyParams.blockCount = 1;
            copyParams.blockLen = static_cast<uint16_t>(curLen * sizeof(DTYPE_X));
            copyParams.srcStride = 0;
            copyParams.dstStride = 0;

            AscendC::DataCopyPadParams padParams;
            padParams.isPad = true;
            padParams.leftPadding = 0;
            padParams.rightPadding = 0;
            padParams.paddingValue = 0;

            AscendC::DataCopyPad(xLocal, xGm[offset], copyParams, padParams);
        }

        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Accumulate(uint32_t curLen, AscendC::LocalTensor<DTYPE_X>& accum)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::Add<DTYPE_X>(accum, accum, xLocal, curLen);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<DTYPE_Y> outLocal = outQueueY.DeQue<DTYPE_Y>();
        if ((this->outerPerCore * sizeof(DTYPE_Y)) % 32 == 0) {
            AscendC::DataCopy(yGm, outLocal, this->outerPerCore);
        } else {
            AscendC::DataCopyParams copyParams;
            copyParams.blockCount = 1;
            copyParams.blockLen = static_cast<uint16_t>(this->outerPerCore * sizeof(DTYPE_Y));
            copyParams.srcStride = 0;
            copyParams.dstStride = 0;
            AscendC::DataCopyPad(yGm, outLocal, copyParams);
        }
        outQueueY.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceWorkBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t outerLen;
    uint32_t reduceLen;
    uint32_t tileLen;
    uint32_t tileNum;
    uint32_t tailLen;
    uint32_t outerPerCore;
    uint32_t startOuter;
    float invReduce;
};

extern "C" __global__ __aicore__ void conv_transpose3d_multiply_max_global_avg_pool_clamp_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dMultiplyMaxGlobalAvgPoolClampCustom op;
    op.Init(x, y,
            tiling_data.outerLen, tiling_data.reduceLen,
            tiling_data.tileLen, tiling_data.tileNum,
            tiling_data.tailLen, tiling_data.outerPerCore,
            tiling_data.invReduce);
    op.Process();
}

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, scale, maxpool_kernel_size):

        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)

        C = x.shape[1]
