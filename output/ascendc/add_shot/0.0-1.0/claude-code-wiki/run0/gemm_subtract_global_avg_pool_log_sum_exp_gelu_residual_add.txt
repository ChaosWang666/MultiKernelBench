I have enough Wiki context. Now I'll analyze the model to find an optimization opportunity.

Key insight:
- `mean(gemm_out - subtract, dim=1)` = `sum_k x[i,k] * w_avg[k] + (mean(bias) - mean(subtract))` where `w_avg[k] = mean_j(W[j,k])`
- `logsumexp` along dim of size 1 is identity
- Final: `y[i,k] = GELU(dot(x[i], w_avg) + const_bias) + x[i,k]`

This avoids the expensive 2048×8192×8192 GEMM entirely. Kernel takes `x`, precomputed `w_avg`, and scalar `const_bias`.

class KernelFused {

    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t inFeatures, uint32_t rowsPerBlock)
    {
        this->inFeatures = inFeatures;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerBlock;
        uint32_t actualRows = 0;
        if (startRow < batchSize) {
            actualRows = (startRow + rowsPerBlock <= batchSize) ? rowsPerBlock : (batchSize - startRow);
        }
        this->actualRows = actualRows;

        xGm.SetGlobalBuffer((__gm__ float*)x + startRow * inFeatures, actualRows * inFeatures);
        yGm.SetGlobalBuffer((__gm__ float*)y + startRow * inFeatures, actualRows * inFeatures);
        wGm.SetGlobalBuffer((__gm__ float*)w, inFeatures);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, 8);

        pipe.InitBuffer(inQueueX, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, inFeatures * sizeof(float));
        pipe.InitBuffer(wBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(tmpBuf, inFeatures * sizeof(float));
        pipe.InitBuffer(reduceWorkBuf, 8192);
        pipe.InitBuffer(geluBuf, 256);
        pipe.InitBuffer(biasBuf, 32);

    {

        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::DataCopy(wLocal, wGm, inFeatures);
        AscendC::DataCopy(biasLocal, biasGm, 8);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

            CopyOut(i);
        }

private:

    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();

        inQueueX.EnQue(xLocal);

    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> wLocal = wBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceWork = reduceWorkBuf.Get<float>();

        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(tmpLocal, tmpLocal, geluLocal, 64);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Muls(tmpLocal, tmpLocal, (float)-0.07134732f, 64);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Add(tmpLocal, tmpLocal, geluLocal, 64);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Exp(tmpLocal, tmpLocal, 64);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Adds(tmpLocal, tmpLocal, 1.0f, 64);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Div(tmpLocal, geluLocal, tmpLocal, 64);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID1);

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, bias=True):
        super().__init__()
        self.gemm = torch.nn.Linear(in_features, out_features, bias=bias)
        self.subtract = torch.nn.Parameter(torch.randn(out_features))
        self.in_features = in_features
        self.out_features = out_features

        x_contig = x.contiguous()
        w_avg = self.gemm.weight.mean(dim=0).contiguous().to(dtype=torch.float32)

            bias_mean = self.gemm.bias.mean()

            bias_mean = torch.zeros([], dtype=torch.float32, device=x.device)
        sub_mean = self.subtract.mean()
        const_val = (bias_mean - sub_mean).to(dtype=torch.float32)
        const_bias = const_val.reshape(1).repeat(8).contiguous()
        return custom_ops_lib.gemm_subtract_global_avg_pool_log_sum_exp_gelu_residual_add_custom(
