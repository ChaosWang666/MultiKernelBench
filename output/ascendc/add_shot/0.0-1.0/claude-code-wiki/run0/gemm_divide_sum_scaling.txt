kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t ALIGN_NUM = 8;

class KernelGemmDivideSumScaling {
public:
    __aicore__ inline KernelGemmDivideSumScaling() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR factor, GM_ADDR y, uint32_t totalRows, uint32_t cols)
    {
        this->cols = cols;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t baseRows = totalRows / blockNum;
        uint32_t extraRows = totalRows % blockNum;
        uint32_t startRow;
        if (blockIdx < extraRows) {
            this->rowsThisBlock = baseRows + 1;
            startRow = blockIdx * this->rowsThisBlock;
        } else {
            this->rowsThisBlock = baseRows;
            startRow = extraRows * (baseRows + 1) + (blockIdx - extraRows) * baseRows;
        }

        if (this->rowsThisBlock == 0) {
            return;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x + startRow * cols, this->rowsThisBlock * cols);
        yGm.SetGlobalBuffer((__gm__ float*)y + startRow, this->rowsThisBlock);
        factorGm.SetGlobalBuffer((__gm__ float*)factor, ALIGN_NUM);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, cols * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, ALIGN_NUM * sizeof(float));
        pipe.InitBuffer(factorQueue, 1, ALIGN_NUM * sizeof(float));
        pipe.InitBuffer(reduceBuf, cols * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->rowsThisBlock == 0) {
            return;
        }

        AscendC::LocalTensor<float> fLocal = factorQueue.AllocTensor<float>();
        AscendC::DataCopy(fLocal, factorGm, ALIGN_NUM);
        factorQueue.EnQue(fLocal);
        fLocal = factorQueue.DeQue<float>();
        this->factorVal = fLocal.GetValue(0);
        factorQueue.FreeTensor(fLocal);

        for (uint32_t i = 0; i < this->rowsThisBlock; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t i)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[i * this->cols], this->cols);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t i)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> work = reduceBuf.Get<float>();

        AscendC::ReduceSum(yLocal, xLocal, work, (int32_t)this->cols);
        inQueueX.FreeTensor(xLocal);

        AscendC::Muls(yLocal, yLocal, this->factorVal, ALIGN_NUM);

        outQueueY.EnQue<float>(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t i)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopyExtParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = sizeof(float);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        AscendC::DataCopyPad(yGm[i], yLocal, copyParams);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> factorQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> factorGm;
    uint32_t cols;
    uint32_t rowsThisBlock;
    float factorVal;
};

extern "C" __global__ __aicore__ void gemm_divide_sum_scaling_custom(GM_ADDR x, GM_ADDR factor, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmDivideSumScaling op;
    op.Init(x, factor, y, tiling_data.totalRows, tiling_data.cols);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib
class ModelNew(torch.nn.Module):
    def __init__(self, input_size, hidden_size, scaling_factor):
        super().__init__()
        self.weight = torch.nn.Parameter(torch.randn(hidden_size, input_size))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = torch.matmul(x, self.weight.T)
        factor = torch.full((8,), self.scaling_factor / 2.0, dtype=x.dtype, device=x.device)
        return custom_ops_lib.gemm_divide_sum_scaling_custom(x, factor)
