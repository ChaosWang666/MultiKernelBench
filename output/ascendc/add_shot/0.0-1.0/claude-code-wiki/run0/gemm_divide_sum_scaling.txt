kernel_src="""
#include "kernel_operator.h"

class KernelGemmDivSumScale {
public:
    __aicore__ inline KernelGemmDivSumScale() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR y,
                                 uint32_t totalRows, uint32_t k, uint32_t rowsPerBlock)
    {
        this->k = k;
        this->rowsPerBlock = rowsPerBlock;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startRow * k, rowsPerBlock * k);
        wGm.SetGlobalBuffer((__gm__ DTYPE_W *)w, k);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + startRow, rowsPerBlock);

        uint32_t outBufBytes = ((rowsPerBlock * sizeof(DTYPE_Y) + 31) / 32) * 32;
        if (outBufBytes < 32) {
            outBufBytes = 32;
        }

        pipe.InitBuffer(wQue, 1, k * sizeof(DTYPE_W));
        pipe.InitBuffer(xQue, 2, k * sizeof(DTYPE_X));
        pipe.InitBuffer(outQue, 1, outBufBytes);
        pipe.InitBuffer(mulBuf, k * sizeof(DTYPE_X));
        pipe.InitBuffer(reduceBuf, 256);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_W> wLocal = wQue.AllocTensor<DTYPE_W>();
        AscendC::DataCopy(wLocal, wGm, this->k);
        wQue.EnQue(wLocal);
        wLocal = wQue.DeQue<DTYPE_W>();

        AscendC::LocalTensor<DTYPE_X> mulLocal = mulBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> reduceLocal = reduceBuf.Get<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Y> outLocal = outQue.AllocTensor<DTYPE_Y>();

        for (uint32_t r = 0; r < this->rowsPerBlock; r++) {
            AscendC::LocalTensor<DTYPE_X> xLocal = xQue.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xLocal, xGm[r * this->k], this->k);
            xQue.EnQue(xLocal);

            xLocal = xQue.DeQue<DTYPE_X>();
            AscendC::Mul(mulLocal, xLocal, wLocal, this->k);
            xQue.FreeTensor(xLocal);

            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum<DTYPE_X>(reduceLocal, mulLocal, mulLocal, this->k);
            AscendC::PipeBarrier<PIPE_V>();

            DTYPE_Y val = reduceLocal.GetValue(0);
            outLocal.SetValue(r, val);
        }

        wQue.FreeTensor(wLocal);
        outQue.EnQue(outLocal);

        outLocal = outQue.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm, outLocal, this->rowsPerBlock);
        outQue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> wQue;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> xQue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mulBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_W> wGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t k;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void gemm_divide_sum_scaling_custom(GM_ADDR x, GM_ADDR w, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmDivSumScale op;
    op.Init(x, w, y, tiling_data.totalRows, tiling_data.k, tiling_data.rowsPerBlock);
    op.Process();
}
