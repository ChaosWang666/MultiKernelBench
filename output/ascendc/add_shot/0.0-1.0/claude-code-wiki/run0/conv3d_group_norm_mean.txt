kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t BATCH_PER_BLOCK = 8;
constexpr int32_t OUT_PER_BLOCK = 8;

class KernelConv3dGroupNormMean {
public:
    __aicore__ inline KernelConv3dGroupNormMean() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t perBatch, uint32_t tileLength,
                                uint32_t numTiles, uint32_t tailLength,
                                float invNum)
    {
        this->perBatch = perBatch;
        this->tileLength = tileLength;
        this->numTiles = numTiles;
        this->tailLength = tailLength;
        this->invNum = invNum;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockBatchOffset = blockIdx * BATCH_PER_BLOCK;
        uint32_t blockElemOffset = blockBatchOffset * perBatch;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockElemOffset,
                            BATCH_PER_BLOCK * perBatch);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + blockBatchOffset,
                            BATCH_PER_BLOCK);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(accBuf, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(workBuf, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, 1, OUT_PER_BLOCK * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> outLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_X> acc = accBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> work = workBuf.Get<DTYPE_X>();

        AscendC::Duplicate<DTYPE_Y>(outLocal, (DTYPE_Y)0, OUT_PER_BLOCK);

        for (int32_t b = 0; b < BATCH_PER_BLOCK; b++) {
            AscendC::Duplicate<DTYPE_X>(acc, (DTYPE_X)0, this->tileLength);

            uint32_t batchBase = static_cast<uint32_t>(b) * this->perBatch;

            for (uint32_t t = 0; t < this->numTiles; t++) {
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(xLocal,
                                  xGm[batchBase + t * this->tileLength],
                                  this->tileLength);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<DTYPE_X> xIn = inQueueX.DeQue<DTYPE_X>();
                AscendC::Add(acc, acc, xIn, this->tileLength);
                inQueueX.FreeTensor(xIn);
            }

            if (this->tailLength > 0) {
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(xLocal,
                                  xGm[batchBase + this->numTiles * this->tileLength],
                                  this->tailLength);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<DTYPE_X> xIn = inQueueX.DeQue<DTYPE_X>();
                AscendC::Add(acc, acc, xIn, this->tailLength);
                inQueueX.FreeTensor(xIn);
            }

            AscendC::ReduceSum<DTYPE_X>(work, acc, work, this->tileLength);
            DTYPE_X partial = work.GetValue(0);
            outLocal.SetValue(b, partial * this->invNum);
        }

        outQueueY.EnQue(outLocal);
        AscendC::LocalTensor<DTYPE_Y> o = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm, o, OUT_PER_BLOCK);
        outQueueY.FreeTensor(o);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t perBatch;
    uint32_t tileLength;
    uint32_t numTiles;
    uint32_t tailLength;
    float invNum;
};

extern "C" __global__ __aicore__ void conv3d_group_norm_mean_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dGroupNormMean op;
    op.Init(x, y,
            tiling_data.perBatch,
            tiling_data.tileLength,
            tiling_data.numTiles,
            tiling_data.tailLength,
            tiling_data.invNum);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, num_groups):
        super().__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.group_norm = nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        x = self.conv(x)
        x = self.group_norm(x)
        x = x.contiguous()
        x = custom_ops_lib.conv3d_group_norm_mean_custom(x)
        return x
