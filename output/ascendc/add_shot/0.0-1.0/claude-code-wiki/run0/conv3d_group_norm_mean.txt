    Conv3dGroupNormMeanCustomTilingData tiling;

        elementPerBatch *= xShape.GetDim(i);
    }

    uint32_t blockDim = BLOCK_DIM;
    if (batchSize < blockDim) {
        blockDim = batchSize;
    }
    if (blockDim == 0) {
        blockDim = 1;
    }
    uint32_t batchPerBlock = batchSize / blockDim;
    if (batchPerBlock == 0) {
        batchPerBlock = 1;
    }

    uint32_t tileLength = 4800;
    if (tileLength > elementPerBatch) {
        tileLength = (elementPerBatch / 8) * 8;
        if (tileLength == 0) {
            tileLength = elementPerBatch;
        }

    tiling.set_batchSize(batchSize);
    tiling.set_elementPerBatch(elementPerBatch);
    tiling.set_tileLength(tileLength);

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMean {
public:
    __aicore__ inline KernelMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                  uint32_t batchSize, uint32_t elementPerBatch,
                                  uint32_t tileLength, uint32_t batchPerBlock)
    {
        this->batchSize = batchSize;
        this->elementPerBatch = elementPerBatch;
        this->tileLength = tileLength;
        this->batchPerBlock = batchPerBlock;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startBatch = blockIdx * this->batchPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + this->startBatch * elementPerBatch,
                             this->batchPerBlock * elementPerBatch);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + this->startBatch, this->batchPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        uint32_t outAligned = ((this->batchPerBlock + 7) / 8) * 8;
        if (outAligned == 0) outAligned = 8;
        pipe.InitBuffer(outQueueY, 1, outAligned * sizeof(DTYPE_Y));
        pipe.InitBuffer(workBuf, 2048);
        pipe.InitBuffer(dstBuf, 32);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();

        for (uint32_t b = 0; b < this->batchPerBlock; b++) {
            float sum = ProcessBatch(b);
            float mean = sum / (float)this->elementPerBatch;
            yLocal.SetValue(b, mean);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        outQueueY.EnQue(yLocal);
        AscendC::LocalTensor<DTYPE_Y> yOut = outQueueY.DeQue<DTYPE_Y>();
        uint32_t outAligned = ((this->batchPerBlock + 7) / 8) * 8;
        if (outAligned == 0) outAligned = 8;
        AscendC::DataCopy(yGm, yOut, outAligned);
        outQueueY.FreeTensor(yOut);
    }

private:
    __aicore__ inline float ProcessBatch(uint32_t b)
    {
        float sum = 0.0f;
        uint32_t processed = 0;
        while (processed < this->elementPerBatch) {
            uint32_t remain = this->elementPerBatch - processed;
            uint32_t currentTile = (remain >= this->tileLength) ? this->tileLength : remain;
            uint32_t copyLen = ((currentTile + 7) / 8) * 8;

            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xLocal, xGm[b * this->elementPerBatch + processed], copyLen);
            inQueueX.EnQue(xLocal);
            xLocal = inQueueX.DeQue<DTYPE_X>();

            AscendC::LocalTensor<DTYPE_X> dst = dstBuf.Get<DTYPE_X>();
            AscendC::LocalTensor<DTYPE_X> work = workBuf.Get<DTYPE_X>();
            AscendC::ReduceSum<DTYPE_X>(dst, xLocal, work, currentTile);

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            float partialSum = dst.GetValue(0);
            sum += partialSum;

            inQueueX.FreeTensor(xLocal);
            processed += currentTile;
        }
        return sum;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> dstBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batchSize;
    uint32_t elementPerBatch;
    uint32_t tileLength;
    uint32_t batchPerBlock;
    uint32_t startBatch;
};

extern "C" __global__ __aicore__ void conv3d_group_norm_mean_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMean op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.elementPerBatch,
            tiling_data.tileLength, tiling_data.batchPerBlock);
    op.Process();
}
"""

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, num_groups):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.group_norm = nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        x = self.conv(x)
        x = self.group_norm(x)
        x = custom_ops_lib.conv3d_group_norm_mean_custom(x.contiguous())
        return x
'''
