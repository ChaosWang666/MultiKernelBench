constexpr int32_t BUFFER_NUM = 1;

class KernelConvTranspose2dGeluGroupNorm {
public:
    __aicore__ inline KernelConvTranspose2dGeluGroupNorm() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                 uint32_t n, uint32_t c, uint32_t h, uint32_t w,

        this->N = n;
        this->C = c;
        this->H = h;
        this->W = w;
        this->G = gIn;
        this->tileSize = tileSizeIn;
        this->eps = epsIn;
        this->channelsPerGroup = C / G;
        this->spatial = H * W;
        this->groupElements = this->channelsPerGroup * this->spatial;

        uint32_t totalGroups = N * G;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t groupsPerBlock = (totalGroups + blockNum - 1) / blockNum;
        this->startGroup = blockIdx * groupsPerBlock;
        uint32_t endGroupCand = this->startGroup + groupsPerBlock;
        this->endGroup = (endGroupCand < totalGroups) ? endGroupCand : totalGroups;

        uint64_t totalElements = (uint64_t)N * C * H * W;
        xGm.SetGlobalBuffer((__gm__ float*)x, totalElements);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalElements);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, C);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, C);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileSize * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, tileSize * sizeof(float));
        pipe.InitBuffer(workBuf, tileSize * sizeof(float));
        pipe.InitBuffer(redBuf, 32);
        pipe.InitBuffer(gammaQue, 1, 256);
        pipe.InitBuffer(betaQue, 1, 256);
    }

    __aicore__ inline void Process() {
        for (uint32_t grp = startGroup; grp < endGroup; ++grp) {
            ProcessOneGroup(grp);
        }
    }

private:
    __aicore__ inline void ComputeGelu(AscendC::LocalTensor<float> dst,
                                        AscendC::LocalTensor<float> src,
                                        AscendC::LocalTensor<float> tmp,
                                        uint32_t count) {
        AscendC::Mul(tmp, src, src, count);
        AscendC::Mul(tmp, tmp, src, count);
        AscendC::Muls(tmp, tmp, 0.044715f, count);
        AscendC::Add(tmp, tmp, src, count);
        AscendC::Muls(tmp, tmp, -1.5957691216057308f, count);
        AscendC::Exp(tmp, tmp, count);
        AscendC::Adds(tmp, tmp, 1.0f, count);
        AscendC::Div(dst, src, tmp, count);
    }

    __aicore__ inline void CopyTileIn(AscendC::LocalTensor<float>& dst, uint64_t gmOffset, uint32_t count) {
        AscendC::DataCopyParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = (uint16_t)(count * sizeof(float));
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        AscendC::DataCopyPadParams padParams{false, 0, 0, 0};
        AscendC::DataCopyPad(dst, xGm[gmOffset], copyParams, padParams);
    }

    __aicore__ inline void CopyTileOut(AscendC::LocalTensor<float>& src, uint64_t gmOffset, uint32_t count) {
        AscendC::DataCopyParams copyParams;
        copyParams.blockCount = 1;
        copyParams.blockLen = (uint16_t)(count * sizeof(float));
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        AscendC::DataCopyPad(yGm[gmOffset], src, copyParams);
    }

    __aicore__ inline void ProcessOneGroup(uint32_t grp) {
        uint32_t n = grp / G;
        uint32_t g = grp % G;
        uint32_t baseChannel = g * channelsPerGroup;
        uint64_t groupBase = ((uint64_t)n * C + (uint64_t)baseChannel) * spatial;

        AscendC::LocalTensor<float> gammaLocal = gammaQue.AllocTensor<float>();
        AscendC::LocalTensor<float> betaLocal = betaQue.AllocTensor<float>();
        uint32_t cpgAligned = ((channelsPerGroup + 7) / 8) * 8;
        AscendC::DataCopy(gammaLocal, gammaGm[baseChannel], cpgAligned);
        AscendC::DataCopy(betaLocal, betaGm[baseChannel], cpgAligned);
        gammaQue.EnQue(gammaLocal);
        betaQue.EnQue(betaLocal);
        gammaLocal = gammaQue.DeQue<float>();
        betaLocal = betaQue.DeQue<float>();

        float localSum = 0.0f;
        float localSumSq = 0.0f;

        for (uint32_t cc = 0; cc < channelsPerGroup; ++cc) {
            uint64_t channelBase = groupBase + (uint64_t)cc * spatial;
            for (uint32_t sp = 0; sp < spatial; sp += tileSize) {
                uint32_t curSize = (sp + tileSize <= spatial) ? tileSize : (spatial - sp);

                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                CopyTileIn(xLocal, channelBase + sp, curSize);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();

                AscendC::LocalTensor<float> gelu = tmpBuf.Get<float>();
                AscendC::LocalTensor<float> work = workBuf.Get<float>();
                ComputeGelu(gelu, xLocal, work, curSize);

                AscendC::LocalTensor<float> redDst = redBuf.Get<float>();
                AscendC::ReduceSum(redDst, gelu, work, curSize);
                localSum += redDst.GetValue(0);

                AscendC::Mul(xLocal, gelu, gelu, curSize);
                AscendC::ReduceSum(redDst, xLocal, work, curSize);
                localSumSq += redDst.GetValue(0);

                inQueueX.FreeTensor(xLocal);
            }
        }

        float invN = 1.0f / (float)groupElements;
        float mean = localSum * invN;
        float var = localSumSq * invN - mean * mean;
        if (var < 0.0f) var = 0.0f;

        AscendC::LocalTensor<float> redDst2 = redBuf.Get<float>();
        redDst2.SetValue(0, var + eps);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Sqrt(redDst2, redDst2, 1);
        AscendC::PipeBarrier<PIPE_ALL>();
        float stdVal = redDst2.GetValue(0);
        float invStd = 1.0f / stdVal;

        for (uint32_t cc = 0; cc < channelsPerGroup; ++cc) {
            float gammaVal = gammaLocal.GetValue(cc);
            float betaVal = betaLocal.GetValue(cc);
            float scale = invStd * gammaVal;
            float biasTerm = betaVal - mean * scale;

            uint64_t channelBase = groupBase + (uint64_t)cc * spatial;
            for (uint32_t sp = 0; sp < spatial; sp += tileSize) {
                uint32_t curSize = (sp + tileSize <= spatial) ? tileSize : (spatial - sp);

                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                CopyTileIn(xLocal, channelBase + sp, curSize);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();

                AscendC::LocalTensor<float> gelu = tmpBuf.Get<float>();
                AscendC::LocalTensor<float> work = workBuf.Get<float>();
                ComputeGelu(gelu, xLocal, work, curSize);

                AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
                AscendC::Muls(yLocal, gelu, scale, curSize);
                AscendC::Adds(yLocal, yLocal, biasTerm, curSize);

                outQueueY.EnQue(yLocal);
                yLocal = outQueueY.DeQue<float>();
                CopyTileOut(yLocal, channelBase + sp, curSize);
                outQueueY.FreeTensor(yLocal);
                inQueueX.FreeTensor(xLocal);
            }
        }

        gammaQue.FreeTensor(gammaLocal);
        betaQue.FreeTensor(betaLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> gammaQue, betaQue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, workBuf, redBuf;
    AscendC::GlobalTensor<float> xGm, gammaGm, betaGm, yGm;

    uint32_t N, C, H, W, G;
    uint32_t channelsPerGroup, spatial, groupElements;
    uint32_t tileSize;
    uint32_t startGroup, endGroup;
    float eps;
};
