kernel_src="""
#include "kernel_operator.h"

class KernelConvtranspose3dReluGroupnormCustom {
public:
    __aicore__ inline KernelConvtranspose3dReluGroupnormCustom() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                 uint32_t totalGroups, uint32_t groupElements,
                                 uint32_t channelsPerGroup, uint32_t spatial,
                                 uint32_t numGroupsPerBatch, uint32_t tileLength,
                                 float epsilon, float invN)
    {
        this->groupElements = groupElements;
        this->channelsPerGroup = channelsPerGroup;
        this->spatial = spatial;
        this->numGroupsPerBatch = numGroupsPerBatch;
        this->tileLength = tileLength;
        this->epsilon = epsilon;
        this->invN = invN;
        this->numTilesPerGroup = groupElements / tileLength;

        uint32_t blockDim = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->groupsPerCore = totalGroups / blockDim;
        this->startGroup = blockIdx * this->groupsPerCore;

        uint64_t totalLen = static_cast<uint64_t>(totalGroups) * static_cast<uint64_t>(groupElements);
        uint32_t totalChannels = numGroupsPerBatch * channelsPerGroup;

        xGm.SetGlobalBuffer((__gm__ float*)x, totalLen);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalLen);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, totalChannels);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, totalChannels);

        pipe.InitBuffer(bufA, tileLength * sizeof(float));
        pipe.InitBuffer(bufB, tileLength * sizeof(float));
        pipe.InitBuffer(reduceBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t g = 0; g < groupsPerCore; g++) {
            uint32_t groupIdx = startGroup + g;
            uint32_t groupInBatch = groupIdx % numGroupsPerBatch;
            uint32_t channelBase = groupInBatch * channelsPerGroup;
            uint64_t groupOffset = static_cast<uint64_t>(groupIdx) * static_cast<uint64_t>(groupElements);
            ProcessOneGroup(groupOffset, channelBase);
        }
    }

private:
    __aicore__ inline void ProcessOneGroup(uint64_t groupOffset, uint32_t channelBase)
    {
        AscendC::LocalTensor<float> aLocal = bufA.Get<float>();
        AscendC::LocalTensor<float> bLocal = bufB.Get<float>();
        AscendC::LocalTensor<float> rLocal = reduceBuf.Get<float>();

        float sum = 0.0f;
        float sumSq = 0.0f;

        for (uint32_t t = 0; t < numTilesPerGroup; t++) {
            uint64_t tileOffset = groupOffset + static_cast<uint64_t>(t) * static_cast<uint64_t>(tileLength);

            AscendC::DataCopy(aLocal, xGm[tileOffset], tileLength);
            AscendC::PipeBarrier<PIPE_ALL>();

            AscendC::Maxs(aLocal, aLocal, (float)0.0f, tileLength);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::DataCopy(yGm[tileOffset], aLocal, tileLength);
            AscendC::PipeBarrier<PIPE_ALL>();

            AscendC::ReduceSum(rLocal, aLocal, bLocal, tileLength);
            AscendC::PipeBarrier<PIPE_V>();
            float partialSum = rLocal.GetValue(0);
            sum += partialSum;

            AscendC::Mul(bLocal, aLocal, aLocal, tileLength);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum(rLocal, bLocal, aLocal, tileLength);
            AscendC::PipeBarrier<PIPE_V>();
            float partialSumSq = rLocal.GetValue(0);
            sumSq += partialSumSq;
        }

        float mean = sum * invN;
        float var = sumSq * invN - mean * mean;
        float varEps = var + epsilon;

        AscendC::Duplicate(rLocal, varEps, 64);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Sqrt(rLocal, rLocal, 64);
        AscendC::PipeBarrier<PIPE_V>();
        float stddev = rLocal.GetValue(0);
        float invStd = 1.0f / stddev;

        for (uint32_t t = 0; t < numTilesPerGroup; t++) {
            uint32_t channelIdx = channelBase + t;
            float gammaC = gammaGm.GetValue(channelIdx);
            float betaC = betaGm.GetValue(channelIdx);
            float aCoef = invStd * gammaC;
            float bCoef = betaC - mean * aCoef;

            uint64_t tileOffset = groupOffset + static_cast<uint64_t>(t) * static_cast<uint64_t>(tileLength);

            AscendC::DataCopy(aLocal, yGm[tileOffset], tileLength);
            AscendC::PipeBarrier<PIPE_ALL>();

            AscendC::Muls(aLocal, aLocal, aCoef, tileLength);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Adds(aLocal, aLocal, bCoef, tileLength);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::DataCopy(yGm[tileOffset], aLocal, tileLength);
            AscendC::PipeBarrier<PIPE_ALL>();
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bufA;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bufB;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    uint32_t groupElements;
    uint32_t channelsPerGroup;
    uint32_t spatial;
    uint32_t numGroupsPerBatch;
    uint32_t tileLength;
    uint32_t numTilesPerGroup;
    uint32_t startGroup;
    uint32_t groupsPerCore;
    float epsilon;
    float invN;
};

extern "C" __global__ __aicore__ void convtranspose3d_relu_groupnorm_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvtranspose3dReluGroupnormCustom op;
    op.Init(x, gamma, beta, y,
            tiling_data.totalGroups, tiling_data.groupElements,
            tiling_data.channelsPerGroup, tiling_data.spatial,
            tiling_data.numGroupsPerBatch, tiling_data.tileLength,
            tiling_data.epsilon, tiling_data.invN);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, groups, bias=False):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, bias=bias)
        self.group_norm = nn.GroupNorm(num_groups=groups, num_channels=out_channels)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x.contiguous()
        return custom_ops_lib.convtranspose3d_relu_groupnorm_custom(
            x, self.group_norm.weight.contiguous(), self.group_norm.bias.contiguous()
        )
