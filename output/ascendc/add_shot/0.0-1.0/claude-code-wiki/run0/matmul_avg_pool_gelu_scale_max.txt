constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulAvgPoolGeluScaleMax {
public:
    __aicore__ inline KernelMatmulAvgPoolGeluScaleMax() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalRows, uint32_t cols,
                                uint32_t poolKernelSize, float scaleFactor)
    {
        this->cols = cols;
        this->poolKernelSize = poolKernelSize;
        this->pooledCols = cols / poolKernelSize;
        this->scaleFactor = scaleFactor;
        this->invPool = 1.0f / (float)poolKernelSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t baseRows = totalRows / blockNum;
        uint32_t extraRows = totalRows % blockNum;
        uint32_t startRow;
        if (blockIdx < extraRows) {
            this->rowsThisBlock = baseRows + 1;
            startRow = blockIdx * this->rowsThisBlock;
        } else {
            this->rowsThisBlock = baseRows;
            startRow = extraRows * (baseRows + 1) + (blockIdx - extraRows) * baseRows;
        }

        if (this->rowsThisBlock == 0) {
            return;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x + startRow * cols, this->rowsThisBlock * cols);
        yGm.SetGlobalBuffer((__gm__ float*)y + startRow, this->rowsThisBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, cols * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, 32);
        pipe.InitBuffer(pooledBuf, this->pooledCols * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->pooledCols * sizeof(float));
        pipe.InitBuffer(reduceBuf, 8 * 1024);
    }

    __aicore__ inline void Process()
    {
        if (this->rowsThisBlock == 0) {
            return;
        }
        for (uint32_t i = 0; i < this->rowsThisBlock; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t i)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[i * this->cols], this->cols);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t i)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> pooled = pooledBuf.Get<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceBuf.Get<float>();

        for (uint32_t p = 0; p < this->pooledCols; p++) {
            AscendC::ReduceSum<float, false>(pooled[p], xLocal[p * this->poolKernelSize],
                                             reduceTmp, (int32_t)this->poolKernelSize);
        }
        inQueueX.FreeTensor(xLocal);

        AscendC::Muls<float>(pooled, pooled, this->invPool, this->pooledCols);

        ApplyGelu(pooled, tmp, this->pooledCols);

        AscendC::Muls<float>(pooled, pooled, this->scaleFactor, this->pooledCols);

        AscendC::ReduceMax<float>(yLocal, pooled, reduceTmp, (int32_t)this->pooledCols, false);

        outQueueY.EnQue<float>(yLocal);
    }

    __aicore__ inline void ApplyGelu(AscendC::LocalTensor<float>& x, AscendC::LocalTensor<float>& tmp, uint32_t n)
    {
        AscendC::Mul<float>(tmp, x, x, n);
        AscendC::Mul<float>(tmp, tmp, x, n);
        AscendC::Muls<float>(tmp, tmp, (float)0.044715, n);
        AscendC::Add<float>(tmp, tmp, x, n);
        AscendC::Muls<float>(tmp, tmp, (float)0.7978845608028654, n);
        AscendC::Tanh<float>(tmp, tmp, n);
        AscendC::Adds<float>(tmp, tmp, (float)1.0, n);
        AscendC::Mul<float>(tmp, tmp, x, n);
        AscendC::Muls<float>(x, tmp, (float)0.5, n);
    }

        copyParams.blockCount = 1;
        copyParams.blockLen = sizeof(float);
        copyParams.srcStride = 0;
        copyParams.dstStride = 0;
        AscendC::DataCopyPad(yGm[i], yLocal, copyParams);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> pooledBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
