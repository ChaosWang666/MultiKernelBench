kernel_src="""
#include "kernel_operator.h"

constexpr int32_t OUT_FEATURES = 8192;
constexpr int32_t POOL_KERNEL = 16;
constexpr int32_t POOLED_LEN = 512;

class KernelFusedOp {
public:
    __aicore__ inline KernelFusedOp() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalBatch,
                                 uint32_t perCoreBatch, float scaleFactor,
                                 float invPoolKernel)
    {
        this->perCoreBatch = perCoreBatch;
        this->scaleFactor = scaleFactor;
        this->invPoolKernel = invPoolKernel;

        uint32_t blockIdx = AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ float*)x + blockIdx * perCoreBatch * OUT_FEATURES,
                             perCoreBatch * OUT_FEATURES);
        yGm.SetGlobalBuffer((__gm__ float*)y + blockIdx * perCoreBatch, perCoreBatch);

        uint32_t maxBufSize = (perCoreBatch < 8 ? 8 : perCoreBatch) * sizeof(float);
        pipe.InitBuffer(xBuf, OUT_FEATURES * sizeof(float));
        pipe.InitBuffer(poolBuf, POOLED_LEN * sizeof(float));
        pipe.InitBuffer(tempBuf, POOLED_LEN * sizeof(float));
        pipe.InitBuffer(maxBuf, maxBufSize);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> pool = poolBuf.Get<float>();
        AscendC::LocalTensor<float> temp = tempBuf.Get<float>();
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();

        for (uint32_t i = 0; i < this->perCoreBatch; i++) {
            ProcessRow(i, xLocal, pool, temp, maxLocal);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        AscendC::DataCopy(yGm, maxLocal, this->perCoreBatch);
    }

private:
    __aicore__ inline void ProcessRow(uint32_t i,
                                       AscendC::LocalTensor<float>& xLocal,
                                       AscendC::LocalTensor<float>& pool,
                                       AscendC::LocalTensor<float>& temp,
                                       AscendC::LocalTensor<float>& maxLocal)
    {
        AscendC::DataCopy(xLocal, xGm[i * OUT_FEATURES], OUT_FEATURES);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);

        for (int32_t j = 0; j < POOLED_LEN; j++) {
            float sum = 0.0f;
            int32_t base = j * POOL_KERNEL;
            for (int32_t k = 0; k < POOL_KERNEL; k++) {
                sum = sum + xLocal.GetValue(base + k);
            }
            pool.SetValue(j, sum * this->invPoolKernel);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        AscendC::Mul(temp, pool, pool, POOLED_LEN);
        AscendC::Mul(temp, temp, pool, POOLED_LEN);
        AscendC::Muls(temp, temp, (float)0.044715f, POOLED_LEN);
        AscendC::Add(temp, temp, pool, POOLED_LEN);
        AscendC::Muls(temp, temp, (float)(-1.595769f), POOLED_LEN);
        AscendC::Exp(temp, temp, POOLED_LEN);
        AscendC::Adds(temp, temp, (float)1.0f, POOLED_LEN);
        AscendC::Div(pool, pool, temp, POOLED_LEN);
        AscendC::Muls(pool, pool, this->scaleFactor, POOLED_LEN);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

        float rowMax = pool.GetValue(0);
        for (int32_t j = 1; j < POOLED_LEN; j++) {
            float v = pool.GetValue(j);
            if (v > rowMax) {
                rowMax = v;
            }
        }
        maxLocal.SetValue(i, rowMax);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, poolBuf, tempBuf, maxBuf;
    AscendC::GlobalTensor<float> xGm, yGm;
    uint32_t perCoreBatch;
    float scaleFactor;
    float invPoolKernel;
};

extern "C" __global__ __aicore__ void matmul_avg_pool_gelu_scale_max_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedOp op;
    op.Init(x, y, tiling_data.totalBatch, tiling_data.perCoreBatch,
            tiling_data.scaleFactor, tiling_data.invPoolKernel);
    op.Process();
}
