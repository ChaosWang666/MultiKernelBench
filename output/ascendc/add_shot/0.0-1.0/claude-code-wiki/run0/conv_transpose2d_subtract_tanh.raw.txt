constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose2dSubtractTanh {
public:
    __aicore__ inline KernelConvTranspose2dSubtractTanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                uint32_t totalLength, uint32_t tileNum,
                                uint32_t tileLength, uint32_t lastTileLength,
                                uint32_t spatialSize, uint32_t numChannels)
    {
        this->blockLength = totalLength / AscendC::GetBlockNum();
        this->tileNum = tileNum;
        this->tileLength = tileLength;
        this->lastTileLength = lastTileLength;
        this->spatialSize = spatialSize;
        this->numChannels = numChannels;
        this->coreOffset = this->blockLength * AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ float *)x + this->coreOffset, this->blockLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->coreOffset, this->blockLength);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, numChannels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, tileLength * sizeof(float));
    }
    __aicore__ inline void Process()
    {
        for (int32_t i = 0; i < this->tileNum; i++) {
            uint32_t tileStart = (uint32_t)i * this->tileLength;
            uint32_t globalStart = this->coreOffset + tileStart;
            uint32_t channelIdx = (globalStart / this->spatialSize) % this->numChannels;
            float biasVal = biasGm.GetValue(channelIdx);
            CopyIn(tileStart, this->tileLength);
            Compute(biasVal, this->tileLength);
            CopyOut(tileStart, this->tileLength);
        }
        if (this->lastTileLength > 0) {
            uint32_t tileStart = (uint32_t)this->tileNum * this->tileLength;
            uint32_t globalStart = this->coreOffset + tileStart;
            uint32_t channelIdx = (globalStart / this->spatialSize) % this->numChannels;
            float biasVal = biasGm.GetValue(channelIdx);
            CopyIn(tileStart, this->lastTileLength);
            Compute(biasVal, this->lastTileLength);
            CopyOut(tileStart, this->lastTileLength);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], len);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(float biasVal, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        AscendC::Adds(tmp, xLocal, -biasVal, len);
        AscendC::Muls(tmp, tmp, -2.0f, len);
        AscendC::Exp(tmp, tmp, len);
        AscendC::Adds(tmp, tmp, 1.0f, len);
        AscendC::Reciprocal(tmp, tmp, len);
        AscendC::Muls(tmp, tmp, 2.0f, len);
        AscendC::Adds(zLocal, tmp, -1.0f, len);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[offset], zLocal, len);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm, biasGm, zGm;

    uint32_t tileNum;
