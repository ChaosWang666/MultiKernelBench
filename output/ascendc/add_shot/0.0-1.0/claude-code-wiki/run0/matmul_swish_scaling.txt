static ge::graphStatus TilingFunc(gert::TilingContext* context)

    auto xShape = context->GetInputShape(0)->GetOriginShape();
    auto weightShape = context->GetInputShape(1)->GetOriginShape();

    uint32_t K = xShape.GetDim(1);
    uint32_t N = weightShape.GetDim(0);

    const float* scalingFactorPtr = attrs->GetAttrPointer<float>(0);
    float scalingFactor = *scalingFactorPtr;

    auto ascendcPlatform = platform_ascendc::PlatformAscendC(context->GetPlatformInfo());
    auto aicNum = ascendcPlatform.GetCoreNumAic();

    matmul_tiling::MultiCoreMatmulTiling cubeTiling(ascendcPlatform);
    cubeTiling.SetDim(aicNum);
    cubeTiling.SetAType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                        matmul_tiling::DataType::DT_FLOAT, false);
    cubeTiling.SetBType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                        matmul_tiling::DataType::DT_FLOAT, true);
    cubeTiling.SetCType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                        matmul_tiling::DataType::DT_FLOAT);
    cubeTiling.SetBiasType(matmul_tiling::TPosition::GM, matmul_tiling::CubeFormat::ND,
                           matmul_tiling::DataType::DT_FLOAT);
    cubeTiling.SetShape(M, N, K);
    cubeTiling.SetOrgShape(M, N, K);
    cubeTiling.SetBias(true);
    cubeTiling.SetBufferSpace(-1, -1, -1);

    if (cubeTiling.GetTiling(tiling.cubeTilingData) == -1) {
        return ge::GRAPH_FAILED;
    }

    tiling.set_scalingFactor(scalingFactor);
    tiling.set_M(M);
    tiling.set_N(N);
    tiling.set_K(K);

    context->SetBlockDim(aicNum);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(),
                        context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());

    size_t userWorkspaceSize = static_cast<size_t>(M) * static_cast<size_t>(N) * sizeof(float);
    size_t systemWorkspaceSize = static_cast<size_t>(ascendcPlatform.GetLibApiWorkSpaceSize());
    size_t* currentWorkspace = context->GetWorkspaceSizes(1);
    currentWorkspace[0] = userWorkspaceSize + systemWorkspaceSize;

    return ge::GRAPH_SUCCESS;

kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t TILE_LEN = 2048;

typedef matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float, false> aType;
typedef matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float, true> bType;
typedef matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float> cType;
typedef matmul::MatmulType<AscendC::TPosition::GM, matmul::CubeFormat::ND, float> biasType;

extern "C" __global__ __aicore__ void matmul_swish_scaling_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    GM_ADDR userWS = GetUserWorkspace(workspace);
    if (userWS == nullptr) {
        return;
    }

    if ASCEND_IS_AIC {
        TPipe cubePipe;
        matmul::Matmul<aType, bType, cType, biasType> mm;
        REGIST_MATMUL_OBJ(&cubePipe, GetSysWorkSpacePtr(), mm, &tilingData.cubeTilingData);

        GlobalTensor<float> xGm;
        GlobalTensor<float> weightGm;
        GlobalTensor<float> biasGm;
        GlobalTensor<float> outGm;
        xGm.SetGlobalBuffer((__gm__ float*)x);
        weightGm.SetGlobalBuffer((__gm__ float*)weight);
        biasGm.SetGlobalBuffer((__gm__ float*)bias);
        outGm.SetGlobalBuffer((__gm__ float*)userWS);

        mm.SetTensorA(xGm);
        mm.SetTensorB(weightGm, true);
        mm.SetBias(biasGm);
        mm.IterateAll(outGm);
        mm.End();
    }

    SyncAll();

    if ASCEND_IS_AIV {
        TPipe vecPipe;
        TQue<TPosition::VECIN, BUFFER_NUM> inQueue;
        TQue<TPosition::VECOUT, BUFFER_NUM> outQueue;
        TBuf<TPosition::VECCALC> tmpBuf;
        TBuf<TPosition::VECCALC> oneBuf;

        vecPipe.InitBuffer(inQueue, BUFFER_NUM, TILE_LEN * sizeof(float));
        vecPipe.InitBuffer(outQueue, BUFFER_NUM, TILE_LEN * sizeof(float));
        vecPipe.InitBuffer(tmpBuf, TILE_LEN * sizeof(float));
        vecPipe.InitBuffer(oneBuf, TILE_LEN * sizeof(float));

        GlobalTensor<float> srcGm;
        GlobalTensor<float> dstGm;
        srcGm.SetGlobalBuffer((__gm__ float*)userWS);
        dstGm.SetGlobalBuffer((__gm__ float*)y);

        LocalTensor<float> oneLocal = oneBuf.Get<float>();
        Duplicate(oneLocal, (float)1.0f, TILE_LEN);

        uint32_t totalLen = tilingData.M * tilingData.N;
        uint32_t aivNum = GetBlockNum();
        uint32_t aivIdx = GetBlockIdx();

        uint32_t lenPerBlock = (totalLen + aivNum - 1) / aivNum;
        lenPerBlock = ((lenPerBlock + 7) / 8) * 8;
        uint32_t startOffset = aivIdx * lenPerBlock;
        if (startOffset >= totalLen) {
            return;
        }
        uint32_t realLen = lenPerBlock;
        if (startOffset + realLen > totalLen) {
            realLen = totalLen - startOffset;
        }

        float scaling = tilingData.scalingFactor;
        uint32_t processed = 0;
        while (processed < realLen) {
            uint32_t curLen = realLen - processed;
            if (curLen > (uint32_t)TILE_LEN) {
                curLen = (uint32_t)TILE_LEN;
            }

            LocalTensor<float> inLocal = inQueue.AllocTensor<float>();
            DataCopy(inLocal, srcGm[startOffset + processed], curLen);
            inQueue.EnQue(inLocal);

            LocalTensor<float> xLocal = inQueue.DeQue<float>();
            LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
            LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

            Muls(tmpLocal, xLocal, (float)-1.0f, curLen);
            Exp(tmpLocal, tmpLocal, curLen);
            Adds(tmpLocal, tmpLocal, (float)1.0f, curLen);
            Div(tmpLocal, oneLocal, tmpLocal, curLen);
            Mul(outLocal, xLocal, tmpLocal, curLen);
            Muls(outLocal, outLocal, scaling, curLen);

            outQueue.EnQue<float>(outLocal);
            LocalTensor<float> yLocal = outQueue.DeQue<float>();
            DataCopy(dstGm[startOffset + processed], yLocal, curLen);

            inQueue.FreeTensor(xLocal);
            outQueue.FreeTensor(yLocal);

            processed += curLen;
        }
    }
}

python_bind_src="""
#include <torch/library.h>
#include <torch/csrc/autograd/custom_function.h>
#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor matmul_swish_scaling_custom_impl_npu(const at::Tensor& x,
                                                 const at::Tensor& weight,
                                                 const at::Tensor& bias,
                                                 double scaling_factor) {
    auto sizes = x.sizes();
    int64_t M = sizes[0];
    int64_t N = weight.size(0);
    at::Tensor result = at::empty({M, N}, x.options());
    float sf = static_cast<float>(scaling_factor);
    EXEC_NPU_CMD(aclnnMatmulSwishScalingCustom, x, weight, bias, sf, result);
    return result;
}

TORCH_LIBRARY_IMPL(myops, PrivateUse1, m) {
    m.impl("matmul_swish_scaling_custom", &matmul_swish_scaling_custom_impl_npu);
}

PYBIND11_MODULE(TORCH_EXTENSION_NAME, m) {
    m.def("matmul_swish_scaling_custom", &matmul_swish_scaling_custom_impl_npu,
          "matmul + swish + scaling fused kernel");
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, scaling_factor):
        super(ModelNew, self).__init__()
        self.matmul = nn.Linear(in_features, out_features)
        self.scaling_factor = scaling_factor

    def forward(self, x):
        return custom_ops_lib.matmul_swish_scaling_custom(
            x.contiguous(),
            self.matmul.weight.contiguous(),
            self.matmul.bias.contiguous(),
            float(self.scaling_factor),
