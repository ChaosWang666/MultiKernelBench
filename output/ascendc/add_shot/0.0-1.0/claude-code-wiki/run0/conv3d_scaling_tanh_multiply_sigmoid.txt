kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusion {
public:
    __aicore__ inline KernelFusion() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR scale, GM_ADDR bias, GM_ADDR z,
                                uint32_t numChannels, uint32_t planeSize,
                                uint32_t tileLength, uint32_t totalPlanes,
                                uint32_t planesPerBlock)
    {
        this->numChannels = numChannels;
        this->planeSize = planeSize;
        this->tileLength = tileLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startPlane = blockIdx * planesPerBlock;
        uint32_t endPlane = startPlane + planesPerBlock;
        if (endPlane > totalPlanes) {
            endPlane = totalPlanes;
        }
        if (startPlane >= totalPlanes) {
            this->myFirstPlane = 0;
            this->myPlanes = 0;
        } else {
            this->myFirstPlane = startPlane;
            this->myPlanes = endPlane - startPlane;
        }
        this->tilesPerPlane = (tileLength == 0) ? 0 : (planeSize / tileLength);

        uint32_t globalOffset = this->myFirstPlane * planeSize;
        uint32_t localCount = this->myPlanes * planeSize;

        xGm.SetGlobalBuffer((__gm__ float*)x + globalOffset, localCount);
        zGm.SetGlobalBuffer((__gm__ float*)z + globalOffset, localCount);
        scaleGm.SetGlobalBuffer((__gm__ float*)scale, numChannels);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, numChannels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->tileLength * sizeof(float));

        uint32_t alignedBytes = ((numChannels * (uint32_t)sizeof(float) + 31u) / 32u) * 32u;
        if (alignedBytes < 32u) {
            alignedBytes = 32u;
        }
        pipe.InitBuffer(scaleBuf, alignedBytes);
        pipe.InitBuffer(biasBuf, alignedBytes);
    }

    __aicore__ inline void Process()
    {
        if (this->myPlanes == 0) {
            return;
        }

        AscendC::LocalTensor<float> scaleLocal = scaleBuf.Get<float>();
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::DataCopy(scaleLocal, scaleGm, this->numChannels);
        AscendC::DataCopy(biasLocal, biasGm, this->numChannels);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);

        for (uint32_t p = 0; p < this->myPlanes; p++) {
            uint32_t channelIdx = (this->myFirstPlane + p) % this->numChannels;
            float scaleVal = scaleLocal.GetValue(channelIdx);
            float biasVal = biasLocal.GetValue(channelIdx);

            for (uint32_t t = 0; t < this->tilesPerPlane; t++) {
                uint32_t tileOffset = p * this->planeSize + t * this->tileLength;
                CopyIn(tileOffset);
                Compute(scaleVal, biasVal);
                CopyOut(tileOffset);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float scaleVal, float biasVal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        AscendC::Muls(xLocal, xLocal, scaleVal, this->tileLength);

        AscendC::Muls(tmp, xLocal, 2.0f, this->tileLength);
        AscendC::Exp(tmp, tmp, this->tileLength);
        AscendC::Adds(tmp, tmp, 1.0f, this->tileLength);
        AscendC::Reciprocal(xLocal, tmp, this->tileLength);
        AscendC::Muls(xLocal, xLocal, -2.0f, this->tileLength);
        AscendC::Adds(xLocal, xLocal, 1.0f, this->tileLength);

        AscendC::Muls(xLocal, xLocal, biasVal, this->tileLength);

        AscendC::Muls(tmp, xLocal, -1.0f, this->tileLength);
        AscendC::Exp(tmp, tmp, this->tileLength);
        AscendC::Adds(tmp, tmp, 1.0f, this->tileLength);
        AscendC::Reciprocal(zLocal, tmp, this->tileLength);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[offset], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scaleBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> zGm;
    AscendC::GlobalTensor<float> scaleGm;
    AscendC::GlobalTensor<float> biasGm;
    uint32_t numChannels;
    uint32_t planeSize;
    uint32_t tileLength;
    uint32_t myFirstPlane;
    uint32_t myPlanes;
    uint32_t tilesPerPlane;
};

extern "C" __global__ __aicore__ void conv3d_scaling_tanh_multiply_sigmoid_custom(
    GM_ADDR x, GM_ADDR scale, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusion op;
    op.Init(x, scale, bias, z,
            tiling_data.numChannels, tiling_data.planeSize,
            tiling_data.tileLength, tiling_data.totalPlanes,
            tiling_data.planesPerBlock);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, scaling_factor, bias_shape):
        super().__init__()
        self.conv = torch.nn.Conv3d(in_channels, out_channels, kernel_size)
        self.scaling_factor = torch.nn.Parameter(torch.randn(bias_shape))
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        x = self.conv(x).contiguous()
        scale_flat = self.scaling_factor.reshape(-1).contiguous()
        bias_flat = self.bias.reshape(-1).contiguous()
        return custom_ops_lib.conv3d_scaling_tanh_multiply_sigmoid_custom(x, scale_flat, bias_flat)
