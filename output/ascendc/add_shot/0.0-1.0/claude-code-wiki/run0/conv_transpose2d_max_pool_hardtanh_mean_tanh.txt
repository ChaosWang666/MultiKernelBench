kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedTail {
public:
    __aicore__ inline KernelFusedTail() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalTasks, uint32_t innerSize,
                                uint32_t tasksPerCore, uint32_t tileLength,
                                uint32_t tileNum, float invInnerSize,
                                float hardtanhMin, float hardtanhMax)
    {
        this->totalTasks = totalTasks;
        this->innerSize = innerSize;
        this->tasksPerCore = tasksPerCore;
        this->tileLength = tileLength;
        this->tileNum = tileNum;
        this->invInnerSize = invInnerSize;
        this->hardtanhMin = hardtanhMin;
        this->hardtanhMax = hardtanhMax;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t start = blockIdx * tasksPerCore;
        uint32_t end = start + tasksPerCore;
        if (end > totalTasks) {
            end = totalTasks;
        }
        this->startTask = start;
        this->numTasks = (end > start) ? (end - start) : 0;

        xGm.SetGlobalBuffer((__gm__ float *)x + this->startTask * this->innerSize,
                            this->numTasks * this->innerSize);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->startTask, this->tasksPerCore);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(accBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(reduceBuf, 32);
        pipe.InitBuffer(workBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(resultsBuf, this->tasksPerCore * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->numTasks == 0) {
            return;
        }

        AscendC::LocalTensor<float> resultsLocal = resultsBuf.Get<float>();
        AscendC::LocalTensor<float> accLocal = accBuf.Get<float>();
        AscendC::LocalTensor<float> rLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> wLocal = workBuf.Get<float>();

        int32_t evtVS = static_cast<int32_t>(
            AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        int32_t evtSV = static_cast<int32_t>(
            AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_V));

        for (uint32_t i = 0; i < this->numTasks; i++) {
            AscendC::Duplicate<float>(accLocal, 0.0f, this->tileLength);

            for (uint32_t j = 0; j < this->tileNum; j++) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[i * this->innerSize + j * this->tileLength], this->tileLength);
                inQueueX.EnQue(xLocal);

                xLocal = inQueueX.DeQue<float>();
                AscendC::Mins<float>(xLocal, xLocal, this->hardtanhMax, this->tileLength);
                AscendC::Maxs<float>(xLocal, xLocal, this->hardtanhMin, this->tileLength);
                AscendC::Add<float>(accLocal, accLocal, xLocal, this->tileLength);
                inQueueX.FreeTensor(xLocal);
            }

            AscendC::ReduceSum<float>(rLocal, accLocal, wLocal, this->tileLength);

            AscendC::SetFlag<AscendC::HardEvent::V_S>(evtVS);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(evtVS);

            float sum = rLocal.GetValue(0);
            resultsLocal.SetValue(i, sum * this->invInnerSize);

            AscendC::SetFlag<AscendC::HardEvent::S_V>(evtSV);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(evtSV);
        }

        AscendC::Tanh<float>(resultsLocal, resultsLocal, this->tasksPerCore);

        int32_t evtVM = static_cast<int32_t>(
            AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_MTE3));
        AscendC::SetFlag<AscendC::HardEvent::V_MTE3>(evtVM);
        AscendC::WaitFlag<AscendC::HardEvent::V_MTE3>(evtVM);

        AscendC::DataCopy(yGm, resultsLocal, this->tasksPerCore);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> resultsBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t totalTasks;
    uint32_t innerSize;
    uint32_t numTasks;
    uint32_t startTask;
    uint32_t tasksPerCore;
    uint32_t tileLength;
    uint32_t tileNum;
    float invInnerSize;
    float hardtanhMin;
    float hardtanhMax;
};

extern "C" __global__ __aicore__ void conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedTail op;
    op.Init(x, y,
            tiling_data.totalTasks, tiling_data.innerSize,
            tiling_data.tasksPerCore, tiling_data.tileLength,
            tiling_data.tileNum, tiling_data.invInnerSize,
            tiling_data.hardtanhMin, tiling_data.hardtanhMax);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding,
                 maxpool_kernel_size, maxpool_stride, hardtanh_min, hardtanh_max):
        super().__init__()
        self.conv_transpose = nn.ConvTranspose2d(in_channels, out_channels, kernel_size,
                                                 stride=stride, padding=padding)
        self.maxpool = nn.MaxPool2d(kernel_size=maxpool_kernel_size, stride=maxpool_stride)
        self.hardtanh_min = float(hardtanh_min)
        self.hardtanh_max = float(hardtanh_max)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.maxpool(x)
        x = x.contiguous()
        x = custom_ops_lib.conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(
            x, self.hardtanh_min, self.hardtanh_max)
        return x
