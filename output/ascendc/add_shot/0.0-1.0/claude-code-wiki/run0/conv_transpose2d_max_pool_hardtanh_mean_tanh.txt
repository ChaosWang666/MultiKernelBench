kernel_src = """
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelHardtanhMeanTanh {
public:
    __aicore__ inline KernelHardtanhMeanTanh() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t totalSamples, uint32_t reduceLength,
                                 uint32_t tileLength, float minVal, float maxVal)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->samplesPerBlock = totalSamples / blockNum;
        this->reduceLength = reduceLength;
        this->tileLength = tileLength;
        this->tileNum = reduceLength / tileLength;
        this->minVal = minVal;
        this->maxVal = maxVal;
        this->invReduce = 1.0f / (float)reduceLength;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockIdx * this->samplesPerBlock * reduceLength,
                             this->samplesPerBlock * reduceLength);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + blockIdx * this->samplesPerBlock,
                             this->samplesPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(accumBuf, tileLength * sizeof(float));
        pipe.InitBuffer(workBuf, tileLength * sizeof(float));
        pipe.InitBuffer(outBuf, 2048 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> means = outBuf.Get<float>();
        AscendC::LocalTensor<float> accum = accumBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();

        for (uint32_t s = 0; s < samplesPerBlock; s++) {
            AscendC::Duplicate(accum, 0.0f, tileLength);
            uint32_t baseOffset = s * reduceLength;

            for (uint32_t t = 0; t < tileNum; t++) {
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(xLocal, xGm[baseOffset + t * tileLength], tileLength);
                inQueueX.EnQue(xLocal);

                AscendC::LocalTensor<DTYPE_X> xIn = inQueueX.DeQue<DTYPE_X>();
                AscendC::Maxs(xIn, xIn, (DTYPE_X)minVal, tileLength);
                AscendC::Mins(xIn, xIn, (DTYPE_X)maxVal, tileLength);
                AscendC::Add(accum, accum, xIn, tileLength);
                inQueueX.FreeTensor(xIn);
            }

            AscendC::ReduceSum<float>(work, accum, work, tileLength);
            AscendC::PipeBarrier<PIPE_V>();
            float meanVal = work.GetValue(0) * invReduce;
            means.SetValue(s, meanVal);
        }

        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Muls(accum, means, 2.0f, samplesPerBlock);
        AscendC::Exp(accum, accum, samplesPerBlock);
        AscendC::Adds(work, accum, 1.0f, samplesPerBlock);
        AscendC::Adds(accum, accum, -1.0f, samplesPerBlock);
        AscendC::Div(means, accum, work, samplesPerBlock);

        AscendC::DataCopy(yGm, means, samplesPerBlock);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t samplesPerBlock;
    uint32_t reduceLength;
    uint32_t tileLength;
    uint32_t tileNum;
    float minVal;
    float maxVal;
    float invReduce;
};

extern "C" __global__ __aicore__ void conv_transpose2d_max_pool_hardtanh_mean_tanh_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelHardtanhMeanTanh op;
    op.Init(x, y, tiling_data.totalSamples, tiling_data.reduceLength,
            tiling_data.tileLength, tiling_data.minVal, tiling_data.maxVal);
    op.Process();
}
