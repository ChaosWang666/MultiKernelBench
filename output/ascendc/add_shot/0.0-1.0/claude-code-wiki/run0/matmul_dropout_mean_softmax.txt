kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelSoftmaxLastDim {
public:
    __aicore__ inline KernelSoftmaxLastDim() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t features)
    {
        this->batchSize = batchSize;
        this->features = features;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        uint32_t rowStart = blockIdx * rowsPerBlock;
        uint32_t rowEnd = rowStart + rowsPerBlock;
        if (rowEnd > batchSize) rowEnd = batchSize;

        this->myRowStart = rowStart;
        this->myRowCount = (rowEnd > rowStart) ? (rowEnd - rowStart) : 0;

        xGm.SetGlobalBuffer((__gm__ float*)x + this->myRowStart * features, this->myRowCount * features);
        yGm.SetGlobalBuffer((__gm__ float*)y + this->myRowStart * features, this->myRowCount * features);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, features * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, features * sizeof(float));
        pipe.InitBuffer(workBuf, 2048);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->myRowCount; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * features], features);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();

        AscendC::ReduceMax<float>(work, xLocal, work, features, false);
        float maxVal = work.GetValue(0);

        AscendC::Adds(yLocal, xLocal, -maxVal, features);

        AscendC::Exp(yLocal, yLocal, features);

        AscendC::ReduceSum<float>(work, yLocal, work, features);
        float sumVal = work.GetValue(0);
        float invSum = 1.0f / sumVal;

        AscendC::Muls(yLocal, yLocal, invSum, features);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[row * features], yLocal, features);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<float> xGm, yGm;
    uint32_t batchSize;
    uint32_t features;
    uint32_t myRowStart;
    uint32_t myRowCount;
};

extern "C" __global__ __aicore__ void matmul_dropout_mean_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSoftmaxLastDim op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.features);
    op.Process();
}

class ModelNew(torch.nn.Module):

        self.matmul = torch.nn.Linear(in_features, out_features)
