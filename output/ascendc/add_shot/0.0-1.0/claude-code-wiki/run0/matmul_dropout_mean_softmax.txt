kernel_src="""
#include "kernel_operator.h"

class KernelSoftmaxRow {
public:
    __aicore__ inline KernelSoftmaxRow() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalRows, uint32_t featureSize, uint32_t rowsPerBlock)
    {
        this->featureSize = featureSize;
        this->rowsPerBlock = rowsPerBlock;
        uint32_t startRow = AscendC::GetBlockIdx() * rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startRow * featureSize, rowsPerBlock * featureSize);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + startRow * featureSize, rowsPerBlock * featureSize);

        pipe.InitBuffer(inQueueX, 1, featureSize * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, 1, featureSize * sizeof(DTYPE_Z));
        pipe.InitBuffer(workBuf, 2048);
        pipe.InitBuffer(redBuf, 64);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsPerBlock; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * this->featureSize], this->featureSize);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> red = redBuf.Get<float>();

        AscendC::ReduceMax<float>(red, xLocal, work, this->featureSize, false);

        AscendC::PipeBarrier<PIPE_ALL>();
        float maxVal = red.GetValue(0);
        float negMax = -maxVal;
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Adds(zLocal, xLocal, negMax, this->featureSize);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Exp(zLocal, zLocal, this->featureSize);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::ReduceSum<float>(red, zLocal, work, this->featureSize);

        AscendC::PipeBarrier<PIPE_ALL>();
        float sumVal = red.GetValue(0);
        float invSum = 1.0f / sumVal;
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Muls(zLocal, zLocal, invSum, this->featureSize);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * this->featureSize], zLocal, this->featureSize);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> redBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t featureSize;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void matmul_dropout_mean_softmax_custom(GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSoftmaxRow op;
    op.Init(x, z, tiling_data.totalRows, tiling_data.featureSize, tiling_data.rowsPerBlock);
    op.Process();
}
