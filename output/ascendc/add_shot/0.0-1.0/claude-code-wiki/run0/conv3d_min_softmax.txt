kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr float POS_LARGE = 1.0e30f;
constexpr float NEG_LARGE = -1.0e30f;

class KernelConv3dMinSoftmax {
public:
    __aicore__ inline KernelConv3dMinSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t dSize, uint32_t hwSize,
                                 uint32_t hwAligned, uint32_t rightPadding)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->dSize = dSize;
        this->hwSize = hwSize;
        this->hwAligned = hwAligned;
        this->rightPadding = rightPadding;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t xOffset = (uint64_t)blockIdx * channels * dSize * hwSize;
        uint64_t yOffset = (uint64_t)blockIdx * channels * hwSize;

        xGm.SetGlobalBuffer((__gm__ float*)x + xOffset, (uint64_t)channels * dSize * hwSize);
        yGm.SetGlobalBuffer((__gm__ float*)y + yOffset, (uint64_t)channels * hwSize);

        pipe.InitBuffer(inQueue, BUFFER_NUM, hwAligned * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, hwAligned * sizeof(float));
        pipe.InitBuffer(reducedBuf, channels * hwAligned * sizeof(float));
        pipe.InitBuffer(maxBuf, hwAligned * sizeof(float));
        pipe.InitBuffer(sumBuf, hwAligned * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        ReduceMinOverD();
        SoftmaxOverC();
    }

private:
    __aicore__ inline void ReduceMinOverD()
    {
        AscendC::LocalTensor<float> reducedData = reducedBuf.Get<float>();

        for (uint32_t c = 0; c < channels; c++) {
            AscendC::LocalTensor<float> minSlice = reducedData[c * hwAligned];
            AscendC::Duplicate<float>(minSlice, POS_LARGE, hwAligned);

            for (uint32_t d = 0; d < dSize; d++) {
                AscendC::LocalTensor<float> in = inQueue.AllocTensor<float>();
                uint64_t offset = (uint64_t)c * dSize * hwSize + (uint64_t)d * hwSize;
                AscendC::DataCopyExtParams copyParams{1, (uint32_t)(hwSize * sizeof(float)), 0, 0, 0};
                AscendC::DataCopyPadExtParams<float> padParams{true, 0, (uint8_t)rightPadding, POS_LARGE};
                AscendC::DataCopyPad(in, xGm[offset], copyParams, padParams);
                inQueue.EnQue(in);

                AscendC::LocalTensor<float> inDq = inQueue.DeQue<float>();
                AscendC::Min(minSlice, minSlice, inDq, hwAligned);
                inQueue.FreeTensor(inDq);
            }
        }
    }

    __aicore__ inline void SoftmaxOverC()
    {
        AscendC::LocalTensor<float> reducedData = reducedBuf.Get<float>();
        AscendC::LocalTensor<float> maxData = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumData = sumBuf.Get<float>();

        AscendC::Duplicate<float>(maxData, NEG_LARGE, hwAligned);
        for (uint32_t c = 0; c < channels; c++) {
            AscendC::Max(maxData, maxData, reducedData[c * hwAligned], hwAligned);
        }

        AscendC::Duplicate<float>(sumData, 0.0f, hwAligned);
        for (uint32_t c = 0; c < channels; c++) {
            AscendC::LocalTensor<float> slice = reducedData[c * hwAligned];
            AscendC::Sub(slice, slice, maxData, hwAligned);
            AscendC::Exp(slice, slice, hwAligned);
            AscendC::Add(sumData, sumData, slice, hwAligned);
        }

        AscendC::Reciprocal(sumData, sumData, hwAligned);

        for (uint32_t c = 0; c < channels; c++) {
            AscendC::LocalTensor<float> out = outQueue.AllocTensor<float>();
            AscendC::Mul(out, reducedData[c * hwAligned], sumData, hwAligned);
            outQueue.EnQue<float>(out);

            AscendC::LocalTensor<float> outDq = outQueue.DeQue<float>();
            AscendC::DataCopyExtParams copyOutParams{1, (uint32_t)(hwSize * sizeof(float)), 0, 0, 0};
            AscendC::DataCopyPad(yGm[(uint64_t)c * hwSize], outDq, copyOutParams);
            outQueue.FreeTensor(outDq);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reducedBuf, maxBuf, sumBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t dSize;
    uint32_t hwSize;
    uint32_t hwAligned;
    uint32_t rightPadding;
};

extern "C" __global__ __aicore__ void conv3d_min_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dMinSoftmax op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.channels, tiling_data.dSize,
            tiling_data.hwSize, tiling_data.hwAligned, tiling_data.rightPadding);
    op.Process();
}
