kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr float NEGATIVE_SLOPE = 0.2f;
constexpr float CLAMP_MIN = -1.0f;
constexpr float CLAMP_MAX = 1.0f;

class KernelConv3dLeakyReluSumClampGelu {
public:
    __aicore__ inline KernelConv3dLeakyReluSumClampGelu() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                uint32_t totalChannels, uint32_t channelSize,
                                uint32_t numChannels, uint32_t tileLength)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t baseCount = totalChannels / blockNum;
        uint32_t remain = totalChannels % blockNum;
        uint32_t startCh;
        uint32_t cntCh;
        if (blockIdx < remain) {
            startCh = blockIdx * (baseCount + 1);
            cntCh = baseCount + 1;
        } else {
            startCh = blockIdx * baseCount + remain;
            cntCh = baseCount;
        }

        this->startChannel = startCh;
        this->channelCount = cntCh;
        this->channelSize = channelSize;
        this->numChannels = numChannels;
        this->tileLength = tileLength;

        uint64_t totalElems = (uint64_t)totalChannels * (uint64_t)channelSize;
        xGm.SetGlobalBuffer((__gm__ float*)x, totalElems);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalElems);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, numChannels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->channelCount; i++) {
            uint32_t globalCh = this->startChannel + i;
            uint32_t c = globalCh % this->numChannels;
            float biasVal = biasGm.GetValue(c);

            uint32_t numTiles = (this->channelSize + this->tileLength - 1) / this->tileLength;
            for (uint32_t t = 0; t < numTiles; t++) {
                uint64_t offset = (uint64_t)globalCh * (uint64_t)this->channelSize
                                + (uint64_t)t * (uint64_t)this->tileLength;
                uint32_t curLen = this->tileLength;
                uint32_t processed = t * this->tileLength;
                if (processed + this->tileLength > this->channelSize) {
                    curLen = this->channelSize - processed;
                }

                CopyIn(offset, curLen);
                Compute(biasVal, curLen);
                CopyOut(offset, curLen);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], len);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float biasVal, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        AscendC::LeakyRelu(yLocal, xLocal, (float)NEGATIVE_SLOPE, len);
        AscendC::Adds(yLocal, yLocal, biasVal, len);
        AscendC::Clip(yLocal, yLocal, (float)CLAMP_MIN, (float)CLAMP_MAX, len);
        AscendC::Gelu(yLocal, yLocal, len);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint64_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[offset], yLocal, len);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t startChannel;
    uint32_t channelCount;
    uint32_t channelSize;
    uint32_t numChannels;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void conv3d_leaky_relu_sum_clamp_gelu_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dLeakyReluSumClampGelu op;
    op.Init(x, bias, y,
            tiling_data.totalChannels, tiling_data.channelSize,
            tiling_data.numChannels, tiling_data.tileLength);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, sum_tensor_shape):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.sum_tensor = nn.Parameter(torch.randn(sum_tensor_shape))

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
        bias_flat = self.sum_tensor.reshape(-1).contiguous()
        return custom_ops_lib.conv3d_leaky_relu_sum_clamp_gelu_custom(x, bias_flat)
