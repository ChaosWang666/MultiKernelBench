        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t channelsPerCore = totalChannels / blockNum;
        uint32_t extraChannels = totalChannels % blockNum;

        if (blockIdx < extraChannels) {
            this->myChannels = channelsPerCore + 1;
            this->startChannel = blockIdx * (channelsPerCore + 1);

        AscendC::DataCopy(inLocal[3 * Waligned],       xGm[baseOff + H * W + (h0 + 1) * W], Waligned);
        inQueue.EnQue(inLocal);

    {
