kernel_src="""
#include "kernel_operator.h"

using namespace AscendC;

class KernelBmmInNormAddMul {
public:
    __aicore__ inline KernelBmmInNormAddMul() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                 uint32_t totalRows, uint32_t colSize, float eps)
    {
        this->colSize = colSize;
        this->eps = eps;

        uint32_t blockId = GetBlockIdx();
        uint32_t blockNum = GetBlockNum();
        uint32_t rowsPerBlock = totalRows / blockNum;
        uint32_t tailRows = totalRows % blockNum;

        if (blockId < tailRows) {
            this->rowsForBlock = rowsPerBlock + 1;
            this->rowOffset = blockId * this->rowsForBlock;
        } else {
            this->rowsForBlock = rowsPerBlock;
            this->rowOffset = blockId * rowsPerBlock + tailRows;
        }

        xGm.SetGlobalBuffer((__gm__ float*)x + this->rowOffset * colSize,
                            this->rowsForBlock * colSize);
        yGm.SetGlobalBuffer((__gm__ float*)y + this->rowOffset * colSize,
                            this->rowsForBlock * colSize);
        zGm.SetGlobalBuffer((__gm__ float*)z + this->rowOffset * colSize,
                            this->rowsForBlock * colSize);

        pipe.InitBuffer(inQueueX, 1, colSize * sizeof(float));
        pipe.InitBuffer(inQueueY, 1, colSize * sizeof(float));
        pipe.InitBuffer(outQueueZ, 1, colSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, colSize * sizeof(float));
        pipe.InitBuffer(reduceBuf, 32 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->rowsForBlock; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx)
    {
        LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        LocalTensor<float> yLocal = inQueueY.AllocTensor<float>();
        DataCopy(xLocal, xGm[rowIdx * this->colSize], this->colSize);
        DataCopy(yLocal, yGm[rowIdx * this->colSize], this->colSize);
        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void Compute(uint32_t rowIdx)
    {
        LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        LocalTensor<float> yLocal = inQueueY.DeQue<float>();
        LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        ReduceSum<float>(reduceLocal, xLocal, tmpLocal, this->colSize);
        event_t evt1 = static_cast<event_t>(GetTPipePtr()->FetchEventID(HardEvent::V_S));
        SetFlag<HardEvent::V_S>(evt1);
        WaitFlag<HardEvent::V_S>(evt1);
        float sum = reduceLocal.GetValue(0);
        float mean = sum / static_cast<float>(this->colSize);

        Adds(tmpLocal, xLocal, -mean, this->colSize);
        PipeBarrier<PIPE_V>();

        Mul(xLocal, tmpLocal, tmpLocal, this->colSize);
        PipeBarrier<PIPE_V>();

        ReduceSum<float>(reduceLocal, xLocal, xLocal, this->colSize);
        event_t evt2 = static_cast<event_t>(GetTPipePtr()->FetchEventID(HardEvent::V_S));
        SetFlag<HardEvent::V_S>(evt2);
        WaitFlag<HardEvent::V_S>(evt2);
        float sqSum = reduceLocal.GetValue(0);
        float var = sqSum / static_cast<float>(this->colSize);
        float invStd = 1.0f / sqrtf(var + this->eps);

        Muls(tmpLocal, tmpLocal, invStd, this->colSize);
        PipeBarrier<PIPE_V>();

        Add(tmpLocal, tmpLocal, yLocal, this->colSize);
        PipeBarrier<PIPE_V>();

        Mul(zLocal, tmpLocal, yLocal, this->colSize);

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx)
    {
        LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        DataCopy(zGm[rowIdx * this->colSize], zLocal, this->colSize);
        outQueueZ.FreeTensor(zLocal);
    }

    TPipe pipe;
    TQue<TPosition::VECIN, 1> inQueueX, inQueueY;
    TQue<TPosition::VECOUT, 1> outQueueZ;
    TBuf<TPosition::VECCALC> tmpBuf, reduceBuf;
    GlobalTensor<float> xGm, yGm, zGm;
    uint32_t colSize;
    uint32_t rowsForBlock;
    uint32_t rowOffset;
    float eps;
};

extern "C" __global__ __aicore__ void bmm_instance_norm_sum_residual_add_multiply_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelBmmInNormAddMul op;
    op.Init(x, y, z, tiling_data.totalRows, tiling_data.colSize, tiling_data.eps);
    op.Process();
}
