kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR y, GM_ADDR bias, GM_ADDR z,
                                uint32_t totalLength, uint32_t channelSize,
                                uint32_t numChannels, uint32_t tileLength)
    {
        this->channelSize = channelSize;
        this->numChannels = numChannels;
        this->tileLength = tileLength;

        uint32_t totalChannelBlocks = totalLength / channelSize;
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->channelsPerBlock = totalChannelBlocks / blockNum;
        this->startChBlock = blockIdx * this->channelsPerBlock;
        this->numTilesPerChannel = channelSize / tileLength;

        yGm.SetGlobalBuffer((__gm__ float *)y + this->startChBlock * channelSize,
                            this->channelsPerBlock * channelSize);
        zGm.SetGlobalBuffer((__gm__ float *)z + this->startChBlock * channelSize,
                            this->channelsPerBlock * channelSize);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, numChannels);

        pipe.InitBuffer(biasQue, 1, ((numChannels * sizeof(float) + 31) / 32) * 32);
        pipe.InitBuffer(inQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasLocal = biasQue.AllocTensor<float>();
        AscendC::DataCopy(biasLocal, biasGm, numChannels);
        biasQue.EnQue(biasLocal);
        biasLocal = biasQue.DeQue<float>();

        for (uint32_t ch = 0; ch < channelsPerBlock; ch++) {
            uint32_t globalCh = (startChBlock + ch) % numChannels;
            float biasVal = biasLocal.GetValue(globalCh);
            float biasPlusOne = biasVal + 1.0f;

            for (uint32_t t = 0; t < numTilesPerChannel; t++) {
                uint32_t offset = ch * channelSize + t * tileLength;
                CopyIn(offset);
                Compute(biasPlusOne);
                CopyOut(offset);
            }
        }

        biasQue.FreeTensor(biasLocal);
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset)
    {
        AscendC::LocalTensor<float> yLocal = inQueueY.AllocTensor<float>();
        AscendC::DataCopy(yLocal, yGm[offset], tileLength);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void Compute(float biasPlusOne)
    {
        AscendC::LocalTensor<float> yLocal = inQueueY.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();

        AscendC::Muls(zLocal, yLocal, 2.0f, tileLength);
        AscendC::Adds(zLocal, zLocal, biasPlusOne, tileLength);
        AscendC::Mul(zLocal, yLocal, zLocal, tileLength);

        outQueueZ.EnQue<float>(zLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[offset], zLocal, tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> biasQue;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t channelSize;
    uint32_t numChannels;
    uint32_t tileLength;
    uint32_t channelsPerBlock;
    uint32_t startChBlock;
    uint32_t numTilesPerChannel;
};

extern "C" __global__ __aicore__ void conv_transpose3d_sum_residual_add_multiply_residual_add_custom(
    GM_ADDR y, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(y, bias, z, tiling_data.totalLength, tiling_data.channelSize,
            tiling_data.numChannels, tiling_data.tileLength);
    op.Process();
}
