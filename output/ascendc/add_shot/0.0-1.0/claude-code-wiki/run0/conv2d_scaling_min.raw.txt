kernel_src="""
#include "kernel_operator.h"
using namespace AscendC;

constexpr int32_t IN_CHANNELS = 64;
constexpr int32_t OUT_CHANNELS = 128;
constexpr int32_t KERNEL_H = 3;
constexpr int32_t KERNEL_W = 3;
constexpr int32_t H_IN = 256;
constexpr int32_t W_IN = 256;
constexpr int32_t H_OUT = 254;
constexpr int32_t W_OUT = 254;
constexpr int32_t W_CHUNK = 128;
constexpr int32_t W_INPUT_STRIDE = 136;

class KernelConv2dScalingMin {
public:
    __aicore__ inline KernelConv2dScalingMin() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR y,
                                  uint32_t batchSize, float scaleFactor)
    {
        this->batchSize = batchSize;
        this->scaleFactor = scaleFactor;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        weightGm.SetGlobalBuffer((__gm__ float*)weight);
        biasGm.SetGlobalBuffer((__gm__ float*)bias);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        pipe.InitBuffer(xBuf, IN_CHANNELS * KERNEL_H * W_INPUT_STRIDE * sizeof(float));
        pipe.InitBuffer(weightChBuf, IN_CHANNELS * KERNEL_H * KERNEL_W * sizeof(float));
        pipe.InitBuffer(biasBuf, OUT_CHANNELS * sizeof(float));
        pipe.InitBuffer(accBuf, W_CHUNK * sizeof(float));
        pipe.InitBuffer(minBuf, W_CHUNK * sizeof(float));
        pipe.InitBuffer(tmpBuf, W_CHUNK * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        LocalTensor<float> biasLocal = biasBuf.Get<float>();
        DataCopy(biasLocal, biasGm, OUT_CHANNELS);
        SetFlag<HardEvent::MTE2_S>(EVENT_ID0);
        WaitFlag<HardEvent::MTE2_S>(EVENT_ID0);

        int32_t blockIdx = static_cast<int32_t>(GetBlockIdx());
        int32_t blockNum = static_cast<int32_t>(GetBlockNum());
        int32_t totalRows = static_cast<int32_t>(batchSize) * H_OUT;
        int32_t rowsPerBlock = (totalRows + blockNum - 1) / blockNum;
        int32_t startRow = blockIdx * rowsPerBlock;
        int32_t endRow = startRow + rowsPerBlock;
        if (endRow > totalRows) endRow = totalRows;

        for (int32_t row = startRow; row < endRow; row++) {
            int32_t b = row / H_OUT;
            int32_t h_out = row % H_OUT;
            ProcessRow(b, h_out, biasLocal);
        }
    }

private:
    __aicore__ inline void ProcessRow(int32_t b, int32_t h_out, LocalTensor<float>& biasLocal)
    {
        LocalTensor<float> xLocal = xBuf.Get<float>();
        LocalTensor<float> wLocal = weightChBuf.Get<float>();
        LocalTensor<float> accLocal = accBuf.Get<float>();
        LocalTensor<float> minLocal = minBuf.Get<float>();
        LocalTensor<float> tmpLocal = tmpBuf.Get<float>();

        for (int32_t w_start = 0; w_start < W_OUT; w_start += W_CHUNK) {
            int32_t w_count = (w_start + W_CHUNK <= W_OUT) ? W_CHUNK : (W_OUT - w_start);
            int32_t input_cols_needed = w_count + KERNEL_W - 1;
            int32_t input_cols_load = (input_cols_needed + 7) / 8 * 8;
            if (w_start + input_cols_load > W_IN) {
                input_cols_load = W_IN - w_start;
            }

            for (int32_t ic = 0; ic < IN_CHANNELS; ic++) {
                for (int32_t kh = 0; kh < KERNEL_H; kh++) {
                    uint64_t srcOffset = (uint64_t)b * IN_CHANNELS * H_IN * W_IN +
                                          (uint64_t)ic * H_IN * W_IN +
                                          (uint64_t)(h_out + kh) * W_IN +
                                          (uint64_t)w_start;
                    int32_t dstOffset = (ic * KERNEL_H + kh) * W_INPUT_STRIDE;
                    DataCopy(xLocal[dstOffset], xGm[srcOffset], input_cols_load);
                }
            }

            SetFlag<HardEvent::MTE2_V>(EVENT_ID1);
            WaitFlag<HardEvent::MTE2_V>(EVENT_ID1);

            Duplicate(minLocal, (float)3.4e38f, w_count);

            for (int32_t c = 0; c < OUT_CHANNELS; c++) {
                uint64_t weightOffset = (uint64_t)c * IN_CHANNELS * KERNEL_H * KERNEL_W;
                DataCopy(wLocal, weightGm[weightOffset], IN_CHANNELS * KERNEL_H * KERNEL_W);

                SetFlag<HardEvent::MTE2_S>(EVENT_ID2);
                WaitFlag<HardEvent::MTE2_S>(EVENT_ID2);

                float biasVal = biasLocal.GetValue(c);
                Duplicate(accLocal, biasVal, w_count);

                for (int32_t ic = 0; ic < IN_CHANNELS; ic++) {
                    for (int32_t kh = 0; kh < KERNEL_H; kh++) {
                        for (int32_t kw = 0; kw < KERNEL_W; kw++) {
                            float w_val = wLocal.GetValue(ic * KERNEL_H * KERNEL_W + kh * KERNEL_W + kw);
                            int32_t xOffset = (ic * KERNEL_H + kh) * W_INPUT_STRIDE + kw;
                            Muls(tmpLocal, xLocal[xOffset], w_val, w_count);
                            Add(accLocal, accLocal, tmpLocal, w_count);
                        }
                    }
                }

                Min(minLocal, minLocal, accLocal, w_count);
            }

            Muls(minLocal, minLocal, scaleFactor, w_count);

            SetFlag<HardEvent::V_MTE3>(EVENT_ID3);
            WaitFlag<HardEvent::V_MTE3>(EVENT_ID3);

            uint64_t outOffset = (uint64_t)b * H_OUT * W_OUT +
                                  (uint64_t)h_out * W_OUT +
                                  (uint64_t)w_start;

            DataCopyExtParams copyParams;
            copyParams.blockCount = 1;
            copyParams.blockLen = (uint32_t)(w_count * sizeof(float));
            copyParams.srcStride = 0;
            copyParams.dstStride = 0;
            DataCopyPad(yGm[outOffset], minLocal, copyParams);

            SetFlag<HardEvent::MTE3_V>(EVENT_ID0);
            WaitFlag<HardEvent::MTE3_V>(EVENT_ID0);
            SetFlag<HardEvent::MTE3_MTE2>(EVENT_ID1);
            WaitFlag<HardEvent::MTE3_MTE2>(EVENT_ID1);
        }
    }

private:
    TPipe pipe;
    TBuf<TPosition::VECCALC> xBuf, weightChBuf, biasBuf, accBuf, minBuf, tmpBuf;
    GlobalTensor<float> xGm, weightGm, biasGm, yGm;
    uint32_t batchSize;
    float scaleFactor;
};

extern "C" __global__ __aicore__ void conv2d_scaling_min_custom(GM_ADDR x, GM_ADDR weight, GM_ADDR bias,
                                                                  GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dScalingMin op;
    op.Init(x, weight, bias, y, tiling_data.batchSize, tiling_data.scaleFactor);
    op.Process();
}

#include <torch/library.h>

#include "pytorch_npu_helper.hpp"
#include <torch/extension.h>

at::Tensor conv2d_scaling_min_custom_impl_npu(const at::Tensor& x, const at::Tensor& weight,
                                                const at::Tensor& bias, double scale_factor) {
    at::Tensor x_c = x.contiguous();
    at::Tensor w_c = weight.contiguous();
    at::Tensor b_c = bias.contiguous();
    auto x_sizes = x_c.sizes();
    auto w_sizes = w_c.sizes();
    int64_t N = x_sizes[0];
    int64_t H = x_sizes[2];
    int64_t W = x_sizes[3];
    int64_t Kh = w_sizes[2];
    int64_t Kw = w_sizes[3];
    int64_t Hout = H - Kh + 1;
    int64_t Wout = W - Kw + 1;
    at::Tensor result = at::empty({N, 1, Hout, Wout}, x_c.options());
    float scale = static_cast<float>(scale_factor);
    EXEC_NPU_CMD(aclnnConv2dScalingMinCustom, x_c, w_c, b_c, scale, result);
    return result;
}
