kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose3dSoftmaxSigmoid {
public:
    __aicore__ inline KernelConvTranspose3dSoftmaxSigmoid() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalBatches, uint32_t channels, uint32_t spatialSize,
                                uint32_t spatialPerCore, uint32_t tileSize, uint32_t tilesPerBatchPerCore)
    {
        this->N = totalBatches;
        this->C = channels;
        this->spatial = spatialSize;
        this->spatialPerCore = spatialPerCore;
        this->tileSize = tileSize;
        this->tilesPerBatchPerCore = tilesPerBatchPerCore;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->coreSpatialStart = blockIdx * spatialPerCore;

        uint32_t totalElems = N * C * spatial;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, totalElems);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y, totalElems);

        pipe.InitBuffer(inQueue, BUFFER_NUM, tileSize * channels * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueue, BUFFER_NUM, tileSize * channels * sizeof(DTYPE_Y));
        pipe.InitBuffer(maxBuf, tileSize * sizeof(float));
        pipe.InitBuffer(sumBuf, tileSize * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t n = 0; n < N; n++) {
            for (uint32_t t = 0; t < tilesPerBatchPerCore; t++) {
                uint32_t sStart = coreSpatialStart + t * tileSize;
                CopyIn(n, sStart);
                Compute();
                CopyOut(n, sStart);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t n, uint32_t sStart)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueue.AllocTensor<DTYPE_X>();
        uint32_t baseOffset = n * C * spatial + sStart;
        for (uint32_t c = 0; c < C; c++) {
            AscendC::DataCopy(xLocal[c * tileSize], xGm[baseOffset + c * spatial], tileSize);
        }
        inQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueue.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueue.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();

        uint32_t total = C * tileSize;

        AscendC::Adds(maxLocal, xLocal, (float)0.0, tileSize);
        for (uint32_t c = 1; c < C; c++) {
            AscendC::Max(maxLocal, maxLocal, xLocal[c * tileSize], tileSize);
        }

        for (uint32_t c = 0; c < C; c++) {
            AscendC::Sub(xLocal[c * tileSize], xLocal[c * tileSize], maxLocal, tileSize);
        }

        AscendC::Exp(xLocal, xLocal, total);

        AscendC::Adds(sumLocal, xLocal, (float)0.0, tileSize);
        for (uint32_t c = 1; c < C; c++) {
            AscendC::Add(sumLocal, sumLocal, xLocal[c * tileSize], tileSize);
        }

        AscendC::Reciprocal(sumLocal, sumLocal, tileSize);

        for (uint32_t c = 0; c < C; c++) {
            AscendC::Mul(xLocal[c * tileSize], xLocal[c * tileSize], sumLocal, tileSize);
        }

        AscendC::Muls(yLocal, xLocal, (float)-1.0, total);
        AscendC::Exp(yLocal, yLocal, total);
        AscendC::Adds(yLocal, yLocal, (float)1.0, total);
        AscendC::Reciprocal(yLocal, yLocal, total);

        outQueue.EnQue<DTYPE_Y>(yLocal);
        inQueue.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t n, uint32_t sStart)
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueue.DeQue<DTYPE_Y>();
        uint32_t baseOffset = n * C * spatial + sStart;
        for (uint32_t c = 0; c < C; c++) {
            AscendC::DataCopy(yGm[baseOffset + c * spatial], yLocal[c * tileSize], tileSize);
        }
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t N;
    uint32_t C;
    uint32_t spatial;
    uint32_t spatialPerCore;
    uint32_t tileSize;
    uint32_t tilesPerBatchPerCore;
    uint32_t coreSpatialStart;
};

extern "C" __global__ __aicore__ void conv_transpose3d_softmax_sigmoid_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dSoftmaxSigmoid op;
    op.Init(x, y,
            tiling_data.totalBatches, tiling_data.channels, tiling_data.spatialSize,
            tiling_data.spatialPerCore, tiling_data.tileSize, tiling_data.tilesPerBatchPerCore);
    op.Process();
}
