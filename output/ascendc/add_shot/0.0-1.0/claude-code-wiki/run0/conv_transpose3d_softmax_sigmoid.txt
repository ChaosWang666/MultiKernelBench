kernel_src="""
#include "kernel_operator.h"

constexpr uint32_t TILE_LEN = 256;

class KernelConvTranspose3dSoftmaxSigmoid {
public:
    __aicore__ inline KernelConvTranspose3dSoftmaxSigmoid() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t batchSize, uint32_t channels, uint32_t spatialSize)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->spatialSize = spatialSize;

        uint32_t tilesPerBatch = spatialSize / TILE_LEN;
        uint32_t tailLen = spatialSize - tilesPerBatch * TILE_LEN;
        if (tailLen > 0) {
            tilesPerBatch += 1;
        }
        this->tilesPerBatch = tilesPerBatch;
        this->tailLen = tailLen;

        uint32_t totalTiles = batchSize * tilesPerBatch;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t tilesPerCore = totalTiles / blockNum;
        uint32_t remainder = totalTiles % blockNum;

        if (blockIdx < remainder) {
            this->startTile = blockIdx * (tilesPerCore + 1);
            this->endTile = this->startTile + tilesPerCore + 1;
        } else {
            this->startTile = remainder * (tilesPerCore + 1) + (blockIdx - remainder) * tilesPerCore;
            this->endTile = this->startTile + tilesPerCore;
        }

        uint64_t totalElems = (uint64_t)batchSize * channels * spatialSize;
        xGm.SetGlobalBuffer((__gm__ float*)x, totalElems);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalElems);

        pipe.InitBuffer(dataBuf, channels * TILE_LEN * sizeof(float));
        pipe.InitBuffer(maxBuf, TILE_LEN * sizeof(float));
        pipe.InitBuffer(sumBuf, TILE_LEN * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> dataLocal = dataBuf.Get<float>();
        AscendC::LocalTensor<float> maxRow = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumRow = sumBuf.Get<float>();

        for (uint32_t t = startTile; t < endTile; t++) {
            uint32_t n = t / tilesPerBatch;
            uint32_t tileIdx = t % tilesPerBatch;
            uint32_t s0 = tileIdx * TILE_LEN;
            uint32_t curLen = TILE_LEN;
            if (tailLen > 0 && tileIdx == tilesPerBatch - 1) {
                curLen = tailLen;
            }
            uint64_t baseOffset = (uint64_t)n * channels * spatialSize + s0;

            for (uint32_t c = 0; c < channels; c++) {
                AscendC::DataCopy(dataLocal[c * TILE_LEN],
                                  xGm[baseOffset + (uint64_t)c * spatialSize],
                                  curLen);
            }
            AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

            AscendC::Adds(maxRow, dataLocal, (float)0.0f, curLen);
            for (uint32_t c = 1; c < channels; c++) {
                AscendC::Max(maxRow, maxRow, dataLocal[c * TILE_LEN], curLen);
            }

            for (uint32_t c = 0; c < channels; c++) {
                AscendC::Sub(dataLocal[c * TILE_LEN], dataLocal[c * TILE_LEN], maxRow, curLen);
                AscendC::Exp(dataLocal[c * TILE_LEN], dataLocal[c * TILE_LEN], curLen);
            }

            AscendC::Duplicate(sumRow, (float)0.0f, curLen);
            for (uint32_t c = 0; c < channels; c++) {
                AscendC::Add(sumRow, sumRow, dataLocal[c * TILE_LEN], curLen);
            }

            AscendC::Reciprocal(sumRow, sumRow, curLen);

            for (uint32_t c = 0; c < channels; c++) {
                AscendC::Mul(dataLocal[c * TILE_LEN], dataLocal[c * TILE_LEN], sumRow, curLen);
                AscendC::Muls(dataLocal[c * TILE_LEN], dataLocal[c * TILE_LEN], (float)-1.0f, curLen);
                AscendC::Exp(dataLocal[c * TILE_LEN], dataLocal[c * TILE_LEN], curLen);
                AscendC::Adds(dataLocal[c * TILE_LEN], dataLocal[c * TILE_LEN], (float)1.0f, curLen);
                AscendC::Reciprocal(dataLocal[c * TILE_LEN], dataLocal[c * TILE_LEN], curLen);
            }

            AscendC::SetFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);

            for (uint32_t c = 0; c < channels; c++) {
                AscendC::DataCopy(yGm[baseOffset + (uint64_t)c * spatialSize],
                                  dataLocal[c * TILE_LEN],
                                  curLen);
            }
            AscendC::SetFlag<AscendC::HardEvent::MTE3_MTE2>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE3_MTE2>(EVENT_ID0);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> dataBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t tilesPerBatch;
    uint32_t tailLen;
    uint32_t startTile;
    uint32_t endTile;
};

extern "C" __global__ __aicore__ void conv_transpose3d_softmax_sigmoid_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dSoftmaxSigmoid op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize);
    op.Process();
}
