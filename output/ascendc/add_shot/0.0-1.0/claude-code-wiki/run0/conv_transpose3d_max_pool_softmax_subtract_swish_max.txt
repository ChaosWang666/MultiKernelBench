kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR subtract, GM_ADDR z,
                                 uint32_t N_, uint32_t C_, uint32_t spatialSize_,
                                 uint32_t tileLen_, uint32_t tilesPerCore_)
    {
        this->C = C_;
        this->spatialSize = spatialSize_;
        this->tileLen = tileLen_;
        this->tilesPerCore = tilesPerCore_;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t perCore = tilesPerCore_ * tileLen_;
        this->coreOffset = blockIdx * perCore;

        uint64_t totalInput = (uint64_t)N_ * (uint64_t)C_ * (uint64_t)spatialSize_;
        uint64_t totalOutput = (uint64_t)N_ * (uint64_t)spatialSize_;

        xGm.SetGlobalBuffer((__gm__ float*)x, totalInput);
        subGm.SetGlobalBuffer((__gm__ float*)subtract, C_);
        outGm.SetGlobalBuffer((__gm__ float*)z, totalOutput);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, C_ * tileLen_ * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLen_ * sizeof(float));
        pipe.InitBuffer(maxBuf, tileLen_ * sizeof(float));
        pipe.InitBuffer(sumBuf, tileLen_ * sizeof(float));
        pipe.InitBuffer(tmpBuf, tileLen_ * sizeof(float));
        uint32_t subAligned = ((C_ + 7) / 8) * 8;
        pipe.InitBuffer(subBuf, subAligned * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> subLocal = subBuf.Get<float>();
        uint32_t subAligned = ((this->C + 7) / 8) * 8;
        AscendC::DataCopy(subLocal, subGm, subAligned);

        auto eventId = AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::MTE2_S);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(eventId);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(eventId);

        for (uint32_t t = 0; t < tilesPerCore; t++) {
            uint32_t outBase = coreOffset + t * tileLen;
            uint32_t n = outBase / spatialSize;
            uint32_t spatialStart = outBase % spatialSize;

            CopyIn(n, spatialStart);
            Compute(subLocal);
            CopyOut(outBase);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t n, uint32_t spatialStart)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        uint64_t baseOffset = (uint64_t)n * (uint64_t)C * (uint64_t)spatialSize + (uint64_t)spatialStart;
        for (uint32_t c = 0; c < C; c++) {
            AscendC::DataCopy(xLocal[c * tileLen], xGm[baseOffset + (uint64_t)c * (uint64_t)spatialSize], tileLen);
        }
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(AscendC::LocalTensor<float>& subLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> maxTile = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumTile = sumBuf.Get<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        AscendC::Duplicate(maxTile, (float)-3.4e38f, tileLen);
        for (uint32_t c = 0; c < C; c++) {
            AscendC::Max(maxTile, maxTile, xLocal[c * tileLen], tileLen);
        }

        AscendC::Duplicate(sumTile, (float)0.0f, tileLen);
        for (uint32_t c = 0; c < C; c++) {
            AscendC::Sub(tmp, xLocal[c * tileLen], maxTile, tileLen);
            AscendC::Exp(xLocal[c * tileLen], tmp, tileLen);
            AscendC::Add(sumTile, sumTile, xLocal[c * tileLen], tileLen);
        }

        AscendC::Duplicate(yLocal, (float)-3.4e38f, tileLen);
        for (uint32_t c = 0; c < C; c++) {
            AscendC::Div(tmp, xLocal[c * tileLen], sumTile, tileLen);
            float subVal = subLocal.GetValue(c);
            AscendC::Adds(tmp, tmp, -subVal, tileLen);
            AscendC::Muls(maxTile, tmp, (float)-1.0f, tileLen);
            AscendC::Exp(maxTile, maxTile, tileLen);
            AscendC::Adds(maxTile, maxTile, (float)1.0f, tileLen);
            AscendC::Div(maxTile, tmp, maxTile, tileLen);
            AscendC::Max(yLocal, yLocal, maxTile, tileLen);
        }

        inQueueX.FreeTensor(xLocal);
        outQueueY.EnQue<float>(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t outBase)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(outGm[outBase], yLocal, tileLen);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf, sumBuf, tmpBuf, subBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> subGm;
    AscendC::GlobalTensor<float> outGm;

    uint32_t C;
    uint32_t spatialSize;
    uint32_t tileLen;
    uint32_t tilesPerCore;
    uint32_t coreOffset;
};

extern "C" __global__ __aicore__ void conv_transpose3d_max_pool_softmax_subtract_swish_max_custom(
    GM_ADDR x, GM_ADDR subtract, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, subtract, z,
            tiling_data.N, tiling_data.C, tiling_data.spatialSize,
            tiling_data.tileLen, tiling_data.tilesPerCore);
    op.Process();
}
