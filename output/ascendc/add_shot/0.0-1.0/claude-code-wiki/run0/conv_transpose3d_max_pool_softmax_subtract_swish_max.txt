kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR sub, GM_ADDR y,
                                 uint32_t batchPerCore, uint32_t channels,
                                 uint32_t innerSize, uint32_t tileLength)
    {
        this->batchPerCore = batchPerCore;
        this->channels = channels;
        this->innerSize = innerSize;
        this->tileLength = tileLength;
        this->tileNum = innerSize / tileLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t batchOffset = (uint64_t)blockIdx * (uint64_t)batchPerCore;

        xGm.SetGlobalBuffer((__gm__ float *)x + batchOffset * channels * innerSize,
                            (uint64_t)batchPerCore * channels * innerSize);
        subGm.SetGlobalBuffer((__gm__ float *)sub, channels);
        yGm.SetGlobalBuffer((__gm__ float *)y + batchOffset * innerSize,
                            (uint64_t)batchPerCore * innerSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf1, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, tileLength * sizeof(float));
        pipe.InitBuffer(subBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> subLocal = subBuf.Get<float>();
        uint32_t subCount = (channels + 7) / 8 * 8;
        AscendC::DataCopy(subLocal, subGm, subCount);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t b = 0; b < batchPerCore; b++) {
            for (uint32_t t = 0; t < tileNum; t++) {
                AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
                AscendC::LocalTensor<float> tmp1 = tmpBuf1.Get<float>();
                AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();

                AscendC::Duplicate<float>(yLocal, -3.4e38f, tileLength);

                uint64_t tileOffset = (uint64_t)t * tileLength;
                uint64_t baseOffset = (uint64_t)b * channels * innerSize;

                for (uint32_t c = 0; c < channels; c++) {
                    AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                    AscendC::DataCopy(xLocal,
                                      xGm[baseOffset + (uint64_t)c * innerSize + tileOffset],
                                      tileLength);
                    inQueueX.EnQue(xLocal);
                    xLocal = inQueueX.DeQue<float>();

                    float subVal = subLocal.GetValue(c);
                    AscendC::Adds(tmp1, xLocal, -subVal, tileLength);
                    inQueueX.FreeTensor(xLocal);

                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::Sigmoid(tmp2, tmp1, tileLength);
                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::Mul(tmp1, tmp1, tmp2, tileLength);
                    AscendC::PipeBarrier<PIPE_V>();
                    AscendC::Max(yLocal, yLocal, tmp1, tileLength);
                    AscendC::PipeBarrier<PIPE_V>();
                }

                outQueueY.EnQue(yLocal);
                AscendC::LocalTensor<float> yOut = outQueueY.DeQue<float>();
                AscendC::DataCopy(yGm[(uint64_t)b * innerSize + tileOffset], yOut, tileLength);
                outQueueY.FreeTensor(yOut);
            }
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf2;
    AscendC::TBuf<AscendC::TPosition::VECCALC> subBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> subGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchPerCore;
    uint32_t channels;
    uint32_t innerSize;
    uint32_t tileLength;
    uint32_t tileNum;
};

extern "C" __global__ __aicore__ void conv_transpose3d_max_pool_softmax_subtract_swish_max_custom(
    GM_ADDR x, GM_ADDR sub, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{

    op.Init(x, sub, y, tiling_data.batchPerCore, tiling_data.channels,

    op.Process();
}

import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, output_padding, pool_kernel_size, pool_stride, pool_padding):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.max_pool = nn.MaxPool3d(kernel_size=pool_kernel_size, stride=pool_stride, padding=pool_padding)
        self.subtract = nn.Parameter(torch.randn(out_channels))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.max_pool(x)
        x = torch.softmax(x, dim=1)
        x = x.contiguous()
        sub = self.subtract.contiguous()
        x = custom_ops_lib.conv_transpose3d_max_pool_softmax_subtract_swish_max_custom(x, sub)
        return x
