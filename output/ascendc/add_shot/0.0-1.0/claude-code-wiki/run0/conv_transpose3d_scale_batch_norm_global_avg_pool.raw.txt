kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGlobalAvgPool {
public:
    __aicore__ inline KernelGlobalAvgPool() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalNC, uint32_t spatialSize,
                                 uint32_t tileLength, uint32_t tileNum, uint32_t ncPerBlock,
                                 float reciprocal)
    {
        this->spatialSize = spatialSize;
        this->tileLength = tileLength;
        this->tileNum = tileNum;
        this->ncPerBlock = ncPerBlock;
        this->reciprocal = reciprocal;
        uint32_t ncStart = ncPerBlock * AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ float *)x + ncStart * spatialSize, ncPerBlock * spatialSize);
        yGm.SetGlobalBuffer((__gm__ float *)y + ncStart, ncPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(reduceDstBuf, 32);
        pipe.InitBuffer(reduceWorkBuf, tileLength * sizeof(float));
        pipe.InitBuffer(outBuf, ncPerBlock * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();

        for (uint32_t local_nc = 0; local_nc < this->ncPerBlock; local_nc++) {
            float acc = 0.0f;
            for (uint32_t i = 0; i < this->tileNum; i++) {
                CopyIn(local_nc, i);
                acc += ReduceTile();
            }
            outLocal.SetValue(local_nc, acc * this->reciprocal);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        AscendC::DataCopy(yGm, outLocal, this->ncPerBlock);
    }

private:
    __aicore__ inline void CopyIn(uint32_t local_nc, uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[local_nc * this->spatialSize + progress * this->tileLength], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline float ReduceTile()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> dstLocal = reduceDstBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = reduceWorkBuf.Get<float>();
        AscendC::ReduceSum<float>(dstLocal, xLocal, workLocal, this->tileLength);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

        float val = dstLocal.GetValue(0);
        inQueueX.FreeTensor(xLocal);

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        return val;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceDstBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceWorkBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t spatialSize;
    uint32_t tileLength;
    uint32_t tileNum;
    uint32_t ncPerBlock;
    float reciprocal;
};

extern "C" __global__ __aicore__ void conv_transpose3d_scale_batch_norm_global_avg_pool_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGlobalAvgPool op;
    op.Init(x, y, tiling_data.totalNC, tiling_data.spatialSize,
            tiling_data.tileLength, tiling_data.tileNum, tiling_data.ncPerBlock,
            tiling_data.reciprocal);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, scale_factor, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size)
        self.scale_factor = scale_factor
        self.batch_norm = nn.BatchNorm3d(out_channels, eps=eps, momentum=momentum)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x * self.scale_factor
        x = self.batch_norm(x)
        N, C = x.shape[0], x.shape[1]
        x = x.contiguous()
        y = custom_ops_lib.conv_transpose3d_scale_batch_norm_global_avg_pool_custom(x)
        return y.view(N, C, 1, 1, 1)
