#include "kernel_operator.h"

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR multiplier, GM_ADDR y,
                                 uint32_t batchSize, uint32_t outChannels,
                                 uint32_t spatialSize, float clampMin, float clampMax)
    {
        this->batchSize = batchSize;
        this->outChannels = outChannels;
        this->spatialSize = spatialSize;
        this->clampMin = clampMin;
        this->clampMax = clampMax;

        xGm.SetGlobalBuffer((__gm__ float *)x);
        multGm.SetGlobalBuffer((__gm__ float *)multiplier);
        yGm.SetGlobalBuffer((__gm__ float *)y);
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t batchesPerBlock = (batchSize + blockNum - 1) / blockNum;
        uint32_t startBatch = blockIdx * batchesPerBlock;
        uint32_t endBatch = startBatch + batchesPerBlock;
        if (endBatch > batchSize) endBatch = batchSize;

        for (uint32_t b = startBatch; b < endBatch; b++) {
            ProcessBatch(b);
        }
    }

private:
    __aicore__ inline float SqrtScalar(float val) {
        if (val <= 0.0f) return 0.0f;
        float y = val;
        for (int32_t i = 0; i < 20; i++) {
            y = 0.5f * (y + val / y);
        }
        return y;
    }

    __aicore__ inline void ProcessBatch(uint32_t b)
    {
        uint32_t C = outChannels;
        uint32_t S = spatialSize;
        uint32_t batchOffset = b * C * S;
        uint32_t outOffset = b * S;
        float invS = 1.0f / (float)S;

        for (uint32_t s = 0; s < S; s++) {
            yGm.SetValue(outOffset + s, -3.4028235e38f);
        }

        for (uint32_t c = 0; c < C; c++) {
            float mult_c = multGm.GetValue(c);
            uint32_t chanOffset = batchOffset + c * S;

            float sum = 0.0f;
            for (uint32_t s = 0; s < S; s++) {
                sum += xGm.GetValue(chanOffset + s) * mult_c;
            }
            float mean = sum * invS;

            float varSum = 0.0f;
            for (uint32_t s = 0; s < S; s++) {
                float v = xGm.GetValue(chanOffset + s) * mult_c;
                float diff = v - mean;
                varSum += diff * diff;
            }
            float var = varSum * invS;
            float inv_std = 1.0f / SqrtScalar(var + 1e-5f);

            for (uint32_t s = 0; s < S; s++) {
                float v = xGm.GetValue(chanOffset + s) * mult_c;
                float normVal = (v - mean) * inv_std;
                if (normVal < clampMin) normVal = clampMin;
                if (normVal > clampMax) normVal = clampMax;
                float finalVal = normVal * mult_c;

                float curMax = yGm.GetValue(outOffset + s);
                if (finalVal > curMax) {
                    yGm.SetValue(outOffset + s, finalVal);
                }
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> multGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t outChannels;
    uint32_t spatialSize;
    float clampMin;
    float clampMax;
};
