#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmFused {
public:
    __aicore__ inline KernelGemmFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR wcs, GM_ADDR cval, GM_ADDR y,
                                uint32_t batchSize, uint32_t inFeatures,
                                uint32_t rowsPerBlock, uint32_t tileLengthK, uint32_t tileNumK)
    {
        this->inFeatures = inFeatures;
        this->rowsPerBlock = rowsPerBlock;
        this->tileLengthK = tileLengthK;
        this->tileNumK = tileNumK;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowStart = blockIdx * rowsPerBlock;

        xGm.SetGlobalBuffer((__gm__ float*)x + rowStart * inFeatures, rowsPerBlock * inFeatures);
        wcsGm.SetGlobalBuffer((__gm__ float*)wcs, inFeatures);
        cvalGm.SetGlobalBuffer((__gm__ float*)cval, 1);
        yGm.SetGlobalBuffer((__gm__ float*)y + rowStart * inFeatures, rowsPerBlock * inFeatures);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLengthK * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, tileLengthK * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLengthK * sizeof(float));
        pipe.InitBuffer(mulBuf, tileLengthK * sizeof(float));
        pipe.InitBuffer(workBuf, tileLengthK * sizeof(float));
        pipe.InitBuffer(sBuf0, 64 * sizeof(float));
        pipe.InitBuffer(sBuf1, 64 * sizeof(float));
        pipe.InitBuffer(sBuf2, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        float cScaled = cvalGm.GetValue(0);
        for (uint32_t r = 0; r < rowsPerBlock; r++) {
            float dot = ComputeDot(r);
            float meanVal = dot + cScaled;
            float geluVal = ComputeGelu(meanVal);
            BroadcastAdd(r, geluVal);
        }
    }

private:
    __aicore__ inline float ComputeDot(uint32_t row)
    {
        float dot = 0.0f;
        for (uint32_t t = 0; t < tileNumK; t++) {
            AscendC::LocalTensor<float> xL = inQueueX.AllocTensor<float>();
            AscendC::LocalTensor<float> wL = inQueueW.AllocTensor<float>();
            AscendC::DataCopy(xL, xGm[row * inFeatures + t * tileLengthK], tileLengthK);
            AscendC::DataCopy(wL, wcsGm[t * tileLengthK], tileLengthK);
            inQueueX.EnQue(xL);
            inQueueW.EnQue(wL);

            AscendC::LocalTensor<float> xD = inQueueX.DeQue<float>();
            AscendC::LocalTensor<float> wD = inQueueW.DeQue<float>();
            AscendC::LocalTensor<float> mulL = mulBuf.Get<float>();
            AscendC::LocalTensor<float> workL = workBuf.Get<float>();
            AscendC::LocalTensor<float> outS = sBuf0.Get<float>();

            AscendC::Mul(mulL, xD, wD, tileLengthK);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum<float>(outS, mulL, workL, tileLengthK);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::event_t evt = static_cast<AscendC::event_t>(AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(evt);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(evt);
            dot += outS.GetValue(0);

            inQueueX.FreeTensor(xD);
            inQueueW.FreeTensor(wD);
        }
        return dot;
    }

    __aicore__ inline float ComputeGelu(float v)
    {
        AscendC::LocalTensor<float> t1 = sBuf0.Get<float>();
        AscendC::LocalTensor<float> t2 = sBuf1.Get<float>();
        AscendC::LocalTensor<float> t3 = sBuf2.Get<float>();

        AscendC::event_t evtSV = static_cast<AscendC::event_t>(AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_V));
        AscendC::SetFlag<AscendC::HardEvent::S_V>(evtSV);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(evtSV);

        AscendC::Duplicate(t1, v, 8);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(t2, t1, t1, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Mul(t2, t2, t1, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Muls(t2, t2, 0.044715f, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Add(t2, t1, t2, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Muls(t2, t2, 1.5957691216057308f, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Exp(t2, t2, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Adds(t3, t2, 1.0f, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Adds(t2, t2, -1.0f, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Div(t2, t2, t3, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Adds(t2, t2, 1.0f, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Muls(t1, t1, 0.5f, 8);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Mul(t1, t1, t2, 8);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::event_t evtVS = static_cast<AscendC::event_t>(AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(evtVS);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(evtVS);
        return t1.GetValue(0);
    }

    __aicore__ inline void BroadcastAdd(uint32_t row, float geluVal)
    {
        for (uint32_t t = 0; t < tileNumK; t++) {
            AscendC::LocalTensor<float> xL = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xL, xGm[row * inFeatures + t * tileLengthK], tileLengthK);
            inQueueX.EnQue(xL);

            AscendC::LocalTensor<float> xD = inQueueX.DeQue<float>();
            AscendC::LocalTensor<float> yL = outQueueY.AllocTensor<float>();
            AscendC::Adds(yL, xD, geluVal, tileLengthK);
            outQueueY.EnQue(yL);
            inQueueX.FreeTensor(xD);

            AscendC::LocalTensor<float> yD = outQueueY.DeQue<float>();
            AscendC::DataCopy(yGm[row * inFeatures + t * tileLengthK], yD, tileLengthK);
            outQueueY.FreeTensor(yD);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mulBuf, workBuf, sBuf0, sBuf1, sBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wcsGm;
    AscendC::GlobalTensor<float> cvalGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t inFeatures;
    uint32_t rowsPerBlock;
    uint32_t tileLengthK;
    uint32_t tileNumK;
};

extern "C" __global__ __aicore__ void gemm_subtract_global_avg_pool_log_sum_exp_gelu_residual_add_custom(
