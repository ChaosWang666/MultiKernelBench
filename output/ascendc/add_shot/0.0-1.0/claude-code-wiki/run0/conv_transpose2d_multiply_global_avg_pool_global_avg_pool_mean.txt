kernel_src="""
#include "kernel_operator.h"

class KernelConvTMeanMul {
public:
    __aicore__ inline KernelConvTMeanMul() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t totalSlices,
                                uint32_t spatialSize,
                                uint32_t slicesPerBlock,
                                uint32_t tileLen,
                                uint32_t tileNum,
                                float scale)
    {
        this->spatialSize = spatialSize;
        this->slicesPerBlock = slicesPerBlock;
        this->tileLen = tileLen;
        this->tileNum = tileNum;
        this->scale = scale;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startSlice = blockIdx * slicesPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + startSlice * spatialSize, slicesPerBlock * spatialSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + startSlice, slicesPerBlock);

        pipe.InitBuffer(inQueueX, 2, tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, 1, slicesPerBlock * sizeof(DTYPE_Y));
        pipe.InitBuffer(accumBuf, tileLen * sizeof(float));
        pipe.InitBuffer(reduceBuf, tileLen * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::LocalTensor<float> accumLocal = accumBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        for (uint32_t i = 0; i < slicesPerBlock; i++) {
            AscendC::Duplicate<float>(accumLocal, 0.0f, tileLen);
            AscendC::PipeBarrier<PIPE_V>();

            for (uint32_t t = 0; t < tileNum; t++) {
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(xLocal, xGm[i * spatialSize + t * tileLen], tileLen);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<DTYPE_X>();
                AscendC::Add(accumLocal, accumLocal, xLocal, tileLen);
                inQueueX.FreeTensor(xLocal);
            }

            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum<float>(reduceLocal, accumLocal, accumLocal, tileLen);

            event_t eventIdVToS = static_cast<event_t>(AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(eventIdVToS);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(eventIdVToS);

            float val = reduceLocal.GetValue(0);
            yLocal.SetValue(i, val * scale);
        }

        event_t eventIdSToMTE3 = static_cast<event_t>(AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_MTE3));
        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(eventIdSToMTE3);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(eventIdSToMTE3);

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        AscendC::LocalTensor<DTYPE_Y> yOut = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm, yOut, slicesPerBlock);
        outQueueY.FreeTensor(yOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t spatialSize;
    uint32_t slicesPerBlock;
    uint32_t tileLen;
    uint32_t tileNum;
    float scale;
};

extern "C" __global__ __aicore__ void conv_transpose2d_multiply_global_avg_pool_global_avg_pool_mean_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);

}
