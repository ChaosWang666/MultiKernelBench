kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose2dMinSumGeluAdd {
public:
    __aicore__ inline KernelConvTranspose2dMinSumGeluAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                uint32_t batchSize, uint32_t channels,
                                uint32_t height, uint32_t width)
    {
        this->channels = channels;
        this->height = height;
        this->width = width;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t batchStride = channels * height * width;

        xGm.SetGlobalBuffer((__gm__ float*)x + blockIdx * batchStride, batchStride);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, 8);
        zGm.SetGlobalBuffer((__gm__ float*)z + blockIdx * width, width);

        pipe.InitBuffer(xQueue, BUFFER_NUM, width * sizeof(float));
        pipe.InitBuffer(biasQueue, 1, 32);
        pipe.InitBuffer(outQueue, 1, width * sizeof(float));
        pipe.InitBuffer(sumBuf, width * sizeof(float));
        pipe.InitBuffer(minBuf, width * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> minLocal = minBuf.Get<float>();

        AscendC::Duplicate(sumLocal, 0.0f, width);

        for (uint32_t h = 0; h < height; h++) {
            AscendC::Duplicate(minLocal, 1.0e30f, width);

            for (uint32_t c = 0; c < channels; c++) {
                AscendC::LocalTensor<float> rowLocal = xQueue.AllocTensor<float>();
                AscendC::DataCopy(rowLocal, xGm[c * height * width + h * width], width);
                xQueue.EnQue(rowLocal);

                AscendC::LocalTensor<float> row = xQueue.DeQue<float>();
                AscendC::Min(minLocal, minLocal, row, width);
                xQueue.FreeTensor(row);
            }

            AscendC::Add(sumLocal, sumLocal, minLocal, width);
        }

        AscendC::LocalTensor<float> biasLocal = biasQueue.AllocTensor<float>();
        AscendC::DataCopy(biasLocal, biasGm, 8);
        biasQueue.EnQue(biasLocal);
        biasLocal = biasQueue.DeQue<float>();
        float biasVal = biasLocal.GetValue(0);
        biasQueue.FreeTensor(biasLocal);

        AscendC::LocalTensor<float> tmpLocal = minLocal;
        AscendC::Mul(tmpLocal, sumLocal, sumLocal, width);
        AscendC::Mul(tmpLocal, tmpLocal, sumLocal, width);
        AscendC::Muls(tmpLocal, tmpLocal, 0.044715f, width);
        AscendC::Add(tmpLocal, tmpLocal, sumLocal, width);
        AscendC::Muls(tmpLocal, tmpLocal, -1.5957691f, width);
        AscendC::Exp(tmpLocal, tmpLocal, width);
        AscendC::Adds(tmpLocal, tmpLocal, 1.0f, width);

        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();
        AscendC::Div(outLocal, sumLocal, tmpLocal, width);
        AscendC::Adds(outLocal, outLocal, biasVal, width);

        outQueue.EnQue(outLocal);
        AscendC::LocalTensor<float> zOut = outQueue.DeQue<float>();
        AscendC::DataCopy(zGm, zOut, width);
        outQueue.FreeTensor(zOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> xQueue;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> biasQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> minBuf;

    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;

    uint32_t channels;
    uint32_t height;
    uint32_t width;
};

extern "C" __global__ __aicore__ void conv_transpose2d_min_sum_gelu_add_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose2dMinSumGeluAdd op;
    op.Init(x, bias, z, tiling_data.batchSize, tiling_data.channels,
            tiling_data.height, tiling_data.width);
    op.Process();
}

class ModelNew(torch.nn.Module):

        super(ModelNew, self).__init__()
        self.conv_transpose = torch.nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride, padding, output_padding)
