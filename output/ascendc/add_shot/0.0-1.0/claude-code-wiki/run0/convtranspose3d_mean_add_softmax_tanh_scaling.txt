kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConvTM {
public:
    __aicore__ inline KernelConvTM() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                 uint32_t B_, uint32_t C_, uint32_t D_, uint32_t H_, uint32_t W_,
                                 float invD_, float scaling_)
    {
        B = B_; C = C_; D = D_; H = H_; W = W_;
        invD = invD_; scaling = scaling_;
        HW = H * W;
        DHW = D * HW;
        CDHW = C * DHW;
        CHW = C * HW;

        uint32_t totalTasks = B * H;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t perCore = (totalTasks + blockNum - 1) / blockNum;
        startTask = blockIdx * perCore;
        endTask = startTask + perCore;
        if (endTask > totalTasks) endTask = totalTasks;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        biasGm.SetGlobalBuffer((__gm__ float*)bias);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, D * W * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, C * W * sizeof(float));
        pipe.InitBuffer(meansBuf, C * W * sizeof(float));
        pipe.InitBuffer(maxBuf, W * sizeof(float));
        pipe.InitBuffer(sumBuf, W * sizeof(float));
        pipe.InitBuffer(biasBuf, C * sizeof(float));

        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::DataCopy(biasLocal, biasGm, C);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
    }

    __aicore__ inline void Process()
    {
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        if (startTask >= endTask) return;

        AscendC::LocalTensor<float> means = meansBuf.Get<float>();
        AscendC::LocalTensor<float> maxVec = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumVec = sumBuf.Get<float>();
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();

        for (uint32_t task = startTask; task < endTask; task++) {
            uint32_t b = task / H;
            uint32_t h = task - b * H;

            // Step 1: mean along D + bias per channel
            for (uint32_t c = 0; c < C; c++) {
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                uint64_t baseOff = (uint64_t)b * CDHW + (uint64_t)c * DHW + (uint64_t)h * W;
                for (uint32_t d = 0; d < D; d++) {
                    AscendC::DataCopy(xLocal[d * W], xGm[baseOff + (uint64_t)d * HW], W);
                }
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();
                for (uint32_t d = 1; d < D; d++) {
                    AscendC::Add(xLocal, xLocal, xLocal[d * W], W);
                }
                AscendC::Muls(xLocal, xLocal, invD, W);
                float bVal = biasLocal.GetValue(c);
                AscendC::Adds(means[c * W], xLocal, bVal, W);
                inQueueX.FreeTensor(xLocal);
            }

            // Step 2: softmax across C into output buffer
            AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
            AscendC::Duplicate(maxVec, -3.402823e38f, W);
            for (uint32_t c = 0; c < C; c++) {
                AscendC::Max(maxVec, maxVec, means[c * W], W);
            }
            for (uint32_t c = 0; c < C; c++) {
                AscendC::Sub(yLocal[c * W], means[c * W], maxVec, W);
            }
            AscendC::Exp(yLocal, yLocal, C * W);
            AscendC::Duplicate(sumVec, 0.0f, W);
            for (uint32_t c = 0; c < C; c++) {
                AscendC::Add(sumVec, sumVec, yLocal[c * W], W);
            }
            for (uint32_t c = 0; c < C; c++) {
                AscendC::Div(yLocal[c * W], yLocal[c * W], sumVec, W);
            }

            // Step 3: tanh + scale
            AscendC::Tanh(yLocal, yLocal, C * W);
            AscendC::Muls(yLocal, yLocal, scaling, C * W);

            outQueueY.EnQue<float>(yLocal);
            yLocal = outQueueY.DeQue<float>();
            uint64_t outBase = (uint64_t)b * CHW + (uint64_t)h * W;
            for (uint32_t c = 0; c < C; c++) {
                AscendC::DataCopy(yGm[outBase + (uint64_t)c * HW], yLocal[c * W], W);
            }
            outQueueY.FreeTensor(yLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> meansBuf, maxBuf, sumBuf, biasBuf;
    AscendC::GlobalTensor<float> xGm, biasGm, yGm;
    uint32_t B, C, D, H, W;
    uint32_t HW, DHW, CDHW, CHW;
    uint32_t startTask, endTask;
    float invD, scaling;
};

extern "C" __global__ __aicore__ void convtranspose3d_mean_add_softmax_tanh_scaling_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTM op;
    op.Init(x, bias, y,
            tiling_data.B, tiling_data.C, tiling_data.D, tiling_data.H, tiling_data.W,
            tiling_data.invD, tiling_data.scaling);
    op.Process();
}

import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
