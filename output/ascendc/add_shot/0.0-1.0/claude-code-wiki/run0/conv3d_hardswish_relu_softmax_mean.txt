kernel_src="""
#include "kernel_operator.h"

class KernelConv3dHardswishReluSoftmaxMean {
public:
    __aicore__ inline KernelConv3dHardswishReluSoftmaxMean() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t batchesPerCore, uint32_t channels,
                                 uint32_t spatial, uint32_t tileSize, float invSpatial)
    {
        this->batchesPerCore = batchesPerCore;
        this->channels = channels;
        this->spatial = spatial;
        this->tileSize = tileSize;
        this->invSpatial = invSpatial;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t inStart = (uint64_t)blockIdx * batchesPerCore * channels * spatial;
        uint64_t outStart = (uint64_t)blockIdx * batchesPerCore * channels;

        xGm.SetGlobalBuffer((__gm__ float *)x + inStart, (uint64_t)batchesPerCore * channels * spatial);
        yGm.SetGlobalBuffer((__gm__ float *)y + outStart, (uint64_t)batchesPerCore * channels);

        pipe.InitBuffer(inQueueX, 2, tileSize * sizeof(float));
        pipe.InitBuffer(redBuf, tileSize * sizeof(float));
        pipe.InitBuffer(outBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> red = redBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();

        for (uint32_t b = 0; b < batchesPerCore; b++) {
            for (uint32_t c = 0; c < channels; c++) {
                uint64_t rowBase = ((uint64_t)b * channels + c) * spatial;
                float channelSum = 0.0f;
                uint32_t pos = 0;

                while (pos < spatial) {
                    uint32_t remain = spatial - pos;
                    uint32_t currentTile = remain < tileSize ? remain : tileSize;
                    uint32_t alignedTile = (currentTile + 63) / 64 * 64;

                    AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                    AscendC::DataCopyExtParams copyParams{1, (uint32_t)(currentTile * sizeof(float)), 0, 0, 0};
                    AscendC::DataCopyPadExtParams<float> padParams{true, 0, (uint8_t)(alignedTile - currentTile), 0.0f};
                    AscendC::DataCopyPad(xLocal, xGm[rowBase + pos], copyParams, padParams);
                    inQueueX.EnQue(xLocal);

                    xLocal = inQueueX.DeQue<float>();
                    AscendC::ReduceSum<float>(red, xLocal, red, alignedTile);
                    AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                    AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                    channelSum += red.GetValue(0);
                    inQueueX.FreeTensor(xLocal);

                    pos += currentTile;
                }

                outLocal.SetValue(c, channelSum * invSpatial);
            }

            AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

            AscendC::DataCopyExtParams outParams{1, (uint32_t)(channels * sizeof(float)), 0, 0, 0};
            AscendC::DataCopyPad(yGm[(uint64_t)b * channels], outLocal, outParams);

            AscendC::SetFlag<AscendC::HardEvent::MTE3_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE3_S>(EVENT_ID0);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> redBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchesPerCore;
    uint32_t channels;
    uint32_t spatial;
    uint32_t tileSize;
    float invSpatial;
};

extern "C" __global__ __aicore__ void conv3d_hardswish_relu_softmax_mean_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dHardswishReluSoftmaxMean op;
    op.Init(x, y, tiling_data.batchesPerCore, tiling_data.channels,
            tiling_data.spatial, tiling_data.tileSize, tiling_data.invSpatial);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch.nn.functional as F
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, num_groups=4, bias=True):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size, bias=bias)
        self.group_norm = nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        x = self.conv(x)
        x = F.hardswish(x)
        x = self.group_norm(x)
        x = x.contiguous()
        return custom_ops_lib.conv3d_hardswish_relu_softmax_mean_custom(x)
