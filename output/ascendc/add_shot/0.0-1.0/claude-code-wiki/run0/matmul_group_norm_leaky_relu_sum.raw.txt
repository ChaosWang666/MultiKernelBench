class KernelMatmulGroupNormLeakyReluSum {
public:
    __aicore__ inline KernelMatmulGroupNormLeakyReluSum() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                uint32_t batchSize, uint32_t hiddenSize,
                                uint32_t numGroups, uint32_t channelsPerGroup,
                                float invN, float epsVal, float negSlope)
    {
        this->batchSize = batchSize;
        this->hiddenSize = hiddenSize;
        this->numGroups = numGroups;
        this->cpg = channelsPerGroup;
        this->invN = invN;
        this->epsVal = epsVal;
        this->negSlope = negSlope;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t rowsPerBlock = (batchSize + blockNum - 1) / blockNum;
        uint32_t startRow = blockIdx * rowsPerBlock;
        uint32_t endRow = startRow + rowsPerBlock;
        if (endRow > batchSize) endRow = batchSize;
        this->startRow = startRow;
        this->rowsThisBlock = (endRow > startRow) ? (endRow - startRow) : 0;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * hiddenSize);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, hiddenSize);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, hiddenSize);
        yGm.SetGlobalBuffer((__gm__ float*)y, batchSize * hiddenSize);

        pipe.InitBuffer(inQueueX, 1, hiddenSize * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, hiddenSize * sizeof(float));
        pipe.InitBuffer(gammaBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(betaBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(tmpBuf, hiddenSize * sizeof(float));
        pipe.InitBuffer(reduceResBuf, 32);
        pipe.InitBuffer(workBuf, 256);
    }

    __aicore__ inline void Process()
    {
        if (rowsThisBlock == 0) return;

        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();
        AscendC::DataCopy(gammaLocal, gammaGm, hiddenSize);
        AscendC::DataCopy(betaLocal, betaGm, hiddenSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t i = 0; i < rowsThisBlock; i++) {
            ProcessRow(startRow + i, gammaLocal, betaLocal);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t row,
                                      AscendC::LocalTensor<float>& gammaLocal,
                                      AscendC::LocalTensor<float>& betaLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * hiddenSize], hiddenSize);
        inQueueX.EnQue(xLocal);

        xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceRes = reduceResBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();

        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t off = g * cpg;

            AscendC::ReduceSum(reduceRes, xLocal[off], work, (int32_t)cpg);
            AscendC::event_t e1 = static_cast<AscendC::event_t>(
                AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(e1);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(e1);
            float sumVal = reduceRes.GetValue(0);
            float mean = sumVal * invN;

            AscendC::event_t e2 = static_cast<AscendC::event_t>(
                AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_V));
            AscendC::SetFlag<AscendC::HardEvent::S_V>(e2);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(e2);

            AscendC::Adds(xLocal[off], xLocal[off], -mean, (int32_t)cpg);
            AscendC::Mul(tmp[off], xLocal[off], xLocal[off], (int32_t)cpg);
            AscendC::ReduceSum(reduceRes, tmp[off], work, (int32_t)cpg);

            AscendC::event_t e3 = static_cast<AscendC::event_t>(
                AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(e3);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(e3);
            float sumSq = reduceRes.GetValue(0);
            float var = sumSq * invN;
            float invStd = 1.0f / sqrt(var + epsVal);

            AscendC::event_t e4 = static_cast<AscendC::event_t>(
                AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_V));
            AscendC::SetFlag<AscendC::HardEvent::S_V>(e4);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(e4);

            AscendC::Muls(xLocal[off], xLocal[off], invStd, (int32_t)cpg);
        }

        AscendC::Mul(xLocal, xLocal, gammaLocal, (int32_t)hiddenSize);
        AscendC::Add(xLocal, xLocal, betaLocal, (int32_t)hiddenSize);

        AscendC::Muls(tmp, xLocal, negSlope, (int32_t)hiddenSize);
        AscendC::Max(xLocal, xLocal, tmp, (int32_t)hiddenSize);

        AscendC::Muls(yLocal, xLocal, 2.0f, (int32_t)hiddenSize);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);

        yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[row * hiddenSize], yLocal, hiddenSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> betaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceResBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t hiddenSize;
    uint32_t numGroups;
    uint32_t cpg;
    uint32_t startRow;
    uint32_t rowsThisBlock;
    float invN;
    float epsVal;
    float negSlope;
};

extern "C" __global__ __aicore__ void matmul_group_norm_leaky_relu_sum_custom(
