kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t TILE_SIZE = 1024;
constexpr int32_t MAX_SPATIAL = 16384;
constexpr int32_t MAX_CHANNELS = 64;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR mult, GM_ADDR out,
                                 uint32_t batchSize, uint32_t channels, uint32_t spatialSize,
                                 float clampMin, float clampMax, float invSpatial, float eps)
    {
        this->batchSizeI = static_cast<int32_t>(batchSize);
        this->channelsI = static_cast<int32_t>(channels);
        this->spatialSizeI = static_cast<int32_t>(spatialSize);
        this->clampMin = clampMin;
        this->clampMax = clampMax;
        this->invSpatial = invSpatial;
        this->eps = eps;

        int32_t blockIdx = AscendC::GetBlockIdx();
        int32_t blockNum = AscendC::GetBlockNum();
        int32_t bpb = (this->batchSizeI + blockNum - 1) / blockNum;
        this->batchStart = blockIdx * bpb;
        this->batchEnd = this->batchStart + bpb;
        if (this->batchEnd > this->batchSizeI) this->batchEnd = this->batchSizeI;
        if (this->batchStart > this->batchSizeI) this->batchStart = this->batchSizeI;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        multGm.SetGlobalBuffer((__gm__ float*)mult);
        outGm.SetGlobalBuffer((__gm__ float*)out);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, TILE_SIZE * sizeof(float));
        pipe.InitBuffer(outQueueOut, BUFFER_NUM, TILE_SIZE * sizeof(float));
        pipe.InitBuffer(multQueue, 1, MAX_CHANNELS * sizeof(float));
        pipe.InitBuffer(tmpBuf, 256 * sizeof(float));
        pipe.InitBuffer(workBuf, 256 * sizeof(float));
        pipe.InitBuffer(maxBuf, MAX_SPATIAL * sizeof(float));
        pipe.InitBuffer(sqrtBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> multLocal = multQueue.AllocTensor<float>();
        int32_t mcount = (this->channelsI + 7) / 8 * 8;
        AscendC::DataCopy(multLocal, multGm, mcount);
        multQueue.EnQue(multLocal);
        multLocal = multQueue.DeQue<float>();

        float mults[MAX_CHANNELS];
        for (int32_t c = 0; c < this->channelsI; c++) {
            mults[c] = multLocal.GetValue(c);
        }
        multQueue.FreeTensor(multLocal);

        for (int32_t n = this->batchStart; n < this->batchEnd; n++) {
            ProcessSample(n, mults);
        }
    }

private:
    __aicore__ inline float ComputeInvStd(float var)
    {
        AscendC::LocalTensor<float> t = sqrtBuf.Get<float>();
        t.SetValue(0, var + this->eps);
        AscendC::Sqrt(t, t, 64);
        float stdVal = t.GetValue(0);
        return 1.0f / stdVal;
    }

    __aicore__ inline void ProcessSample(int32_t n, float* mults)
    {
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::Duplicate(maxLocal, -3.4e38f, this->spatialSizeI);

        int32_t sp = this->spatialSizeI;
        int32_t C = this->channelsI;

        for (int32_t c = 0; c < C; c++) {
            float multVal = mults[c];
            int32_t baseOffset = n * C * sp + c * sp;

            float sum = 0.0f;
            float sumSq = 0.0f;
            for (int32_t i = 0; i < sp; i += TILE_SIZE) {
                int32_t len = (i + TILE_SIZE > sp) ? (sp - i) : TILE_SIZE;
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[baseOffset + i], len);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();

                AscendC::Muls(xLocal, xLocal, multVal, len);

                AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
                AscendC::LocalTensor<float> work = workBuf.Get<float>();
                AscendC::ReduceSum<float>(tmp, xLocal, work, len);
                sum += tmp.GetValue(0);

                AscendC::Mul(xLocal, xLocal, xLocal, len);
                AscendC::ReduceSum<float>(tmp, xLocal, work, len);
                sumSq += tmp.GetValue(0);

                inQueueX.FreeTensor(xLocal);
            }

            float mean = sum * this->invSpatial;
            float var = sumSq * this->invSpatial - mean * mean;
            if (var < 0.0f) var = 0.0f;
            float invStd = ComputeInvStd(var);
            float negMean = -mean;

            for (int32_t i = 0; i < sp; i += TILE_SIZE) {
                int32_t len = (i + TILE_SIZE > sp) ? (sp - i) : TILE_SIZE;
                AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[baseOffset + i], len);
                inQueueX.EnQue(xLocal);
                xLocal = inQueueX.DeQue<float>();

                AscendC::Muls(xLocal, xLocal, multVal, len);
                AscendC::Adds(xLocal, xLocal, negMean, len);
                AscendC::Muls(xLocal, xLocal, invStd, len);
                AscendC::Maxs(xLocal, xLocal, this->clampMin, len);
                AscendC::Mins(xLocal, xLocal, this->clampMax, len);
                AscendC::Muls(xLocal, xLocal, multVal, len);

                AscendC::Max(maxLocal[i], maxLocal[i], xLocal, len);

                inQueueX.FreeTensor(xLocal);
            }
        }

        int32_t outOffset = n * sp;
        for (int32_t i = 0; i < sp; i += TILE_SIZE) {
            int32_t len = (i + TILE_SIZE > sp) ? (sp - i) : TILE_SIZE;
            AscendC::LocalTensor<float> outLocal = outQueueOut.AllocTensor<float>();
            AscendC::Adds(outLocal, maxLocal[i], 0.0f, len);
            outQueueOut.EnQue<float>(outLocal);
            outLocal = outQueueOut.DeQue<float>();
            AscendC::DataCopy(outGm[outOffset + i], outLocal, len);
            outQueueOut.FreeTensor(outLocal);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueOut;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> multQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sqrtBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> multGm;
    AscendC::GlobalTensor<float> outGm;
    int32_t batchSizeI;
    int32_t channelsI;
    int32_t spatialSizeI;
    float clampMin;
    float clampMax;
    float invSpatial;
    float eps;
    int32_t batchStart;
    int32_t batchEnd;
};

extern "C" __global__ __aicore__ void conv3d_multiply_instance_norm_clamp_multiply_max_custom(
    GM_ADDR x, GM_ADDR mult, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, mult, z, tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize,
            tiling_data.clampMin, tiling_data.clampMax, tiling_data.invSpatial, tiling_data.eps);
    op.Process();
}
