kernel_src="""
#include "kernel_operator.h"

class KernelConvLseFused {
public:
    __aicore__ inline KernelConvLseFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                 uint32_t batch, uint32_t channels)
    {
        this->batch = batch;
        this->channels = channels;

        xGm.SetGlobalBuffer((__gm__ float*)x, batch * channels);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, channels);
        zGm.SetGlobalBuffer((__gm__ float*)z, batch);

        pipe.InitBuffer(xBuf, batch * channels * sizeof(float));
        pipe.InitBuffer(biasBuf, channels * sizeof(float));
        pipe.InitBuffer(tmpBuf, channels * sizeof(float));
        pipe.InitBuffer(workBuf, 256 * sizeof(float));
        pipe.InitBuffer(maxBuf, 64 * sizeof(float));
        pipe.InitBuffer(sumBuf, 64 * sizeof(float));
        pipe.InitBuffer(outBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();
        AscendC::LocalTensor<float> maxArr = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumArr = sumBuf.Get<float>();
        AscendC::LocalTensor<float> out = outBuf.Get<float>();

        AscendC::DataCopy(xLocal, xGm, this->batch * this->channels);
        AscendC::DataCopy(biasLocal, biasGm, this->channels);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t b = 0; b < this->batch; b++) {
            AscendC::Add(tmp, xLocal[b * this->channels], biasLocal, this->channels);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceMax<float>(work, tmp, work, this->channels, false);
            AscendC::PipeBarrier<PIPE_ALL>();

            float maxVal = work.GetValue(0);
            maxArr.SetValue(b, maxVal);

            AscendC::Adds(tmp, tmp, -maxVal, this->channels);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Exp(tmp, tmp, this->channels);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::ReduceSum<float>(work, tmp, work, this->channels);
            AscendC::PipeBarrier<PIPE_ALL>();

            float sumVal = work.GetValue(0);
            sumArr.SetValue(b, sumVal);
        }

        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Ln(sumArr, sumArr, this->batch);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Add(out, maxArr, sumArr, this->batch);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Muls(out, out, 10.0f, this->batch);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::DataCopy(zGm, out, this->batch);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t batch;
    uint32_t channels;
};

extern "C" __global__ __aicore__ void convtranspose2d_globalavgpool_biasadd_logsumexp_sum_multiply_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvLseFused op;
    op.Init(x, bias, z, tiling_data.batch, tiling_data.channels);
    op.Process();
}
