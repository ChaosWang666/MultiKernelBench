    {

    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();

        // Mish(x) = x * tanh(softplus(x)) = x * tanh(ln(1 + exp(x)))
        // softplus = ln(1 + exp(x))
        AscendC::Exp(tmp1, xLocal, this->tileLength);
        AscendC::Adds(tmp1, tmp1, (float)1.0f, this->tileLength);
        AscendC::Ln(tmp1, tmp1, this->tileLength);

        // tanh(softplus) = (exp(2*sp) - 1) / (exp(2*sp) + 1)
        AscendC::Muls(tmp2, tmp1, (float)2.0f, this->tileLength);
        AscendC::Exp(tmp2, tmp2, this->tileLength);
        AscendC::Adds(tmp1, tmp2, (float)-1.0f, this->tileLength);
        AscendC::Adds(tmp2, tmp2, (float)1.0f, this->tileLength);
        AscendC::Div(tmp1, tmp1, tmp2, this->tileLength);

        // mish = x * tanh(softplus)
        AscendC::Mul(zLocal, xLocal, tmp1, this->tileLength);

        // add value

        // hardtanh(x, -1, 1) = min(max(x, -1), 1)

        AscendC::Mins(zLocal, zLocal, (float)1.0f, this->tileLength);
