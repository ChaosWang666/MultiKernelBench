kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMatmulSwishSumGroupNorm {
public:
    __aicore__ inline KernelMatmulSwishSumGroupNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR gamma, GM_ADDR beta,
                                 GM_ADDR z, uint32_t batchSize, uint32_t outFeatures,
                                 uint32_t numGroups, uint32_t channelsPerGroup,
                                 uint32_t rowsPerCore, float invChannelsPerGroup, float eps)
    {
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->channelsPerGroup = channelsPerGroup;
        this->invCpg = invChannelsPerGroup;
        this->epsVal = eps;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t coreRowStart = blockIdx * rowsPerCore;
        uint32_t coreRowEnd = coreRowStart + rowsPerCore;
        if (coreRowEnd > batchSize) {
            coreRowEnd = batchSize;
        }
        this->myRowCount = (coreRowEnd > coreRowStart) ? (coreRowEnd - coreRowStart) : 0;

        xGm.SetGlobalBuffer((__gm__ float*)x + coreRowStart * outFeatures, this->myRowCount * outFeatures);
        zGm.SetGlobalBuffer((__gm__ float*)z + coreRowStart * outFeatures, this->myRowCount * outFeatures);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, outFeatures);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, outFeatures);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, outFeatures);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(biasBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(gammaBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(betaBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(sigmoidBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(reduceWorkBuf, 1024 * sizeof(float));
        pipe.InitBuffer(reduceDstBuf, 32 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->myRowCount == 0) {
            return;
        }
        LoadConstants();
        for (uint32_t row = 0; row < this->myRowCount; row++) {
            CopyIn(row);
            Compute();
            CopyOut(row);
        }
    }

private:
    __aicore__ inline void LoadConstants()
    {
        AscendC::LocalTensor<float> biasL = biasBuf.Get<float>();
        AscendC::LocalTensor<float> gammaL = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaL = betaBuf.Get<float>();
        AscendC::DataCopy(biasL, biasGm, outFeatures);
        AscendC::DataCopy(gammaL, gammaGm, outFeatures);
        AscendC::DataCopy(betaL, betaGm, outFeatures);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(0);
    }

    __aicore__ inline void CopyIn(uint32_t row)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[row * outFeatures], outFeatures);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::LocalTensor<float> biasL = biasBuf.Get<float>();
        AscendC::LocalTensor<float> gammaL = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaL = betaBuf.Get<float>();
        AscendC::LocalTensor<float> sigL = sigmoidBuf.Get<float>();
        AscendC::LocalTensor<float> workL = reduceWorkBuf.Get<float>();
        AscendC::LocalTensor<float> dstL = reduceDstBuf.Get<float>();

        uint32_t len = this->outFeatures;
        uint32_t cpg = this->channelsPerGroup;

        AscendC::Muls(sigL, xLocal, -1.0f, len);
        AscendC::Exp(sigL, sigL, len);
        AscendC::Adds(sigL, sigL, 1.0f, len);
        AscendC::Reciprocal(sigL, sigL, len);
        AscendC::Mul(xLocal, xLocal, sigL, len);

        AscendC::Add(xLocal, xLocal, biasL, len);

        for (uint32_t g = 0; g < this->numGroups; g++) {
            uint32_t off = g * cpg;

            AscendC::ReduceSum<float>(dstL, xLocal[off], workL, cpg);
            float mean = dstL.GetValue(0) * this->invCpg;

            AscendC::Adds(xLocal[off], xLocal[off], -mean, cpg);

            AscendC::Mul(sigL[off], xLocal[off], xLocal[off], cpg);
            AscendC::ReduceSum<float>(dstL, sigL[off], workL, cpg);
            float var = dstL.GetValue(0) * this->invCpg;

            float invStd = 1.0f / sqrtf(var + this->epsVal);

            AscendC::Muls(xLocal[off], xLocal[off], invStd, cpg);
            AscendC::Mul(xLocal[off], xLocal[off], gammaL[off], cpg);
            AscendC::Add(zLocal[off], xLocal[off], betaL[off], cpg);
        }

        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t row)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[row * outFeatures], zLocal, outFeatures);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> betaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sigmoidBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceWorkBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceDstBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> gammaGm;
    AscendC::GlobalTensor<float> betaGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t outFeatures;
    uint32_t numGroups;
    uint32_t channelsPerGroup;
    uint32_t myRowCount;
    float invCpg;
    float epsVal;
};

extern "C" __global__ __aicore__ void matmul_swish_sum_group_norm_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR gamma, GM_ADDR beta,
    GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulSwishSumGroupNorm op;
    op.Init(x, bias, gamma, beta, z,
            tiling_data.batchSize, tiling_data.outFeatures,
            tiling_data.numGroups, tiling_data.channelsPerGroup,
            tiling_data.rowsPerCore,
            tiling_data.invChannelsPerGroup, tiling_data.eps);
    op.Process();
}

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, num_groups, bias_shape):
        super(ModelNew, self).__init__()
        self.matmul = torch.nn.Linear(in_features, out_features)
        self.bias = torch.nn.Parameter(torch.randn(bias_shape))
        self.group_norm = torch.nn.GroupNorm(num_groups, out_features)

    def forward(self, x):
        return custom_ops_lib.matmul_swish_sum_group_norm_custom(
            x,
            self.matmul.weight,
