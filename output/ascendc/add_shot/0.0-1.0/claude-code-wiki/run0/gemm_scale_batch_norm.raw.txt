kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmScaleBatchNorm {
public:
    __aicore__ inline KernelGemmScaleBatchNorm() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR a, GM_ADDR b, GM_ADDR z,
                                 uint32_t rowsPerCore, uint32_t outFeatures)
    {
        this->rowsPerCore = rowsPerCore;
        this->outFeatures = outFeatures;

        uint32_t coreOffset = rowsPerCore * outFeatures * AscendC::GetBlockIdx();
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + coreOffset, rowsPerCore * outFeatures);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + coreOffset, rowsPerCore * outFeatures);
        aGm.SetGlobalBuffer((__gm__ DTYPE_A *)a, outFeatures);
        bGm.SetGlobalBuffer((__gm__ DTYPE_B *)b, outFeatures);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, outFeatures * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, outFeatures * sizeof(DTYPE_Z));
        pipe.InitBuffer(aBuf, outFeatures * sizeof(DTYPE_A));
        pipe.InitBuffer(bBuf, outFeatures * sizeof(DTYPE_B));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_A> aLocal = aBuf.Get<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = bBuf.Get<DTYPE_B>();
        AscendC::DataCopy(aLocal, aGm, outFeatures);
        AscendC::DataCopy(bLocal, bGm, outFeatures);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

        for (uint32_t i = 0; i < rowsPerCore; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * outFeatures], outFeatures);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(uint32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<DTYPE_A> aLocal = aBuf.Get<DTYPE_A>();
        AscendC::LocalTensor<DTYPE_B> bLocal = bBuf.Get<DTYPE_B>();

        AscendC::Mul(zLocal, xLocal, aLocal, outFeatures);
        AscendC::Add(zLocal, zLocal, bLocal, outFeatures);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[progress * outFeatures], zLocal, outFeatures);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> aBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> bBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_A> aGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t rowsPerCore;
    uint32_t outFeatures;
};

extern "C" __global__ __aicore__ void gemm_scale_batch_norm_custom(
    GM_ADDR x, GM_ADDR a, GM_ADDR b, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmScaleBatchNorm op;
    op.Init(x, a, b, z, tiling_data.rowsPerCore, tiling_data.outFeatures);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, scale_shape, eps=1e-5, momentum=0.1):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.scale = nn.Parameter(torch.randn(scale_shape))
        self.bn = nn.BatchNorm1d(out_features, eps=eps, momentum=momentum)

    def forward(self, x):
        y = self.gemm(x)
        y = y * self.scale

        if self.training:
            mean = y.mean(dim=0)
            var = y.var(dim=0, unbiased=False)
        else:
            mean = self.bn.running_mean
            var = self.bn.running_var

        inv_std = 1.0 / torch.sqrt(var + self.bn.eps)
        a_param = (self.bn.weight * inv_std).contiguous()
        b_param = (self.bn.bias - mean * a_param).contiguous()

        y = y.contiguous()
        return custom_ops_lib.gemm_scale_batch_norm_custom(y, a_param, b_param)
'''
