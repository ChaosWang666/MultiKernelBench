The approach is to keep ConvTranspose3d and BatchNorm3d in PyTorch (too complex for custom kernel) and fuse the two consecutive AvgPool3d(kernel_size=2) into a single 4×4×4 average pooling custom kernel. Two consecutive AvgPool3d(2) is mathematically equivalent to averaging over a 4×4×4 window with stride 4 (divisor = 64).

kernel_src="""
#include "kernel_operator.h"

class KernelFusedPool {
public:
    __aicore__ inline KernelFusedPool() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t totalSlices, uint32_t inD, uint32_t inH, uint32_t inW,
                                 uint32_t outD, uint32_t outH, uint32_t outW)
    {
        this->blockIdx = AscendC::GetBlockIdx();
        this->blockNum = AscendC::GetBlockNum();
        this->totalSlices = totalSlices;
        this->inD = inD;
        this->inH = inH;
        this->inW = inW;
        this->outD = outD;
        this->outH = outH;
        this->outW = outW;
        this->inSliceSize = inD * inH * inW;
        this->outSliceSize = outD * outH * outW;

        xGm.SetGlobalBuffer((__gm__ float *)x, (uint64_t)totalSlices * inSliceSize);
        yGm.SetGlobalBuffer((__gm__ float *)y, (uint64_t)totalSlices * outSliceSize);
    }

    __aicore__ inline void Process()
    {
        uint32_t slicesPerBlock = (totalSlices + blockNum - 1) / blockNum;
        uint32_t startSlice = blockIdx * slicesPerBlock;
        uint32_t endSlice = startSlice + slicesPerBlock;
        if (endSlice > totalSlices) endSlice = totalSlices;

        float invPool = 1.0f / 64.0f;

        for (uint32_t s = startSlice; s < endSlice; s++) {
            uint64_t inBase = (uint64_t)s * inSliceSize;
            uint64_t outBase = (uint64_t)s * outSliceSize;

            for (uint32_t od = 0; od < outD; od++) {
                for (uint32_t oh = 0; oh < outH; oh++) {
                    for (uint32_t ow = 0; ow < outW; ow++) {
                        float sum = 0.0f;
                        for (uint32_t dd = 0; dd < 4; dd++) {
                            uint32_t id = od * 4 + dd;
                            for (uint32_t dh = 0; dh < 4; dh++) {
                                uint32_t ih = oh * 4 + dh;
                                uint64_t rowBase = inBase + (uint64_t)id * inH * inW + (uint64_t)ih * inW;
                                uint32_t iwBase = ow * 4;
                                sum += xGm.GetValue(rowBase + iwBase);
                                sum += xGm.GetValue(rowBase + iwBase + 1);
                                sum += xGm.GetValue(rowBase + iwBase + 2);
                                sum += xGm.GetValue(rowBase + iwBase + 3);
                            }
                        }
                        uint64_t outIdx = outBase + (uint64_t)od * outH * outW + (uint64_t)oh * outW + ow;
                        yGm.SetValue(outIdx, sum * invPool);
                    }
                }
            }
        }
    }

private:
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t blockIdx, blockNum;
    uint32_t totalSlices, inD, inH, inW, outD, outH, outW;
    uint32_t inSliceSize, outSliceSize;
};

extern "C" __global__ __aicore__ void conv_transpose3d_batch_norm_avg_pool_avg_pool_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedPool op;
    op.Init(x, y, tiling_data.totalSlices, tiling_data.inD, tiling_data.inH, tiling_data.inW,
            tiling_data.outD, tiling_data.outH, tiling_data.outW);
    op.Process();
}
