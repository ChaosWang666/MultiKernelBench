kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;
constexpr int32_t W_ELEM = 64;
constexpr int32_t NUM_ROWS = 16;
constexpr int32_t OUT_PAD = 32;

class KernelFusedAvgPool {
public:
    __aicore__ inline KernelFusedAvgPool() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t N_in, uint32_t C_in,
                                 uint32_t D_in_, uint32_t H_in_, uint32_t W_in_,
                                 uint32_t D_out_, uint32_t H_out_, uint32_t W_out_,
                                 uint32_t totalJobs_, uint32_t jobsPerBlock_,
                                 float invK_)
    {
        this->N = N_in;
        this->C = C_in;
        this->D_in = D_in_;
        this->H_in = H_in_;
        this->W_in = W_in_;
        this->D_out = D_out_;
        this->H_out = H_out_;
        this->W_out = W_out_;
        this->totalJobs = totalJobs_;
        this->jobsPerBlock = jobsPerBlock_;
        this->invK = invK_;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->startJob = blockIdx * jobsPerBlock_;
        uint32_t ej = this->startJob + jobsPerBlock_;
        if (ej > totalJobs_) ej = totalJobs_;
        this->endJob = ej;

        xGm.SetGlobalBuffer((__gm__ float*)x);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, NUM_ROWS * W_ELEM * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, OUT_PAD * sizeof(float));
        pipe.InitBuffer(sumBuf, W_ELEM * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t job = this->startJob; job < this->endJob; job++) {
            ProcessJob(job);
        }
    }

private:
    __aicore__ inline void ProcessJob(uint32_t job)
    {
        uint32_t ohIdx = job % this->H_out;
        uint32_t t1 = job / this->H_out;
        uint32_t odIdx = t1 % this->D_out;
        uint32_t t2 = t1 / this->D_out;
        uint32_t cIdx = t2 % this->C;
        uint32_t nIdx = t2 / this->C;

        uint32_t baseD = 4 * odIdx;
        uint32_t baseH = 4 * ohIdx;

        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();

        AscendC::DataCopyExtParams copyInParams;
        copyInParams.blockCount = 1;
        copyInParams.blockLen = 63 * sizeof(float);
        copyInParams.srcStride = 0;
        copyInParams.dstStride = 0;
        AscendC::DataCopyPadExtParams<float> padParams;
        padParams.isPad = true;
        padParams.paddingValue = 0;
        padParams.leftPadding = 0;
        padParams.rightPadding = 1;

        uint64_t sliceBase = ((uint64_t)nIdx * this->C + cIdx) * (uint64_t)this->D_in * this->H_in * this->W_in;

        for (int32_t dk = 0; dk < 4; dk++) {
            uint64_t dOff = sliceBase + (uint64_t)(baseD + dk) * this->H_in * this->W_in;
            for (int32_t hk = 0; hk < 4; hk++) {
                uint64_t rowOff = dOff + (uint64_t)(baseH + hk) * this->W_in;
                int32_t rowIdx = dk * 4 + hk;
                AscendC::DataCopyPad(xLocal[rowIdx * W_ELEM], xGm[rowOff], copyInParams, padParams);
            }
        }

        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> sumRow = sumBuf.Get<float>();

        AscendC::Add(sumRow, xLocal, xLocal[W_ELEM], W_ELEM);
        for (int32_t r = 2; r < NUM_ROWS; r++) {
            AscendC::Add(sumRow, sumRow, xLocal[r * W_ELEM], W_ELEM);
        }

        inQueueX.FreeTensor(xLocal);

        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        AscendC::PipeBarrier<PIPE_V>();
        AscendC::TEventID eventIdVS = static_cast<AscendC::TEventID>(0);
        AscendC::SetFlag<AscendC::HardEvent::V_S>(eventIdVS);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(eventIdVS);

        for (int32_t ow = 0; ow < 15; ow++) {
            float s0 = sumRow.GetValue(4 * ow);
            float s1 = sumRow.GetValue(4 * ow + 1);
            float s2 = sumRow.GetValue(4 * ow + 2);
            float s3 = sumRow.GetValue(4 * ow + 3);
            yLocal.SetValue(ow, (s0 + s1 + s2 + s3) * this->invK);
        }

        AscendC::TEventID eventIdSMTE = static_cast<AscendC::TEventID>(0);
        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(eventIdSMTE);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(eventIdSMTE);

        outQueueY.EnQue<float>(yLocal);
        yLocal = outQueueY.DeQue<float>();

        uint64_t outOffset = ((uint64_t)nIdx * this->C + cIdx) * (uint64_t)this->D_out * this->H_out * this->W_out
                            + (uint64_t)odIdx * this->H_out * this->W_out
                            + (uint64_t)ohIdx * this->W_out;

        AscendC::DataCopyExtParams copyOutParams;
        copyOutParams.blockCount = 1;
        copyOutParams.blockLen = 15 * sizeof(float);
        copyOutParams.srcStride = 0;
        copyOutParams.dstStride = 0;

        AscendC::DataCopyPad(yGm[outOffset], yLocal, copyOutParams);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t N, C, D_in, H_in, W_in, D_out, H_out, W_out;
    uint32_t totalJobs, jobsPerBlock;
    uint32_t startJob, endJob;
    float invK;
};

extern "C" __global__ __aicore__ void conv_transpose3d_batch_norm_avg_pool_avg_pool_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedAvgPool op;
    op.Init(x, y,
            tiling_data.N, tiling_data.C,
            tiling_data.D_in, tiling_data.H_in, tiling_data.W_in,
            tiling_data.D_out, tiling_data.H_out, tiling_data.W_out,
            tiling_data.totalJobs, tiling_data.jobsPerBlock,
            tiling_data.invK);
    op.Process();
}
