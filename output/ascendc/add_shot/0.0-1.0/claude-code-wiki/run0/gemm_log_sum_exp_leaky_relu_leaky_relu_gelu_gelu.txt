kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedGemmLSE {
public:
    __aicore__ inline KernelFusedGemmLSE() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR bias, GM_ADDR z,
                                 uint32_t kDim, uint32_t nDim, uint32_t rowsPerCore)
    {
        this->kDim = kDim;
        this->nDim = nDim;
        this->rowsPerCore = rowsPerCore;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerCore;

        xGm.SetGlobalBuffer((__gm__ float*)x + startRow * kDim, rowsPerCore * kDim);
        wGm.SetGlobalBuffer((__gm__ float*)w, nDim * kDim);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, nDim);
        zGm.SetGlobalBuffer((__gm__ float*)z + startRow, rowsPerCore);

        pipe.InitBuffer(xBuf, kDim * sizeof(float));
        pipe.InitBuffer(yBuf, nDim * sizeof(float));
        pipe.InitBuffer(biasBuf, nDim * sizeof(float));
        pipe.InitBuffer(maxBuf, 64 * sizeof(float));
        pipe.InitBuffer(sumBuf, 64 * sizeof(float));
        pipe.InitBuffer(scalarBuf, 64 * sizeof(float));
        pipe.InitBuffer(workBuf, 1024 * sizeof(float));
        pipe.InitBuffer(outBuf, 64 * sizeof(float));
        pipe.InitBuffer(wQueue, BUFFER_NUM, kDim * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> yLocal = yBuf.Get<float>();
        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();

        AscendC::DataCopy(biasLocal, biasGm, nDim);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_S>(EVENT_ID0);
        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID1);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID1);

        for (uint32_t i = 0; i < rowsPerCore; i++) {
            AscendC::DataCopy(xLocal, xGm[i * kDim], kDim);
            AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

            for (uint32_t j = 0; j < nDim; j++) {
                AscendC::LocalTensor<float> wLocal = wQueue.AllocTensor<float>();
                AscendC::DataCopy(wLocal, wGm[j * kDim], kDim);
                wQueue.EnQue(wLocal);
                wLocal = wQueue.DeQue<float>();

                AscendC::Mul(wLocal, xLocal, wLocal, kDim);
                AscendC::PipeBarrier<PIPE_V>();
                AscendC::ReduceSum<float>(wLocal, wLocal, workLocal, kDim);

                AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

                float val = wLocal.GetValue(0) + biasLocal.GetValue(j);
                yLocal.SetValue(j, val);

                wQueue.FreeTensor(wLocal);
            }

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

            AscendC::ReduceMax<float>(scalarLocal, yLocal, workLocal, nDim);
            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            float max_val = scalarLocal.GetValue(0);
            maxLocal.SetValue(i, max_val);

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

            AscendC::Adds(yLocal, yLocal, -max_val, nDim);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Exp(yLocal, yLocal, nDim);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum<float>(scalarLocal, yLocal, workLocal, nDim);

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            float sum_exp = scalarLocal.GetValue(0);
            sumLocal.SetValue(i, sum_exp);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        AscendC::Ln(sumLocal, sumLocal, rowsPerCore);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Add(outLocal, maxLocal, sumLocal, rowsPerCore);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Muls(scalarLocal, outLocal, 0.0001f, rowsPerCore);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Max(outLocal, outLocal, scalarLocal, rowsPerCore);
        AscendC::PipeBarrier<PIPE_V>();

        for (int k = 0; k < 2; k++) {
            AscendC::Muls(scalarLocal, outLocal, 0.70710678118654752f, rowsPerCore);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Erf(scalarLocal, scalarLocal, rowsPerCore);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Adds(scalarLocal, scalarLocal, 1.0f, rowsPerCore);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Mul(outLocal, outLocal, scalarLocal, rowsPerCore);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::Muls(outLocal, outLocal, 0.5f, rowsPerCore);
            AscendC::PipeBarrier<PIPE_V>();
        }

        AscendC::SetFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);
        AscendC::DataCopy(zGm, outLocal, rowsPerCore);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> yBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> biasBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> outBuf;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> wQueue;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t kDim;
    uint32_t nDim;
    uint32_t rowsPerCore;
};

extern "C" __global__ __aicore__ void gemm_log_sum_exp_leaky_relu_leaky_relu_gelu_gelu_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR z,
    GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedGemmLSE op;
    op.Init(x, weight, bias, z, tiling_data.kDim, tiling_data.nDim, tiling_data.rowsPerCore);
    op.Process();
}
