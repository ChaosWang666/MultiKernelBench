kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmGroupNormMinBiasAdd {
public:
    __aicore__ inline KernelGemmGroupNormMinBiasAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                 uint32_t batchSize, uint32_t outFeatures,
                                 uint32_t channelsPerBlock)
    {
        this->batchSize = batchSize;
        this->outFeatures = outFeatures;
        this->channelsPerBlock = channelsPerBlock;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t channelStart = blockIdx * channelsPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x, batchSize);
        biasGm.SetGlobalBuffer((__gm__ DTYPE_BIAS *)bias + channelStart, channelsPerBlock);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + channelStart * batchSize, channelsPerBlock * batchSize);

        pipe.InitBuffer(inQueueX, 1, batchSize * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueBias, 1, channelsPerBlock * sizeof(DTYPE_BIAS));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, batchSize * sizeof(DTYPE_Z));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_X> xAlloc = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xAlloc, xGm, this->batchSize);
        inQueueX.EnQue(xAlloc);
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();

        AscendC::LocalTensor<DTYPE_BIAS> biasAlloc = inQueueBias.AllocTensor<DTYPE_BIAS>();
        AscendC::DataCopy(biasAlloc, biasGm, this->channelsPerBlock);
        inQueueBias.EnQue(biasAlloc);
        AscendC::LocalTensor<DTYPE_BIAS> biasLocal = inQueueBias.DeQue<DTYPE_BIAS>();

        int32_t loopCount = static_cast<int32_t>(this->channelsPerBlock);
        for (int32_t c = 0; c < loopCount; c++) {
            AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
            float biasVal = biasLocal.GetValue(c);
            AscendC::Adds(zLocal, xLocal, biasVal, this->batchSize);
            outQueueZ.EnQue<DTYPE_Z>(zLocal);

            AscendC::LocalTensor<DTYPE_Z> zOut = outQueueZ.DeQue<DTYPE_Z>();
            AscendC::DataCopy(zGm[c * this->batchSize], zOut, this->batchSize);
            outQueueZ.FreeTensor(zOut);
        }

        inQueueX.FreeTensor(xLocal);
        inQueueBias.FreeTensor(biasLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_BIAS> biasGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t batchSize;
    uint32_t outFeatures;
    uint32_t channelsPerBlock;
};

extern "C" __global__ __aicore__ void gemm_group_norm_min_bias_add_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormMinBiasAdd op;
    op.Init(x, bias, z, tiling_data.batchSize, tiling_data.outFeatures, tiling_data.channelsPerBlock);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, bias_shape):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.group_norm = nn.GroupNorm(num_groups, out_features)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.out_features = out_features

    def forward(self, x):
        x = self.gemm(x)
        x = self.group_norm(x)
        x_min = torch.min(x, dim=1)[0].contiguous()
        batch_size = x_min.shape[0]
        bias_flat = self.bias.reshape(-1).contiguous()
        out = custom_ops_lib.gemm_group_norm_min_bias_add_custom(x_min, bias_flat)
        return out.reshape(1, self.out_features, batch_size, 1)
'''
