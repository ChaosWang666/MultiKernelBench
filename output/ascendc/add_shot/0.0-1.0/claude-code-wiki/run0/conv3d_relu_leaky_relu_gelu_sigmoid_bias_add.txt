kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr uint32_t MAX_CHANNEL = 1024;

class KernelConv3dReluLeakyReluGeluSigmoidBiasAdd {
public:
    __aicore__ inline KernelConv3dReluLeakyReluGeluSigmoidBiasAdd() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                  uint32_t batchSize, uint32_t channelNum,
                                  uint32_t innerLength, uint32_t tileNum,
                                  uint32_t tileLength)
    {
        this->batchSize = batchSize;
        this->channelNum = channelNum;
        this->innerLength = innerLength;
        this->tileNum = tileNum;
        this->tileLength = tileLength;
        this->channelIdx = AscendC::GetBlockIdx();

        uint32_t total = batchSize * channelNum * innerLength;
        xGm.SetGlobalBuffer((__gm__ float *)x, total);
        yGm.SetGlobalBuffer((__gm__ float *)y, total);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, channelNum);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf1, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, this->tileLength * sizeof(float));
        pipe.InitBuffer(biasQueue, 1, MAX_CHANNEL * sizeof(float));

        LoadBias();
    }

    __aicore__ inline void Process()
    {
        for (uint32_t n = 0; n < this->batchSize; n++) {
            uint32_t baseOffset = (n * this->channelNum + this->channelIdx) * this->innerLength;
            for (uint32_t i = 0; i < this->tileNum; i++) {
                uint32_t offset = baseOffset + i * this->tileLength;
                CopyIn(offset);
                Compute();
                CopyOut(offset);
            }
        }
    }

private:
    __aicore__ inline void LoadBias()
    {
        AscendC::LocalTensor<float> biasLocal = biasQueue.AllocTensor<float>();
        uint32_t loadLen = (this->channelNum + 7) / 8 * 8;
        if (loadLen > MAX_CHANNEL) {
            loadLen = MAX_CHANNEL;
        }
        AscendC::DataCopy(biasLocal, biasGm, loadLen);
        biasQueue.EnQue(biasLocal);
        biasLocal = biasQueue.DeQue<float>();
        this->biasVal = biasLocal.GetValue(this->channelIdx);
        biasQueue.FreeTensor(biasLocal);
    }

    __aicore__ inline void CopyIn(uint32_t offset)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> t1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> t2 = tmpBuf2.Get<float>();

        AscendC::Maxs(t1, xLocal, 0.0f, this->tileLength);

        AscendC::Mul(t2, t1, t1, this->tileLength);
        AscendC::Mul(t2, t2, t1, this->tileLength);
        AscendC::Muls(t2, t2, 0.044715f, this->tileLength);
        AscendC::Add(t2, t2, t1, this->tileLength);
        AscendC::Muls(t2, t2, -1.5957691f, this->tileLength);
        AscendC::Exp(t2, t2, this->tileLength);
        AscendC::Adds(t2, t2, 1.0f, this->tileLength);
        AscendC::Div(t1, t1, t2, this->tileLength);

        AscendC::Muls(t2, t1, -1.0f, this->tileLength);
        AscendC::Exp(t2, t2, this->tileLength);
        AscendC::Adds(t2, t2, 1.0f, this->tileLength);
        AscendC::Duplicate(t1, 1.0f, this->tileLength);
        AscendC::Div(t1, t1, t2, this->tileLength);

        AscendC::Adds(yLocal, t1, this->biasVal, this->tileLength);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[offset], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> biasQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> biasGm;
    uint32_t batchSize;
    uint32_t channelNum;
    uint32_t innerLength;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t channelIdx;
    float biasVal;
};

extern "C" __global__ __aicore__ void conv3d_relu_leaky_relu_gelu_sigmoid_bias_add_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dReluLeakyReluGeluSigmoidBiasAdd op;
    op.Init(x, bias, y,
            tiling_data.batchSize,
            tiling_data.channelNum,
            tiling_data.innerLength,
            tiling_data.tileNum,
            tiling_data.tileLength);
    op.Process();
}
