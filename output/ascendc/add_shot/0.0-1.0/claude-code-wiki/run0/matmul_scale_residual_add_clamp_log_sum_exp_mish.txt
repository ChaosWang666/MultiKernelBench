kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batchSize, uint32_t hiddenSize, uint32_t rowsPerCore,
                                float combinedScale, float clampMin, float clampMax)
    {
        this->hiddenSize = hiddenSize;
        this->rowsPerCore = rowsPerCore;
        this->combinedScale = combinedScale;
        this->clampMin = clampMin;
        this->clampMax = clampMax;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t startRow = blockIdx * rowsPerCore;

        inputGm.SetGlobalBuffer((__gm__ float*)x + startRow * hiddenSize, rowsPerCore * hiddenSize);
        outputGm.SetGlobalBuffer((__gm__ float*)y + startRow, rowsPerCore);

        uint32_t rowBytes = rowsPerCore * sizeof(float);
        uint32_t alignedRowBytes = ((rowBytes + 31U) / 32U) * 32U;
        if (alignedRowBytes < 32U) { alignedRowBytes = 32U; }

        pipe.InitBuffer(inQueue, BUFFER_NUM, hiddenSize * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, alignedRowBytes);
        pipe.InitBuffer(sumBuf, alignedRowBytes);
        pipe.InitBuffer(reduceTmpBuf, 64);
        pipe.InitBuffer(tmp1Buf, alignedRowBytes);
        pipe.InitBuffer(tmp2Buf, alignedRowBytes);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceTmpBuf.Get<float>();

        for (uint32_t i = 0; i < rowsPerCore; i++) {
            AscendC::LocalTensor<float> rowLocal = inQueue.AllocTensor<float>();
            AscendC::DataCopy(rowLocal, inputGm[i * hiddenSize], hiddenSize);
            inQueue.EnQue(rowLocal);
            rowLocal = inQueue.DeQue<float>();

            AscendC::Muls(rowLocal, rowLocal, combinedScale, static_cast<int32_t>(hiddenSize));
            AscendC::Mins(rowLocal, rowLocal, clampMax, static_cast<int32_t>(hiddenSize));
            AscendC::Maxs(rowLocal, rowLocal, clampMin, static_cast<int32_t>(hiddenSize));
            AscendC::Exp(rowLocal, rowLocal, static_cast<int32_t>(hiddenSize));

            AscendC::ReduceSum<float>(reduceTmp, rowLocal, reduceTmp, static_cast<int32_t>(hiddenSize));

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

            float sumVal = reduceTmp.GetValue(0);
            sumLocal.SetValue(i, sumVal);

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

            inQueue.FreeTensor(rowLocal);
        }

        AscendC::LocalTensor<float> tmp1 = tmp1Buf.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmp2Buf.Get<float>();

        int32_t count = static_cast<int32_t>(rowsPerCore);

        AscendC::Ln(tmp1, sumLocal, count);

        AscendC::Exp(tmp2, tmp1, count);
        AscendC::Adds(tmp2, tmp2, 1.0f, count);
        AscendC::Ln(tmp2, tmp2, count);

        AscendC::Muls(tmp2, tmp2, -2.0f, count);
        AscendC::Exp(tmp2, tmp2, count);

        AscendC::LocalTensor<float> outLocal = outQueue.AllocTensor<float>();

        AscendC::Muls(outLocal, tmp2, -1.0f, count);
        AscendC::Adds(outLocal, outLocal, 1.0f, count);
        AscendC::Adds(tmp2, tmp2, 1.0f, count);
        AscendC::Div(tmp2, outLocal, tmp2, count);

        AscendC::Mul(tmp1, tmp1, tmp1, count);
        AscendC::Mul(outLocal, tmp1, tmp2, count);

        outQueue.EnQue(outLocal);
        outLocal = outQueue.DeQue<float>();
        AscendC::DataCopy(outputGm, outLocal, rowsPerCore);
        outQueue.FreeTensor(outLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf, reduceTmpBuf, tmp1Buf, tmp2Buf;
    AscendC::GlobalTensor<float> inputGm;
    AscendC::GlobalTensor<float> outputGm;
    uint32_t hiddenSize;
    uint32_t rowsPerCore;
    float combinedScale;
    float clampMin;
    float clampMax;
};

extern "C" __global__ __aicore__ void matmul_scale_residual_add_clamp_log_sum_exp_mish_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, y,
            tiling_data.batchSize, tiling_data.hiddenSize, tiling_data.rowsPerCore,
            tiling_data.combinedScale, tiling_data.clampMin, tiling_data.clampMax);
    op.Process();
}
