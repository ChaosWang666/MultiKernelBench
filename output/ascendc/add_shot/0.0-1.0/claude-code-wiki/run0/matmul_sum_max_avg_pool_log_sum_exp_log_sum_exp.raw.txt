kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelRowSum {
public:
    __aicore__ inline KernelRowSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t featureSize, uint32_t rowsPerCore)
    {
        this->featureSize = featureSize;
        this->rowsPerCore = rowsPerCore;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockIdx * rowsPerCore * featureSize, rowsPerCore * featureSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + blockIdx * rowsPerCore, rowsPerCore);
        pipe.InitBuffer(inQueueX, BUFFER_NUM, featureSize * sizeof(DTYPE_X));
        pipe.InitBuffer(workBuf, featureSize * sizeof(DTYPE_X));
        pipe.InitBuffer(accBuf, 64 * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> accLocal = accBuf.Get<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_X> workLocal = workBuf.Get<DTYPE_X>();

        for (uint32_t i = 0; i < rowsPerCore; i++) {
            CopyIn(i);
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
            AscendC::ReduceSum<DTYPE_X>(workLocal, xLocal, workLocal, featureSize);
            inQueueX.FreeTensor(xLocal);

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

            DTYPE_X val = workLocal.GetValue(0);
            accLocal.SetValue(i, val);

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        AscendC::DataCopy(yGm, accLocal, rowsPerCore);
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[progress * featureSize], featureSize);
        inQueueX.EnQue(xLocal);
    }

    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t featureSize;
    uint32_t rowsPerCore;
};

extern "C" __global__ __aicore__ void matmul_sum_max_avg_pool_log_sum_exp_log_sum_exp_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelRowSum op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.featureSize, tiling_data.rowsPerCore);
    op.Process();
}
