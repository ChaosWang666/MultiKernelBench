kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFusedMatReduce {
public:
    __aicore__ inline KernelFusedMatReduce() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y,
                                uint32_t batchSize, uint32_t inFeatures, uint32_t batchPerCore)
    {
        this->inFeatures = inFeatures;
        this->batchPerCore = batchPerCore;

        uint32_t coreOffset = AscendC::GetBlockIdx() * batchPerCore;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + coreOffset * inFeatures, batchPerCore * inFeatures);
        wGm.SetGlobalBuffer((__gm__ DTYPE_W *)w, inFeatures);
        bGm.SetGlobalBuffer((__gm__ DTYPE_B *)b, 1);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + coreOffset, batchPerCore);

        uint32_t outBytes = batchPerCore * sizeof(DTYPE_Y);
        uint32_t outBufBytes = ((outBytes + 31) / 32) * 32;
        if (outBufBytes < 32) {
            outBufBytes = 32;
        }

        pipe.InitBuffer(wBuf, inFeatures * sizeof(DTYPE_W));
        pipe.InitBuffer(xQue, BUFFER_NUM, inFeatures * sizeof(DTYPE_X));
        pipe.InitBuffer(tmpBuf, inFeatures * sizeof(DTYPE_X));
        pipe.InitBuffer(workBuf, 4096);
        pipe.InitBuffer(reduceBuf, 32);
        pipe.InitBuffer(outQue, 1, outBufBytes);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_W> wLocal = wBuf.Get<DTYPE_W>();
        AscendC::DataCopy(wLocal, wGm, this->inFeatures);

        AscendC::LocalTensor<DTYPE_X> tmpLocal = tmpBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> workLocal = workBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> reduceLocal = reduceBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> outLocal = outQue.AllocTensor<DTYPE_Y>();

        float biasSum = bGm.GetValue(0);

        for (uint32_t i = 0; i < this->batchPerCore; i++) {
            AscendC::LocalTensor<DTYPE_X> xLocal = xQue.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xLocal, xGm[i * this->inFeatures], this->inFeatures);
            xQue.EnQue(xLocal);

            AscendC::LocalTensor<DTYPE_X> xL = xQue.DeQue<DTYPE_X>();
            AscendC::Mul(tmpLocal, xL, wLocal, this->inFeatures);
            AscendC::PipeBarrier<PIPE_V>();
            AscendC::ReduceSum<DTYPE_X>(reduceLocal, tmpLocal, workLocal, this->inFeatures);
            AscendC::PipeBarrier<PIPE_V>();
            float dotResult = reduceLocal.GetValue(0);
            outLocal.SetValue(i, dotResult + biasSum);
            xQue.FreeTensor(xL);
        }

        outQue.EnQue<DTYPE_Y>(outLocal);
        AscendC::LocalTensor<DTYPE_Y> outFinal = outQue.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm, outFinal, this->batchPerCore);
        outQue.FreeTensor(outFinal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> wBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> xQue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQue;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_W> wGm;
    AscendC::GlobalTensor<DTYPE_B> bGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t inFeatures;
    uint32_t batchPerCore;
};

extern "C" __global__ __aicore__ void matmul_sum_max_avg_pool_log_sum_exp_log_sum_exp_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR b, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedMatReduce op;
    op.Init(x, w, b, y, tiling_data.batchSize, tiling_data.inFeatures, tiling_data.batchPerCore);
    op.Process();
}
