kernel_src="""
#include "kernel_operator.h"

class KernelConvTranspose3dLayerNormGeluScaling {
public:
    __aicore__ inline KernelConvTranspose3dLayerNormGeluScaling() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
                                uint32_t totalRows, uint32_t normSize, uint32_t alignedNorm,
                                float invNormSize, float eps)
    {
        this->totalRows = totalRows;
        this->normSize = normSize;
        this->alignedNorm = alignedNorm;
        this->invNormSize = invNormSize;
        this->eps = eps;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t rowsPerBlock = (totalRows + blockNum - 1) / blockNum;
        this->startRow = blockIdx * rowsPerBlock;
        uint32_t endCand = this->startRow + rowsPerBlock;
        this->endRow = (endCand < totalRows) ? endCand : totalRows;
        if (this->startRow > totalRows) {
            this->startRow = totalRows;
        }

        uint64_t totalElems = (uint64_t)totalRows * (uint64_t)normSize;
        xGm.SetGlobalBuffer((__gm__ float*)x, totalElems);
        yGm.SetGlobalBuffer((__gm__ float*)y, totalElems);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, normSize);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, normSize);

        pipe.InitBuffer(gammaBuf, alignedNorm * sizeof(float));
        pipe.InitBuffer(betaBuf, alignedNorm * sizeof(float));
        pipe.InitBuffer(xBuf, alignedNorm * sizeof(float));
        pipe.InitBuffer(tmpBuf, alignedNorm * sizeof(float));
        pipe.InitBuffer(redBuf, 32);
    }

    __aicore__ inline void Process()
    {
        if (this->startRow >= this->endRow) {
            return;
        }

        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();

        AscendC::DataCopy(gammaLocal, gammaGm, alignedNorm);
        AscendC::DataCopy(betaLocal, betaGm, alignedNorm);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t row = startRow; row < endRow; row++) {
            ProcessOneRow(row, gammaLocal, betaLocal);
        }
    }

private:
    __aicore__ inline void ProcessOneRow(uint32_t row,
                                          AscendC::LocalTensor<float>& gammaLocal,
                                          AscendC::LocalTensor<float>& betaLocal)
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> redLocal = redBuf.Get<float>();

        uint64_t offset = (uint64_t)row * (uint64_t)normSize;

        AscendC::DataCopy(xLocal, xGm[offset], alignedNorm);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::ReduceSum(redLocal, xLocal, tmpLocal, normSize);
        AscendC::PipeBarrier<PIPE_ALL>();
        float sum = redLocal.GetValue(0);
        float mean = sum * invNormSize;

        AscendC::Adds(xLocal, xLocal, -mean, normSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Mul(tmpLocal, xLocal, xLocal, normSize);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::ReduceSum(redLocal, tmpLocal, tmpLocal, normSize);
        AscendC::PipeBarrier<PIPE_ALL>();
        float sumSq = redLocal.GetValue(0);
        float var = sumSq * invNormSize;

        redLocal.SetValue(0, var + eps);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Sqrt(redLocal, redLocal, 1);
        AscendC::PipeBarrier<PIPE_ALL>();
        float stdVal = redLocal.GetValue(0);
        float invStd = 1.0f / stdVal;

        AscendC::Muls(xLocal, xLocal, invStd, normSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Mul(xLocal, xLocal, gammaLocal, normSize);
        AscendC::PipeBarrier<PIPE_ALL>();
        AscendC::Add(xLocal, xLocal, betaLocal, normSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::Mul(tmpLocal, xLocal, xLocal, normSize);
        AscendC::Mul(tmpLocal, tmpLocal, xLocal, normSize);
        AscendC::Muls(tmpLocal, tmpLocal, 0.044715f, normSize);
        AscendC::Add(tmpLocal, tmpLocal, xLocal, normSize);
        AscendC::Muls(tmpLocal, tmpLocal, -1.5957691216057308f, normSize);
        AscendC::Exp(tmpLocal, tmpLocal, normSize);
        AscendC::Adds(tmpLocal, tmpLocal, 1.0f, normSize);
        AscendC::Div(xLocal, xLocal, tmpLocal, normSize);
        AscendC::PipeBarrier<PIPE_ALL>();

        AscendC::DataCopy(yGm[offset], xLocal, alignedNorm);
        AscendC::PipeBarrier<PIPE_ALL>();
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> betaBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> redBuf;
    AscendC::GlobalTensor<float> xGm, gammaGm, betaGm, yGm;
    uint32_t totalRows;
    uint32_t normSize;
    uint32_t alignedNorm;
    float invNormSize;
    float eps;
    uint32_t startRow;
    uint32_t endRow;
};

extern "C" __global__ __aicore__ void conv_transpose3d_layer_norm_gelu_scaling_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dLayerNormGeluScaling op;
    op.Init(x, gamma, beta, y,
            tiling_data.totalRows, tiling_data.normSize, tiling_data.alignedNorm,
            tiling_data.invNormSize, tiling_data.eps);
    op.Process();
}

model_src='''
import torch
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, bias=True, eps=1e-5, scaling_factor=1.0):
        super().__init__()
        self.conv_transpose = torch.nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, bias=bias)
        self.layer_norm = torch.nn.LayerNorm(out_channels, eps=eps)
        self.eps = float(eps)
        self.scaling_factor = scaling_factor

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x.contiguous()
        gamma = self.layer_norm.weight.contiguous()
        beta = self.layer_norm.bias.contiguous()
        y = custom_ops_lib.conv_transpose3d_layer_norm_gelu_scaling_custom(x, gamma, beta, self.eps)
        y = y * self.scaling_factor
        return y
'''
