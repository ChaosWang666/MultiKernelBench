namespace optiling {
const uint32_t MAX_BLOCK_DIM = 16;
const uint32_t MIN_BATCH_PER_CORE_ALIGN = 8;
const uint32_t TILE_LEN = 8192;

static ge::graphStatus TilingFunc(gert::TilingContext* context)
{
    MatmulSigmoidSumCustomTilingData tiling;
    auto xShape = context->GetInputShape(0)->GetOriginShape();
    uint32_t batchSize = xShape.GetDim(0);
    uint32_t hiddenSize = xShape.GetDim(1);

    uint32_t blockDim = 1;
    for (uint32_t d = MAX_BLOCK_DIM; d >= 1; d /= 2) {
        if (batchSize % d == 0) {
            uint32_t bpc = batchSize / d;
            if (bpc % MIN_BATCH_PER_CORE_ALIGN == 0) {
                blockDim = d;
                break;
            }
        }
        if (d == 1) break;
    }
    if (blockDim == 0) {
        blockDim = 1;
    }

    uint32_t batchPerCore = batchSize / blockDim;

    uint32_t tileLen = TILE_LEN;
    if (tileLen > hiddenSize) {
        tileLen = hiddenSize;
    }
    while (tileLen > 0 && (hiddenSize % tileLen != 0)) {
        tileLen /= 2;
    }
    if (tileLen == 0) {
        tileLen = hiddenSize;
    }
    uint32_t numTiles = hiddenSize / tileLen;

    context->SetBlockDim(blockDim);
    tiling.set_batchSize(batchSize);
    tiling.set_hiddenSize(hiddenSize);
    tiling.set_batchPerCore(batchPerCore);
    tiling.set_tileLen(tileLen);
    tiling.set_numTiles(numTiles);
    tiling.SaveToBuffer(context->GetRawTilingData()->GetData(), context->GetRawTilingData()->GetCapacity());
    context->GetRawTilingData()->SetDataSize(tiling.GetDataSize());
    size_t *currentWorkspace = context->GetWorkspaceSizes(1);

}

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSigmoidSum {
public:
    __aicore__ inline KernelSigmoidSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batchSize, uint32_t hiddenSize,
                                uint32_t batchPerCore, uint32_t tileLen, uint32_t numTiles)
    {
        this->hiddenSize = hiddenSize;
        this->batchPerCore = batchPerCore;
        this->tileLen = tileLen;
        this->numTiles = numTiles;

        uint32_t coreOffset = AscendC::GetBlockIdx() * batchPerCore;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + coreOffset * hiddenSize, batchPerCore * hiddenSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + coreOffset, batchPerCore);

        pipe.InitBuffer(xQue, BUFFER_NUM, tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(tmpBuf, tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(workBuf, tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(reduceBuf, 32);

        uint32_t outBytes = batchPerCore * sizeof(DTYPE_Y);
        uint32_t outBufBytes = ((outBytes + 31) / 32) * 32;
        if (outBufBytes < 32) {
            outBufBytes = 32;
        }
        pipe.InitBuffer(outQue, 1, outBufBytes);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_X> tmpLocal = tmpBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> workLocal = workBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_X> reduceLocal = reduceBuf.Get<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> outLocal = outQue.AllocTensor<DTYPE_Y>();

        for (uint32_t i = 0; i < this->batchPerCore; i++) {
            float batchSum = 0.0f;
            for (uint32_t t = 0; t < this->numTiles; t++) {
                AscendC::LocalTensor<DTYPE_X> xLocal = xQue.AllocTensor<DTYPE_X>();
                AscendC::DataCopy(xLocal, xGm[i * this->hiddenSize + t * this->tileLen], this->tileLen);
                xQue.EnQue(xLocal);

                AscendC::LocalTensor<DTYPE_X> xL = xQue.DeQue<DTYPE_X>();

                AscendC::Muls(tmpLocal, xL, -1.0f, this->tileLen);
                AscendC::PipeBarrier<PIPE_V>();
                AscendC::Exp(tmpLocal, tmpLocal, this->tileLen);
                AscendC::PipeBarrier<PIPE_V>();
                AscendC::Adds(tmpLocal, tmpLocal, 1.0f, this->tileLen);
                AscendC::PipeBarrier<PIPE_V>();
                AscendC::Reciprocal(tmpLocal, tmpLocal, this->tileLen);
                AscendC::PipeBarrier<PIPE_V>();

                AscendC::ReduceSum<DTYPE_X>(reduceLocal, tmpLocal, workLocal, this->tileLen);
                xQue.FreeTensor(xL);

                AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
                AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

                batchSum += reduceLocal.GetValue(0);

                AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
                AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            }
            outLocal.SetValue(i, batchSum);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);

        outQue.EnQue<DTYPE_Y>(outLocal);
        AscendC::LocalTensor<DTYPE_Y> outFinal = outQue.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm, outFinal, this->batchPerCore);
        outQue.FreeTensor(outFinal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> xQue;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t hiddenSize;
    uint32_t batchPerCore;
    uint32_t tileLen;
    uint32_t numTiles;
};

extern "C" __global__ __aicore__ void matmul_sigmoid_sum_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSigmoidSum op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.hiddenSize,
            tiling_data.batchPerCore, tiling_data.tileLen, tiling_data.numTiles);
    op.Process();
}
