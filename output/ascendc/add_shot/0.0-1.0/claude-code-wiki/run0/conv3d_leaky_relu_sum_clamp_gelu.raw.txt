kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv3dPostFuse {
public:
    __aicore__ inline KernelConv3dPostFuse() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR sum, GM_ADDR y,
                                 uint32_t numChannels, uint32_t spatialSize,
                                 uint32_t slicesPerCore, uint32_t tileLength,
                                 uint32_t tilesPerSlice, uint32_t tailLength,
                                 float alpha)
    {
        this->numChannels = numChannels;
        this->spatialSize = spatialSize;
        this->slicesPerCore = slicesPerCore;
        this->tileLength = tileLength;
        this->tilesPerSlice = tilesPerSlice;
        this->tailLength = tailLength;
        this->alpha = alpha;

        uint32_t blockOffset = AscendC::GetBlockIdx() * slicesPerCore * spatialSize;
        uint32_t blockLength = slicesPerCore * spatialSize;

        xGm.SetGlobalBuffer((__gm__ float *)x + blockOffset, blockLength);
        yGm.SetGlobalBuffer((__gm__ float *)y + blockOffset, blockLength);
        sumGm.SetGlobalBuffer((__gm__ float *)sum, numChannels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf1, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf2, tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t s = 0; s < this->slicesPerCore; s++) {
            uint32_t globalSliceIdx = AscendC::GetBlockIdx() * this->slicesPerCore + s;
            uint32_t c = globalSliceIdx % this->numChannels;
            float scalar = sumGm.GetValue(c);

            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

            for (uint32_t t = 0; t < this->tilesPerSlice; t++) {
                uint32_t tileOffset = s * this->spatialSize + t * this->tileLength;
                uint32_t curLen = (t == this->tilesPerSlice - 1) ? this->tailLength : this->tileLength;
                CopyIn(tileOffset, curLen);
                Compute(scalar, curLen);
                CopyOut(tileOffset, curLen);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], len);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float scalar, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> t1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> t2 = tmpBuf2.Get<float>();

        // LeakyReLU(x, 0.2) = max(x, 0) + 0.2 * min(x, 0)
        AscendC::Maxs(t1, xLocal, (float)0.0f, len);
        AscendC::Mins(t2, xLocal, (float)0.0f, len);
        AscendC::Muls(t2, t2, this->alpha, len);
        AscendC::Add(yLocal, t1, t2, len);

        // + sum_tensor[c]
        AscendC::Adds(yLocal, yLocal, scalar, len);

        // clamp [-1, 1]
        AscendC::Mins(yLocal, yLocal, (float)1.0f, len);
        AscendC::Maxs(yLocal, yLocal, (float)-1.0f, len);

        // GELU (tanh approx): x / (1 + exp(-1.5957691*x - 0.0713548*x^3))
        AscendC::Mul(t1, yLocal, yLocal, len);                 // x^2
        AscendC::Mul(t2, t1, yLocal, len);                     // x^3
        AscendC::Muls(t2, t2, (float)0.0713548162f, len);      // 0.07135 * x^3
        AscendC::Muls(t1, yLocal, (float)1.5957691216f, len);  // 1.5958 * x
        AscendC::Add(t2, t2, t1, len);                         // u
        AscendC::Muls(t2, t2, (float)-1.0f, len);              // -u
        AscendC::Exp(t2, t2, len);                             // exp(-u)
        AscendC::Adds(t2, t2, (float)1.0f, len);               // 1 + exp(-u)
        AscendC::Div(yLocal, yLocal, t2, len);                 // x / (1 + exp(-u))

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[offset], yLocal, len);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf2;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> sumGm;

    uint32_t numChannels;
    uint32_t spatialSize;
    uint32_t slicesPerCore;
    uint32_t tileLength;
    uint32_t tilesPerSlice;
    uint32_t tailLength;
    float alpha;
};

extern "C" __global__ __aicore__ void conv3d_leaky_relu_sum_clamp_gelu_custom(
    GM_ADDR x, GM_ADDR sum, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dPostFuse op;
    op.Init(x, sum, y,
            tiling_data.numChannels, tiling_data.spatialSize,
            tiling_data.slicesPerCore, tiling_data.tileLength,
            tiling_data.tilesPerSlice, tiling_data.tailLength,
            tiling_data.alpha);
    op.Process();
}

import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, sum_tensor_shape):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.sum_tensor = nn.Parameter(torch.randn(sum_tensor_shape))

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
        sum_flat = self.sum_tensor.contiguous().view(-1)
