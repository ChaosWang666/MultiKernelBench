#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelFusedPool {
public:
    __aicore__ inline KernelFusedPool() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                uint32_t totalLengthPerBatch, uint32_t spatialSize,
                                uint32_t channelCount, uint32_t batchesPerBlock,
                                uint32_t tileNum, uint32_t tileLength, uint32_t lastTileLength)
    {
        this->totalLengthPerBatch = totalLengthPerBatch;
        this->spatialSize = spatialSize;
        this->channelCount = channelCount;
        this->batchesPerBlock = batchesPerBlock;
        this->tileNum = tileNum;
        this->tileLength = tileLength;
        this->lastTileLength = lastTileLength;
        this->invSpatialSize = 1.0f / (float)spatialSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t xOffset = blockIdx * batchesPerBlock * totalLengthPerBatch;
        uint32_t zOffset = blockIdx * batchesPerBlock;

        xGm.SetGlobalBuffer((__gm__ float*)x + xOffset, batchesPerBlock * totalLengthPerBatch);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, channelCount);
        zGm.SetGlobalBuffer((__gm__ float*)z + zOffset, batchesPerBlock);

        uint32_t maxTileLen = tileLength;
        if (lastTileLength > maxTileLen) maxTileLen = lastTileLength;

        pipe.InitBuffer(inQueueX, BUFFER_NUM, maxTileLen * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, batchesPerBlock * sizeof(float));
        pipe.InitBuffer(workBuf, maxTileLen * sizeof(float));
        pipe.InitBuffer(reduceBuf, 8 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        float biasSum = ComputeBiasSum();

        AscendC::LocalTensor<float> outLocal = outQueueZ.AllocTensor<float>();

        for (uint32_t b = 0; b < batchesPerBlock; b++) {
            float totalSum = 0.0f;
            uint32_t batchOffset = b * totalLengthPerBatch;

            for (uint32_t t = 0; t < tileNum; t++) {
                totalSum += LoadAndReduce(batchOffset + t * tileLength, tileLength);
            }
            if (lastTileLength > 0) {
                totalSum += LoadAndReduce(batchOffset + tileNum * tileLength, lastTileLength);
            }

            float result = totalSum * invSpatialSize + biasSum;
            outLocal.SetValue(b, result);
        }

        outQueueZ.EnQue<float>(outLocal);

        outLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm, outLocal, batchesPerBlock);
        outQueueZ.FreeTensor(outLocal);
    }

private:
    __aicore__ inline float ComputeBiasSum()
    {
        AscendC::LocalTensor<float> biasLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(biasLocal, biasGm, channelCount);
        inQueueX.EnQue(biasLocal);
        biasLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::ReduceSum(reduceLocal, biasLocal, workLocal, channelCount);
        AscendC::PipeBarrier<PIPE_V>();
        float biasSum = reduceLocal.GetValue(0);

        inQueueX.FreeTensor(biasLocal);
        return biasSum;
    }

    __aicore__ inline float LoadAndReduce(uint32_t offset, uint32_t length)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], length);
        inQueueX.EnQue(xLocal);
        xLocal = inQueueX.DeQue<float>();

        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::ReduceSum(reduceLocal, xLocal, workLocal, length);
        AscendC::PipeBarrier<PIPE_V>();
        float val = reduceLocal.GetValue(0);

        inQueueX.FreeTensor(xLocal);
        return val;
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf, reduceBuf;
    AscendC::GlobalTensor<float> xGm, biasGm, zGm;

    uint32_t totalLengthPerBatch;
    uint32_t spatialSize;
    uint32_t channelCount;
    uint32_t batchesPerBlock;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t lastTileLength;
    float invSpatialSize;
};

extern "C" __global__ __aicore__ void conv3d_divide_max_global_avg_pool_bias_add_sum_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
