kernel_src="""
#include "kernel_operator.h"

class KernelFusedBiasAddSum {
public:
    __aicore__ inline KernelFusedBiasAddSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR y,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t channelsAligned, uint32_t batchAligned)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->channelsAligned = channelsAligned;
        this->batchAligned = batchAligned;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * channels);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, channelsAligned);
        yGm.SetGlobalBuffer((__gm__ float*)y, batchAligned);

        pipe.InitBuffer(xBuf, channelsAligned * sizeof(float));
        pipe.InitBuffer(biasBuf, channelsAligned * sizeof(float));
        pipe.InitBuffer(tmpBuf, channelsAligned * sizeof(float));
        pipe.InitBuffer(workBuf, channelsAligned * sizeof(float));
        pipe.InitBuffer(reduceBuf, 32);
        pipe.InitBuffer(outBuf, batchAligned * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (AscendC::GetBlockIdx() != 0) {
            return;
        }

        AscendC::LocalTensor<float> biasLocal = biasBuf.Get<float>();
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> outLocal = outBuf.Get<float>();

        AscendC::Duplicate<float>(biasLocal, 0.0f, channelsAligned);
        AscendC::DataCopy(biasLocal, biasGm, channelsAligned);

        for (uint32_t b = 0; b < batchSize; b++) {
            AscendC::Duplicate<float>(xLocal, 0.0f, channelsAligned);
            AscendC::DataCopy(xLocal, xGm[b * channels], channelsAligned);
            AscendC::Add(tmpLocal, xLocal, biasLocal, channels);
            AscendC::ReduceSum<float>(reduceLocal, tmpLocal, workLocal, channels);
            float s = reduceLocal.GetValue(0);
            outLocal.SetValue(b, s);
        }

        AscendC::DataCopy(yGm, outLocal, batchAligned);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf, biasBuf, tmpBuf, workBuf, reduceBuf, outBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t channelsAligned;
    uint32_t batchAligned;
};

extern "C" __global__ __aicore__ void conv3d_divide_max_global_avg_pool_bias_add_sum_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFusedBiasAddSum op;
    op.Init(x, bias, y, tiling_data.batchSize, tiling_data.channels,
            tiling_data.channelsAligned, tiling_data.batchAligned);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, divisor, pool_size, bias_shape, sum_dim):
        super(ModelNew, self).__init__()
        self.conv = nn.Conv3d(in_channels, out_channels, kernel_size)
        self.divisor = divisor
        self.max_pool = nn.MaxPool3d(pool_size)
        self.global_avg_pool = nn.AdaptiveAvgPool3d((1, 1, 1))
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.sum_dim = sum_dim

    def forward(self, x):
        x = self.conv(x)
        x = x / self.divisor
        x = self.max_pool(x)
        x = self.global_avg_pool(x)
        b, c = x.shape[0], x.shape[1]
        x_flat = x.reshape(b, c).contiguous()
        bias_flat = self.bias.reshape(-1).contiguous()
        y = custom_ops_lib.conv3d_divide_max_global_avg_pool_bias_add_sum_custom(x_flat, bias_flat)
        return y.reshape(b, 1, 1, 1)
