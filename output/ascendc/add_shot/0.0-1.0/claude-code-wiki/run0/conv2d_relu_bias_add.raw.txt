kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv2dReluBiasAdd {
public:
    __aicore__ inline KernelConv2dReluBiasAdd() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                 uint32_t outChannels, uint32_t spatialSize,
                                 uint32_t slicesPerBlock, uint32_t alignedSpatialSize,
                                 uint32_t totalSlices)
    {
        this->outChannels = outChannels;
        this->spatialSize = spatialSize;
        this->slicesPerBlock = slicesPerBlock;
        this->alignedSpatialSize = alignedSpatialSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t startSlice = (uint64_t)blockIdx * slicesPerBlock;
        uint64_t blockOffset = startSlice * spatialSize;
        uint64_t blockSize = (uint64_t)slicesPerBlock * spatialSize;
        uint64_t totalSize = (uint64_t)totalSlices * spatialSize;

        xGm.SetGlobalBuffer((__gm__ float *)x + blockOffset, blockSize);
        zGm.SetGlobalBuffer((__gm__ float *)z + blockOffset, blockSize);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, outChannels);

        this->startChannel = (uint32_t)(startSlice % outChannels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, alignedSpatialSize * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, alignedSpatialSize * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < slicesPerBlock; i++) {
            uint32_t channel = (startChannel + i) % outChannels;
            float biasVal = biasGm.GetValue(channel);
            CopyIn(i);
            Compute(biasVal);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t progress)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        uint64_t offset = (uint64_t)progress * spatialSize;
        AscendC::DataCopyExtParams params{1, spatialSize * (uint32_t)sizeof(float), 0, 0, 0};
        AscendC::DataCopyPadExtParams<float> padParams{false, 0, 0, 0.0f};
        AscendC::DataCopyPad(xLocal, xGm[offset], params, padParams);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float biasVal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        AscendC::Maxs(zLocal, xLocal, 0.0f, alignedSpatialSize);
        AscendC::PipeBarrier<PIPE_V>();
        AscendC::Adds(zLocal, zLocal, biasVal, alignedSpatialSize);
        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t progress)
    {
        AscendC::LocalTensor<float> zLocal = outQueueZ.DeQue<float>();
        uint64_t offset = (uint64_t)progress * spatialSize;
        AscendC::DataCopyExtParams params{1, spatialSize * (uint32_t)sizeof(float), 0, 0, 0};
        AscendC::DataCopyPad(zGm[offset], zLocal, params);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t outChannels;
    uint32_t spatialSize;
    uint32_t slicesPerBlock;
    uint32_t alignedSpatialSize;
    uint32_t startChannel;
};

extern "C" __global__ __aicore__ void conv2d_relu_bias_add_custom(GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv2dReluBiasAdd op;
    op.Init(x, bias, z,
            tiling_data.outChannels, tiling_data.spatialSize,
            tiling_data.slicesPerBlock, tiling_data.alignedSpatialSize,
            tiling_data.totalSlices);
    op.Process();
}
