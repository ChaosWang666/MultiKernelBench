kernel_src = """
#include "kernel_operator.h"

constexpr float EPS = 1e-5f;

class KernelMSSGN {
public:
    __aicore__ inline KernelMSSGN() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR gn_w, GM_ADDR gn_b, GM_ADDR y,
                                 uint32_t batchPerCore, uint32_t channels,
                                 uint32_t groups, uint32_t channelsPerGroup)
    {
        this->batchPerCore = batchPerCore;
        this->channels = channels;
        this->groups = groups;
        this->cpg = channelsPerGroup;

        uint32_t blockOffset = batchPerCore * channels * AscendC::GetBlockIdx();
        xGm.SetGlobalBuffer((__gm__ float*)x + blockOffset, batchPerCore * channels);
        yGm.SetGlobalBuffer((__gm__ float*)y + blockOffset, batchPerCore * channels);
        biasGm.SetGlobalBuffer((__gm__ float*)bias, channels);
        gnWGm.SetGlobalBuffer((__gm__ float*)gn_w, channels);
        gnBGm.SetGlobalBuffer((__gm__ float*)gn_b, channels);

        pipe.InitBuffer(inQueueX, 2, channels * sizeof(float));
        pipe.InitBuffer(outQueueY, 2, channels * sizeof(float));
        pipe.InitBuffer(biasQueue, 1, channels * sizeof(float));
        pipe.InitBuffer(gnWQueue, 1, channels * sizeof(float));
        pipe.InitBuffer(gnBQueue, 1, channels * sizeof(float));
        pipe.InitBuffer(tmpBuf, channels * sizeof(float));
        pipe.InitBuffer(reduceResBuf, 32);
        pipe.InitBuffer(workBuf, 256);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasLocal = biasQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> gnWLocal = gnWQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> gnBLocal = gnBQueue.AllocTensor<float>();
        AscendC::DataCopy(biasLocal, biasGm, channels);
        AscendC::DataCopy(gnWLocal, gnWGm, channels);
        AscendC::DataCopy(gnBLocal, gnBGm, channels);
        biasQueue.EnQue(biasLocal);
        gnWQueue.EnQue(gnWLocal);
        gnBQueue.EnQue(gnBLocal);
        biasLocal = biasQueue.DeQue<float>();
        gnWLocal = gnWQueue.DeQue<float>();
        gnBLocal = gnBQueue.DeQue<float>();

        for (uint32_t i = 0; i < batchPerCore; i++) {
            CopyIn(i);
            Compute(biasLocal, gnWLocal, gnBLocal);
            CopyOut(i);
        }

        biasQueue.FreeTensor(biasLocal);
        gnWQueue.FreeTensor(gnWLocal);
        gnBQueue.FreeTensor(gnBLocal);
    }

private:
    __aicore__ inline void CopyIn(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[rowIdx * channels], channels);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(AscendC::LocalTensor<float>& biasLocal,
                                     AscendC::LocalTensor<float>& gnWLocal,
                                     AscendC::LocalTensor<float>& gnBLocal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceRes = reduceResBuf.Get<float>();
        AscendC::LocalTensor<float> work = workBuf.Get<float>();

        AscendC::Sigmoid(tmp, xLocal, channels);
        AscendC::Mul(xLocal, xLocal, tmp, channels);

        AscendC::Add(xLocal, xLocal, biasLocal, channels);

        for (uint32_t g = 0; g < groups; g++) {
            uint32_t offset = g * cpg;

            AscendC::ReduceSum(reduceRes, xLocal[offset], work, cpg);
            AscendC::event_t eventVS1 = static_cast<AscendC::event_t>(
                AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(eventVS1);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(eventVS1);
            float mean = reduceRes.GetValue(0) / (float)cpg;

            AscendC::event_t eventSV1 = static_cast<AscendC::event_t>(
                AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_V));
            AscendC::SetFlag<AscendC::HardEvent::S_V>(eventSV1);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(eventSV1);

            AscendC::Adds(xLocal[offset], xLocal[offset], -mean, cpg);
            AscendC::Mul(tmp[offset], xLocal[offset], xLocal[offset], cpg);

            AscendC::ReduceSum(reduceRes, tmp[offset], work, cpg);
            AscendC::event_t eventVS2 = static_cast<AscendC::event_t>(
                AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
            AscendC::SetFlag<AscendC::HardEvent::V_S>(eventVS2);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(eventVS2);
            float var = reduceRes.GetValue(0) / (float)cpg;
            float invStd = 1.0f / sqrt(var + EPS);

            AscendC::event_t eventSV2 = static_cast<AscendC::event_t>(
                AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::S_V));
            AscendC::SetFlag<AscendC::HardEvent::S_V>(eventSV2);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(eventSV2);

            AscendC::Muls(xLocal[offset], xLocal[offset], invStd, cpg);
        }

        AscendC::Mul(yLocal, xLocal, gnWLocal, channels);
        AscendC::Add(yLocal, yLocal, gnBLocal, channels);

        inQueueX.FreeTensor(xLocal);
        outQueueY.EnQue<float>(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t rowIdx)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[rowIdx * channels], yLocal, channels);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, 2> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 2> outQueueY;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> biasQueue, gnWQueue, gnBQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf, reduceResBuf, workBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> gnWGm;
    AscendC::GlobalTensor<float> gnBGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchPerCore;
    uint32_t channels;
    uint32_t groups;
    uint32_t cpg;
};

extern "C" __global__ __aicore__ void matmul_swish_sum_group_norm_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR gn_weight, GM_ADDR gn_bias, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelMSSGN op;
    op.Init(x, bias, gn_weight, gn_bias, y,
            tiling_data.batchPerCore, tiling_data.channels,
            tiling_data.groups, tiling_data.channelsPerGroup);
    op.Process();
}

model_src = '''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, bias_shape):
        super(ModelNew, self).__init__()
        self.matmul = nn.Linear(in_features, out_features)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.group_norm = nn.GroupNorm(num_groups, out_features)

    def forward(self, x):
        x = self.matmul(x)
        x = x.contiguous()
        x = custom_ops_lib.matmul_swish_sum_group_norm_custom(
            x, self.bias.contiguous(),
            self.group_norm.weight.contiguous(),
            self.group_norm.bias.contiguous())
        return x
