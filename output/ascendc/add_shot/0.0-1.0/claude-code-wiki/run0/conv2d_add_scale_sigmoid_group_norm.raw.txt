            blockDim--;
        }
    }
    if (blockDim == 0) {
        blockDim = 1;

        tileLength = ((spatial + 7) / 8) * 8;
        if (tileLength == 0) {
            tileLength = 8;

    if (tileNum == 0) {

    tiling.set_totalChunks(totalChunks);

kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelFused {
public:
    __aicore__ inline KernelFused() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR scale, GM_ADDR y,
                                uint32_t totalChunks, uint32_t channels, uint32_t spatial,
                                uint32_t tileLength, uint32_t tileNum, uint32_t tailLength)
    {
        this->channels = channels;
        this->spatial = spatial;
        this->tileLength = tileLength;
        this->tileNum = tileNum;
        this->tailLength = tailLength;

        uint32_t blockDim = AscendC::GetBlockNum();
        this->chunksPerCore = totalChunks / blockDim;
        this->startChunk = this->chunksPerCore * static_cast<uint32_t>(AscendC::GetBlockIdx());

        xGm.SetGlobalBuffer((__gm__ float *)x + this->startChunk * this->spatial,
                            this->chunksPerCore * this->spatial);
        yGm.SetGlobalBuffer((__gm__ float *)y + this->startChunk * this->spatial,
                            this->chunksPerCore * this->spatial);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, this->channels);
        scaleGm.SetGlobalBuffer((__gm__ float *)scale, this->channels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->chunksPerCore; ++i) {
            uint32_t globalChunk = this->startChunk + i;
            uint32_t ch = globalChunk % this->channels;
            float b = biasGm.GetValue(ch);
            float s = scaleGm.GetValue(ch);

            for (uint32_t t = 0; t < this->tileNum; ++t) {
                uint32_t curLen = (t == this->tileNum - 1) ? this->tailLength : this->tileLength;
                uint32_t offset = i * this->spatial + t * this->tileLength;
                CopyIn(offset, curLen);
                Compute(b, s, curLen);
                CopyOut(offset, curLen);
            }
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();

        AscendC::DataCopyExtParams params;
        params.blockCount = 1;
        params.blockLen = static_cast<uint32_t>(len * sizeof(float));
        params.srcStride = 0;
        params.dstStride = 0;

        AscendC::DataCopyPadExtParams<float> pad;
        pad.isPad = false;
        pad.leftPadding = 0;
        pad.rightPadding = 0;
        pad.paddingValue = 0.0f;

        AscendC::DataCopyPad(xLocal, xGm[offset], params, pad);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline void Compute(float b, float s, uint32_t len)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();

        AscendC::Adds(xLocal, xLocal, b, len);
        AscendC::Muls(xLocal, xLocal, s, len);
        AscendC::Muls(yLocal, xLocal, -1.0f, len);
        AscendC::Exp(yLocal, yLocal, len);
        AscendC::Adds(yLocal, yLocal, 1.0f, len);
        AscendC::Reciprocal(yLocal, yLocal, len);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }

    __aicore__ inline void CopyOut(uint32_t offset, uint32_t len)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();

        AscendC::DataCopyExtParams params;
        params.blockCount = 1;
        params.blockLen = static_cast<uint32_t>(len * sizeof(float));
        params.srcStride = 0;
        params.dstStride = 0;

        AscendC::DataCopyPad(yGm[offset], yLocal, params);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> scaleGm;
    uint32_t channels;
    uint32_t spatial;
    uint32_t tileLength;
    uint32_t tileNum;
    uint32_t tailLength;
    uint32_t chunksPerCore;
    uint32_t startChunk;
};

extern "C" __global__ __aicore__ void conv2d_add_scale_sigmoid_group_norm_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR scale, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelFused op;
    op.Init(x, bias, scale, y,
            tiling_data.totalChunks, tiling_data.channels, tiling_data.spatial,
            tiling_data.tileLength, tiling_data.tileNum, tiling_data.tailLength);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(torch.nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, num_groups, bias_shape, scale_shape):
        super().__init__()
        self.conv = nn.Conv2d(in_channels, out_channels, kernel_size)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.scale = nn.Parameter(torch.randn(scale_shape))
        self.group_norm = nn.GroupNorm(num_groups, out_channels)

    def forward(self, x):
        x = self.conv(x)
        x = x.contiguous()
        b = self.bias.view(-1).contiguous()
        s = self.scale.view(-1).contiguous()
        x = custom_ops_lib.conv2d_add_scale_sigmoid_group_norm_custom(x, b, s)
        x = self.group_norm(x)
        return x
