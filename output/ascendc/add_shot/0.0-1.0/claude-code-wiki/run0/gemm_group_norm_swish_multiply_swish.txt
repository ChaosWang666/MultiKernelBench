kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelSwishMulSwish {
public:
    __aicore__ inline KernelSwishMulSwish() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR y,
                                uint32_t batchSize, uint32_t outFeatures,
                                uint32_t rowsPerCore, uint32_t tileLength,
                                uint32_t tilesPerRow)
    {
        this->outFeatures = outFeatures;
        this->tileLength = tileLength;
        this->tilesPerRow = tilesPerRow;
        this->totalTiles = rowsPerCore * tilesPerRow;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t coreOffset = blockIdx * rowsPerCore * outFeatures;
        uint32_t coreElems = rowsPerCore * outFeatures;

        xGm.SetGlobalBuffer((__gm__ float *)x + coreOffset, coreElems);
        yGm.SetGlobalBuffer((__gm__ float *)y + coreOffset, coreElems);
        wGm.SetGlobalBuffer((__gm__ float *)w, outFeatures);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t t = 0; t < this->totalTiles; t++) {
            uint32_t xOff = t * this->tileLength;
            uint32_t wOff = (t % this->tilesPerRow) * this->tileLength;
            CopyIn(xOff, wOff);
            Compute();
            CopyOut(xOff);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t xOff, uint32_t wOff)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::LocalTensor<float> wLocal = inQueueW.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[xOff], this->tileLength);
        AscendC::DataCopy(wLocal, wGm[wOff], this->tileLength);
        inQueueX.EnQue(xLocal);
        inQueueW.EnQue(wLocal);
    }

    __aicore__ inline void Compute()
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> wLocal = inQueueW.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        uint32_t len = this->tileLength;

        AscendC::Muls(tmp, xLocal, -1.0f, len);
        AscendC::Exp(tmp, tmp, len);
        AscendC::Adds(tmp, tmp, 1.0f, len);
        AscendC::Reciprocal(tmp, tmp, len);
        AscendC::Mul(xLocal, xLocal, tmp, len);

        AscendC::Mul(xLocal, xLocal, wLocal, len);

        AscendC::Muls(tmp, xLocal, -1.0f, len);
        AscendC::Exp(tmp, tmp, len);
        AscendC::Adds(tmp, tmp, 1.0f, len);
        AscendC::Reciprocal(tmp, tmp, len);
        AscendC::Mul(yLocal, xLocal, tmp, len);

        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueW.FreeTensor(wLocal);
    }

    __aicore__ inline void CopyOut(uint32_t xOff)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[xOff], yLocal, this->tileLength);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> wGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t outFeatures;
    uint32_t tileLength;
    uint32_t tilesPerRow;
    uint32_t totalTiles;
};

extern "C" __global__ __aicore__ void gemm_group_norm_swish_multiply_swish_custom(
    GM_ADDR x, GM_ADDR w, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelSwishMulSwish op;
    op.Init(x, w, y, tiling_data.batchSize, tiling_data.outFeatures,
            tiling_data.rowsPerCore, tiling_data.tileLength,
            tiling_data.tilesPerRow);
    op.Process();
}

model_src='''
import torch
import torch_npu
import torch.nn as nn
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_features, out_features, num_groups, multiply_weight_shape):
        super(ModelNew, self).__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.group_norm = nn.GroupNorm(num_groups, out_features)
        self.multiply_weight = nn.Parameter(torch.randn(multiply_weight_shape))

    def forward(self, x):
        x = self.gemm(x)
        x = self.group_norm(x)
        x = custom_ops_lib.gemm_group_norm_swish_multiply_swish_custom(
            x.contiguous(), self.multiply_weight.contiguous()
        )
        return x
