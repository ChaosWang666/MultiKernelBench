kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelGemmGroupNormSwishMultiplySwish {
public:
    __aicore__ inline KernelGemmGroupNormSwishMultiplySwish() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR mw, GM_ADDR y,
                                uint32_t batchSize, uint32_t outFeatures, uint32_t numGroups,
                                uint32_t groupSize, uint32_t batchPerCore,
                                float invGroupSize, float eps)
    {
        this->outFeatures = outFeatures;
        this->numGroups = numGroups;
        this->groupSize = groupSize;
        this->batchPerCore = batchPerCore;
        this->invGroupSize = invGroupSize;
        this->eps = eps;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint64_t rowOffset = (uint64_t)blockIdx * (uint64_t)batchPerCore * (uint64_t)outFeatures;

        xGm.SetGlobalBuffer((__gm__ float *)x + rowOffset, (uint64_t)batchPerCore * outFeatures);
        yGm.SetGlobalBuffer((__gm__ float *)y + rowOffset, (uint64_t)batchPerCore * outFeatures);
        gammaGm.SetGlobalBuffer((__gm__ float *)gamma, outFeatures);
        betaGm.SetGlobalBuffer((__gm__ float *)beta, outFeatures);
        mwGm.SetGlobalBuffer((__gm__ float *)mw, outFeatures);

        pipe.InitBuffer(xInQueue, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(yOutQueue, BUFFER_NUM, outFeatures * sizeof(float));
        pipe.InitBuffer(gammaBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(betaBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(mwBuf, outFeatures * sizeof(float));
        pipe.InitBuffer(scalarBuf, 32);
        pipe.InitBuffer(reduceBuf, 1024);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();
        AscendC::LocalTensor<float> mwLocal = mwBuf.Get<float>();
        AscendC::DataCopy(gammaLocal, gammaGm, outFeatures);
        AscendC::DataCopy(betaLocal, betaGm, outFeatures);
        AscendC::DataCopy(mwLocal, mwGm, outFeatures);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t i = 0; i < batchPerCore; i++) {
            CopyIn(i);
            Compute(i);
            CopyOut(i);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t i)
    {
        AscendC::LocalTensor<float> xLocal = xInQueue.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[(uint64_t)i * outFeatures], outFeatures);
        xInQueue.EnQue(xLocal);
    }

    __aicore__ inline void Compute(uint32_t i)
    {
        AscendC::LocalTensor<float> xLocal = xInQueue.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = yOutQueue.AllocTensor<float>();
        AscendC::LocalTensor<float> gammaLocal = gammaBuf.Get<float>();
        AscendC::LocalTensor<float> betaLocal = betaBuf.Get<float>();
        AscendC::LocalTensor<float> mwLocal = mwBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();
        AscendC::LocalTensor<float> reduceTmp = reduceBuf.Get<float>();

        // Phase 1: center x per group (x := x - mean_g)
        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t off = g * groupSize;
            AscendC::ReduceSum<float>(scalarLocal, xLocal[off], reduceTmp, groupSize);
            AscendC::PipeBarrier<PIPE_ALL>();
            float sum = scalarLocal.GetValue(0);
            float mean = sum * invGroupSize;
            AscendC::Adds<float>(xLocal[off], xLocal[off], -mean, groupSize);
        }
        AscendC::PipeBarrier<PIPE_ALL>();

        // Phase 2: variance per group and normalize (x := x / sqrt(var+eps))
        AscendC::Mul<float>(yLocal, xLocal, xLocal, outFeatures);
        AscendC::PipeBarrier<PIPE_ALL>();

        for (uint32_t g = 0; g < numGroups; g++) {
            uint32_t off = g * groupSize;
            AscendC::ReduceSum<float>(scalarLocal, yLocal[off], reduceTmp, groupSize);
            AscendC::PipeBarrier<PIPE_ALL>();
            float sumsq = scalarLocal.GetValue(0);
            float var = sumsq * invGroupSize;
            float varPlusEps = var + eps;
            if (varPlusEps < eps) {
                varPlusEps = eps;
            }
            AscendC::Duplicate<float>(scalarLocal, varPlusEps, 8);
            AscendC::PipeBarrier<PIPE_ALL>();
            AscendC::Sqrt<float>(scalarLocal, scalarLocal, 8);
            AscendC::PipeBarrier<PIPE_ALL>();
            float stdVal = scalarLocal.GetValue(0);
            float invStd = 1.0f / stdVal;
            AscendC::Muls<float>(xLocal[off], xLocal[off], invStd, groupSize);
        }
        AscendC::PipeBarrier<PIPE_ALL>();

        // Apply gamma and beta: y = x * gamma + beta
        AscendC::Mul<float>(yLocal, xLocal, gammaLocal, outFeatures);
        AscendC::Add<float>(yLocal, yLocal, betaLocal, outFeatures);

        // Swish 1: y = y * sigmoid(y)
        AscendC::Muls<float>(xLocal, yLocal, -1.0f, outFeatures);
        AscendC::Exp<float>(xLocal, xLocal, outFeatures);
        AscendC::Adds<float>(xLocal, xLocal, 1.0f, outFeatures);
        AscendC::Reciprocal<float>(xLocal, xLocal, outFeatures);
        AscendC::Mul<float>(yLocal, yLocal, xLocal, outFeatures);

        // Multiply by weight
        AscendC::Mul<float>(yLocal, yLocal, mwLocal, outFeatures);

        // Swish 2: y = y * sigmoid(y)
        AscendC::Muls<float>(xLocal, yLocal, -1.0f, outFeatures);
        AscendC::Exp<float>(xLocal, xLocal, outFeatures);
        AscendC::Adds<float>(xLocal, xLocal, 1.0f, outFeatures);
        AscendC::Reciprocal<float>(xLocal, xLocal, outFeatures);
        AscendC::Mul<float>(yLocal, yLocal, xLocal, outFeatures);

        xInQueue.FreeTensor(xLocal);
        yOutQueue.EnQue(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t i)
    {
        AscendC::LocalTensor<float> yLocal = yOutQueue.DeQue<float>();
        AscendC::DataCopy(yGm[(uint64_t)i * outFeatures], yLocal, outFeatures);
        yOutQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> xInQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> yOutQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> gammaBuf, betaBuf, mwBuf, scalarBuf, reduceBuf;
    AscendC::GlobalTensor<float> xGm, yGm, gammaGm, betaGm, mwGm;
    uint32_t outFeatures;
    uint32_t numGroups;
    uint32_t groupSize;
    uint32_t batchPerCore;
    float invGroupSize;
    float eps;
};

extern "C" __global__ __aicore__ void gemm_group_norm_swish_multiply_swish_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR mw, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmGroupNormSwishMultiplySwish op;
    op.Init(x, gamma, beta, mw, y,
            tiling_data.batchSize, tiling_data.outFeatures, tiling_data.numGroups,
            tiling_data.groupSize, tiling_data.batchPerCore,
            tiling_data.invGroupSize, tiling_data.eps);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, num_groups, multiply_weight_shape):
        super().__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.group_norm = nn.GroupNorm(num_groups, out_features)
        self.multiply_weight = nn.Parameter(torch.randn(multiply_weight_shape))

    def forward(self, x):
        x = self.gemm(x)
        x = x.contiguous()
        return custom_ops_lib.gemm_group_norm_swish_multiply_swish_custom(
            x,
            self.group_norm.weight.contiguous(),
            self.group_norm.bias.contiguous(),
            self.multiply_weight.contiguous(),
