kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose3dMaxMaxSum {
public:
    __aicore__ inline KernelConvTranspose3dMaxMaxSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t channels, uint32_t innerSize, uint32_t blockDim)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->innerSize = innerSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t batchesPerCore = (batchSize + blockDim - 1) / blockDim;
        this->batchStart = blockIdx * batchesPerCore;
        uint32_t end = this->batchStart + batchesPerCore;
        if (end > batchSize) end = batchSize;
        this->batchCount = (this->batchStart < batchSize) ? (end - this->batchStart) : 0;

        xGm.SetGlobalBuffer((__gm__ float*)x, batchSize * channels * innerSize);
        yGm.SetGlobalBuffer((__gm__ float*)y, batchSize * innerSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->innerSize * sizeof(float));
        pipe.InitBuffer(outQueueY, 1, this->innerSize * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->batchCount; i++) {
            uint32_t b = this->batchStart + i;
            ProcessBatch(b);
        }
    }

private:
    __aicore__ inline void ProcessBatch(uint32_t b)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        AscendC::Duplicate(yLocal, (float)0, this->innerSize);

        for (uint32_t c = 0; c < this->channels; c++) {
            AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
            AscendC::DataCopy(xLocal, xGm[(b * this->channels + c) * this->innerSize], this->innerSize);
            inQueueX.EnQue(xLocal);

            xLocal = inQueueX.DeQue<float>();
            AscendC::Add(yLocal, yLocal, xLocal, this->innerSize);
            inQueueX.FreeTensor(xLocal);
        }

        outQueueY.EnQue<float>(yLocal);
        AscendC::LocalTensor<float> yOut = outQueueY.DeQue<float>();
        AscendC::DataCopy(yGm[b * this->innerSize], yOut, this->innerSize);
        outQueueY.FreeTensor(yOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t innerSize;
    uint32_t batchStart;
    uint32_t batchCount;
};

extern "C" __global__ __aicore__ void conv_transpose3d_max_max_sum_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dMaxMaxSum op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.channels, tiling_data.innerSize, tiling_data.blockDim);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, stride=stride, padding=padding)
        self.max_pool1 = nn.MaxPool3d(kernel_size=2)
        self.max_pool2 = nn.MaxPool3d(kernel_size=3)

    def forward(self, x):
        x = self.conv_transpose(x)
        x = self.max_pool1(x)
        x = self.max_pool2(x)
        x = x.contiguous()
        x = custom_ops_lib.conv_transpose3d_max_max_sum_custom(x)
        return x
