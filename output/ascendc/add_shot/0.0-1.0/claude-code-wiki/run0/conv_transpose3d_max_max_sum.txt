kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTranspose3dMaxMaxSum {
public:
    __aicore__ inline KernelConvTranspose3dMaxMaxSum() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize, uint32_t channels, uint32_t spatialSize)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->spatialSize = spatialSize;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        if (blockIdx >= this->batchSize) {
            return;
        }

        uint32_t batchStride = channels * spatialSize;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockIdx * batchStride, batchStride);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + blockIdx * spatialSize, spatialSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->spatialSize * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, 1, this->spatialSize * sizeof(DTYPE_Y));
    }

    __aicore__ inline void Process()
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        if (blockIdx >= this->batchSize) {
            return;
        }

        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueueY.AllocTensor<DTYPE_Y>();
        AscendC::Duplicate(yLocal, 0.0f, this->spatialSize);

        for (uint32_t c = 0; c < this->channels; c++) {
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
            AscendC::DataCopy(xLocal, xGm[c * this->spatialSize], this->spatialSize);
            inQueueX.EnQue(xLocal);

            xLocal = inQueueX.DeQue<DTYPE_X>();
            AscendC::Add(yLocal, yLocal, xLocal, this->spatialSize);
            inQueueX.FreeTensor(xLocal);
        }

        outQueueY.EnQue<DTYPE_Y>(yLocal);
        yLocal = outQueueY.DeQue<DTYPE_Y>();
        AscendC::DataCopy(yGm, yLocal, this->spatialSize);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t spatialSize;
};

extern "C" __global__ __aicore__ void conv_transpose3d_max_max_sum_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTranspose3dMaxMaxSum op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.channels, tiling_data.spatialSize);
    op.Process();
}
