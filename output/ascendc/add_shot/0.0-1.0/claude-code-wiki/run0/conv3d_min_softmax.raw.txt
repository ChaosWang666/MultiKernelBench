kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelConv3dMinSoftmax {
public:
    __aicore__ inline KernelConv3dMinSoftmax() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
        uint32_t N, uint32_t C, uint32_t D, uint32_t H, uint32_t W)
    {
        this->N = N;
        this->C = C;
        this->D = D;
        this->H = H;
        this->W = W;
        this->HW = H * W;
        this->CHW = C * this->HW;
        this->DHW = D * this->HW;
        this->CDHW = C * this->DHW;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t nPerBlock = (N + blockNum - 1) / blockNum;
        uint32_t sN = blockIdx * nPerBlock;
        uint32_t eN = sN + nPerBlock;
        if (eN > N) eN = N;
        if (sN > N) sN = N;
        this->startN = sN;
        this->endN = eN;
        uint32_t localN = eN - sN;

        xGm.SetGlobalBuffer((__gm__ float*)x + sN * this->CDHW, localN * this->CDHW);
        yGm.SetGlobalBuffer((__gm__ float*)y + sN * this->CHW, localN * this->CHW);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->DHW * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->CHW * sizeof(float));
        pipe.InitBuffer(maxBuf, 1024 * sizeof(float));
        pipe.InitBuffer(sumBuf, 1024 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();

        uint32_t nRange = endN - startN;
        for (uint32_t ni = 0; ni < nRange; ni++) {
            AscendC::LocalTensor<float> outL = outQueueZ.AllocTensor<float>();

            for (uint32_t c = 0; c < C; c++) {
                AscendC::LocalTensor<float> xAlloc = inQueueX.AllocTensor<float>();
                AscendC::DataCopy(xAlloc, xGm[ni * CDHW + c * DHW], DHW);
                inQueueX.EnQue(xAlloc);

                AscendC::LocalTensor<float> xL = inQueueX.DeQue<float>();
                AscendC::Adds(outL[c * HW], xL, (float)0.0f, HW);
                for (uint32_t d = 1; d < D; d++) {
                    AscendC::Min(outL[c * HW], outL[c * HW], xL[d * HW], HW);
                }
                inQueueX.FreeTensor(xL);
            }

            AscendC::Adds(maxLocal, outL, (float)0.0f, HW);
            for (uint32_t c = 1; c < C; c++) {
                AscendC::Max(maxLocal, maxLocal, outL[c * HW], HW);
            }

            AscendC::Duplicate<float>(sumLocal, (float)0.0f, HW);
            for (uint32_t c = 0; c < C; c++) {
                AscendC::Sub(outL[c * HW], outL[c * HW], maxLocal, HW);
                AscendC::Exp(outL[c * HW], outL[c * HW], HW);
                AscendC::Add(sumLocal, sumLocal, outL[c * HW], HW);
            }

            for (uint32_t c = 0; c < C; c++) {
                AscendC::Div(outL[c * HW], outL[c * HW], sumLocal, HW);
            }

            outQueueZ.EnQue<float>(outL);

            AscendC::LocalTensor<float> outD = outQueueZ.DeQue<float>();
            AscendC::DataCopy(yGm[ni * CHW], outD, CHW);
            outQueueZ.FreeTensor(outD);
        }
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t N;
    uint32_t C;
    uint32_t D;
    uint32_t H;
    uint32_t W;
    uint32_t HW;
    uint32_t CHW;
    uint32_t DHW;
    uint32_t CDHW;
    uint32_t startN;
    uint32_t endN;
};

extern "C" __global__ __aicore__ void conv3d_min_softmax_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConv3dMinSoftmax op;
    op.Init(x, y, tiling_data.N, tiling_data.C, tiling_data.D, tiling_data.H, tiling_data.W);
    op.Process();
}
