kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t RES_LEN = 64;

class KernelLogSumExp {
public:
    __aicore__ inline KernelLogSumExp() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR z, uint32_t totalRows, uint32_t rowLen)
    {
        this->rowLen = rowLen;
        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        this->rowsPerBlock = totalRows / blockNum;

        uint32_t rowOffset = blockIdx * this->rowsPerBlock;
        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + rowOffset * rowLen, this->rowsPerBlock * rowLen);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + rowOffset, this->rowsPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, rowLen * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueZ, 1, this->rowsPerBlock * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpBuf1, rowLen * sizeof(float));
        pipe.InitBuffer(tmpBuf2, rowLen * sizeof(float));
        pipe.InitBuffer(resBuf, RES_LEN * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> tmp1 = tmpBuf1.Get<float>();
        AscendC::LocalTensor<float> tmp2 = tmpBuf2.Get<float>();
        AscendC::LocalTensor<float> res = resBuf.Get<float>();

        AscendC::Duplicate<float>(res, (float)1.0, RES_LEN);

        for (uint32_t r = 0; r < rowsPerBlock; r++) {
            CopyIn(r);
            AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();

            AscendC::ReduceMax<float>(res, xLocal, tmp2, rowLen, false);
            float maxVal = res.GetValue(0);

            AscendC::Adds<float>(tmp1, xLocal, -maxVal, rowLen);
            inQueueX.FreeTensor(xLocal);

            AscendC::Exp<float>(tmp1, tmp1, rowLen);

            AscendC::Duplicate<float>(res, (float)1.0, RES_LEN);
            AscendC::ReduceSum<float>(res, tmp1, tmp2, rowLen);

            AscendC::Ln<float>(res, res, RES_LEN);
            float logSum = res.GetValue(0);

            zLocal.SetValue(r, logSum + maxVal);
        }

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        AscendC::LocalTensor<DTYPE_Z> zOut = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm, zOut, rowsPerBlock);
        outQueueZ.FreeTensor(zOut);
    }

private:
    __aicore__ inline void CopyIn(uint32_t r)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::DataCopy(xLocal, xGm[r * rowLen], rowLen);
        inQueueX.EnQue(xLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf1;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf2;
    AscendC::TBuf<AscendC::TPosition::VECCALC> resBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t rowLen;
    uint32_t rowsPerBlock;
};

extern "C" __global__ __aicore__ void gemm_sigmoid_sum_log_sum_exp_custom(
    GM_ADDR x, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelLogSumExp op;
    op.Init(x, z, tiling_data.totalRows, tiling_data.rowLen);
    op.Process();
}
"""
