kernel_src="""
#include "kernel_operator.h"
#include "lib/matmul_intf.h"

using namespace AscendC;

class GemmFusedAIV {
public:
    __aicore__ inline GemmFusedAIV() {}

    __aicore__ inline void Init(GM_ADDR y, GM_ADDR gnGamma, GM_ADDR gnBeta,
                                uint32_t M, uint32_t N, uint32_t numGroups,
                                uint32_t channelsPerGroup, float eps,
                                float invChannelsPerGroup, TPipe* pipe)
    {
        this->pipePtr = pipe;
        this->M = M;
        this->N = N;
        this->numGroups = numGroups;
        this->channelsPerGroup = channelsPerGroup;
        this->eps = eps;
        this->invChannelsPerGroup = invChannelsPerGroup;

        uint32_t blockIdx = GetBlockIdx();
        uint32_t blockNum = GetBlockNum();
        if (blockNum == 0) {
            blockNum = 1;
        }
        this->rowsPerBlock = M / blockNum;
        uint32_t startRow = blockIdx * this->rowsPerBlock;

        yGm.SetGlobalBuffer((__gm__ float*)y + startRow * N, this->rowsPerBlock * N);
        gnGammaGm.SetGlobalBuffer((__gm__ float*)gnGamma, N);
        gnBetaGm.SetGlobalBuffer((__gm__ float*)gnBeta, N);

        pipePtr->InitBuffer(rowBuf, N * sizeof(float));
        pipePtr->InitBuffer(gammaBuf, N * sizeof(float));
        pipePtr->InitBuffer(betaBuf, N * sizeof(float));
        pipePtr->InitBuffer(tmp1Buf, N * sizeof(float));
        pipePtr->InitBuffer(tmp2Buf, N * sizeof(float));
        pipePtr->InitBuffer(diffBuf, channelsPerGroup * sizeof(float));
        pipePtr->InitBuffer(sqBuf, channelsPerGroup * sizeof(float));
        pipePtr->InitBuffer(reduceBuf, 32);
        pipePtr->InitBuffer(workBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        LocalTensor<float> gamma = gammaBuf.Get<float>();
        LocalTensor<float> beta = betaBuf.Get<float>();
        LocalTensor<float> row = rowBuf.Get<float>();
        LocalTensor<float> tmp1 = tmp1Buf.Get<float>();
        LocalTensor<float> tmp2 = tmp2Buf.Get<float>();
        LocalTensor<float> diff = diffBuf.Get<float>();
        LocalTensor<float> sq = sqBuf.Get<float>();
        LocalTensor<float> reduceTmp = reduceBuf.Get<float>();
        LocalTensor<float> workTmp = workBuf.Get<float>();

        DataCopy(gamma, gnGammaGm, this->N);
        DataCopy(beta, gnBetaGm, this->N);
        event_t evt0 = pipePtr->FetchEventID(HardEvent::MTE2_V);
        SetFlag<HardEvent::MTE2_V>(evt0);
        WaitFlag<HardEvent::MTE2_V>(evt0);

        for (uint32_t r = 0; r < this->rowsPerBlock; r++) {
            ProcessRow(r, row, gamma, beta, tmp1, tmp2, diff, sq, reduceTmp, workTmp);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t r,
                                       LocalTensor<float>& row,
                                       LocalTensor<float>& gamma,
                                       LocalTensor<float>& beta,
                                       LocalTensor<float>& tmp1,
                                       LocalTensor<float>& tmp2,
                                       LocalTensor<float>& diff,
                                       LocalTensor<float>& sq,
                                       LocalTensor<float>& reduceTmp,
                                       LocalTensor<float>& workTmp)
    {
        DataCopy(row, yGm[r * this->N], this->N);
        event_t evtIn = pipePtr->FetchEventID(HardEvent::MTE2_V);
        SetFlag<HardEvent::MTE2_V>(evtIn);
        WaitFlag<HardEvent::MTE2_V>(evtIn);

        Maxs(row, row, -1.0f, this->N);
        PipeBarrier<PIPE_V>();
        Mins(row, row, 1.0f, this->N);
        PipeBarrier<PIPE_V>();

        Exp(tmp1, row, this->N);
        PipeBarrier<PIPE_V>();
        Adds(tmp1, tmp1, 1.0f, this->N);
        PipeBarrier<PIPE_V>();
        Ln(tmp1, tmp1, this->N);
        PipeBarrier<PIPE_V>();
        Muls(tmp1, tmp1, 2.0f, this->N);
        PipeBarrier<PIPE_V>();
        Exp(tmp1, tmp1, this->N);
        PipeBarrier<PIPE_V>();
        Adds(tmp2, tmp1, 1.0f, this->N);
        Adds(tmp1, tmp1, -1.0f, this->N);
        PipeBarrier<PIPE_V>();
        Div(tmp1, tmp1, tmp2, this->N);
        PipeBarrier<PIPE_V>();
        Mul(row, row, tmp1, this->N);
        PipeBarrier<PIPE_V>();

        for (uint32_t g = 0; g < this->numGroups; g++) {
            uint32_t offset = g * this->channelsPerGroup;

            ReduceSum<float>(reduceTmp, row[offset], workTmp, this->channelsPerGroup);
            event_t evtVS1 = pipePtr->FetchEventID(HardEvent::V_S);
            SetFlag<HardEvent::V_S>(evtVS1);
            WaitFlag<HardEvent::V_S>(evtVS1);
            float sumVal = reduceTmp.GetValue(0);
            float meanVal = sumVal * this->invChannelsPerGroup;

            event_t evtSV1 = pipePtr->FetchEventID(HardEvent::S_V);
            SetFlag<HardEvent::S_V>(evtSV1);
            WaitFlag<HardEvent::S_V>(evtSV1);

            Adds(diff, row[offset], -meanVal, this->channelsPerGroup);
            PipeBarrier<PIPE_V>();
            Mul(sq, diff, diff, this->channelsPerGroup);
            PipeBarrier<PIPE_V>();
            ReduceSum<float>(reduceTmp, sq, workTmp, this->channelsPerGroup);
            event_t evtVS2 = pipePtr->FetchEventID(HardEvent::V_S);
            SetFlag<HardEvent::V_S>(evtVS2);
            WaitFlag<HardEvent::V_S>(evtVS2);
            float varSum = reduceTmp.GetValue(0);
            float varVal = varSum * this->invChannelsPerGroup;
            float invStd = 1.0f / sqrt(varVal + this->eps);

            event_t evtSV2 = pipePtr->FetchEventID(HardEvent::S_V);
            SetFlag<HardEvent::S_V>(evtSV2);
            WaitFlag<HardEvent::S_V>(evtSV2);

            Muls(diff, diff, invStd, this->channelsPerGroup);
            PipeBarrier<PIPE_V>();
            Mul(diff, diff, gamma[offset], this->channelsPerGroup);
            PipeBarrier<PIPE_V>();
            Add(row[offset], diff, beta[offset], this->channelsPerGroup);
            PipeBarrier<PIPE_V>();
        }

        event_t evtOut = pipePtr->FetchEventID(HardEvent::V_MTE3);
        SetFlag<HardEvent::V_MTE3>(evtOut);
        WaitFlag<HardEvent::V_MTE3>(evtOut);
        DataCopy(yGm[r * this->N], row, this->N);
        event_t evtBack = pipePtr->FetchEventID(HardEvent::MTE3_MTE2);
        SetFlag<HardEvent::MTE3_MTE2>(evtBack);
        WaitFlag<HardEvent::MTE3_MTE2>(evtBack);
    }

    TPipe* pipePtr;
    GlobalTensor<float> yGm, gnGammaGm, gnBetaGm;
    TBuf<TPosition::VECCALC> rowBuf, gammaBuf, betaBuf, tmp1Buf, tmp2Buf;
    TBuf<TPosition::VECCALC> diffBuf, sqBuf, reduceBuf, workBuf;
    uint32_t M, N, numGroups, channelsPerGroup, rowsPerBlock;
    float eps, invChannelsPerGroup;
};

extern "C" __global__ __aicore__ void gemm_bias_add_hardtanh_mish_group_norm_custom(
    GM_ADDR x, GM_ADDR weight, GM_ADDR bias, GM_ADDR gn_gamma, GM_ADDR gn_beta, GM_ADDR y,
    GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tilingData, tiling);
    TPipe pipe;

    if ASCEND_IS_AIC {
        using AT = MatmulType<TPosition::GM, CubeFormat::ND, float>;
        using BT = MatmulType<TPosition::GM, CubeFormat::ND, float>;
        using CT = MatmulType<TPosition::GM, CubeFormat::ND, float>;
        using BiasT = MatmulType<TPosition::GM, CubeFormat::ND, float>;
        Matmul<AT, BT, CT, BiasT> mm;
        REGIST_MATMUL_OBJ(&pipe, GetSysWorkSpacePtr(), mm, &tilingData.mmTiling);

        GlobalTensor<float> xGm, weightGm, biasGm, yGm;
        xGm.SetGlobalBuffer((__gm__ float*)x);
        weightGm.SetGlobalBuffer((__gm__ float*)weight);
        biasGm.SetGlobalBuffer((__gm__ float*)bias);
        yGm.SetGlobalBuffer((__gm__ float*)y);

        mm.SetTensorA(xGm);
        mm.SetTensorB(weightGm);
        mm.SetBias(biasGm);
        mm.IterateAll(yGm);
        mm.End();
    }

    SyncAll();

    if ASCEND_IS_AIV {
        GemmFusedAIV op;
        op.Init(y, gn_gamma, gn_beta,
                tilingData.M, tilingData.N, tilingData.numGroups,
                tilingData.channelsPerGroup, tilingData.eps,
                tilingData.invChannelsPerGroup, &pipe);
        op.Process();
    }
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(torch.nn.Module):
    def __init__(self, in_features, out_features, bias_shape, num_groups):
        super().__init__()
        self.gemm = nn.Linear(in_features, out_features)
        self.bias = nn.Parameter(torch.randn(bias_shape))
        self.groupnorm = nn.GroupNorm(num_groups=num_groups, num_channels=out_features)
        self._weight_t_cache = None
        self._combined_bias_cache = None

    def _prepare_cache(self):
        wt = self.gemm.weight.detach().t().contiguous()
        cb = (self.gemm.bias.detach() + self.bias.detach()).contiguous()
        self._weight_t_cache = wt
        self._combined_bias_cache = cb

    def forward(self, x):
        if (self._weight_t_cache is None
                or self._weight_t_cache.device != x.device):
            self._prepare_cache()
            self._weight_t_cache = self._weight_t_cache.to(x.device)
            self._combined_bias_cache = self._combined_bias_cache.to(x.device)
        return custom_ops_lib.gemm_bias_add_hardtanh_mish_group_norm_custom(
            x.contiguous(),
            self._weight_t_cache,
            self._combined_bias_cache,
            self.groupnorm.weight.contiguous(),
            self.groupnorm.bias.contiguous(),
        )
