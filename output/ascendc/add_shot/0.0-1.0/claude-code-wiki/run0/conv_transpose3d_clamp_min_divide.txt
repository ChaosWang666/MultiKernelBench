kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelClampMinDivide {
public:
    __aicore__ inline KernelClampMinDivide() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t totalLength, uint32_t tileLength, float minValue, float divisor)
    {
        this->totalLength = totalLength;
        this->tileLength = tileLength;
        this->minValue = minValue;
        this->invDivisor = 1.0f / divisor;

        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t blockIdx = AscendC::GetBlockIdx();

        uint32_t avgPerBlock = (totalLength + blockNum - 1) / blockNum;
        avgPerBlock = (avgPerBlock + 7) / 8 * 8;

        uint32_t startOffset = avgPerBlock * blockIdx;
        if (startOffset >= totalLength) {
            this->blockOffset = totalLength;
            this->blockLength = 0;
        } else {
            this->blockOffset = startOffset;
            uint32_t endOffset = startOffset + avgPerBlock;
            if (endOffset > totalLength) {
                endOffset = totalLength;
            }
            this->blockLength = endOffset - startOffset;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x, totalLength);
        yGm.SetGlobalBuffer((__gm__ float *)y, totalLength);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(float));
        pipe.InitBuffer(outQueueY, BUFFER_NUM, this->tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        if (this->blockLength == 0) {
            return;
        }
        uint32_t numFullTiles = this->blockLength / this->tileLength;
        uint32_t tail = this->blockLength - numFullTiles * this->tileLength;
        uint32_t totalTiles = numFullTiles + (tail > 0 ? 1 : 0);

        for (uint32_t i = 0; i < totalTiles; i++) {
            uint32_t curLen = (i < numFullTiles) ? this->tileLength : tail;
            uint32_t curOff = i * this->tileLength;
            CopyIn(curOff, curLen);
            Compute(curLen);
            CopyOut(curOff, curLen);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t localOffset, uint32_t length)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        uint32_t alignedLen = (length + 7) / 8 * 8;
        AscendC::DataCopy(xLocal, xGm[this->blockOffset + localOffset], alignedLen);
        inQueueX.EnQue(xLocal);
    }
    __aicore__ inline void Compute(uint32_t length)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> yLocal = outQueueY.AllocTensor<float>();
        uint32_t alignedLen = (length + 7) / 8 * 8;
        AscendC::Maxs(yLocal, xLocal, this->minValue, alignedLen);
        AscendC::Muls(yLocal, yLocal, this->invDivisor, alignedLen);
        outQueueY.EnQue<float>(yLocal);
        inQueueX.FreeTensor(xLocal);
    }
    __aicore__ inline void CopyOut(uint32_t localOffset, uint32_t length)
    {
        AscendC::LocalTensor<float> yLocal = outQueueY.DeQue<float>();
        uint32_t alignedLen = (length + 7) / 8 * 8;
        AscendC::DataCopy(yGm[this->blockOffset + localOffset], yLocal, alignedLen);
        outQueueY.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueY;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t totalLength;
    uint32_t blockOffset;
    uint32_t blockLength;
    uint32_t tileLength;
    float minValue;
    float invDivisor;
};

extern "C" __global__ __aicore__ void conv_transpose3d_clamp_min_divide_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelClampMinDivide op;
    op.Init(x, y, tiling_data.totalLength, tiling_data.tileLength, tiling_data.minValue, tiling_data.divisor);
    op.Process();
}
