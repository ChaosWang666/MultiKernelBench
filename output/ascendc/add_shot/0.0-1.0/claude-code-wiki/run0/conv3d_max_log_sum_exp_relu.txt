kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConv3dMaxLogSumExpReluCustom {
public:
    __aicore__ inline KernelConv3dMaxLogSumExpReluCustom() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t batchSize, uint32_t channels,
                                 uint32_t dhwSize, uint32_t tileLength,
                                 uint32_t coresPerBatch, uint32_t outputPerCore,
                                 uint32_t tileCount)
    {
        this->batchSize = batchSize;
        this->channels = channels;
        this->dhwSize = dhwSize;
        this->tileLength = tileLength;
        this->coresPerBatch = coresPerBatch;
        this->outputPerCore = outputPerCore;
        this->tileCount = tileCount;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t batchIdx = blockIdx / coresPerBatch;
        uint32_t withinBatchIdx = blockIdx % coresPerBatch;

        uint64_t xOffset = (uint64_t)batchIdx * channels * dhwSize + (uint64_t)withinBatchIdx * outputPerCore;
        uint64_t yOffset = (uint64_t)batchIdx * dhwSize + (uint64_t)withinBatchIdx * outputPerCore;

        xGm.SetGlobalBuffer((__gm__ float*)x + xOffset, (uint64_t)channels * dhwSize);
        yGm.SetGlobalBuffer((__gm__ float*)y + yOffset, outputPerCore);

        pipe.InitBuffer(inQueue, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(mBuf, tileLength * sizeof(float));
        pipe.InitBuffer(sBuf, tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, tileLength * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t t = 0; t < tileCount; t++) {
            uint32_t tileOffset = t * tileLength;
            ComputeTile(tileOffset);
        }
    }

private:
    __aicore__ inline void ComputeTile(uint32_t tileOffset)
    {
        AscendC::LocalTensor<float> m = mBuf.Get<float>();
        AscendC::LocalTensor<float> s = sBuf.Get<float>();
        AscendC::LocalTensor<float> tmp = tmpBuf.Get<float>();

        AscendC::Duplicate<float>(m, -1.0e30f, tileLength);

        for (uint32_t c = 0; c < channels; c++) {
            AscendC::LocalTensor<float> in = inQueue.AllocTensor<float>();
            AscendC::DataCopy(in, xGm[(uint64_t)c * dhwSize + tileOffset], tileLength);
            inQueue.EnQue(in);
            AscendC::LocalTensor<float> inDq = inQueue.DeQue<float>();
            AscendC::Max(m, m, inDq, tileLength);
            inQueue.FreeTensor(inDq);
        }

        AscendC::Duplicate<float>(s, 0.0f, tileLength);

        for (uint32_t c = 0; c < channels; c++) {
            AscendC::LocalTensor<float> in = inQueue.AllocTensor<float>();
            AscendC::DataCopy(in, xGm[(uint64_t)c * dhwSize + tileOffset], tileLength);
            inQueue.EnQue(in);
            AscendC::LocalTensor<float> inDq = inQueue.DeQue<float>();
            AscendC::Sub(tmp, inDq, m, tileLength);
            AscendC::Exp(tmp, tmp, tileLength);
            AscendC::Add(s, s, tmp, tileLength);
            inQueue.FreeTensor(inDq);
        }

        AscendC::LocalTensor<float> out = outQueue.AllocTensor<float>();
        AscendC::Ln(tmp, s, tileLength);
        AscendC::Add(out, tmp, m, tileLength);
        AscendC::Maxs(out, out, 0.0f, tileLength);

        outQueue.EnQue<float>(out);
        AscendC::LocalTensor<float> outDq = outQueue.DeQue<float>();
        AscendC::DataCopy(yGm[tileOffset], outDq, tileLength);
        outQueue.FreeTensor(outDq);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> mBuf, sBuf, tmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t channels;
    uint32_t dhwSize;
    uint32_t tileLength;
    uint32_t coresPerBatch;
    uint32_t outputPerCore;
    uint32_t tileCount;
};

extern "C" __global__ __aicore__ void conv3d_max_log_sum_exp_relu_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);

            tiling_data.tileLength, tiling_data.coresPerBatch, tiling_data.outputPerCore,
            tiling_data.tileCount);
    op.Process();
}
