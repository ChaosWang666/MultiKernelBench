kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t MIN_OUT_BYTES = 32;

class KernelMatmulMaxPoolSumScale {
public:
    __aicore__ inline KernelMatmulMaxPoolSumScale() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                uint32_t batchSize,
                                uint32_t outFeatures,
                                uint32_t tileNum,
                                uint32_t tileLength,
                                uint32_t batchPerBlock,
                                float scaleFactor)
    {
        this->batchSize = batchSize;
        this->outFeatures = outFeatures;
        this->tileNum = tileNum;
        this->tileLength = tileLength;
        this->batchPerBlock = batchPerBlock;
        this->scaleFactor = scaleFactor;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t xOffset = blockIdx * batchPerBlock * outFeatures;
        uint32_t yOffset = blockIdx * batchPerBlock;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + xOffset, batchPerBlock * outFeatures);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + yOffset, batchPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueueY, 1, MIN_OUT_BYTES);
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Y> resLocal = outQueueY.AllocTensor<DTYPE_Y>();
        uint32_t outSlots = MIN_OUT_BYTES / sizeof(DTYPE_Y);
        for (uint32_t i = 0; i < outSlots; i++) {
            resLocal.SetValue(i, (DTYPE_Y)0);
        }

        for (uint32_t b = 0; b < this->batchPerBlock; b++) {
            float batchSum = 0.0f;
            for (uint32_t t = 0; t < this->tileNum; t++) {
                CopyIn(b, t);
                batchSum += ComputeTile();
            }
            resLocal.SetValue(b, batchSum * this->scaleFactor);
        }

        outQueueY.EnQue<DTYPE_Y>(resLocal);
        CopyOut();
    }

private:
    __aicore__ inline void CopyIn(uint32_t b, uint32_t t)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        uint32_t offset = b * this->outFeatures + t * this->tileLength;
        AscendC::DataCopy(xLocal, xGm[offset], this->tileLength);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline float ComputeTile()
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();

        float sum = 0.0f;
        uint32_t pairs = this->tileLength / 2;
        for (uint32_t i = 0; i < pairs; i++) {
            float a = xLocal.GetValue(2 * i);
            float b = xLocal.GetValue(2 * i + 1);
            float m = (a > b) ? a : b;
            sum += m;
        }

        inQueueX.FreeTensor(xLocal);
        return sum;
    }

    __aicore__ inline void CopyOut()
    {
        AscendC::LocalTensor<DTYPE_Y> resLocal = outQueueY.DeQue<DTYPE_Y>();
        uint32_t outSlots = MIN_OUT_BYTES / sizeof(DTYPE_Y);
        AscendC::DataCopy(yGm, resLocal, outSlots);
        outQueueY.FreeTensor(resLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueY;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t batchSize;
    uint32_t outFeatures;
    uint32_t tileNum;
    uint32_t tileLength;
    uint32_t batchPerBlock;
    float scaleFactor;
};

extern "C" __global__ __aicore__ void matmul_max_pool_sum_scale_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMatmulMaxPoolSumScale op;
    op.Init(x, y,
            tiling_data.batchSize,
            tiling_data.outFeatures,
            tiling_data.tileNum,
            tiling_data.tileLength,
            tiling_data.batchPerBlock,
            tiling_data.scaleFactor);
    op.Process();
}
