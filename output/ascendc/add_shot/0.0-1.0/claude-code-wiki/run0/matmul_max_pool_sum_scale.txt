#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;
constexpr int32_t MAX_TILE_LEN = 4096;
constexpr int32_t FP32_PER_REPEAT = 64;
constexpr uint64_t MASK_EVEN_FP32 = 0x5555555555555555ULL;

class KernelMaxPoolSumScale {
public:
    __aicore__ inline KernelMaxPoolSumScale() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t batchSize,
                                 uint32_t outFeatures, uint32_t tileLen, float scaleFactor)
    {
        this->batchSize = batchSize;
        this->outFeatures = outFeatures;
        this->tileLen = tileLen;
        this->scaleFactor = scaleFactor;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();
        uint32_t baseRows = batchSize / blockNum;
        uint32_t extra = batchSize % blockNum;
        if (blockIdx < extra) {
            this->rowsThisBlock = baseRows + 1;
            this->rowStart = blockIdx * (baseRows + 1);
        } else {
            this->rowsThisBlock = baseRows;
            this->rowStart = extra * (baseRows + 1) + (blockIdx - extra) * baseRows;
        }

        if (this->rowsThisBlock == 0) {
            return;
        }

        xGm.SetGlobalBuffer((__gm__ float *)x, batchSize * outFeatures);
        yGm.SetGlobalBuffer((__gm__ float *)y, batchSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, (MAX_TILE_LEN + 16) * sizeof(float));
        pipe.InitBuffer(maxBuf, MAX_TILE_LEN * sizeof(float));
        pipe.InitBuffer(sumBuf, 32);
        pipe.InitBuffer(reduceTmpBuf, 16 * 1024);
    }

    __aicore__ inline void Process()
    {
        if (this->rowsThisBlock == 0) {
            return;
        }
        for (uint32_t i = 0; i < this->rowsThisBlock; i++) {
            ProcessRow(this->rowStart + i);
        }
    }

private:
    __aicore__ inline void ProcessRow(uint32_t batchIdx)
    {
        float rowSum = 0.0f;
        uint32_t rowBase = batchIdx * this->outFeatures;
        uint32_t numTiles = this->outFeatures / this->tileLen;

        for (uint32_t t = 0; t < numTiles; t++) {
            CopyIn(rowBase + t * this->tileLen);
            float tileSum = ComputeTileSum();
            rowSum += tileSum;
        }

        yGm.SetValue(batchIdx, rowSum * this->scaleFactor);
    }

    __aicore__ inline void CopyIn(uint32_t srcOffset)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[srcOffset], this->tileLen);
        inQueueX.EnQue(xLocal);
    }

    __aicore__ inline float ComputeTileSum()
    {
        AscendC::LocalTensor<float> xData = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> maxLocal = maxBuf.Get<float>();
        AscendC::LocalTensor<float> sumLocal = sumBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = reduceTmpBuf.Get<float>();

        AscendC::Duplicate<float>(maxLocal, 0.0f, this->tileLen);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::BinaryRepeatParams params;
        params.dstBlkStride = 1;
        params.src0BlkStride = 1;
        params.src1BlkStride = 1;
        params.dstRepStride = 8;
        params.src0RepStride = 8;
        params.src1RepStride = 8;

        uint32_t repeats = this->tileLen / FP32_PER_REPEAT;
        AscendC::Max<float>(maxLocal, xData, xData[1], MASK_EVEN_FP32, repeats, params);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::ReduceSum<float>(sumLocal, maxLocal, tmpLocal, this->tileLen);
        inQueueX.FreeTensor(xData);

        AscendC::PipeBarrier<PIPE_ALL>();
        return sumLocal.GetValue(0);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TBuf<AscendC::TPosition::VECCALC> maxBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> sumBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceTmpBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;
    uint32_t batchSize;
    uint32_t outFeatures;
    uint32_t tileLen;
    float scaleFactor;
    uint32_t rowStart;
    uint32_t rowsThisBlock;
};

extern "C" __global__ __aicore__ void matmul_max_pool_sum_scale_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMaxPoolSumScale op;
    op.Init(x, y, tiling_data.batchSize, tiling_data.outFeatures, tiling_data.tileLen, tiling_data.scaleFactor);
    op.Process();
}
