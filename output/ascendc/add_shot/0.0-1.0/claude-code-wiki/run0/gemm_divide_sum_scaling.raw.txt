kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelGemmDivideSumScaling {
public:
    __aicore__ inline KernelGemmDivideSumScaling() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR w, GM_ADDR z,
                                  uint32_t batchPerBlock, uint32_t inputSize,
                                  uint32_t tileNum, uint32_t tileLength)
    {
        this->batchPerBlock = batchPerBlock;
        this->inputSize = inputSize;
        this->tileNum = tileNum;
        this->tileLength = tileLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + blockIdx * batchPerBlock * inputSize, batchPerBlock * inputSize);
        wGm.SetGlobalBuffer((__gm__ DTYPE_W *)w, inputSize);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + blockIdx * batchPerBlock, batchPerBlock);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->tileLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueW, BUFFER_NUM, this->tileLength * sizeof(DTYPE_W));
        pipe.InitBuffer(outQueueZ, 1, 256);
        pipe.InitBuffer(accumBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(tmpBuf, this->tileLength * sizeof(float));
        pipe.InitBuffer(reduceBuf, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> accumLocal = accumBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();

        for (uint32_t b = 0; b < batchPerBlock; b++) {
            AscendC::Duplicate<float>(accumLocal, 0.0f, this->tileLength);
            AscendC::PipeBarrier<PIPE_V>();

            for (uint32_t t = 0; t < tileNum; t++) {
                AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
                AscendC::LocalTensor<DTYPE_W> wLocal = inQueueW.AllocTensor<DTYPE_W>();
                AscendC::DataCopy(xLocal, xGm[b * this->inputSize + t * this->tileLength], this->tileLength);
                AscendC::DataCopy(wLocal, wGm[t * this->tileLength], this->tileLength);
                inQueueX.EnQue(xLocal);
                inQueueW.EnQue(wLocal);

                AscendC::LocalTensor<DTYPE_X> xL = inQueueX.DeQue<DTYPE_X>();
                AscendC::LocalTensor<DTYPE_W> wL = inQueueW.DeQue<DTYPE_W>();
                AscendC::Mul(tmpLocal, xL, wL, this->tileLength);
                AscendC::PipeBarrier<PIPE_V>();
                AscendC::Add(accumLocal, accumLocal, tmpLocal, this->tileLength);
                AscendC::PipeBarrier<PIPE_V>();
                inQueueX.FreeTensor(xL);
                inQueueW.FreeTensor(wL);
            }

            AscendC::ReduceSum<float>(reduceLocal, accumLocal, tmpLocal, this->tileLength);
            AscendC::PipeBarrier<PIPE_V>();

            AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
            float sumVal = reduceLocal.GetValue(0);
            zLocal.SetValue(b, sumVal);
            AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
            AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        }

        AscendC::SetFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_MTE3>(EVENT_ID0);
        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        AscendC::LocalTensor<DTYPE_Z> zOut = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm, zOut, this->batchPerBlock);
        outQueueZ.FreeTensor(zOut);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueW;
    AscendC::TQue<AscendC::TPosition::VECOUT, 1> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> accumBuf, tmpBuf, reduceBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_W> wGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t batchPerBlock;
    uint32_t inputSize;
    uint32_t tileNum;
    uint32_t tileLength;
};

extern "C" __global__ __aicore__ void gemm_divide_sum_scaling_custom(GM_ADDR x, GM_ADDR w, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelGemmDivideSumScaling op;
    op.Init(x, w, z, tiling_data.batchPerBlock, tiling_data.inputSize, tiling_data.tileNum, tiling_data.tileLength);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib


class ModelNew(nn.Module):
    def __init__(self, input_size, hidden_size, scaling_factor):
        super(ModelNew, self).__init__()
        self.weight = nn.Parameter(torch.randn(hidden_size, input_size))
        self.scaling_factor = scaling_factor

    def forward(self, x):
        w = self.weight.sum(dim=0).mul(self.scaling_factor / 2.0).contiguous()
        return custom_ops_lib.gemm_divide_sum_scaling_custom(x.contiguous(), w)
