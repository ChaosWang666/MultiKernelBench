kernel_src="""
#include "kernel_operator.h"

class KernelInstNormDiv {
public:
    __aicore__ inline KernelInstNormDiv() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y,
                                 uint32_t ncTotal, uint32_t spatialSize,
                                 uint32_t spatialSizeAligned, uint32_t ncPerCore,
                                 uint32_t tailNum, float invSpatial,
                                 float eps, float invDivideBy)
    {
        uint32_t blockIdx = AscendC::GetBlockIdx();
        this->spatialSize = spatialSize;
        this->invSpatial = invSpatial;
        this->eps = eps;
        this->invDivideBy = invDivideBy;

        if (blockIdx < tailNum) {
            this->ncStart = blockIdx * (ncPerCore + 1U);
            this->ncCount = ncPerCore + 1U;
        } else {
            this->ncStart = tailNum * (ncPerCore + 1U) + (blockIdx - tailNum) * ncPerCore;
            this->ncCount = ncPerCore;
        }

        uint64_t total = static_cast<uint64_t>(ncTotal) * static_cast<uint64_t>(spatialSize);
        xGm.SetGlobalBuffer((__gm__ float*)x, total);
        yGm.SetGlobalBuffer((__gm__ float*)y, total);

        pipe.InitBuffer(xBuf, spatialSizeAligned * sizeof(float));
        pipe.InitBuffer(tmpBuf, spatialSizeAligned * sizeof(float));
        pipe.InitBuffer(workBuf, 32U * 1024U);
        pipe.InitBuffer(resBuf, 32U);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t i = 0; i < this->ncCount; i++) {
            ProcessOne(this->ncStart + i);
        }
    }

private:
    __aicore__ inline void ProcessOne(uint32_t nc)
    {
        AscendC::LocalTensor<float> xLocal = xBuf.Get<float>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::LocalTensor<float> resLocal = resBuf.Get<float>();

        uint64_t offset = static_cast<uint64_t>(nc) * static_cast<uint64_t>(this->spatialSize);

        AscendC::DataCopyExtParams copyInParams;
        copyInParams.blockCount = 1;
        copyInParams.blockLen = this->spatialSize * static_cast<uint32_t>(sizeof(float));
        copyInParams.srcStride = 0;
        copyInParams.dstStride = 0;
        AscendC::DataCopyPadExtParams<float> padParams{true, 0U, 8U, 0.0f};
        AscendC::DataCopyPad(xLocal, xGm[offset], copyInParams, padParams);

        AscendC::SetFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE2_V>(EVENT_ID0);

        AscendC::ReduceSum<float>(resLocal, xLocal, workLocal, this->spatialSize);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

        float sum = resLocal.GetValue(0);
        float mean = sum * this->invSpatial;
        float negMean = -mean;

        AscendC::Adds(xLocal, xLocal, negMean, this->spatialSize);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::Mul(tmpLocal, xLocal, xLocal, this->spatialSize);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::ReduceSum<float>(resLocal, tmpLocal, workLocal, this->spatialSize);

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

        float sumsq = resLocal.GetValue(0);
        float var = sumsq * this->invSpatial;
        float varEps = var + this->eps;

        resLocal.SetValue(0, varEps);

        AscendC::SetFlag<AscendC::HardEvent::S_V>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::S_V>(EVENT_ID0);

        AscendC::Sqrt(resLocal, resLocal, 1);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::SetFlag<AscendC::HardEvent::V_S>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(EVENT_ID0);

        float stdVal = resLocal.GetValue(0);
        float invStd = 1.0f / stdVal;
        float scale = invStd * this->invDivideBy;

        AscendC::Muls(xLocal, xLocal, scale, this->spatialSize);
        AscendC::PipeBarrier<PIPE_V>();

        AscendC::SetFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::V_MTE3>(EVENT_ID0);

        AscendC::DataCopyExtParams copyOutParams;
        copyOutParams.blockCount = 1;
        copyOutParams.blockLen = this->spatialSize * static_cast<uint32_t>(sizeof(float));
        copyOutParams.srcStride = 0;
        copyOutParams.dstStride = 0;
        AscendC::DataCopyPad(yGm[offset], xLocal, copyOutParams);

        AscendC::SetFlag<AscendC::HardEvent::MTE3_MTE2>(EVENT_ID0);
        AscendC::WaitFlag<AscendC::HardEvent::MTE3_MTE2>(EVENT_ID0);
    }

    AscendC::TPipe pipe;
    AscendC::TBuf<AscendC::TPosition::VECCALC> xBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> resBuf;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> yGm;

    uint32_t spatialSize;
    uint32_t ncStart;
    uint32_t ncCount;
    float invSpatial;
    float eps;
    float invDivideBy;
};

extern "C" __global__ __aicore__ void conv2d_instance_norm_divide_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelInstNormDiv op;
    op.Init(x, y,
            tiling_data.ncTotal, tiling_data.spatialSize,
            tiling_data.spatialSizeAligned, tiling_data.ncPerCore,
            tiling_data.tailNum, tiling_data.invSpatial,
            tiling_data.eps, tiling_data.invDivideBy);
    op.Process();
}
