kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelReluGroupNorm {
public:
    __aicore__ inline KernelReluGroupNorm() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR z, GM_ADDR tiling)
    {
        GET_TILING_DATA(tiling_data, tiling);
        this->numGroups = tiling_data.numGroups;
        this->channelsPerGroup = tiling_data.channelsPerGroup;
        this->spatialSize = tiling_data.spatialSize;
        this->groupSize = tiling_data.groupSize;
        this->outChannels = tiling_data.outChannels;
        this->epsilon = tiling_data.epsilon;
        uint32_t rowsPerCore = tiling_data.rowsPerCore;
        uint32_t extraRows = tiling_data.extraRows;

        if (AscendC::GetBlockIdx() < extraRows) {
            this->myRows = rowsPerCore + 1;
            this->startRow = (rowsPerCore + 1) * AscendC::GetBlockIdx();
        } else {
            this->myRows = rowsPerCore;
            this->startRow = (rowsPerCore + 1) * extraRows +
                             rowsPerCore * (AscendC::GetBlockIdx() - extraRows);
        }

        uint32_t N = tiling_data.totalRows / this->numGroups;
        uint32_t totalGmSize = N * this->outChannels * this->spatialSize;

        xGm.SetGlobalBuffer((__gm__ float*)x, totalGmSize);
        zGm.SetGlobalBuffer((__gm__ float*)z, totalGmSize);
        gammaGm.SetGlobalBuffer((__gm__ float*)gamma, this->outChannels);
        betaGm.SetGlobalBuffer((__gm__ float*)beta, this->outChannels);

        pipe.InitBuffer(inQueue, BUFFER_NUM, this->spatialSize * sizeof(float));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->spatialSize * sizeof(float));
        pipe.InitBuffer(gammaQueue, 1, this->outChannels * sizeof(float));
        pipe.InitBuffer(betaQueue, 1, this->outChannels * sizeof(float));
        pipe.InitBuffer(tmpBufRelu, this->spatialSize * sizeof(float));
        pipe.InitBuffer(tmpBufWork, 1024 * sizeof(float));
        pipe.InitBuffer(tmpBufResult, 64 * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> gammaLocal = gammaQueue.AllocTensor<float>();
        AscendC::DataCopy(gammaLocal, gammaGm[0], this->outChannels);
        gammaQueue.EnQue(gammaLocal);
        gammaLocal = gammaQueue.DeQue<float>();

        AscendC::LocalTensor<float> betaLocal = betaQueue.AllocTensor<float>();
        AscendC::DataCopy(betaLocal, betaGm[0], this->outChannels);
        betaQueue.EnQue(betaLocal);
        betaLocal = betaQueue.DeQue<float>();

        AscendC::LocalTensor<float> tmpRelu = tmpBufRelu.Get<float>();
        AscendC::LocalTensor<float> tmpWork = tmpBufWork.Get<float>();
        AscendC::LocalTensor<float> tmpResult = tmpBufResult.Get<float>();

        float zero = 0.0f;
        float fGroupSize = static_cast<float>(this->groupSize);

        for (uint32_t row = 0; row < this->myRows; row++) {
            uint32_t globalRow = this->startRow + row;
            uint32_t batchIdx = globalRow / this->numGroups;
            uint32_t groupIdx = globalRow % this->numGroups;
            uint32_t gmBase = batchIdx * this->outChannels * this->spatialSize +
                              groupIdx * this->channelsPerGroup * this->spatialSize;
            uint32_t gammaBase = groupIdx * this->channelsPerGroup;

            float sum = 0.0f;
            float sumSq = 0.0f;

            for (uint32_t c = 0; c < this->channelsPerGroup; c++) {
                uint32_t chOff = gmBase + c * this->spatialSize;

                AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[chOff], this->spatialSize);
                inQueue.EnQue(xLocal);
                xLocal = inQueue.DeQue<float>();

                AscendC::Maxs(tmpRelu, xLocal, zero, this->spatialSize);

                AscendC::ReduceSum<float>(tmpResult, tmpRelu, tmpWork, this->spatialSize);
                sum += tmpResult.GetValue(0);

                AscendC::Mul(tmpRelu, tmpRelu, tmpRelu, this->spatialSize);
                AscendC::ReduceSum<float>(tmpResult, tmpRelu, tmpWork, this->spatialSize);
                sumSq += tmpResult.GetValue(0);

                inQueue.FreeTensor(xLocal);
            }

            float mean = sum / fGroupSize;
            float var = sumSq / fGroupSize - mean * mean;

            AscendC::Duplicate<float>(tmpResult, var + this->epsilon, 8);
            AscendC::Rsqrt(tmpResult, tmpResult, 8);
            float invStd = tmpResult.GetValue(0);

            for (uint32_t c = 0; c < this->channelsPerGroup; c++) {
                uint32_t chOff = gmBase + c * this->spatialSize;
                uint32_t globalCh = gammaBase + c;
                float gamma_c = gammaLocal.GetValue(globalCh);
                float beta_c = betaLocal.GetValue(globalCh);
                float scale = invStd * gamma_c;
                float shift = beta_c - mean * scale;

                AscendC::LocalTensor<float> xLocal = inQueue.AllocTensor<float>();
                AscendC::DataCopy(xLocal, xGm[chOff], this->spatialSize);
                inQueue.EnQue(xLocal);
                xLocal = inQueue.DeQue<float>();

                AscendC::LocalTensor<float> zLocal = outQueue.AllocTensor<float>();
                AscendC::Maxs(zLocal, xLocal, zero, this->spatialSize);
                AscendC::Muls(zLocal, zLocal, scale, this->spatialSize);
                AscendC::Adds(zLocal, zLocal, shift, this->spatialSize);

                outQueue.EnQue(zLocal);
                inQueue.FreeTensor(xLocal);

                zLocal = outQueue.DeQue<float>();
                AscendC::DataCopy(zGm[chOff], zLocal, this->spatialSize);
                outQueue.FreeTensor(zLocal);
            }
        }

        gammaQueue.FreeTensor(gammaLocal);
        betaQueue.FreeTensor(betaLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::QuePosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::QuePosition::VECIN, 1> gammaQueue, betaQueue;
    AscendC::TQue<AscendC::QuePosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::QuePosition::VECCALC> tmpBufRelu, tmpBufWork, tmpBufResult;
    AscendC::GlobalTensor<float> xGm, gammaGm, betaGm, zGm;

    uint32_t numGroups;
    uint32_t channelsPerGroup;
    uint32_t spatialSize;
    uint32_t groupSize;
    uint32_t outChannels;
    uint32_t myRows;
    uint32_t startRow;
    float epsilon;
};

extern "C" __global__ __aicore__ void convtranspose3d_relu_groupnorm_custom(
    GM_ADDR x, GM_ADDR gamma, GM_ADDR beta, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelReluGroupNorm op;
    op.Init(x, gamma, beta, y, tiling);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, groups, bias=False):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose3d(in_channels, out_channels, kernel_size, bias=bias)
        self.num_groups = groups
        self.epsilon = 1e-5
        self.gamma = nn.Parameter(torch.ones(out_channels))
        self.beta = nn.Parameter(torch.zeros(out_channels))

    def forward(self, x):
        x = self.conv_transpose(x)
        return custom_ops_lib.convtranspose3d_relu_groupnorm_custom(
            x, self.gamma, self.beta, self.num_groups, self.epsilon)
