kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelMeanSub {
public:
    __aicore__ inline KernelMeanSub() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, uint32_t ncPerCore, uint32_t spatialLen,
                                 uint32_t tileLen, uint32_t fullTiles, uint32_t remainder, float invSpatial)
    {
        this->ncPerCore = ncPerCore;
        this->spatialLen = spatialLen;
        this->tileLen = tileLen;
        this->fullTiles = fullTiles;
        this->remainder = remainder;
        this->invSpatial = invSpatial;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t coreOffset = blockIdx * ncPerCore * spatialLen;
        uint32_t coreSize = ncPerCore * spatialLen;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + coreOffset, coreSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + coreOffset, coreSize);

        pipe.InitBuffer(inQueue, BUFFER_NUM, this->tileLen * sizeof(DTYPE_X));
        pipe.InitBuffer(outQueue, BUFFER_NUM, this->tileLen * sizeof(DTYPE_Y));
        pipe.InitBuffer(reduceBuf, 32);
        pipe.InitBuffer(workBuf, this->tileLen * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        for (uint32_t g = 0; g < ncPerCore; g++) {
            float mean = ComputeMean(g);
            SubtractMean(g, mean);
        }
    }

private:
    __aicore__ inline float ComputeMean(uint32_t group)
    {
        float sum = 0.0f;
        uint32_t baseOffset = group * spatialLen;

        for (uint32_t t = 0; t < fullTiles; t++) {
            sum += LoadAndSum(baseOffset + t * tileLen, tileLen, false);
        }
        if (remainder > 0) {
            sum += LoadAndSum(baseOffset + fullTiles * tileLen, remainder, true);
        }
        return sum * invSpatial;
    }

    __aicore__ inline float LoadAndSum(uint32_t offset, uint32_t curLen, bool usePad)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueue.AllocTensor<DTYPE_X>();
        if (usePad) {
            AscendC::DataCopyExtParams params{1, curLen * (uint32_t)sizeof(DTYPE_X), 0, 0, 0};
            AscendC::DataCopyPadExtParams<DTYPE_X> padParams{false, 0, 0, 0};
            AscendC::DataCopyPad(xLocal, xGm[offset], params, padParams);
        } else {
            AscendC::DataCopy(xLocal, xGm[offset], curLen);
        }
        inQueue.EnQue(xLocal);
        xLocal = inQueue.DeQue<DTYPE_X>();

        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> workLocal = workBuf.Get<float>();
        AscendC::ReduceSum<float>(reduceLocal, xLocal, workLocal, curLen);

        event_t eventId = static_cast<event_t>(AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(eventId);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(eventId);

        float result = reduceLocal.GetValue(0);
        inQueue.FreeTensor(xLocal);
        return result;
    }

    __aicore__ inline void SubtractMean(uint32_t group, float mean)
    {
        uint32_t baseOffset = group * spatialLen;

        for (uint32_t t = 0; t < fullTiles; t++) {
            DoSubTile(baseOffset + t * tileLen, tileLen, mean, false);
        }
        if (remainder > 0) {
            DoSubTile(baseOffset + fullTiles * tileLen, remainder, mean, true);
        }
    }

    __aicore__ inline void DoSubTile(uint32_t offset, uint32_t curLen, float mean, bool usePad)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueue.AllocTensor<DTYPE_X>();
        if (usePad) {
            AscendC::DataCopyExtParams params{1, curLen * (uint32_t)sizeof(DTYPE_X), 0, 0, 0};
            AscendC::DataCopyPadExtParams<DTYPE_X> padParams{false, 0, 0, 0};
            AscendC::DataCopyPad(xLocal, xGm[offset], params, padParams);
        } else {
            AscendC::DataCopy(xLocal, xGm[offset], curLen);
        }
        inQueue.EnQue(xLocal);
        xLocal = inQueue.DeQue<DTYPE_X>();

        AscendC::LocalTensor<DTYPE_Y> yLocal = outQueue.AllocTensor<DTYPE_Y>();
        AscendC::Adds(yLocal, xLocal, -mean, curLen);
        outQueue.EnQue<DTYPE_Y>(yLocal);
        inQueue.FreeTensor(xLocal);

        yLocal = outQueue.DeQue<DTYPE_Y>();
        if (usePad) {
            AscendC::DataCopyExtParams params{1, curLen * (uint32_t)sizeof(DTYPE_Y), 0, 0, 0};
            AscendC::DataCopyPad(yGm[offset], yLocal, params);
        } else {
            AscendC::DataCopy(yGm[offset], yLocal, curLen);
        }
        outQueue.FreeTensor(yLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueue;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueue;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> workBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    uint32_t ncPerCore;
    uint32_t spatialLen;
    uint32_t tileLen;
    uint32_t fullTiles;
    uint32_t remainder;
    float invSpatial;
};

extern "C" __global__ __aicore__ void conv_transpose3d_batch_norm_subtract_custom(GM_ADDR x, GM_ADDR y, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelMeanSub op;
    op.Init(x, y, tiling_data.ncPerCore, tiling_data.spatialLen, tiling_data.tileLen,
            tiling_data.fullTiles, tiling_data.remainder, tiling_data.invSpatial);
    op.Process();
}

import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, stride, padding, bias=True):
        super(ModelNew, self).__init__()

        x = self.conv_transpose(x)
        x = self.batch_norm(x)
        x = x.contiguous()
