kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 1;

class KernelBmmInstanceNormSumResidualAddMultiply {
public:
    __aicore__ inline KernelBmmInstanceNormSumResidualAddMultiply() {}
    __aicore__ inline void Init(GM_ADDR x, GM_ADDR y, GM_ADDR z,
                                uint32_t rowLength, uint32_t rowsPerCore,
                                float epsilon, float invN)
    {
        this->rowLength = rowLength;
        this->rowsPerCore = rowsPerCore;
        this->epsilon = epsilon;
        this->invN = invN;

        uint32_t coreOffset = rowsPerCore * rowLength * AscendC::GetBlockIdx();
        uint32_t coreSize = rowsPerCore * rowLength;

        xGm.SetGlobalBuffer((__gm__ DTYPE_X *)x + coreOffset, coreSize);
        yGm.SetGlobalBuffer((__gm__ DTYPE_Y *)y + coreOffset, coreSize);
        zGm.SetGlobalBuffer((__gm__ DTYPE_Z *)z + coreOffset, coreSize);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, this->rowLength * sizeof(DTYPE_X));
        pipe.InitBuffer(inQueueY, BUFFER_NUM, this->rowLength * sizeof(DTYPE_Y));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, this->rowLength * sizeof(DTYPE_Z));
        pipe.InitBuffer(tmpBuf, this->rowLength * sizeof(float));
        pipe.InitBuffer(reduceBuf, 256);
        pipe.InitBuffer(scalarBuf, 32);
    }

    __aicore__ inline void Process()
    {
        for (uint32_t r = 0; r < rowsPerCore; r++) {
            CopyIn(r);
            Compute(r);
            CopyOut(r);
        }
    }

private:
    __aicore__ inline void CopyIn(uint32_t r)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.AllocTensor<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.AllocTensor<DTYPE_Y>();
        AscendC::DataCopy(xLocal, xGm[r * rowLength], rowLength);
        AscendC::DataCopy(yLocal, yGm[r * rowLength], rowLength);
        inQueueX.EnQue(xLocal);
        inQueueY.EnQue(yLocal);
    }

    __aicore__ inline void Compute(uint32_t r)
    {
        AscendC::LocalTensor<DTYPE_X> xLocal = inQueueX.DeQue<DTYPE_X>();
        AscendC::LocalTensor<DTYPE_Y> yLocal = inQueueY.DeQue<DTYPE_Y>();
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.AllocTensor<DTYPE_Z>();
        AscendC::LocalTensor<float> tmpLocal = tmpBuf.Get<float>();
        AscendC::LocalTensor<float> reduceLocal = reduceBuf.Get<float>();
        AscendC::LocalTensor<float> scalarLocal = scalarBuf.Get<float>();

        AscendC::Mul(tmpLocal, xLocal, xLocal, rowLength);

        AscendC::ReduceSum<float>(scalarLocal, xLocal, reduceLocal, rowLength);

        event_t eventVS0 = static_cast<event_t>(AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(eventVS0);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(eventVS0);
        float sumVal = scalarLocal.GetValue(0);

        AscendC::ReduceSum<float>(scalarLocal, tmpLocal, reduceLocal, rowLength);

        event_t eventVS1 = static_cast<event_t>(AscendC::GetTPipePtr()->FetchEventID(AscendC::HardEvent::V_S));
        AscendC::SetFlag<AscendC::HardEvent::V_S>(eventVS1);
        AscendC::WaitFlag<AscendC::HardEvent::V_S>(eventVS1);
        float sumSqVal = scalarLocal.GetValue(0);

        float mean = sumVal * this->invN;
        float var = sumSqVal * this->invN - mean * mean;
        float varEps = var + this->epsilon;
        if (varEps < 0.0f) { varEps = this->epsilon; }
        float invStd = 1.0f / sqrt(varEps);
        float negMean = -mean;

        AscendC::Adds(zLocal, xLocal, negMean, rowLength);
        AscendC::Muls(zLocal, zLocal, invStd, rowLength);
        AscendC::Add(zLocal, zLocal, yLocal, rowLength);
        AscendC::Mul(zLocal, zLocal, yLocal, rowLength);

        outQueueZ.EnQue<DTYPE_Z>(zLocal);
        inQueueX.FreeTensor(xLocal);
        inQueueY.FreeTensor(yLocal);
    }

    __aicore__ inline void CopyOut(uint32_t r)
    {
        AscendC::LocalTensor<DTYPE_Z> zLocal = outQueueZ.DeQue<DTYPE_Z>();
        AscendC::DataCopy(zGm[r * rowLength], zLocal, rowLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX, inQueueY;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::TBuf<AscendC::TPosition::VECCALC> tmpBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> reduceBuf;
    AscendC::TBuf<AscendC::TPosition::VECCALC> scalarBuf;
    AscendC::GlobalTensor<DTYPE_X> xGm;
    AscendC::GlobalTensor<DTYPE_Y> yGm;
    AscendC::GlobalTensor<DTYPE_Z> zGm;
    uint32_t rowLength;
    uint32_t rowsPerCore;
    float epsilon;
    float invN;
};

extern "C" __global__ __aicore__ void bmm_instance_norm_sum_residual_add_multiply_custom(
    GM_ADDR x, GM_ADDR y, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling)
{
    GET_TILING_DATA(tiling_data, tiling);
    KernelBmmInstanceNormSumResidualAddMultiply op;
    op.Init(x, y, z, tiling_data.rowLength, tiling_data.rowsPerCore,
            tiling_data.epsilon, tiling_data.invN);
    op.Process();
}
