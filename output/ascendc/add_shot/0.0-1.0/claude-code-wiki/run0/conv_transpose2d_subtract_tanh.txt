kernel_src="""
#include "kernel_operator.h"

constexpr int32_t BUFFER_NUM = 2;

class KernelConvTSubTanh {
public:
    __aicore__ inline KernelConvTSubTanh() {}

    __aicore__ inline void Init(GM_ADDR x, GM_ADDR bias, GM_ADDR z,
                                 uint32_t totalLength, uint32_t channels,
                                 uint32_t spatialSize, uint32_t tileLength)
    {
        this->channels = channels;
        this->spatialSize = spatialSize;
        this->tileLength = tileLength;

        uint32_t blockIdx = AscendC::GetBlockIdx();
        uint32_t blockNum = AscendC::GetBlockNum();

        uint32_t totalPlanes = totalLength / spatialSize;
        this->planesPerCore = totalPlanes / blockNum;
        this->startPlane = blockIdx * this->planesPerCore;

        uint64_t coreOffset = static_cast<uint64_t>(this->startPlane) * spatialSize;
        uint64_t coreLength = static_cast<uint64_t>(this->planesPerCore) * spatialSize;

        xGm.SetGlobalBuffer((__gm__ float *)x + coreOffset, coreLength);
        zGm.SetGlobalBuffer((__gm__ float *)z + coreOffset, coreLength);
        biasGm.SetGlobalBuffer((__gm__ float *)bias, channels);

        pipe.InitBuffer(inQueueX, BUFFER_NUM, tileLength * sizeof(float));
        pipe.InitBuffer(outQueueZ, BUFFER_NUM, tileLength * sizeof(float));
        uint32_t biasAligned = ((channels + 7) / 8) * 8;
        pipe.InitBuffer(inQueueBias, 1, biasAligned * sizeof(float));
    }

    __aicore__ inline void Process()
    {
        AscendC::LocalTensor<float> biasLocal = inQueueBias.AllocTensor<float>();
        AscendC::DataCopy(biasLocal, biasGm, this->channels);
        inQueueBias.EnQue(biasLocal);
        biasLocal = inQueueBias.DeQue<float>();

        for (uint32_t p = 0; p < this->planesPerCore; p++) {
            uint32_t globalPlaneIdx = this->startPlane + p;
            uint32_t channelIdx = globalPlaneIdx % this->channels;
            float biasVal = biasLocal.GetValue(channelIdx);
            uint64_t planeBase = static_cast<uint64_t>(p) * this->spatialSize;
            ProcessPlane(planeBase, biasVal);
        }

        inQueueBias.FreeTensor(biasLocal);
    }

private:
    __aicore__ inline void ProcessPlane(uint64_t planeBase, float biasVal)
    {
        uint32_t numFullTiles = this->spatialSize / this->tileLength;
        for (uint32_t i = 0; i < numFullTiles; i++) {
            ProcessTile(planeBase + static_cast<uint64_t>(i) * this->tileLength, biasVal);
        }
        uint32_t covered = numFullTiles * this->tileLength;
        if (covered < this->spatialSize) {
            uint64_t tailOffset = planeBase + static_cast<uint64_t>(this->spatialSize) - this->tileLength;
            ProcessTile(tailOffset, biasVal);
        }
    }

    __aicore__ inline void ProcessTile(uint64_t offset, float biasVal)
    {
        AscendC::LocalTensor<float> xLocal = inQueueX.AllocTensor<float>();
        AscendC::DataCopy(xLocal, xGm[offset], this->tileLength);
        inQueueX.EnQue(xLocal);

        xLocal = inQueueX.DeQue<float>();
        AscendC::LocalTensor<float> zLocal = outQueueZ.AllocTensor<float>();
        float negBias = -biasVal;
        AscendC::Adds(zLocal, xLocal, negBias, this->tileLength);
        AscendC::Tanh(zLocal, zLocal, this->tileLength);
        outQueueZ.EnQue<float>(zLocal);
        inQueueX.FreeTensor(xLocal);

        zLocal = outQueueZ.DeQue<float>();
        AscendC::DataCopy(zGm[offset], zLocal, this->tileLength);
        outQueueZ.FreeTensor(zLocal);
    }

private:
    AscendC::TPipe pipe;
    AscendC::TQue<AscendC::TPosition::VECIN, BUFFER_NUM> inQueueX;
    AscendC::TQue<AscendC::TPosition::VECIN, 1> inQueueBias;
    AscendC::TQue<AscendC::TPosition::VECOUT, BUFFER_NUM> outQueueZ;
    AscendC::GlobalTensor<float> xGm;
    AscendC::GlobalTensor<float> biasGm;
    AscendC::GlobalTensor<float> zGm;
    uint32_t channels;
    uint32_t spatialSize;
    uint32_t tileLength;
    uint32_t planesPerCore;
    uint32_t startPlane;
};

extern "C" __global__ __aicore__ void conv_transpose2d_subtract_tanh_custom(
    GM_ADDR x, GM_ADDR bias, GM_ADDR z, GM_ADDR workspace, GM_ADDR tiling) {
    GET_TILING_DATA(tiling_data, tiling);
    KernelConvTSubTanh op;
    op.Init(x, bias, z, tiling_data.totalLength, tiling_data.channels,
            tiling_data.spatialSize, tiling_data.tileLength);
    op.Process();
}

model_src='''
import torch
import torch.nn as nn
import torch_npu
import custom_ops_lib

class ModelNew(nn.Module):
    def __init__(self, in_channels, out_channels, kernel_size, bias_shape, stride=2, padding=1, output_padding=1):
        super(ModelNew, self).__init__()
        self.conv_transpose = nn.ConvTranspose2d(in_channels, out_channels, kernel_size, stride=stride, padding=padding, output_padding=output_padding)
        self.bias = nn.Parameter(torch.randn(bias_shape))

    def forward(self, x):
        x = self.conv_transpose(x)
        x = x.contiguous()
        bias_flat = self.bias.contiguous().view(-1)
        return custom_ops_lib.conv_transpose2d_subtract_tanh_custom(x, bias_flat)
